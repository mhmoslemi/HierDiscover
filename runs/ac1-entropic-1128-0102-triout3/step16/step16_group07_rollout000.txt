def _sweep_families_simple(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=float)
    # Uniform already considered.
    # Geometric
    for eps in np.logspace(-5, -1, 50):
        if time.time() >= deadline: break
        r = 1.0-eps
        geo = r**idx
        geo /= geo.sum()
        score = _objective_score(geo)
        if score < best_score:
            best_score = score
            best_seq = geo.copy()
    # Power
    for p in np.linspace(-2, 2, 81):
        if time.time() >= deadline: break
        seq = (idx+1)**p
        seq = np.maximum(seq, 0)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    return best_seq, best_score

def _random_dirichlet_trials_more(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # sample concentration parameter log-uniform
        log_c = rng.uniform(-6, 3)  # c in [1e-3, 20]
        c = np.exp(log_c)
        # shape c for each dimension? Dirichlet with all alphas = c
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0: continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    return best_seq, best_score

"""
propose_candidate – a multi‑stage optimiser for the autocorrelation peak.

The routine works on a fixed length (default n = 4096) and follows
four phases:

1. Quick parametric sweep (geometric / power families) to get a decent
   starting point.
2. Massive Dirichlet sampling with a wide range of concentration
   parameters to discover good sparse or dense shapes.
3. Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
   sub‑gradient.  This respects the simplex geometry and is much more
   aggressive than plain projected gradient descent.
4. A short simulated‑annealing hill‑climb that moves tiny fractions of
   mass between coordinates, accepting occasional uphill moves.
All phases respect the global 1000 s budget.  The best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear self‑convolution using an FFT of length L_fft."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]

    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Exact score: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Small parametric sweep
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # geometric families (r close to 1)
    for eps in np.logspace(-5, -1, 60):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        sc = _objective_score(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} score={sc:.6f}")

    # power families a_i ∝ (i+1)^p
    for p in np.linspace(-2.0, 2.0, 101):
        if time.time() >= deadline:
            break
        seq = (idx + 1.0) ** p
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] power p={p:.3f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet sampling (broad concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # draw concentration parameter log‑uniformly in a wide interval
        log_c = rng.uniform(-6.0, 3.0)          # c ∈ [1e‑3 , 20]
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Mirror‑descent (exponentiated‑gradient) on the exact max‑conv
# ----------------------------------------------------------------------
def _eg_descent(a0: np.ndarray,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                max_iters: int = 25000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (entropy‑mirror) descent using the exact sub‑gradient
    of max(conv(a,a)).  The update is
        a ← a * exp(-η_t * grad)   (followed by L1‑normalisation).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta0 = 0.7                     # initial step size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")

        # exact sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)            # shape (n,)
        grad = 2.0 * n * subgrad

        # learning‑rate schedule η_t = η0 / sqrt(t)
        eta = eta0 / np.sqrt(it)

        # clip to keep the exponent numerically stable
        grad_clipped = np.clip(grad, -20.0, 20.0)

        # multiplicative update and re‑normalise onto the simplex
        a = a * np.exp(-eta * grad_clipped)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # occasional jitter to escape flat plateaux
        if it % 3000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Simple simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Tiny‑mass transfers between coordinates with a classical exponential
    cooling schedule.  The move size is deliberately tiny (≤0.5 % of the
    source mass) to fine‑tune the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _objective_score(a)

    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()
        print(f"[{time.time() - start_time:.2f}s] SA init best={cur_score:.6f}")

    total_time = max(0.0, deadline - time.time())
    if total_time <= 0.0:
        return best_seq, best_score

    T0 = 0.02
    T_end = 1e-6
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # exponential cooling
        T = T0 * ((T_end / T0) ** (elapsed / (deadline - start_time)))

        # pick a source with non‑zero mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a tiny fraction of src to dst
        frac = rng.random() * 0.004 + 1e-5   # between 1e‑5 and 4e‑3
        delta = a[src] * frac
        if delta <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective_score(a_prop)
        dE = new_score - cur_score
        accept = False
        if dE < -1e-12:
            accept = True
        else:
            prob = np.exp(-dE / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that
    minimises `evaluate_sequence`.  The search is deterministic for a
    given seed and respects the 1000 s time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional external starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] External starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – fast parametric families
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # Phase 2 – intensive Dirichlet sampling
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # Phase 3 – mirror‑descent on exact max‑convolution
    #         We launch a few restarts (different random spikes)
    # ------------------------------------------------------------------
    while time.time() < deadline - 80.0:
        # generate a fresh start: either the current best or a fresh Dirichlet
        if np.random.rand() < 0.6:
            start = best_seq.copy()
        else:
            # a very sparse start – Dirichlet with tiny concentration
            c = np.exp(np.random.uniform(-5, -1))
            a = np.random.gamma(shape=c, scale=1.0, size=n)
            a /= a.sum()
            start = a

        # give each run at most ~30 s
        sub_deadline = min(deadline - 30.0, time.time() + 25.0)
        best_seq, best_score = _eg_descent(
            start, start_time, sub_deadline, best_seq, best_score, max_iters=18000
        )
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing fine‑tune
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score≈{final_score:.6f}")

    return [float(x) for x in best_seq]
