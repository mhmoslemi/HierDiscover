def _next_pow2(x: int) -> int:

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1). Uses an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

        theta = cssv[rho] / (rho + 1.0)
