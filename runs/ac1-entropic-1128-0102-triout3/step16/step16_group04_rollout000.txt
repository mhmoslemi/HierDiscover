"""
Simple random‑subset search for a low‑max‑convolution step function.

The score 2·n·max(conv(a,a)) / (∑a)² is scale‑invariant, so we keep ∑a = 1.
A binary mask with a fixed number *k* of non‑zero entries (each weight = 1/k)
produces a convolution whose expected maximum is about ½·(1/n), i.e. a
score close to 1, far better than the previous ~1.5.  The algorithm repeatedly
samples random supports (k in a wide range), evaluates the exact score via
FFT‑based convolution, and keeps the best mask found before the time budget
expires.  A tiny amount of local hill‑climbing (swap one index in/out) is also
performed when time permits.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence as a plain Python list of non‑negative floats summing to 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)−1) via FFT.
    a is assumed real; result is real.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _score_of_mask(mask: np.ndarray, k: int, n: int, L_fft: int) -> float:
    """
    Compute the exact evaluator score for a binary mask:
        a_i = mask_i / k   (∑a = 1)
    The convolution maximum equals max(mask*mask) / k².
    """
    # mask contains 0/1 floats
    conv = _conv_self_fft(mask, L_fft)
    max_n = conv.max()                 # integer count of ordered pairs
    return float(2.0 * n * max_n / (k * k))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    evaluate_sequence.  The routine uses a random‑subset strategy, evaluates
    candidates via FFT, and keeps the best result found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin so we finish cleanly
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------
    # Problem size – keep the classic 4096 for a fair comparison.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # length for a*a FFT

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (score = 2)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_score = 2.0
    best_mask = np.ones(n, dtype=np.float64)   # uniform mask (all ones)
    best_k = n                                 # each entry weight = 1/n
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter sequence is available, evaluate it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            # normalise to sum‑1
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = float(evaluate_sequence(starter.tolist()))
                if sc < best_score:
                    best_score = sc
                    best_mask = starter.copy()
                    best_k = None                     # not a binary mask
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random‑subset search
    # ------------------------------------------------------------------
    it = 0
    # we keep a list of indices that are currently non‑zero for quick swaps
    # (filled only when best_mask is binary)
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Choose a support size k (between 10% and 90% of n)
        # --------------------------------------------------------------
        k = int(rng.integers(int(0.1 * n), int(0.9 * n) + 1))

        # --------------------------------------------------------------
        # Generate a random binary mask with exactly k ones
        # --------------------------------------------------------------
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=k, replace=False)
        mask[idx] = 1.0

        # --------------------------------------------------------------
        # Evaluate score
        # --------------------------------------------------------------
        cur_score = _score_of_mask(mask, k, n, L_fft)

        # --------------------------------------------------------------
        # Keep the best result
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_mask = mask
            best_k = k
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} k={k} → score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Occasionally perform a cheap local swap on the current best mask
        # --------------------------------------------------------------
        if best_k is not None and rng.random() < 0.05 and time.time() < deadline:
            # pick an occupied index and a free index and swap them
            occupied = np.flatnonzero(best_mask)
            free = np.flatnonzero(best_mask == 0.0)
            if occupied.size and free.size:
                i = rng.choice(occupied)
                j = rng.choice(free)
                # perform the swap
                trial_mask = best_mask.copy()
                trial_mask[i] = 0.0
                trial_mask[j] = 1.0
                trial_score = _score_of_mask(trial_mask, best_k, n, L_fft)
                if trial_score < best_score:
                    best_score = trial_score
                    best_mask = trial_mask
                    # best_k unchanged
                    print(f"[{time.time() - start_time:.2f}s] swap i={i} j={j} → score={trial_score:.6f}")

    # ------------------------------------------------------------------
    # Construct final probability vector from the best binary mask
    # ------------------------------------------------------------------
    if best_k is not None:
        # binary mask case – each selected entry gets equal mass 1/k
        final_vec = best_mask / float(best_k)
    else:
        # starter was already a full vector
        final_vec = np.maximum(best_mask, 0.0)

    # Normalise just in case (numerical safety)
    total = final_vec.sum()
    if total > 0.0:
        final_vec = final_vec / total
    else:
        final_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final verification using the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(final_vec.tolist())
        print(f"[{time.time() - start_time:.2f}s] Final evaluated score = {final_score:.6f}")
    except Exception:
        # If for any reason the evaluator cannot be called, just report the
        # internally computed estimate.
        print(f"[{time.time() - start_time:.2f}s] Final internal score ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the platform.
    # ------------------------------------------------------------------
    return [float(v) for v in final_vec]
