        conv = np.fft.irfft(Af * Af, n=L_fft)[:2*n-1]
        return conv

    def max_conv(a):
        conv = conv_full(a)
        maxb = conv.max()
        return maxb, conv

    def evaluate_obj(a):
        maxb, _ = max_conv(a)
        # sum is 1 due to simplex
        return 2 * n * maxb

    def project_simplex(v):
        # Works for any vector
        # Implementation Duchi et al.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond].max()
        theta = cssv[cond].sum() / rho
        w = np.maximum(v - theta, 0)
        return w

def pgd(a0, max_iters, start_time, deadline):
    a = a0.copy()
    best_a, best_score = a.copy(), evaluate_obj(a)
    grad_sq_sum = np.zeros_like(a)  # for AdaGrad maybe
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        maxb, conv = max_conv(a)
        # get active
        eps = 1e-12
        active = np.where(conv >= maxb - eps)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        # compute grad
        # correlation w ⋆ a (valid)
        w_rev = w[::-1]
        # use FFT
        Lc = L_corr
        Wr = np.fft.rfft(w_rev, n=Lc)
        Ar = np.fft.rfft(a, n=Lc)
        full = np.fft.irfft(Wr * Ar, n=Lc)
        start = n - 1
        grad = 2.0 * full[start:start+n]
        # step size
        eta = 1.0 / np.sqrt(t) * 0.5  # tune
        # momentum? not now
        # update in primal space: move against gradient
        a_new = a - eta * grad
        # project onto simplex
        a = project_simplex(a_new)
        # evaluate
        cur_score = evaluate_obj(a)
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD iter={t} score={cur_score:.6f}")
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
The score is scale‑invariant, therefore we keep the vector on the
probability simplex (∑a_i = 1).  The optimisation proceeds in three
stages.

1. **Random warm‑starts** – a few thousand Dirichlet draws are generated
   (with randomly chosen concentration) and the best few are kept as
   seed vectors.

2. **Projected sub‑gradient/Adam descent** – for each seed we minimise
   the maximum convolution value.  For a fixed vector ``a`` let

        B = a ∗ a                (full linear convolution)
        M = max B
        K = {k | B_k = M}        (active set)

   The sub‑gradient of ``M`` w.r.t. ``a`` is

        g = 2·corr(w , a)          with   w_k = 1/|K|   if k∈K else 0

   (`corr` is the valid linear correlation).  After a gradient step the
   iterate is projected back onto the simplex by the O(n) algorithm of
   Duchi et al.  An Adam‑style momentum is used to accelerate convergence.

3. **Final refinement** – a quick scan of convex mixtures with the uniform
   distribution followed by a light random‑pair hill‑climb (mass transfers
   between two coordinates).

The whole routine is time‑aware (default budget ≈ 1000 s).  Intermediate
scores are printed.  The returned list contains non‑negative floats that
approximately sum to 1 (the evaluator will re‑normalise them automatically).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _corr_valid(w: np.ndarray, a: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Valid correlation (length = len(a)):
        (w ⋆ a)[i] = Σ_k w_k a_{k-i}   (zero outside the range)
    Implemented as convolution of reversed w with a and taking the middle slice.
    """
    w_rev = w[::-1]
    Wr = np.fft.rfft(w_rev, n=L_corr)
    Ar = np.fft.rfft(a, n=L_corr)
    full = np.fft.irfft(Wr * Ar, n=L_corr)
    n = a.shape[0]
    start = n - 1                     # start index of the "valid" part
    return full[start:start + n]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implementation of Duchi et al. (2008).  Returns a vector with non‑negative
    entries that sum to 1.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are negative – fall back to the uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (fast internal version – identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score (2·n·max(conv)) for a probability vector."""
    conv = _conv_self_fft(a, L_fft)
    maxb = conv.max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      L_corr: int,
                      beta1: float,
                      beta2: float,
                      eps_adam: float,
                      eta0: float,
                      t: int,
                      m: np.ndarray,
                      v: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One projected Adam step on the sub‑gradient of max(conv(a,a)).
    Returns the new a, and the updated Adam moments m and v.
    """
    # 1) convolution and active set
    conv = _conv_self_fft(a, L_fft)
    maxb = conv.max()
    # tolerance – a tiny absolute value relative to the magnitude of maxb
    tol = maxb * 1e-12 + 1e-15
    active = np.where(conv >= maxb - tol)[0]

    # 2) sub‑gradient g = 2·corr(w , a)   (w uniform on active set)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / float(active.size)
    grad = 2.0 * _corr_valid(w, a, L_corr)

    # 3) Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    # bias‑corrected moments
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # 4) step (learning‑rate schedule)
    eta = eta0 / np.sqrt(t)        # simple decay
    step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)

    # 5) projected update
    a_next = _project_onto_simplex(a - step)
    return a_next, m, v


def _run_descent(seed_a: np.ndarray,
                 n: int,
                 L_fft: int,
                 L_corr: int,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform the Adam‑based projected sub‑gradient descent starting from
    ``seed_a``.  Returns the best vector found and its score.
    """
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.2            # base step size

    a = seed_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_from_a(best_a, n, L_fft)

    t = 0
    while time.time() < deadline:
        t += 1
        a, m, v = _subgradient_step(a, n, L_fft, L_corr,
                                    beta1, beta2, eps_adam,
                                    eta0, t, m, v)
        cur_score = _score_from_a(a, n, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Descent iter={t:,}  score={cur_score:.6f}")
        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.002:
            jitter = np.random.normal(scale=0.001, size=n)
            a = _project_onto_simplex(a + jitter)

    return best_a, best_score


def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Mix with uniform and perform a cheap random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _score_from_a(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # ---- convex mixing with the uniform distribution ----
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * best_a + (1.0 - lam) * uniform
        cand = _project_onto_simplex(cand)
        sc = _score_from_a(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # ---- random pair hill‑climb ----
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # transfer a tiny amount δ from i to j
        delta = rng.uniform(0.0, min(best_a[i], 0.005))
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not needed – sum stays 1 and all entries stay ≥0
        sc = _score_from_a(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score
