"""
Optimising the worst‑case self‑convolution of a non‑negative step‑function.

The score is 2·n·max(conv(a,a)) / (sum(a)²).  Because the objective is
scale‑invariant we constrain the vector to the probability simplex
(∑a_i = 1).  Uniform weights give a score of 2, but a judicious
distribution can flatten the convolution and push the score towards the
theoretical lower bound ≈1.

The optimiser works in three phases:

1️⃣ Warm‑start – a handful of Dirichlet draws (different
   concentration parameters) are sampled; the best one is kept.
   If the global variable ``height_sequence_1`` exists we also try it.

2️⃣ Directed hill‑climbing – at each step we look at the index *k* where
   the current convolution attains its maximum.  The contribution of
   every coordinate a_i to this peak is a_i·a_{k‑i}.  We move a small
   amount of mass from the coordinate with the largest contribution to
   a randomly chosen recipient.  The convolution is recomputed using an
   FFT (O(n log n) with n=4096).  Improvements are kept; otherwise the
   move is discarded.  This simple, problem‑specific gradient‑free
   scheme quickly reduces the peak of the convolution.

3️⃣ Final polishing – a short “mix‑with‑uniform + tiny random pair”
   refinement (the same routine used in the reference solution) is
   applied to squeeze the last bits of improvement.

All work respects a 1000 s wall‑time budget; intermediate scores are
printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, power‑of‑two helpers)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)–1) using a power‑of‑two FFT.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# 0️⃣ Helper to evaluate the exact score (no calls to evaluate_sequence)
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv) (sum of a is 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣ Warm‑start using Dirichlet draws (and optional global starter)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sample a few Dirichlet vectors, keep the best one."""
    while time.time() < deadline:
        # concentration drawn log‑uniformly in a wide interval
        log_c = rng.uniform(np.log(0.1), np.log(10.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑draw c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Directed local search – move mass away from contributors to the current peak
# ----------------------------------------------------------------------
def _directed_local_search(a_start: np.ndarray,
                           start_score: float,
                           n: int,
                           L_fft: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that repeatedly reduces the contribution of the
    coordinates that dominate the current maximal convolution coefficient.
    """
    a = a_start.copy()
    conv = _conv_self_fft(a, L_fft)
    max_idx = int(np.argmax(conv))
    best_a = a.copy()
    best_score = start_score

    # maximal fraction of the source coordinate that may be moved in a single step
    max_step_frac = 0.03   # up to 3 % of a[i] (still tiny because a[i]≈1/n)

    while time.time() < deadline:
        k = max_idx                       # index where conv attains its maximum
        idx = np.arange(n)
        partner = k - idx                 # partner index that forms a product a[i]*a[partner]
        mask = (partner >= 0) & (partner < n)

        # contribution of every i to the peak (zero where partner out of range)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[partner[mask]]

        # source = index with largest contribution (must have a[i] > 0)
        src_candidates = np.where(mask)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(contrib[src_candidates])]

        # choose a distinct destination uniformly at random
        j = rng.choice(np.delete(np.arange(n), i))

        # propose a small transfer δ from i to j
        max_delta = min(a[i] * max_step_frac, a[i])
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # recompute convolution and score
        cand_conv = _conv_self_fft(cand, L_fft)
        cand_score = _score_from_conv(cand_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            conv = cand_conv
            max_idx = int(np.argmax(conv))
            print(f"[{time.time() - start_time:.2f}s] move i={i}→j={j} δ={delta:.5e} → score={cand_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Final mixing / tiny hill‑climb (taken from the reference solution)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Convex mix with the uniform distribution and a light random‑pair hill‑climb."""
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)

    uniform = np.full(n, 1.0 / n)

    # convex mixture with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → score={sc:.6f}")

    # tiny random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.005))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _score_from_conv(_conv_self_fft(cand, L_fft), n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats (step‑function heights) that aims
    to minimise ``evaluate_sequence``.  The routine respects a 1000 s
    time budget, prints progress, and uses a combination of random
    warm‑starts and a directed hill‑climbing/finishing refinement.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin – leave a few hundred ms for the wrapper
    deadline = start_time + budget_s - 0.15

    # problem size – the value 4096 works well (fits into memory, give
    # enough degrees of freedom for a good flattening of the convolution)
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline initialisation (uniform)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _score_from_conv(_conv_self_fft(best_a, L_fft), n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to use a globally supplied starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _score_from_conv(_conv_self_fft(starter, L_fft), n)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start via Dirichlet draws (≈5 % of the budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * budget_s)
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score, warm_deadline, rng)

    # ------------------------------------------------------------------
    # 2️⃣  Directed local search (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # keep a slice of the remaining time for the final polishing stage
        ls_deadline = deadline - 5.0   # leave ~5 s for the final mixing phase
        print(f"[{time.time() - start_time:.2f}s] Starting directed local search")
        best_a, best_score = _directed_local_search(best_a, best_score,
                                                    n, L_fft, ls_deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] After local search best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final mixing with uniform + very light hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement (mix + hill‑climb)")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # plain Python list as required by the evaluator
    return [float(v) for v in best_a]
