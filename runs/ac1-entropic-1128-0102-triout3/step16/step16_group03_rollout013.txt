"""
Hybrid optimiser for the step‑function evaluation problem (new version).

The programme consists of five phases:

1️⃣  Quick Dirichlet/Beta random sampling – finds a decent seed.
2️⃣  Many short Adam‑optimisation runs from freshly sampled seeds.
3️⃣  A long Adam run starting from the current best.
4️⃣  Deterministic hill‑climbing with occasional simulated‑annealing moves.
5️⃣  Final convex mixing with the uniform distribution (golden‑section).

All phases respect a global countdown (default 1000 s).  The optimiser works
with the *softmax* approximation of the max‑convolution, which provides a
smooth sub‑gradient and helps to escape plateaus.  Adam’s adaptive step‑size
combined with a decreasing temperature gives a more aggressive descent than
the previously used exponentiated‑gradient scheme.

The routine returns the best non‑negative sequence found, normalised to sum = 1.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length = 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """Use the global variable `height_sequence_1` if it is present."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return None
    return a / s

# 1️⃣  Broad random search (Dirichlet + Beta seeds)
# ----------------------------------------------------------------------
def _broad_random_search(n: int,
                         best_seq: np.ndarray,
                         best_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                   (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3)]
    beta_seeds = [v for a, b in beta_params if (v := _beta_shape(a, b, n)) is not None]

    while time.time() < deadline:
        # Dirichlet‑like draw
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        samp = rng.gamma(shape=c, scale=1.0, size=n)
        a = samp / samp.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # occasional deterministic Beta seed
        if rng.random() < 0.005 and beta_seeds:
            a = rng.choice(beta_seeds)
            sc = _objective_score(a, n)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max helper (smooth max of convolution)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temp: float) -> np.ndarray:
    """Return a probability vector proportional to exp((conv‑max)/temp)."""
    if temp <= 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    max_v = conv.max()
    scaled = (conv - max_v) / temp
    # exp of large negative numbers underflows to 0 – that is fine.
    exp_vals = np.exp(scaled)
    s = exp_vals.sum()
    if s == 0 or np.isnan(s):
        # fallback to uniform if everything underflowed
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a,a)  w.r.t a.
    Using the identity  ∇_a Σ w_k·(a*a)_k = 2·correlate(w, a).
    """
    # np.correlate with mode='valid' gives exactly Σ_j w_{i+j}·a_j   for i=0..n-1
    return 2.0 * np.correlate(w, a, mode='valid')

# 2️⃣  Short Adam runs from random seeds
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   L_fft: int,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser (additive updates) on the simplex.
    Uses the soft‑max approximation of the max‑convolution.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    last_improve_it = 0
    restart_interval = 300_000

    while time.time() < deadline:
        if max_iters is not None and t >= max_iters:
            break
        t += 1

        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = t
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} → score={cur_score:.6f}")

        # temperature decays linearly with remaining time
        remaining = deadline - time.time()
        total = deadline - start_time
        temp = max(1e-5, 0.05 * (remaining / total))

        w = _softmax_weights(conv, temp)
        grad = _gradient_softmax(a, w)               # shape (n,)

        # Adam update (minimisation)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule: start 0.2, decay linearly to 0.01
        lr = 0.2 * (1.0 - (t / (max_iters or 1e9))) + 0.01
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = a  # fallback – should never happen

        a = a_new

        # stall detection → random perturbation
        if t - last_improve_it > restart_interval:
            rng = np.random.default_rng()
            perturb = rng.normal(scale=0.02, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            ss = a.sum()
            if ss > 0:
                a = a / ss
            last_improve_it = t

    return best_seq, best_score


def _short_random_adam_phase(n: int,
                             L_fft: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             start_time: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly draw a Dirichlet candidate and run a short Adam optimisation
    (few hundred iterations).  Keeps any improvement.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a0 = sample / sample.sum()
        # give each short run a few seconds or a fixed iteration budget
        sub_deadline = min(time.time() + 0.5, deadline)
        best_seq, best_score = _adam_optimize(
            a0, n, L_fft, start_time, sub_deadline, best_seq, best_score,
            max_iters=400
        )
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Long Adam run from the current best
# ----------------------------------------------------------------------
def _long_adam_phase(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Run Adam with essentially unlimited iterations until the phase deadline."""
    return _adam_optimize(a0, n, L_fft, start_time, deadline, best_seq, best_score, max_iters=None)


# ----------------------------------------------------------------------
# 4️⃣  Hill‑climbing (same as the previous implementation)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.7:
            i = np.argmax(a)
        else:
            i = rng.integers(n)
        if rng.random() < 0.7:
            j = np.argmin(a)
        else:
            j = rng.integers(n)
        if i == j or a[i] <= 1e-14:
            continue
        eps = min(0.02 * a[i], 1e-4)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Fine mixing with the uniform distribution (golden‑section)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget (≈1000 s) and
    prints intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the manager will kill the process after `budget_s` seconds
    deadline = start_time + budget_s - 0.25

    n = 4096                     # problem size (matching the previous best)
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise with uniform and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – broad random search (≈10 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.10 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _broad_random_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )

    # ------------------------------------------------------------------
    # Phase 2 – many short Adam runs from fresh Dirichlet seeds
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.45 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Short‑Adam restarts")
        best_seq, best_score = _short_random_adam_phase(
            n, L_fft, best_seq, best_score, start_time, phase2_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After short‑Adam best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long Adam optimisation from the current best
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * (deadline - start_time)
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Long Adam optimisation")
        best_seq, best_score = _long_adam_phase(
            best_seq, n, L_fft, start_time, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After long‑Adam best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – hill‑climbing refinements (≈10 % of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, L_fft, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – fine mixing with the uniform distribution (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
