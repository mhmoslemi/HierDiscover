def propose_candidate(seed=0,budget_s=1000,**kwargs):
    import time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 4096
    # base uniform sequence
    uniform = np.full(n, 1.0 / n)
    def conv_self(a):
        L = 1 << ((2*n-2).bit_length()) # next pow2
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A*A, n=L)[:2*n-1]
        return conv

    def score_of(a):
        conv = conv_self(a)
        return float(2.0 * n * conv.max())

    best_seq = uniform.copy()
    best_score = score_of(best_seq)
    print(f"[{time.time()-start_time:.2f}s] uniform score {best_score:.6f}")

    # candidate families
    idx = np.arange(n, dtype=np.float64)
    # arcsine-like shapes with varying exponent and offset
    alphas = np.linspace(0.2,0.8,13)   # shape exponent a (=> exponent a-1)
    deltas = [0.0, 0.25, 0.5, 0.75, 1.0]
    for a in alphas:
        for delta in deltas:
            if time.time() >= deadline:
                break
            # compute log-weights to avoid under/overflow
            # weight_i = (i+delta)^{a-1} * (n-1-i+delta)^{a-1}
            # handle i+delta>0, n-1-i+delta>0 always
            log_w = (a-1)*(np.log(idx + delta + 1e-12) + np.log(n-1-idx + delta + 1e-12))
            w = np.exp(log_w - np.max(log_w))   # scale to avoid overflow
            s = w.sum()
            if s == 0 or not np.isfinite(s):
                continue
            base = w / s
            # try plain shape
            sc = score_of(base)
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time()-start_time:.2f}s] a={a:.3f} delta={delta:.2f} plain score={sc:.6f}")
            # mixtures with uniform
            for lam in np.linspace(0.0,1.0,11):
                seq = lam*uniform + (1.0 - lam)*base
                seq /= seq.sum()
                sc = score_of(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] a={a:.3f} delta={delta:.2f} lam={lam:.2f} mix score={sc:.6f}")

    # random Dirichlet draws for final polish
    rng = np.random.default_rng(seed)
    draws = 2000
    for _ in range(draws):
        if time.time() >= deadline:
            break
        c = 10**rng.uniform(-2.0, 0.7)   # concentration 0.01 to 5
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = score_of(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    # small projected gradient descent to polish (optional)
    # we will run a few iterations only if time permits
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        for it in range(200):
            if time.time() >= deadline:
                break
            conv = conv_self(a)
            cur_sc = 2.0 * n * conv.max()
            grad = _subgradient_maxconv(a, conv)
            d = -(grad - grad.mean())
            # feasibility step
            neg = np.where(d < 0)[0]
            if neg.size:
                max_step = np.min(-a[neg] / d[neg])
            else:
                max_step = 1.0
            step = max_step * 0.9
            while step > 1e-12:
                anew = a + step * d
                if np.any(anew < 0):
                    step *= 0.5
                    continue
                conv_new = conv_self(anew)
                new_sc = 2.0 * n * conv_new.max()
                if new_sc <= cur_sc:
                    a = anew
                    break
                step *= 0.5
            # evaluate
            cur_sc = 2.0 * n * conv_self(a).max()
            if cur_sc < best_score:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time()-start_time:.2f}s] PGD polish it={it} score={cur_sc:.6f}")

    # final normalisation and safeguard
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total

"""
Propose a candidate step‑function sequence that (almost) minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation is deliberately simple and fast:

*  a  = 1/√[(i+δ)(n‑1‑i+δ)] (the discrete arcsine law) is a known
   distribution whose self‑convolution is almost uniform,
   giving a very small value of the objective.
*  A tiny sweep over the exponent a (β‑distribution family) and a
   small offset δ is performed; each candidate is optionally mixed with
   the uniform distribution (parameter λ).  The best among all these
   candidates is kept.
*  A modest number of random Dirichlet draws are checked to see if a
   purely random point is even better.
*  If time permits a few projected‑gradient‑descent polishing steps are
   applied.
*  The final vector is normalised to sum = 1 and returned as a plain
   ``list[float]`` ready for ``evaluate_sequence``.

All heavy work (convolutions, gradients) uses the FFT and runs well
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a power‑of‑two FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.size
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at a.
    The gradient is averaged over every shift k that attains the maximum.
    """
    n = a.size
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine builds a small family of analytically motivated
    candidates (arcsine‑like β‑distributions, mixtures with the uniform
    distribution) and a handful of random Dirichlet samples, keeps the best
    one, and finally applies a few projected‑gradient‑descent polishing
    steps.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    # ------------------------------------------------------------------
    # Problem size (the same size that was used in the reference code)
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families – arcsine / beta shapes with offset & mix
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # exponents a (β‑distribution shape parameter) – we concentrate around 0.5
    alphas = np.linspace(0.20, 0.80, 13)          # 0.5 is the classic arcsine
    # half‑grid offsets δ – moving the singularities a little away from the ends
    deltas = [0.0, 0.25, 0.5, 0.75, 1.0]

    for a in alphas:
        for δ in deltas:
            if time.time() >= deadline:
                break

            # weight_i = (i+δ)^{a‑1}·(n‑1‑i+δ)^{a‑1}  (log‑domain for stability)
            log_w = (a - 1.0) * (np.log(idx + δ + 1e-12) +
                                 np.log(n - 1 - idx + δ + 1e-12))
            # shift to avoid overflow, exponentiate
            w = np.exp(log_w - np.max(log_w))
            s = w.sum()
            if s == 0 or not np.isfinite(s):
                continue
            base = w / s                       # already summed to 1

            # pure shape
            sc = _objective_score(base)
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time() - start_time:.2f}s] a={a:.3f} δ={δ:.2f} plain  score={sc:.6f}")

            # mixture with the uniform distribution – λ∈[0,1]
            for lam in np.linspace(0.0, 1.0, 11):
                seq = lam * uniform + (1.0 - lam) * base
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] a={a:.3f} δ={δ:.2f} λ={lam:.2f} mix score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws – a cheap final “polish”
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)
    dirichlet_trials = 2000
    for _ in range(dirichlet_trials):
        if time.time() >= deadline:
            break
        # concentration parameter c ≈ 0.01 … 5  (log‑uniform)
        logc = rng.uniform(-2.0, 0.7)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        total = a.sum()
        if total == 0:
            continue
        a /= total
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Light projected‑gradient polishing (only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:               # keep a few seconds free
        a = best_seq.copy()
        # at most a few hundred cheap iterations – enough to clean up tiny noise
        for it in range(200):
            if time.time() >= deadline:
                break
            conv = _conv_self(a)
            cur_sc = float(2.0 * n * conv.max())
            grad = _subgradient_maxconv(a, conv)
            # descent direction that preserves the simplex constraint
            d = -(grad - grad.mean())

            # largest feasible step without leaving the non‑negative orthant
            neg = np.where(d < 0)[0]
            if neg.size:
                max_step = np.min(-a[neg] / d[neg])
            else:
                max_step = 1.0
            step = max_step * 0.9                     # start a bit shy of the bound

            # simple back‑tracking line search
            while step > 1e-12:
                anew = a + step * d
                if np.any(anew < 0):
                    step *= 0.5
                    continue
                new_sc = float(2.0 * n * _conv_self(anew).max())
                if new_sc <= cur_sc:
                    a = anew
                    break
                step *= 0.5

            # record improvement
            cur_sc = float(2.0 * n * _conv_self(a).max())
            if cur_sc < best_score:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] PGD polish it={it} score={cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
