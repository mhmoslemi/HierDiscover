"""
A compact multi‑phase optimiser for the non‑negative step‑function problem.

The routine

* searches a few parametric families (geometric, power, beta, exponential,
  skewed exponential) with fine grids,
* draws many random points from Dirichlet distributions,
* finally performs a cheap gradient‑guided “mass‑transfer’’ local search that
  repeatedly moves a fraction of the current highest‑gradient coefficient to the
  lowest‑gradient coefficient.
All phases respect the global time budget (default 1000 s).  The best vector
found (always normalised to sum = 1) is returned as a plain ``list[float]``.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level helpers (FFT‑based convolution, objective, sub‑gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective assuming sum(seq)=1 → 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    For every active shift (where conv ≈ max) we add the corresponding
    contribution a_{k‑i} to the gradient of a_i.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (fine‑grained sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    # eps = 1‑r   (r close to 1 gives almost uniform)
    eps_grid = np.logspace(-6, -0.2, 350)
    lam_grid = np.linspace(0.0, 1.0, 11)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in lam_grid:
            seq = lam * (1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    p_grid = np.linspace(-4.0, 2.0, 250)
    for p in p_grid:
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 6.0, 120)
    for a in alphas:
        if time.time() >= deadline:
            break
        # log‑density of symmetric Beta(α,α) on {1,…,n}
        log_seq = (a - 1.0) * (np.log(idx + 1.0) + np.log(np.maximum(n - idx, 1)))
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.2, 5.0, 25)
    for a in alphas:
        if time.time() >= deadline:
            break
        left = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = left + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay a_i ∝ exp(-λ·min(i, n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    lam_grid = np.logspace(-4, -0.1, 120)
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_skewed_exponential(n, best_seq, best_score, deadline):
    """Piecewise exponential: left λ₁, right λ₂."""
    mid = n // 2
    left_idx = np.arange(mid, dtype=np.float64)
    right_idx = np.arange(mid, n, dtype=np.float64)
    lam_left_grid = np.logspace(-4, -0.2, 20)
    lam_right_grid = np.logspace(-4, -0.2, 20)
    for lam_l in lam_left_grid:
        for lam_r in lam_right_grid:
            if time.time() >= deadline:
                break
            seq = np.empty(n, dtype=np.float64)
            seq[:mid] = np.exp(-lam_l * left_idx)
            seq[mid:] = np.exp(-lam_r * (n - 1 - right_idx))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SkewExp λL={lam_l:.1e} λR={lam_r:.1e} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 2500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration between 0.01 and 4 (log‑uniform)
        logc = rng.uniform(-2.0, 0.6)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Local gradient‑guided mass transfer
# ----------------------------------------------------------------------
def _local_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Repeatedly move a small fraction of probability mass from the
    coefficient with the largest sub‑gradient to the one with the smallest.
    Simple back‑tracking reduces the step if no improvement occurs.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-9:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LS it={it:,} new best {cur_score:.6f}")
            # early stop if we have reached the target bound
            if cur_score <= 1.503:
                break

        grad = _subgradient_maxconv(a, conv)
        src = np.argmax(grad)      # index that most hurts the max‑conv
        dst = np.argmin(grad)      # index that helps most if we give it mass
        if a[src] <= 1e-12:
            break  # nothing to move

        # try a fraction of the source mass; start fairly aggressive
        frac = 0.3
        step = a[src] * frac
        improved = False
        while step > 1e-12:
            a_new = a.copy()
            a_new[src] -= step
            a_new[dst] += step
            # keep feasibility
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()
            conv_new = _conv_self(a_new)
            new_score = float(2.0 * n * conv_new.max())
            if new_score < cur_score:
                a = a_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random jitter to escape possible plateaus
            eps = 1e-7
            a += np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method consists of:
        1. fine parametric sweeps,
        2. many random Dirichlet samples,
        3. a cheap gradient‑guided local optimiser.
    The best vector (normalised to sum = 1) is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size (a few thousand entries gives a decent trade‑off)
    # --------------------------------------------------------------
    n = 2048
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (from the previous run, if available)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Broad parametric sweep (time‑controlled)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0  # reserve ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_skewed_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Simple gradient‑guided local improvement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting local gradient descent")
        best_seq, best_score = _local_gradient_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After local descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
