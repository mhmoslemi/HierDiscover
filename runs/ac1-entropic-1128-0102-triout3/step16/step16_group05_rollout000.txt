import time, numpy as np

def _next_pow2(x):

def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

def _objective(a):
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()

def _subgrad(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _exp_gradient_descent(a0, deadline, best_seq, best_score, rng, n):
    a = a0.copy()
    step = 0.1   # initial learning rate
    min_step = 1e-8
    decay = 0.9995   # per iteration
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it} score={cur_score:.6f}")
        grad = _subgrad(a, conv)
        # Exponential update
        a = a * np.exp(-step * grad)
        # renormalize
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # adapt step: if no improvement for N iterations, reduce step.
        step = max(step * decay, min_step)
        # maybe occasional jitter
        if it % 2000 == 0:
            # jitter random indices
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-6 + 1e-9)
            a[i1] -= eps
            a[i2] += eps
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_seq, best_score

def _hill_climb(a0, deadline, best_seq, best_score, rng):
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgrad(a, conv)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0: break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15: continue
        # move small amount
        eps = a[src] * (rng.random() * 0.005 + 1e-6)   # smaller move maybe 0.5%?
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure nonnegative: if src becomes negative, skip
        if a_prop[src] < -1e-12:
            continue
        # renorm (should already sum to 1)
        # compute score
        cur_score = 2.0 * n * _conv_self(a_prop).max()
        if cur_score < best_score:
            a = a_prop
            best_seq = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                # jitter
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-6 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = np.maximum(a, 0.0)

                no_improve = 0
    return best_seq, best_score

            if s > 0:

    if time.time() < deadline:

    total = best_seq.sum()
    if total > 0:

    else:

"""
Propose a non‑negative step function that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1. Choose a moderate size `n` (here 8192) that still allows fast FFTs.
2. Initialise the probability vector `a` from the global starter if present,
   otherwise from a uniform distribution (or a simple U‑shaped guess).
3. Perform an **exponentiated‑gradient descent** on the simplex:
   - Sub‑gradient of the `max` of the self‑convolution is computed analytically.
   - The update `a ← a * exp(-η * g)` automatically keeps the entries ≥0.
   - After each update the vector is renormalised to sum‑to‑1.
   - A simple multiplicative decay of the learning rate is used.
4. When the time budget runs low, hand over to a short **hill‑climb**:
   - Move a small fraction of mass from the index with the largest
     sub‑gradient (most “over‑represented’’) to the one with the smallest.
   - Accept the move only if the score improves; otherwise apply occasional
     random jitter to escape plateaus.
5. Return the best sequence (as a plain Python list) found within the budget.

The whole routine respects a 1000 s wall‑clock limit and prints intermediate
scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global time marker (used in helper prints)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a (real, not circular) using FFT.
    The result has length 2*n-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged if the maximum is attained
    at several shifts).  Returns a vector of the same size as a.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to decide which entries are active
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a_{k-i}
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the probability simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          rng: np.random.Generator,
                          n: int) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the simplex.
    The learning rate η is decayed multiplicatively.
    """
    a = a0.copy()
    # initial step size – empirically 0.1 works well for n≈8192
    eta = 0.10
    min_eta = 1e-8
    decay = 0.9995  # decay per iteration
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it} score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_maxconv(a, conv)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # decay learning rate
        eta = max(eta * decay, min_eta)

        # occasional tiny jitter to escape flat regions
        if it % 4000 == 0:
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
            a[i1] -= jitter
            a[i2] += jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb fine‑tuner
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb: move a tiny amount of mass from the index with the
    largest sub‑gradient to the one with the smallest.  If no improvement is
    observed for a while we inject a random jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source – highest gradient among entries that still have mass
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – lowest gradient (could be the same)
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move a tiny fraction (up to 0.5 %) of the source mass
        eps = a[src] * (rng.random() * 5e-3 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # keep numeric stability
        if a_prop[src] < -1e-12:
            continue

        cur_score = 2.0 * n * _conv_self(a_prop).max()
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 4000:
                # random jitter to jump out of a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0

    return best_seq, best_score
