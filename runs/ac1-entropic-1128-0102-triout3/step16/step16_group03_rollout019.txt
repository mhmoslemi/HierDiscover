"""
Hybrid optimizer for the step‑function evaluation problem.

Main ideas
----------
*   Gather a diverse pool of simple parametric seeds:
    - uniform,
    - exponential profiles  a_i ∝ exp(-λ·i)  with λ spread over several decades,
    - Beta‑shaped densities,
    - a few Dirichlet draws for extra randomness.
*   Pick the best seed and improve it with **additive projected gradient descent**
    on the simplex, using the sub‑gradient of the max‑convolution.
    The update is  a ← a – η·∇,  η decays like 1/√t and after every step the vector
    is projected back onto the probability simplex (clip at zero and renormalise).
    Occasional uphill moves (simulated‑annealing) and random perturbations help
    to escape plateaux.
*   Finish with a cheap golden‑section search over λ∈[0,1] for the convex mixture
    with the uniform distribution.
*   The whole routine respects the 1000 s budget, prints intermediate scores
    and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size  # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _exp_shape(lam: float, n: int) -> np.ndarray:
    """Exponential profile a_i ∝ exp(-lam·i) (lam≥0), normalised to sum 1."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# 1️⃣  Additive projected gradient descent
# ----------------------------------------------------------------------
def _additive_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Plain (additive) gradient descent on the simplex.
    The update is a ← a – η·grad  with η ≈ 0.3/√t.
    After each step we clip negatives to 0 and renormalise.
    """
    a = a0.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Additive it={it:,}  score={cur_score:.6f}")

        # sub‑gradient of the current max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # step size η – classic 1/√t schedule (slightly aggressive)
        eta = 0.30 / math.sqrt(it)

        # additive update + projection onto the simplex
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:                                   # fallback – stay where we are
            a_new = a

        # evaluate the trial point
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()

        # accept improvements; otherwise accept a tiny uphill move (annealing)
        if new_score < cur_score:
            a = a_new
        else:
            # temperature decays linearly with remaining time
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        # every few‑tens of thousands of iterations inject a small random shake
        if it % 50000 == 0:
            rng = np.random.default_rng()
            perturb = rng.normal(scale=0.02, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few golden‑section
    evaluations.  The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score
