"""
Search for a non‑negative step‑function (list of floats) minimizing

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages within the allotted 1000 s:

1. **Parameterized random search** – generate candidates from a few simple
   families (Beta distributions, exponential tails, Dirichlet draws,
   classic windows) and keep the best one seen.

2. **Targeted max‑descent** – repeatedly identify the index where the
   convolution attains its current maximum and take a multiplicative
   mirror‑descent step that directly reduces this entry (the gradient of
   the active convolution is simply the reversed vector).  The step size
   decays like 1/√t.

3. **Light hill‑climbing** – random pairwise mass transfers to fine‑tune
   the solution.

All candidates are kept on the simplex (∑a_i = 1) and are non‑negative,
so the evaluator’s clipping does not affect the final result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global constants (problem size is fixed to the size of the known starter)
# ----------------------------------------------------------------------
N = 4096                         # length of the step‑function
X = (np.arange(N, dtype=np.float64) + 0.5) / N   # points in (0,1) for Beta shapes
UNIFORM = np.full(N, 1.0 / N)    # baseline uniform distribution
L_FFT = 1 << ((2 * N - 2).bit_length())   # power‑of‑two length for FFT

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[float, int, np.ndarray]:
    """Return (max value, index of the max, full convolution) of a*a."""
    Af = np.fft.rfft(a, n=L_FFT)
    conv = np.fft.irfft(Af * Af, n=L_FFT)[: 2 * N - 1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv


def _score(a: np.ndarray) -> float:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    max_val, _, _ = _conv_max(a)
    return float(2.0 * N * max_val)


# ----------------------------------------------------------------------
# Shape generators (all return a normalized ndarray, sum == 1)
# ----------------------------------------------------------------------
def _shape_uniform() -> np.ndarray:
    return UNIFORM.copy()


def _shape_hann() -> np.ndarray:
    i = np.arange(N, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (N - 1)))
    w /= w.sum()
    return w


def _shape_triangular() -> np.ndarray:
    i = np.arange(N, dtype=np.float64)
    mid = (N - 1) / 2.0
    w = np.where(i <= mid, i + 1.0, N - i)
    w /= w.sum()
    return w


def _shape_beta(p: float, q: float) -> np.ndarray:
    """Beta(p,q) density on the interval (0,1), evaluated at mid‑points."""
    # log‑space to avoid overflow
    log_a = (p - 1.0) * np.log(X) + (q - 1.0) * np.log(1.0 - X)
    a = np.exp(log_a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return UNIFORM.copy()
    a /= s
    return a


def _shape_exponential(lam: float, direction: int = 1) -> np.ndarray:
    """exp(-lam * i) (direction=1) or its mirror (direction=-1)."""
    if direction == 1:
        exponents = -lam * np.arange(N, dtype=np.float64)
    else:
        exponents = -lam * np.arange(N - 1, -1, -1, dtype=np.float64)
    a = np.exp(exponents)
    a /= a.sum()
    return a


def _shape_dirichlet(conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with total concentration `conc`."""
    sample = rng.gamma(shape=conc, scale=1.0, size=N)
    a = sample / sample.sum()
    return a


def _mix_with_uniform(a: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·a + (1‑λ)·uniform, renormalised."""
    mixed = lam * a + (1.0 - lam) * UNIFORM
    mixed /= mixed.sum()
    return mixed
