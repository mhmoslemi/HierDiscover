"""
Propose a step‑function that (empirically) beats the previous best score
(~1.5078).  The search uses three stages:

1.  Initialise from the global starter (if any) or from a uniform vector.
    A short burst of random Dirichlet draws is performed to obtain a good
    starting point.

2.  A projected sub‑gradient descent on the exact max‑convolution objective
    (the same routine that gave the earlier baseline).  This runs for a
    modest part of the time budget and quickly reaches the neighbourhood of
    the former optimum.

3.  Simulated‑annealing refinement: at every step we move a small amount of
    mass from the index with the largest sub‑gradient to the index with the
    smallest sub‑gradient.  The move is accepted if it improves the score,
    otherwise it is accepted with a Metropolis probability that gradually
    cools.  This lets the optimiser escape shallow plateaus that stopped the
    pure sub‑gradient method.

4.  A final tiny‑mass hill‑climb (identical to the one used previously)
    performs a last polishing pass.

All phases respect a global 1000 s wall‑clock budget.  The returned list is
normalised to unit L¹‑norm, which is the scale expected by the public
``evaluate_sequence`` routine.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, using FFT)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return the exact evaluation score for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (sum(a)=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=200):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as before, but a bit faster)
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq, deadline, best_seq, best_score):
    """Projected sub‑gradient descent on the max‑convolution objective."""
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.2
    min_step = 1e-12
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={iteration} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        # back‑tracking line search
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-12)

        # occasional jitter to shake off plateaus
        if iteration % 3000 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny‑mass hill climb (final polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq, deadline):
    """
    Refinement using a Metropolis‑style walk.
    Moves mass from the index with the largest sub‑gradient to the one with
    the smallest sub‑gradient; acceptance probability is exp(-Δ/T), where
    the temperature T is linearly cooled to zero.
    """
    a = start_seq.copy()
    n = a.shape[0]
    cur_score = float(2.0 * n * _conv_self(a).max())
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # initial temperature – a small fraction of the current score
    T0 = max(0.01, cur_score * 0.005)
    anneal_start = time.time()

    while time.time() < deadline:
        # compute gradient on the current state
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # fall back to a random pair occasionally
            src, dst = rng.choice(n, size=2, replace=False)

        # step size shrinks together with the temperature
        elapsed = time.time() - anneal_start
        fraction = elapsed / (deadline - anneal_start)
        T = max(T0 * (1.0 - fraction), 1e-12)
        move_frac = rng.uniform(0.0, 0.05) * (T / T0 + 0.01)  # 0‑5% of source mass
        delta = a[src] * move_frac
        if delta <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = float(2.0 * n * _conv_self(a_prop).max())
        delta_score = new_score - cur_score

        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_a = a.copy()
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")

        # occasional random jitter to explore new basins
        if rng.random() < 0.001:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The optimiser runs four stages:
        1.  Initialisation (global starter or uniform + a few Dirichlet draws);
        2.  Projected sub‑gradient descent (≈15 % of the budget);
        3.  Simulated‑annealing refinement (≈70 % of the budget);
        4.  Final tiny‑mass hill‑climb (remaining time).
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ----- problem size -------------------------------------------------
    n = 4096                     # a good compromise between flexibility and speed
    uniform = np.full(n, 1.0 / n)

    # ----- initialise ---------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # try to use the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ----- Phase 0 : quick Dirichlet exploration ------------------------
    phase0_deadline = start_time + budget_s * 0.08
    if time.time() < phase0_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase0_deadline, trials=500
        )

    # ----- Phase 1 : sub‑gradient descent --------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    if time.time() < phase1_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(
            best_seq, phase1_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After SG → {best_score:.6f}")

    # ----- Phase 2 : simulated annealing --------------------------------
    phase2_deadline = start_time + budget_s * 0.85
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, phase2_deadline)
        print(f"[{time.time() - start_time:.2f}s] After SA → {best_score:.6f}")

    # ----- Phase 3 : final hill‑climb -----------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ----- normalise and return -----------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
