def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2*n-1)

    def compute_score(a):
        conv = _conv_self(a, L_fft)
        return 2 * n * conv.max()

    # Uniform initial
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = compute_score(best_seq)
    print(f"Uniform baseline score={best_score}")

    # Try previous starter if exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Normalize to sum 1
            starter = np.maximum(starter, 0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = compute_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"Starter reduction score={sc}")
    except Exception:
        pass

    # Phase 1: Dirichlet random sampling
    rng = np.random.default_rng(seed + 12345)
    trials = 0
    while time.time() < deadline - 200:  # keep ~200s for refinement
        # random concentration type
        log_c = rng.uniform(-6, 2)  # c in approx [0.0025, 7.4]
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        score = compute_score(a)
        trials += 1
        if score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"[{time.time()-start_time:.1f}s] Dirichlet trial {trials} c={c:.3f} score={score:.6f}")

    # Phase 2: PGD descent from best_seq
    # We'll run PGD while time left > 50s
    a = best_seq.copy()
    if time.time() < deadline - 50:
        a, pgd_score = _pgd_descent(a, start_time, deadline - 50, best_seq, best_score)
        if pgd_score < best_score:
            best_seq = a.copy()
            best_score = pgd_score
            print(f"[{time.time()-start_time:.1f}s] After PGD score={best_score}")

    # Phase 3: Simulated annealing hill climb
    if time.time() < deadline:
        a, final_score = _anneal_hill_climb(a, start_time, deadline, best_seq, best_score)
        if final_score < best_score:
            best_seq = a.copy()
            best_score = final_score
            print(f"[{time.time()-start_time:.1f}s] Final SA score={best_score}")

    # Normalise and return list
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"Finished best_score≈{best_score}")
    return [float(x) for x in best_seq]

def _anneal_hill_climb(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    rng = np.random.default_rng()
    cur_score = compute_score(a)
    # SA parameters
    T0 = cur_score * 0.05   # start temp
    total_iters = 0
    # exponential decay schedule that ends near zero at deadline
    total_time = deadline - time.time()
    # We'll compute decay per iteration based on target final temperature close to 0
    # Use T = T0 * (0.001)**(elapsed/total_time)
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # Compute temperature
        T = T0 * (0.001 ** (elapsed / (deadline - start_time)))
        # pick src and dst indices: src with large a maybe > epsilon, dst any
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(n)
        if src == dst:
            continue
        # Choose epsilon fraction up to 0.2 of source mass
        frac = rng.random() * 0.2
        eps = a[src] * frac
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # Clip negative (should not happen) and renorm? Keep sum same automatically.
        if a_prop[src] < -1e-12:
            continue
        # No need to renorm because sum unchanged.
        new_score = compute_score(a_prop)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time()-start_time:.1f}s] SA best {best_score}")
        total_iters += 1
    return best_seq, best_score

def _pgd_descent(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 1.0
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it} score={cur_score:.6f}")
        subgrad = _subgradient_maxconv(a, conv)
        grad = 2 * n * subgrad
        it += 1
        eta_t = eta / (1 + it * 1e-4)
        a_new = a - eta_t * grad
        a_new = _project_simplex(a_new)
        # Check improvement
        new_score = 2 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score - 1e-12:
            a = a_new
            # Optionally increase step adaptation
        else:
            # Reduce step size if not improving
            eta *= 0.9
        if eta < 1e-8:
            break
    return best_seq, best_score

"""
Propose a step‑function (probability vector) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm is deliberately simple and robust:

1.  **Broad random search** – many Dirichlet / power‑law samples are
    generated (≈10⁵ candidates).  Each candidate is normalised to sum = 1
    and scored with a fast FFT‑based convolution.  The best few
    candidates are retained.

2.  **Projected sub‑gradient descent** – starting from the current best
    vector a sub‑gradient of the exact max‑convolution is used.
    A decaying step size together with occasional random jitter helps
    to escape plateaus.

3.  **Simulated‑annealing hill‑climb** – a final phase performs large
    random mass transfers between positions.  Moves are accepted
    according to a temperature schedule; the best solution seen is
    returned.

All phases respect the global time budget (default 1000 s).  The code
uses only NumPy (no external solvers) and works purely on the CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray, L_fft: int) -> float:
    """2·n·max(conv(a,a)) for a already normalised (sum = 1)."""
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 1.0                     # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)          # ∂max/∂a
        grad = 2.0 * n * subgrad                         # ∂objective/∂a

        # decaying step size
        eta_t = eta / (1.0 + it * 1e-4)
        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        cand_score = 2.0 * n * _conv_self(a_candidate, L_fft).max()
        if cand_score < cur_score - 1e-12:
            a = a_candidate
            # if we improve, keep step size roughly unchanged
        else:
            # otherwise slowly shrink the step
            eta *= 0.995
            # occasional random jitter to escape flat regions
            if it % 500 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = _project_simplex(np.maximum(a_jit, 0.0))
                a = a_jit

        if eta < 1e-9:
            break
    return best_seq, best_score


def _anneal_hill_climb(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.  Randomly transfers a fraction of mass
    from one index to another; moves are accepted with Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # Exponential temperature schedule, ending near zero at the deadline
    T0 = cur_score * 0.05 + 1e-12
    total_time = deadline - start_time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = T0 * (0.001 ** (elapsed / total_time))   # final temperature ≈0

        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        frac = rng.random() * 0.20
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:      # numerical safety
            continue

        new_score = 2.0 * n * _conv_self(a_prop, L_fft).max()
        Δ = new_score - cur_score
        if Δ < -1e-12 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine consists of a massive random Dirichlet / power‑law sweep,
    a projected sub‑gradient descent, and a final simulated‑annealing
    refinement.  It returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # problem size – 4096 works well and matches the reference implementation
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline (uniform) and optional external starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        # `height_sequence_1` may be defined by the surrounding harness
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, L_fft)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – massive random sampling
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    sampling_deadline = deadline - 300.0   # reserve ~300 s for refinement
    trials = 0
    while time.time() < sampling_deadline:
        # two families of random candidates
        if rng.random() < 0.5:
            # Dirichlet‑like (Gamma) sampling, concentration log‑uniform
            log_c = rng.uniform(-6.0, 2.0)      # c ∈ [≈0.0025, ≈7.4]
            c = np.exp(log_c)
            a = rng.gamma(shape=c, scale=1.0, size=n)
        else:
            # Power‑law of a uniform base: a_i = (U_i)^p
            p = rng.uniform(0.1, 5.0)
            a = np.power(rng.random(size=n), p)

        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        trials += 1

        sc = _objective_score(a, L_fft)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Trial {trials:05d} score={sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (main refinement)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 50.0:
        pgd_deadline = time.time() + 0.7 * remaining   # 70 % of what is left
        best_seq, best_score = _pgd_descent(best_seq, start_time,
                                            pgd_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal_hill_climb(best_seq, start_time,
                                                  deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
