"""
A two‑phase optimizer for the max‑convolution bound.

Phase 1 – many random Dirichlet draws (sparse and dense) to obtain a good
starting point on the probability simplex.

Phase 2 – smooth‑max gradient descent.  The true objective
    f(a) = 2·n·max(conv(a,a))
is approximated by the smooth max
    Fβ(a) = (1/β)·log Σ exp(β·conv(a,a)).
Its gradient is 2·(a ⋆ w) where w are the soft‑max weights of the
convolution entries.  An exponentiated‑gradient (mirror‑descent) update
keeps the iterates on the simplex.  The temperature β is increased
geometrically, letting the optimisation first see a smoothed landscape
and then sharpen around the true max.

A short projected‑gradient refinement (as in the reference code) is run
at the end to polish the best candidate.

All work respects a 1000 s budget; the best sequence found is returned as
a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation of a (length n) with w (length 2·n‑1):
        (a ⋆ w)[i] = Σ_j a[j]·w[i+j] .
    Implemented via FFT.
    """
    n = a.shape[0]
    L = _next_pow2(n + w.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(np.conj(Af) * Wf, n=L)
    return corr_full[:n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum of a is forced to 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet initialisations (dense + very sparse)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(
    n: int, max_trials: int, deadline: float
) -> tuple[np.ndarray | None, float]:
    """
    Return the best probability vector seen among a number of Dirichlet draws.
    Concentration is sampled log‑uniformly from 10⁻⁴ … 10 to obtain both
    dense (large c) and extremely sparse (tiny c) candidates.
    """
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # log₁₀ concentration uniformly in [-4, 1] → c ∈ [1e‑4, 10]
        conc_log = rng.uniform(-4.0, 1.0)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.5g} "
                f"score={sc:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max (soft‑max) gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray, cur_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Minimise the true max‑convolution objective by optimising a smooth
    approximation.  The schedule of the smoothing parameter β grows
    geometrically; each stage runs a limited number of exponentiated‑gradient
    steps and keeps the best feasible vector seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # β schedule – start very smooth, finish very sharp
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    # allocate roughly equal wall‑clock time to each β‑stage
    total_time = deadline - time.time()
    time_per_stage = max(total_time / len(beta_schedule), 0.5)

    for beta in beta_schedule:
        stage_deadline = min(deadline, time.time() + time_per_stage)

        # learning rate inversely proportional to β (larger β → steeper landscape)
        lr = 0.4 / max(beta, 1e-12)

        while time.time() < stage_deadline:
            conv = _conv_self(a)                 # length 2·n‑1
            # soft‑max weights w_k = exp(β·conv_k) / Σ exp(β·conv)
            maxc = conv.max()
            exp_vals = np.exp(beta * (conv - maxc))   # stabilise
            w = exp_vals / exp_vals.sum()

            # gradient of the smooth max (2·a ⋆ w)
            grad = 2.0 * _cross_corr(a, w)            # shape (n,)

            # --- exponentiated‑gradient (mirror descent on the simplex) ---
            # work in log‑space to avoid under/overflow
            log_factor = -lr * grad
            log_factor -= np.max(log_factor)          # shift for numerical stability
            a_new = a * np.exp(log_factor)
            s = a_new.sum()
            if s > 0.0:
                a_new /= s
            else:
                a_new = np.full_like(a, 1.0 / n)

            # evaluate the true (non‑smoothed) objective
            cand_score = _objective_score(a_new)

            # keep the globally best vector
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_new.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"β={beta:.1f} new best {best_score:.6f}"
                )

            # accept the move if it improves the true objective,
            # otherwise keep the old point (simple greedy acceptance)
            if cand_score < cur_score:
                a = a_new
                cur_score = cand_score
            else:
                # occasional stochastic acceptance to escape tiny plateaus
                if np.random.rand() < 0.003:
                    a = a_new

            # occasional tiny random perturbation (helps avoid being stuck)
            if np.random.rand() < 0.001:
                noise = np.random.normal(scale=1e-4, size=n)
                a = _project_simplex(np.maximum(a + noise, 0.0))
                cur_score = _objective_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final projected‑gradient refinement (as in the reference code)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _projected_gradient_descent(
    a0: np.ndarray, cur_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Fine‑tune the vector by moving opposite to the sub‑gradient and re‑projecting."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5  # initial step length

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score
