"""
Adam‑smoothed max‑convolution optimiser.

The evaluation function depends only on the maximal entry of the
self‑convolution of a non‑negative sequence normalised to sum=1:

    score = 2·n·max(conv(a,a))

We minimise this by a smooth approximation of the max, namely a
soft‑max with temperature β.  The gradient of the smoothed objective is
simple: it is 2·conv(a, w) where w are the soft‑max weights derived from
the current convolution.  An Adam optimiser is used on the smooth
objective, with a projection onto the probability simplex after each
step.  The temperature β is increased gradually so that the algorithm
first moves easily on a smooth landscape and later refines the true
max‑objective.  Random Dirichlet / mixture initialisers give a good
starting point; the whole search respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    return _conv_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum(v)=1, v_i ≥ 0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self_fft(a).max())


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    trials: int = 300,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)  # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}"
            )
    return best_seq, best_score


def _random_mixture_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    trials: int = 250,
) -> tuple[np.ndarray, float]:
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution objective
# ----------------------------------------------------------------------


def _adam_smoothing_opt(
    init_a: np.ndarray,
    init_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the smooth approximation
        fβ(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    with a gradually increasing temperature β.
    The iterate is projected onto the probability simplex each step.
    """
    a = init_a.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = init_score

    t = 0
    beta = 5.0               # start with a modest smoothing
    beta_mul = 1.02          # increase temperature slowly
    beta_max = 5000.0

    while time.time() < deadline:
        t += 1

        # ----- evaluate current objective -----
        conv = _conv_self_fft(a)          # length 2n‑1, symmetric
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t} score={cur_score:.6f} β={beta:.2f}"
            )

        # ----- soft‑max weights -----
        z = beta * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)         # stabilised
        w = exp_z / exp_z.sum()           # length 2n‑1, symmetric

        # ----- gradient of the smoothed objective -----
        # grad = 2 * conv(a, w)   (full convolution, then slice the central n entries)
        conv_full = _conv_fft(a, w)       # length 3n‑2
        start_idx = len(w) - 1            # = 2n‑2
        grad = 2.0 * conv_full[start_idx : start_idx + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ----- projection onto the simplex -----
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # ----- temperature schedule -----
        beta = min(beta * beta_mul, beta_max)

    return best_a, best_score

    # --------------------------------------------------------------
    # half‑sine bump
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine /= sine.sum()
    sc = _objective_score(sine)
    if sc < best_score:
        best_score, best_seq = sc, sine.copy()
        print(f"[{time.time() - start_time:.2f}s] Sine bump → {sc:.6f}")

    # triangular bump
    tri = 1.0 - np.abs(np.arange(n) - (n - 1) / 2) * 2.0 / (n - 1)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    sc = _objective_score(tri)
    if sc < best_score:
        best_score, best_seq = sc, tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular bump → {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Adam optimiser on smoothed max‑convolution (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        best_seq, best_score = _adam_smoothing_opt(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Adam optimiser → best score {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
