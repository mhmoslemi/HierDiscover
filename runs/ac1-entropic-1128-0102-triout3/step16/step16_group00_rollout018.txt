"""
Propose a better step‑function using three ideas:

1️⃣  Greedy construction of a sparse “Sidon‑like’’ support.
    We keep a set S ⊂ {0,…,n‑1} such that each ordered sum i+j
    appears at most C times (C=2 or 3).  With a uniform weight on S
    the maximal self‑convolution value is C/|S|²; for a large enough
    support this already beats the previous ≈1.508 bound.

2️⃣  Random Dirichlet restarts – cheap way of finding diverse starting
    points.

3️⃣  A smooth‑max optimisation (Adam) on the whole simplex.
    The non‑smooth max in the objective is approximated by a
    soft‑max with a temperature that grows from a smooth regime to a
    hard max.  The gradient of the soft‑max is obtained analytically:
    ∇ L = 2 conv( w , a_rev )   (w = soft‑max weights,
    a_rev = a[::-1]).  Adam updates are projected back onto the
    probability simplex after each step.

The whole search respects the 1000 s budget; the best sequence found
is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _objective_full(seq: np.ndarray) -> float:
    """Exact objective for any (non‑negative) sequence (no safety checks)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Greedy Sidon‑like construction
# ----------------------------------------------------------------------
def _sidon_initial(n: int, max_occ: int, deadline: float):
    """
    Build a set S ⊂ [0,n) such that each ordered sum i+j occurs at most
    `max_occ` times.  Return a uniform distribution on the best set found
    and its objective (computed analytically, no FFT needed).
    """
    best_set = set()
    best_m = 0
    best_maxcnt = max_occ
    rng = np.random.default_rng()
    # repeat many random orderings until time runs out
    while time.time() < deadline:
        order = rng.permutation(n)
        cur_set = set()
        cnt = {}                # sum → ordered count
        for x in order:
            # quick rejection test
            if cnt.get(2 * x, 0) + 1 > max_occ:
                continue
            ok = True
            for y in cur_set:
                s = x + y
                if cnt.get(s, 0) + 2 > max_occ:
                    ok = False
                    break
            if not ok:
                continue
            # accept x
            cnt[2 * x] = cnt.get(2 * x, 0) + 1
            for y in cur_set:
                s = x + y
                cnt[s] = cnt.get(s, 0) + 2
            cur_set.add(x)
        m = len(cur_set)
        if m > best_m:
            best_set = cur_set.copy()
            best_m = m
            best_maxcnt = max(cnt.values()) if cnt else 0
            # we keep the best set seen so far
    if best_m == 0:
        return None, np.inf
    # uniform distribution on the support
    seq = np.zeros(n, dtype=np.float64)
    w = 1.0 / best_m
    seq[list(best_set)] = w
    # analytical score: max_counts / m²  multiplied by 2·n
    score = 2.0 * n * best_maxcnt / (best_m * best_m)
    return seq, score

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):

# 3️⃣  Soft‑max + Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, cur_score: float, deadline: float):
    """
    Smooth‑max Adam optimisation.
    Starts from a0 (already on the simplex) and returns a
    (possibly) better candidate before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    # FFT lengths (fixed)
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n - 1)          # for w * a_rev

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.6           # initial learning rate
    beta_start, beta_end = 2.0, 250.0   # soft‑max temperature schedule

    iteration = 0
    while time.time() < deadline:
        # ----- self‑convolution (full) ---------------------------------
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]

        # ----- soft‑max weights ----------------------------------------
        # linear increase of temperature
        elapsed = time.time() - start_time
        total = deadline - (start_time + 0.0)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        # soft‑max (numerically stable)
        z = beta * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        w = exp_z / np.sum(exp_z)

        # ----- gradient of soft‑max objective ---------------------------
        # compute 2 * conv( w , a_rev )
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_grad)
        A_rev = np.fft.rfft(a_rev, n=L_grad)
        conv_w_arev = np.fft.irfft(W * A_rev, n=L_grad)
        # slice the needed part: indices n-1 .. n-1+n-1
        grad = 2.0 * conv_w_arev[n - 1 : n - 1 + n]

        # ----- Adam update ---------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project back to the simplex (keeps non‑negativity and sum=1)
        a = _project_simplex(np.maximum(a, 0.0))

        # ----- keep the best -------------------------------------------
        cur = _objective_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new best {best_score:.6f}")

        # optional learning‑rate decay
        if iteration % 300 == 0:
            lr *= 0.85

    return best_a, best_score

# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Fine‑tune the vector by moving opposite to the sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        # sub‑gradient of the max‑convolution (average over active shifts)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs several complementary
    heuristics (Sidon‑set start, random Dirichlet restarts,
    Adam with a smooth max, final projected‑gradient fine‑tune)
    within the 1000 s time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size – can be increased if time permits
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # (optional) global starter supplied by the environment
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Greedy Sidon‑like construction (≈25 s)
    # ------------------------------------------------------------------
    sidon_deadline = min(deadline, start_time + 25.0)
    if time.time() < sidon_deadline:
        seq, sc = _sidon_initial(n, max_occ=2, deadline=sidon_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] Sidon start score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 55.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=500, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam optimisation (major work phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        best_seq, best_score = _adam_optimize(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Adam finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final projected‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Gradient descent finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
