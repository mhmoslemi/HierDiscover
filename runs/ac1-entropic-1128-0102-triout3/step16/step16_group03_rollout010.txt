"""
Hybrid optimizer for the step‑function upper‑bound problem.

The algorithm works in three clear stages:

1️⃣  **Exploratory sampling of parametrised families** –  a dense random
    sweep over Beta‑shaped densities (α,β ∈ [0.05,10]) and over a
    cosine‑dipped shape (a = 1 + c·cos(π·(i+½)/n), c∈[‑0.99,0]).
    The best few candidates are retained.

2️⃣  **Adam‑style exponentiated‑gradient descent** –  starting from the
    best sampled seed, we run a multiplicative mirror‑descent with
    per‑coordinate Adam moments (first‑ and second‑order).  The update
    uses the sub‑gradient of the current maximal convolution entry and
    projects back onto the simplex by renormalisation.  A decaying
    step‑size (∝1/√t) together with the Adam scaling quickly escapes
    plateaux and finds a lower local minimum.

3️⃣  **Targeted hill‑climbing** –  we repeatedly identify the shift k
    attaining the current maximum of the self‑convolution, then move a
    tiny amount of mass from the index that contributes most to that
    shift to an index that contributes little.  Improvements are always
    accepted; occasional uphill moves are allowed with a small
    temperature schedule.

A final short golden‑section line‑search mixes the obtained sequence
with the uniform distribution; the best mixture is kept.  All phases
respect the allotted time budget (default 1000 s).  The routine returns
a normalised list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat everything within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a_{k‑i}
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _cosine_shape(c: float, n: int) -> np.ndarray:
    """Cosine‑dipped shape: a_i ∝ 1 + c·cos(π·(i+0.5)/n),  c∈[‑1,0]."""
    xs = np.cos(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    a = 1.0 + c * xs
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def _normalize(a: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    b = np.maximum(a, 0.0)
    s = b.sum()
    if s <= 0:
        return np.full_like(b, 1.0 / b.size)
    return b / s


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 1️⃣  Parametric random search (Beta + cosine families)
# ----------------------------------------------------------------------


def _parametric_search(n: int,
                       best_seq: np.ndarray,
                       best_score: float,
                       deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # log‑uniform sampling range for α,β
    log_min, log_max = np.log(0.05), np.log(10.0)

    while time.time() < deadline:
        # ----- Beta draw -------------------------------------------------
        log_a = rng.uniform(log_min, log_max)
        log_b = rng.uniform(log_min, log_max)
        a_val, b_val = np.exp(log_a), np.exp(log_b)
        cand = _beta_shape(a_val, b_val, n)
        if cand is not None:
            sc = _objective_score(cand, n)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a_val:.3f},β={b_val:.3f} → score={sc:.6f}")

        # ----- Cosine draw -----------------------------------------------
        c_val = rng.uniform(-0.99, 0.0)          # negative dip, never below –1
        cand = _cosine_shape(c_val, n)
        sc = _objective_score(cand, n)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine c={c_val:.3f} → score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam‑style exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _adam_expgrad(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam moments (multiplicative update).
    η_t decays as η₀/√t; the per‑coordinate step is η_t·m̂/(√v̂+ε).
    """
    a = a0.copy()
    rng = np.random.default_rng()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    eta0 = 0.6                      # base learning rate
    it = 0
    last_improve_it = 0
    restart_interval = 200_000

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Adam‑EG it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the current max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule
        eta_t = eta0 / math.sqrt(it)

        # multiplicative factor (exponentiated‑gradient style)
        factor = np.exp(-eta_t * m_hat / (np.sqrt(v_hat) + eps))
        # keep values in a numerically safe range
        factor = np.clip(factor, 1e-12, 1e12)

        a_new = a * factor
        a_new = _normalize(a_new)

        # occasional random kick to escape flat plateaux
        if it - last_improve_it > restart_interval:
            perturb = rng.normal(scale=0.01, size=n) * a_new
            a_new = _normalize(np.maximum(a_new + perturb, 0.0))
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Adam‑EG random kick at it={it}")

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climbing (max‑conv based mass transfer)
# ----------------------------------------------------------------------


def _targeted_hill_climb(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of probability mass from the index that contributes
    most to the current maximal convolution shift towards an index that
    contributes little.  Accept improvements always; uphill moves are accepted
    with a low temperature.
    """
    a = a0.copy()
    conv = _conv_self(a, L_fft)
    cur_score = 2.0 * n * conv.max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # find the shift attaining the current maximum
        k = int(conv.argmax())

        # indices i that appear in this shift
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_arr = np.arange(lo, hi + 1, dtype=int)
        j_arr = k - i_arr
        contrib = a[i_arr] * a[j_arr]

        # donor = index with largest contribution (i.e. large a_i * a_j)
        donor = i_arr[np.argmax(contrib)]
        # receiver = index with smallest *total* probability (or random)
        if rng.random() < 0.8:
            receiver = int(np.argmin(a))
        else:
            receiver = rng.integers(n)

        if donor == receiver or a[donor] < 1e-14:
            # nothing to move – try next iteration
            conv = _conv_self(a, L_fft)   # keep conv fresh
            continue

        eps = min(0.02 * a[donor], 1e-4)   # tiny transfer
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        # evaluate new candidate
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            a, conv, cur_score = a_new, conv_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            # simulated‑annealing style uphill acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, conv, cur_score = a_new, conv_new, new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section)
# ----------------------------------------------------------------------


def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few
    golden‑section steps.  The best mixture replaces best_seq/score.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = _normalize(seq)
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = _normalize(lam1 * a + (1.0 - lam1) * uniform)
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.5f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = _normalize(lam2 * a + (1.0 - lam2) * uniform)
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.5f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the runner will cut us off after `budget_s` seconds
    deadline = start_time + budget_s - 0.25

    # problem size – a power of two is convenient for FFTs
    n = 8192          # 2 × the size used previously; still very fast
    L_fft = _next_pow2(2 * n - 1)

    # ---- 0️⃣  Uniform baseline and optional global starter ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ---- 1️⃣  Parametric random search (≈10 % of budget) ----
    phase1_deadline = start_time + 0.10 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _parametric_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )

    # ---- 2️⃣  Adam‑style exponentiated‑gradient descent (≈60 % of budget) ----
    phase2_deadline = start_time + 0.70 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑exp‑grad")
        best_seq, best_score = _adam_expgrad(
            best_seq, n, L_fft, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam‑exp‑grad best≈{best_score:.6f}")

    # ---- 3️⃣  Targeted hill‑climbing (≈20 % of budget) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting targeted hill‑climb")
        best_seq, best_score = _targeted_hill_climb(
            best_seq, n, L_fft, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best≈{best_score:.6f}")

    # ---- 4️⃣  Fine mixing with uniform (remaining time) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ---- Final normalisation and return ----
    best_seq = _normalize(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
