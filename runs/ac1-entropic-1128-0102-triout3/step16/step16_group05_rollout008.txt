   # initialization
   rng = np.random.default_rng(seed)

   n = 4096
   # initial a: from global if exists else uniform

   best_score = _objective_score_normed(a)

   decay = 0.9999
   max_iter = large number, but loop until time.

   # number of iterations since last recompute conv via FFT
   recompute_interval = 2000
   iter_since_recalc = 0
   # track iteration count

   last_report = start_time

   while time.time() < deadline:
       it += 1
       # compute kmax and overlap
       kmax = int(conv.argmax())
       # compute overlap weights
       lo = max(0, kmax - (n-1))
       hi = min(kmax, n-1)
       overlap = np.zeros(n)
       if lo <= hi:
           idx = np.arange(lo, hi+1)
           overlap[idx] = a[kmax - idx]
       # source
       src_candidates = np.where(a > 1e-15)[0]
       if src_candidates.size == 0:

       # dest = minimal overlap, prefer low a
       min_ov = overlap.min()
       dest_candidates = np.where(overlap == min_ov)[0]
       # among those, choose one with minimal a to avoid increasing overlap
       dest = dest_candidates[np.argmin(a[dest_candidates])]
       if src == dest:
           # choose random
           dst = rng.integers(n)
           if dst == src:
               continue
           dest = dst

       eps = a[src] * step_factor
       if eps < 1e-12:
           eps = min_step_factor

       # update conv
       conv[src:src+n] -= 2.0 * eps * a
       conv[dest:dest+n] += 2.0 * eps * a

       # update a
       a[src] -= eps
       a[dest] += eps

       # recompute error corrections small negative conv
       # just clip small negatives to zero occasionally
       if it % 5000 == 0:

       cur_max = conv.max()

           best_a = a.copy()
           # report

def _subgradient_maxconv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    # For each active shift k:
    # contributions: for i from lo to hi: a_{k - i}

        hi = min(n-1, k)

        # contribution to grad[i] is a[k - i]
        idx = np.arange(lo, hi+1)
        grad[idx] += a[k - idx]

        grad = 2.0 * grad / active.size
    return grad

"""
Hill‑climb with fast incremental convolution updates.

We keep the sequence `a` normalised to sum = 1 (so the evaluation
function equals `2·n·max(conv(a,a))`).  The algorithm works in two
phases.

1.  A targeted hill‑climb: at each step we locate the index `kmax`
    where the self‑convolution attains its maximum.  Moving a small
    amount of mass from a position that contributes most to this peak
    to a position that contributes the least reduces (or does not
    increase) the peak.  The convolution is updated **in‑place** using
    the linear formula 
    `conv ← conv – 2·eps·shift(src) + 2·eps·shift(dst)`,
    which costs O(n) per move, allowing millions of cheap moves.

    Periodically the full convolution is recomputed via FFT to correct
    accumulated numerical error.

2.  When the time budget is almost exhausted we run a few iterations
    of a projected sub‑gradient descent (the same routine used in the
    previous baseline) to polish the solution.

The whole routine respects a 1000 s wall‑clock limit, prints progress
and finally returns the best sequence found as a plain ``list``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (real, non‑circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self_fft(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent used for polishing
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    A very short projected sub‑gradient descent (uses FFT each iteration).
    Only a few thousand steps are performed – enough for a final polish.
    """
    a = a0.copy()
    n = a.shape[0]

    step = 5e-3
    min_step = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} "
                  f"best score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # simple back‑tracking line search
        improved = False
        step_try = step
        for _ in range(6):
            cand = a - step_try * direction
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                step = min(step_try * 1.02, 5e-2)
                improved = True
                break
            step_try *= 0.5
        if not improved:
            step = max(step * 0.5, min_step)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Incremental hill‑climb (core of the new search)
# ----------------------------------------------------------------------
def _incremental_hill_climb(a: np.ndarray,
                           conv: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climb that moves mass from the index contributing most to the
    current maximal convolution value to the index contributing least.
    Convolution updates are performed in O(n) using the analytic formula.
    The routine also performs periodic full recomputation of the convolution
    (via FFT) to counteract numerical drift.
    """
    n = a.shape[0]
    L = conv.shape[0]

    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    # parameters
    step_factor = 0.35          # fraction of source mass moved each step
    min_step_factor = 1e-7
    decay = 0.9999
    recompute_every = max(2000, n // 2)   # how often to recompute conv via FFT
    iter_since_recalc = 0
    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # index where the convolution attains its maximum
        kmax = int(conv.argmax())

        # overlap weight w_i = a_{kmax-i} (zero when out of range)
        w = np.zeros(n, dtype=np.float64)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            w[idx] = a[kmax - idx]

        # source: have mass and maximal overlap weight
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(w[src_candidates])]

        # destination: minimal overlap weight (prefer a location with little mass)
        min_w = w.min()
        dst_candidates = np.where(w == min_w)[0]
        if dst_candidates.size == 0:
            dst = rng.integers(n)
        else:
            dst = dst_candidates[np.argmin(a[dst_candidates])]

        if src == dst:
            # pick a random different destination
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        eps = a[src] * step_factor
        if eps < 1e-12:
            eps = max(a[src] * min_step_factor, 1e-12)
        # ensure we never move more than what is available
        eps = min(eps, a[src])

        # ---- update convolution in place (linearised formula) ----
        # conv ← conv – 2·eps·shift(src) + 2·eps·shift(dst)
        # shift(i) is simply the original vector `a` placed starting at position i
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a

        # ---- update the sequence ----
        a[src] -= eps
        a[dst] += eps

        # small numerical clean‑up
        if it % 5000 == 0:
            np.maximum(conv, 0.0, out=conv)

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score * (1 - 1e-12):
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            step_factor = min(step_factor * 1.02, 0.7)
            print(f"[{time.time() - start_time:.2f}s] Hill it={it} "
                  f"new best={cur_score:.6f}")
        else:
            no_improve += 1
            step_factor = max(step_factor * decay, min_step_factor)

        # periodic full recomputation to avoid drift
        iter_since_recalc += 1
        if iter_since_recalc >= recompute_every:
            conv = _conv_self_fft(a)
            iter_since_recalc = 0

        # occasional random jitter if we are stuck
        if no_improve > 5000:
            src = rng.choice(np.where(a > 1e-15)[0])
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
            eps = a[src] * (0.01 * rng.random() + 1e-9)
            conv[src:src + n] -= 2.0 * eps * a
            conv[dst:dst + n] += 2.0 * eps * a
            a[src] -= eps
            a[dst] += eps
            no_improve = 0
            step_factor = max(step_factor, min_step_factor)

    return best_a, best_score

# Main entry point used by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising the evaluation function.
    The routine performs a fast incremental hill‑climb followed by a brief
    projected sub‑gradient polishing phase.  It respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # --------------------------------------------------------------
    # Problem size – moderate to keep each iteration cheap.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise from a global starter if available, else uniform.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                a = starter / s
            else:
                a = uniform.copy()
        else:
            a = uniform.copy()
    except Exception:
        a = uniform.copy()

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                 # ensure exact normalisation
    conv = _conv_self_fft(a)

    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – fast incremental hill‑climb (majority of the budget)
    # --------------------------------------------------------------
    hill_deadline = deadline - 30.0          # reserve ~30 s for polishing
    if time.time() < hill_deadline:
        best_a, best_score = _incremental_hill_climb(
            a, conv, hill_deadline, rng)

    # --------------------------------------------------------------
    # Phase 2 – short projected sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting polishing")
        best_a, best_score = _subgradient_descent(
            best_a, deadline, best_a, best_score, rng)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    # final sanity check (optional – the platform will call evaluate_sequence)
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print(f"[{time.time() - start_time:.2f}s] Final evaluation = {final_val:.6f}")
    except Exception:
        # evaluate_sequence may not be available in the local test harness
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a.tolist()]
