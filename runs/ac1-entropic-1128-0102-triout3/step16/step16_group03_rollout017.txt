"""
Hybrid optimiser (Adam + smooth‑max) for the step‑function evaluation problem.

The algorithm consists of four stages (all bounded by a 1000 s wall‑clock budget).

1️⃣  Broad random search – draws from Dirichlet distributions with a
    log‑uniform concentration and a handful of deterministic Beta‑shaped
    seeds.  The best candidate (measured by the exact objective
    2·n·max(conv)) is retained.

2️⃣  Adam optimiser on the simplex (softmax parametrisation).  The loss is a
    smooth approximation of the ℓ∞‑norm of the convolution:
        L_β(a) = (1/β)·log∑_k exp(β·(a*a)[k]).
    β is increased linearly so the optimisation starts with a smooth
    landscape and gradually focuses on the true maximum.
    The gradient of L_β can be written in closed form using a correlation
    (computed by FFT), which makes each iteration O(n log n).  Adam’s
    adaptive steps and an occasional random perturbation help escape
    plateaus.

3️⃣  Targeted hill‑climbing – tiny mass‑transfer moves (largest → smallest)
    with simulated‑annealing style acceptance, plus occasional random
    swaps of two positions.  Improves any remaining local wiggles.

4️⃣  Fine mixing with the uniform distribution – golden‑section search for
    λ∈[0,1] in  λ·a + (1‑λ)·uniform, keeping the best mixture.

All phases respect the global time budget and print intermediate scores.
The final sequence is normalised to sum 1, non‑negative and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT of length L.
    L must be ≥ 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a probability distribution over the convolution indices that
    approximates the arg‑max of conv (soft‑max with temperature 1/β).
    Returns an array w of the same shape as conv with sum(w)=1.
    """
    # numerical stability – subtract the maximum before exponentiating
    m = conv.max()
    scaled = np.exp(beta * (conv - m))
    w = scaled / scaled.sum()
    return w


def _gradient_theta(theta: np.ndarray,
                    a: np.ndarray,
                    w: np.ndarray,
                    L: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max objective with respect to the
    soft‑max parameters theta.  Uses FFT to obtain the required correlation.
    """
    n = a.shape[0]

    # FFT of a (reuse in the correlation)
    A_fft = np.fft.rfft(a, n=L)

    # FFT of the weight vector w (length 2·n‑1)
    W_fft = np.fft.rfft(w, n=L)

    # Linear correlation w ⋆ a (full length 3·n‑2) via IFFT( W·conj(A) )
    corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=L)
    # Only the first n entries correspond to lags -(n‑1) … 0
    # grad_a_i = 2 * corr_full[n‑1‑i]  i=0…n‑1
    corr_first_n = corr_full[:n]                     # lags –(n‑1)…‑1
    grad_a = 2.0 * corr_first_n[::-1]                # reversed -> indices 0…n‑1

    # Gradient w.r.t. soft‑max parameters:
    # dL/dθ = a * (grad_a - <a,grad_a>)
    dot = np.dot(a, grad_a)
    grad_theta = a * (grad_a - dot)
    return grad_theta


def _softmax(theta: np.ndarray) -> np.ndarray:
    """Stable soft‑max on a vector of arbitrary real numbers."""
    shift = theta.max()
    exp_shift = np.exp(theta - shift)
    return exp_shift / exp_shift.sum()


def _objective_score(a: np.ndarray, n: int) -> float:
    """Exact objective value 2·n·max(conv(a,a)) (a already normalised)."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable `height_sequence_1` exists, initialise from it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 1️⃣  Broad random search (Dirichlet + deterministic Beta seeds)
# ----------------------------------------------------------------------
def _broad_random_search(n: int,
                         best_seq: np.ndarray,
                         best_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # a few deterministic Beta seeds (evaluated once)
    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                   (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3),
                   (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]
    beta_seeds = [_beta_shape(a, b, n) for a, b in beta_params]
    beta_seeds = [v for v in beta_seeds if v is not None]

    while time.time() < deadline:
        # ---- Dirichlet draw (log‑uniform concentration) ----
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s > 0:
            a = sample / s
            sc = _objective_score(a, n)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet(c={c:.3f}) → score={sc:.6f}")

        # ---- occasional Beta seed evaluation ----
        if rng.random() < 0.005 and beta_seeds:
            v = beta_seeds[rng.integers(len(beta_seeds))]
            sc = _objective_score(v, n)
            if sc < best_score:
                best_score, best_seq = sc, v.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam optimiser on the simplex (smooth‑max)
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   n: int,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parameters θ (so that a = softmax(θ)).
    A smooth‑max is used during optimisation (temperature β grows linearly).
    """
    # initialise θ from the current best a (log‑softmax)
    theta = np.log(start_a + 1e-30)

    m = np.zeros_like(theta)
    v = np.zeros_like(theta)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    last_improve_it = 0
    restart_interval = 250_000   # if no improvement for this many iters, perturb

    # FFT length that works for both convolution (2n‑1) and correlation (3n‑2)
    L = _next_pow2(3 * n)

    while time.time() < deadline:
        it += 1

        # ----- transform θ → a (soft‑max) -----
        a = _softmax(theta)

        # ----- exact convolution and score -----
        conv = _conv_self(a, L)                     # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ----- smooth‑max weights -----
        beta = min(2000.0, 0.005 * it)               # linear schedule
        w = _smooth_max_weights(conv, beta)         # same length as conv (2n‑1)

        # ----- gradient w.r.t. θ -----
        grad_theta = _gradient_theta(theta, a, w, L)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decaying step size
        lr = 0.08 / (math.sqrt(it) + 1.0)

        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional perturbation if stuck -----
        if it - last_improve_it > restart_interval:
            perturb = np.random.default_rng().normal(scale=0.1, size=n)
            theta += perturb
            last_improve_it = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climbing with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Small deterministic mass‑transfer moves (largest → smallest) plus
    occasional random swaps.  Accept if improvement, otherwise with a
    temperature‑scaled probability.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- mass transfer move -----
        if rng.random() < 0.75:
            i = np.argmax(a)                 # donor (largest)
        else:
            i = rng.integers(n)              # random donor
        if rng.random() < 0.75:
            j = np.argmin(a)                 # receiver (smallest)
        else:
            j = rng.integers(n)              # random receiver

        if i != j and a[i] > 1e-14:
            eps = min(0.03 * a[i], 5e-4)      # tiny fraction of donor
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
            new_score = 2.0 * n * _conv_self(a_new, _next_pow2(2 * n - 1)).max()
            if new_score < cur_score:
                a, cur_score = a_new, new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, a_new.copy()
                    print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved → {new_score:.6f}")
                continue
            # annealing acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score

        # ----- occasional random swap (helps explore different orderings) -----
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a_new = a.copy()
                a_new[i], a_new[j] = a_new[j], a_new[i]
                new_score = 2.0 * n * _conv_self(a_new, _next_pow2(2 * n - 1)).max()
                if new_score < cur_score or np.random.rand() < 0.01:
                    a, cur_score = a_new, new_score
                    if new_score < best_score:
                        best_score, best_seq = new_score, a_new.copy()
                        print(f"[{time.time() - start_time:.2f}s] Random swap → {new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few golden‑section
    evaluations.  The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressivity
    # ------------------------------------------------------------------
    n = 4096

    # ---- 0️⃣ Initialise with uniform & possible global starter ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Broad random search (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _broad_random_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )

    # ------------------------------------------------------------------
    # 2️⃣ Adam optimiser with smooth‑max (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, n, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Hill‑climbing refinement (≈20 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Fine mixing with uniform (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ---- Final normalisation and return ----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
