"""Hybrid parametric initialization + simulated annealing + gradient refinement to minimise evaluate_sequence.

The algorithm:
- Pre‑compute a distance vector.
- Scan a few parametric families (power and exponential) to obtain a good seed.
- Run a long simulated‑annealing loop that repeatedly transfers a random amount of mass from a source index to a destination.
  Moves are either directed by the sub‑gradient (most/least contributing indices) or random.
  Acceptance follows Metropolis with a temperature that cools exponentially.
- Occasionally recompute the gradient and perform a directed move.
- Keep the best vector seen.
- Finish with a few projected‑gradient steps.
- Return the best candidate as a plain list.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)), assuming sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())

def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    simulated‑annealing walk on the probability simplex (mass transfers
    between coordinates).  The walk mixes directed moves (using the
    sub‑gradient of the max‑convolution) with pure random moves and
    accepts uphill steps with a temperature that cools exponentially.
    A short projected‑gradient refinement is applied at the end.
    The best vector found within the time budget is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    n = 4096
    # Distance from the centre; used by parametric families
    centre = (n - 1) / 2.0
    dist = np.abs(np.arange(n, dtype=float) - centre)
    max_dist = dist.max()

    best_seq = None
    best_score = np.inf

    # Phase 1 – parametric families (cheap exhaustive scan)
    # -----------------------------------------------------------------

            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] power p={p:.2f}, d={d:.0f} → {sc:.6f}")

        sc = _objective_score(a)

            best_seq = a.copy()

        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] default uniform score {best_score:.6f}")

    # Phase 2 – long simulated‑annealing on the simplex
    # -----------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")

            else:

    # -----------------------------------------------------------------

    total = best_seq.sum()
    if total > 0.0:

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective {final_obj:.6f}")
    return [float(x) for x in best_seq.tolist()]

Hybrid parametric initialisation + simulated‑annealing + gradient refinement.

The routine proceeds in three stages:

1️⃣  Scan a few low‑dimensional parametric families (power law and exponential
    shapes) to obtain a good‐starting vector on the simplex.
2️⃣  Run a long simulated‑annealing walk that repeatedly transfers a small
    amount of mass from one coordinate to another.  Moves are a mixture of
    random transfers and directed transfers (from the index that contributes
    most to the current maximal convolution to the one that contributes least).
    Acceptance follows the Metropolis rule with an exponentially cooling temperature.
3️⃣  Finish with a short projected‑gradient descent (back‑tracking line‑search)
    to fine‑tune the best candidate.

The best vector encountered within the 1000 s budget is returned as a plain
list of floats.  All helper functions are written at top level and avoid
closures, matching the challenge constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# ----------------------------------------------------------------------

    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    n = 4096
    centre = (n - 1) / 2.0
    dist = np.abs(np.arange(n, dtype=float) - centre)   # distance from centre
    max_dist = dist.max()

    best_seq = None
    best_score = np.inf

    for p in p_vals:
        for d in d_vals:

                break

            else:
