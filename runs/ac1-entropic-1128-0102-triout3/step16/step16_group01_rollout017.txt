"""
A fast, high‑precision search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Very fine parametric sweep** – the sequence is restricted to the
    family a_i ∝ (i+1)^p · r^i (geometric × power law).  Both the
    geometric rate `r` (expressed via λ = -log r) and the exponent `p`
    are scanned on dense grids that focus on the region where the
    optimum of this family is known to lie (λ ≈ 0.5·n⁻¹ … 1.5·n⁻¹,
    p ≈ -5 … 2).  The best vector from the sweep is stored.

2️⃣  **Mirror‑descent refinements** – starting from the best sweep
    candidate we run two classic optimisation phases that operate on
    the exact L∞‑objective:

        • a projected sub‑gradient descent with a back‑tracking line‑search
          (few thousand iterations, aggressive step‑size decay);
        • an exponentiated‑gradient (multiplicative) update with a slowly
          annealed learning‑rate, keeping the vector on the simplex.

3️⃣  **Final random hill‑climb** – tiny mass‑transfer moves (simulated‑annealing
    acceptance) are performed until the time budget expires, guaranteeing
    that any last improvement is captured.

All phases respect the 1000 s budget, stay on the probability simplex,
and finally return the best sequence found as a plain Python list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – assumes the sum is ≈ 1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family a_i ∝ (i+1)^p * r^i  (geometric × power)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """
    Build a vector a_i ∝ (i+1)^p * exp(-lam*i)   (lam = -log r).
    The result is normalised to unit L1 norm.
    """
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)                # pre‑computed log(i+1)
    log_shape = -lam * i + p * log_idx       # log of unnormalised shape
    # shift for numerical stability
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes the diagonal correctly
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (multiplicative updates)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) updates:
        a ← a * exp(- η * g)   ,   η annealed.
    After each update we renormalise to stay on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.8
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # decreasing step length
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(10):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # very gentle uphill move – may help escape tiny plateaus
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fine parametric sweep over (λ, p) – ≈70 s.
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    # λ = -log(r)  – we scan the range where the optimum is expected.
    min_lambda = 4e-05          # very close to 0 (r≈0.99996)
    max_lambda = 1.5e-03        # r≈0.9985
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    # power exponent
    p_grid = np.linspace(-6.0, 3.0, 91)                  # step 0.1

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        # pre‑compute the λ‑dependent part once per λ
        base = -lam * np.arange(n, dtype=np.float64)
        for p in p_grid:
            # log‑shape = -lam*i + p*log(i+1)
            log_shape = base + p * np.log(np.arange(1, n + 1, dtype=np.float64))
            log_shape -= np.max(log_shape)          # stabilise exp
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6f} "
                      f"p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (more intensive now that we have a good start).
    # --------------------------------------------------------------
    if time.time() < deadline - 250.0:
        sub_deadline = deadline - 250.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_deadline, start_time,
                                                    best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient phase (fine‑grained multiplicative updates).
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        exp_deadline = deadline - 150.0
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_deadline, start_time,
                                              best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 4️⃣  Final random hill‑climb – mass transfers with a tiny acceptance rule.
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 5e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 2e-3   # ≤0.2 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
