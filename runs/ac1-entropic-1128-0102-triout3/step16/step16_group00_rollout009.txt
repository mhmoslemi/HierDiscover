"""
Propose a step‑function (non‑negative vector on the simplex) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a hybrid of three simple optimisation stages:

1️⃣  A short burst of random Dirichlet draws to obtain a decent starting point.
2️⃣  Simulated‑annealing on the simplex (mass moves guided by the sub‑gradient).
3️⃣  Fine‑tuning with a mirror‑descent style exponential‑gradient update
    (KL‑projection is implicit in the soft‑max normalisation).

All heavy work (FFT self‑convolution, sub‑gradient) is vectorised.
The routine respects a 1000 s budget – it stops as soon as the time limit is
reached and returns the best vector seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, sub‑gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    The returned vector has the same length as a.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution from the partner entry
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 accounts for symmetry
    return grad


def _objective_score(a: np.ndarray) -> float:
    """Assumes sum(a)=1. Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (used only by the final PGD stage – kept for safety)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Simulated annealing (mass transfer guided by sub‑gradient)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, cur_score: float, deadline: float):
    """Anneal the vector by moving mass from the most to the least contributing indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # temperature proportional to current objective (small but non‑zero)
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9999
    max_frac_start = 0.30   # up‑front we may move a large fraction of the source mass
    max_frac_end = 0.01     # later we fine‑tune

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # linear schedule for the allowed transfer fraction
        t_rem = deadline - time.time()
        total_rem = deadline - start_time
        frac = max_frac_end + (max_frac_start - max_frac_end) * max(t_rem / total_rem, 0.0)

        delta = a[src] * rng.random() * frac
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0:
            continue
        cand_score = _objective_score(cand)
        diff = cand_score - cur_score
        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")
        T = max(T * decay, 1e-12)
    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exponential‑gradient (mirror‑descent) refinement
# ----------------------------------------------------------------------
def _exponential_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Mirror‑descent on the simplex using the KL‑proximal (exponential) update:
        a_{new} ∝ a * exp(-η * ∇f(a))
    where ∇f is a sub‑gradient of the max‑convolution objective.
    """
    a = np.copy(a0)
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    it = 0
    eta0 = 0.03            # initial step size (tuned experimentally)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG new best {best_score:.6f}")

        # sub‑gradient of the *objective* (2·n·max(conv))
        grad_a = _subgradient_maxconv(a, conv) * (2.0 * n)

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # exponential update (clip to avoid overflow)
        delta = -eta * grad_a
        delta = np.clip(delta, -50.0, 50.0)
        a = a * np.exp(delta)
        # re‑normalise to stay on the simplex
        s = a.sum()
        if s == 0.0:                     # numerical safeguard
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional random jitter to help escape flat plateaux
        if it % 2000 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Optional final projected‑gradient descent (very short, just in case)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Plain projected gradient descent with a simple back‑tracking line‑search."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv) * (2.0 * n)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break
        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the allotted time.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size (use the length of a global starter if present)
    # ------------------------------------------------------------------
    n = 4096
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            # normalise the starter (it may not be a probability vector)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                best_seq = starter.copy()
                best_score = _objective_score(best_seq)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {best_score:.6f}")
            else:
                best_seq = np.full(n, 1.0 / n)
                best_score = _objective_score(best_seq)
        else:
            raise ValueError
    except Exception:
        # fallback: uniform vector
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisers (≈20 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 20.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=500, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing (≈600 s)
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, start_time + 620.0)
    if time.time() < anneal_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, best_score, anneal_deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exponential‑gradient (mirror‑descent) refinement (≈300 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        eg_deadline = deadline - 5.0
        print(f"[{time.time() - start_time:.2f}s] Starting exponential‑gradient refinement")
        best_seq, best_score = _exponential_gradient_descent(best_seq, eg_deadline)
        print(f"[{time.time() - start_time:.2f}s] Exponential‑gradient finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Very short final projected‑gradient polishing (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final PGD polish")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final PGD finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
