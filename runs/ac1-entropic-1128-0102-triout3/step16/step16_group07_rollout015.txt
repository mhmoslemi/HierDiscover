def _lbfgs_optimize(a0, start_time, deadline, best_seq, best_score):
    import scipy.optimize

    n = a0.shape[0]
    # initial x = log(a0) - log mean maybe
    x0 = np.log(np.maximum(a0, 1e-16))

    # We'll define ranges of tau values:
    tau_values = [5.0, 20.0, 80.0, 200.0, 500.0]
    for tau in tau_values:
        if time.time() >= deadline:
            break
        def func_and_grad(x):
            # compute a = softmax(x)
            max_x = np.max(x)  # for numeric stability? Not needed but for a.
            # compute a = exp(x - max_x) then normalize
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()
            # compute convolution
            conv = _conv_self(a)
            # smooth max
            m = conv.max()
            # stable log-sum-exp
            shifted = conv - m
            exp_shift = np.exp(tau * shifted)
            sum_exp = exp_shift.sum()
            smooth_max = m + np.log(sum_exp) / tau
            # objective
            obj = 2.0 * n * smooth_max
            # compute weights w = exp(tau*(conv - m)) / sum_exp
            w = exp_shift / sum_exp
            # gradient wrt a
            a_rev = a[::-1]
            # conv of w and a_rev
            L = _next_pow2(3 * n - 2)
            w_fft = np.fft.rfft(w, n=L)
            a_rev_fft = np.fft.rfft(a_rev, n=L)
            conv_w_full = np.fft.irfft(w_fft * a_rev_fft, n=L)[:n]
            grad_a = 2.0 * conv_w_full
            # gradient wrt x (softmax param)
            dot = np.dot(grad_a, a)
            grad_x = a * (grad_a - dot)
            return obj, grad_x.astype(np.float64)
        # run minimize
        options = {'maxiter': 2000, 'disp': False}
        # Provide jac=True
        res = scipy.optimize.minimize(fun=func_and_grad,
                                      x0=x0,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options=options)
        # map result to a
        x_opt = res.x
        exp_x = np.exp(x_opt - np.max(x_opt))
        a_opt = exp_x / exp_x.sum()
        sc = _objective_score(a_opt)
        if sc < best_score:
            best_score, best_seq = sc, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] L-BFGS tau={tau} score={sc:.6f}")
        # prepare for next tau: start from current x
        x0 = x_opt
    return best_seq, best_score

"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)²)

The pipeline differs from the reference in three ways:

1.  The parametric sweep now includes Gaussian and raised‑cosine families,
    which were missing before.
2.  A deterministic L‑BFGS optimisation of a smooth‑max surrogate is
    performed (instead of the hand‑crafted Adam loop).  The smoothness
    parameter τ is increased gradually, giving a fast‑converging quasi‑Newton
    refinement that usually beats the Adam stage.
3.  The Dirichlet sampling loop now runs as long as time permits, giving many
    more random starts, and the final hill‑climb is executed after the L‑BFGS
    pass.

All steps respect the global time budget (default = 1000 s).  The routine
returns a plain Python list of floats that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

    # ---- logistic -------------------------------------------------------

    for mu_f in mu_factors:

        mu = mu_f * n
        for s_f in s_factors:

            s_val = s_f * n
            if s_val < 1e-6:

            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)

        if s == 0:

        sc = _objective_score(seq)

def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        trials += 1
        # draw concentration from a wide log‑uniform range
        log_c = rng.uniform(np.log(0.01), np.log(50.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet #{trials} c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  L‑BFGS optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    start_time: float,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimises a smooth approximation of max(conv) using L‑BFGS‑B.
    The parameters τ (smoothness) are increased gradually.
    """
    # initialise from the current point (soft‑max parameters)
    x0 = np.log(np.maximum(a0, 1e-16))

    # schedule of τ values – start smooth, finish sharp
    tau_schedule = [5.0, 20.0, 80.0, 200.0, 500.0]

    for tau in tau_schedule:
        if time.time() >= deadline:
            break

        # closures we need inside the optimiser
        L_fft = _next_pow2(3 * n - 2)

        def func_and_grad(x):
            # a = soft‑max(x)
            max_x = np.max(x)                     # stability
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()

            # convolution a*a
            conv = _conv_self(a)

            # smooth max via log‑sum‑exp (stable)
            m = conv.max()
            shifted = conv - m
            exp_shift = np.exp(tau * shifted)
            sum_exp = exp_shift.sum()
            smooth_max = m + np.log(sum_exp) / tau
            obj = 2.0 * n * smooth_max

            # gradient wrt a (same as for the Adam step)
            w = exp_shift / sum_exp                # soft‑max weights over shifts
            a_rev = a[::-1]
            w_fft = np.fft.rfft(w, n=L_fft)
            a_rev_fft = np.fft.rfft(a_rev, n=L_fft)
            conv_w_full = np.fft.irfft(w_fft * a_rev_fft, n=L_fft)[:n]
            grad_a = 2.0 * conv_w_full

            # chain rule to get gradient wrt x (soft‑max parametrisation)
            dot = grad_a @ a
            grad_x = a * (grad_a - dot)

            return obj, grad_x.astype(np.float64)

        # L‑BFGS‑B call – limit iterations to keep under the global deadline
        maxiter = 500
        opt_res = scipy.optimize.minimize(fun=func_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': maxiter,
                                                   'disp': False})
        # map back to a probability vector
        x_opt = opt_res.x
        exp_x = np.exp(x_opt - np.max(x_opt))
        a_opt = exp_x / exp_x.sum()

        sc = _objective_score(a_opt)
        if sc < best_score:
            best_score, best_seq = sc, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS τ={tau:.0f} score={sc:.6f}")

        # feed the result into the next τ‑stage
        x0 = x_opt

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic projected gradient descent (Euclidean projection onto the simplex)
    using the exact sub‑gradient of max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.5          # initial step size
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        eta_t = eta / (1.0 + it * 1e-4)
        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.995
            if no_improve % 500 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = _project_simplex(np.maximum(a_jit, 0.0))
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-8:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing (unchanged)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The pipeline consists of:
        1. parametric sweep (including Gaussian / raised‑cosine)
        2. unrestricted Dirichlet sampling
        3. L‑BFGS optimisation of a smooth‑max surrogate
        4. projected sub‑gradient descent on the exact max‑convolution
        5. final hill‑climb / simulated‑annealing
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # problem size (larger n gives a slightly better theoretical limit)
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # baseline – uniform distribution
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1) – if supplied by the judge
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (larger families)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0    # reserve ~250 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling (as many as time allows)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣  L‑BFGS optimisation of the smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting L‑BFGS optimisation")
        best_seq, best_score = _lbfgs_optimize(
            best_seq, n, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After L‑BFGS best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent on exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD descent")
        best_seq, best_score = _pgd_descent(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # final normalisation and return
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
