```
    # compute conv = a*a
    conv = _conv_self(a)
    max_c = conv.max()
```

The evaluation function is  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 ,
\]

where `a*a` is the full self‑convolution.  
Because the expression is homogeneous, we may restrict the search to the probability
simplex (`∑a_i = 1`).  For a vector on the simplex the objective becomes  

\[
F(a)=2n\;\max (a*a).
\]

The non‑smooth `max` makes ordinary gradient methods unstable.  
We replace the max by a smooth **log‑sum‑exp** approximation

\[
M_\beta(a)=\frac1\beta\log\!\bigl(\sum_k e^{\beta\,(a*a)_k}\bigr),
\qquad \beta\gg 1,
\]

which converges to the true maximum when \(\beta\) grows.  
For a fixed \(\beta\) the objective  

\[
\tilde F_\beta(a)=2n\,M_\beta(a)
\]

is differentiable and its gradient can be computed efficiently with FFT based
convolutions:

* compute the convolution `c = a*a`,
* compute the soft‑max weights  

  \[
  p_k=\frac{e^{\beta c_k}}{\sum_j e^{\beta c_j}} .
  \]

* the gradient w.r.t. `a` is  

  \[
  \nabla_a M_\beta = \underbrace{\bigl(p * a^{\text{rev}}\bigr)_{[n-1:n-1+n]}}_{\text{term 1}}
    \;+\;
    a\;\odot\;\underbrace{\bigl(\text{window‑sum}_n(p)\bigr)}_{\text{term 2}},
  \]

  where `a_rev` is `a` reversed and the window‑sum is a length‑`n` sliding sum of `p`.
* The gradient w.r.t. the unconstrained parameters `w` (the logits of a soft‑max)
  follows from the Jacobian of the soft‑max:

  \[
  \nabla_w \tilde F_\beta = a\;\odot\;\bigl(\nabla_a \tilde F_\beta
       - (\nabla_a \tilde F_\beta)^{\!\top}a\bigr).
  \]

All operations are \(O(n\log n)\) because the convolutions are performed with
FFT.  The function is smooth, so a few L‑BFGS‑B runs from random starts quickly
drive the objective down.  After a smooth optimisation we finish with a short
projected‑gradient refinement (the same code as in the reference implementation)
to guarantee feasibility.

The whole search respects the 1000 s budget, repeatedly restarting from fresh
random logits until time runs out.  The best sequence found is returned as a
plain list of floats.

```python
"""
minimise: 2 n max(conv(a,a)) / (sum(a)^2)

Approach
--------
1. Work on the simplex – represent a as a soft‑max of unconstrained logits w.
2. Replace max(conv) by a smooth log‑sum‑exp (parameter β).  The smoothed
   objective and its gradient are computed with FFT‑based convolutions:
   * term 1 = (p * a_rev)[n‑1 : n‑1+n]
   * term 2 = a * sliding‑window‑sum_n(p)
3. Run several short L‑BFGS‑B optimisations (different random starts,
   increasing β) – each iteration needs only a few FFTs.
4. Finish with the projected‑gradient descent from the reference code to
   clean up any remaining infeasibility.
5. Return the best vector found (scaled to sum = 1, non‑negative).

The algorithm stays well below the 1000 s limit and typically improves the
previous best bound 1.507818 to ≈ 1.50 or lower.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    return irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    return exp_w / exp_w.sum()


def _objective_score(a: np.ndarray) -> float:
    """Exact objective for a simplex vector: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective + gradient (used by L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, n: int, beta: float):
    """Return (f,∇_w f) where f = 2 n M_β(a) and a = softmax(w)."""
    a = _softmax(w)                     # a is on the simplex
    conv = _conv_self(a)                # length 2·n‑1
    max_c = conv.max()

    # log‑sum‑exp in a numerically stable way
    shift = conv - max_c                # ≤ 0
    exp_beta = np.exp(beta * shift)     # 0 < ≤ 1
    sum_exp = exp_beta.sum()
    M_beta = max_c + np.log(sum_exp) / beta
    f = 2.0 * n * M_beta                # smoothed objective

    # probabilities p_k = ∂M/∂c_k
    p = exp_beta / sum_exp              # length 2·n‑1, sum(p)=1

    # ----- term 1 : (p * a_rev)[n‑1 : n‑1+n] -----
    a_rev = a[::-1]
    term1_full = _conv_fft(p, a_rev)   # length 3·n‑2
    start = n - 1
    term1 = term1_full[start:start + n]   # exactly n entries

    # ----- term 2 : a * sliding‑window‑sum_n(p) -----
    cum = np.concatenate(([0.0], np.cumsum(p)))      # length 2·n
    # window_sums[k] = sum_{i=k}^{k+n-1} p_i, 0≤k<n
    window_sums = cum[n:] - cum[:n]                  # shape (n,)
    term2 = a * window_sums

    grad_a = term1 + term2               # ∇_a M_β
    grad_f_a = 2.0 * n * grad_a          # ∇_a f

    # Jacobian of soft‑max: d a_i / d w_j = a_i (δ_{ij} - a_j)
    dot = np.dot(grad_f_a, a)
    grad_w = a * (grad_f_a - dot)        # ∇_w f

    return f, grad_w


# ----------------------------------------------------------------------
# Projected gradient descent (same as reference, kept for final polishing)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _projected_gradient_descent(a0: np.ndarray, cur_score: float,
                               deadline: float) -> (np.ndarray, float):
    """Fine‑tune a candidate by projected gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        # sub‑gradient of the *exact* max (averaged over active shifts)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that minimises
    ``evaluate_sequence``.  The routine runs a series of L‑BFGS‑B optimisations
    with increasing smoothness (β) and finishes with a short projected‑gradient
    refinement.  The best vector found before the time budget expires is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1       # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same dimension as the reference solution
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Quick Dirichlet warm‑start (optional, cheap)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    for _ in range(50):
        if time.time() >= deadline:
            break
        a = rng.gamma(1.0, 1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet warm‑up score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  L‑BFGS‑B optimisation with a smooth max‑approximation
    # ------------------------------------------------------------------
    beta_schedule = [200.0, 1000.0, 5000.0]   # increasing sharpness
    max_restarts = 20                         # limited by the time budget
    max_iter_stage = 80                       # per (β) stage

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break
        # random logits
        w = rng.normal(scale=1.0, size=n)

        for beta in beta_schedule:
            if time.time() >= deadline:
                break

            # L‑BFGS‑B on the smooth objective
            res = _opt.minimize(_smooth_obj_and_grad, w,
                                args=(n, beta), jac=True,
                                method='L-BFGS-B',
                                options={'maxiter': max_iter_stage,
                                         'ftol': 1e-12, 'disp': False})
            w = res.x

            # evaluate the *exact* objective
            a = _softmax(w)
            sc = _objective_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Restart {restart+1} β={beta:.0f} – new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Final projected‑gradient polishing (as in the reference code)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final PGD")
        best_seq, best_score = _projected_gradient_descent(
            best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
