def _adam_optimize(start_seq, deadline, best_seq, best_score):
    a = start_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t} score={cur_score:.6f}")
        # compute subgradient of max conv
        grad = _subgradient_maxconv(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.1  # initial step size; maybe adapt with iteration: step = 0.1 / sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

"""
Optimiser for the step‑function evaluation problem (target ≤ 1.503).

The algorithm consists of three stages:

1.  **Diverse warm‑starts** – a collection of random Dirichlet draws,
    decreasing exponential shapes and possibly a global starter.
2.  **Smoothed Adam optimisation** – minimise a smooth approximation of the
    max‑convolution (τ·log‑sum‑exp) with an Adam‑style update and Euclidean
    projection onto the probability simplex.  The temperature τ is annealed,
    allowing the optimiser to first explore a wide basin and later focus on
    the true max.
3.  **Tiny‑mass hill‑climb** – a final greedy local search that moves a very
    small amount of mass from a “source” index to a “target” index whenever
    it strictly improves the true objective.

All phases respect a shared 1000 s wall‑clock budget.  The best sequence
found (normalised to sum = 1) is returned as a plain Python list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a."""
    return _conv_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Exact evaluation score 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) (average over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of τ·log‑sum‑exp(conv/τ) w.r.t. a.
    w are the normalised soft‑max weights (sum to 1) of the same length as conv.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution w * a_rev (full linear) – length = len(w)+n-1 = 3n‑2
    full = _conv_fft(w, a_rev)
    # needed slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _rng():
    """Singleton RNG – seeded from the global seed."""
    return np.random.default_rng()


def _dirichlet_initial(n: int) -> np.ndarray:
    """Draw a random point on the simplex using a log‑uniform concentration."""
    logc = _rng().uniform(-2.0, 0.7)          # concentration ∈ [0.01, 5]
    c = 10 ** logc
    a = _rng().gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


def _exponential_initial(n: int) -> np.ndarray:
    """Monotonically decreasing exponential shape (parameter r close to 1)."""
    r = _rng().uniform(0.97, 0.9999)
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a += 1e-14                     # avoid exact zeros
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed Adam optimisation
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation τ·log‑sum‑exp(conv/τ).
    The true objective is monitored and the best ever sequence is kept.
    """
    a = start_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    tau = 5e-3                     # initial temperature

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)       # length 2n‑1
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t} score={cur_score:.6f}")

        # smooth‑max weights
        w = np.exp(conv / tau)
        w /= w.sum()

        # gradient of the smooth objective
        grad = _grad_from_weights(a, w)

        # Adam update (projected onto simplex after the step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step_size = 0.15           # nominal step size in the projected space
        a_new = a - step_size * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a = _project_simplex(a_new)

        # anneal temperature slowly
        if t % 800 == 0:
            tau *= 0.92
            if tau < 1e-5:
                tau = 1e-5

        # occasional random jitter to escape plateaus
        if t % 3000 == 0:
            rng = _rng()
            i, j = rng.choice(n, size=2, replace=False)
            eps_mass = a[i] * (rng.random() * 0.002 + 1e-6)
            a[i] -= eps_mass
            a[j] += eps_mass
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny‑mass hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Greedy local search moving a very small fraction of mass."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = _rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # choose a source (large gradient) and a target (small gradient)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.003 + 1e-6)   # ≤ 0.3 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to break a possible plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.002 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields an
    evaluation score below the target 1.503.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # problem size – a moderate n gives a good speed / quality trade‑off
    n = 2048
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – several diverse warm‑starts, each followed by Adam
    # ------------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.70   # ~70 % of budget for optimisation
    while time.time() < phase1_end:
        # pick a random initialisation method
        if _rng().random() < 0.55:
            a0 = _dirichlet_initial(n)
        else:
            a0 = _exponential_initial(n)
        if a0 is None:
            continue
        # run a short Adam optimisation (≈5 s) on this start
        sub_deadline = time.time() + 5.0
        best_seq, best_score = _adam_optimize(a0, sub_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 2 – a final, longer Adam run from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Long Adam from current best")
        best_seq, best_score = _adam_optimize(best_seq,
                                              deadline - 0.2,   # leave room for hill‑climb
                                              best_seq,
                                              best_score)

    # ------------------------------------------------------------------
    # Phase 3 – final tiny‑mass hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Normalise and return the result
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
