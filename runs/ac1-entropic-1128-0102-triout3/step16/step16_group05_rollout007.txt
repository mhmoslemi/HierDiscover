"""
A multi‑stage optimizer for the step‑function bound.

1. **Initial seeding** – Use the historic best (`height_sequence_1`) if
   present, otherwise try a battery of analytic window shapes (uniform,
   Kaiser, Chebyshev, DPSS) and a quick random Dirichlet sweep.  All candidates
   are normalised to sum = 1 and scored with a fast FFT‑based convolution.

2. **Aggressive hill‑climb** – Repeatedly moves a small fraction of mass from
   the “most over‑represented’’ index (largest gradient of the active
   max‑convolution constraints) to the “least represented’’ one.
   Several step‑sizes are tested each iteration; the best improving move is
   accepted.  Random jitter is injected after long stagnation.

3. **Simulated annealing fine‑tuning** – A classic SA loop proposes random
   source→dest transfers (up to a few per‑cent of the source mass) and
   accepts them either if they improve the objective or with a temperature‑
   dependent probability.  Temperature decays linearly to zero.

The three phases respect a 1000 s wall‑clock budget.  The final vector is
projected onto the probability simplex and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Phase 1 – generate a diverse pool of seeds
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator):
    """Yield a sequence of candidate vectors (already normalised)."""
    # uniform
    yield np.full(n, 1.0 / n)

    # try scipy windows if available
    try:
        from scipy.signal import windows as spw
        # Kaiser (beta from 0 to 14)
        for beta in np.linspace(0, 14, 7):
            w = spw.kaiser(n, beta)
            w = np.maximum(w, 0.0)
            if w.sum() > 0:
                yield w / w.sum()

        # Chebyshev (attenuation from 20 to 200 dB)
        for att in [20, 40, 60, 80, 120, 180, 240]:
            w = spw.chebwin(n, att)
            w = np.maximum(w, 0.0)
            if w.sum() > 0:
                yield w / w.sum()

        # DPSS (time‑bandwidth product NW)
        for NW in [2.0, 2.5, 3.0, 3.5, 4.0]:
            try:
                w = spw.dpss(n, NW, Kmax=1, sym=False)[0]
                w = np.maximum(w, 0.0)
                if w.sum() > 0:
                    yield w / w.sum()
            except Exception:
                pass
    except Exception:
        pass

    # a few random Dirichlet draws with varying concentration
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(40):
            vec = rng.dirichlet(np.full(n, conc))
            yield vec

# Phase 2 – aggressive hill‑climb with multi‑step sizes
# ----------------------------------------------------------------------
def hill_climb_multi(a0: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float,
                     rng: np.random.Generator):
    """Hill‑climb that tests several step fractions each iteration."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0
    fractions = (0.001, 0.003, 0.007, 0.015, 0.03, 0.06)  # up to 6 % of source

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # choose source (largest gradient among entries that still have mass)
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-14:
                continue

        improved = False
        best_local = best_score
        best_candidate = None

        for f in fractions:
            eps = a[src] * f
            if eps <= 0:
                continue
            a_c = a.copy()
            a_c[src] -= eps
            a_c[dst] += eps
            # maintain feasibility (simple projection not needed as sum unchanged)
            # evaluate
            score_c = _objective_score(a_c)
            if score_c < best_local:
                best_local = score_c
                best_candidate = a_c
                improved = True

        if improved and best_candidate is not None:
            a = best_candidate
            if best_local < best_score:
                best_score = best_local
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to escape flat region
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated annealing fine‑tuning
# ----------------------------------------------------------------------
def annealing_search(a0: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float,
                     rng: np.random.Generator):
    """Simple SA that moves mass between random indices."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # linear temperature schedule
    total_time = deadline - time.time()
    while time.time() < deadline:
        # update temperature (relative to remaining fraction)
        remaining = max(0.0, deadline - time.time())
        temp = (remaining / total_time) * cur_score * 0.01  # small scale

        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.choice(n)
        if src == dst:
            continue
        # propose epsilon up to a few percent of source mass
        max_eps = a[src] * min(0.05, max(0.001, temp / cur_score))
        eps = rng.random() * max_eps
        if eps <= 0:
            continue

        a_c = a.copy()
        a_c[src] -= eps
        a_c[dst] += eps

        new_score = _objective_score(a_c)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            a = a_c
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved to {best_score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = annealing_search(
            best_seq, deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Final sanity normalisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    # return as plain Python list
    return [float(x) for x in best_seq]
