    best = a.copy()
    best_score = _objective_score(a)
    tau = 5.0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute softmax weights
        w = np.exp(tau * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:

        else:
            w = w / w_sum

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine builds on the sub‑gradient descent / hill‑climb that was used
previously, but adds a final simulated‑annealing phase (larger random mass
moves with a temperature schedule) to escape shallow plateaus.  Several
random restarts are performed within the 1000 s wall‑clock budget; the best
candidate found over all restarts is returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (same as the original helper code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Local optimiser – sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(conv).  A simple back‑tracking line
    search adapts the step size; random jitter is added when no improvement
    can be found.
    """
    a = a0.copy()
    n = a.shape[0]

    step_scale = 5e-3            # initial step magnitude (L2 norm)
    min_step = 1e-10
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} score={cur_score:.6f}")

        # sub‑gradient
        grad = _subgradient_maxconv(a, conv)
        grad_norm = np.linalg.norm(grad) + 1e-16

        # back‑tracking line search
        step = step_scale
        improved = False
        for _ in range(6):
            direction = grad / grad_norm
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            new_score = _objective_score(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # shrink step and inject a tiny jitter
            step_scale = max(step_scale * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score

# Hill‑climb (small deterministic moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that repeatedly moves a tiny fraction of mass from the
    “over‑represented’’ index to the “under‑represented’’ one.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.01 + 1e-8)   # up to 1 %
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

# Simulated annealing – larger random moves with temperature decay
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Large random mass transfers (up to 20 % of a source entry) accepted with a
    Metropolis probability.  The temperature decays linearly, allowing the
    search to escape plateaux and settle near a better local minimum.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    total = max(deadline - time.time(), 0.0)
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        # temperature drops from ~1e‑2 to ~1e‑6
        temp = max(1e-6, (1.0 - frac) * 1e-2)

        # pick a source with non‑zero mass
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # move a random fraction (up to 20 %) of the source mass
        eps = a[src] * (rng.random() * 0.2 + 1e-9)
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # maintain feasibility (non‑negative, sum = 1)
        if a_new[src] < -1e-12:
            continue

        new_score = _objective_score(a_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal new best {best_score:.6f}")

    return best_a, best_score

        # initialise a random probability vector (Dirichlet with α=0.5)
        alpha = np.full(n, 0.5)
        a0 = rng.gamma(alpha, 1.0)
        a0 = _project_simplex(a0)

        best_seq, best_score = _subgradient_descent(
            a0, slice_deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Stage 2 – deterministic hill‑climb (use a chunk of the budget)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 150.0   # reserve ~2.5 min for annealing
    if time.time() < hill_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, hill_deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing (final refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, rng)

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
