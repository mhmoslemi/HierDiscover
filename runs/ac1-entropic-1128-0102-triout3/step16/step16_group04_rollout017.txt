def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_self_fft(a, L):
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[:2*a.shape[0]-1]

def _project_simplex(v):
    # returns projected vector onto simplex ∑ v_i =1, v_i >=0
    # Compute sorted v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Search for a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1️⃣ **Warm‑start** – a mixture of the global starter (if present),
   the uniform vector and many Dirichlet draws with random
   concentration.  The best of these cheap candidates is kept.

2️⃣ **Variance‑flattening gradient descent** – the convolution is forced
   to be as flat as possible (minimising the L²‑variance of the
   convolution).  The gradient of the variance is obtained analytically
   via FFT and the iterate is projected onto the probability simplex
   (non‑negative, sum = 1) with the standard Duchi projection.

3️⃣ **Targeted hill‑climb** – while time permits the algorithm repeatedly
   identifies the index `k` where the convolution attains its current
   maximum, picks the contributing pair `(i, k‑i)` that makes the largest
   contribution, and transfers a tiny amount of mass from the larger
   member of the pair to a random other coordinate.  Accept only
   improving moves; this aggressively reduces the peak of the
   convolution.

After hill‑climbing a short uniform‑mixing + random‑pair fine‑tuning
stage is executed (as in the reference implementation).  The whole
routine respects the 1000 s budget, prints progress, and finally
returns a plain Python list of floats that sum (approximately) to one.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a)=1 is assumed)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _variance_gradient(a: np.ndarray,
                       conv: np.ndarray,
                       mean: float,
                       L_corr: int) -> np.ndarray:
    """
    Gradient of   Σ_k (conv_k – mean)²   w.r.t. a.
    Formula:   ∇ = 4·corr(conv‑mean , a),
    where the correlation is the valid part of the convolution of
    (conv‑mean) reversed with a.
    """
    diff = conv - mean                      # shape (2n‑1,)
    diff_rev = diff[::-1]
    Dr = np.fft.rfft(diff_rev, n=L_corr)
    Ar = np.fft.rfft(a, n=L_corr)
    full = np.fft.irfft(Dr * Ar, n=L_corr)
    start = a.shape[0] - 1                  # pick the “valid” centre part
    grad = 4.0 * full[start:start + a.shape[0]]
    return grad


def _max_contributions(a: np.ndarray, k: int) -> tuple[int, int, float]:
    """
    For a given convolution index k, return the pair (i, j) with the
    largest product a_i·a_j (where i+j=k) together with that product.
    """
    n = a.shape[0]
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start > i_end:
        return -1, -1, 0.0
    idx = np.arange(i_start, i_end + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    max_pos = np.argmax(contrib)
    i = int(idx[max_pos])
    j = int(k - i)
    return i, j, float(contrib[max_pos])


# ----------------------------------------------------------------------
# 1️⃣  Warm‑start (Dirichlet draws, uniform, global starter)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in a sensible range
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Variance‑flattening gradient descent
# ----------------------------------------------------------------------
def _grad_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  L_corr: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent minimising the variance of the self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    mean = 1.0 / (2 * n - 1)          # total mass of the convolution is 1
    it = 0
    eta0 = 0.8                       # initial stepsize

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GradIter={it:,}  score={cur_score:.6f}")

        # gradient of variance
        grad = _variance_gradient(a, conv, mean, L_corr)

        # classic 1/√t schedule
        eta = eta0 / np.sqrt(it + 1)

        # gradient step and projection onto simplex
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # occasional tiny random jitter to escape plateaus
        if np.random.rand() < 0.004:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climb on the maximum of the convolution
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly attacks the current maximal convolution entry:
    identify the (i,j) pair that contributes most to this entry and move a
    small amount of probability mass from the larger of the two to a random
    location.  Accept only improving moves.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        i, j, contrib = _max_contributions(best_a, k_max)
        if i == -1:  # should not happen
            continue

        # move mass from the larger partner to a random destination
        src = i if best_a[i] >= best_a[j] else j
        if best_a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if dst == src:
            continue
        # delta is a modest fraction of the source weight
        delta = rng.uniform(0.0, min(best_a[src], 0.005))
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # re‑normalise to counteract tiny floating errors
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        new_score = _objective_score(cand, n, L_fft)
        if new_score < best_score:
            best_score, best_a = new_score, cand
            print(f"[{time.time() - start_time:.2f}s] Hill i={src}→{dst} δ={delta:.5f} → score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing + light random‑pair hill‑climb (as in the reference)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixing with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.004))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Light hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score
