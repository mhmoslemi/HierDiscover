def _max_subgradient_descent(a, n, L_fft, deadline, start_time):
    best_a = a.copy()
    conv = _conv_self_fft(a, L_fft)
    best_score = 2.0 * n * conv.max()
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute conv
        conv = _conv_self_fft(a, L_fft)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad iter={iteration} score={cur_score:.6f}")

        # compute gradient
        inds = max_idx - np.arange(n)
        mask = (inds >= 0) & (inds < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[inds[mask]]

        # step size
        eta = 0.75 / np.sqrt(iteration)  # maybe adjust

        # multiplicative update
        factor = np.exp(-eta * grad)
        # clip factor to avoid overflow/underflow
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        # optional jitter
        if np.random.rand() < 0.001:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_a, best_score

def _hill_climb_refinement(a, n, L_fft, deadline, start_time):
    best_a = a.copy()
    conv = _conv_self_fft(best_a, L_fft)
    best_score = 2.0 * n * conv.max()
    rng = np.random.default_rng()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j: continue
        max_delta = min(best_a[i], 0.01)  # 0.01 is some max move
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        conv = _conv_self_fft(cand, L_fft)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = cand
            print(f"[{time.time()-start_time:.2f}s] Hillclimb i={i} j={j} δ={delta:.6f} → score={new_score:.6f}")

        # maybe occasional mixing with uniform
        if iteration % 1000 == 0 and np.random.rand() < 0.05:
            lam = np.random.rand()
            uniform = np.full(n, 1.0/n)
            mixed = lam * best_a + (1-lam) * uniform
            mixed = mixed / mixed.sum()
            conv = _conv_self_fft(mixed, L_fft)
            sc = 2.0 * n * conv.max()
            if sc < best_score:
                best_score = sc
                best_a = mixed
                print(f"[{time.time()-start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")
    return best_a, best_score

assistantfinal**Solution Overview**

The score to minimise is  

\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum a)^2}\;\;,
\]

which is *scale invariant*.  
We therefore keep the vector on the probability simplex
\(\sum a_i = 1\) and minimise  

\[
F(a)=2n\;\max (a*a).
\]

`F` is the maximum of a set of quadratic functions and is non‑smooth.
The optimiser consists of three time‑aware phases.

1. **Warm start** – a large number of Dirichlet draws are evaluated
   (the evaluation is cheap thanks to an FFT based convolution).  
   The best draw becomes the initial point.

2. **Sub‑gradient mirror descent** – we minimise a *smooth* surrogate of
   `max(conv)` by using the exact sub‑gradient of the maximum
   (i.e. the gradient of the single active convolution entry).  
   The update is an exponentiated‑gradient step (mirror descent on the
   simplex) with a classic \( \eta_t = \eta_0/\sqrt t\) schedule.
   The current best ever vector is memorised.

3. **Final random‑pair hill‑climb** – starting from the best vector
   found in phase 2 we perform cheap random mass transfers
   `a_i ← a_i‑δ , a_j ← a_j+δ`.  
   After every transfer the exact score is recomputed with an FFT; a
   transfer is kept only if it strictly improves the score.
   Occasional mixing with the uniform distribution is also tried.

All phases are stopped as soon as the overall time budget (`budget_s`)
is exhausted, guaranteeing a valid output before the 1000 s limit.

The code below follows the described strategy, is completely self‑contained,
uses only `numpy` and the provided `evaluate_sequence` function,
and returns a plain Python list of non‑negative floats whose sum is (approximately) 1.

```python
#!/usr/bin/env python3
"""
Step‑function optimiser for the evaluation function

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three time‑aware stages:

1. Warm‑start: many Dirichlet draws are evaluated (FFT convolution).
2. Sub‑gradient mirror descent: exact gradient of the active
   convolution entry is used; exponentiated‑gradient updates keep the
   vector on the simplex.
3. Final hill‑climb: random pairwise mass transfers improve the
   exact score; occasional mixing with the uniform distribution is
   tried.

The routine respects a 1000 s budget and prints progress.
It returns a plain list of non‑negative floats (approximately normalised).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    The result is trimmed to the useful length.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(Af * Af, n=L_fft)
    return conv_full[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# 0️⃣  Utility to compute the exact score (sum(a) is 1 → simplification)
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv) (sum of a is 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Warm‑start with many Dirichlet draws
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random Dirichlet draws; keep the best one."""
    while time.time() < deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        conv = _conv_self_fft(a, L_fft)
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑start Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Sub‑gradient mirror descent (exact max entry)
# ----------------------------------------------------------------------
def _subgrad_descent(a: np.ndarray,
                     n: int,
                     L_fft: int,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of
    max(conv).  The update is multiplicative (exponentiated gradient) and
    the current best point is remembered.
    """
    best_a = a.copy()
    conv = _conv_self_fft(a, L_fft)
    best_score = _score_from_conv(conv, n)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        conv = _conv_self_fft(a, L_fft)
        max_idx = int(conv.argmax())
        cur_score = _score_from_conv(conv, n)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad iter={it:,} score={cur_score:.6f}")

        # ---- exact sub‑gradient of max(conv) ----
        # gradient_i = 2 * a_{max_idx - i}  (if index in range)
        idx = max_idx - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # ---- stepsize schedule (η_t = η0 / sqrt(t)) ----
        eta0 = 0.8
        eta = eta0 / np.sqrt(it)

        # ---- exponentiated‑gradient update (mirror descent) ----
        factor = np.exp(-eta * grad)
        factor = np.clip(factor, 1e-12, 1e12)          # protect from overflow
        a = a * factor
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # ---- occasional random jitter to avoid tiny plateaus ----
        if rng.random() < 0.002:
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final refinement – random pair hill‑climbing + uniform mix
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                n: int,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two positions are accepted only if they
    improve the exact score.  Occasionally a convex mixture with the
    uniform distribution is examined.
    """
    best_a = a.copy()
    conv = _conv_self_fft(best_a, L_fft)
    best_score = _score_from_conv(conv, n)

    it = 0
    uniform = np.full(n, 1.0 / n)

    while time.time() < deadline:
        it += 1

        # ----- random pair transfer -----
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        max_delta = min(best_a[i], 0.01)   # at most 0.01 of the total mass
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        conv = _conv_self_fft(cand, L_fft)
        sc = _score_from_conv(conv, n)

        if sc + 1e-12 < best_score:          # strict improvement
            best_score, best_a = sc, cand
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

        # ----- occasional mixing with uniform -----
        if it % 1000 == 0 and rng.random() < 0.05:
            lam = rng.random()
            mixed = lam * best_a + (1.0 - lam) * uniform
            mixed = mixed / mixed.sum()
            conv = _conv_self_fft(mixed, L_fft)
            sc = _score_from_conv(conv, n)
            if sc < best_score - 1e-12:
                best_score, best_a = sc, mixed
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    return best_a, best_score
```
