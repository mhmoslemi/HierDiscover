def _smooth_descent(a0, lam_initial, lam_final, lam_decay_iters, deadline, best_seq, best_score, rng):
    a = a0.copy()
    n = len(a)
    step_scale = 0.02
    lam = lam_initial
    it = 0
    while time.time() < deadline and lam > lam_final:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it} lam={lam:.2e} score={cur_score:.6f}")
        # compute smooth gradient
        grad = _smooth_max_grad(a, lam)
        grad_norm = np.linalg.norm(grad) + 1e-16
        direction = grad / grad_norm
        # try backtracking step
        step = step_scale
        improved = False
        for _ in range(5):
            cand = a - step * direction
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            new_score = _objective_score(cand)
            if new_score < cur_score:
                a = cand
                cur_score = new_score
                step_scale = min(step_scale * 1.02, 0.05)
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            step_scale = max(step_scale * 0.5, 1e-5)
            # jitter
            i1,i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-4 + 1e-8)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a,0.0))
        it += 1
        # schedule lam decay
        if it % lam_decay_iters == 0:
            lam *= 0.7
            if lam < lam_final:
                lam = lam_final
    return best_seq, best_score, a

"""
Multi‑restart smoothed gradient descent + sub‑gradient + hill‑climb.

The objective is   2·n·max(conv(a,a))   with   sum(a)=1 , a_i≥0 .
We work on the simplex (project after each update) and minimise a
smooth approximation of the max (log‑sum‑exp) with a decreasing
smoothing parameter λ.  Each run starts from a different random
Dirichlet seed (or from a hand‑crafted shape) so the search explores
many basins.  After the smoothed phase we fall back to the classic
sub‑gradient descent and finish with a tiny hill‑climb.  The whole
procedure respects the 1000 s wall‑clock budget and returns the best
sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution and objective helpers (the sum is always 1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Sub‑gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active indices)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    step_scale = 5e-3
    min_step = 1e-10
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        grad_norm = np.linalg.norm(grad) + 1e-16

        # back‑tracking line search
        step = step_scale
        improved = False
        for _ in range(6):
            direction = grad / grad_norm
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)
            new_score = _objective_score(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            step *= 0.5

        if not improved:
            step_scale = max(step_scale * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Hill‑climb (tiny mass moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Directed hill‑climb that moves a tiny fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        cur_score = _objective_score(a_prop)

        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max‑gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_max_grad_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        S_λ(a) = λ·log Σ_k exp(conv_k/λ)
    Returns ∂S_λ/∂a  (size n).
    """
    n = a.shape[0]
    max_c = conv.max()
    # stable soft‑max weights
    exps = np.exp((conv - max_c) / lam)
    sum_exps = exps.sum()
    if sum_exps == 0.0:
        sum_exps = 1.0

    a_rev = a[::-1]
    L = _next_pow2(len(a_rev) + len(exps) - 1)
    Af = rfft(a_rev, n=L)
    Bf = rfft(exps, n=L)
    full = irfft(Af * Bf, n=L)[: len(a_rev) + len(exps) - 1]
    # slice the part that corresponds to the gradient entries
    grad = 2.0 * full[n - 1:n - 1 + n] / sum_exps
    return grad


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    lam_start: float,
                    lam_end: float,
                    lam_decay: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Projected gradient descent on the smooth max (log‑sum‑exp).
    λ is exponentially reduced from lam_start to lam_end.
    """
    a = a0.copy()
    n = a.shape[0]
    step_scale = 0.02
    lam = lam_start
    it = 0

    while time.time() < deadline and lam > lam_end:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it} λ={lam:.2e} score={cur_score:.6f}")

        grad = _smooth_max_grad_from_conv(a, conv, lam)
        grad_norm = np.linalg.norm(grad) + 1e-16
        direction = grad / grad_norm

        # single back‑track step
        step = step_scale
        a_cand = a - step * direction
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = _project_simplex(a_cand)
        cand_score = _objective_score(a_cand)

        if cand_score < cur_score:
            a = a_cand
            cur_score = cand_score
            step_scale = min(step_scale * 1.02, 0.05)
        else:
            step_scale = max(step_scale * 0.5, 1e-5)
            # tiny random jitter to escape plateaus
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-4 + 1e-9)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

        it += 1
        if it % 2000 == 0:
            lam = max(lam * lam_decay, lam_end)

    return best_seq, best_score, a


# ----------------------------------------------------------------------
# Hand‑crafted families (used for the very first seeds)
# ----------------------------------------------------------------------
def _family_uniform(n):
    return np.full(n, 1.0 / n)


def _family_triangular(n):
    i = np.arange(n)
    a = (i + 1) * (n - i)
    return a / a.sum()


def _family_power(n, p):
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** p
    return a / a.sum()


def _family_exponential(n, c):
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - centre))
    return a / a.sum()


def _family_sine(n):
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n)
    return a / a.sum()


def _family_arcsine(n):
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return a / a.sum()
