def _adam_optimize(a0, cur_score, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t_iter = 0
    while time.time() < deadline:
        t_iter += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new best {best_score:.6f}")
        a = cand
        # gradually decay learning rate
        lr = max(lr * 0.999, 1e-4)
    return best_a, best_score

"""
Optimising the max‑auto‑convolution of a non‑negative step function.

The search consists of three loosely coupled stages, all respecting a
1000 s time budget.

1️⃣  **Broad random initialisation** – a large set of Dirichlet draws with
    widely varying concentration parameters.  Each draw is normalised so
    that its ℓ₁‑norm equals 1 (the objective is homogeneous, therefore any
    scaling is irrelevant).  The best sample becomes the starting point.

2️⃣  **Adam‑based accelerated sub‑gradient descent** – the objective
    `f(a)=2·n·max(conv(a,a))` (with sum = 1) is convex.  Its sub‑gradient
    is obtained by averaging the gradients of every active (maximal)
    convolution entry.  An Adam optimiser (first‑ and second‑moment
    estimates) supplies adaptive step sizes, while after each update the
    point is projected back onto the probability simplex (non‑negative,
    unit‑sum).  This phase is the heavy‑lifting phase.

3️⃣  **Refinement by back‑tracking projected gradient descent** – a
    standard line‑search that moves opposite to the same sub‑gradient and
    projects onto the simplex.  It finishes the optimisation with fine
    adjustments.

All heavy numerical work (full self‑convolution and the projection) uses
pure NumPy/FFT routines.  The code constantly checks the remaining time
and returns the best candidate found so far, guaranteeing a valid answer
before the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}.
    Implements the O(n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerically degenerate case
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration parameter c chosen log‑uniformly in [10⁻³, 10⁰·⁷]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s                     # ℓ₁‑normalisation
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam‑accelerated sub‑gradient descent
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, cur_score: float, deadline: float):
    """
    Adam optimiser on the convex objective `2*n*max(conv(a,a))`
    with projection onto the simplex after every step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                      # initial learning rate
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        # enforce feasibility (non‑negativity) before projection
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new best {best_score:.6f}")

        a = cand

        # modest learning‑rate decay to stabilise later iterations
        lr = max(lr * 0.9995, 1e-4)

        # occasional tiny random jitter to escape flat plateaux
        if it % 200 == 0:
            jitter = np.random.default_rng().normal(scale=1e-5, size=n)
            cand_jit = _project_simplex(np.maximum(a + jitter, 0.0))
            cand_jit_score = _objective_score(cand_jit)
            if cand_jit_score < best_score:
                best_score = cand_jit_score
                best_a = cand_jit.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam jitter improved to {best_score:.6f}")
            a = cand_jit

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final projected‑gradient refinement (back‑tracking line‑search)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Fine‑tune the vector by moving opposite to the sub‑gradient and re‑projecting."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5          # initial step length
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        # try a few decreasing step factors
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)   # shrink step if no progress
        else:
            step = min(step * 1.05, 1.0)    # modestly increase step when improving
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """
    # make the start time globally visible to the helper functions
    global start_time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑accelerated sub‑gradient descent – heavy lifting phase
    # ------------------------------------------------------------------
    adam_deadline = min(deadline, start_time + 720.0)   # about 12 min
    if time.time() < adam_deadline:

        best_seq, best_score = _adam_optimize(best_seq, best_score, adam_deadline)

    tot = best_seq.sum()
    if tot > 0.0:

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
