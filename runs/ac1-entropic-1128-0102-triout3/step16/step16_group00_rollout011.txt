"""
Advanced smooth‑max gradient descent on the probability simplex.

The objective (with sum(a)=1) is

    F(a) = 2·n·max_i (a∗a)[i]                      (1)

where “∗” is the (linear) convolution.
The function is homogeneous, therefore we may (and do) keep the
sequence on the simplex (a_i ≥ 0, Σa_i = 1).

A direct sub‑gradient of the max is very noisy.  We replace the max
by a smooth log‑sum‑exp approximation

    S_ε(a) = ε·log Σ_i exp( (a∗a)[i] / ε )          (2)

which satisfies  S_ε → max as ε → 0.
The gradient of (2) w.r.t. a is

    ∇_a S_ε = 2· ( w ⊛ a )                         (3)

where w_i = exp( (a∗a)[i] / ε ) / Σ_j exp( (a∗a)[j] / ε )
and “⊛” denotes the cross‑correlation (i.e. correlation of w with a).
Both the convolution and the correlation are evaluated with an FFT,
so each iteration costs O(n log n).

The variables are re‑parameterised with a soft‑max:
    a = softmax(z)  (z ∈ ℝⁿ, no constraints)
so that the simplex constraints are satisfied automatically.
Using the Jacobian of the soft‑max we obtain

    ∇_z F = a ∘ (∇_a F – ⟨∇_a F , a⟩)           (4)

where “∘” is the element‑wise product.

The optimisation is performed with the Adam optimiser
(β₁=0.9, β₂=0.999).  ε is decreased slowly (geometric decay) so
that the algorithm first follows a smooth landscape and then
focuses on the true max‑objective.  Random Dirichlet seeds are
generated for a brief warm‑up period; the best seed is then
refined with Adam for the remaining time.

The function returns the best sequence found (as a plain list of
floats).  All numbers are non‑negative and automatically normalised
to sum to 1, so ``evaluate_sequence`` can be called directly on the
result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using an FFT of length L.
    If L is None a suitable power‑of‑2 is chosen automatically.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _smooth_max_and_weights(conv: np.ndarray, eps: float):
    """
    Compute the smooth max (log‑sum‑exp) of the convolution
    together with the normalized soft‑max weights w_i.
    """
    max_val = conv.max()
    # shift by max to avoid overflow
    shifted = conv - max_val
    # exponent in the log‑sum‑exp
    exp_weights = np.exp(shifted / eps)
    sum_exp = exp_weights.sum()
    smooth_max = max_val + eps * np.log(sum_exp)
    w = exp_weights / sum_exp
    return smooth_max, w


def _grad_smooth_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max (3) with respect to a.
    Implemented via cross‑correlation of w with a using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)          # length for correlation
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)          # w has length 2n‑1
    # correlation (w ⊛ a) – real‑valued
    corr = np.fft.irfft(Wf * np.conj(Af), n=L)
    offset = w.shape[0] - 1           # index of zero lag
    grad = 2.0 * corr[offset : offset + n]
    return grad


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   time_deadline: float,
                   eps_start: float = 0.10,
                   eps_min: float = 1e-6,
                   eps_decay: float = 0.9995,
                   lr: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate.
    Returns the best (exact) sequence seen and its exact score.
    """
    n = start_a.shape[0]
    # initialise Adam state on the *z* (log‑softmax) parameters
    z = np.log(start_a + 1e-30)               # avoid log(0)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = start_a.copy()
    best_score = _objective_score(best_a)

    eps = eps_start
    it = 0
    while time.time() < time_deadline:
        it += 1
        # --------------- a = softmax(z) -----------------
        a = _softmax(z)                         # stays on simplex

        # --------------- convolution & smooth max -------
        # Compute self‑convolution once (FFT length chosen for corr later)
        conv = _conv_self(a)                    # length 2n‑1
        smooth_max, w = _smooth_max_and_weights(conv, eps)

        # exact objective (for bookkeeping)
        exact_score = 2.0 * n * conv.max()
        if exact_score < best_score:
            best_score = exact_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"New best (exact) = {best_score:.6f}")

        # --------------- gradient w.r.t. a -------------
        grad_a = _grad_smooth_wrt_a(a, w)       # shape (n,)

        # --------------- gradient w.r.t. z (softmax) ---
        inner = np.dot(grad_a, a)               # scalar ⟨grad_a , a⟩
        grad_z = a * (grad_a - inner)           # (4)

        # --------------- Adam update -------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------- epsilon schedule ---------------
        eps = max(eps * eps_decay, eps_min)

        # --------------- occasional random jitter -------
        # If we get stuck for a while, add a tiny perturbation.
        if it % 2000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-3, size=n)
            z += jitter

    return best_a, best_score


# ----------------------------------------------------------------------
# Random Dirichlet starter (quick exploration)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             max_trials: int,
                             deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration parameter sampled log‑uniformly
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score
