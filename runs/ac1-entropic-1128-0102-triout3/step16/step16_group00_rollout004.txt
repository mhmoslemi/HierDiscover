def _local_hill_climbing(a0, cur_score, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    while time.time() < deadline:
        # pick source and dest random
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer a random fraction up to 10% of src
        delta = a[src] * rng.random() * 0.1
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time()-start_time:.2f}s] hillclimb new best {best_score:.6f}")
        elif cand_score < cur_score and rng.random() < 0.1:
            # occasional move acceptance
            a = cand
            cur_score = cand_score
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
   np.random.seed(seed)
   global start_time
   start_time = time.time()
   deadline = start_time + budget_s - 0.1
   
   n = 4096
   uniform = np.full(n, 1.0/n)

   best_seq = uniform.copy()
   best_score = _objective_score(best_seq)
   print(f"[{time.time()-start_time:.2f}s] start uniform score {best_score}")

   # load global starter if exists
   try:
       starter = np.asarray(height_sequence_1, dtype=float)
       if starter.ndim == 1 and starter.size > 0:
           # pad or truncate to length n
           if starter.size != n:
               if starter.size < n:
                   starter = np.concatenate([starter, np.zeros(n-starter.size)])
               else:
                   starter = starter[:n]
           starter = np.maximum(starter, 0.0)
           s = starter.sum()
           if s > 0:
               starter = starter / s
               sc = _objective_score(starter)
               if sc < best_score:
                   best_seq = starter.copy()
                   best_score = sc
                   print(f"[{time.time()-start_time:.2f}s] loaded starter score {sc}")
   except Exception:
       pass

   # Phase 1: random Dirichlet
   dir_deadline = min(deadline, start_time + 30.0)
   seq, sc = _random_dirichlet_trials(n, max_trials=500, deadline=dir_deadline)
   if seq is not None and sc < best_score:
       best_seq = seq.copy()
       best_score = sc
       print(f"[{time.time()-start_time:.2f}s] after Dirichlet best {best_score}")

   # Phase 2: smooth descent multi restart
   beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0]
   num_restarts = 5
   # compute per-restart deadline portion
   remaining = deadline - time.time()
   # allocate most remaining to descent
   descent_deadline = time.time() + remaining * 0.85  # 85% for descent
   # For each restart, we use portion of remaining time.
   for r in range(num_restarts):
       if time.time() >= descent_deadline:
           break
       # initialize from best_seq with some perturbation
       a0 = best_seq.copy()
       # add small random jitter to avoid same start:
       rng = np.random.default_rng()
       jitter = rng.normal(scale=0.01, size=n)
       a0 = np.maximum(a0 + jitter, 0.0)
       a0 = _project_simplex(a0)
       # run descent
       sub_deadline = min(deadline, time.time() + (descent_deadline - time.time())/(num_restarts - r))
       a_opt, sc_opt = _smooth_descent(a0, beta_schedule, sub_deadline)
       if sc_opt < best_score:
           best_seq = a_opt.copy()
           best_score = sc_opt
           print(f"[{time.time()-start_time:.2f}s] restart {r} new best {best_score}")

   # Phase 3: local hill climbing for remaining time
   if time.time() < deadline:
       print(f"[{time.time()-start_time:.2f}s] starting hill climbing")
       best_seq, best_score = _local_hill_climbing(best_seq, best_score, deadline)

   # final normalise and truncate
   best_seq = np.maximum(best_seq, 0.0)
   tot = best_seq.sum()
   if tot > 0:
       best_seq = best_seq / tot
   else:
       best_seq = uniform.copy()
   print(f"[{time.time()-start_time:.2f}s] final best objective ≈ {best_score:.6f}")
   return [float(x) for x in best_seq]

"""
A multi‑phase optimizer for the self‑convolution objective

    2·n·max(conv(a,a)) / (sum(a)^2)

All vectors are normalised to sum = 1, so the denominator is fixed and the
objective reduces to   2·n·max(conv(a,a)).
The algorithm consists of

1️⃣  a quick sweep of random Dirichlet draws and a few deterministic shapes
    (sine, Hann, triangular) to obtain a good starting point;

2️⃣  a *smooth* projected‑gradient descent.  The non‑smooth max is replaced
    by a log‑sum‑exp (temperature β) which yields a smooth gradient.
    β is increased in stages (0.5 → … → 2000) so the optimisation first
    follows a gentle landscape and then focuses on the true maximum;

3️⃣  a short random hill‑climbing (mass‑transfer) phase for final fine‑tuning.

All heavy convolutions are performed with FFTs, the Euclidean projection onto
the probability simplex uses the O(n) algorithm of Duchi et al., and the whole
search respects the 1000 s time budget.  The best sequence found is returned
as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing variable used by the helper functions for printing
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using an FFT of length power‑of‑2."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Smooth approximation of max(conv) using log‑sum‑exp with temperature β."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max with respect to a.
    Uses the identity   grad = 2 * (a_rev * w)[offset]
    where w are the normalized exp‑weights.
    """
    # compute softmax weights over the convolution entries
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    # cross‑correlation a ⋆ w  (equivalently convolution of a_rev with w)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet initialisation
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth projected‑gradient descent (β‑schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    score0: float,
                    beta_vals,
                    deadline: float):
    """
    Gradient descent on the smooth approximation of max‑convolution.
    β is increased gradually; after each stage we project back onto the simplex.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    for beta in beta_vals:
        step = 0.5                     # initial step length for this β
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            improved = False
            # try a few geometric step reductions
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} new best {best_score:.6f}")
                    break
            # adapt step size
            if not improved:
                step = max(step * 0.5, 1e-12)
            else:
                step = min(step * 1.05, 1.0)

            # stop if step has collapsed
            if step < 1e-12:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Final random hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                score0: float,
                deadline: float):
    """Very cheap random mass‑transfer moves for the last few seconds."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # move a small random fraction (up to 5 % of the source mass)
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional uphill move (simple Metropolis with tiny temperature)
            a = cand
            cur_score = cand_score
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1           # safety margin

    # ------------------------------------------------------------------
    # Problem size – a good trade‑off between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet + a few deterministic shapes
    # ------------------------------------------------------------------
    dir_deadline = min(deadline, start_time + 30.0)
    if time.time() < dir_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=500, deadline=dir_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # deterministic initial shapes
    shapes = []
    # sine shape
    shapes.append(np.sin(np.pi * (np.arange(n) + 1) / (n + 1)))
    # Hann (raised‑cosine) window
    shapes.append(0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    # symmetric triangular window
    triangular = np.minimum(np.arange(n), n - 1 - np.arange(n))
    shapes.append(triangular.astype(float))

    for shp in shapes:
        shp = np.maximum(shp, 0.0)
        shp = _project_simplex(shp)
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth projected‑gradient descent with β‑schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate most of the remaining budget to this phase
        remaining = deadline - time.time()
        descent_deadline = time.time() + remaining * 0.85
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]

        # several restarts with a small random perturbation around the current best
        restarts = 8
        for r in range(restarts):
            if time.time() >= descent_deadline:
                break
            # add a tiny random jitter to escape possible flat regions
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            sub_deadline = min(descent_deadline,
                               time.time() + (descent_deadline - time.time()) / (restarts - r))
            a_opt, sc_opt = _smooth_descent(a0, a0_score, beta_schedule, sub_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – very short hill‑climbing tweak for the last seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting hill‑climbing")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
