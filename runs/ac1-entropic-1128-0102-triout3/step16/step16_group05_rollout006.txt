"""
Heuristic optimizer for the step‑function bound

    2·n·max(conv(a,a)) / (sum(a)**2) .

The algorithm has three stages

1. Initialisation – start from the global starter (if any), otherwise from a
   uniform or random Dirichlet vector.  The vector is projected onto the
   probability simplex (sum = 1, aᵢ ≥ 0).

2. Smoothed projected gradient descent – we replace the nonsmooth
   “max‑convolution’’ by a log‑sum‑exp (soft‑max) with a temperature β.
   The gradient of the smooth objective can be computed in O(n log n) by
   FFT‑based convolutions.  β is increased during the run so that the
   optimisation first moves easily on a smooth landscape and then fine‑tunes
   the true max.  After every step the iterate is projected back onto the
   simplex.

3. Final hill‑climb – a short greedy local search that moves a tiny amount
   of mass from the index with the largest sub‑gradient of the exact
   max‑convolution to the index with the smallest one.  This cleans up any
   remaining plateaus.

The whole routine stops before the given wall‑clock budget (default 1000 s)
and returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w | w_i ≥ 0, sum w_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum(a)=1) and sub‑gradient utilities
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective value for a normalised to sum=1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _smooth_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of 2·n·max(conv(a,a)) using
    log‑sum‑exp with temperature beta, together with its gradient.
    Returns (smooth_obj, grad) where grad has the same shape as a.
    """
    n = a.shape[0]

    # 1️⃣  self‑convolution
    conv = _conv_self(a)                 # length 2n‑1
    max_c = conv.max()

    # 2️⃣  soft‑max weights (numerically stable)
    shifted = conv - max_c               # ≤ 0
    exp_beta = np.exp(beta * shifted)    # in (0,1]
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp               # probability vector, sum=1

    # smooth max value
    smooth_max = max_c + (1.0 / beta) * np.log(sum_exp)
    smooth_obj = 2.0 * n * smooth_max

    # 3️⃣  gradient of smooth max:
    #    grad_smooth = 2 * sum_i w_i * a shifted by i
    a_rev = a[::-1]
    conv_wa = _conv_full_fft(w, a_rev)          # length 3n‑2
    # slice that corresponds to sum_i w_i * a_{i‑k}
    grad_smooth = 2.0 * conv_wa[n - 1 : n - 1 + n]   # length n
    grad = 2.0 * n * grad_smooth

    return smooth_obj, grad


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective.
    The temperature beta is increased during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    step = 0.1               # initial step magnitude (in Euclidean norm)
    min_step = 1e-7
    it = 0

    while time.time() < deadline:
        it += 1

        # a simple schedule for beta
        if it < 2000:
            beta = 1.0
        elif it < 8000:
            beta = 5.0
        elif it < 30000:
            beta = 20.0
        else:
            beta = 50.0

        # smooth objective and gradient at the current point
        smooth_obj, grad = _smooth_and_grad(a, beta)

        # exact objective (cheap – we already have the convolution)
        conv = _conv_self(a)
        exact_score = 2.0 * n * conv.max()
        if exact_score < best_score:
            best_score = exact_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] GD it={it:6d} exact={exact_score:.6f} beta={beta}")

        # --- line search on the smooth objective ---
        # we move opposite to the gradient
        grad_norm = np.linalg.norm(grad) + 1e-16
        direction = -grad / grad_norm          # unit descent direction
        cur_step = step

        improved = False
        for _ in range(8):
            a_candidate = a + cur_step * direction
            a_candidate = _project_simplex(a_candidate)
            cand_smooth, _ = _smooth_and_grad(a_candidate, beta)
            if cand_smooth < smooth_obj:
                a = a_candidate
                smooth_obj = cand_smooth
                # modestly increase step size on success
                step = min(step * 1.02, 0.5)
                improved = True
                break
            cur_step *= 0.5

        if not improved:
            # shrink step size and inject a tiny random jitter
            step = max(step * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a small fraction of mass from the
    index with the largest sub‑gradient (i.e. most “over‑represented’’)
    to the index with the smallest sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move up to 1 % of the source mass (a bit larger than before)
        eps = a[src] * (rng.random() * 0.015 + 1e-8)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time()-start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to get off the plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     n: int = 4096,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine uses a smoothed gradient descent followed by a short hill‑climb,
    honouring the supplied wall‑clock budget.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # If the caller provided a different n we honour it (must be >0).
    if n <= 0:
        n = 4096

    # ------------------------------------------------------------------
    # Initialise the iterate
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                init = starter.copy()
            else:
                init = None
        else:
            init = None
    except Exception:
        init = None

    if init is None:
        # fall back to a random Dirichlet (concentration 0.7 gives a mild
        # amount of sparsity) – the simplex projection will enforce sum=1.
        gamma = rng.gamma(shape=0.7, scale=1.0, size=n)
        init = gamma / gamma.sum()

    init = np.maximum(init, 0.0)
    init = _project_simplex(init)

    best_seq = init.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Initial exact score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smoothed gradient descent (majority of the budget)
    # ------------------------------------------------------------------
    gd_deadline = deadline - 120.0   # reserve ~2 min for hill‑climb
    if time.time() < gd_deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting smoothed gradient descent")
        best_seq, best_score = _gradient_descent(
            best_seq, gd_deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Stage 2 – final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Finished – best exact score ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
