assistantfinal**Solution Overview**

The goal is to minimise  

```
score(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)
```

for a non‑negative step‑function `a`.  
All sequences are re‑scaled to have `sum(a)=1` – therefore the
objective reduces to  

```
score(a) = 2 * n * max(conv(a,a))                (sum(a)=1)
```

The search consists of four independent phases that run sequentially
until the time budget expires.

1. **Rich parametric sweep** – a large collection of analytic families
   (geometric, power, (a)symmetric Beta, exponentially decaying,
   logistic, polynomial, cosine and two‑block families).  
   Each family is scanned on a fine grid, the best candidate of the
   whole sweep is retained.

2. **Massive random Dirichlet sampling** – draws with a very wide
   range of concentration parameters (including extremely spiky and
   almost‑uniform draws) and immediate evaluation.

3. **Population‑based evolutionary optimisation** – a small population
   (size ≈ 30) of the best candidates found so far is evolved with
   mutation, crossover and a simple tournament selection.  Every
   individual is projected onto the simplex (non‑negative, sum = 1)
   before evaluation.  The evolution runs as long as time permits and
   constantly updates the global best.

4. **Fine‑grained projected gradient descent** – starting from the
   current best sequence a short (≈ 50 iterations) sub‑gradient descent
   is performed to polish the solution.

All convolution operations use a fast FFT implementation, the
projection onto the simplex runs in linear time and every function is
written as a top‑level routine so that the time‑budget checks are cheap.

The final sequence is normalised to unit sum and returned as a plain
Python list – ready for `evaluate_sequence`.

```python
"""
A fast multi‑phase optimiser for the step‑function objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm performs:
1️⃣ a dense parametric sweep over many analytic families,
2️⃣ massive random Dirichlet draws,
3️⃣ a small population evolutionary search (mutation + crossover),
4️⃣ a short projected‑gradient polishing step.

All phases respect the global time limit (default 1000 s).  The best
vector (normalised to sum = 1) is returned as a plain list of floats.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# low level helpers
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """2 * n * max(conv(seq,seq))  – assumes sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# 1️⃣ parametric families
# ---------------------------------------------------------------

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for eps in np.logspace(-6, -1, 400):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        # blend with uniform – the extreme case (lam=0) is pure geometric
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * (np.full(n, 1.0 / n)) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-3.0, 3.0, 301):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for a in np.linspace(0.05, 8.0, 121):
        if time.time() >= deadline:
            break
        # symmetric Beta( a , a )
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (a - 1.0) * np.log(np.maximum(n - idx, 1))
        seq = np.exp(log_seq - np.max(log_seq))
        seq_sum = seq.sum()
        if seq_sum == 0 or not np.isfinite(seq_sum):
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 8.0, 33)
    for a in alphas:
        log_a = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential: a_i ∝ exp(-λ·min(i,n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-5, -0.2, 120):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_logistic(n, best_seq, best_score, deadline):
    """Logistic (sigmoid) shape centred at the middle of the array."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for scale in np.logspace(-3, -0.1, 80):
        if time.time() >= deadline:
            break
        seq = 1.0 / (1.0 + np.exp((idx - centre) / scale))
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Logistic scale={scale:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_polynomial(n, best_seq, best_score, deadline):
    """Polynomial (parabolic) family: a_i ∝ max(0, 1 - |i-centre|/centre)^p."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    rel = np.abs(idx - centre) / centre
    for p in np.linspace(0.5, 6.0, 71):
        if time.time() >= deadline:
            break
        seq = np.power(np.maximum(0.0, 1.0 - rel), p)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Poly p={p:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_cosine(n, best_seq, best_score, deadline):
    """Cosine bump: a_i ∝ max(0, cos(π·(i-centre)/width))."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for width_factor in np.linspace(0.1, 1.0, 45):
        if time.time() >= deadline:
            break
        width = width_factor * centre
        arg = np.pi * (idx - centre) / width
        seq = np.maximum(0.0, np.cos(arg))
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine w={width_factor:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the extremes."""
    for L_frac in np.linspace(0.01, 0.4, 40):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

# 2️⃣ random Dirichlet initialisers
# ---------------------------------------------------------------

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=1200):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration between 0.01 and 5 (log‑uniform)
        logc = rng.uniform(-2.0, 0.7)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ---------------------------------------------------------------
# 3️⃣ simple evolutionary optimiser
# ---------------------------------------------------------------

def _evolutionary_search(n, best_seq, best_score, deadline,
                         pop_size=30, child_per_gen=20,
                         mutate_sigma=0.03):
    """Population based optimisation (mutation + crossover)."""
    rng = np.random.default_rng()
    # initialise population with a few deterministic seeds + random Dirichlets
    population = []
    population.append(best_seq.copy())
    population.append(np.full(n, 1.0 / n))
    # a few hand‑crafted seeds
    population.append(np.exp(-np.arange(n, dtype=np.float64) / n))
    population.append(np.sqrt(np.arange(1, n + 1, dtype=np.float64)))
    # fill the rest with random Dirichlet draws
    while len(population) < pop_size:
        if time.time() >= deadline:
            break
        a = rng.gamma(shape=1.0, scale=1.0, size=n)
        a /= a.sum()
        population.append(a)

    # evaluate once
    scores = np.array([_objective_score(p) for p in population])
    # keep the best seen so far
    best_idx = np.argmin(scores)
    if scores[best_idx] < best_score:
        best_score, best_seq = scores[best_idx], population[best_idx].copy()
        print(f"[{time.time() - start_time:.2f}s] Initial pop best score={best_score:.6f}")

    # evolutionary loop
    while time.time() < deadline:
        # tournament selection – pick two parents from the top half
        # (simple rank‑based selection)
        order = np.argsort(scores)
        elite = order[:pop_size // 2]

        children = []
        for _ in range(child_per_gen):
            # crossover with probability 0.3
            if rng.random() < 0.3:
                p1, p2 = rng.choice(elite, size=2, replace=False)
                alpha = rng.random()
                child = alpha * population[p1] + (1.0 - alpha) * population[p2]
            else:
                # mutation of a single elite individual
                p = rng.choice(elite)
                child = population[p].copy()
                # add zero‑mean Gaussian noise
                child += mutate_sigma * rng.normal(size=n)
                child = np.maximum(child, 0.0)
            # project back onto the simplex
            child = _project_simplex(child)
            children.append(child)

        # evaluate children
        child_scores = np.array([_objective_score(c) for c in children])
        # insert children into the population, removing the worst individuals
        for c, sc in zip(children, child_scores):
            worst_idx = np.argmax(scores)
            if sc < scores[worst_idx]:
                population[worst_idx] = c
                scores[worst_idx] = sc

        # update global best
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_score:
            best_score = float(scores[best_idx])
            best_seq = population[best_idx].copy()
            print(f"[{time.time() - start_time:.2f}s] Evo new best score={best_score:.6f}")

        # shrink mutation strength slowly to fine‑tune
        mutate_sigma *= 0.9995
        # occasional reset of bad individuals to keep diversity
        if rng.random() < 0.02:
            bad = np.argmax(scores)
            a = rng.gamma(shape=1.0, scale=1.0, size=n)
            a /= a.sum()
            population[bad] = a
            scores[bad] = _objective_score(a)

    return best_seq, best_score

# 4️⃣ short projected gradient polishing
# ---------------------------------------------------------------

def _projected_gradient_descent(a0, deadline, best_seq, best_score, max_iter=80):
    """Sub‑gradient descent on the simplex (few iterations, heavy line‑search)."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        # descent direction – keep sum zero
        d = -(grad - grad.mean())
        # maximal feasible step size
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        step = max_step * 0.8

        # back‑tracking line‑search (simple)
        while step > 1e-14:
            a_new = a + step * d
            if np.any(a_new < 0):
                step *= 0.5
                continue
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score <= cur_score:
                a = a_new
                break
            step *= 0.5
        # if no progress, add a tiny jitter
        if step <= 1e-14:
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine consists of a parametric sweep,
    massive Dirichlet sampling, a lightweight evolutionary search and a
    short projected‑gradient polishing phase.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # -----------------------------------------------------------
    # problem size
    # -----------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # -----------------------------------------------------------
    # 1️⃣ broad parametric sweep (≈ 250 s)
    # -----------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_logistic(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_polynomial(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_cosine(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # -----------------------------------------------------------
    # 2️⃣ massive Dirichlet draws (≈ 150 s)
    # -----------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=2000
        )

    # -----------------------------------------------------------
    # 3️⃣ evolutionary search (≈ 300 s)
    # -----------------------------------------------------------
    evo_deadline = deadline - 100.0
    if time.time() < evo_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting evolutionary search")
        best_seq, best_score = _evolutionary_search(
            n, best_seq, best_score, evo_deadline,
            pop_size=30,
            child_per_gen=25,
            mutate_sigma=0.02,
        )
        print(f"[{time.time() - start_time:.2f}s] After EA best score={best_score:.6f}")

    # -----------------------------------------------------------
    # 4️⃣ short projected‑gradient polishing (≈  70 s)
    # -----------------------------------------------------------
    if time.time() < deadline:
        pgd_deadline = deadline - 10.0
        print(f"[{time.time() - start_time:.2f}s] Starting final projected‑gradient polish")
        best_seq, best_score = _projected_gradient_descent(
            best_seq, pgd_deadline, best_seq, best_score, max_iter=80
        )
        print(f"[{time.time() - start_time:.2f}s] Final polished score={best_score:.6f}")

    # -----------------------------------------------------------
    # final normalisation and return
    # -----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best internal score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
```
