def _greedy_mass_transfer(a, deadline, best_seq, best_score):
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute subgradient
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # random perturbation
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue
        # Transfer fraction; start large fraction maybe 0.5, reduce if not accepted.
        transferred = False
        for frac in [0.5, 0.25, 0.1, 0.05, 0.01]:
            eps = a[src] * frac
            if eps <= 1e-15:
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                a = a_new
                transferred = True
                print(f"[{time.time() - start_time:.2f}s] Greedy it={it} src={src} dst={dst} frac={frac:.3f} score={new_score:.6f}")
                break
        if not transferred:
            # random jitter to escape
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * 0.1
            a[i1] = max(0.0, a[i1] - eps)
            a[i2] = min(1.0, a[i2] + eps)
            # renorm
            a = np.maximum(a, 0)
            total = a.sum()
            if total > 0:
                a /= total
        # maybe early stop if time low.
    return best_seq, best_score, a

def _smooth_gradient_descent(a, deadline, best_seq, best_score):
    n = a.shape[0]
    p_values = [4, 8, 16, 32, 64, 128, 256]
    while time.time() < deadline:
        for p in p_values:
            # Determine inner iteration count based on time left.
            it = 0
            while time.time() < deadline:
                it += 1
                grad = _gradient_softmax(a, p)
                # zero-mean direction
                d = -(grad - grad.mean())
                # compute feasible step
                idx_neg = np.where(d < 0)[0]
                if idx_neg.size > 0:
                    max_step = np.min(-a[idx_neg] / d[idx_neg])
                else:
                    max_step = 1.0
                step = max_step * 0.7  # aggressive
                # line search
                cur_score = _objective_score(a)
                while step > 1e-14:
                    a_new = a + step * d
                    if np.any(a_new < 0):
                        step *= 0.5
                        continue
                    new_score = _objective_score(a_new)
                    if new_score <= cur_score:
                        a = a_new
                        break
                    step *= 0.5
                if step <= 1e-14:
                    # jitter
                    a = a + np.random.uniform(-1e-7, 1e-7, size=n)
                    a = np.clip(a, 0, None)
                    a /= a.sum()
                # record best
                cur_score = _objective_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SoftGrad p={p} it={it:,} score={cur_score:.6f}")

"""
propose_candidate
=================

A multi‑stage optimiser for the “max‑convolution” bound.
The routine works on a vector `a` (non‑negative, sum = 1) of length *n*
and minimises

    2·n·max(conv(a,a))  .

The pipeline:

1. **Quick parametric sweep** – geometric and skewed‑exponential families
   give a decent starting shape.
2. **Greedy mass‑transfer** – iteratively moves probability mass from the
   indices that most contribute to the current maximal convolution value to
   the least‑contributing ones (a deterministic hill‑climb).
3. **Continuation gradient descent** – a smoothed objective
   `log‑sum‑exp` of the convolution is minimised with increasing
   smoothing parameter `p`.  The gradient is obtained by a weighted
   cross‑correlation using FFTs; each step is projected onto the simplex.
4. **Final simulated‑annealing** – a stochastic hill‑climb with a
   temperature schedule to polish the solution.

All phases respect the global 1000 s budget.  The best vector found is
normalised to sum = 1 and returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _softmax_gradient(a: np.ndarray, p: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        (1/p)·log Σ exp(p·conv_i)
    where conv = a*a.  Implements the weighted cross‑correlation via FFT.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    m = conv.max()
    # weights w_i = exp(p·(conv_i−m)) ∈ (0,1]
    w = np.exp(p * (conv - m))

    # linear cross‑correlation of a (length n) with w (length 2n‑1)
    L = _next_pow2(n + (2 * n - 1) - 1)  # ≥ 3n‑2
    Af = np.fft.fft(a, n=L)
    Wf = np.fft.fft(w, n=L)
    corr = np.fft.ifft(Af * np.conj(Wf)).real
    # first n entries correspond to indices 0…n‑1
    corr = corr[:n]
    grad = 2.0 * corr
    return grad


# ----------------------------------------------------------------------
# Simple parametric families (coarse sweep)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    """Geometric decay a_i ∝ r^i mixed with uniform."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 80)          # fewer points → faster
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} → {sc:.6f}")
    return best_seq, best_score


def _sweep_skewed_exponential(n, best_seq, best_score, deadline):
    """Piecewise exponential: left and right sides have independent rates."""
    mid = n // 2
    left_idx = np.arange(mid, dtype=np.float64)
    right_idx = np.arange(mid, n, dtype=np.float64)
    left_grid = np.logspace(-4, -0.2, 15)
    right_grid = np.logspace(-4, -0.2, 15)

    for lam_l in left_grid:
        for lam_r in right_grid:
            if time.time() >= deadline:
                break
            seq = np.empty(n, dtype=np.float64)
            seq[:mid] = np.exp(-lam_l * left_idx)
            seq[mid:] = np.exp(-lam_r * (n - 1 - right_idx))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SkewExp λL={lam_l:.1e} λR={lam_r:.1e} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _greedy_mass_transfer(a0, deadline, best_seq, best_score):
    """Deterministic hill‑climb: move mass from high‑gradient to low‑gradient indices."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy it={it} new best {cur_score:.6f}")

        # sub‑gradient of the max‑convolution (higher → source, lower → dest)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        improved = False
        for frac in (0.5, 0.25, 0.1, 0.05, 0.01):
            eps = a[src] * frac
            if eps <= 1e-15:
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
            new_score = _objective_score(a_new)
            if new_score < cur_score:
                a = a_new
                improved = True
                print(f"[{time.time() - start_time:.2f}s] Greedy it={it} src={src}→dst={dst} eps={eps:.3e} score={new_score:.6f}")
                break
        if not improved:
            # small random jitter to avoid stagnation
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = a[i1] * 0.02
            a[i1] = max(0.0, a[i1] - jitter)
            a[i2] = a[i2] + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score, a


# ----------------------------------------------------------------------
# Continuation gradient descent on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0, deadline, best_seq, best_score):
    """Iteratively minimise a log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    # increasing smoothing parameter
    p_vals = [4, 8, 16, 32, 64, 128, 256]
    for p in p_vals:
        # give each p a few inner iterations (time‑controlled)
        it = 0
        while time.time() < deadline:
            it += 1
            grad = _softmax_gradient(a, float(p))
            # keep direction in the tangent space of the simplex
            d = -(grad - grad.mean())

            # largest feasible step respecting non‑negativity
            idx_neg = np.where(d < 0)[0]
            if idx_neg.size:
                max_step = np.min(-a[idx_neg] / d[idx_neg])
            else:
                max_step = 1.0
            step = max_step * 0.7

            cur_score = _objective_score(a)
            # back‑tracking line search
            while step > 1e-14:
                a_new = a + step * d
                if np.any(a_new < 0):
                    step *= 0.5
                    continue
                new_score = _objective_score(a_new)
                if new_score <= cur_score:
                    a = a_new
                    break
                step *= 0.5

            # if step vanished, add a microscopic random perturbation
            if step <= 1e-14:
                a = a + np.random.uniform(-1e-7, 1e-7, size=n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            cur_score = _objective_score(a)
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SoftGrad p={p} it={it} → {cur_score:.6f}")
                if best_score <= 1.503:
                    return best_seq, best_score, a

            # limit inner iterations per p to keep within the global budget
            if it >= 60:
                break
            if time.time() >= deadline:
                break
    return best_seq, best_score, a

# Simulated‑annealing hill‑climb (as in the previous baseline)
# ----------------------------------------------------------------------
def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score
    cur_a = a.copy()

    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        T = (1.0 - frac) * cur_score * 0.001   # temperature ≈ 0.1 % of the score

        if it % 100 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))
            pos = grad.max()
            neg = grad.min()

        prob = np.maximum(grad, 0)
        prob_sum = prob.sum()
        if prob_sum > 0:
            src = rng.choice(n, p=prob / prob_sum)
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst:
            continue

        max_frac = 0.25
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop)
        score_new = float(2.0 * n * conv_new.max())
        delta = score_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * (rng.random() * max_frac)

            a = np.clip(a, 0.0, None)
