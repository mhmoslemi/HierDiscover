    a=a0.copy()
    best_a=a.copy()

    beta = 5.0
    # schedule: while time left >something and beta <500:
    while time.time()<deadline and beta<500:
        # run a few steps at this beta
        for step in range(20):
            if time.time()>=deadline: break
            conv = _conv_self(a) # length 2n-1
            # compute softmax weights
            maxc = conv.max()
            w = np.exp(beta*(conv - maxc))
            w_sum = w.sum()
            w = w / w_sum
            # gradient using correlation
            w_rev = w[::-1]
            L = _next_pow2(n + len(w) - 1)
            A = np.fft.rfft(a, L)
            W = np.fft.rfft(w_rev, L)
            corr = np.fft.irfft(A * W, n=L)[: n + len(w) - 1]
            grad = 2.0 * corr[len(w)-1 - np.arange(n)]  # length n
            # Determine step size: use simple backtracking to reduce objective (actual max conv)
            step_size = 1.0
            cur_score = _objective_score(a)

"""
Hybrid non‑negative step‑function optimizer.

The routine builds a set of seed vectors (uniform, two‑block, random
Dirichlet draws, optional external starter) and refines each of them
with two complementary stages:

1. **Smooth‑max gradient descent** – we minimise a smooth
   approximation of the L∞ objective
        2·n·max(conv(a,a))
   by replacing the max with a soft‑max (log‑sum‑exp).  The gradient
   of this surrogate is the weighted autocorrelation
        ∇J(a) = 2·(a ⋆ w)          (w are the soft‑max weights)
   which is computed efficiently by FFTs.  After each step the
   iterate is projected onto the probability simplex (non‑negative,
   sum = 1).  A simple back‑tracking line‑search guarantees monotone
   improvement of the true objective.

2. **Aggressive simulated‑annealing** – starting from the best
   vector obtained by the first stage we perform a long hill‑climb.
   The temperature starts at ≈ 5 % of the current score and cools
   linearly to zero, allowing frequent uphill moves and large mass
   transfers (up to 40 % of a source entry).  If no improvement is
   seen for many iterations a random reshuffle is performed.

Both stages are time‑controlled; the whole search stops before the
budget (default 1000 s) expires and returns the best sequence found.
All numbers are guaranteed to be non‑negative and sum to 1, so the
output can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    A small safety check guards against a zero‑sum vector.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx = 1}.  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣ Smooth‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth surrogate of the L∞ objective.
    The surrogate is 2·n·log‑sum‑exp(beta·conv)/beta.
    As beta grows the surrogate approaches the true objective.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    # initialise best from caller
    best_local_seq = a.copy()
    best_local_score = cur_score

    beta = 5.0                 # start with a modest smoothing
    while time.time() < deadline and beta < 500.0:
        # a few inner gradient steps at the current beta
        for _ in range(12):
            if time.time() >= deadline:
                break
            conv = _conv_self(a)                    # length 2·n‑1
            # soft‑max weights
            maxc = conv.max()
            w = np.exp(beta * (conv - maxc))
            w /= w.sum()

            # gradient = 2·(a ⋆ w)  (⋆ = correlation)
            w_rev = w[::-1]
            L = _next_pow2(n + w_rev.size - 1)
            A = np.fft.rfft(a, L)
            W = np.fft.rfft(w_rev, L)
            corr = np.fft.irfft(A * W, n=L)[: n + w_rev.size - 1]
            grad = 2.0 * corr[w_rev.size - 1 - np.arange(n)]

            # descent direction (negative gradient)
            d = -grad

            # compute a feasible step size (keep a ≥ 0)
            idx_neg = np.where(d < 0)[0]
            if idx_neg.size:
                max_step = np.min(-a[idx_neg] / d[idx_neg])
                step = max_step * 0.9
            else:
                step = 1.0

            # back‑tracking line search on the **real** objective
            cur_score = _objective_score(a)
            while step > 1e-12:
                a_new = a + step * d
                a_new = _project_simplex(a_new)          # enforce feasibility
                new_score = _objective_score(a_new)
                if new_score <= cur_score:
                    a = a_new
                    cur_score = new_score
                    if cur_score < best_local_score:
                        best_local_score = cur_score
                        best_local_seq = a.copy()
                        print(f"[{time.time()-start_time:.2f}s] "
                              f"beta={beta:.1f} step={step:.2e} "
                              f"score={cur_score:.6f}")
                    break
                step *= 0.5

            # if back‑track failed, add a tiny jitter to escape flat region
            if step <= 1e-12:
                eps = 1e-7
                a = a + np.random.uniform(-eps, eps, size=n)
                a = _project_simplex(a)

        # increase smoothing – the surrogate becomes a better approximation
        beta *= 1.5

    # compare with the external best
    if best_local_score < best_score:
        best_score = best_local_score
        best_seq = best_local_seq.copy()

    return best_seq, best_score

def _improved_simulated_annealing(a0: np.ndarray,
                                  deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climb with a relatively high initial temperature (≈5 % of the
    current score) and a linear cooling schedule.  Mass is transferred
    between randomly chosen indices; the fraction moved may be as large
    as 40 %.  If the algorithm stalls for many iterations a random
    reshuffle is performed.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    cur_best_seq = a.copy()
    cur_best_score = cur_score

    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        # temperature linearly decays from ≈5 % of the start score to 0
        T = (1.0 - frac) * cur_best_score * 0.05

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue

        max_frac = 0.4               # large possible jumps

        a_prop[dst] += eps

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop

            if cur_score < cur_best_score:
