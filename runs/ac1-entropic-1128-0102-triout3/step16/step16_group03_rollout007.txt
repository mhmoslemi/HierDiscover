def proj_simplex(v):
    # returns projection onto simplex sum=1, v_i>=0
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v) - 1
    ind = np.arange(len(v)) + 1
    cond = sorted_v - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimizer (gradient‑descent + simulated annealing) for the
step‑function evaluation problem.

The objective (after normalising the sum to 1) is   f(a)=2·n·max(conv(a,a)),
where conv(a,a) is the full linear self‑convolution.  We minimise f over the
probability simplex.  The algorithm consists of four stages:

1️⃣  Fast Dirichlet/Beta sampling – a brief broad random search that quickly
    finds a decent starting point.

2️⃣  Projected Adam (additive) descent on the simplex.  The gradient of the
    current maximum convolution entry is obtained analytically; Adam’s
    adaptive steps are followed by a projection onto the simplex.

3️⃣  Simulated‑annealing local refinement – random mass‑transfer moves are
    accepted with a temperature‑scaled Metropolis criterion, allowing the
    search to escape shallow plateaux.

4️⃣  Fine mixing with the uniform distribution – a cheap golden‑section search
    over λ in a·λ+(1‑λ)·uniform, keeping the best mixture.

All phases respect the global time‑budget (default 1000 s).  The final
sequence is normalised to sum 1, clipped to non‑negative numbers and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum (within a tiny tolerance) the
    gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat everything within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # each term a_i * a_{k-i} contributes a_{k-i} to ∂/∂a_i
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because each product appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.allclose(v, 0):
        # zero vector → uniform
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically unstable case – fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray | None, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

# 1️⃣  Broad random search (Dirichlet + a few Beta seeds)
# ----------------------------------------------------------------------
def _broad_random_search(n: int,
                         best_seq: np.ndarray,
                         best_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # a handful of deterministic Beta seeds (eval’d once)
    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                   (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3)]
    beta_seeds = []
    for a, b in beta_params:
        v = _beta_shape(a, b, n)
        if v is not None:
            beta_seeds.append(v)

    while time.time() < deadline:
        # Dirichlet draw (gamma trick)
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        samp = rng.gamma(shape=c, scale=1.0, size=n)
        s = samp.sum()
        if s == 0:
            continue
        a = samp / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # occasional evaluation of a random Beta seed (cheap)
        if rng.random() < 0.005 and beta_seeds:
            v = beta_seeds[rng.integers(len(beta_seeds))]
            sc = _objective_score(v, n)
            if sc < best_score:
                best_seq, best_score = v.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Projected Adam descent on the simplex
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Additive Adam optimiser with projection onto the simplex.
    The update direction is the negative sub‑gradient of max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2                       # base learning rate
    it = 0
    last_improve = 0
    # stall detection after many iterations → inject small random noise
    restart_interval = 250_000

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the current max‑convolution (direction of increase)
        grad = _subgradient_maxconv(a, conv)
        # we step opposite to reduce the objective
        grad = -grad

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a_new = a + step * m_hat

        # projection back onto the simplex
        a = _proj_simplex(a_new)

        # occasional stochastic kick to escape flat plateaux
        if it - last_improve > restart_interval:
            noise = np.random.default_rng().normal(scale=0.01, size=n) * a
            a = _proj_simplex(a + noise)
            last_improve = it

    return best_seq, best_score

# 3️⃣  Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves accepted with a temperature‑scaled
    Metropolis rule.  The temperature decays linearly with the remaining
    time budget.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self_fft(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # pick donor and receiver
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 0.0:
            continue

        # transfer a small fraction of donor's mass
        max_transfer = min(a[i], 0.04)          # up to 4 % of donor
        delta = rng.random() * max_transfer
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        conv_new = _conv_self_fft(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved to {new_score:.6f}")
        else:
            # temperature that linearly decays to near‑zero
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-3 * (remaining / total)
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few golden‑section
    evaluations.  The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score

    phase2_deadline = start_time + 0.70 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected Adam descent")
        best_seq, best_score = _adam_descent(
            best_seq, n, L_fft, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing refinement (≈20 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:            # keep a few seconds for final mixing
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, n, L_fft, start_time, deadline - 5.0, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After SA best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fine mixing with uniform (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Convert to plain Python list
    return [float(v) for v in best_seq.tolist()]
