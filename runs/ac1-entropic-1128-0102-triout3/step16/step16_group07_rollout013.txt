def _exp_gradient_descent(a0, start_time, deadline, best_seq, best_score):

        grad=_subgradient_maxconv(a,conv)

        # exponentiated update, ensure positivity

"""Propose a step-function sequence minimizing the max self‑convolution.
The algorithm combines a broad sweep of analytic windows, massive
Dirichlet sampling, multi‑start projected‑gradient descent and an
aggressive simulated‑annealing hill‑climb.  All steps respect the
global 1000 s budget.
"""

import time
import math
import numpy as np

# ---------- low‑level utilities ----------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all active indices)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) (sum(seq) assumed = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# ---------- projected gradient descent ----------
def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.5
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad
        eta_t = eta / (1.0 + it * 1e-4)
        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-14:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.995
            if no_improve % 400 == 0:
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-9:
            break

    return best_seq, best_score

# ---------- enhanced hill‑climb / simulated annealing ----------
def _enhanced_hill_climb(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score

    # temperature schedule
    T0 = 0.08
    decay = (1e-7 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        # compute gradient for guiding the source selection
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # pick a source index with large positive gradient (mass contributes to the peak)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]

        # pick a destination index with small (or negative) gradient
        dst = np.argmin(grad)

        if src == dst:
            # fallback random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        # transfer a modest fraction of src's mass
        eps_frac = rng.random() * 0.07 + 1e-5  # up to ~7 %
        eps = a[src] * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = math.exp(-Δ / max(T, 1e-14))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Enhanced hill best={best_score:.6f}")

    return best_seq, best_score

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used internally: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

) -> tuple[np.ndarray, float]:
    """Classic projected gradient descent using the exact sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.5
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad
        eta_t = eta / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-14:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.995
            if no_improve % 400 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-9:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Enhanced hill‑climb / simulated‑annealing refinement
# ----------------------------------------------------------------------
def _enhanced_hill_climb(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Metropolis moves that shift a moderate fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score

    # temperature schedule
    T0 = 0.08
    decay = (1e-7 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        eps_frac = rng.random() * 0.07 + 1e-5  # up to ~7 % of src mass
        eps = a[src] * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = math.exp(-Δ / max(T, 1e-14))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Enhanced hill best={best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The pipeline consists of analytic families, Dirichlet sampling,
    multi‑start PGD, and a final aggressive simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # safety margin

    # problem size – 4096 gives a good trade‑off between expressivity and speed
    n = 4096

    # -----------------------------------------------------------------
    # Uniform baseline
    # -----------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] External starter score={sc:.6f}"
                    )
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1️⃣ Analytic families (including windows)
    # -----------------------------------------------------------------
    phase_deadline = deadline - 300.0  # reserve ≈300 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families_extended(
            n, best_seq, best_score, phase_deadline
        )
        print(
            f"[{time.time() - start_time:.2f}s] After families best score={best_score:.6f}"
        )

    # -----------------------------------------------------------------
    # 2️⃣ Random Dirichlet sampling
    # -----------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials_extended(
            n, best_seq, best_score, phase_deadline
        )
        print(
            f"[{time.time() - start_time:.2f}s] After Dirichlet best score={best_score:.6f}"
        )

    # -----------------------------------------------------------------
    # 3️⃣ Multi‑start projected gradient descent
    # -----------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time() - start_time:.2f}s] Starting multi‑start PGD")
        while time.time() < deadline - 30.0:
            # fresh random start (symmetric Dirichlet)
            rng = np.random.default_rng()
            log_c = rng.uniform(np.log(0.02), np.log(10.0))
            c = np.exp(log_c)
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
            sub_deadline = min(time.time() + 5.0, deadline - 30.0)
            best_seq, best_score = _pgd_descent(
                a0, start_time, sub_deadline, best_seq, best_score
            )
        print(
            f"[{time.time() - start_time:.2f}s] Multi‑start PGD finished best≈{best_score:.6f}"
        )

    # -----------------------------------------------------------------
    # 4️⃣ Final aggressive hill‑climb / simulated annealing
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final enhanced hill‑climb")
        best_seq, best_score = _enhanced_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(
            f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}"
        )

    # -----------------------------------------------------------------
    # Normalise and return
    # -----------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
