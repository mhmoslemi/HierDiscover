def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3

    n = 8192
    L_fft = _next_pow2(2*n - 1)

    def conv_self(a):
        Af = np.fft.rfft(a, n=L_fft)
        return np.fft.irfft(Af*Af, n=L_fft)[:2*n-1]

    def obj(a):
        # assume sum > 0
        return 2*n * conv_self(a).max()

    # initial uniform
    best_seq = np.full(n, 1.0/n)

    # Make list of top candidates
    top_seqs = []

    # Stage: random sampling
    while time.time() < deadline - 200:  # preserve 200s for optimization
        c = np.exp(np.random.uniform(np.log(0.01), np.log(20)))  # concentration

        a /= a_sum
        sc = obj(a)
        if sc < best_score:

        # keep top few
        # store if within some margin
        if len(top_seqs) < 5:
            top_seqs.append((sc, a.copy()))
        else:
            # keep best 5
            max_sc = max(top_seqs, key=lambda t: t[0])
            if sc < max_sc[0]:
                top_seqs.remove(max_sc)
                top_seqs.append((sc, a.copy()))

    # Stage: short Adam from each top candidate
    for idx, (sc0, a0) in enumerate(top_seqs):
        # allocate time piece: maybe 30 secs each
        sub_deadline = min(deadline, time.time() + 30)

        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] Adam restart {idx} improved to {best_score:.6f}")

    # Stage: longer Adam from best
    if time.time() < deadline - 80:

"""
Propose a step‑function (probability vector) with a low value of
evaluate_sequence(seq) = 2·n·max(conv(a,a))/(sum a)^2.

The algorithm runs within the 1000 s budget and consists of:

1️⃣  A fast random search (Dirichlet / exponential draws) to obtain
    several promising starting points.
2️⃣  Short Adam runs on a smooth‑max surrogate for each start.
3️⃣  A longer projected sub‑gradient descent on the exact max‑convolution
    (the most effective local optimiser for this non‑convex problem).
4️⃣  A simulated‑annealing hill‑climb that makes larger mass transfers.
5️⃣  Normalisation onto the probability simplex before returning.

The problem size is increased to n = 8192, which gives the optimiser
more freedom (the objective is scale‑invariant).  All stages respect the
global deadline; the routine prints intermediate scores so the progress
can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L_fft."""
    Af = np.fft.rfft(a, n=L_fft)
    # element‑wise product gives the convolution in the frequency domain
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a (exact, not smooth).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to decide which entries are active
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    # gradient initially zero
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i runs over valid indices contributing to shift k
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a[k - idx] is the partner from the other side
        grad[idx] += a[k - idx]
    if active.size:
        # factor 2 because each product appears twice except the centre term
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, L_fft: int) -> float:
    """
    Score used internally (identical to evaluate_sequence when sum(a)=1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   L_fft: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The smooth‑max uses a temperature τ that is gradually increased.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0                # start temperature
    tau_factor = 1.2
    it = 0

    # pre‑allocate FFT size for the extra convolution inside the gradient
    L_extra = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        # exact convolution (only needed for logging)
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ---- smooth‑max (log‑sum‑exp) over convolution entries ----
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()                     # weights sum to 1

        # gradient of smooth‑max: (a ⋆ w) shifted and doubled
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=L_extra) *
                              np.fft.rfft(a_rev, n=L_extra),
                              n=L_extra)[:n]
        grad = 2.0 * conv_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # slowly increase the temperature so that the surrogate
        # approaches the true max
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float,
                 L_fft: int) -> tuple[np.ndarray, float]:
    """
    Projected (onto simplex) sub‑gradient descent on the exact
    max‑convolution objective.  The step size decays slowly; occasional
    random jitter helps to escape flat plateaux.
    """
    a = a0.copy()
    n = a.shape[0]
    eta = 0.5                # initial step size
    it = 0
    no_improve = 0
    cur_score = _objective_score(a, L_fft)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        # decaying step size (very mild decay)
        eta_t = eta / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate, L_fft)

        if new_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.995           # shrink step when stuck
            # random jitter every few hundred failures
            if no_improve % 500 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a, L_fft)

        if eta < 1e-8:
            break

    return best_seq, best_score


def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      L_fft: int) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing / hill‑climb that transfers a tiny amount
    of mass from a high‑gradient location to a low‑gradient one.
    The temperature decays to almost zero by the end.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # Occasionally pick random pair to avoid dead ends
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # Transfer a small fraction of the source mass
        frac = rng.random() * 0.01 + 1e-5      # up to ~1 % of src
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the caller's housekeeping
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size – larger n gives the optimiser more freedom.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Uniform baseline (always a safe start)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a previous global solution is available, try it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, L_fft)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet / exponential sampling (quick exploration)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    top_candidates = []                       # store up to 5 promising starts
    while time.time() < deadline - 200.0:     # reserve ~200 s for the heavy phase
        # draw concentration parameter on a log‑scale (0.01 … 20)
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

        # maintain a small pool of the best candidates for restarts
        if len(top_candidates) < 5:
            top_candidates.append((sc, a.copy()))
        else:
            # replace the worst in the pool if this one is better
            worst_idx = np.argmax([t[0] for t in top_candidates])
            if sc < top_candidates[worst_idx][0]:
                top_candidates[worst_idx] = (sc, a.copy())

    # ------------------------------------------------------------------
    # 2️⃣  Short Adam runs from the few best starts
    # ------------------------------------------------------------------
    for idx, (sc0, a0) in enumerate(top_candidates):
        sub_deadline = min(deadline - 30.0, time.time() + 30.0)   # at most 30 s each
        if time.time() >= sub_deadline:
            break
        best_seq, best_score = _adam_optimize(a0, start_time, sub_deadline,
                                              best_seq, best_score, L_fft)
        print(f"[{time.time() - start_time:.2f}s] Adam restart {idx} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long projected sub‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        best_seq, best_score = _pgd_descent(best_seq, start_time, sub_deadline,
                                            best_seq, best_score, L_fft)
        print(f"[{time.time() - start_time:.2f}s] PGD final best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb / simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _final_hill_climb(best_seq, start_time,
                                                deadline, best_seq,
                                                best_score, L_fft)
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise onto the probability simplex and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
