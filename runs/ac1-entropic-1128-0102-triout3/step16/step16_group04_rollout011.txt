def compute_score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 2).bit_length()) # next pow2
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a*sum_a))

"""
Simple stochastic optimizer for the step‑function problem.

The score is  2·n·max(conv(a,a)) / (sum(a)²) .
Because the objective is homogeneous, only the *shape* of the vector matters;
the algorithm works directly with the normalized (sum = 1) vector.

Strategy
--------
1.  Initialise a candidate.
    * If the global variable ``height_sequence_1`` exists, it is taken,
      renormalised and used as a starting point.
    * Otherwise a uniform vector is used.
2.  Warm‑up: a few Dirichlet draws (different concentrations) are sampled
    quickly; the best of those replaces the current best.
3.  Simulated‑annealing style hill‑climbing.
    In each iteration a *move* is proposed:
        – transfer a random amount between two indices,
        – reshuffle the mass on a small random subset,
        – mix the current vector with the uniform distribution.
    The new vector is accepted if it improves the score, or with probability
    exp((old‑score−new‑score)/T) where the temperature T is linearly cooled.
    The best ever vector is kept.
4.  After the time budget the best vector is returned (non‑negative, summed to 1).

The implementation uses an FFT‑based convolution (O(n log n)) and is
restricted to a single CPU.  It respects the 1000 s budget and prints
intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures – needed for the sandbox)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _score_of(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective used by ``evaluate_sequence``.
    The vector is assumed to be non‑negative; it is normalised internally.
    """
    n = a.shape[0]
    if a.sum() < 0.01:          # protection against division by (almost) zero
        return float("inf")
    # normalise to keep the scale invariant (helps numerical stability)
    a = a / a.sum()
    L = _next_pow2(2 * n - 1)   # length for zero‑padded FFT
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Move generators
# ----------------------------------------------------------------------
def _pair_transfer(a: np.ndarray, rng: np.random.Generator, max_delta: float = 0.02):
    """Move a random amount from one index to another."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    allowed = min(a[i], max_delta)
    if allowed <= 0.0:
        return a
    delta = rng.random() * allowed
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def _subset_reshuffle(a: np.ndarray, rng: np.random.Generator, max_subset: int = 6):
    """Pick a small random subset and redistribute its total mass uniformly."""
    n = a.shape[0]
    k = rng.integers(2, max_subset + 1)
    idx = rng.choice(n, size=k, replace=False)
    total = a[idx].sum()
    if total == 0.0:
        return a
    # new proportions from a Dirichlet (α = 1 → uniform on simplex)
    new_weights = rng.random(k)
    new_weights /= new_weights.sum()
    b = a.copy()
    b[idx] = total * new_weights
    return b


def _mix_with_uniform(a: np.ndarray, rng: np.random.Generator, uniform: np.ndarray):
    """Convex (or slightly extrapolating) combination with the uniform vector."""
    lam = rng.uniform(-0.15, 0.30)          # allow a little overshoot
    b = (1.0 - lam) * a + lam * uniform
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return uniform.copy()
    return b / s


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list sums to roughly one (the evaluator rescales
    automatically, but this keeps numbers in a stable range).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to guarantee termination before the external timeout
    deadline = start_time + budget_s - 0.10

    # --------------------------------------------------------------
    # Problem size – moderate (fast FFT) but large enough to beat 1.507
    # --------------------------------------------------------------
    try:
        # If the global starter exists, reuse its length
        n = int(len(height_sequence_1))
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0.0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        n = 2048                     # fallback size
        starter = None

    uniform = np.full(n, 1.0 / n)

    # ----------- initial best candidate --------------------------------
    if starter is not None and starter.size == n:
        best_a = starter.copy()
        best_score = _score_of(best_a)
        print(f"[0.00s] Starter (from height_sequence_1) score={best_score:.6f}")
    else:
        best_a = uniform.copy()
        best_score = _score_of(best_a)
        print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # --------------- warm‑start with Dirichlet samples -----------------
    warm_deadline = min(deadline, start_time + 0.10 * (deadline - start_time))
    while time.time() < warm_deadline:
        # draw concentration from a log‑uniform range
        log_c = rng.uniform(np.log(0.1), np.log(5.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _score_of(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

    # --------------- simulated‑annealing hill‑climb --------------------
    current = best_a.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule (starts around 0.5, ends near 0)
        fraction = (time.time() - start_time) / (deadline - start_time)
        T = max(1e-8, 0.5 * (1.0 - fraction))

        # choose a move type
        r = rng.random()
        if r < 0.45:
            cand = _pair_transfer(current, rng, max_delta=0.02)
        elif r < 0.75:
            cand = _subset_reshuffle(current, rng, max_subset=7)
        else:
            cand = _mix_with_uniform(current, rng, uniform)

        # normalise and guard against accidental zero‑vector
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            cand = uniform.copy()
        else:
            cand = cand / s

        cand_score = _score_of(cand)

        # Acceptance rule (Metropolis)
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / T):
            current, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] New best score {best_score:.6f}  iter={it:,}")

        # every few thousand steps, snap back to the best known solution
        if it % 5000 == 0:
            current, cur_score = best_a.copy(), best_score

    # --------------------------------------------------------------
    # Final normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    tot = best_a.sum()
    if tot > 0.0:
        best_a = best_a / tot
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
