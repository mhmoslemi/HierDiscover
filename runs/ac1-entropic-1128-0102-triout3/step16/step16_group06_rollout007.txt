def _adam_descent(a_start, deadline, best_seq, best_score):
    a = a_start.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1  # initial learning rate for parameter space (not scaled).
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        maxconv = conv.max()
        cur_score = 2.0 * n * maxconv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t} score={cur_score:.6f}")
        # subgradient.
        grad = _subgradient_maxconv(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # step length scaling: we might need to scale step to keep within simplex
        a_new = a - step
        # project onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        # compute new score
        new_conv = _conv_self(a_new)
        new_score = 2.0 * n * new_conv.max()
        # Accept if improved else maybe still accept with probability? For simplicity, accept improvement else continue with same a.
        if new_score < cur_score:
            a = a_new
        else:
            # maybe occasional random jitter to explore
            if t % 500 == 0:
                rng = np.random.default_rng()
                i1,i2 = rng.choice(n, size=2, replace=False)
                eps = a[i1] * (rng.random() * 0.01 + 1e-6)
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))
    return best_seq, best_score

"""
Propose a step–function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where n = len(a).  The algorithm works in the following stages:

1.  A cheap analytic scan of the one‑parameter geometric families
    a_i ∝ r^i (r∈(0,1)) – the best of these are often already better
    than the previous best bound 1.5078.

2.  A large number of random Dirichlet draws and left‑geometric / right‑geometric
    mixtures to obtain diverse good starting points.

3.  A projected Adam (adaptive‑momentum) sub‑gradient descent on the
    L∞‑norm of the self‑convolution, keeping the vector on the probability
    simplex after each step.

4.  A final tiny‑mass hill‑climb (simulated‑annealing style) that shuffles
    tiny amounts of mass from “bad” to “good’’ coordinates.

All phases respect the global 1000 s wall‑clock budget.  The best
sequence found (normalised to sum = 1) is returned as a plain list of
Python floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (vector is assumed to be normalised to sum=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (normalised to sum=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration parameter – log-uniform in a wide range
        log_c = rng.uniform(-2.0, 0.7)          # c ∈ [0.01, 5]
        c = 10.0 ** log_c
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=1500):
    """
    Mixtures of left‑geometric, right‑geometric and uniform components.
    """
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.9999)               # geometric ratio
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.5f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Analytic scan of the pure geometric family a_i ∝ r^i
# ----------------------------------------------------------------------


def _best_geometric_seq(n, best_seq, best_score, deadline, samples=2000):
    """Search over r∈(0,1) for the pure geometric decreasing sequence."""
    # The search is performed in log‑space for better coverage near 1.
    rng = np.random.default_rng()
    ks = np.arange(2 * n - 1, dtype=np.float64)
    t = np.concatenate([np.arange(1, n + 1), np.arange(n - 1, 0, -1)], dtype=np.float64)

    for _ in range(samples):
        if time.time() >= deadline:
            break
        # pick log‑r uniformly in a small interval that contains the optimum
        log_r = rng.uniform(-1e-4, np.log(0.99999))   # r very close to 1 is interesting
        r = np.exp(log_r)
        if r <= 0 or r >= 1:
            continue
        # normalisation constant
        denom = 1.0 - r ** n
        if denom == 0:
            continue
        c = (1.0 - r) / denom
        # convolution (analytic, no FFT)
        conv_vals = (c * c) * t * (r ** ks)
        M = conv_vals.max()
        sc = 2.0 * n * M
        if sc < best_score:
            best_score, best_seq = sc, (c * (r ** np.arange(n)))
            print(f"[{time.time() - start_time:.2f}s] Geom r={r:.7f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------


def _adam_descent(start_seq, deadline, best_seq, best_score):
    """Projected Adam on the L∞‑self‑convolution objective (sum normalised to 1)."""
    a = start_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base learning‑rate (in the simplex space)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution (already 2·… incorporated later)
        grad = _subgradient_maxconv(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # improvement test (simple – accept only if better)
        new_score = 2.0 * n * _conv_self(a_new).max()
        if new_score < cur_score:
            a = a_new
        else:
            # occasional jitter to leave flat plateaus
            if t % 500 == 0:
                rng = np.random.default_rng()
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.02 + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final tiny‑mass hill‑climb (simulated‑annealing flavour)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    it = 0
    temperature = 0.05                     # initial acceptance temperature

    while time.time() < deadline:
        it += 1
        if it % 5000 == 0:
            # cooling schedule
            temperature *= 0.9
            if temperature < 1e-6:
                temperature = 1e-6

        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source – large positive gradient, destination – small (or negative) gradient
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # propose a transfer of a few percent of the source mass
        frac = rng.random() * 0.04 + 0.001                 # 0.1% – 4%
        eps = min(a[src] * frac, a[src] * 0.5)             # never move more than half
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = best_score
        new_score = float(2.0 * n * _conv_self(a_prop).max())

        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            # accept uphill with a probability that decays with temperature
            if rng.random() < np.exp(-(new_score - best_score) / temperature):
                a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The search proceeds through several cheap exploratory phases,
    a longer Adam‑based sub‑gradient descent, and a final hill‑climb.
    The total wall‑clock time never exceeds ``budget_s`` seconds.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05        # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate length keeps the FFT cheap while allowing
    # enough degrees of freedom to approach the continuous optimum.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if the host provides one.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – analytic geometric scan (very cheap)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.10
    if time.time() < phase1_deadline:
        best_seq, best_score = _best_geometric_seq(
            n, best_seq, best_score, phase1_deadline, samples=4000
        )

    # ------------------------------------------------------------------
    # Phase 2 – many random Dirichlet draws
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase2_deadline, trials=3000
        )

    # ------------------------------------------------------------------
    # Phase 3 – random left/right geometric mixtures
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.45
    if time.time() < phase3_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase3_deadline, trials=2500
        )

    # ------------------------------------------------------------------
    # Phase 4 – Adam projected sub‑gradient descent
    # ------------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.80
    if time.time() < phase4_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descent")
        best_seq, best_score = _adam_descent(
            best_seq, phase4_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
