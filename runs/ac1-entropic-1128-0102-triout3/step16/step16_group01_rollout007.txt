def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    end = start + budget_s - 0.1

    # Determine n
    n = 4096  # could adapt if height_sequence_1 length is known.

    # Initialize with given best or uniform.
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim == 1 and init.size == n:
            a = np.maximum(init, 0.0)
            total = a.sum()
            if total > 0:
                a = a / total
            else:
                a = np.full(n, 1.0/n)
        else:
            a = np.full(n, 1.0/n)
    except Exception:
        a = np.full(n, 1.0/n)

    # Compute conv via FFT (full length 2n-1)
    L = 1 << ((2*n - 2).bit_length())  # next pow2 >= 2n-1
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # SA parameters:
    T0 = 0.1 * cur_score  # initial temperature
    # We'll schedule T = T0 * (1 - (elapsed / total))^p maybe linear.
    total_time = end - start

    # Moves: variable magnitude (fraction up to max_move = 0.02)
    max_move_frac = 0.02

    it = 0
    while time.time() < end:
        it += 1
        # choose source depending on current distribution: weighted by a
        # to bias moves from larger masses
        src = rng.choice(n, p=a)  # sample according to a
        if a[src] < 1e-12:
            continue
        # choose dst (any other index)
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1

        # epsilon: fraction of source mass
        frac = rng.random() * max_move_frac
        eps = a[src] * frac
        if eps < 1e-12:
            continue

        # incremental update conv
        # compute new conv from old conv
        conv_new = conv.copy()
        # shift contributions
        # conv_new[src:src+n] -= 2*eps * a
        conv_new[src:src+n] -= 2.0 * eps * a
        conv_new[dst:dst+n] += 2.0 * eps * a

        # d*d adjustments
        idx2src = src + src
        idx2dst = dst + dst
        idxsum = src + dst
        eps2 = eps * eps
        conv_new[idx2src] += eps2
        conv_new[idx2dst] += eps2
        conv_new[idxsum] -= 2.0 * eps2

        # new max
        max_new = conv_new.max()
        new_score = 2.0 * n * max_new  # sum stays 1

        # Temperature schedule
        elapsed = time.time() - start
        T = T0 * (1 - elapsed/total_time)  # linear decay to 0

        # Acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score)/max(T, 1e-12)):
            # accept
            # update a and conv
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] iter={it} new best {best_score:.6f}")

        # Periodically recompute conv from scratch to reduce drift
        if it % 5000 == 0:
            Af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
            cur_score = 2.0 * n * conv.max()

    # final after loop
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0/n)
    print(f"Finished, best score approx {best_score:.6f}")
    return best_a.tolist()

"""
Simulated‑annealing mass‑transfer optimizer.

The algorithm starts from the best known sequence (if the global variable
`height_sequence_1` is present, otherwise from the uniform distribution) and
performs a long‑running Metropolis‑style search that repeatedly moves a small
fraction of probability mass from one index to another.  The self‑convolution
is updated incrementally (O(n) per move) instead of recomputed via FFT, which
allows tens of thousands of proposals per second.  Temperature and step size
are annealed so the search explores widely at first and then greedily refines.
A final short exponentiated‑gradient fine‑tuning phase (exact sub‑gradient of
the max‑convolution) polishes the best vector found.

The returned list is a non‑negative simplex vector (sum≈1) and can be fed to
`evaluate_sequence`.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Exact full self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes diagonal correctly (2*a_i)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient fine‑tuning (exact max‑objective)
# ----------------------------------------------------------------------


def _expgrad_fine_tune(a0: np.ndarray,
                       L_fft: int,
                       iters: int,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Short EG refinement on the exact L∞ objective.
    Returns (best_a, best_score) where score = 2·n·max(conv) (sum = 1).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = _full_conv_fft(a)
    best_score = 2.0 * n * conv.max()
    cur_score = best_score
    eta0 = 0.8

    for it in range(1, iters + 1):
        # worst index
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.001 * it)

        # multiplicative update
        mult = np.exp(-eta * g)
        a = a * mult
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # recompute convolution (cheap enough for a few thousand iters)
        conv = _full_conv_fft(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional jitter to avoid being stuck
        if it % 500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
            conv = _full_conv_fft(a)
            cur_score = 2.0 * n * conv.max()
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
        2 * n * max(conv(a,a)) / (sum(a)**2)

    The method is a long simulated‑annealing mass‑transfer procedure
    (≈800 s) followed by a short exponentiated‑gradient polishing phase.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.14      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – use the length of the global starter if available,
    # otherwise fall back to 4096.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size == 0:
            raise ValueError
        n = starter.size
    except Exception:
        n = 4096
        starter = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise starter to a probability simplex
    starter = np.maximum(starter, 0.0)
    total = starter.sum()
    if total > 0.0:
        a = starter / total
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial convolution (full) and score (sum = 1 → score = 2·n·max_conv)
    # ------------------------------------------------------------------
    L_fft = _next_pow2(2 * n - 1)
    conv = _full_conv_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] start length={n}, "
          f"initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing mass‑transfer phase
    # ------------------------------------------------------------------
    # Allocate most of the budget to this phase; keep ~30 s for fine‑tune.
    fine_tune_time = 30.0
    anneal_deadline = deadline - fine_tune_time
    it = 0

    # Parameters for the schedule
    total_anneal = anneal_deadline - start_time
    max_move_frac_start = 0.05   # maximal fraction of source mass at the start
    max_move_frac_end = 0.001    # at the very end

    while time.time() < anneal_deadline:
        it += 1
        elapsed = time.time() - start_time
        # Annealed temperature (linear decay to 0)
        T = cur_score * 0.05 * (1.0 - elapsed / total_anneal)
        T = max(T, 1e-14)

        # Adaptive move size (large early, very small later)
        move_frac = max_move_frac_end + (max_move_frac_start - max_move_frac_end) * (
            1.0 - elapsed / total_anneal)

        # --- pick a source index (biased by current mass) ---
        src = rng.choice(n, p=a)                     # probability‑proportional
        if a[src] < 1e-16:
            continue

        # --- pick a distinct destination ---
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # --- amount of mass to move ---
        eps = a[src] * rng.random() * move_frac
        if eps < 1e-16:
            continue

        # ----- Incremental convolution update -----
        # conv_new = conv - 2*eps*shift(src) + 2*eps*shift(dst) + d*d terms
        conv_new = conv.copy()
        # shift contributions (vectorised slice updates)
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        # d*d term (exact)
        eps2 = eps * eps
        idx2src = src + src
        idx2dst = dst + dst
        idxsum = src + dst
        conv_new[idx2src] += eps2
        conv_new[idx2dst] += eps2
        conv_new[idxsum] -= 2.0 * eps2

        # ----- evaluate new score -----
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max     # sum stays 1

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            # commit the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] it={it:,} new best {best_score:.6f}")

        # Periodically (every few thousand steps) recompute the convolution
        # from scratch to avoid accumulation of floating‑point drift.
        if it % 5000 == 0:
            conv = _full_conv_fft(a)
            cur_score = 2.0 * n * conv.max()
            # Also renormalise a (tiny drift may happen)
            a_sum = a.sum()
            if abs(a_sum - 1.0) > 1e-12:
                a = a / a_sum

    # ------------------------------------------------------------------
    # Exponentiated‑gradient polishing phase (exact max‑objective)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # use at most ~10 000 iterations (fast enough for the remaining time)
        iters = int(min(15000, remaining * 200))   # rough 200 it/s estimate
        best_a, best_score = _expgrad_fine_tune(best_a,
                                                L_fft,
                                                iters,
                                                rng)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
