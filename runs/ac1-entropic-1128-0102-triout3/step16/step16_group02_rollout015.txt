    # Very light random refinement (only if the analytic family failed)
    # --------------------------------------------------------------
    if best_seq is None or best_score > 1.503:
        print("Analytic family did not reach target – running a tiny random search.")
        rng = np.random.default_rng(seed + 12345)
        # we perform a bounded number of cheap Dirichlet draws
        for _ in range(2000):
            if time.time() - start > budget_s - 0.1:
                break
            # concentration >0 yields a dense vector; 0.1 gives a bit of sparsity.
            a = rng.gamma(shape=0.3, scale=1.0, size=n)
            a /= a.sum()
            conv = np.convolve(a, a, mode='full')
            score = float(2.0 * n * conv.max())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] Random candidate score={score:.6f}")
                if best_score <= 1.503:
                    break

    # --------------------------------------------------------------
    # Safety normalisation – the returned list must be non‑negative and sum > 0.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Final best score≈{best_score:.6f}")
    # Return plain Python floats as required by the evaluation harness.
    return [float(v) for v in best_seq]
