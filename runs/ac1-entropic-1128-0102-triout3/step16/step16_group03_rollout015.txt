def _project_to_simplex(v):
    n = v.shape[0]
    # shift v to ensure sum = 1
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries <= 0
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    if w.sum() == 0:
        # fallback uniform
        w = np.full(n, 1.0 / n)
    else:
        w /= w.sum()
    return w
```

"""
Hybrid optimizer with smoothed max‑convolution gradient descent.

The optimiser works in four phases (time‑controlled):

1️⃣  Broad random sampling (Dirichlet + a few extreme Beta shapes) to get a
    decent starting point.

2️⃣  Smoothed‑objective gradient descent on the probability simplex.
    The true objective 2·n·max(conv) is approximated by
        f_γ(a) = (2·n/γ)·log‑sum‑exp(γ·conv(a,a))
    which is smooth for moderate γ.  We cycle γ from small to large,
    use a simple projected‑gradient step with a 1/√t step‑size,
    and occasionally restart from a fresh Dirichlet draw when progress stalls.
    This tends to escape the shallow plateaux that trap the previous
    exponentiated‑gradient method.

3️⃣  Targeted hill‑climbing (mass transfer between indices) with
    simulated‑annealing style acceptance.

4️⃣  Fine mixing with the uniform distribution – a golden‑section search for
    the optimal convex combination λ·a+(1−λ)·uniform.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x and y (length len(x)+len(y)-1) via FFT."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, L_self: int) -> np.ndarray:
    """Full convolution a*a."""
    return _conv_fft(a, a, L_self)


def _objective_score(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) – a is assumed normalised (sum=1)."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho (the last index where u_i - cssv_i / i > 0)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive → return uniform vector
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise (numerical safety)
    s = w.sum()
    if s == 0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Discrete Beta(α,β) density on n points, normalised."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf

# 1️⃣  Broad random search (Dirichlet + Beta seeds)
# ----------------------------------------------------------------------
def _broad_random_search(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # Deterministic Beta seeds (evaluated once)
    beta_params = [
        (0.2, 0.2),
        (0.3, 0.3),
        (0.5, 0.5),
        (0.2, 0.8),
        (0.8, 0.2),
        (0.3, 0.7),
        (0.7, 0.3),
    ]
    beta_seeds = []
    for a, b in beta_params:
        v = _beta_shape(a, b, n)
        if v is not None:
            beta_seeds.append(v)

    while time.time() < deadline:
        # ---- Symmetric Dirichlet draw (single concentration) ----
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s > 0:
            a = sample / s
            sc = _objective_score(a, n)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # ---- Dirichlet with a random concentration vector ----
        # (adds extra diversity)
        if rng.random() < 0.2:
            base_c = rng.uniform(np.log(0.01), np.log(100.0))
            alphas = rng.gamma(shape=np.exp(base_c), scale=1.0, size=n)
            sample = rng.gamma(shape=alphas, scale=1.0)
            s = sample.sum()
            if s > 0:
                a = sample / s
                sc = _objective_score(a, n)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Random‑α Dirichlet → score={sc:.6f}")

        # ---- Beta seeds (cheap, occasional) ----
        if rng.random() < 0.005 and beta_seeds:
            v = beta_seeds[rng.integers(len(beta_seeds))]
            sc = _objective_score(v, n)
            if sc < best_score:
                best_score, best_seq = sc, v.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")
    return best_seq, best_score

# 2️⃣  Smoothed‑objective gradient descent with restarts
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Stable softmax for a vector of length 2·n‑1."""
    shift = conv.max()
    exps = np.exp(gamma * (conv - shift))
    total = exps.sum()
    if total == 0:
        # all underflow – fallback to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


def _grad_softmax_maxconv(
    a: np.ndarray,
    conv_self: np.ndarray,
    gamma: float,
    L_grad: int,
) -> np.ndarray:
    """
    Gradient of the smoothed objective (2·n/γ)·log‑sum‑exp(γ·conv).
    Returns ∇_a f_γ(a).
    """
    n = a.shape[0]
    w = _softmax_weights(conv_self, gamma)          # length 2·n‑1
    # full convolution w * a
    conv_wa = _conv_fft(w, a, L_grad)               # length (2n‑1)+n‑1 = 3n‑2
    h = conv_wa[n - 1 : n - 1 + n]                  # slice of length n
    grad = 4.0 * n * h                              # factor 4·n from derivation
    return grad


def _annealed_gradient_descent(
    a0: np.ndarray,
    n: int,
    L_self: int,
    L_grad: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution.
    Gamma is increased cyclically (0.5 → 100).  If no improvement for a long
    stretch we randomly restart from a Dirichlet draw.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    it = 0
    last_improve_it = 0
    restart_interval = 80_000          # iterations without improvement → restart
    gamma_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    gamma_idx = 0

    # Pre‑compute an initial objective value for the current a
    conv_self = _conv_self(a, L_self)
    cur_score = _objective_score(a, n)

    while time.time() < deadline:
        it += 1
        # cycle gamma
        gamma = gamma_schedule[gamma_idx]
        gamma_idx = (gamma_idx + 1) % len(gamma_schedule)

        # gradient of smoothed objective
        grad = _grad_softmax_maxconv(a, conv_self, gamma, L_grad)

        # classic 1/√t step‑size (tunable)
        eta = 0.8 / math.sqrt(it)

        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new)

        # evaluate the exact objective for the trial point
        conv_new = _conv_self(a_new, L_self)
        new_score = float(2.0 * n * conv_new.max())

        # simulated‑annealing style acceptance (helps escaping shallow minima)
        if new_score < cur_score:
            accept = True
        else:
            # temperature decays linearly with remaining time
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 3e-4 * (remaining / total)
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12))

        if accept:
            a = a_new
            conv_self = conv_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                last_improve_it = it
                print(f"[{time.time() - start_time:.2f}s] New best {new_score:.6f} (γ={gamma:.1f})")

        # restart if we have been stagnant for a while
        if it - last_improve_it > restart_interval:
            # random restart (Dirichlet with random concentration)
            log_c = rng.uniform(np.log(0.01), np.log(30.0))
            c = np.exp(log_c)
            sample = rng.gamma(shape=c, scale=1.0, size=n)
            a = sample / sample.sum()
            conv_self = _conv_self(a, L_self)
            cur_score = _objective_score(a, n)
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Restart (c={c:.3f}) – score={cur_score:.6f}")

    return best_seq, best_score

# 3️⃣  Targeted hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    n: int,
    L_self: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from a donor to a receiver."""
    a = a0.copy()
    cur_score = _objective_score(a, n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # choose donor – mostly the current maximum entry
        if rng.random() < 0.7:
            donor = np.argmax(a)
        else:
            donor = rng.integers(n)

        # choose receiver – mostly the current minimum entry
        if rng.random() < 0.7:
            rec = np.argmin(a)
        else:
            rec = rng.integers(n)

        if donor == rec or a[donor] < 1e-14:
            continue

        # move a small fraction of donor's mass
        eps = min(0.02 * a[donor], 1e-4)  # at most 2 % of donor or a hard cap
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[rec] += eps

        new_score = _objective_score(a_new, n)

        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            # annealing acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(
    a: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform via golden‑section."""
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    # golden‑section loop
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = (lam1 * a + (1.0 - lam1) * uniform)
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = (lam2 * a + (1.0 - lam2) * uniform)
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin for the runner
    deadline = start_time + budget_s - 0.25

    # Problem size – 4096 is a sweet spot between quality and speed
    n = 4096
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ---- 0️⃣ Initialise with uniform and optional global starter ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ---- 1️⃣ Broad random search (≈10 % of the budget) ----
    phase1_end = start_time + 0.10 * (deadline - start_time)
    if time.time() < phase1_end:
        best_seq, best_score = _broad_random_search(
            n, best_seq, best_score, phase1_end, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After random search best score≈{best_score:.6f}")

    # ---- 2️⃣ Smoothed gradient descent with restarts (≈70 % of budget) ----
    phase2_end = start_time + 0.80 * (deadline - start_time)
    if time.time() < phase2_end:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed gradient descent")
        best_seq, best_score = _annealed_gradient_descent(
            best_seq,
            n,
            L_self,
            L_grad,
            start_time,
            phase2_end,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After grad‑desc best≈{best_score:.6f}")

    # ---- 3️⃣ Hill‑climbing local refinement (≈15 % of budget) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq,
            n,
            L_self,
            start_time,
            deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best≈{best_score:.6f}")

    # ---- 4️⃣ Fine mixing with uniform (remaining time) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq,
            n,
            deadline,
            start_time,
            best_seq,
            best_score,
        )

    # ---- Final normalisation and return ----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
