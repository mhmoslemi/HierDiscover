def project_to_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

        score = 2·n·max(conv(a,a)) / (sum(a)²)

The optimizer is a three‑phase heuristic:

1️⃣  Warm‑start – a few Dirichlet draws (including the optional global
    starter `height_sequence_1`) are sampled and the best is kept.

2️⃣  Sub‑gradient descent on the true max‑convolution (no smooth‑max
    approximation).  At each iteration we locate the index `k*` where the
    self‑convolution is maximal, compute the sub‑gradient
        g_i = 2·a_{k*‑i}   (zero when the index is out of range)
    then take an additive step `a ← a – η·g` and re‑project onto the
    probability simplex by simple clipping + renormalisation.  The
    stepsize follows η = η0 / sqrt(t) with a relatively large initial
    η0 so that the algorithm makes visible progress from the start.
    Random jitter is added occasionally to escape plateaus.

3️⃣  Final random‑pair hill‑climb – with the remaining budget a cheap
    random‑pair transfer move is repeatedly tried:
        move a small δ from i to j,
    keeping the vector on the simplex.  Moves are accepted only when
    they improve the exact score (computed via FFT for speed).  This
    phase performs a fine‑grained polishing of the best candidate found
    in phase 2.

All heavy work (convolution, gradients) is done via FFT, giving an
O(n log n) cost per iteration.  The routine respects the 1000 s budget
and returns a plain Python list of non‑negative floats whose sum is
≈ 1 (the evaluator will re‑normalise anyway).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution / correlation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of length L_fft."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (ignores the 1/(sum)^2 factor)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _subgrad_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current maximum index `max_idx`.
    Returns a vector g where g_i = 2·a_{max_idx‑i} when the index is inside
    the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    # i runs from lo..hi inclusive
    i = np.arange(lo, hi + 1, dtype=int)
    grad[i] = 2.0 * a[max_idx - i]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Warm‑start via Dirichlet draws (optional global starter)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter sampled log‑uniformly
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Additive projected sub‑gradient descent on max(conv(a,a)).
    Uses a simple 1/√t stepsize schedule and occasional random jitter.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n, L_fft)

    iteration = 0
    eta0 = 0.6          # initial stepsize (tuned experimentally)

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self_fft(a, L_fft)
        max_idx = int(conv.argmax())

        # sub‑gradient for the current maximal index
        grad = _subgrad_max(a, max_idx)

        # stepsize schedule
        eta = eta0 / np.sqrt(iteration)

        # additive move + simple projection (clip + renormalise)
        a_tmp = a - eta * grad
        a_tmp = np.maximum(a_tmp, 0.0)
        sm = a_tmp.sum()
        if sm > 0.0:
            a = a_tmp / sm
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional jitter to avoid getting stuck
        if np.random.rand() < 0.01:
            eps = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

        cur_score = _objective_score(a, n, L_fft)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad it={iteration:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Final random‑pair hill‑climb (pure exploitation)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                n: int,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer a small amount of mass between two random indices.
    Accept only improving moves.  This cheap local search finishes the
    optimisation.
    """
    best_a = a.copy()
    best_score = _objective_score(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move a tiny fraction of the mass from i to j
        delta = rng.uniform(0.0, min(best_a[i], 0.001))
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list contains non‑negative entries and (approximately)
    unit sum.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin to avoid overrunning the budget
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good balance of speed / quality
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # for a*a
    # ------------------------------------------------------------------
    # 0️⃣ Initialise – uniform or the optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Try to use a globally supplied starter ``height_sequence_1`` if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start (≈10 % of the total time)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.10 * budget_s)
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score, warm_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent (≈75 % of the total time)
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, start_time + 0.85 * budget_s)

    # ------------------------------------------------------------------
    # Normalise & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
