"""
Optimiser for the step‑function evaluation problem.

The search proceeds in five stages:

1.  Initialise with a short sweep of random Dirichlet draws.
2.  Add a small set of geometric / uniform mixtures.
3.  Projected sub‑gradient descent (the same routine used previously)
    – this quickly reaches a decent local minimum.
4.  A smooth‑max gradient descent:
    * the true objective is 2·n·max(conv(a,a)).
    * we replace the hard max by a smooth soft‑max (parameter α)
      which yields a dense gradient.
    * an adaptive back‑tracking line‑search and a slowly increasing α
      drive the iterates past the plateaus that trap the pure sub‑gradient
      method.
5.  A final tiny‑mass hill‑climb to polish any remaining wiggle‑room.

All iterates are projected onto the probability simplex (sum = 1,
non‑negative) after each update.  The whole algorithm respects a 1000 s
wall‑clock budget and returns the best sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (a is real, non‑negative)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum = 1, v ≥ 0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – the value minimised by evaluate_sequence."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (sum(a)=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=300):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=250):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (phase 3)
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq, deadline, best_seq, best_score):
    """
    Direct minimise 2·n·max(conv(a,a)) by sub‑gradient steps.
    Each iterate is projected onto the simplex.
    """
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.2
    min_step = 1e-12
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={iteration} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        # back‑tracking line search
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-12)

        # occasional jitter to leave flat plateaus
        if iteration % 4000 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (phase 4)
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the α‑smoothed max of the self‑convolution.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = _conv_self(a)            # length 2n‑1
    maxv = conv.max()
    # soft‑max weights (stable exponentiation)
    w = np.exp(alpha * (conv - maxv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2 * (w * a_rev) convolved, sliced to length n
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad


def _smooth_gradient_descent(start_seq, deadline, best_seq, best_score):
    """
    Gradient descent on a smooth approximation of max(conv(a,a)).
    Uses back‑tracking line‑search and a slowly increasing smoothing
    parameter α.
    """
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.2
    min_step = 1e-12
    iteration = 0
    alpha = 5.0               # start with a very smooth max
    alpha_inc = 0.02          # increase each iteration (capped later)

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGrad it={iteration} score={cur_score:.6f}")

        # gradient of the smoothed max
        grad = _smooth_max_gradient(a, alpha)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        # back‑tracking line search
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-12)

        # slowly tighten the smooth max
        alpha = min(alpha + alpha_inc, 200.0)

        # periodic tiny jitter to avoid stagnation
        if iteration % 1200 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.004 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny‑mass hill‑climb (final polishing, phase 5)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to shake out of a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) minimizing
    evaluate_sequence().  The workflow is:

    * random Dirichlet / mixture exploration,
    * projected sub‑gradient descent,
    * smooth‑max gradient descent,
    * a final tiny‑mass hill‑climb.

    The total wall‑clock time never exceeds ``budget_s`` seconds.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4096                         # keep the same size as the reference run
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈5 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.05
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=500
        )

    # --------------------------------------------------------------
    # Phase 2 – random mixtures (≈5 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.10
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=300
        )

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (≈25 % of budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.35
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(
            best_seq, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – smooth‑max gradient descent (≈45 % of budget)
    # --------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.80
    if time.time() < phase4_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
        best_seq, best_score = _smooth_gradient_descent(
            best_seq, phase4_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth‑grad → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final tiny‑mass hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
