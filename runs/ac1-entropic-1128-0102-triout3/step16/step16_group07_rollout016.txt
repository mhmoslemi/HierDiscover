"""
Improved search for a non‑negative step function that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²).

The algorithm consists of
  * a dense sweep of the geometric family (the empirically best family)
    with a fine grid around the analytically optimal scale ε≈c/n,
  * a modest broad sweep of the beta family (extended parameter range),
  * many random Dirichlet samples (larger concentration range),
  * an extended projected sub‑gradient descent from the current best,
  * a simple simulated‑annealing hill‑climb for final refinement.

All phases respect the overall 1000 s time budget.
The best sequence found (normalised to sum = 1) is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq))  (seq is already normalised to sum=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Focused geometric sweep
# ----------------------------------------------------------------------
def _sweep_geometric(n: int,
                     best_seq: np.ndarray,
                     best_score: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Fine sweep of a·r^i (geometric) with optional mixing with uniform."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    # analytically optimal ε is about c/n with c≈2 → explore a factor of 10 around it
    eps_center = 2.0 / n
    eps_min = eps_center * 0.2
    eps_max = eps_center * 5.0
    eps_grid = np.logspace(np.log10(eps_min), np.log10(eps_max), 400)

    # mixing fractions – only coarse points are needed
    lam_grid = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 0.9])

    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0.0:
            continue
        geo /= s                               # normalised geometric
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.2e} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Expanded beta‑type sweep
# ----------------------------------------------------------------------
def _sweep_beta(n: int,
                best_seq: np.ndarray,
                best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Beta‑type densities with an extended parameter grid."""
    idx = np.arange(n, dtype=np.float64) + 1.0       # i from 1..n
    rev = np.arange(n, 0, -1, dtype=np.float64)    # n‑i+1
    # broaden the search a bit beyond the previous 4×4 grid
    alphas = np.linspace(0.05, 8.0, 81)
    betas  = np.linspace(0.05, 8.0, 81)

    for a in alphas:
        if time.time() >= deadline: break
        log_a = (a - 1.0) * np.log(idx)
        for b in betas:
            if time.time() >= deadline: break
            log_seq = log_a + (b - 1.0) * np.log(rev)
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if not np.isfinite(s) or s == 0.0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Massive Dirichlet sampling (larger concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Draw many Dirichlet vectors with concentration from 0.01 to 200."""
    rng = np.random.default_rng()
    # allocate a modest budget of trials; the loop is timed, not count‑limited
    while time.time() < deadline:
        # sample concentration uniformly in log‑space
        log_c = rng.uniform(np.log(0.01), np.log(200.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Prolonged projected sub‑gradient descent
# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution.
    The schedule is longer than in the reference implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # start with a relatively aggressive step, decay slowly
    eta0 = 1.0
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # exact sub‑gradient
        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        # step size decays ~1/√it (common in sub‑gradient methods)
        eta = eta0 / np.sqrt(it)

        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        # accept if improvement, otherwise keep old iterate
        if new_score < cur_score - 1e-15:
            a = a_candidate
        else:
            # occasional tiny random jitter to escape flat regions
            if it % 300 == 0:
                jitter = np.random.default_rng().normal(scale=1e-4, size=n)
                a = _project_simplex(a + jitter)

    return best_seq, best_score

# 5️⃣  Final simulated‑annealing hill‑climb (tiny moves)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Very fine‑grained hill‑climb based on moving tiny mass fractions."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)

    # temperature schedule: start modest and cool to ~1e‑7
    T0 = 0.02
    total = max(0.0, deadline - time.time())
    decay = (1e-7 / T0) ** (1.0 / total) if total > 0 else 1.0

    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]
        # choose a destination at random (could be same – then we retry)
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a tiny fraction (log‑uniformly distributed)
        frac = np.exp(rng.uniform(np.log(1e-6), np.log(1e-2)))   # between 1e‑6 and 1e‑2
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a normalised list of floats (sum = 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # a small safety margin for the outer harness
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size (keep 4096 – the reference size)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Baseline initialisation (uniform)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional external starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] External starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Focused geometric sweep (most promising family)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0          # reserve time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Extended beta‑type sweep (broader range)
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Massive Dirichlet sampling (larger concentration range)
    # --------------------------------------------------------------
    phase_deadline = deadline - 100.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 4️⃣  Prolonged projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 40.0:
        print(f"[{time.time() - start_time:.2f}s] Starting long PGD")
        pgd_deadline = deadline - 25.0
        best_seq, best_score = _pgd_descent(
            best_seq, pgd_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final fine‑grained hill‑climb / simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
