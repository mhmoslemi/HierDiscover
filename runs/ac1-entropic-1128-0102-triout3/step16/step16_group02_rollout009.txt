    eps = 1e-12

        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        # a_{k - idx} picks indexes
        grad[idx] += a[k - idx]
    # average over active

        2·n·max(conv(a,a)) / (sum(a)**2)

is as small as possible.

Method
------
1.  Initialise many random points on the simplex using Dirichlet draws
    (the concentration is varied over several decades).
2.  For each start run a *smooth* projected gradient descent.
    Instead of the sub‑gradient of the hard max we minimise a smooth
    approximation
            g_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
    (log‑sum‑exp).  Its gradient can be written as a single convolution,
    so each iteration needs only a handful of FFTs.
    The smoothing parameter α is increased gradually – the algorithm
    first moves mass globally and then focuses on the current peak.
3.  Keep the best vector seen over all starts.
4.  Refine it with a lightweight simulated‑annealing hill‑climb that
    repeatedly transfers a small fraction of mass from indices with a
    large “peak‑gradient’’ to low‑gradient ones.
5.  Normalise the final vector to sum 1 and return it as a plain list.

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, next power of two, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) under the convention sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    The formula reduces to a single convolution:
        grad = 2·(a_rev * w)[n‑1 : n‑1+n],
    where w are normalized softmax weights based on conv.
    """
    n = a.shape[0]

    # soft‑max weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    w = exps / exps.sum()                     # length 2·n‑1, all ≥ 0, Σw = 1

    # convolution a_rev * w  (a_rev is a reversed copy of a)
    a_rev = a[::-1]
    L = _next_pow2(3 * n)                     # enough for length n+2n‑1‑1
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    full = np.fft.irfft(Af * Wf, n=L)         # length L
    grad = 2.0 * full[n - 1 : n - 1 + n]      # extract the n relevant entries
    return grad


def _smooth_gd(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    init_alpha: float = 0.5,
    alpha_inc: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with the smooth log‑sum‑exp objective.
    Returns the best point (and its score) found along the trajectory.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_score = _objective_score(a)

        # gradually increase smoothness
        alpha = init_alpha + alpha_inc * it

        # gradient of the smoothed objective
        grad = _smooth_gradient(a, conv, alpha)

        # move inside the simplex: direction with zero sum
        d = -(grad - grad.mean())

        # maximal feasible step size without leaving the non‑negative orthant
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        step = max_step * 0.9

        # back‑tracking line‑search (monotone improvement)
        while step > 1e-12:
            a_new = a + step * d
            if np.any(a_new < 0):
                step *= 0.5
                continue
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                break
            step *= 0.5

        # if line‑search fails, add a tiny jitter to escape stalls
        if step <= 1e-12:
            eps = 1e-7
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()                     # re‑normalise (should stay ≈1)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray, deadline: float, best_seq: np.ndarray, best_score: float
) -> tuple[np.ndarray, float]:
    """Transfer a fraction of mass from high‑gradient to low‑gradient indices."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score
    cur_a = a.copy()
    total_time = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    # pre‑compute a gradient vector (averaged over active max‑shifts)
    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = (1.0 - frac) * cur_score * 0.001  # start temperature ≈0.1 % of score

        # refresh gradient every 100 iterations (cheap)
        if it % 100 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))

        # source = index with large positive gradient, destination = random
        prob = np.maximum(grad, 0)
        if prob.sum() > 0:
            src = rng.choice(n, p=prob / prob.sum())
        else:
            src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a modest fraction of the source mass
        max_frac = 0.25
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        conv_new = _conv_self(a_prop)
        score_new = float(2.0 * n * conv_new.max())
        delta = score_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random shake‑up if stuck
        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * (rng.random() * max_frac)
            a[i1] -= eps2
            a[i2] += eps2
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = float(2.0 * n * _conv_self(a).max())
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the *hard* max (used by the SA)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) – averaged over all active shifts."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (the sequence is automatically normalised to
    sum = 1 before being handed to ``evaluate_sequence``).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size (keep it moderate for speed)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter is present, try it first
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Random Dirichlet initialisers + smooth GD
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    max_total_iters = 0
    # allocate roughly half of the remaining time to this phase
    phase_deadline = deadline - 200.0
    while time.time() < phase_deadline:
        # draw concentration from a wide log‑uniform range
        logc = rng.uniform(-1.8, 1.0)   # ≈ 0.016 … 10
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()

        # smooth projected gradient descent (a few hundred iterations)
        iters = 300
        a_opt, sc = _smooth_gd(a, iters, deadline, init_alpha=0.5, alpha_inc=0.02)

        if sc < best_score:
            best_score, best_seq = sc, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after GD score={best_score:.6f}")

        # stop early if we run out of time
        if time.time() >= deadline - 5.0:
            break

    # --------------------------------------------------------------
    # 2️⃣ Final simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final simulated‑annealing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
