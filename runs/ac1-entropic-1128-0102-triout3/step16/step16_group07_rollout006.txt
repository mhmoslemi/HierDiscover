"""
search_sa_pgd.py

A two‑phase optimiser for the step‑function bound.
Phase 1 – Simulated‑annealing guided by the exact sub‑gradient of
max(conv(a,a)).  At each iteration we move a small amount of mass from
indices that increase the current peak (large sub‑gradient) to indices
that have a small sub‑gradient.  A temperature schedule allows occasional
up‑hill moves to escape plateaux.

Phase 2 – Projected sub‑gradient descent (same idea as the reference
implementation) to polish the best solution found by the annealer.

Both phases respect the global time budget (default 1000 s) and keep the
sequence on the probability simplex (non‑negative, sum = 1).  The routine
returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑to‑real)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1.
    Identical to ``evaluate_sequence`` up to the safety–clamping steps.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search proceeds with a guided simulated‑annealing
    phase followed by a few hundred projected‑sub‑gradient steps.
    The returned object is a plain list of floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – reuse the size of the previous best if available,
    # otherwise fall back to 4096.
    # --------------------------------------------------------------
    try:
        # global variable supplied by the platform (may be missing)
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = init_seq.size
    except Exception:
        n = 4096
        init_seq = None

    # --------------------------------------------------------------
    # Initialise a feasible probability vector (sum=1, all ≥0)
    # --------------------------------------------------------------
    if init_seq is not None and init_seq.ndim == 1 and init_seq.size > 0:
        # Clip negatives, pad/truncate to the chosen length
        seq = np.maximum(init_seq, 0.0)
        if seq.size > n:
            seq = seq[:n]
        elif seq.size < n:
            seq = np.pad(seq, (0, n - seq.size))
    else:
        seq = np.full(n, 1.0 / n)

    seq_sum = seq.sum()
    if seq_sum > 0:
        seq /= seq_sum
    else:
        seq = np.full(n, 1.0 / n)

    best_seq = seq.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Guided simulated annealing
    # --------------------------------------------------------------
    # Reserve ~250 s for the final polishing phase
    sa_deadline = deadline - 250.0
    if sa_deadline <= time.time():
        sa_deadline = time.time() + 1.0  # tiny fallback

    cur_seq = best_seq.copy()
    cur_score = best_score
    L_fft = _next_pow2(2 * n - 1)

    # pre‑compute the current convolution once
    conv_cur = _conv_self(cur_seq, L_fft)

    # SA hyper‑parameters (exponential schedules)
    T_start = max(0.01, best_score * 0.02)      # initial temperature
    T_end = max(1e-8, best_score * 1e-6)       # final temperature
    move_start = 0.03                          # up to 3 % of the source mass
    move_end = 0.0003                          # down to 0.03 % at the end

    rng = np.random.default_rng()

    while time.time() < sa_deadline:
        elapsed = time.time() - start_time
        total_sa = sa_deadline - start_time
        progress = elapsed / total_sa if total_sa > 0 else 1.0

        # temperature and move‑size schedules (exponential decay)
        T = T_start * (T_end / T_start) ** progress
        max_move_frac = move_start * (move_end / move_start) ** progress

        # ------------------------------------------------------------------
        # Choose source (high sub‑gradient) and destination (low sub‑gradient)
        # ------------------------------------------------------------------
        subgrad = _subgradient_maxconv(cur_seq, conv_cur)

        # source: among indices that currently carry mass, pick one with the
        # largest sub‑gradient.
        viable = cur_seq > 1e-12
        if not np.any(viable):
            break
        # restrict to viable indices for source selection
        src_candidates = np.where(viable)[0]
        # pick the viable index with the largest sub‑gradient
        src = src_candidates[np.argmax(subgrad[src_candidates])]

        # destination: index with the smallest sub‑gradient (any index allowed)
        dst = int(np.argmin(subgrad))
        if dst == src:
            # pick a different destination at random
            dst = rng.integers(n)
            if dst == src:
                continue

        # --------------------------------------------------------------
        # Propose a move: transfer a fraction of source mass to destination
        # --------------------------------------------------------------
        delta = cur_seq[src] * max_move_frac * rng.random()
        if delta < 1e-12:
            continue

        prop_seq = cur_seq.copy()
        prop_seq[src] -= delta
        prop_seq[dst] += delta
        # Numerical cleanup
        prop_seq = np.maximum(prop_seq, 0.0)

        # No renormalisation is needed – the move preserves the total sum,
        # but we renormalise occasionally to combat floating‑point drift.
        if (np.abs(prop_seq.sum() - 1.0) > 1e-10):
            prop_seq /= prop_seq.sum()

        # ------------------------------------------------------------------
        # Evaluate the proposal (single convolution)
        # ------------------------------------------------------------------
        conv_new = _conv_self(prop_seq, L_fft)
        new_score = 2.0 * n * conv_new.max()

        # keep the global best
        if new_score < best_score:
            best_score = new_score
            best_seq = prop_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New best {best_score:.6f}")

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move accepted with probability exp(-Δ/T)
            prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = prop_seq
            cur_score = new_score
            conv_cur = conv_new

        # occasional full re‑normalisation to prevent drift
        if rng.random() < 0.001:
            cur_seq = _project_simplex(cur_seq)
            conv_cur = _conv_self(cur_seq, L_fft)
            cur_score = 2.0 * n * conv_cur.max()

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (polishing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # a small number of iterations with decaying step‑size
        a = best_seq.copy()
        L_fft = _next_pow2(2 * n - 1)
        eta = 0.5               # initial step size
        it = 0
        while time.time() < deadline - 1.0:
            it += 1
            conv = _conv_self(a, L_fft)
            max_val = conv.max()
            cur_obj = 2.0 * n * max_val
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] PGD improvement {best_score:.6f}")

            subgrad = _subgradient_maxconv(a, conv)
            grad = 2.0 * n * subgrad
            # decaying step size
            eta_t = eta / (1.0 + it * 1e-4)
            a_candidate = a - eta_t * grad
            a_candidate = _project_simplex(a_candidate)

            new_obj = 2.0 * n * _conv_self(a_candidate, L_fft).max()
            if new_obj < cur_obj:
                a = a_candidate
                # keep current step size
            else:
                # shrink step size when we cannot improve
                eta *= 0.995
                if eta < 1e-8:
                    break

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Final exact evaluation (for logging)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
