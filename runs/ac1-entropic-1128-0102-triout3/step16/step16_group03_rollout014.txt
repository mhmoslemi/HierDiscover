def _project_simplex(v):
    # V is np.ndarray, returns vector with nonnegative entries sum=1.
    # Implementation based on Duchi et al.
    # Ensure v is float.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        # all entries non-positive, set to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:

        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimizer for the step‑function evaluation problem (improved version).

Main ideas
~~~~~~~~~~
1️⃣  Quick exploration of the 2‑parameter Beta family (edge‑heavy shapes) on a fine
    grid – these distributions are known to give very low autocorrelation peaks.

2️⃣  Smooth‑gradient descent on the simplex.  We minimise a *soft‑max* of the
    convolution (log‑sum‑exp) which is a smooth approximation of the true
    objective  maxₖ (a*a)[k].  The gradient is obtained by a cheap FFT‑based
    cross‑correlation.  After each update the point is projected onto the
    probability simplex (additive mirror descent with Euclidean projection).

    The smoothing parameter α is increased geometrically during the run so that
    the optimisation first moves globally (small α) and later focuses on the
    true max (large α).

3️⃣  A deterministic hill‑climbing phase that moves a tiny amount of mass from
    the largest entry to the smallest (plus occasional random moves) and accepts
    uphill steps with a small temperature.

4️⃣  Final fine mixing with the uniform distribution using a golden‑section search.

All phases respect the global time budget (default 1000 s).  The best sequence
found is returned as a list of non‑negative floats that sum to one, ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)) – the quantity minimised by the optimiser."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Normalised Beta(α,β) density on n discrete points (mid‑point rule)."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]                      # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1                     # last true index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray | None, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf

# ----------------------------------------------------------------------
# 1️⃣  Systematic Beta‑shape search
# ----------------------------------------------------------------------
def _beta_grid_search(n: int,
                      best_seq: np.ndarray,
                      best_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Exhaustive search on a log‑grid of (α,β) values."""
    alphas = np.geomspace(0.05, 10.0, 40)   # 40 values spanning several decades
    betas = alphas.copy()
    rng = np.random.default_rng()
    # symmetric grid
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                return best_seq, best_score
            v = _beta_shape(a, b, n)
            if v is None:
                continue
            sc = _objective_score(v, n)
            if sc < best_score:
                best_score, best_seq = sc, v.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Beta(α={a:.3g},β={b:.3g}) → score={sc:.6f}")

    # a few random Beta draws to fill the space
    for _ in range(200):
        if time.time() >= deadline:
            break
        a = 10 ** rng.uniform(-2, 2)   # log‑uniform between 0.01 and 100
        b = 10 ** rng.uniform(-2, 2)
        v = _beta_shape(a, b, n)
        if v is None:
            continue
        sc = _objective_score(v, n)
        if sc < best_score:
            best_score, best_seq = sc, v.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Random Beta(α={a:.3g},β={b:.3g}) → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_grad_descent(a0: np.ndarray,
                         n: int,
                         L: int,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smoothed version of max(conv) using an additive update
    followed by Euclidean projection onto the simplex.
    """
    a = a0.copy()
    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0
    step0 = 0.5               # initial step size
    alpha = 5.0               # initial smoothing parameter
    max_alpha = 200.0
    eval_every = 10           # compute true objective only periodically

    # pre‑compute index array for the gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    while time.time() < deadline:
        it += 1
        # ----- Fourier transforms -----
        A_fft = np.fft.rfft(a, n=L)

        # ----- full self‑convolution (for the soft‑max) -----
        conv_full = np.fft.irfft(A_fft * A_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- soft‑max weights -----
        max_val = conv.max()
        # numerical safety: shift by max_val
        exp_w = np.exp(alpha * (conv - max_val))
        w = exp_w / exp_w.sum()                     # w is a probability vector

        # ----- gradient = 2·(a ⋆ w) (cross‑correlation) -----
        w_rev = w[::-1]
        W_rev_fft = np.fft.rfft(w_rev, n=L)
        corr_full = np.fft.irfft(A_fft * W_rev_fft, n=L)
        grad = 2.0 * corr_full[idx]

        # ----- additive mirror descent step -----
        step = step0 / math.sqrt(it)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # ----- occasional evaluation of the true objective -----
        if it % eval_every == 0:
            # evaluate score of the *new* point
            conv_new = _conv_self(a_new, _next_pow2(2 * n - 1))
            cur_score = float(2.0 * n * conv_new.max())
            if cur_score < best_score:
                best_score, best_seq = cur_score, a_new.copy()
                last_improve_it = it
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Smooth‑grad it={it:,} → score={cur_score:.6f}")

        # ----- stagnation handling -----
        if it - last_improve_it > 200_000:
            # random perturbation to escape flat plateaux
            perturb = rng.normal(scale=0.02, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            a = a / a.sum()
            last_improve_it = it
        else:
            a = a_new

        # ----- slowly increase the smoothing parameter -----
        if it % 5_000 == 0:
            alpha = min(alpha * 2.0, max_alpha)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """Small mass‑transfer moves with occasional uphill acceptance."""
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # donor
        if rng.random() < 0.7:
            i = np.argmax(a)           # largest entry
        else:
            i = rng.integers(n)        # random
        # receiver
        if rng.random() < 0.7:
            j = np.argmin(a)           # smallest entry
        else:
            j = rng.integers(n)

        if i == j or a[i] <= 1e-14:
            continue

        eps = min(0.02 * a[i], 1e-4)    # move at most 2 % of donor mass
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Hill‑climb improved to {new_score:.6f}")
        else:
            # annealing acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """Optimise λ∈[0,1] for λ·a + (1−λ)·uniform via a few golden‑section steps."""
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq /= seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the runner will kill the process after `budget_s` seconds
    deadline = start_time + budget_s - 0.25

    # Problem size – 4096 is a good trade‑off between quality and speed
    n = 4096
    L_full = _next_pow2(3 * n - 2)            # for the smooth‑gradient phase
    L_conv = _next_pow2(2 * n - 1)            # for exact convolutions elsewhere

    # ---- 0️⃣ Initialise with uniform & optional global starter ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ---- 1️⃣ Beta‑shape grid search (≈10 % of the budget) ----
    phase1_deadline = start_time + 0.10 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _beta_grid_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After Beta grid best score≈{best_score:.6f}")

    # ---- 2️⃣ Smooth‑gradient descent (≈60 % of the budget) ----
    phase2_deadline = start_time + 0.70 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑gradient descent")
        best_seq, best_score = _smooth_grad_descent(
            best_seq, n, L_full, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth‑grad best score≈{best_score:.6f}")

    # ---- 3️⃣ Hill‑climbing refinements (≈20 % of the budget) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, L_conv, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ---- 4️⃣ Fine mixing with uniform (remaining time) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ---- Final normalisation and return ----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
