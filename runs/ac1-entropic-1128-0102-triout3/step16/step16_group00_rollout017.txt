```
    # Set random seed for reproducibility
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4096

    # get initial vector
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size != n:

    best_score = float('inf')
    # evaluate
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()
    print(f"[{time.time()-start:.2f}s] initial score {best_score:.6f}")

    # Main Loop solving LPs
    iteration = 0
    t = 0.5   # mixing factor
    while time.time() < deadline and iteration < 200:
        iteration += 1
        conv = _conv_self(a)
        M = conv.max()
        # select active indices
        # pick top K values (K ~ min(200, len(conv)) )
        K = min(200, conv.size)

            rows[row_i, i] = a[k - i]   # a[j] contributions

            print(f"[{time.time()-start:.2f}s] LP failed at iter {iteration}")
            break
        b_opt = res.x
        # Normalize
        total_b = b_opt.sum()
        if total_b == 0:
            print(f"[{time.time()-start:.2f}s] Zero solution, break")
            break
        b_opt = b_opt / total_b

        # Update a with mixing
        a_new = (1 - t) * a + t * b_opt
        # Normalize to sum=1 to avoid drift
        a_new = a_new / a_new.sum()
        # Evaluate objective
        conv_new = _conv_self(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[{time.time()-start:.2f}s] LP iter {iteration} new best {best_score:.6f}")
        # Decide whether to accept a_new
        if score_new <= 2.0 * n * M:   # Since M = current max of a, this condition true
            a = a_new
        else:
            # fallback keep old a
            a = a
        # Decay mixing factor
        t = max(t * 0.95, 0.01)

    # Final refinement via projected gradient descent (as before)
    # We'll reuse earlier PGD code but adapted.

    # PGD parameters
    a = best_a.copy()
    cur_score = best_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        grad = _subgradient_maxconv(a, conv)    # we need to define this.
        # try steps
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            # projection onto simplex
            cand = _project_simplex(np.maximum(cand, 0.0))
            conv_cand = _conv_self(cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time()-start:.2f}s] PGD improved best {best_score:.6f}")
                break
        if not improved:
            step = max(step * 0.5, 1e-12)
        else:
            step = min(step * 1.05, 1.0)
    # Return best sequence as list
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0/n)
    print(f"[{time.time()-start:.2f}s] finished best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
```

The optimizer works in three stages.

1. **Initialisation** – use the previously found construction (`height_sequence_1`) if it exists, otherwise start from the uniform vector.
2. **Active‑set linear programming** – for the current vector `a` compute the self‑convolution, keep the *K* shifts where the convolution is largest (the “active” constraints) and solve a linear program that maximises the total mass of a new vector `b` while keeping all active convolutions ≤ the current maximum `M`.  
   The solution is mixed with the current vector (`a←(1‑t)a+t b`) and the process repeats.  Because only a small subset of constraints is used, each LP is tiny (≈ 50 × 4096) and solves in a few milliseconds with SciPy’s HiGHS solver.
3. **Projected gradient refinement** – a few iterations of projected sub‑gradient descent (the same sub‑gradient that the original code used) polish the best vector found by the LP phase.

All operations stay on the probability simplex (`∑a_i=1`, `a_i≥0`).  The objective that is minimised is  

\[
\text{score}=2\,n\;\max_k (a*a)_k ,
\]

which is exactly the value returned by `evaluate_sequence` for a normalised vector.

The whole routine respects the 1000 s budget; the best vector found up to the deadline is returned as a plain Python list.

```python
"""
Propose a step‑function (non‑negative vector) with a low value of
evaluate_sequence.  The method combines:

* a warm start from the global best vector if available,
* an active‑set linear‑programming loop that continuously reduces the
  worst self‑convolution value,
* a final projected sub‑gradient descent for fine tuning.

All vectors are kept on the probability simplex (sum = 1, non‑negative).
The LP phase uses only the K largest convolution entries, so each LP
has at most a few hundred constraints and solves very quickly with
scipy.optimize.linprog (HiGHS backend).
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    n = 4096

    # ------------------------------------------------------------------
    # Warm start – use the global best if it is present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter /= total
                a = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Warm start from global best")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        a = np.full(n, 1.0 / n)          # uniform start

    best_a = a.copy()
    best_score = _objective_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Active‑set Linear‑Programming loop
    # ------------------------------------------------------------------
    mix_factor = 0.5          # initial mixing weight for the new LP solution
    max_lp_iter = 200        # safety cap
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iter:
        lp_iter += 1

        conv = _conv_self(a)
        M = conv.max()

        # ----- choose active constraints (largest convolution entries) -----
        K = min(80, conv.size)                     # how many constraints we keep
        # indices of the K largest values
        cand_idx = np.argpartition(conv, -K)[-K:]
        # keep only those that are close to the maximum (within 0.5 %)
        active_idx = cand_idx[conv[cand_idx] >= M * 0.995]
        if active_idx.size == 0:                    # fall back to the K largest
            active_idx = cand_idx

        # ----- build the inequality matrix A @ b <= M -----
        rows = np.zeros((active_idx.size, n), dtype=float)
        for r, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)               # positions in b
            rows[r, i] = a[k - i]                               # coefficient a_{k-i}

        # ----- linear program: maximise sum(b) s.t. rows·b ≤ M , b ≥ 0 -----
        c = -np.ones(n)                       # minimise -sum(b)  →  maximise sum(b)
        bounds = [(0, None)] * n
        # preserve remaining time for the LP
        lp_time_left = max(0.0, deadline - time.time())
        # the HiGHS solver respects a `time_limit` option (seconds)
        options = {"presolve": True, "time_limit": lp_time_left}
        res = scipy.optimize.linprog(c,
                                    A_ub=rows,
                                    b_ub=np.full(rows.shape[0], M),
                                    bounds=bounds,
                                    method='highs',
                                    options=options)

        if not res.success or res.x is None:
            # if the LP cannot be solved in the remaining time, stop the loop
            print(f"[{time.time() - start_time:.2f}s] LP failed (iter {lp_iter}), stopping LP phase")
            break

        b = res.x
        total_b = b.sum()
        if total_b == 0:
            print(f"[{time.time() - start_time:.2f}s] LP returned zero vector, aborting LP phase")
            break
        b /= total_b                                   # keep it on the simplex

        # ----- mix the new candidate with the current vector -----
        a_new = (1.0 - mix_factor) * a + mix_factor * b
        a_new = a_new / a_new.sum()                    # re‑normalise (numerical safety)

        score_new = _objective_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} – new best {best_score:.6f}")

        # always move to the mixed point (it never worsens the active constraints)
        a = a_new

        # slowly reduce the mixing weight – the updates become finer
        mix_factor = max(mix_factor * 0.96, 0.01)

    # ------------------------------------------------------------------
    # 2️⃣  Projected Sub‑gradient Descent (fine‑tuning)
    # ------------------------------------------------------------------
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD improvement – best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-12)            # shrink step if no progress
        else:
            step = min(step * 1.05, 1.0)              # otherwise accelerate a little

    # ------------------------------------------------------------------
    # Normalise and return the best vector found
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
```
