```
def project_simplex(v):
    # ensure non-neg, sum to 1
    # returns projection onto simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _simulated_annealing(a0, n, time_deadline):
    a = a0.copy()
    conv = _conv_self(a, _next_pow2(2*n-1))
    cur_max = conv.max()
    cur_score = 2.0*n*cur_max
    best_a = a.copy()
    best_score = cur_score
    start = time.time()
    while time.time() < time_deadline:
        # choose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

"""
Hybrid search for the step‑function minimisation task.

The algorithm proceeds in three clear phases:

1️⃣  Broad random sampling – many Dirichlet draws with a log‑uniform
    concentration produce a short list of the best few candidates.

2️⃣  Local refinement – a fast simulated‑annealing / mass‑transfer hill‑climb.
    A move transfers a tiny amount ε of probability mass from a “heavy”
    index (large sub‑gradient) to a “light” index, updates the full
    convolution by O(n) slice arithmetic (no FFT inside the inner loop)
    and accepts the move with a temperature‑scaled Metropolis rule.
    The best candidate seen during the allotted time is retained.

3️⃣  Fine mixing – golden‑section search of the convex combination
    λ·a + (1‑λ)·uniform to exploit the remaining improvement potential.

All phases respect the global time budget (default 1000 s).  The final
sequence is normalised to sum 1, clipped to non‑negative values and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv(a,a)).  Input a is already normalised (sum=1)."""
    L_fft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Classic O(n log n) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣  Broad random search – keep the best K candidates
# ----------------------------------------------------------------------
def _broad_random_search(n: int,
                         best_seq: np.ndarray,
                         best_score: float,
                         K: int,
                         deadline: float,
                         start_time: float):
    rng = np.random.default_rng()
    top = [(best_score, best_seq.copy())]  # list of (score, sequence)
    while time.time() < deadline:
        # log‑uniform concentration between 0.01 and 30
        log_c = rng.uniform(np.log(0.01), np.log(30.0))
        c = math.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best {sc:.6f}")

        # keep a short list of the best K candidates
        if len(top) < K:
            top.append((sc, a.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if sc < top[-1][0]:
                top[-1] = (sc, a.copy())
                top.sort(key=lambda x: x[0])

    return best_seq, best_score, top

# 2️⃣  Simulated‑annealing / mass‑transfer hill climbing
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float):
    """
    Perform many tiny mass‑transfer moves a[i]←a[i]‑ε, a[j]←a[j]+ε.
    Convolution is updated by O(n) slice arithmetic, therefore
    many thousands of iterations fit easily into the time budget.
    """
    a = a0.copy()
    L_fft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L_fft)                 # full convolution (length 2n‑1)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a, best_score = a.copy(), cur_score

    rng = np.random.default_rng()
    it = 0
    # temperature scaling – start a bit higher, then decay linearly
    T0 = 1e-4

    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # choose donor (probability ∝ a) and receiver (uniform)
        # --------------------------------------------------------------
        if np.all(a == 0):
            break
        donor = rng.choice(n, p=a)                     # more mass → more likely to donate
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # --------------------------------------------------------------
        # amount to move – a modest fraction of the donor mass
        # --------------------------------------------------------------
        max_move = 0.02 * a[donor]                      # at most 2 % of the donor
        eps = rng.random() * max_move
        if eps < 1e-12:
            continue

        # --------------------------------------------------------------
        # fast convolution update (linear + quadratic terms)
        # --------------------------------------------------------------
        conv_new = conv.copy()
        # linear part: 2*ε*(shift_receiver - shift_donor)
        conv_new[donor:donor + n] -= 2.0 * eps * a
        conv_new[receiver:receiver + n] += 2.0 * eps * a
        # quadratic part (ε² terms)
        conv_new[2 * donor] += eps * eps
        conv_new[2 * receiver] += eps * eps
        conv_new[donor + receiver] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)               # linear cooling
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            # update a and conv permanently
            a[donor] -= eps
            a[receiver] += eps
            conv = conv_new
            cur_score, cur_max = new_score, new_max
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved to {best_score:.6f}")

        # --------------------------------------------------------------
        # occasional random jitter to shake out of plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.001:
            pert = rng.normal(scale=0.01, size=n) * a
            a = np.maximum(a + pert, 0.0)
            a /= a.sum()
            conv = _conv_self(a, L_fft)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA jitter → {best_score:.6f}")

    return best_a, best_score

# 4️⃣  Fine mixing with the uniform distribution (golden‑section)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform via golden‑section."""
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    # keep the known ends implicit – best_seq already stores the best known point
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = np.maximum(best_seq, 0.0)
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → {sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = np.maximum(best_seq, 0.0)
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → {sc2:.6f}")

        if hi - lo < 1e-5:
            break
