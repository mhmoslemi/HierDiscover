"""
Search for a non‑negative step function that minimises the bound
`evaluate_sequence`.  The pipeline is:

1.  A lightweight sweep over several analytic families (geometric,
    power, beta, stretched‑exp, logistic, sine‑power).  This gives a
    decent starting point very quickly.
2.  A large number of Dirichlet draws with many concentration values.
3.  Adam optimisation of a smooth‑max surrogate of the self‑convolution
    (τ‑soft‑max).  The optimiser works on the probability simplex.
4.  Projected sub‑gradient descent on the exact max‑convolution.
5.  A long simulated‑annealing phase that repeatedly moves a random
    fraction of mass from a source index to a destination index and
    accepts the move with a temperature schedule.  This stage is good
    at escaping the plateaus left by the previous gradient stages.
6.  A short hill‑climb / simulated‑annealing refinement (identical to the
    reference implementation).

All stages respect the global 1000 s budget; the best sequence seen so
far is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:

    # number of attempts is limited only by remaining time
    while time.time() < deadline:

        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:

        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (as in the reference)
        a_rev = a[::-1]
        Lg = _next_pow2(3 * n - 2)
        conv_w = np.fft.irfft(np.fft.rfft(w, n=Lg) *
                              np.fft.rfft(a_rev, n=Lg),
                              n=Lg)[:n]
        grad = 2.0 * conv_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score

def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.5          # initial step size
    it = 0
    cur_score = _objective_score(a)
    no_improve = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad
        eta_t = eta / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.995
            if no_improve % 500 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-8:
            break
    return best_seq, best_score

# 5️⃣  Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    rng = np.random.default_rng()
    # temperature schedule: start warm, cool exponentially to a tiny floor
    T0 = 0.04
    T_min = 1e-6
    anneal_start = time.time()
    while time.time() < deadline:
        # exponential decay of temperature over the annealing stage
        elapsed = time.time() - anneal_start
        total = max(1e-12, deadline - anneal_start)
        fraction = elapsed / total
        T = T0 * (T_min / T0) ** fraction

        # source index drawn proportional to current mass (big masses move more)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue

        # the scale of the transferred mass shrinks with the temperature
        max_frac = 0.5 * (T / T0) + 0.001   # between ~0.001 and ~0.5
        eps_frac = rng.uniform(0.001, max_frac)
        eps = a[src] * eps_frac
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        new_score = _objective_score(a_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal new best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing (as in reference)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (the best sequence found).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialisation (uniform or external starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = min(deadline, start_time + 150.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more trials)
    # --------------------------------------------------------------
    phase_deadline = min(deadline, phase_deadline + 150.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation of smooth‑max surrogate
    # --------------------------------------------------------------
    phase_deadline = min(deadline, phase_deadline + 250.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent on exact max‑convolution
    # --------------------------------------------------------------
    phase_deadline = min(deadline, phase_deadline + 150.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD descent")
        best_seq, best_score = _pgd_descent(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Simulated annealing (large‑move mass transfers)
    # --------------------------------------------------------------
    phase_deadline = min(deadline, phase_deadline + 150.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After annealing best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final hill‑climb / simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
