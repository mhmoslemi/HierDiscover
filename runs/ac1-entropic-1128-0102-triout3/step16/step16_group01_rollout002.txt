"""
A fast‐mix of random starts, an expanded parametric sweep (geometric×power + Beta
pdf shapes), momentum‑accelerated smooth‑max gradient descent, exponentiated‑gradient
on the true L∞ objective and a final hill‑climb.  The search works on the
probability simplex (∑a_i=1) – the evaluation function is scale‑invariant
so normalising to a simplex is safe.  The code respects the 1000 s budget and
keeps the best‑found vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective (exact) – assumes sum≈1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape: a_i ∝ (i+1)^{alpha‑1}·(n-i)^{beta‑1},
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    j = n - np.arange(n, dtype=np.float64) + 1.0      # n … 1
    # guard against negative exponents producing NaNs for zero entries
    shape = np.exp((alpha - 1.0) * np.log(i) + (beta_ - 1.0) * np.log(j))
    # replace any NaNs (possible for 0^neg) with 0
    shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
    total = shape.sum()
    if total == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """p_k = exp(beta*conv_k) / Σexp(...). Numerically stable."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k (smooth‑max objective):
        g_i = 2·Σ_j a_j·p_{i+j}
    Computed via FFT as a cross‑correlation.
    """
    n = a.shape[0]
    # reverse p for correlation
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # The correlation length is n + (2n‑1)‑1 = 3n‑2.
    # Desired slice is indices (n‑1) … (2n‑2) inclusive (length n).
    start = n - 1
    g = corr[start:start + n][::-1]        # reverse to obtain correct ordering
    return 2.0 * g


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes diagonal terms correctly (2·a_i)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (multiplicative updates)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) updates on the true L∞‑objective.
    a ← a·exp(-η·g), then re‑project onto the simplex.
    The step‑size η decays as 1/√t for smoother convergence.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.9                     # a slightly larger initial step
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # η decays like 1/√it (log‑schedule works well in practice)
        eta = eta0 / (np.sqrt(it) + 1.0)

        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        # occasional jitter (helps escape flat plateaux)
        if it % 4000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with back‑tracking line‑search)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # simple diminishing step size
        step = 0.35 / (np.sqrt(it) + 1.0)

        # back‑tracking: try up to 8 halvings
        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # allow occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        # jitter to keep dynamics alive
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

def _smoothmax_phase(a0: np.ndarray,
                     L_fft: int,
                     L_grad: int,
                     betas: list[float],
                     deadline: float,
                     start_time: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimises the smooth‑max surrogate  f_β(a)= Σ_k p_k·conv_k, where
    p_k ∝ exp(β·conv_k).  For each β we run a Nesterov‑accelerated gradient
    descent (projected onto the simplex).  β values are increased
    geometrically, ending at a large β so that the surrogate approximates the
    true max.
    """
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)

    for beta in betas:
        # allocate a modest slice of the remaining budget to each β
        per_beta = min(45.0, (deadline - time.time()) / max(len(betas) - betas.index(beta), 1))
        stage_deadline = time.time() + per_beta
        v = np.zeros_like(a)               # momentum buffer
        mu = 0.93                           # momentum coefficient
        it = 0
        while time.time() < stage_deadline:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta)
            g = _grad_softmax(a, p, L_grad)

            # step length ∝ 1/√it, normalised by gradient norm
            step_len = 0.40 / (np.sqrt(it) + 1.0)
            step = step_len / (np.linalg.norm(g) + 1e-12)

            # Nesterov‑style update
            v = mu * v - step * g
            a_candidate = a + v
            a_candidate = _project_simplex(a_candidate)

            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.0f} it={it} "
                      f"score={cand_score:.6f}")

            # accept if we improve the true objective; otherwise probabilistic uphill
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
            else:
                prob = np.exp(-(cand_score - cur_score) * 4.0 / (np.sqrt(it) + 1.0))
                if rng.random() < prob:
                    a = a_candidate
                    cur_score = cand_score

            # occasional jitter
            if it % 5000 == 0:
                eps = 1e-4 * rng.random(a.shape[0]) * a
                a = _project_simplex(a + eps)

    return best_seq, best_score

    # ------------------------------------------------------------------
    # 4️⃣ Projected sub‑gradient descent with back‑tracking (≈160 s).
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_deadline, start_time,
                                                    best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Final random hill‑climb (mass transfer moves) – polish remaining time.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.4 % of source mass (slightly larger than before)
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_sc = _eval_fast(prop, L_fft)
        if new_sc < best_score:
            best_seq = prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # optional sanity check using the provided evaluation function
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Final evaluate_sequence result = {final_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
