"""
propose_candidate

Search for a non‑negative step function (probability vector) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²).

The algorithm proceeds in several fast, complementary stages:

1.  Baseline and optional global starter.
2.  Exhaustive sweep of analytic families.
    – Uniform, geometric, symmetric geometric (U‑shape), power,
      beta (with very small α,β), stretched‑exponential,
      logistic, sine‑power, cosine‑power.
3.  Random initialisers (Dirichlet, log‑normal, per‑coordinate Gamma).
4.  Two complementary non‑smooth optimisers:
    • Accelerated (Nesterov) projected sub‑gradient descent.
    • Exponentiated‑gradient sub‑gradient descent (mirror descent on the simplex).
5.  Final simulated‑annealing hill‑climb.

All stages respect the global 1000 s budget and keep the best
candidate seen.  The returned list is normalised to sum to 1 and
contains only non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For each active shift k (where conv attains its maximum) we add the
    shifted copy of a to the gradient and finally average.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) for a sequence with sum = 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# 1️⃣  Analytic families

def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # ----------------------- uniform -----------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ----------------------- geometric -----------------------
    eps_grid = np.logspace(-5, -0.2, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ----------------------- symmetric geometric (U‑shape) -----------------------
    r_grid = np.logspace(-3, -0.001, 80)  # r close to 0 → very spiked ends
    for r in r_grid:
        if time.time() >= deadline:
            break
        geo_left = np.power(r, idx)                 # high at left
        geo_right = np.power(r, n - 1 - idx)        # high at right (mirror)
        sym_geo = geo_left + geo_right
        s = sym_geo.sum()
        if s == 0:
            continue
        sym_geo /= s
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * sym_geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SymGeo r={r:.3e} lam={lam:.2f} score={sc:.6f}")

    # ----------------------- power law -----------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ----------------------- reversed power (U‑shape) -----------------------
    for p in np.linspace(0.0, 2.0, 21):
        if time.time() >= deadline:
            break
        rev_pow = np.power(n - idx, p)
        s = rev_pow.sum()
        if s == 0:
            continue
        rev_pow /= s
        sc = _objective_score(rev_pow)
        if sc < best_score:
            best_score, best_seq = sc, rev_pow.copy()
            print(f"[{time.time() - start_time:.2f}s] RevPower p={p:.2f} score={sc:.6f}")

    # ----------------------- beta‑type (extended range) -----------------------
    alphas = np.concatenate((
        np.logspace(np.log10(0.02), np.log10(0.2), 12),
        np.linspace(0.25, 4.0, 38)
    ))
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}")

    # ----------------------- stretched‑exponential -----------------------
    scales = np.linspace(0.15, 1.0, 9)
    p_vals = np.linspace(0.2, 6.0, 45)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    # ----------------------- logistic -----------------------
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.01, 0.25, 13)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ----------------------- sine‑power (central hump) -----------------------
    p_vals = np.linspace(0.1, 6.0, 45)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    # ----------------------- cosine‑power (U‑shape) -----------------------
    p_vals = np.linspace(0.2, 8.0, 41)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Cos‑power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_initialisations(n: int,
                            best_seq: np.ndarray,
                            best_score: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        method = rng.choice(['dirichlet', 'lognorm', 'percoord_gamma'])
        if method == 'dirichlet':
            # concentration spanning several orders of magnitude
            log_c = rng.uniform(np.log(1e-4), np.log(1e2))
            c = np.exp(log_c)
            a = rng.gamma(shape=c, scale=1.0, size=n)
        elif method == 'lognorm':
            mu = rng.normal(0.0, 0.5)
            sigma = rng.exponential(0.5) + 0.1
            a = np.exp(rng.normal(mu, sigma, size=n))
        else:  # per‑coordinate Gamma (asymmetric Dirichlet)
            alphas = rng.uniform(0.02, 0.5, size=n)
            a = rng.gamma(shape=alphas, scale=1.0, size=n)

        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] {method} init score={sc:.6f}")
    return best_seq, best_score

# 3️⃣  Accelerated projected sub‑gradient descent (Nesterov)
# ----------------------------------------------------------------------
def _accelerated_subgradient(a0: np.ndarray,
                             start_time: float,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    y = a.copy()
    t = 1.0
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    it = 0
    eta0 = 0.6  # initial step‑size factor
    while time.time() < deadline:
        it += 1
        conv = _conv_self(y, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, y.copy()
            print(f"[{time.time() - start_time:.2f}s] Accel it={it:,} score={cur_score:.6f}")

        subg = _subgradient_maxconv(y, conv)
        grad = 2.0 * n * subg

        # diminishing step size ~ 1/√k
        eta = eta0 / np.sqrt(it)
        a_next = y - eta * grad
        a_next = _project_simplex(a_next)

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_next + ((t - 1.0) / t_next) * (a_next - a)

        a = a_next
        t = t_next

        # occasional small random shake to escape plateaus
        if it % 800 == 0:
            jitter = np.random.default_rng().normal(scale=0.001, size=n)
            y = np.maximum(y + jitter, 0.0)
            y = _project_simplex(y)

    return best_seq, best_score

# 4️⃣  Exponentiated‑gradient sub‑gradient descent (mirror descent)
# ----------------------------------------------------------------------
def _expgrad_subgradient(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    it = 0
    eta0 = 0.7
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad it={it:,} score={cur_score:.6f}")

        subg = _subgradient_maxconv(a, conv)
        # scale the gradient to keep updates moderate
        scale = np.max(np.abs(subg)) + 1e-12
        eta = eta0 / (np.sqrt(it) * scale)

        factor = np.exp(-eta * subg)  # element‑wise
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # tiny random perturbations every few thousand steps
        if it % 1500 == 0:
            jitter = np.random.default_rng().normal(scale=5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum

    return best_seq, best_score

# 5️⃣  Final hill‑climb / simulated annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    T0 = 0.025
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Analytic families (incl. new U‑shape families)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0  # leave ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random initialisers (Dirichlet, log‑normal, per‑coord Gamma)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_initialisations(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Accelerated projected sub‑gradient descent
    # --------------------------------------------------------------
    accel_deadline = deadline - 120.0
    if time.time() < accel_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting accelerated PGD")
        best_seq, best_score = _accelerated_subgradient(
            best_seq, start_time, accel_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After accel best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient sub‑gradient descent
    # --------------------------------------------------------------
    exp_deadline = deadline - 60.0
    if time.time() < exp_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting exp‑grad descent")
        best_seq, best_score = _expgrad_subgradient(
            best_seq, start_time, exp_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After exp‑grad best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
