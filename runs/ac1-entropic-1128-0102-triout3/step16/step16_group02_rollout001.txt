"""
A smooth‑max (log‑sum‑exp) surrogate for the max‑convolution objective and
Adam optimisation on the probability simplex.

The algorithm works with a fixed length `n` (4096 by default) and repeatedly
optimises the surrogate
    S_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
which converges to the true max as α → ∞.  Starting with a modest α we obtain a
smooth landscape, then α is increased gradually so the optimiser focuses on the
actual peak of the self‑convolution.  After each Adam step the vector is
projected onto the simplex (non‑negative, sum = 1).  Random restarts and
occasional jitter help to escape local minima.  The best sequence encountered
within the time budget is returned as a plain list of floats ready for
``evaluate_sequence``.
"""

import time

# Low‑level utilities (FFT helpers)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) (seq must sum to 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):

        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    v = np.zeros_like(a)
    it = 0                         # iteration counter for the current restart
    total_it = 0                   # global iteration counter
    no_improve = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Compute the exact self‑convolution (needed for the true objective)
        # ------------------------------------------------------------------
        Af1 = np.fft.rfft(a, n=L1)
        conv_full = np.fft.irfft(Af1 * Af1, n=L1)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv

        # Update the global best if we improved the true objective
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter={total_it:,}  best_score={best_score:.6f}")
            if best_score <= 1.503:          # target reached – stop early
                break
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Smooth‑max surrogate (log‑sum‑exp) and its gradient
        # ------------------------------------------------------------------
        # grad of the smooth max w.r.t. a is
        #   2 * (a  *  w_rev)   where w is the softmax over conv entries
        #   w_k ∝ exp(α·conv_k)
        conv_shift = conv_full - max_conv               # shifted for stability
        exp_term = np.exp(alpha_start * conv_shift) if total_it == 0 else np.exp(alpha * conv_shift)
        w = exp_term / exp_term.sum()                  # shape (2n‑1,)

        # Reverse w for the convolution that yields the gradient
        w_rev = w[::-1]

        Af2 = np.fft.rfft(a, n=L2)
        Wf = np.fft.rfft(w_rev, n=L2)
        grad_conv = np.fft.irfft(Af2 * Wf, n=L2)

        # Extract the n relevant entries (see derivation in analysis)
        grad = 2.0 * grad_conv[2 * n - 2 - np.arange(n)]

        # ------------------------------------------------------------------
        # Adam update (descent direction = -gradient)
        # ------------------------------------------------------------------
        it += 1
        total_it += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Linear decay of the learning rate
        lr = lr0 / np.sqrt(it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step                                 # move opposite to grad

        # Project back onto the simplex (keeps non‑negativity and Σ=1)
        a = _simplex_projection(a_new)

        # ------------------------------------------------------------------
        # Gradually increase the smoothness parameter α (continuation)
        # ------------------------------------------------------------------
        # α is increased linearly over the first few thousand iterations;
        # after that it stays at α_end.
        if total_it < 3000:
            alpha = alpha_start + (alpha_end - alpha_start) * (total_it / 3000.0)
        else:
            alpha = alpha_end

        # ------------------------------------------------------------------
        # Occasionally jitter the point if we have not improved for a while
        # ------------------------------------------------------------------
        if no_improve >= no_improve_limit:
            a += np.random.normal(scale=jitter_scale, size=n)
            a = _simplex_projection(a)
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] jitter applied (iter={total_it})")

        # ------------------------------------------------------------------
        # Forced restart after many iterations – allows fresh exploration
        # ------------------------------------------------------------------
        if it >= restart_interval:
            a = _random_start()
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] restart #{total_it // restart_interval}")

    # ----------------------------------------------------------------------
    # Normalise the best vector (safety) and return as plain Python list
    # ----------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
