"""
Propose a step‑function (non‑negative vector) that minimizes

        score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages within the 1000 s budget:

1. **Warm‑start** – many Dirichlet draws (different concentrations) are
   sampled; the best‑scoring vector is kept as a starting point.
   If a global variable ``height_sequence_1`` is present it is also tried.

2. **Maximum‑gradient mirror descent** – at each iteration the exact index
   *k* where the convolution attains its maximum is located.
   The gradient of that max‑convolution is
       ∇ₖ = 2·a_{k‑i}   (zero outside the support).
   An exponentiated‑gradient (mirror‑descent on the simplex) step
   ``a ← a·exp(-η·∇ₖ)`` is taken, with a diminishing stepsize
   ``η = η₀/√t``.  Small random jitter is added occasionally to escape
   plateaus.

3. **Final refinement** – a short convex mixture sweep with the uniform
   distribution followed by a cheap random‑pair hill‑climb.

All operations are done with FFT‑based convolutions for speed.
The routine is time‑aware and prints intermediate scores; the returned
list contains non‑negative floats that (approximately) sum to 1, which the
evaluator will automatically re‑normalise.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _gradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the *k*‑th convolution entry (the one attaining the maximum)
    with respect to a.  For each i,
        ∂conv_k/∂a_i = 2·a_{k‑i}   (zero when the index is out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx_i = np.arange(i_start, i_end + 1)
    idx_j = k - idx_i                 # always in [0, n‑1]
    g[idx_i] = 2.0 * a[idx_j]
    return g


def _local_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Warm‑start with Dirichlet draws
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # log‑uniform concentration in a useful range
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _local_score(a, n, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑start c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Maximum‑gradient mirror descent
# ----------------------------------------------------------------------
def _max_gradient_descent(a0: np.ndarray,
                          n: int,
                          L_fft: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _local_score(a, n, L_fft)

    it = 0
    eta0 = 0.7                     # initial stepsize (can be tuned)

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {it:,}  score={cur_score:.6f}")

        # gradient of the *active* max entry
        g = _gradient_max(a, k_max)

        # classic 1/√t stepsize
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        factor = np.exp(-eta * g)
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional random jitter to avoid being stuck
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2, endpoint=False)
            delta = min(a[i], 0.01) * rng.random()
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final refinement (mix with uniform + hill‑climb)
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_score = _local_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # convex mixtures with the uniform distribution
    for lam in np.linspace(0.0, 1.0, 41):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _local_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → score={sc:.6f}")

    # cheap random‑pair hill‑climb
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(best_a[i], 0.005) * rng.random()
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        sc = _local_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score
