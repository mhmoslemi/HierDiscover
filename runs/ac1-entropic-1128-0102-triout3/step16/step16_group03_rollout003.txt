"""
Hybrid optimizer for the step‑function evaluation problem.

New ideas compared with the previous runner:
*   A smooth surrogate for the ``max`` in the objective using a soft‑max
    (log‑sum‑exp) with a schedule of increasing β.  This yields a
    differentiable objective whose gradient can be computed efficiently
    via FFT‑based correlation.
*   Additive updates followed by Euclidean projection onto the simplex
    (non‑negative, unit‑sum) instead of multiplicative mirror‑descent.
*   A short deterministic sweep of Beta‑shaped seeds (including bell‑shapes
    with α,β>1) before the smooth optimisation.
*   The later phases (hill‑climb and mixing with the uniform distribution)
    are retained from the earlier code.

The algorithm respects a 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv(a,a)).  a is already normalised (sum=1)."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    # make sure we are working with a float copy
    v = v.astype(float, copy=True)

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray | None, float]:
    """If a global variable ``height_sequence_1`` exists, use it as a starter."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """Golden‑section search for λ∈[0,1] in λ·a+(1‑λ)·uniform."""
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


def _hill_climb(a0: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer hill‑climb with simulated‑annealing acceptance."""
    a = a0.copy()
    cur_score = _objective_score(a, n)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # donor – either the biggest or a random index
        if rng.random() < 0.7:
            i = np.argmax(a)
        else:
            i = rng.integers(n)
        # receiver – either the smallest or a random index
        if rng.random() < 0.7:
            j = np.argmin(a)
        else:
            j = rng.integers(n)

        if i == j or a[i] <= 1e-14:
            continue

        eps = min(0.02 * a[i], 1e-4)            # move at most 2 % or 1e‑4
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = _objective_score(a_new, n)
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            # simulated‑annealing style uphill acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints
    progress and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25  # tiny safety margin

    # problem size – 4096 works well and matches earlier runs
    n = 4096
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for correlation w * a_rev

    # ------------------------------------------------------------------
    # 0️⃣  Initialise with uniform & optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic Beta‑shaped seeds (including bell shapes)
    # ------------------------------------------------------------------
    beta_params = [
        (0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
        (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3),
        (2.0, 2.0), (3.0, 3.0), (4.0, 4.0), (5.0, 5.0),   # bell shapes
        (0.5, 2.0), (2.0, 0.5),                         # skewed versions
    ]
    for a_par, b_par in beta_params:
        seq = _beta_shape(a_par, b_par, n)
        if seq is not None:
            sc = _objective_score(seq, n)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta({a_par},{b_par}) → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth gradient descent (soft‑max surrogate)
    # ------------------------------------------------------------------
    a = best_seq.copy()               # start point for the smooth optimisation
    beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000]
    total_iters = 0
    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        # step‑size schedule (decays like 1/√t)
        step0 = 0.5
        max_inner = 200_000            # generous upper bound for each β

        for it in range(1, max_inner + 1):
            if time.time() >= deadline:
                break
            total_iters += 1

            # ----- convolution a*a (needed for the soft‑max) -----
            Af = np.fft.rfft(a, n=L_conv)
            conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]
            max_c = conv.max()

            # ----- soft‑max weights w -----
            # stability: subtract max before exponentiation
            exp_terms = np.exp(beta * (conv - max_c))
            sumexp = exp_terms.sum()
            w = exp_terms / sumexp
            # smooth objective (for diagnostics only)
            smooth_obj = 2.0 * n * (max_c + math.log(sumexp) / beta)

            # ----- correlation w * a_rev -----
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_corr)
            ARf = np.fft.rfft(a_rev, n=L_corr)
            corr_full = np.fft.irfft(Wf * ARf, n=L_corr)
            start_idx = n - 1
            corr_slice = corr_full[start_idx:start_idx + n]        # length n

            # ----- gradient of the smooth surrogate -----
            grad = (4.0 * n / beta) * corr_slice

            # ----- additive step + projection onto the simplex -----
            eta = step0 / math.sqrt(it)          # 1/√t step‑size
            a_candidate = a - eta * grad
            a_candidate = _project_onto_simplex(a_candidate)

            # ----- occasional hard evaluation -----
            if it % 150 == 0:
                cand_score = _objective_score(a_candidate, n)
                if cand_score < best_score:
                    best_score, best_seq = cand_score, a_candidate.copy()
                    print(f"[{time.time() - start_time:.2f}s] β={beta} it={total_iters} → score={cand_score:.6f}")

            a = a_candidate

        # after each β we make a quick hard‑evaluation of the current a
        cur_score = _objective_score(a, n)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] End β={beta} → hard score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted hill‑climbing (simulated‑annealing style)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, n, L_conv,
                                           start_time, deadline,
                                           best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fine mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(best_seq, n,
                                                     deadline, start_time,
                                                     best_seq, best_score)

    # ------------------------------------------------------------------
    # 5️⃣  Final normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
