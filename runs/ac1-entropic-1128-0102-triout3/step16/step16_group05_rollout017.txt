"""
Propose a step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm (≈ 1000 s budget)

1. Choose a moderate problem size (n = 1024).  This keeps the linear‑program
   (LP) small enough to solve repeatedly while still allowing the objective
   to approach its theoretical lower bound ≈ 1.

2. Initialise a non‑uniform distribution.
   * If the global variable ``height_sequence_1`` exists we down‑sample it to
     length n and normalise it to a probability vector.
   * Otherwise we use a smooth “Hann” window (zero at the ends, bell shaped
     in the middle) and normalise it.

3. Iteratively improve the distribution by solving the linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k = 0 … 2n‑2
                    b_j ≥ 0

   where * denotes the linear self‑convolution,
   f is the current distribution and M = max(f * f).  
   The LP is linear in the unknown b, while the bound M depends on f.
   After solving we obtain a candidate b with total mass S = Σ b_j .
   Scaling b to unit sum (g = b / S) yields a new distribution with
   max‑convolution ≤ M / S², i.e. a strictly smaller objective whenever
   S > 1.  We replace f by g and repeat (typically a handful of iterations
   suffices).

   The constraint matrix A (size (2n‑1) × n) is extremely sparse – each row
   contains at most n entries – and is built with ``scipy.sparse`` on the
   fly.

4. After the LP phase we run a very light hill‑climb that repeatedly moves
   a tiny fraction of mass from the index with the largest gradient
   (computed from the active convolution constraints) to the index with the
   smallest gradient.  This finishes the optimisation and removes any tiny
   numerical artefacts.

5. The final vector is projected onto the probability simplex, converted to
   a plain Python list and returned.

All steps respect the 1000 s wall‑clock budget and print intermediate
scores for monitoring.
"""

import numpy as np
from numpy.fft import rfft, irfft
import scipy.sparse as sp
import scipy.optimize as opt

# Low‑level utilities (must be top‑level for the sandbox)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2n‑1) × n) such that
        (A @ b)[k] = (f * b)[k]
    for any non‑negative vector b.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # each row k corresponds to shift k
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # j runs over the non‑zero entries of f that contribute to this shift
        for j in range(lo, hi + 1):
            rows.append(k)          # row index
            cols.append(k - j)      # column index (b_{k‑j})
            data.append(f[j])       # coefficient = f_j
    return sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

        # New objective score
        new_score = _objective_score(g)
        print(f"[{time.time() - start_time:.2f}s] LP #{it}: sum_b={sum_b:.6f}, score={new_score:.6f}")

        if new_score < best_score:
            best_score = new_score
            best_f = g.copy()

        # Use the new vector for the next iteration
        f = g

        # Early stopping if we already beat the target
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # Final tiny hill‑climb (mass‑redistribution moves)
    # --------------------------------------------------------------
    def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
        """Very light local optimisation after the LP phase."""
        a = a.copy()
        best_a = a.copy()
        best_val = _objective_score(a)
        n_local = a.shape[0]
        no_improve = 0
        while time.time() < deadline:
            # compute sub‑gradient of max(conv) (as in the previous code)
            conv = _conv_self(a)
            max_val = conv.max()
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]
            grad = np.zeros_like(a)
            for k in active:
                lo = max(0, k - (n_local - 1))
                hi = min(n_local - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad[idx] += a[k - idx]
            if active.size:
                grad = 2.0 * grad / active.size

            # source = index with large gradient & positive mass,
            # destination = index with tiny gradient
            src_candidates = np.where(a > 1e-15)[0]
            if src_candidates.size == 0:
                break
            src = src_candidates[np.argmax(grad[src_candidates])]
            dst = np.argmin(grad)

            if src == dst:
                src, dst = rng.choice(n_local, size=2, replace=False)
                if a[src] < 1e-15:
                    continue

            eps = a[src] * (rng.random() * 0.01 + 1e-9)
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            cur_val = _objective_score(a_prop)
            if cur_val < best_val:
                a = a_prop
                best_a = a.copy()
                best_val = cur_val
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] Hill step → {best_val:.6f}")
            else:
                no_improve += 1
                if no_improve > 3000:
                    # random jitter to escape a plateau
                    i1, i2 = rng.choice(n_local, size=2, replace=False)
                    jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                    a[i1] -= jitter
                    a[i2] += jitter
                    a = _project_simplex(np.maximum(a, 0.0))
                    no_improve = 0
        return best_a, best_val

    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_f, best_score = _hill_climb(best_f, deadline)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0:
        best_f = best_f / total
    else:
        best_f = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_f]
