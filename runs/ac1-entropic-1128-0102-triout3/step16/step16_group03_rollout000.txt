"""
Hybrid random + simulated‑annealing optimizer.

The routine works in three phases (time‑controlled):
  1. Broad parametric random search – samples many Beta, exponential and
     power‑law shapes (very cheap to generate) and keeps the best.
  2. Simulated‑annealing (SA) on the full 4096‑dimensional simplex.
     Small random mass transfers are proposed; uphill moves are accepted
     with a decreasing temperature to escape local minima.
  3. Fine mixing with the uniform distribution – a golden‑section search
     over λ∈[0,1] for the convex combination λ·best + (1‑λ)·uniform.

All phases respect a global time‑budget (default 1000 s).  The final
sequence is normalised to sum 1, contains only non‑negative numbers and
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv(a,a))  (a already normalised, sum =1)."""
    L_fft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric shape generators
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _exp_shape(lam: float, n: int) -> np.ndarray:
    """Symmetric exponential shape: e^{-lam·i}+e^{-lam·(n‑1‑i)}."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s


def _power_shape(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape: (i+1)^{-p} + (n‑i)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    a = np.power(idx, -p) + np.power(n + 1 - idx, -p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s

# ----------------------------------------------------------------------
# 1️⃣  Broad parametric random search
# ----------------------------------------------------------------------
def _broad_parametric_search(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Sample many Beta / exponential / power‑law shapes and keep the best.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose a family with probabilities (Beta 0.5, Exp 0.25, Power 0.25)
        r = rng.random()
        if r < 0.5:  # Beta
            log_a = rng.uniform(np.log(0.02), np.log(20.0))
            log_b = rng.uniform(np.log(0.02), np.log(20.0))
            a = _beta_shape(np.exp(log_a), np.exp(log_b), n)
            if a is None:
                continue
        elif r < 0.75:  # Exponential
            lam = rng.uniform(0.0, 6.0)          # allow a slightly broader range
            a = _exp_shape(lam, n)
        else:  # Power‑law
            p = rng.uniform(0.0, 4.0)
            a = _power_shape(p, n)

        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Parametric sample → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Simulated‑annealing on the full simplex
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Classic SA: repeatedly transfer a small amount of mass from a donor
    index to a receiver index.  Accept improvements outright; otherwise
    accept with probability exp(-Δ/ T) where temperature T decays linearly.
    """
    a = a0.copy()
    cur_score = _objective_score(a, n)
    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0
    # initial temperature – a few percent of the score gives a reasonable
    # acceptance rate for uphill moves at the start.
    T0 = best_score * 0.02

    while time.time() < deadline:
        it += 1

        # ---- propose a move -------------------------------------------------
        # donate from a relatively large entry, receive at a smaller one
        if rng.random() < 0.7:
            donor = np.argmax(a)
        else:
            donor = rng.integers(n)

        if a[donor] <= 1e-14:
            continue

        if rng.random() < 0.7:
            receiver = np.argmin(a)
        else:
            receiver = rng.integers(n)

        if donor == receiver:
            continue

        # move a random fraction of donor's mass (max 5 % of donor)
        max_transfer = min(a[donor] * 0.05, 0.005)  # absolute cap to keep steps tiny
        delta = rng.uniform(0.0, max_transfer)

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # ---- evaluate -------------------------------------------------------
        new_score = _objective_score(a_new, n)

        # ---- accept / reject -------------------------------------------------
        accepted = False
        if new_score < cur_score:
            accepted = True
        else:
            # linear temperature schedule
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            if T > 0 and rng.random() < math.exp(-(new_score - cur_score) / T):
                accepted = True

        if accepted:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                last_improve_it = it
                print(f"[{time.time() - start_time:.2f}s] SA improved to {new_score:.6f}")

        # ---- restart if stuck for a long while -------------------------------
        if it - last_improve_it > 300_000:
            # perturb the current best a little and continue
            perturb = rng.normal(scale=0.03, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            cur_score = _objective_score(a, n)
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] SA perturbation after stall")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(
    a: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using golden‑section.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.5f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.5f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the runner will kill the process after budget_s seconds
    deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # problem size (the previous best used 4096 – a good compromise)
    # ------------------------------------------------------------------
    n = 4096

    # ---- start from uniform and possibly from a global starter -------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric random search (≈15 % of the time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _broad_parametric_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] End of parametric search – best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on the full simplex (≈70 % of time)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * (deadline - start_time) - 0.10 * (deadline - start_time)
    # keep ~10 % for the final mixing
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, n, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After SA – best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine mixing with the uniform distribution (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
