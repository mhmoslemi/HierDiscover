"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The method is a fast greedy “mass‑transfer”
heuristic:

* Start from an (almost) uniform distribution on the same length as the
  best known construction (`height_sequence_1`).  A tiny random
  perturbation breaks the perfect symmetry.
* Keep the self‑convolution `b = a * a` (length 2 n − 1) updated
  incrementally.  Moving an amount `δ` from position `i` to position `j`
  changes `b` by simple slice operations, so each iteration costs only
  O(n) vector work.
* In every step the index `k` where `b` attains its maximum is found.
  For that `k` the gradient of `b_k` w.r.t. the coefficients is
  `∂b_k/∂a_i = 2·a_{k‑i}` (zero when the index is out of range).  The
  algorithm moves a fraction of the coefficient with the largest gradient
  (the “heavy” side) to a coefficient with the smallest gradient (the
  “light” side).  This strictly reduces the current maximum unless the
  situation is already flat.
* Every few thousand iterations the whole convolution is recomputed by
  FFT to avoid drift from rounding errors, and the whole vector is
  renormalised to keep the sum exactly 1 (the objective is scale‑invariant).
* The best sequence seen during the allotted time is returned.

The routine uses only NumPy and the already‑imported ``evaluate_sequence``
function; no external files or network access are required.
"""

import time
import numpy as np

def _convolve_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (linear convolution) using a power‑of‑two FFT of given size."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * len(a) - 1]


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """Return a sequence of non‑negative heights with low evaluate_sequence score."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Problem size – use the length of the current best construction if
    #     available, otherwise fall back to a reasonable size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 2000  # fallback; any positive length works

    # ------------------------------------------------------------------
    # 2.  Initialise a near‑uniform distribution (sum = 1, scale‑invariant).
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    # tiny random noise to break perfect symmetry
    a += (np.random.rand(n) - 0.5) * 1e-6
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # ------------------------------------------------------------------
    # 3.  Prepare convolution (FFT size = next power of two >= 2n‑1).
    # ------------------------------------------------------------------
    fft_size = 1 << ((2 * n - 1).bit_length())
    b = _convolve_fft(a, fft_size)                     # b = a*a
    best_a = a.copy()
    best_score = 2.0 * n * b.max() / (a.sum() ** 2)

    start = time.time()
    deadline = start + budget_s - 0.5   # spare a little margin
    it = 0
    # parameters controlling the mass‑transfer step
    frac_move = 0.5            # fraction of the donor mass that is moved
    min_eps = 1e-12            # avoid zero moves
    renorm_interval = 2000     # how often we re‑normalise & recompute via FFT
    random_shake_interval = 3000

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 4.  Locate the current peak of the convolution.
        # ------------------------------------------------------------------
        k_peak = int(np.argmax(b))

        # ------------------------------------------------------------------
        # 5.  Compute the gradient weights w_i = a_{k_peak - i} (zero outside range).
        # ------------------------------------------------------------------
        idx = np.arange(n)
        j = k_peak - idx
        mask = (j >= 0) & (j < n)
        w = np.zeros_like(a)
        w[mask] = a[j[mask]]

        # ------------------------------------------------------------------
        # 6.  Choose donor (large weight) and receiver (small weight).
        # ------------------------------------------------------------------
        # donor must have some mass left to give away
        donor_mask = (a > min_eps)
        weighted = w * donor_mask
        if weighted.max() == 0.0:               # pathological – fall back to random
            i_donor = np.random.choice(np.where(a > min_eps)[0])
            i_recv = np.random.randint(n)
        else:
            i_donor = int(np.argmax(weighted))
            # receiver: any index with minimal weight (preferably different)
            i_recv_candidates = np.where(w == w.min())[0]
            i_recv = int(
                np.random.choice(
                    i_recv_candidates[i_recv_candidates != i_donor]
                    if i_recv_candidates.size > 1
                    else i_recv_candidates
                )
            )

        # ------------------------------------------------------------------
        # 7.  Determine transferred amount.
        # ------------------------------------------------------------------
        eps = a[i_donor] * frac_move
        if eps < min_eps:
            eps = min_eps

        # ------------------------------------------------------------------
        # 8.  Incrementally update the convolution.
        # ------------------------------------------------------------------
        a_old = a.copy()          # needed for the shift terms
        delta_donor = -eps
        delta_recv = eps

        # donor contribution
        b[i_donor : i_donor + n] += 2.0 * delta_donor * a_old
        b[2 * i_donor] += delta_donor * delta_donor

        # receiver contribution
        b[i_recv : i_recv + n] += 2.0 * delta_recv * a_old
        b[2 * i_recv] += delta_recv * delta_recv

        # cross term (i_donor + i_recv)
        b[i_donor + i_recv] += 2.0 * delta_donor * delta_recv

        # apply the mass transfer to a
        a[i_donor] += delta_donor
        a[i_recv] += delta_recv

        # ------------------------------------------------------------------
        # 9.  Periodic renormalisation to keep the sum exactly 1 and
        #     recompute the whole convolution (removes accumulated round‑off).
        # ------------------------------------------------------------------
        if (it % renorm_interval) == 0:

            if total != 0.0:

            b = _convolve_fft(a, fft_size)

        # ------------------------------------------------------------------
        # 10.  Evaluate current score (cheap, no extra convolution).
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * b.max() / (a.sum() ** 2)
        if cur_score < best_score:

                if delta > min_eps:
                    a[i1] -= delta
                    a[i2] += delta
            # recompute full convolution after the shake
            b = _convolve_fft(a, fft_size)

        it += 1

    # ----------------------------------------------------------------------
    # 12.  Final polishing – brute‑force scan of a few simple parametric families
    #     (linear, quadratic, exponential) and keep the best among them.
    # ----------------------------------------------------------------------
    def _score(seq):
        seq = np.asarray(seq, dtype=np.float64)
        conv = _convolve_fft(seq, fft_size)
        return 2.0 * n * conv.max() / (seq.sum() ** 2)

    # linear decreasing (a_i ∝ n‑i)
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum()
    sc = _score(lin)
    if sc < best_score:
        best_score, best_a = sc, lin.copy()
        print(f"linear decreasing improved score to {best_score:.6f}")

    quad = np.square(np.arange(n, 0, -1, dtype=np.float64))
    quad = quad / quad.sum()
    sc = _score(quad)

        exp_seq /= exp_seq.sum()

        if sc < best_score:
            best_score, best_a = sc, exp_seq.copy()
            print(f"exponential r={r:.3f} improved score to {best_score:.6f}")

    print(f"Finished after {it} iterations – best score = {best_score:.6f}")
    return best_a.tolist()
