    """Gradient-descent based optimizer for the step‐function bound.
    Uses a smooth soft‑max approximation of the max‑convolution and FFTs for
    fast evaluation of the objective and its gradient.  The algorithm runs
    for up to ``budget_s`` seconds and returns the best sequence found.
    """

    end_time = time.time() + budget_s - 5.0   # keep a margin

    # Helper to compute initial sequence
    def make_uniform(n):
        a = np.ones(n, dtype=float)
        target_sum = np.sqrt(2.0 * n)
        a *= target_sum / n
        return a

    # Determine a few starting points

        n0 = len(starts[0])
    else:

    starts.append(make_uniform(n0))

    # 3) a random Dirichlet‑type start (non‑negative, normalized)
    n_dir = n0
    rand = np.random.rand(n_dir) + 0.001
    rand = rand / rand.sum() * np.sqrt(2.0 * n_dir)
    starts.append(rand)

    best_seq = None
    best_score = np.inf

    # Core optimisation routine
    for start_idx, a0 in enumerate(starts):
        if time.time() >= end_time:
            break
        a = a0.astype(float)
        n = len(a)
        target_sum = np.sqrt(2.0 * n)

        # normalise sum – the objective is scale invariant, normalisation
        # improves numerical stability
        a *= target_sum / a.sum()

        # pad size for FFT (≥ 3·n to prevent circular wrap‑around)
        m = 1 << ((3 * n - 2).bit_length())

        # hyper‑parameters – they are slightly annealed inside the loop
        eta0 = 0.5            # initial step size
        alpha = 5.0           # soft‑max temperature
        it_without_improve = 0
        score = np.inf
        iteration = 0

        while time.time() < end_time:
            iteration += 1

            # ----- convolution a*a (linear) ---------------------------------
            A = np.fft.rfft(a, n=m)
            C = A * A
            conv = np.fft.irfft(C, n=m)[:2 * n - 1]

            c_max = np.max(conv)

            sum_exp = np.sum(exp_vals)

            # ----- simple gradient step ------------------------------------
            eta = eta0 / np.sqrt(iteration)          # decreasing step size

            a_new = np.maximum(a_new, 0.0)

            sum_new = np.sum(a_new)

            a_new *= target_sum / sum_new

"""
Gradient‑descent optimizer for the step‑function bound.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is homogeneous: scaling a does not change the score.
We therefore fix ‖a‖₁ = √(2 n) (the same normalisation used in the
state‑of‑the‑art code) – this turns the problem into minimising
the peak of the linear self‑convolution.

The algorithm uses a smooth soft‑max approximation of the max,
computes the gradient analytically (the derivative of the
soft‑max of a·a is a simple cross‑correlation) and takes
gradient steps with a decreasing learning rate.
All convolutions and correlations are performed with FFTs
so each iteration costs O(m log m) with
    m = next power of two ≥ 3 n  (no circular wrap‑around).

Several initialisations are tried (the previous best sequence,
a uniform sequence and a random Dirichlet start).  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np

def _linear_conv_fft(a: np.ndarray, m: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length m."""
    A = np.fft.rfft(a, n=m)
    C = A * A
    conv = np.fft.irfft(C, n=m)[: 2 * len(a) - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Linear cross‑correlation (valid part) of w with a:
        (w ⋆ a)[i] = Σ_j w[i+j] * a[j] ,   i = 0 … n‑1
    Implemented via FFT:  irfft(FFT(w)·conj(FFT(a))).
    Returns the first n entries.
    """
    W = np.fft.rfft(w, n=m)
    A = np.fft.rfft(a, n=m)
    cross = np.fft.irfft(W * np.conj(A), n=m)[: len(a)]
    return cross


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact score (no soft‑max) from a previously computed convolution."""
    return float(2.0 * n * np.max(conv) / (total * total))


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Run a time‑limited optimisation and return the best coefficient list."""
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # 1.  Choose starting points
    # ------------------------------------------------------------------
    starts = []

    # a) previous construction, if it exists
    try:
        hs = np.array(height_sequence_1, dtype=float)  # global from the runner
        if hs.ndim == 1 and hs.size > 0:
            starts.append(hs.copy())
    except Exception:
        pass

    # b) a uniform sequence of comparable length
    if starts:
        n0 = len(starts[0])
    else:
        n0 = np.random.randint(3000, 7000)
    uniform = np.ones(n0, dtype=float)
    target_sum = np.sqrt(2.0 * n0)
    uniform *= target_sum / n0
    starts.append(uniform)

    # c) a random Dirichlet‑type start (still normalized)
    rand = np.random.rand(n0) + 1e-6
    rand = rand / rand.sum() * np.sqrt(2.0 * n0)
    starts.append(rand)

    best_seq = None
    best_score = np.inf

    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 2.  Optimisation loop for every start
    # ------------------------------------------------------------------
    for start_id, a0 in enumerate(starts):
        if time.time() >= deadline:
            break

        a = a0.astype(float)
        n = len(a)
        target_sum = np.sqrt(2.0 * n)

        # normalise sum – the objective is scale‑invariant, normalisation helps numerics
        a *= target_sum / a.sum()

        # FFT length (must accommodate linear conv of length 2n‑1 and a correlation of length n)
        m = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

        # hyper‑parameters
        eta0 = 0.5          # initial step size
        alpha = 5.0         # soft‑max “temperature”
        iter_without_improve = 0
        iteration = 0

        while time.time() < deadline:
            iteration += 1

            # --------------------------------------------------------------
            # 2.1  Compute convolution a*a (linear) and soft‑max of its max
            # --------------------------------------------------------------
            conv = _linear_conv_fft(a, m)
            c_max = np.max(conv)

            # soft‑max approximation of max (prevent overflow by shifting)
            shifted = conv - c_max
            exp_vals = np.exp(alpha * shifted)
            sum_exp = np.sum(exp_vals)
            w = exp_vals / sum_exp                 # soft‑max weights, sum to 1
            M = c_max + np.log(sum_exp) / alpha    # smooth max value

            # --------------------------------------------------------------
            # 2.2  Gradient of the smoothed max
            # --------------------------------------------------------------
            # dM/da = 2 * (w ⋆ a)   (valid part, length n)
            cross = _cross_corr_fft(w, a, m)

            # --------------------------------------------------------------
            if iteration % 5 == 0:

                    a += jitter

                    a *= target_sum / np.sum(a)
                    iter_without_improve = 0

    # ------------------------------------------------------------------
    # 3.  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:

        uniform *= np.sqrt(2.0 * n_fallback) / n_fallback
        best_seq = uniform

    # Return a plain Python list – the harness will call evaluate_sequence().
    return best_seq.tolist()
