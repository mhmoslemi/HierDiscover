"""Generate a low‑evaluation step‑function.

The construction uses a power‑law (i**p) ramp that is scaled so that the
largest entry is just below the hard cap 1000.  For a polynomial of degree
p≥2 the convolution becomes almost flat, giving an evaluation score

    2·n·max(f*f) / (∑f)^2 = 2·((p+1)!)² / (2p+1)!,

which decreases quickly with p (≈0.008 for p=6).  The routine tests a few
choices of length and power inside the time budget and returns the best
sequence found.
"""

import time


def _scaled_power_ramp(length: int, power: int) -> list[float]:
    """
    Return a list ``[ (i**power) * scale for i in range(length) ]`` where
    ``scale`` makes the last element just below the limit 1000.
    """
    # avoid division by zero for length==1
    max_unscaled = (length - 1) ** power
    # a tiny safety factor keeps the value ≤1000 after the final clipping
    scale = 1000.0 / max_unscaled * 0.999999
    return [(i ** power) * scale for i in range(length)]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a power‑law ramp, try a few lengths / powers and keep the best
    (lowest) value returned by ``evaluate_sequence``.
    """
    import numpy as np  # imported only for the random seed (deterministic here)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    np.random.seed(seed)  # deterministic but not otherwise used

    best_seq = None
    best_score = float("inf")

    # candidate lengths (all comfortably below the practical limits)
    lengths = [1500, 2000, 2500]

    # try powers 1 … 8 – higher powers give lower scores, but we keep the
    # search tiny to stay well inside the time budget.
    for power in range(1, 9):
        for n in lengths:
            if time.time() > deadline:
                break

            seq = _scaled_power_ramp(n, power)

            # evaluate; any exception (e.g. overflow) is ignored
            try:
                score = evaluate_sequence(seq)
            except Exception:   # pragma: no cover
                continue

            if score < best_score:
                best_score = score
                best_seq = seq
                print(
                    f"New best – length={n}, power={power}, "
                    f"score={score:.6f}"
                )

            # a score far below the target is already sufficient
            if best_score < 0.01:
                break
        if best_score < 0.01 or time.time() > deadline:
            break

    # fallback – should never happen, but keeps the function total
    if best_seq is None:
        best_seq = [float(i) for i in range(20)]
        best_score = evaluate_sequence(best_seq)

    print(f"Returning sequence length={len(best_seq)} with final score={best_score:.6f}")
    return best_seq
