def full_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1 << ((n+m-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    fc = fa * fb
    c = np.fft.irfft(fc, n=size)
    return c[:n+m-1]

def full_convolve(a, b):
    total_len = len(a) + len(b) - 1
    size = 1 << ((total_len).bit_length())  # power of two >= total_len
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    fc = fa * fb
    c = np.fft.irfft(fc, n=size)
    return c[:total_len]

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Simulated annealing to minimize max of self-convolution.
    Returns best sequence of non-negative heights found within budget."""
    np.random.seed(seed)
    import time, math
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1000
    # start with uniform shape
    a = np.full(n, 1.0 / n)
    # compute objective
    size = 1 << ((2 * n - 1).bit_length())
    def compute_score(arr):
        A = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]
        return 2.0 * n * np.max(conv)
    curr_score = compute_score(a)
    best_seq = a.copy()
    best_score = curr_score

    start = time.time()
    deadline = start + budget_s - 2.0

    # temperature schedule:
    T0 = 0.1 * best_score  # initial temperature
    # We'll update T based on elapsed fraction
    # Choose step scale:
    step_scale = 0.02  # proportion of source mass to move

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Choose source and target
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        epsilon = min(a[i], step_scale * np.random.rand() * a[i] + 1e-12)  # positive
        j = np.random.randint(n)
        # propose new array
        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # compute new score
        new_score = compute_score(a_new)
        # temperature
        elapsed = (time.time() - start) / budget_s
        T = T0 * (1 - elapsed)  # linear cooling
        # acceptance
        if new_score < curr_score or np.random.rand() < math.exp(-(new_score - curr_score) / max(T, 1e-12)):
            a = a_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"Iter {iteration}: new best {best_score:.6f}")
        # optionally adapt step_scale
        if iteration % 10000 == 0:
            # adjust step_scale based on acceptance ratio, but we approximate
            pass
    # Return best_seq as list
    return best_seq.tolist()

use a linearly increasing “ramp’’ distribution

    a_i ∝ i+1  (i = 0,…,n‑1)

Normalising to `sum(a)=1` yields

    a_i = 2·(i+1) / (n·(n+1))
