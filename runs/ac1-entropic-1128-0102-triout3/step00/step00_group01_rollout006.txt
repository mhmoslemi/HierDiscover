def gradient_descent_flatten(a, steps=100, step_factor=0.2):
    a = np.array(a, dtype=float)
    sum_original = a.sum() if a.sum()>0 else 1.0
    for _ in range(steps):
        n = len(a)
        conv = compute_self_convolution(a)
        sum_a = a.sum()
        if sum_a < 1e-8:
            # reinitialize
            a = np.random.rand(n)
            sum_a = a.sum()
        c = sum_a**2 / (2*n - 1)
        diff = conv - c
        # compute gradient: 4 * correlation of a and diff
        a_rev = a[::-1]
        size = 1 << ((3*n -2).bit_length())
        fa_rev = np.fft.rfft(a_rev, n=size)
        fdiff = np.fft.rfft(diff, n=size)
        grad_full = np.fft.irfft(fa_rev * fdiff, n=size)
        # slice gradient
        grad = 4 * grad_full[n-1:n-1+n]
        # Determine step size
        max_abs = np.max(np.abs(grad)) + 1e-12
        lr = step_factor / max_abs  # ensures max update = step_factor
        # Update
        a_new = a - lr * grad
        # enforce nonnegative
        a_new = np.clip(a_new, 0, None)
        # Renormalize sum to original (scale invariance)
        sum_new = a_new.sum()
        if sum_new < 1e-8:
            break
        a_new *= sum_original / sum_new
        a = a_new
    return a

    fa = np.fft.rfft(a, n=size)

    return conv


def flatten_gradient_step(a: np.ndarray, step_factor: float = 0.2) -> np.ndarray:
    """
    One gradient‑descent step that reduces the variance of the
    self‑convolution of ``a``.
    The exact gradient of  Σ_k (b_k - μ)^2  with respect to ``a`` is
    4 * correlation(a, b-μ).  The implementation uses FFTs.
    ``step_factor`` controls the maximal absolute change of any entry.
    """
    n = a.shape[0]
    # convolution and statistics
    b = self_convolution(a)
    s = a.sum()
    if s == 0.0:
        return a
    mu = s * s / (2 * n - 1)               # average of the convolution
    diff = b - mu

    # correlation a ⋆ diff  (non‑circular, “full” correlation)
    a_rev = a[::-1]
    size = 1 << ((3 * n - 2).bit_length())   # length for a_rev (n) + diff (2n‑1) -1
    fa_rev = np.fft.rfft(a_rev, n=size)
    fdiff = np.fft.rfft(diff, n=size)
    corr_full = np.fft.irfft(fa_rev * fdiff, n=size)   # length 3n‑2
    grad = 4.0 * corr_full[n - 1 : n - 1 + n]          # extract n relevant entries

    # scale step so that the largest absolute update equals step_factor
    max_abs = np.abs(grad).max()
    if max_abs == 0.0:
        return a
    lr = step_factor / max_abs
    a_new = a - lr * grad
    # non‑negativity
    a_new = np.clip(a_new, 0.0, None)

    # keep the original scale (the objective is scale‑invariant)
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new *= s / sum_new
    return a_new
