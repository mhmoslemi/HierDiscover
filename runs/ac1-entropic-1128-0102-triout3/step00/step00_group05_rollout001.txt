    """Return a non‑negative vector b that maximises sum(b) with the
    linear constraints (f * b)_k ≤ bound for all k.
    """

A simple projected‑subgradient / LP optimizer for the convolution max problem.

The algorithm works with a fixed length `n`.  It starts from a random
non‑negative sequence whose ℓ₁‑norm is normalised to √(2 n) (the same
normalisation used by `evaluate_sequence`).  In each iteration it

1. evaluates the current objective value (which equals the maximal entry
   of the self‑convolution after normalisation);
2. solves a linear programme that finds a new vector `b` with a *larger*
   ℓ₁‑norm while keeping the convolution `a * b` bounded by the current
   maximal value;
3. rescales `b` to have the same sum as the current sequence, obtains a
   direction `d = b_scaled – a` and performs a short line‑search over
   several step sizes `t`.  The first candidate that does not worsen the
   objective is accepted; the best candidate seen so far is stored.
4. If the LP fails, a tiny random perturbation is applied to escape
   possible stagnation.

The loop runs until the allotted time budget (default 1000 s) elapses and
the best sequence found is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, bound: float):
    """
    Solve the linear programme:
        maximise   sum b_j
        subject to (f_seq * b)_k ≤ bound   for all k
                   b_j ≥ 0

    Returns the optimal `b` as a NumPy array, or ``None`` if the solver fails.
    """
    n = len(f_seq)
    # maximize sum(b) → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]          # coefficient f_i for variable b_j
        rows.append(row)
        rhs.append(bound)

    # non‑negativity constraints: -b ≤ 0
    rows.append(-np.eye(n))
    rhs.append(np.zeros(n))

    A = np.vstack(rows)
    b = np.hstack(rhs)

    # use the HiGHS solver; a small time limit keeps the whole search fast
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        method="highs",
        options={"disp": False, "time_limit": 1.0},
    )
    if res.success:
        return res.x
    return None
