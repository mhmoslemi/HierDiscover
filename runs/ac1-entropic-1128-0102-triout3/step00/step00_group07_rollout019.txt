def gradient_step(seq, lr):
    n = len(seq)
    a = np.array(seq, dtype=np.float64)
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    c = b[k]
    s = a.sum()
    if s < 1e-8: return seq
    # compute term for each i:
    idxs = np.arange(n)
    k_minus_i = k - idxs
    mask = (k_minus_i >= 0) & (k_minus_i < n)
    a_k_i = np.zeros_like(a)
    a_k_i[mask] = a[k_minus_i[mask]]
    grad = (2.0 * n / s**3) * (a_k_i * s - 2.0 * c)
    new_a = a - lr * grad
    new_a = np.clip(new_a, 0, 1000)
    return new_a.tolist()

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + max(0, budget_s - 1.0)

    # Choose a set of lengths to try (shorter lengths evaluate faster)
    length_candidates = [500, 1000, 2000, 4000, 8000]
    best_seq = None
    best_score = np.inf

    # 1. Explore simple parametric families
    for n in length_candidates:
        # uniform baseline (quick)
        seq = [1000.0] * n
        sc = evaluate_sequence(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq

        # exponential families
        for r in np.linspace(0.90, 0.9999, 20):
            seq = (1000.0 * (r ** np.arange(n))).tolist()
            sc = evaluate_sequence(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq

        # power families
        for p in np.linspace(0.5, 5.0, 15):
            seq = (1000.0 * ((n - np.arange(n)) / n) ** p).tolist()
            sc = evaluate_sequence(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq

        # random sorted
        for _ in range(5):
            arr = np.random.rand(n)
            seq = (np.sort(arr)[::-1] * 1000.0).tolist()
            sc = evaluate_sequence(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq

    # If a global sequence exists we also test it as starting point
    try:
        seq0 = list(height_sequence_1)
        sc0 = evaluate_sequence(seq0)
        if sc0 < best_score:
            best_score, best_seq = sc0, seq0
    except Exception:
        pass

    # 2. Gradient‑descent refinement with random restarts
    def gradient_step(a, lr):
        a = np.array(a, dtype=np.float64)
        n = len(a)
        b = np.convolve(a, a)
        k = int(np.argmax(b))
        c = b[k]
        s = a.sum()
        # avoid division by zero
        if s < 1e-8:
            return a.tolist()
        # compute a_{k-i}
        idx = np.arange(n)
        km = k - idx
        mask = (km >= 0) & (km < n)
        a_k = np.zeros_like(a)
        a_k[mask] = a[km[mask]]
        grad = (2.0 * n / s**3) * (a_k * s - 2.0 * c)
        new_a = a - lr * grad
        np.clip(new_a, 0.0, 1000.0, out=new_a)
        return new_a.tolist()

    # perform a few independent runs
    for restart in range(5):
        # start from the best sequence found so far
        cur = best_seq.copy()
        n = len(cur)
        lr = 1.0   # initial learning rate (tuned experimentally)
        # exponential cooling
        while time.time() < deadline:
            # one gradient descent step
            new = gradient_step(cur, lr)
            cur_score = evaluate_sequence(new)
            if cur_score < best_score:
                best_score, best_seq = cur_score, new.copy()
                # keep the improved sequence for future steps
                cur = new
                # reset learning rate a little to keep making progress
                lr = 1.0
            else:
                # occasional stochastic acceptance to escape plateaus
                if np.random.rand() < 0.01:
                    cur = new
                # decay the learning rate gently
                lr *= 0.9995
                # if lr becomes too small we break and restart
                if lr < 1e-4:
                    break
        # random perturbation before next restart
        # create a fresh random start with the same length
        cur = (np.random.rand(n) * 1000.0).tolist()
        # small smoothing to avoid all zeros
        cur = [max(1e-3, x) for x in cur]

    print("Best score found:", best_score, "length:", len(best_seq))
    return best_seq

Search for a step‑function (a non‑negative list of heights) that gives a low

The algorithm works in three stages:

   numerator of the evaluation function while preserving the denominator.

   performed (adding a small Gaussian bump to a random entry) to escape

    return evaluate_sequence(seq)

    return np.convolve(seq, seq)

    try:
        return _evaluate(seq)

        return np.inf

    """Geometric‑decay shape: a[i] = height * r**i."""
    return (height * (r ** np.arange(n))).tolist()

    """
    return (height * ((n - np.arange(n)) / n) ** p).tolist()

def _redistribute(seq, epsilon=0.01):
    """
    One greedy redistribution step.

    * Find the convolution coefficient with maximal value (index k).
    * Compute the contribution of each position i to that coefficient:
          contrib[i] = a[i] * a[k-i]   (zero if k-i out of range).
    * Choose a source index with the largest contribution and a
      destination index with a very small contribution.
    * Move a small fraction (epsilon * a[source]) of mass from source to
      destination, clip to feasible range, keep the total sum unchanged.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.size

    # convolution and index of the current maximum
    b = _convolution(a)
    k = int(np.argmax(b))

    # contributions to b[k]
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[i[mask]] * a[j[mask]]

    # source: index with maximal contribution
    src = int(np.argmax(contrib))

    # destination: pick among the 10‑percent smallest contributions
    threshold = np.percentile(contrib, 10)
    candidates = np.where(contrib <= threshold)[0]
    if candidates.size == 0:
        # fallback – any index different from src
        dst = (src + n // 2) % n
    else:
        dst = int(np.random.choice(candidates))

    if delta <= 0.0:

    a[src] -= delta
    a[dst] += delta

    # enforce the problem bounds
    np.clip(a, 0.0, 1000.0, out=a)
    return a.tolist()
