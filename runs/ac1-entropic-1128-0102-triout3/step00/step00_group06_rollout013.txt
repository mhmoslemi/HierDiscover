"""
Simple heuristic optimizer for the step‑function evaluation problem.

The objective (for a sequence with sum 1) is:
    O(a) = 2 * n * max(conv(a, a))

where n = len(a) and conv is the full (non‑reversed) convolution.
The optimum is obtained when the convolution is as flat as possible.
We use a projected sub‑gradient descent on the active index k* that
realises max(conv).  The sub‑gradient of max(conv) at a is:

    ∂max/∂a_i = 2 * a_{k* - i}   (if 0 ≤ k*‑i < n, otherwise 0)

To keep the sum of the coefficients equal to 1 we add the constant
vector (2/n) which makes the update direction have zero sum.
A small step is taken, the result is projected onto the non‑negative
orthant and re‑normalised.  The process is repeated inside a fast
random‑restart loop; the best sequence found before the time budget
expires is returned.

The algorithm runs entirely in memory, uses only NumPy and standard
library modules and respects the 1000 s time limit.
"""
import time
import numpy as np

def _objective(a: np.ndarray) -> float:
    """Return the raw objective value 2*n*max(conv(a,a)) for a with sum 1."""
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max()


def _local_optimize(a: np.ndarray,
                    max_iters: int = 500,
                    step_start: float = 0.15,
                    step_decay: float = 0.99,
                    step_growth: float = 1.02) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the current maximal convolution entry.
    Returns the improved vector and its objective value.
    """
    n = a.size
    step = step_start
    best_obj = _objective(a)

    for _ in range(max_iters):
        # current convolution and the index of the peak
        conv = np.convolve(a, a)
        max_b = conv.max()
        k = conv.argmax()

        # partner[i] = a_{k-i} if the pair (i, k-i) exists, else 0
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        partner = np.empty_like(a)
        partner.fill(0.0)
        partner[mask] = a[idx[mask]]

        # sub‑gradient of max_b (scaled by 2) and correction for the sum constraint
        # d = -2*partner + 2/n  (sum(d) == 0)
        d = -2.0 * partner + (2.0 / n)

        # tentative step
        a_new = a + step * d
        a_new = np.clip(a_new, 0.0, None)

        # re‑normalise to keep sum == 1 (scale invariance of the original objective)
        s = a_new.sum()
        if s == 0.0:               # collapse – restart from a random point
            break
        a_new /= s

        new_obj = _objective(a_new)

        # accept only if we improve; otherwise shrink the step size
        if new_obj < best_obj:
            a = a_new
            best_obj = new_obj
            step = min(step * step_growth, 0.5)   # allow a slightly larger step after success
        else:
            step *= step_decay
            if step < 1e-7:
                break

    return a, best_obj


def propose_candidate(seed: int = 123,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine performs many random restarts and projected sub‑gradient
    refinements within the given time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Helper to evaluate the external function (only used for sanity‑check)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence   # noqa: F821 – provided by the runner
    except NameError:
        # If the evaluator is not present (e.g. during local testing) we
        # fall back to the internal objective which is mathematically equal.
        def evaluate_sequence(seq):
            seq = [float(x) for x in seq]
            n = len(seq)
            b = np.convolve(seq, seq)
            return float(2.0 * n * max(b) / (sum(seq) ** 2))
    # ------------------------------------------------------------------

    # start with the known construction if it exists
    best_seq = None
    best_val = float('inf')
    if 'height_sequence_1' in globals():
        try:
            cand = list(height_sequence_1)
            val = evaluate_sequence(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
                print(f"Initial best from height_sequence_1: {best_val:.6f}")
        except Exception:
            pass

    # Main stochastic search
    # we use a moderate length; longer vectors give more freedom to flatten
    # the convolution, but also increase the constant factor 2*n.
    # Empirically n≈200–400 works well within the time budget.
    while time.time() < deadline:
        # Randomly pick a length for this restart (helps escape size‑related traps)
        n = np.random.randint(150, 401)          # 150 … 400
        # Dirichlet with small concentration to obtain varied, non‑uniform starts
        a = np.random.dirichlet(np.ones(n) * 0.4)

        # Local refinement
        a, val = _local_optimize(a,
                                 max_iters=800,
                                 step_start=0.12)

        # Record the best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best (n={n}) → {best_val:.6f}")

        # Occasionally perform a large random perturbation of the best sequence
        # to possibly discover a different basin of attraction.
        if best_seq is not None and np.random.rand() < 0.03:
            pert = np.array(best_seq, copy=True)
            # add Gaussian noise and renormalise
            noise = np.random.randn(pert.size) * 0.02
            pert = np.clip(pert + noise, 0.0, None)
            if pert.sum() == 0.0:
                pert = np.random.dirichlet(np.ones(pert.size))
            else:
                pert /= pert.sum()
            pert, val2 = _local_optimize(pert,
                                         max_iters=500,
                                         step_start=0.08)
            if val2 < best_val:
                best_val = val2
                best_seq = pert.copy()
                print(f"[{time.time():.1f}] Improved after perturb → {best_val:.6f}")

    # final sanity check using the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        if final_val < best_val:
            best_val = final_val
    except Exception:
        pass

    print(f"Search finished. Best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
