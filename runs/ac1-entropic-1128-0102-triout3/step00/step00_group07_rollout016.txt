def propose_candidate(seed=0, budget_s=950):
    import numpy as np, time, random

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    best_score = np.inf
    best_seq = None

    # parameters for search
    lengths = [50,100,200,400,800]  # maybe adapt
    for n in lengths:
        # Initialize with random Dirichlet
        a = rng.random(n)
        a = a / a.sum()
        # Optionally apply smoothing to shape (e.g., piecewise linear)
        cur = a.copy()
        cur_score = evaluate_sequence(cur.tolist())
        if cur_score < best_score:
            best_score = cur_score; best_seq = cur.tolist()
        # local search
        T0=1.0
        T= T0
        iter=0
        while time.time()<deadline:
            # Simulated annealing style
            # Choose random move
            i = rng.integers(0,n)
            j = rng.integers(0,n)
            if i==j: continue
            # choose delta proportion
            delta = (rng.random()*0.2) * cur[i]  # up to 20% of its value
            if delta <= 0: continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            # ensure nonnegativity maybe still ok as we take delta <= cur[i]
            new_score = evaluate_sequence(new.tolist())
            # Accept if better or with probability exp(-(new-old)/T)
            d = new_score - cur_score
            if d < 0 or rng.random() < np.exp(-d/T):
                cur = new
                cur_score = new_score
                # track best
                if cur_score < best_score:
                    best_score = cur_score; best_seq = cur.tolist()
            # cooling schedule
            iter+=1
            if iter % 1000 == 0:
                T = T0 * 0.99**(iter//1000)  # decays
    return best_seq

def project_onto_simplex(v):
    # returns closest vector to v on simplex sum=1, v>=0
    # algorithm from Duchi et al.
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=950):
    import numpy as np, time, random

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # choose length
    n = 3000  # try moderate size
    a = sqrt_series(n)  # sum=1

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score: {best_score}")

    # Parameters
    eta = 0.1
    temperature = 0.1
    iter = 0

    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = conv.argmax()

        # compute gradient
        grad = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        i_idx = np.arange(low, high + 1)
        j_idx = k - i_idx
        # sum contributions
        grad[i_idx] += a[j_idx]
        grad[i_idx] += a[i_idx]   # double count

        # step direction: move opposite to gradient to reduce max
        step = eta * grad
        # ensure we don't make negative entries
        step = np.minimum(step, a * 0.9)
        a_new = a - step

        # add random noise to maintain variability
        if rng.random() < 0.01:
            noise = (rng.random(n) - 0.5) * 1e-3
            a_new += noise

        # project onto simplex
        a_new = project_onto_simplex(a_new)

        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            continue

        # acceptance
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"Iter {iter} new best {best_score:.6f}")
        # simple hill climb: accept if improved, else maybe accept with probability
        if score <= evaluate_sequence(a.tolist()):
            a = a_new
        else:
            # Metropolis acceptance
            delta = score - evaluate_sequence(a.tolist())
            prob = np.exp(-delta / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                a = a_new

        # decay parameters
        iter += 1
        if iter % 1000 == 0:
            eta *= 0.9
            temperature *= 0.9

    # final best sequence as list
    return best_a.tolist()

"""
Gradient‑descent optimizer for the convolution‑based evaluation.

The algorithm starts from the classic “half‑binomial” (central‑binomial)
sequence a_i = C(2i,i)/4^i, which gives a flat auto‑convolution (the
coefficients of (1‑z)^‑1/2).  After normalising the vector to unit L1‑norm,
the evaluation is ≈ π/2 ≈ 1.57, already better than the uniform bound 2.

From this seed we perform an iterative projected gradient descent that
reduces the current maximal convolution coefficient.  For the index k where
the convolution attains its maximum we move a small amount of mass away
from the entries that contribute most to that peak (the gradient of the
peak).  After each step the vector is projected back onto the probability
simplex (non‑negative, sum = 1).  A simple simulated‑annealing acceptance
rule allows occasional uphill moves to escape shallow local minima.

The loop runs until the allotted time expires (default 950 s, leaving a
safety margin for the caller).  The best sequence encountered during the
search is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be defined at top level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    Generate the first n coefficients of the series
        a_i = C(2i,i) / 4**i,
    i.e. the coefficients of (1 - x)^(-1/2).
    The returned vector is normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_{i} = a_{i-1} * (2*i-1) / (2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { x | x_i >= 0,  sum_i x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]                # the last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete convolution x * x using real FFT.
    Result length is 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    # rfft gives N/2+1 complex numbers for real input of length N
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

    The routine starts from the half‑binomial (sqrt) sequence, then performs
    a projected gradient descent with a simple simulated‑annealing acceptance
    rule.  The best sequence found before the time limit is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # parameters
    n = 2500                     # length of the sequence (adjustable)
    max_iter = 10_000_000        # safety bound, loop will stop by deadline
    eta_start = 0.12             # initial step size (relative to gradient)
    temp_start = 0.08            # initial temperature for uphill moves
    decay_every = 2000           # iteration interval for parameter decay

    # ------------------------------------------------------------------
    # initialise from the half‑binomial series (the known good construction)
    a = sqrt_series(n)           # a already sums to 1
    conv = conv_fft(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val           # because sum(a)=1
    best_a = a.copy()
    best_score = cur_score

    # quick sanity print
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation loop
    deadline = time.time() + budget_s - 5.0   # a small safety margin
    eta = eta_start
    temperature = temp_start
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute convolution and locate the peak index
        conv = conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # gradient of the peak (sub‑gradient of max_k conv)
        # b_k = sum_i a_i a_{k-i}
        # derivative w.r.t. a_i (when k-i is a valid index):
        #   g_i = a_{k-i} + a_i   (the second term accounts for the symmetric pair)
        # for indices outside the support the derivative is zero.
        grad = np.zeros_like(a)

        # valid i satisfy 0 <= k-i < n
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k - idx

        # contributions
        grad[idx] += a[jdx]          # term a_i * a_{k-i}
        grad[idx] += a[idx]          # the symmetric partner (adds a_i)

        # steepest‑descent step (move opposite to the gradient)
        step = eta * grad
        # make sure we do not create negative entries
        step = np.minimum(step, a * 0.95)   # keep at least 5 % of each entry
        a_trial = a - step

        # occasional small random perturbation to increase diversity
        if rng.random() < 0.007:
            noise = (rng.random(n) - 0.5) * 1e-4
            a_trial += noise

        # project back onto the simplex (non‑negative, sum = 1)
        a_trial = project_onto_simplex(a_trial)

        # evaluate new candidate (the evaluation is cheap because sum=1)
        conv_trial = conv_fft(a_trial)
        max_val_trial = conv_trial.max()
        trial_score = 2.0 * n * max_val_trial

        # keep the best ever seen
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # acceptance rule (always accept improvements)
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            # Metropolis acceptance for occasional uphill moves
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # otherwise reject and keep current a

        # decay step size and temperature periodically
        if (it + 1) % decay_every == 0:
            eta *= 0.93
            temperature *= 0.92
            # safety: keep them positive
            eta = max(eta, 1e-5)
            temperature = max(temperature, 1e-6)

        it += 1

    # ------------------------------------------------------------------
    # final check (evaluate with the provided function for safety)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, final reported score={final_score:.6f}")

    return best_a.tolist()
