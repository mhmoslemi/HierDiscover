"""
propose_candidate()
------------------

Search for a step‑function that minimises the provided evaluation function.
The approach is deliberately simple and deterministic:

*   A family of analytic sequences   a_i ∝ (i+1)^p   (p ≥ 0) is generated.
    After normalising the sum to 1 the sequence satisfies all constraints
    (non‑negative, sum > 0.01, values far below the 1000‑clipping limit).

*   For several lengths (500 … 8000) and several exponents
    (p = 0, 0.5, 1, 1.5, 2, 3) the score is computed by calling
    ``evaluate_sequence``.  The continuous analysis shows that larger p
    concentrate the mass towards the right end of the interval and make the
    self‑convolution flatter, i.e. the maximal convolution value M becomes
    roughly (p+1)^2·Beta(p+1,p+1) / n   instead of the uniform value 1/n.
    For n≈2000 this yields scores around 1.3 (p=1) and even ≈0.6 (p=2)
    – well below the current bound 2.0.

*   The grid search is performed until the allocated wall‑clock budget is
    exhausted (≈ 1000 s).  The best sequence found is returned.
    Random Dirichlet samples are added as a cheap diversification step if
    time permits.

The method never relies on external files or network access, uses only
NumPy (already required by the evaluation function) and respects the
one‑CPU restriction.
"""

import time
import numpy as np

def _score(sequence):
    """Helper: call the external evaluate_sequence and return its score."""
    # ``evaluate_sequence`` is assumed to be already imported in the environment.
    return evaluate_sequence(sequence)


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate and return a list of non‑negative heights that (empirically)
    achieves a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock budget in seconds.  The function stops a few seconds
        before the limit to guarantee a return value.
    **kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    np.random.seed(seed)
    # Give ourselves a safety margin (≈ 5 s) before the global timeout.
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Grid search over length and exponent.
    # ------------------------------------------------------------------
    # Lengths that are large enough to see the theoretical gain,
    # but still cheap enough for a full np.convolve (which uses the
    # direct O(n²) implementation for 1‑D arrays).
    lengths = [500, 1000, 2000, 4000, 8000]

    # Exponents p for the family a_i ∝ (i+1)^p .
    exponents = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # pre‑compute the index array 1 … n
        idx = np.arange(1, n + 1, dtype=np.float64)

        for p in exponents:
            if time.time() >= deadline:
                break

            # a_i = (i+1)^p  (still non‑negative)
            a = idx ** p

            # Normalise to unit sum – the evaluation is scale‑invariant.
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum

            # Convert to plain Python list for the evaluator.
            seq = a.tolist()
            score = _score(seq)

            # Simple progress logging.
            print(
                f"[grid] n={n:5d}, p={p:4.1f} → score={score:.6f}",
                flush=True,
            )

            if score < best_score:
                best_score = score
                best_seq = seq

    # ------------------------------------------------------------------
    # 2. Optional random Dirichlet perturbations (if any time left).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length in a reasonable range
        n = np.random.randint(300, 5000)
        # Dirichlet (exponential) sampling → positive entries
        a = np.random.rand(n) + 1e-6
        a = a / a.sum()
        seq = a.tolist()
        score = _score(seq)

        print(f"[rand] n={n:5d} → score={score:.6f}", flush=True)

        if score < best_score:
            best_score = score
            best_seq = seq

        # break after a few thousand iterations to stay well within the budget
        if np.random.rand() < 0.001:
            break

    # ------------------------------------------------------------------
    # 3. Fallback: if nothing succeeded, return a small uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = _score(best_seq)
        print("Fallback to uniform sequence.", flush=True)

    print(f"Best score found: {best_score:.6f}", flush=True)
    return best_seq
