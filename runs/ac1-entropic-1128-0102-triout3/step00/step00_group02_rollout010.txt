"""
Simple stochastic search for a low‑value step sequence.

The evaluation function (after normalising the L1‑norm to sqrt(2·n))
is  eval = max_k (a*a)_k .
For a binary support set S (a_i ∈ {0,1}) with |S| = m and weight w = sqrt(2·n)/m
the value becomes

    eval = 2·n·max_count / m²

where max_count is the maximal number of ordered pairs (i,j)∈S² with i+j = k.
Thus the problem reduces to finding a set S with many elements but with a
nearly flat sumset.  A random set already gives a ratio close to 2; we improve
it by a greedy hill‑climbing that repeatedly removes an element participating
in the current peak and inserts a new element that does not increase that peak.

The algorithm:
* repeatedly draw a random size m (5‑40 % of n) and a random set S.
* evaluate its score using an FFT‑based convolution (O(n log n)).
* perform a bounded number of local moves that try to lower the maximal
  count; keep the best sequence ever seen.
* finally turn the best set into a real‑valued step sequence by giving every
  selected position the uniform weight w = sqrt(2·n)/|S|, which guarantees the
  required L1‑norm.
The whole search runs within the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _fft_max_convolution(binary_vec: np.ndarray, fft_len: int) -> int:
    """
    Compute max_{k} (binary_vec * binary_vec)_k using FFT.
    binary_vec must contain only 0 and 1.
    """
    # Forward FFT
    f = np.fft.rfft(binary_vec, n=fft_len)
    # Element‑wise product (convolution theorem)
    conv = np.fft.irfft(f * f, n=fft_len)
    # Keep only the first 2·n‑1 entries (the true linear convolution)
    # Round to nearest integer to avoid tiny floating errors.
    conv = np.rint(conv[: 2 * len(binary_vec) - 1])
    return int(conv.max())


def _score_from_set(S: set, n: int, fft_len: int):
    """
    For a given support set S compute
        eval = 2·n·max_count / |S|²
    and also return the maximal count and the index where it occurs.
    """
    m = len(S)
    if m == 0:
        return np.inf, np.inf, -1, None

    vec = np.zeros(n, dtype=np.float64)
    if m:
        vec[list(S)] = 1.0

    # FFT‑based max convolution
    f = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)
    conv = np.rint(conv[: 2 * n - 1]).astype(np.int64)

    max_count = int(conv.max())
    k_max = int(conv.argmax())

    score = 2.0 * n * max_count / (m * m)
    return score, max_count, k_max, conv


def _random_set(n: int, m: int) -> set:
    """Return a random subset of {0,…,n‑1} of size m."""
    # Using numpy's choice is faster than Python's random.sample for large n
    return set(np.random.choice(n, size=m, replace=False))


def _hill_climb(
    S: set,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
):
    """
    Try to improve a set S by swapping out elements that contribute to the current peak.
    Stops after max_iters steps or when the time budget expires.
    Returns (best_set, best_score, best_max_count).
    """
    best_set = S.copy()
    best_score, best_max, _, _ = _score_from_set(best_set, n, fft_len)

    S_cur = S.copy()
    score_cur, max_cur, k_max_cur, conv_cur = _score_from_set(
        S_cur, n, fft_len
    )

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        # Identify indices that participate in the current maximum.
        # For a given sum k, an element i participates iff (k‑i) ∈ S.
        participants = [i for i in S_cur if (k_max_cur - i) in S_cur]
        if not participants:
            # No obvious participant – fall back to a random swap.
            i_out = np.random.choice(list(S_cur))
            j_in = np.random.randint(n)
            while j_in in S_cur:
                j_in = np.random.randint(n)
            new_set = S_cur.copy()
            new_set.remove(i_out)
            new_set.add(j_in)
        else:
            # Remove a random participant.
            i_out = np.random.choice(participants)
            # Find a new index that does NOT create the same peak.
            # Try a few random candidates.
            attempts = 0
            while attempts < 15:
                j_in = np.random.randint(n)
                if j_in in S_cur:
                    attempts += 1
                    continue
                if (k_max_cur - j_in) in S_cur:
                    attempts += 1
                    continue
                break
            else:
                # Fallback to a completely random index.
                j_in = np.random.randint(n)
                while j_in in S_cur:
                    j_in = np.random.randint(n)

            new_set = S_cur.copy()
            new_set.remove(i_out)
            new_set.add(j_in)

        new_score, new_max, new_k_max, _ = _score_from_set(
            new_set, n, fft_len
        )

        # Accept only improving moves (greedy).
        if new_score < score_cur:
            S_cur = new_set
            score_cur, max_cur, k_max_cur = new_score, new_max, new_k_max
            if new_score < best_score:
                best_score, best_max, best_set = new_score, new_max, new_set
        # otherwise keep current set

    return best_set, best_score, best_max

# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Stochastic optimisation of a step function.
    Returns a list of non‑negative heights that (hopefully) beats the current bound.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # a small safety margin

    # Length of the reference construction (and a convenient default length)
    n = len(height_sequence_1)  # provided by the environment
    target_l1 = np.sqrt(2.0 * n)  # the normalisation used by the evaluator

    # Pre‑compute FFT length (next power of two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Start with the known uniform construction as a baseline
    best_seq = list(height_sequence_1)
    best_score = float(evaluate_sequence(best_seq))
    best_set = set(range(n))  # uniform has every index selected

    print(f"Initial uniform score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Randomly pick a density for the support set
        # 5 %–40 % of the indices give a good trade‑off between sparsity and count variance.
        m_min = max(2, int(0.05 * n))
        m_max = max(m_min + 1, int(0.40 * n))
        m = np.random.randint(m_min, m_max + 1)

        # Random initial set of size m
        cur_set = _random_set(n, m)
        cur_score, cur_max, cur_k_max, _ = _score_from_set(cur_set, n, fft_len)

        # Local improvement (greedy hill‑climbing)
        # Allow up to a few thousand elementary moves per start.
        max_local_iters = 3000
        improved_set, improved_score, improved_max = _hill_climb(
            cur_set, n, fft_len, max_local_iters, deadline
        )

        # Record global best
        if improved_score < best_score:
            best_score = improved_score
            best_set = improved_set
            print(
                f"New global best: score={best_score:.6f}, "
                f"|S|={len(best_set)}, max_count={improved_max}"
            )

    # ------------------------------------------------------------------
    # Build the final float sequence from the best support set.
    # All selected positions receive the same weight so that the L1‑norm
    # equals sqrt(2·n).  This guarantees that evaluate_sequence returns exactly
    # the maximal convolution count (i.e. the score we have been optimising).
    # ------------------------------------------------------------------
    final_seq = [0.0] * n
    if best_set:
        weight = target_l1 / len(best_set)
        for idx in best_set:
            final_seq[idx] = weight

    final_score = evaluate_sequence(final_seq)
    print(f"Final sequence score reported by evaluator: {final_score:.6f}")

    return final_seq
