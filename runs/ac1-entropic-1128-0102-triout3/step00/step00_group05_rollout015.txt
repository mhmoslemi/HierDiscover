Simplex projection algorithm:

def project_onto_simplex(v):
    """Projects vector v onto probability simplex (sum=1, v>=0)"""

    v = np.asarray(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # sort
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Projects vector v onto the probability simplex (nonnegative, sum=1)."""
    # works for any 1D numpy array
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000
    best_seq = None
    best_score = float('inf')
    # Run few restarts
    restarts = 0
    while time.time() < deadline:
        # Decide start: with probability use height_seq else random
        if restarts == 0 and 'height_sequence_1' in globals():
            init = np.array(height_sequence_1, dtype=float)
        else:
            init = np.random.rand(n)
        # normalize to sum 1
        if init.sum() == 0:
            init = np.ones(n) / n
        else:
            init = init / init.sum()
        # Gradient descent
        a = init
        # set step based on n
        step = 0.5  # initial?
        # loop for limited iterations maybe while time
        iteration = 0
        while time.time() < deadline and iteration < 2000:
            conv = np.convolve(a, a)
            max_v = conv.max()
            score = 2 * n * max_v  # sum =1
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # logging
                print(f"New best score {best_score:.6f} at iter {iteration}, restart {restarts}")
            # compute gradient for max indices
            max_idxs = np.flatnonzero(conv == max_v)
            # gradient vector
            g = np.zeros_like(a)
            # compute contributions
            idx_arr = np.arange(n)
            for idx in max_idxs:
                j = idx - idx_arr
                mask = (j >= 0) & (j < n)
                g[mask] += 2 * a[j[mask]]
            # try step sizes
            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                step_try = step * factor
                w = a - step_try * g
                # project onto simplex
                a_new = simplex_projection(w)
                conv_new = np.convolve(a_new, a_new)
                score_new = 2 * n * conv_new.max()
                if score_new < score:
                    a = a_new
                    step = step_try
                    improved = True
                    break
            if not improved:
                # random perturbation, small exchange
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i != j:
                    eps = min(a[i], 0.01)  # move small amount
                    a[i] -= eps
                    a[j] += eps
                    a = simplex_projection(a)  # ensure sum=1
            iteration += 1
        restarts += 1
        # optional early break if time
    # Return best sequence as list of float
    if best_seq is None:
        # fallback random
        best_seq = np.random.rand(n)
        best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()

"""
search.py – Projected gradient descent on the max‑convolution functional.

The goal is to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

subject to a_i ≥ 0.  Because the objective is invariant to an overall
scale, we keep the vector on the probability simplex (sum = 1).
Starting from a random point (or the supplied SOTA sequence) we repeatedly
  * compute the convolution,
  * locate the index/indices where the convolution attains its maximum,
  * build a sub‑gradient of the max‑operator,
  * take a small step opposite to this gradient,
  * project back onto the simplex.

If no improvement is found a tiny random mass transfer is performed to
escape possible plateaus.  The loop runs until the time budget (≈1000 s)
is exhausted and the best sequence seen is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays")
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:               # all entries are zero or negative
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient step for the max‑convolution objective
def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected‑gradient step that tries to lower the current
    max‑convolution value.  `a` is assumed to lie on the simplex.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_idx = np.flatnonzero(conv == max_val)

    # sub‑gradient of max(conv) at the current point
    g = np.zeros_like(a)
    idx_vec = np.arange(n)

    for k in max_idx:
        # indices j such that a_j contributes to conv_k
        j = k - idx_vec
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]

    # try a few step‑sizes (backtracking line‑search)
    for factor in (1.0, 0.5, 0.25, 0.125):
        trial = a - step * factor * g
        trial = _project_simplex(trial)
        trial_max = np.convolve(trial, trial).max()
        if trial_max < max_val:                # improvement
            return trial, step * factor, trial_max

    # no improvement – return the original point unchanged
    return a, step, max_val


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    evaluate_sequence().  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # Length of the problem – use the known SOTA length if available,
    # otherwise fall back to a moderate size.
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = 2000   # reasonable default for a fresh run

    # bookkeeping
    best_seq = None
    best_score = float('inf')

    # Parameters for the optimisation loops
    max_iters_per_restart = 4000
    initial_step = 0.5       # starting step‑size for the gradient step
    perturb_strength = 1e-3  # magnitude of random mass‑exchange when stuck

    restart = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Initialise a starting point
        if restart == 0 and 'height_sequence_1' in globals():
            a = np.array(height_sequence_1, dtype=float)
            if a.sum() == 0:
                a = np.ones_like(a)
        else:
            a = np.random.rand(n)
            if a.sum() == 0:
                a = np.ones_like(a)

        a = a / a.sum()               # put on the simplex
        step = initial_step
        # ------------------------------------------------------------------
        # Local optimisation loop
        for it in range(max_iters_per_restart):
            if time.time() >= deadline:
                break

            a, step, cur_max = _gradient_step(a, step)

            # current objective (sum(a) = 1 by construction)
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.1f}s] "
                      f"new best = {best_score:.6f} (restart {restart}, iter {it})")

            # if the gradient step made no progress, inject a tiny random
            # perturbation to encourage escape from flat regions
            if step < 1e-12 or np.abs(cur_max - np.convolve(a, a).max()) < 1e-12:
                i, j = np.random.randint(n, size=2)
                eps = min(a[i], perturb_strength)
                if eps > 0 and i != j:
                    a[i] -= eps
                    a[j] += eps
                    a = _project_simplex(a)

        restart += 1

    # ----------------------------------------------------------------------
    # Fallback – should never happen but keeps the contract safe
    if best_seq is None:
        best_seq = np.random.rand(n)
        best_seq = best_seq / best_seq.sum()

    # Convert to plain Python list as the caller expects
    return best_seq.tolist()
