def propose_move(seq):
    conv = fast_convolution(seq)
    k_max = np.argmax(conv)   # index of max
    g = np.zeros_like(seq)
    low = k_max - np.arange(n)
    # vectorized: compute reversed portion
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = seq[idx[mask]]
    # Choose i with probability proportional to g (or g^2) to target high contribution
    probs = g / g.sum() if g.sum()>0 else uniform
    i = np.random.choice(n, p=probs)
    # choose j uniformly among indices where g is low (e.g., choose j from complement)
    # simple: choose j random uniform
    j = np.random.randint(n)
    # ensure i != j
    while j == i:
        j = np.random.randint(n)
    # transfer epsilon
    eps = seq[i] * (np.random.rand() * 0.1)   # up to 10%
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    return new_seq
