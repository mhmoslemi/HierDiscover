def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Choose length
    n = 2400  # not too big
    # Build central binomial coefficient sequence up to n
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i-1] * (2*i-1) / (2*i)
    # At this point seq sum approx...
    # Normalize sum to 1 (optional)
    seq = seq / np.sum(seq)  # scaling doesn't affect O
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("Initial score:", best_score)
    # Local search
    while time.time() < deadline:
        # Randomly select a donor index based on product contributions
        # Compute convolution and max index
        b = np.convolve(seq, seq)
        max_idx = int(np.argmax(b))
        # compute contributions for max_idx
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        # compute product of each overlapping pair
        # product_i = seq[i] * seq[max_idx - i]
        # Let's vectorize:
        i_vals = np.arange(start, end+1)
        prod_vals = seq[i_vals] * seq[max_idx - i_vals]
        donor_idx = i_vals[np.argmax(prod_vals)]
        # Choose target index with smallest value not equal donor
        # maybe random among small values
        # we can choose target as index with smallest seq value
        target_idx = np.argmin(seq)
        if target_idx == donor_idx:
            # pick second smallest
            sorted_idx = np.argsort(seq)
            for idx in sorted_idx:
                if idx != donor_idx:
                    target_idx = idx
                    break
        # Transfer a small fraction
        donor_val = seq[donor_idx]
        if donor_val == 0:
            continue
        delta = donor_val * 0.02  # 2% move
        seq[donor_idx] -= delta
        seq[target_idx] += delta
        # Re-normalize to keep sum unchanged? sum unchanged due to transfer.
        # Evaluate score
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"New best: {best_score:.6f} at time {time.time():.2f}")
        else:
            # revert
            seq[donor_idx] += delta
            seq[target_idx] -= delta
        # occasional random perturbation:
        if np.random.rand() < 0.01:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                # small random swap
                amount = min(seq[i], seq[j]) * np.random.rand() * 0.05
                seq[i] -= amount
                seq[j] += amount
    return best_seq.tolist()

"""
propose_candidate
-----------------

Generate a high‑quality non‑negative step function (a list of heights) that
minimises the security evaluation function.

The strategy is three‑fold:

1. **Theoretical seed** – use the optimal analytic construction known from
   probability theory.  The coefficients
   C_k = (2k choose k) / 4**k (central binomial probabilities) satisfy

        (C * C)_k = 1   for 0 ≤ k < n

   after normalisation.  With this seed the evaluation value is
   2 n / (∑C_k)² ≈ π/2 ≈ 1.57, already a large improvement over the
   naïve constant sequence (value 2).

2. **Fallback seed** – if a previously found sequence `height_sequence_1`
   is available, evaluate it as well and keep the better one.

3. **Local greedy refinement** – repeatedly locate the shift `k` where the
   convolution `b = a ∗ a` is maximal, pick the pair (i, j) contributing
   the largest product a_i·a_j to that shift and move a tiny amount of
   mass from the larger index to the smallest index in the whole vector.
   The sum of the coefficients stays constant, while the maximal
   convolution value is usually reduced, thus lowering the objective.
   Simple simulated‑annealing style acceptance of occasional worsening
   moves helps escape shallow local minima.

The whole optimisation runs for the allotted time (default 1000 s,
leaving a small safety margin).  The best sequence found is returned.
"""

import time
import math
import random
import numpy as np

# The evaluation function `evaluate_sequence` is provided in the
# execution environment – we do **not** import it here.


def _central_binomial_sequence(length: int) -> np.ndarray:
    """
    Return the first `length` central‑binomial coefficients
    C_k = (2k choose k) / 4**k (as double precision floats).

    The recurrence C_{k+1} = C_k * (2k+1) / (2(k+1)) is numerically stable.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        # C_k = C_{k-1} * (2*k-1) / (2*k)
        seq[k] = seq[k - 1] * (2 * k - 1) / (2.0 * k)
    return seq

    """Thin wrapper converting a NumPy array to a Python list for the checker."""
    return evaluate_sequence(seq.tolist())

    # 1.  Choose a working length.
    #     If the reference construction `height_sequence_1` exists we reuse
    #     its length; otherwise we fall back to a modest size (~2500).
    # ------------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_len = 2500

    # ------------------------------------------------------------------
    # 2.  Build two candidate seeds:
    #     - the analytical central‑binomial construction,
    #     - the previously known construction (if any).
    # ------------------------------------------------------------------
    # analytical seed
    cb_seq = _central_binomial_sequence(base_len)
    # normalise to unit L1‑norm (scaling does not affect the objective)
    cb_seq = cb_seq / cb_seq.sum()

    # previous SOTA seed (if available)
    try:
        prev_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev_seq = prev_seq / prev_seq.sum()
        have_prev = True
    except Exception:
        have_prev = False

    # ------------------------------------------------------------------
    # 3.  Choose the better of the two as the starting point.
    # ------------------------------------------------------------------
    best_seq = cb_seq.copy()
    best_score = _evaluate(best_seq)
    print(f"Initial analytical (central‑binomial) score: {best_score:.6f}")

    if have_prev:
        prev_score = _evaluate(prev_seq)
        print(f"Initial previous construction score:   {prev_score:.6f}")
        if prev_score < best_score:
            best_seq = prev_seq.copy()
            best_score = prev_score
            print("→ Keeping the previous construction as start.")
    else:
        print("No previous construction found.")

    # ------------------------------------------------------------------
    # 4.  Local greedy refinement.
    #
    #    The main loop runs until the time budget is exhausted.
    #    In each iteration we locate the shift with the maximal convolution
    #    value, identify the pair of indices (i, j) producing the largest
    #    product a_i·a_j for that shift and move a small fraction of mass
    #    from the larger of the two to the overall smallest coefficient.
    #
    #    A tiny amount of simulated annealing (accepting a worse move with
    #    probability decaying with time) is added to avoid stagnation.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a few seconds for return

    # pre‑compute a mask for the smallest element (used as a cheap sink)
    def _smallest_index(arr):
        return int(np.argmin(arr))

    # helper to compute the objective quickly
    def _objective(arr):
        # writing the evaluation directly saves a list conversion
        n = len(arr)
        b = np.convolve(arr, arr)
        max_b = b.max()
        sum_a = arr.sum()
        if sum_a < 0.01:
            return float('inf')
        return float(2.0 * n * max_b / (sum_a ** 2))

    # start the refinement loop
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution once per iteration
        conv = np.convolve(best_seq, best_seq)
        k_max = int(conv.argmax())          # shift where max occurs
        max_val = conv[k_max]

        # identify overlapping indices for this shift
        start = max(0, k_max - (len(best_seq) - 1))
        end = min(len(best_seq) - 1, k_max)
        idx_range = np.arange(start, end + 1)
        partner = k_max - idx_range
        products = best_seq[idx_range] * best_seq[partner]

        # index contributing the largest product
        donor_pos = int(idx_range[products.argmax()])

        # partner of the donor in this shift
        donor_partner = int(k_max - donor_pos)

        # decide which of the two to pull mass from (the larger one)
        if best_seq[donor_pos] >= best_seq[donor_partner]:
            source = donor_pos
        else:
            source = donor_partner

        # choose a sink (the current smallest coefficient)
        sink = _smallest_index(best_seq)
        if sink == source:
            # if by chance they coincide pick the second smallest
            cand = np.argsort(best_seq)
            for cand_idx in cand:
                if cand_idx != source:
                    sink = int(cand_idx)
                    break

        # amount to transfer – a small fraction of the source value
        frac = 0.02                     # 2 % move each step
        delta = best_seq[source] * frac
        if delta <= 0.0:
            continue

        # perform the move
        best_seq[source] -= delta
        best_seq[sink]   += delta

        # re‑evaluate (fast objective)
        cur_score = _objective(best_seq)

        # Acceptance test
        if cur_score < best_score:
            best_score = cur_score
            # keep the modification
        else:
            # simulated‑annealing style: accept with tiny probability that
            # decays as the run progresses.
            temperature = max(0.01, (deadline - time.time()) / budget_s)
            if random.random() < math.exp(-(cur_score - best_score) / temperature):
                best_score = cur_score
            else:
                # revert the move
                best_seq[source] += delta
                best_seq[sink]   -= delta

        # occasional random shake‑up to escape plateaus
        if it % 250 == 0:
            # random pairwise redistribution of a tiny amount
            i, j = np.random.choice(len(best_seq), 2, replace=False)
            amount = min(best_seq[i], best_seq[j]) * 0.005
            best_seq[i] -= amount
            best_seq[j] += amount

        # reporting progress every few seconds
        if it % 500 == 0:
            now = time.time()
            remaining = max(0.0, deadline - now)
            print(f"[{now:.1f}] iteration {it:5d} – score {best_score:.6f} – "
                  f"time left {remaining:.1f}s")

    # ------------------------------------------------------------------
    # 5.  Return the best sequence discovered (as plain Python list).
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations – best score: {best_score:.6f}")
    return best_seq.tolist()
