"""
Propose a step‑function sequence with a provably low evaluation score.

The construction uses a *Sidon* (B2) set S of size p in the range
[0, p²‑1] (p a prime).  On the whole interval we place a small uniform
height `b = 1/p`.  On the Sidon positions we raise the height to `1`.
Because the Sidon property guarantees that any sum i+j occurs at most
once (ordered pairs give at most two contributions), the maximal value
of the convolution is exactly `2`.  The total sum of the sequence is
`2·p‑1`, so for n = p² we obtain

    evaluation = 2·n·max_b / (sum²) = 4·p² / (2·p‑1)² → 1   (as p → ∞)

For moderate primes (e.g. p=101) the score is ≈1.01, well below the
target 1.503.  The algorithm tries a few primes and keeps the best
candidate found within the time budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def is_prime(num: int) -> bool:
    """Return True if num is a prime (simple trial division)."""
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    limit = int(math.isqrt(num))
    for d in range(3, limit + 1, 2):
        if num % d == 0:
            return False
    return True


def generate_primes(limit: int) -> list[int]:
    """Return a list of all prime numbers up to *limit* (inclusive)."""
    return [p for p in range(3, limit + 1) if is_prime(p)]


def sidon_set(p: int) -> list[int]:
    """
    Classical Sidon construction for a prime p.
    The set { i·p + (i² mod p) | i = 0,…,p‑1 } has size p and lives in [0,p²‑1].
    """
    return [i * p + (i * i) % p for i in range(p)]


def build_sequence(p: int, big: float = 1.0) -> list[float]:
    """
    Build the sequence of length n = p².
    *big* is the height on the Sidon positions; the background height is
    ``big / p`` (i.e. 1/p when big=1).
    """
    n = p * p
    small = big / p                     # background height = 1/p
    seq = np.full(n, small, dtype=float)
    for idx in sidon_set(p):
        seq[idx] = big                  # raise Sidon positions
    return seq.tolist()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best sequence within the allowed time.
    The method builds Sidon‑based candidates for a few prime values of p
    (up to ~150) and returns the one with the smallest evaluation score.
    """
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # leave a tiny margin

    # a small set of primes – larger primes give scores closer to 1
    prime_limit = 200                     # safe upper bound for quick convolutions
    primes = generate_primes(prime_limit)

    # shuffle to avoid any bias; we will try the larger primes first
    primes.sort(reverse=True)              # better scores tend to larger p
    best_seq = None
    best_score = float("inf")

    for p in primes:
        if time.time() > deadline:
            break
        # build candidate (big=1 is sufficient because the evaluation is scale‑invariant)
        seq = build_sequence(p, big=1.0)

        # evaluate – this is the expensive part (np.convolve inside evaluate_sequence)
        try:
            score = evaluate_sequence(seq)
        except Exception as e:
            # any unexpected failure -> skip this candidate
            print(f"Evaluation failed for p={p}: {e}")
            continue

        print(f"p={p:<3}  n={p*p:<6}  score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"  ** new best (p={p}) with score {score:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny local refinement: try a few random perturbations of
    # the background height (all non‑Sidon entries).  Because the score
    # is almost scale‑invariant this rarely improves the result, but it
    # is cheap and may catch corner cases where the convolution peak is
    # slightly larger than 2.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # extract data of the current best solution
        n = len(best_seq)
        p = int(round(math.isqrt(n)))      # recover p (since n = p²)
        sid = set(sidon_set(p))

        # background indices
        bg_indices = [i for i in range(n) if i not in sid]

        # try a handful of random tweaks
        for _ in range(30):
            if time.time() > deadline:
                break
            # propose a tiny multiplicative change to the background height
            factor = 1.0 + random.uniform(-0.05, 0.05)   # +/-5 %
            new_seq = best_seq.copy()
            for i in bg_indices:
                new_seq[i] = max(0.0, best_seq[i] * factor)

            # keep the Sidon heights unchanged
            try:
                new_score = evaluate_sequence(new_seq)
            except Exception:
                continue

            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq
                print(f"  *** improvement by background tweak: score {new_score:.6f}")

    # if nothing succeeded (should not happen) fall back to a trivial uniform seq
    if best_seq is None:
        print("Warning: no valid candidate found – returning a uniform sequence.")
        best_seq = [1.0] * 10

    print(f"Returning sequence of length {len(best_seq)} with best score {best_score:.6f}")
    return best_seq
