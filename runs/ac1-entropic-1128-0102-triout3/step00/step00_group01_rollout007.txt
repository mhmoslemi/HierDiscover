    n = len(a)
    # forward
    size1 = 2*n-1
    m1 = 1 << ((size1-1).bit_length())

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_step(a, beta, lr):
    n = a.shape[0]
    # conv via FFT
    size_conv = 2*n - 1
    m_conv = 1 << ((size_conv - 1).bit_length())
    A = np.fft.rfft(a, m_conv)
    conv = np.fft.irfft(A * A, m_conv)[:size_conv]
    max_conv = np.max(conv)
    # softmax weights
    shifted = conv - max_conv
    exp_beta = np.exp(beta * shifted)
    w = exp_beta / np.sum(exp_beta)  # shape (2n-1,)
    # gradient via convolution of a_rev and w
    a_rev = a[::-1]
    size_grad = n + (2*n-1) - 1  # 3n-2
    m_grad = 1 << ((size_grad - 1).bit_length())
    AR = np.fft.rfft(a_rev, m_grad)
    WR = np.fft.rfft(w, m_grad)
    grad_full = np.fft.irfft(AR * WR, m_grad)[:size_grad]
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    return a_new, max_conv, conv

"""
Propose a candidate step‑function using a projected‑gradient descent
optimizer.  The objective we minimise is the evaluation function

    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Scaling all coefficients does not change the value, therefore we fix
sum(a)=1 and minimise 2·n·max(conv), i.e. minimise the maximal entry
of the self‑convolution.  The optimisation is performed with a smooth
approximation of the max (soft‑max) and the iterate is always projected
onto the probability simplex (non‑negative, sum = 1).  The gradient of the
soft‑max objective can be expressed as a convolution, which we evaluate
with FFT for speed.  Several random restarts with different lengths are
tried within the time budget; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real‑to‑real)."""
    n = a.shape[0]
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    conv = np.fft.irfft(A * A, m)[:size]
    return conv


def _gradient_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a  ( 1/β log Σ_k exp(β·conv_k) ).
    """
    n = a.shape[0]

    # --- convolution a * a -------------------------------------------------
    size_conv = 2 * n - 1
    m_conv = 1 << ((size_conv - 1).bit_length())
    A = np.fft.rfft(a, m_conv)
    conv = np.fft.irfft(A * A, m_conv)[:size_conv]

    # --- soft‑max weights --------------------------------------------------
    # subtract the max for numerical stability
    max_c = conv.max()
    shifted = conv - max_c
    exp_beta = np.exp(beta * shifted)
    w = exp_beta / exp_beta.sum()               # shape (2n‑1,)

    # --- gradient = 2 * (a_rev  *  w) ------------------------------------
    a_rev = a[::-1]
    size_grad = n + size_conv - 1               # = 3n‑2
    m_grad = 1 << ((size_grad - 1).bit_length())
    AR = np.fft.rfft(a_rev, m_grad)
    WR = np.fft.rfft(w, m_grad)
    grad_full = np.fft.irfft(AR * WR, m_grad)[:size_grad]
    # extract the part that corresponds to the original indices
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad, max_c


def _step(a: np.ndarray, beta: float, lr: float) -> tuple[np.ndarray, float]:
    """
    One gradient‑descent step.

    Returns the new vector (projected onto the simplex) and the
    current maximal convolution value.
    """
    grad, max_c = _gradient_softmax(a, beta)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new)
    return a_new, max_c


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within the allotted time.
    The routine performs several independent runs with different lengths,
    each run being a projected‑gradient descent optimisation of a smooth
    approximation of the evaluation objective.
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # keep the best sequence seen so far
    best_score = np.inf
    best_seq = None

    # Parameters for the optimisation loop
    MIN_N = 2000
    MAX_N = 8000
    MAX_RESTARTS = 1000   # generous upper bound – loop stops on time
    restart_counter = 0

    while time.time() < end_time and restart_counter < MAX_RESTARTS:
        restart_counter += 1

        # ------------------------------------------------------------------
        # Choose a random problem size and an initial vector
        # ------------------------------------------------------------------
        n = np.random.randint(MIN_N, MAX_N + 1)

        # start from a random Dirichlet‑type point (all entries positive)
        a = np.random.rand(n)
        a = a / a.sum()                     # enforce sum = 1

        # hyper‑parameters for this run (they are modestly adaptive)
        beta = 1.0
        beta_max = 3000.0
        beta_growth = 1.025

        lr = 0.5
        lr_decay = 0.9994

        # stop each sub‑run after a fixed number of iterations or when out of time
        max_iters = 5000
        last_improve_iter = 0
        it = 0

        # ------------------------------------------------------------------
        # Main optimisation loop for the current restart
        # ------------------------------------------------------------------
        while it < max_iters and time.time() < end_time:
            it += 1

            a, max_c = _step(a, beta, lr)

            # current objective (sum(a) is always 1)
            cur_score = 2.0 * n * max_c

            # keep the global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve_iter = it
                print(
                    f"[{time.time():.2f}] New best: {best_score:.6f} "
                    f"(n={n}, iter={it}, beta={beta:.2f}, lr={lr:.5f})"
                )

            # simple learning‑rate / beta schedule
            beta = min(beta_max, beta * beta_growth)
            lr *= lr_decay

            # occasional tiny random jitter to escape plateaus
            if np.random.rand() < 0.007:
                a += 1e-4 * np.random.randn(n)
                a = _project_simplex(a)

            # if we have made no progress for a while, break early
            if it - last_improve_iter > 1200:
                break

        # ------------------------------------------------------------------
        # After the inner loop we try a quick random perturbation and keep
        # the best we have.
        # ------------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.3:
            # start a new random point for the next restart
            a = np.random.rand(n)
            a = a / a.sum()
            # (no need to carry over the state – we just continue with a fresh start)

    # ----------------------------------------------------------------------
    # Fallback: if for some reason no sequence was found (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(10) / 10.0

    # Return a plain Python list – the competition harness will call
    # evaluate_sequence() on it.
    return best_seq.tolist()
