    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # base shape...
    i = np.arange(1, n+1)
    a = 1.0/np.sqrt(i + offset)
    a = a/np.sum(a)
    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)
    step = 0.05 # adjust to relative.

    while time.time() < deadline:
        # random move
        i_idx, j_idx = np.random.randint(0,n,2)
        if i_idx == j_idx:
            continue
        # propose new distribution by moving mass from i_idx to j_idx
        # pick fraction
        frac = np.random.rand()*step
        delta = frac * a[i_idx]  # moving this proportion
        new_a = a.copy()
        new_a[i_idx] -= delta
        new_a[j_idx] += delta
        # ensure nonneg
        if new_a[i_idx] < 0:
            # adjust delta to not go negative
            delta = a[i_idx]
            new_a[i_idx] = 0
            new_a[j_idx] = a[j_idx] + delta
        # renormalize to sum 1
        new_a = new_a / np.sum(new_a)
        score = evaluate_sequence(new_a.tolist())
        if score < best_score:

            best_seq = new_a.tolist()
            a = new_a
            # maybe reduce step for fine search
        else:
            # occasional acceptance with small probability
            if np.random.rand() < 0.001:
                a = new_a
        # adapt step
        if time.time() % 10 < 0.01:
            step = max(step * 0.99, 1e-5)
    return best_seq

sequence whose self‑convolution is almost flat and whose evaluation
value is ≈π/2≈1.57 – a clear improvement over the uniform sequence
(value 2.0).  Starting from this analytically‑motivated initial point,
a simple stochastic local‑search (mass transfer between two random
coordinates, accepting only improving moves) refines the shape
further.  The inner loop uses an FFT based convolution, therefore a
single score evaluation is O(n log n) and many iterations fit easily
into the 1000 s budget.

The function `propose_candidate` returns the best sequence found.
"""
import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast evaluation of a sequence (the same formula as
# evaluate_sequence but without the security checks).  It is only used
# internally – the final result will be checked by the provided
# evaluate_sequence.
def fast_score(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a,a))/(sum(a)**2) using FFT."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:                 # degenerate – not a valid candidate
        return np.inf
    # length for a circular convolution that contains the linear one
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """
    Initialise a positive sequence based on the discretised kernel
    a_i = 1 / sqrt(i + 1 + offset) and normalise it to ℓ¹‑norm 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops a few seconds before
        the limit so that the caller can still obtain the result).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a reasonable length.
    #     The value does not depend on n (asymptotically it tends to
    #     π/2), but a few thousand elements give a good approximation
    #     of the continuous kernel while staying fast.
    n = 3000

    # ------------------------------------------------------------------
    # 2.  Scan a few offsets for the analytical shape and keep the best.
    best_seq = None
    best_val = np.inf
    offsets = [0.0, 0.1, 0.5, 1.0, 2.0]   # small grid around the ideal 0
    for off in offsets:
        seq = build_initial_sequence(n, off)
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
    print(f"[init] best value after offset scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Hill‑climbing: repeatedly move a tiny amount of mass from one
    #     index to another (preserving the total ℓ¹‑norm).  Only moves
    #     that improve the score are kept (with a tiny chance of
    #     accepting a worse move to escape tiny plateaus).
    current = best_seq.copy()
    current_val = best_val

    step = 0.05          # relative step size for the transferred mass
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # amount to move = step * current[i] * U[0,1)
        delta = step * current[i] * np.random.rand()
        if delta > current[i]:          # avoid negative entries
            delta = current[i]

        new_seq = current.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # re‑normalise to keep sum exactly 1 (helps numerical stability)
        s = new_seq.sum()
        if s <= 0.0:          # should never happen, but guard against it
            continue
        new_seq /= s

        new_val = fast_score(new_seq)

        # accept if we improve, otherwise keep a tiny acceptance probability
        if new_val < current_val:
            current, current_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq
        else:
            if np.random.rand() < 1e-3:   # occasional uphill move
                current, current_val = new_seq, new_val

        # adapt the step size slowly – smaller step when we are close to a
        # local optimum
        if it % 5000 == 0 and it > 0:
            step = max(step * 0.9, 1e-5)

        # occasional progress report
        if it % 2000 == 0:
            print(f"[iter {it:6d}] best={best_val:.6f}, cur={current_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # convert to plain Python list (the caller expects a list of floats)
    return best_seq.tolist()
