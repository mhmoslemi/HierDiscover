def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Starting sequences: include uniform, random, height_sequence_1, maybe zeros plus random mass.
    seeds = []
    n0 = 6091
    seeds.append(list(height_sequence_1))
    seeds.append([1/n0]*n0) # uniform
    # random size between 1000 and 8000
    for _ in range(3):
        n = np.random.randint(2000, 8000)
        a = np.random.rand(n)
        a = a / a.sum()
        seeds.append(a.tolist())

    best_seq = None
    best_score = np.inf
    # helper functions
    ...

    for seq in seeds:
        # enforce sum normalization to 1 (or not needed)
        seq = np.array(seq, dtype=float)
        seq = seq / seq.sum()
        # iteration loop
        while time.time() < deadline:
            n = len(seq)
            # compute max conv
            conv = np.convolve(seq, seq)
            M = conv.max()
            # solve LP: maximize sum(g) with conv(seq, g) <= M, g>=0
            g = solve_convolution_lp(seq.tolist(), M)
            if g is None:
                # small random perturbation
                idx = np.random.randint(n)
                seq[idx] += np.random.normal(scale=0.01)
                seq = np.maximum(seq, 0)
                seq = seq / seq.sum()
                continue
            # normalize g to have same sum as seq
            sum_g = g.sum()
            if sum_g == 0:
                break
            g_norm = g * (seq.sum() / sum_g)
            # try mixing with larger t to reduce M
            best_t = 0
            best_seq_candidate = seq
            best_M = M
            # test t values
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
                cand = (1 - t) * seq + t * g_norm
                cand = np.maximum(cand, 0)
                cand_sum = cand.sum()
                if cand_sum == 0:
                    continue
                cand = cand / cand_sum   # maintain sum=1
                conv_cand = np.convolve(cand, cand)
                M_cand = conv_cand.max()
                if M_cand < best_M:
                    best_M = M_cand
                    best_t = t
                    best_seq_candidate = cand
            # update seq to best candidate
            seq = best_seq_candidate
            # compute evaluate_sequence score
            score = evaluate_sequence(seq.tolist())
            if not np.isfinite(score):
                continue
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best score {best_score:.6f} length {len(seq)}")
            # maybe optionally random restart if stuck.
            # early exit if score < target 1.503?
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    return best_seq

"""
propose_candidate
-----------------

Search for a low‑score step‑function by exploiting a parametric family

    a_i = (i + c) ** (-α) ,   i = 0,…,n‑1

which is known to give good autocorrelation properties (α≈0.5).  
The algorithm

1. picks a length (uses the length of the provided `height_sequence_1`
   if available, otherwise a default of 2000),
2. performs a coarse grid search over (α, c) and evaluates the
   objective

        score = 2·n·max_k (a*a)_k

   using a fast FFT‑based convolution,
3. refines the best parameters with a finer grid,
4. runs a short hill‑climbing phase that moves a small amount of mass
   between two random positions while keeping the vector non‑negative
   and normalised,
5. returns the best sequence found within the time budget.

All operations respect the constraints of `evaluate_sequence`
(non‑negative numbers, clipping at 1000, sum > 0).  The implementation
uses only NumPy and the standard library and finishes well within the
1000 s limit.
"""

import time
import numpy as np

def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max value of the discrete self‑convolution a*a using FFT."""
    # real FFT for speed (the sequence is real)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _score_from_array(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max_conv, where n = len(a) and a is already normalised."""
    n = len(a)
    max_conv = _fft_max_convolution(a, fft_len)
    return 2.0 * n * max_conv


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 2.0   # leave a small margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the reference length if present
    except Exception:
        n = 2000                            # fallback length

    # FFT length – next power of two that can hold a full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Helper to build a normalised candidate from parameters
    # ------------------------------------------------------------------
    def build_candidate(alpha: float, c: float) -> np.ndarray:
        """Return a normalised array a_i = (i + c)^(-alpha)."""
        i = np.arange(n, dtype=np.float64)
        # avoid division by zero – c must be > 0
        a = (i + c) ** (-alpha)
        s = a.sum()
        if s == 0.0:
            return None
        return a / s

    # ------------------------------------------------------------------
    # 3. Coarse grid search over (alpha, c)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_alpha = None
    best_c = None

    alphas = np.linspace(0.30, 0.80, 11)               # 0.30 … 0.80
    c_candidates = [1, 2, 5, 10, 20, 50, 100, 200,
                    500, 1000, min(2000, n // 2)]

    for alpha in alphas:
        for c in c_candidates:
            if time.time() > finish_time:
                break
            a = build_candidate(alpha, float(c))
            if a is None:
                continue
            sc = _score_from_array(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_alpha = alpha
                best_c = float(c)
                print(f"[coarse] alpha={alpha:.3f} c={c} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine refinement around the current best parameters
    # ------------------------------------------------------------------
    if best_seq is not None:
        for _ in range(3):               # three refinement passes
            # tighter neighbourhoods
            a_vals = np.linspace(max(0.05, best_alpha - 0.05),
                                 min(1.5, best_alpha + 0.05), 9)
            c_step = max(1, int(best_c * 0.1))
            c_vals = np.arange(max(1, int(best_c - 5 * c_step)),
                               min(n - 1, int(best_c + 5 * c_step) + 1),
                               c_step)
            for alpha in a_vals:
                for c in c_vals:
                    if time.time() > finish_time:
                        break
                    a = build_candidate(alpha, float(c))
                    if a is None:
                        continue
                    sc = _score_from_array(a, fft_len)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        best_alpha = alpha
                        best_c = float(c)
                        print(f"[refine] α={alpha:.4f} c={c} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Simple hill‑climbing to polish the solution
    # ------------------------------------------------------------------
    while time.time() < finish_time:
        # propose a small random transfer of mass between two indices
        a = best_seq.copy()
        i1, i2 = np.random.choice(n, size=2, replace=False)
        # magnitude of the transfer (up to ~1 % of the donor)
        delta = (np.random.rand() - 0.5) * 0.02
        # keep non‑negativity
        if a[i1] + delta < 0.0:
            delta = -a[i1] + 1e-12
        if a[i2] - delta < 0.0:
            delta = a[i2] - 1e-12
        a[i1] += delta
        a[i2] -= delta

        # renormalise (the sum may have drifted by ~1e‑15)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        sc = _score_from_array(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[hill] improvement → score={sc:.6f}")

        # occasional random restart from the parametric family
        if np.random.rand() < 0.02:
            r_alpha = np.random.uniform(0.35, 0.70)
            r_c = np.random.uniform(1.0, n / 2.0)
            a = build_candidate(r_alpha, r_c)
            if a is not None:
                sc = _score_from_array(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    best_alpha, best_c = r_alpha, r_c
                    print(f"[restart] α={r_alpha:.3f} c={r_c:.1f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never happens in practice)
        best_seq = np.ones(n) / n
    return best_seq.tolist()
