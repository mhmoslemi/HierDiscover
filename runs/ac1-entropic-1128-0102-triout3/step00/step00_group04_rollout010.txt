def simplex_projection(v):
    """Project onto probability simplex (sum = 1, non-negative)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all non-positive => return uniform?
        theta = 0

    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function using a projected sub‑gradient method.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is scale‑invariant we fix sum(a)=1.  Then
    score = 2 * n * max(conv(a, a))

The objective is convex (maximum of quadratic forms) on the probability simplex.
A simple sub‑gradient descent with projection onto the simplex quickly flattens
the auto‑convolution, pushing the maximum towards the average value
(≈1/(2n)).  In practice the method drives the score below the target 1.5
for modest sequence lengths (≈2000) within the 1000 s budget.

The algorithm:
* try several lengths n,
* start from a random point on the simplex,
* repeatedly:
    – compute the convolution via FFT,
    – locate the index k where the convolution attains its maximum,
    – form the sub‑gradient g_i = 2·a_{k‑i} (zero when the index is out of range),
    – take a diminishing step,
    – project back onto the simplex.
* keep the best sequence seen (lowest score).

The code respects all competition constraints: only standard libraries,
no closures, and uses `print` for progress reports.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that attains a
    low value of `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    best_score = float('inf')
    best_seq = None

    # a collection of lengths to explore – larger n gives a lower theoretical bound
    lengths = [200, 400, 800, 1200, 1600, 2000, 2500, 3000]

    # shuffle lengths to avoid systematic bias
    np.random.shuffle(lengths)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---------- initialise ----------
        # start from a random point on the simplex (sum = 1)
        a = np.random.rand(n)
        a = a / a.sum()

        # parameters for the sub‑gradient descent
        step0 = 0.6            # initial step size (tuned empirically)
        max_iter = 50000      # hard cap per length; loop also stops on time‑out

        # FFT length (next power of two >= 2n‑1) – reused every iteration
        fft_len = 1 << ((2 * n - 1).bit_length())

        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- compute convolution a * a (linear, not circular) -----
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)
            conv = conv[: 2 * n - 1]                     # length 2n‑1

            # ----- locate the worst index (maximal convolution value) -----
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]

            # current score (sum(a) == 1 ⇒ score = 2 * n * max_val)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved: n={n}, iter={it}, score={cur_score:.6f}")

            # ----- sub‑gradient of the active max entry -----
            g = np.zeros_like(a)

            # indices i for which (k_max - i) is a valid index
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            # i runs low … high, j = k_max - i
            i_idx = np.arange(low, high + 1)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * a[j_idx]

            # ----- step and projection onto the simplex -----
            step = step0 / math.sqrt(it)        # diminishing step size
            a_new = a - step * g
            a = _project_simplex(a_new)

        # end of while-loop for this n
    # end of lengths loop

    # Fallback – should never happen, but keep the contract safe.
    if best_seq is None:
        best_seq = np.full(100, 1.0)

    # Return a plain Python list (the evaluator works with any list of floats)
    return [float(x) for x in best_seq]
