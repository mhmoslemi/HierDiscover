def project_onto_simplex(v):
    # v: 1D np.ndarray
    # returns projection onto {x >=0, sum x =1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

            best_seq = p.copy()

        # Random move
        # Determine magnitude based on iteration
        step_frac = base_step / (1 + iter_idx * 0.001)  # decreasing

        # Choose two distinct indices
        i = rng.integers(n)
        j = rng.integers(n-1)
        if j >= i:
            j += 1

        delta = min(p[i], step_frac * p[i])
        if delta <= 0:
            continue

        p_new[i] -= delta
        p_new[j] += delta

        # Ensure nonnegatives:
        if p_new[i] < 0:

        fft_pnew = np.fft.rfft(p_new, n=size)

        max_c_new = conv_new.max()
        score_new = 2.0 * n * max_c_new

            p = p * np.exp(-eta * grad)

            s = p.sum()

            else:

    return best_seq.tolist()

"""
Propose a coefficient list for the step‑function optimisation problem.

The algorithm works on the probability vector p (non‑negative, sum = 1) that
represents the normalised heights.  The objective to minimise is

    score(p) = 2 * n * max(conv(p,p)) / (sum(p)**2)      (n = len(p))

Since the score is scale invariant we keep sum(p)=1 throughout the search.

* We start from a random perturbation of the given construction (if
  `height_sequence_1` exists) or from a uniform distribution.
* The core iteration is a simple *hill‑climbing* move:
    – locate the index `k_max` where the auto‑convolution is maximal,
    – find the pair (i, j) that contributes most to this peak,
    – transfer a small amount of mass from the index with the largest
      contribution to a random index.
  The move is accepted if it lowers the score, otherwise it may be
  accepted with a small Metropolis probability.
* Every few hundred iterations a soft‑max based gradient step is applied.
  The soft‑max weights concentrate on the largest convolution entries,
  the gradient is computed by a second FFT‑based convolution and an
  exponentiated‑gradient update keeps the iterate inside the simplex.
* The step size and the soft‑max temperature are annealed gradually.
* Throughout the run the best sequence (lowest score) is stored and
  finally returned.

The whole routine respects the 1000 s budget, uses only one CPU core and
does not write to the file system.  All helper functions are defined at
module level (no closures) as required by the evaluation framework.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(p: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution p * p using FFT of given size."""
    ft = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(ft * ft, n=size)
    return conv[: 2 * p.size - 1]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Compute soft‑max weights over the convolution entries."""
    m = conv.max()
    # shift to avoid overflow, the shift does not change the arg‑max
    exp_vals = np.exp(beta * (conv - m))
    return exp_vals / exp_vals.sum()


def _gradient_via_fft(p: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of  sum_k w_k * (p*p)_k   with respect to p.
    The formula is  grad_i = 2 * Σ_j p_j * w_{i+j}.
    Implemented with a single FFT‑based convolution.
    """
    n = p.size
    p_rev = p[::-1]
    size = _next_pow2(len(weights) + n - 1)
    fw = np.fft.rfft(weights, n=size)
    fr = np.fft.rfft(p_rev, n=size)
    conv_rev = np.fft.irfft(fw * fr, n=size)[: len(weights) + n - 1]
    # grad_i = 2 * conv_rev[(n-1) - i]
    idx = (n - 1) - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds
        before the limit to avoid a hard kill).

    Returns
    -------
    list[float]
        The best coefficient list found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise the probability vector p (sum(p) = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        n = init.size
        if init.sum() <= 0:
            p = np.full(n, 1.0 / n)
        else:
            p = init / init.sum()
    else:
        # fallback length – a few thousand entries give a good trade‑off
        n = 3000
        p = np.full(n, 1.0 / n)

    # break the perfect symmetry of a uniform start
    p += 0.001 * rng.random(n)
    p = np.maximum(p, 0.0)
    p /= p.sum()

    best_p = p.copy()
    best_score = evaluate_sequence(best_p.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT size for the (p * p) convolution
    # ------------------------------------------------------------------
    fft_size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin
    iteration = 0

    # annealing / step‑size parameters
    move_fraction = 0.2      # fraction of the current mass that may be moved
    beta = 1.0               # soft‑max temperature
    beta_inc = 1.02

    while time.time() < deadline:
        iteration += 1

        # ---- current convolution & score ---------------------------------
        conv = _conv_fft(p, fft_size)
        max_c = conv.max()
        # sum(p) == 1 (up to numerical noise)
        cur_score = 2.0 * n * max_c / (p.sum() ** 2)

        # store best solution
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[{iteration:06d}] new best score = {best_score:.6f}")

        # ---- targeted hill‑climbing move ---------------------------------
        # index of the maximal convolution entry
        k_max = int(conv.argmax())

        # all i that have a valid partner j = k_max - i
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idxs = np.arange(i_low, i_high + 1, dtype=int)
        partners = k_max - idxs
        contributions = p[idxs] * p[partners]

        # choose the pair that contributes most to the peak
        i_h = idxs[np.argmax(contributions)]
        j_h = partners[np.argmax(contributions)]

        # pick a random target index that is different from the chosen pair
        while True:
            t = rng.integers(n)
            if t != i_h and t != j_h:
                break

        # amount of mass to move (a fraction of the source entry)
        delta = min(p[i_h], move_fraction * p[i_h] + 1e-12)

        if delta > 0:
            p_candidate = p.copy()
            p_candidate[i_h] -= delta
            p_candidate[t] += delta
        else:
            # fallback to a completely random redistribution
            i_rand, j_rand = rng.choice(n, size=2, replace=False)
            delta = min(p[i_rand], move_fraction * p[i_rand] + 1e-12)
            if delta == 0:
                continue
            p_candidate = p.copy()
            p_candidate[i_rand] -= delta
            p_candidate[j_rand] += delta

        # stay on the simplex (numerical drift is removed)
        p_candidate = np.maximum(p_candidate, 0.0)
        p_candidate /= p_candidate.sum()

        # evaluate the candidate move (cheap – we reuse the FFT routine)
        conv_cand = _conv_fft(p_candidate, fft_size)
        cand_score = 2.0 * n * conv_cand.max() / (p_candidate.sum() ** 2)

        # Metropolis acceptance (temperature is deliberately low)
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) * 0.2):
            p = p_candidate
            # occasionally recompute the exact score for the new state
            # (already done via cur_score on the next loop)

        # ---- occasional soft‑max gradient step ---------------------------
        if iteration % 500 == 0:
            # increase temperature slowly, never exceeding 200
            beta = min(200.0, beta * beta_inc)

            # soft‑max over the convolution entries
            w = _softmax_weights(conv, beta)

            # gradient (vector of size n)
            grad = _gradient_via_fft(p, w)

            # exponentiated‑gradient update – automatically respects positivity
            eta = 0.08 / (1.0 + iteration * 0.001)
            p = p * np.exp(-eta * grad)
            p = np.maximum(p, 0.0)
            p_sum = p.sum()
            if p_sum > 0:
                p /= p_sum
            else:
                p = np.full(n, 1.0 / n)

        # slowly cool the move size
        if iteration % 1000 == 0:
            move_fraction = max(0.01, move_fraction * 0.9)

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return best_p.tolist()
