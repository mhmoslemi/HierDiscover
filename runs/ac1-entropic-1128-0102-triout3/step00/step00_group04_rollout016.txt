import numpy as np, time, math

def max_convolution(seq):
    """
    Return the maximum entry of the linear convolution of seq with itself.
    Utilizes FFT for O(n log n) speed.
    """
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:2 * n - 1]
    return float(np.max(conv.real))

def project_to_sum(vec, target):
    """Scale a non‑negative vector to have the exact L1 sum = target."""
    total = np.sum(vec)
    if total == 0:
        # avoid division by zero – return uniform distribution
        return np.full_like(vec, target / len(vec))
    return vec * (target / total)

def random_mask_sequence(n, target_sum):
    """
    Create a random 0/1 mask of length n and assign equal weight
    so that the L1‑norm equals target_sum.
    """
    # pick a random density between 0.4 and 0.95
    p = np.random.uniform(0.4, 0.95)
    k = max(1, int(round(p * n)))
    # choose distinct indices
    idx = np.random.choice(n, k, replace=False)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = 1.0
    seq = project_to_sum(seq, target_sum)
    return seq

"""
Random‑mask hill‑climbing search for a low‑evaluation step function.

The evaluation function is scale‑invariant; after fixing the L1‑norm
to S = sqrt(2 n) the score equals the maximum entry of the
self‑convolution.  Uniform sequences give a score of 2, while a
randomly sparsified mask spreads the pairwise products over many
different sums, typically lowering the peak.  The algorithm works in
two phases:

1.  Randomly generate masks of varying density, assign equal weight to
    the selected positions, normalise the sum to S and keep the best
    candidate (fast FFT based convolution).

2.  Perform cheap local perturbations of the current best sequence
    (add small Gaussian noise to a random subset, re‑normalise) and
    accept any improvement.

The method is completely self‑contained, uses only NumPy/FFT for
speed and respects the 1000 s time budget.
"""

import time
import math
import numpy as np


def max_convolution(seq: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution of `seq` with itself.
    FFT is used for O(n log n) speed.
    """
    n = len(seq)
    # next power‑of‑two size for an efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    return float(np.max(conv.real))


def project_to_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its L1‑norm becomes exactly `target`.
    If the vector is all zeros a uniform distribution is returned.
    """
    total = np.sum(vec)
    if total == 0.0:
        return np.full_like(vec, target / len(vec))
    return vec * (target / total)


def random_mask_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Build a random 0/1 mask of length `n`, set the chosen entries to the same
    positive value and normalise the whole vector to have L1‑norm `target_sum`.
    The mask density is sampled uniformly from [0.4, 0.95] (values outside this
    range are rarely useful for the objective).
    """
    p = np.random.uniform(0.4, 0.95)          # density of non‑zeros
    k = max(1, int(round(p * n)))            # number of active indices
    idx = np.random.choice(n, k, replace=False)
    seq = np.zeros(n, dtype=np.float64)
    seq[idx] = 1.0
    return project_to_sum(seq, target_sum)


def local_perturb(seq: np.ndarray, target_sum: float,
                  step_scale: float = 0.05) -> np.ndarray:
    """
    Produce a mild random perturbation of `seq`:
      * add Gaussian noise to a small random subset of entries,
      * clip negatives to zero,
      * re‑scale to the exact L1‑sum `target_sum`.
    The `step_scale` controls the magnitude of the noise relative to the
    typical entry size.
    """
    n = len(seq)
    new_seq = seq.copy()
    # modify roughly 5 % of the coordinates (at least one)
    m = max(1, int(0.05 * n))
    idx = np.random.choice(n, m, replace=False)
    noise = np.random.randn(m) * step_scale * np.mean(seq)
    new_seq[idx] += noise
    np.maximum(new_seq, 0.0, out=new_seq)
    return project_to_sum(new_seq, target_sum)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds
    and returns the best list discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a little margin

    # problem size – a trade‑off between solution quality and speed
    n = 2500
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # initialise with any previously known construction
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            tmp = np.array(height_sequence_1, dtype=np.float64)
            tmp = project_to_sum(tmp, target_sum)
            score = max_convolution(tmp)
            best_seq = tmp.tolist()
            best_score = score
            print(f"Start from height_sequence_1, score = {best_score:.6f}")
        except Exception as exc:
            print("Failed to evaluate height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # Phase 1 – random mask exploration
    # ------------------------------------------------------------------
    trials = 0
    while time.time() < deadline:
        cand = random_mask_sequence(n, target_sum)
        cand_score = max_convolution(cand)
        trials += 1
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            print(f"Trial {trials}: new best score = {best_score:.6f}")

        # occasional cheap local move starting from the current best
        if best_seq is not None and np.random.rand() < 0.02:
            pert = local_perturb(np.array(best_seq), target_sum, step_scale=0.1)
            pert_score = max_convolution(pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.tolist()
                print(f"Local tweak improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – focused hill climbing around the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        current = np.array(best_seq, dtype=np.float64)
        for i in range(2000):
            if time.time() >= deadline:
                break
            candidate = local_perturb(current, target_sum, step_scale=0.02)
            cand_score = max_convolution(candidate)
            if cand_score < best_score:
                best_score = cand_score
                current = candidate
                print(f"Refinement step {i+1}: score = {best_score:.6f}")
        best_seq = current.tolist()

    print(f"Search finished after {trials} random trials. "
          f"Best score = {best_score:.6f}")
    return best_seq
