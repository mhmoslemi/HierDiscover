def project_onto_simplex(v):
    """Project onto the probability simplex {x >= 0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def solve_convolution_lp(f, M):
    # objective: maximize sum(g) i.e., -np.ones(n) as c (minimize -sum)
    n = len(f)
    c = -np.ones(n)
    # Build constraints for each k: sum_i f_i * g_{k-i} <= M
    # This is linear in g.
    A = []
    b = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] += f[i]
        A.append(row)
        b.append(M)
    # Non-negativity constraints are implicit in bounds
    bounds = [(0, None)] * n
    # Use linprog
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': 5})
    if result.success:
        return result.x
    else:
        return None
```

def propose_candidate(seed=0,budget_s=1000):
    import numpy as np, time
    from scipy.optimize import linprog
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    # decide length
    max_len = 800
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = min(max_len, len(base))
        f = base[:n]
    except Exception:
        n = max_len
        f = np.random.rand(n)

    # ensure non-neg
    f = np.maximum(f, 0)
    # scaling: set sum to sqrt(2n)
    target_sum = np.sqrt(2 * n)
    if f.sum() <= 0:
        f = np.full(n, target_sum / n)
    else:
        f = f * (target_sum / f.sum())

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"Start length {n}, initial score {best_score}")

    # small initial disturbance to break symmetry
    f += 0.01 * target_sum * (np.random.rand(n) - 0.5)
    f = np.maximum(f, 0)
    f = f * (target_sum / f.sum())

    # loop
    while time.time() < deadline:
        M = np.max(np.convolve(f, f))
        g = solve_convolution_lp(f, M)
        if g is None:
            # random bump
            idx = np.random.randint(n)
            f[idx] = max(0, f[idx] + np.random.randn() * 0.01 * target_sum)
            f = f * (target_sum / f.sum())
            continue
        # mix
        t = 0.01
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        # ensure not exceed 1000 each: clip
        f_new = np.minimum(f_new, 1000.0)
        # update sum automatically (no renorm)
        # compute score
        try:
            score = evaluate_sequence(f_new.tolist())
        except Exception:
            score = float('inf')
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            print(f"Improved score {best_score:.6f} at time {time.time():.2f}, sum {f_new.sum():.2f}, max {np.max(np.convolve(f_new,f_new)):.4f}")
        # accept new
        f = f_new
        # also occasionally add random jitter to escape plateau
        if np.random.rand() < 0.01:
            f += 0.005 * target_sum * np.random.randn(n)
            f = np.maximum(f, 0)
            # optionally renorm to stay within bound
    # Return best
    result = [float(x) for x in best_f.tolist()]
    print(f"Final best score {best_score:.6f}")

"""
Propose a step‑function (non‑negative list of heights) that has a smaller
value of the evaluation function

    2 * n * max(conv(a,a)) / (sum(a)**2)

than the trivial uniform construction.  The method follows the
state‑of‑the‑art iterative scheme:

* start from a (slightly perturbed) uniform sequence of length n,
  normalized so that Σ a = √(2n) (the normalisation used in the reference
  algorithm);
* given the current sequence f, compute its convolution bound M = max(f*f);
* solve a linear programme that **maximises** the total mass of a new
  sequence g subject to the *cross‑convolution* constraints  f*g ≤ M
  (these constraints are linear in g);
* replace f by a small convex combination (step size t≈0.01) of f and g.
  Because Σg ≥ Σf while the bound M stays (almost) unchanged,
  the ratio 2·n·M/Σ² strictly decreases.
* repeat until the time budget expires, keeping the best sequence seen.

The LP is solved with SciPy’s “highs” interior‑point solver; the problem
size is limited to a few hundred entries (default n=800) so that many
iterations fit comfortably into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise   sum(g)
        subject to  conv(f_seq, g) ≤ bound   (for all shifts)
                    g ≥ 0

    The constraints are linear in g because f_seq is fixed.
    Returns the optimal g (as a NumPy array) or None if the solver fails.
    """
    n = len(f_seq)
    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n)

    # Build the matrix A_ub so that A_ub @ g ≤ b_ub encodes conv(f,g) ≤ bound.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # For a given shift k, the coefficient of g[j] is f[k-j] if the index is valid.
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # i runs over the indices of f that appear in this convolution term.
        # The corresponding index in g is j = k - i.
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, bound, dtype=float)

    # Non‑negativity bounds for the variables.
    bounds = [(0.0, None)] * n

    # Use the modern HiGHS solver; it is fast for these modest sizes.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None

    # Return the best sequence as a plain Python list of floats.
    # ------------------------------------------------------------------
    result = [float(x) for x in best_seq.tolist()]

    return result
