    n = len(f)
    rows = []
    cols = []
    data = []

def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """Solve LP: maximize sum(b) subject to conv(f,b) <= rhs, b>=0."""
    from scipy.optimize import linprog
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b) => minimize -sum(b)
    # Build convolution matrix A for constraints conv(f, b) <= rhs.
    # Matrix shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            i_start = k - j_end
            i_end = k - j_start
            # f slice from i_start to i_end inclusive
            # assign reversed order to columns j_start..j_end
            # We'll fill row k entries column j_start..j_end with coer.
            # to avoid loops, we loop j and assign corresponding f value.
            # Simpler: for j in range(j_start, j_end+1):
            #    rows.append(k); cols.append(j); data.append(f_seq[k - j])
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                data.append(f_seq[k - j])
    # nonnegativity constraints: -I b <= 0
    # Add rows with -I
    for i in range(n):
        rows.append(2 * n - 1 + i)
        cols.append(i)
        data.append(-1.0)
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1 + n, n))
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])
    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None),
                  method='highs',
                  options={'time_limit': time_limit, 'disp': False})
    if res.success:
        return res.x
    else:
        return None

"""
Hybrid LP‑guided descent with occasional random perturbations.

The objective to minimise is

    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Scaling a → c·a leaves E unchanged, therefore we keep every
candidate normalised to   sum(a) = sqrt(2·n) .
For a normalised sequence the score equals max(conv(a,a)).

Algorithm
---------

1. Initialise a candidate:
   * use the supplied `height_sequence_1` if available,
   * otherwise start from a random positive vector.
   The vector is normalised to the required sum.

2. Repeatedly improve the candidate until the time budget expires.

   a)  Compute the current maximal convolution value M.
   b)  Solve the linear programme

          maximise   Σ b_i
          subject to conv(a,b) ≤ M ,   b_i ≥ 0 .

       The LP finds a direction `g` that can increase the total mass
       without increasing the current worst convolution entry.

   c)  Normalise `g` to the same total mass as `a` and try a few
       convex combinations  a' = (1‑t)·a + t·g  with
       t ∈ {0.02,0.05,0.1,0.2,0.4}.  The best improvement (if any) is kept.
       The current candidate is updated whenever a strictly better score
       is observed.

   d)  If the LP fails (or no improvement is found) a tiny random
       redistribution of mass between two coordinates is performed.
       The vector is re‑normalised afterwards.

   e)  Every few hundred iterations without progress a fresh random
       restart is performed to escape possible stagnation.

3. The best sequence seen during the whole run is returned.

The LP is built as a sparse Toeplitz‑like matrix where each row
represents one convolution constraint; only non‑zero coefficients are stored.
`scipy.optimize.linprog` with the HiGHS solver solves the LP under a short
time limit, keeping the whole search well within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired ℓ₁‑norm for a sequence of length n."""
    return np.sqrt(2.0 * n)


def _normalise(seq, target):
    """Scale seq so that its sum equals target (leaving zero entries unchanged)."""
    total = np.sum(seq)
    if total == 0:
        # uniform fallback
        return [target / len(seq)] * len(seq)
    factor = target / total
    return [float(x * factor) for x in seq]


def _max_convolution(seq):
    """Return max(conv(seq,seq))."""
    return np.max(np.convolve(seq, seq))


def _convolution_matrix(f):
    """
    Build the (2n‑1) × n matrix A such that A·b = conv(f,b) (vectorised).
    Only non‑zero entries are stored (CSR format).
    """
    n = len(f)
    rows = []
    cols = []
    data = []

    # For each convolution index k we fill the appropriate slice of f.
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # column indices j_start … j_end
        # corresponding f‑coefficients are f[k-j] = f[ reversed slice ]
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve:
        maximise Σ b_i
        s.t. conv(f_seq, b) ≤ rhs ,  b_i ≥ 0

    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                       # maximise sum(b)  →  minimise -sum(b)
    A_ub = _convolution_matrix(f_seq)     # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, rhs)

    # HiGHS is the default and works with sparse matrices.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"time_limit": time_limit, "disp": False},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) with a small
    evaluation score.  The function respects the 1000 s time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a small safety margin

    # ------------------------------------------------------------------
    # Choose an initial length and starting sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
        cur = list(height_sequence_1)
    else:
        n = 2401                # a reasonable default (same as the SOTA length)
        cur = [np.random.rand() for _ in range(n)]

    # Normalise to the invariant sum √(2n)
    target = _target_sum(n)
    cur = _normalise(cur, target)

    # best‑ever bookkeeping
    best_seq = cur[:]
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # current candidate (may diverge from best)
    cur_score = best_score
    no_improve = 0

    # candidate mixing factors
    t_values = (0.02, 0.05, 0.1, 0.2, 0.4)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Compute the present worst convolution value M
        # --------------------------------------------------------------
        M = _max_convolution(cur)

        # --------------------------------------------------------------
        # 2) Solve the LP to obtain a promising direction g
        # --------------------------------------------------------------
        g = _solve_lp(cur, M, time_limit=2.0)

        if g is not None:
            # Normalise g to the same ℓ₁‑norm as cur
            sum_g = np.sum(g)
            if sum_g > 0:
                g = _normalise(g, target)

                # ------------------------------------------------------
                # 3) Try a few convex combinations (line‑search on t)
                # ------------------------------------------------------
                improved_this_round = False
                for t in t_values:
                    cand = [(1.0 - t) * a + t * b for a, b in zip(cur, g)]
                    # safeguard against accidental zero‑sum (should not happen)
                    cand = _normalise(cand, target)

                    cand_score = evaluate_sequence(cand)

                    # keep the global best
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand[:]
                        print(f"[{time.time() - start_time:5.1f}s] New best = {best_score:.6f} (t={t})")

                    # accept improvements for the current trajectory
                    if cand_score < cur_score - 1e-12:
                        cur = cand
                        cur_score = cand_score
                        improved_this_round = True
                        # break early – we already moved forward
                        break

                if not improved_this_round:
                    no_improve += 1
                else:
                    no_improve = 0
            else:
                # pathological g, fallback to random tweak
                no_improve += 1
                g = None

        else:
            # LP failed – perform a tiny random redistribution
            no_improve += 1

        # --------------------------------------------------------------
        # 4) If LP gave no progress, do a small random redistribution
        # --------------------------------------------------------------
        if g is None or no_improve > 0:
            i, j = np.random.randint(n, size=2)
            if i != j:
                # move a random fraction of mass from i to j
                eps = (np.random.rand() - 0.5) * 0.05 * cur[i]  # up to ±2.5 %
                new_i = cur[i] + eps
                new_j = cur[j] - eps
                if new_i >= 0 and new_j >= 0:
                    cur[i] = new_i
                    cur[j] = new_j
                    # re‑normalise to keep the invariant sum
                    cur = _normalise(cur, target)
                    cur_score = evaluate_sequence(cur)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cur[:]
                        print(f"[{time.time() - start_time:5.1f}s] Random tweak improved to {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Stagnation handling – occasional full restart
        # --------------------------------------------------------------
        if no_improve >= 300:
            print(f"[{time.time() - start_time:5.1f}s] Stagnation detected, restarting.")
            cur = [np.random.rand() for _ in range(n)]
            cur = _normalise(cur, target)
            cur_score = evaluate_sequence(cur)
            no_improve = 0

    # final report
    final_score = evaluate_sequence(best_seq)
    print(f"[done] Best score found = {final_score:.6f}")
    return best_seq
