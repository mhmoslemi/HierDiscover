def project_simplex(v):
    # returns projection onto simplex [w_i >=0, sum w_i = 1].
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # v is 1D numpy array
    # returns w that is nonnegative and sums to 1.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_to_simplex(v):
    # based on Duchi et al. (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_conv(p):
    n = p.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fp * fp, size)
    conv = conv[: 2 * n - 1]

    conv = np.maximum(conv, 0.0)
    return conv

"""
Projected gradient descent to flatten a self‑convolution.

The evaluation function for a list ``a`` is

    E(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is invariant under scaling, we can treat ``a`` as a
probability distribution (sum = 1).  The goal is to minimise the
maximum entry of the convolution, i.e. to make the distribution of the
sum of two independent draws as flat as possible.

The algorithm works with a vector ``p`` (non‑negative, sum = 1):
    1. Compute the convolution ``c = p * p`` (via FFT for speed).
    2. Find the index k where ``c[k]`` is maximal.
    3. The sub‑gradient of the max‑convolution at that index is
       g_i = p_{k-i} + p_i (the second term appears only when i == k-i).
    4. Take a small step opposite to the gradient and project back onto
       the probability simplex.
    5. Repeat until the maximum convolution entry is close to its
       theoretical lower bound 1/(2n-1) or the time budget expires.
    6. Several random restarts are performed; the best sequence found
       (lowest evaluation score) is returned.

The implementation respects all competition rules: only top‑level
functions, no file I/O, and prints occasional progress messages.
"""

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0, sum_i w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return the uniform vertex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_convolution(p: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution p * p using an FFT.
    The result length is 2*len(p)-1.
    """
    n = p.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fp = np.fft.rfft(p, size)
    conv_full = np.fft.irfft(fp * fp, size)
    conv = conv_full[: 2 * n - 1]                    # truncate to exact length
    # eliminate tiny negative round‑off values
    np.maximum(conv, 0, out=conv)
    return conv


def max_conv_gradient(p: np.ndarray, kmax: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at index kmax)
    with respect to the probability vector p.
    """
    n = p.shape[0]
    grad = np.zeros_like(p)
    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    i = np.arange(i_start, i_end + 1, dtype=int)
    j = kmax - i
    # contribution p_{k-i}
    grad[i] += p[j]
    # diagonal contribution (i == j) appears twice in the derivative
    diag = i == j
    grad[i[diag]] += p[j[diag]]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    performing many random restarts of a projected‑gradient descent that tries
    to flatten the self‑convolution of the sequence.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate the current probability vector using the exact
    #         evaluation function (scale‑invariant, sum = 1)
    # ------------------------------------------------------------------
    def current_score(p_vec):
        n = p_vec.shape[0]
        conv = compute_convolution(p_vec)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Optional: start from the provided global construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.sum() > 1e-12:
            init /= init.sum()
            sc = current_score(init)
            best_seq = init.copy()
            best_score = sc
            print(f"[init] starting score from height_sequence_1 = {sc:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – many random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a problem size; larger n gives a lower theoretical bound
        n = np.random.randint(300, 1500)           # 300‑1500 works fast with FFT
        # random Dirichlet start (avoid the uniform stationary point)
        p = np.random.rand(n)
        p /= p.sum()

        # inner gradient‑descent loop
        max_iters = 3000
        for it in range(max_iters):
            # time check
            if time.time() >= deadline:
                break

            conv = compute_convolution(p)
            max_val = conv.max()
            # current evaluation score (scale invariant)
            score = 2.0 * n * max_val

            # keep global best
            if score < best_score:
                best_score = score
                best_seq = p.copy()
                print(
                    f"[improve] new best = {best_score:.6f} (n={n}, it={it})"
                )
                # stop early if target is already reached
                if best_score <= 1.503:
                    return best_seq.tolist()

            # stopping criterion – when we are already close to the average
            avg = 1.0 / (2 * n - 1)
            if max_val <= avg * 1.02:    # within 2 % of the lower bound
                break

            # sub‑gradient step for the maximal entry
            kmax = int(np.argmax(conv))
            grad = max_conv_gradient(p, kmax)

            # step‑size schedule – larger when the gap is large
            gap = max_val - avg
            lr = 0.5 * gap / max_val   # adaptive learning rate
            p -= lr * grad
            # project back onto the simplex (non‑negative, sum = 1)
            p = project_to_simplex(p)

            # occasional random jitter to escape tiny plateaus
            if it % 200 == 0 and it > 0:
                jitter = np.random.randn(n) * 0.001
                p = np.maximum(p + jitter, 0.0)
                p = project_to_simplex(p)

        # end of inner loop – try a new random length if time permits
    # ------------------------------------------------------------------

    # Fallback in the very unlikely case that nothing was found
    if best_seq is None:
        best_seq = np.ones(10) / 10.0

    return best_seq.tolist()
