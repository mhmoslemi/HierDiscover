def propose_candidate(seed=12345, budget_s=1000):

    np.random.seed(seed)
    # Determine n as length of height_sequence_1 if exists else default
    try:
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
    except NameError:
        n = 1000
        base_seq = np.ones(n) / n
    # Initialize best as uniform:
    best = np.full(n, 1.0/n)
    best_score = evaluate_sequence(best.tolist())
    # Also try starting from base_seq if provided
    candidates = [best.copy(), base_seq.copy()]
    # schedule for temperature
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    # Precompute FFT size:
    size = 1 << ((2*n-1).bit_length())
    def conv(arr):
        A = np.fft.rfft(arr, size)
        c = np.fft.irfft(A * A, size)[:2*n-1]
        return c
    # Helper to compute max conv quickly
    for curr in candidates:
        # Run local search on curr
        a = np.maximum(curr, 1e-12) # avoid zeros? keep.
        a = a / a.sum()
        # start temperature
        T = 0.01
        patience = 0
        while time.time() < deadline:
            # compute conv and max
            c = conv(a)
            max_val = c.max()
            k = int(c.argmax())
            # contributions
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            i_range = np.arange(i_min, i_max+1)
            prod = a[i_range] * a[k - i_range]
            # find index with largest product
            idx = i_range[prod.argmax()]
            jdx = k - idx
            # choose donor = idx if a[idx]>=a[jdx] else jdx
            donor = idx if a[idx] >= a[jdx] else jdx
            # choose receiver = index with minimal a
            receiver = np.argmin(a)
            if donor == receiver:
                # pick second min
                receiver = np.argsort(a)[1]
            # step size epsilon proportional to donor value
            eps = min(a[donor] * 0.05, 0.001)  # dynamic
            # propose new a
            a_new = a.copy()
            a_new[donor] -= eps
            a_new[receiver] += eps
            # compute new max
            new_c = conv(a_new)
            new_max = new_c.max()
            delta = new_max - max_val
            accept = False
            if delta < 0:
                accept = True
            else:
                # probabilistic acceptance
                prob = np.exp(-delta / T) if T > 1e-8 else 0.0
                if np.random.rand() < prob:
                    accept = True
            if accept:
                a = a_new
                max_val = new_max
                # update best if improved (lower max)
                score = 2 * n * max_val  # because sum is 1
                if score < best_score:
                    best_score = score
                    best = a.copy()

                patience = 0
            else:
                patience += 1
            # cooling schedule
            if patience > 1000:
                T *= 0.9
                patience = 0

                break
    # Finally, convert best to list

"""
Propose a candidate sequence of non‑negative heights that (hopefully) lowers the
evaluation score 2·n·max(conv(a,a))/sum(a)^2 below the uniform bound 2.0.

The algorithm is a simple stochastic hill‑climber / simulated‑annealing
procedure:

1.  Initialise a few diverse starting points (uniform, random, and the
    previously found construction if available).
2.  Repeatedly evaluate the self‑convolution via an FFT (fast O(n log n)).
3.  Find the index k where the convolution attains its maximum.
    The contribution of each entry a[i] to this maximum is
        c[i] = a[i] * a[k‑i]   (if the partner index exists).
4.  Pick the pair (i, j=k‑i) with the largest product.  Move a small amount
    of mass from the larger of the two entries to an index that currently has
    the smallest value.  This reduces the dominant product while keeping the
    total sum constant.
5.  Occasionally (10 % of the moves) perform a random redistribution of a
    small mass between two random positions – this injects noise to escape
    plateaus.
6.  Accept a move if it strictly reduces the maximal convolution value.
    Otherwise accept it with a temperature‑controlled probability
    (simulated annealing) to allow occasional uphill steps.
7.  Keep the best sequence seen so far and stop when the time budget is
    exhausted (the caller gives 1000 s; we stop a few seconds early).

The routine works for any length `n`.  If the global variable
`height_sequence_1` exists it is used as an additional starting point.
All numbers are kept non‑negative; the final list is returned as plain
Python floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy import optimize

# ----- Helper functions -----------------------------------------------------

def fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return the aperiodic convolution a*a using an FFT (real input)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return c


def move_mass(a: np.ndarray, donor: int, receiver: int,
              amount: float) -> np.ndarray:
    """Return a copy of a with `amount` moved from donor to receiver."""
    b = a.copy()
    b[donor] -= amount
    b[receiver] += amount
    return b


def random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative vector of length n with sum 1."""
    x = rng.exponential(scale=1.0, size=n)
    x /= x.sum()
    return x


# ----- Main candidate generator -------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the caller allows ~1000 s).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a safety margin

    # --------------------------------------------------------------------
    # Determine the problem size.  Use the length of the provided
    # `height_sequence_1` if it exists; otherwise fall back to 2000.
    # --------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        base_seq = np.array(height_sequence_1, dtype=float)  # noqa: F821
    except Exception:
        n = 2000
        base_seq = np.full(n, 1.0 / n)

    # Normalise the base sequence to have sum = 1 (the objective is scale‑invariant)
    base_seq = np.maximum(base_seq, 0.0)
    if base_seq.sum() == 0:
        base_seq = np.full(n, 1.0 / n)
    else:
        base_seq = base_seq / base_seq.sum()

    # Keep track of the best sequence discovered so far
    best_seq = base_seq.copy()
    best_score = 2.0 * n * fft_convolve(best_seq).max()   # same as evaluate_sequence for sum==1
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------------
    # Prepare a few initial candidates: uniform, the historic solution,
    # and a few random vectors.
    # --------------------------------------------------------------------
    candidates = [
        np.full(n, 1.0 / n),   # uniform
        best_seq.copy()
    ]
    for _ in range(3):
        candidates.append(random_start(n, rng))

    # --------------------------------------------------------------------
    # Parameters for the stochastic search
    # --------------------------------------------------------------------
    # Amount of mass moved in a deterministic "targeted" step
    eps_factor = 0.02           # fraction of the donor's current mass
    eps_min    = 1e-6           # absolute lower bound
    # Acceptance temperature (starts relatively warm and cools down)
    temperature = 0.01
    temp_decay   = 0.9995
    # How often we perform a purely random move (to add exploration)
    random_move_prob = 0.1
    # Number of consecutive non‑improving iterations after which we
    # optionally reduce the temperature more aggressively.
    stagnation_counter = 0
    stagnation_limit   = 2000

    # --------------------------------------------------------------------
    # Main optimisation loop (over all starts)
    # --------------------------------------------------------------------
    for start_idx, a in enumerate(candidates):
        a = np.maximum(a, 0.0)
        a /= a.sum()                     # enforce sum == 1

        # Local copy of the best score for this start
        local_best_score = best_score
        local_best_seq   = a.copy()

        # Simple annealing loop
        while time.time() < deadline:
            # ------------------------------------------------------------
            # 1) compute convolution and locate the current worst index k
            # ------------------------------------------------------------
            conv = fft_convolve(a)
            max_val = conv.max()
            k = int(conv.argmax())

            # ------------------------------------------------------------
            # 2) decide which move to propose
            # ------------------------------------------------------------
            if rng.random() < random_move_prob:
                # ---- random redistribution move ----
                i, j = rng.choice(n, size=2, replace=False)
                # move a small random fraction of the donor's mass
                donor = i if a[i] > a[j] else j
                recv  = j if donor == i else i
                eps = min(eps_min + eps_factor * a[donor], a[donor] * 0.5)
                a_new = move_mass(a, donor, recv, eps)
            else:
                # ---- targeted move that attacks the dominant product ----
                # valid i's for this k
                i_min = max(0, k - (n - 1))
                i_max = min(n - 1, k)
                idx_range = np.arange(i_min, i_max + 1)
                partner = k - idx_range
                products = a[idx_range] * a[partner]

                # index producing the largest product
                worst_local = idx_range[products.argmax()]
                worst_partner = k - worst_local

                # Choose the larger entry as donor
                if a[worst_local] >= a[worst_partner]:
                    donor = worst_local
                else:
                    donor = worst_partner

                # Receiver = index with the smallest current value (different from donor)
                recv = np.argmin(a)
                if recv == donor:
                    # second smallest if the absolute minimum is the donor
                    recv = np.argsort(a)[1]

                # Amount to move (a modest fraction of the donor)
                eps = min(eps_min + eps_factor * a[donor], a[donor] * 0.5)
                a_new = move_mass(a, donor, recv, eps)

            # ------------------------------------------------------------
            # 3) evaluate the new candidate (only the maximal convolution)
            # ------------------------------------------------------------
            conv_new = fft_convolve(a_new)
            new_max = conv_new.max()
            delta = new_max - max_val

            # ------------------------------------------------------------
            # 4) accept / reject (simulated annealing)
            # ------------------------------------------------------------
            accept = False
            if delta < 0:
                accept = True
            else:
                # probability ~ exp(-Δ / T)
                if temperature > 0:
                    prob = np.exp(-delta / temperature)
                    if rng.random() < prob:
                        accept = True

            if accept:
                a = a_new
                max_val = new_max
                # Check if this is the best overall solution so far
                score = 2.0 * n * max_val   # sum == 1 → convenient form
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[improve] new best score={best_score:.6f} "
                          f"time={time.time() - start_time:.1f}s")
                stagnation_counter = 0
            else:
                stagnation_counter += 1

            # ------------------------------------------------------------
            # 5) annealing schedule & optional temperature reset
            # ------------------------------------------------------------
            temperature *= temp_decay
            if stagnation_counter >= stagnation_limit:
                # make the search a bit more exploratory
                temperature = max(temperature, 0.001)
                stagnation_counter = 0

            # Small safeguard against numeric drift
            if abs(a.sum() - 1.0) > 1e-12:
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # end of while – time exhausted for this start
        print(f"[finish] start {start_idx} completed, best score so far={best_score:.6f}")

    # --------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------------
    return best_seq.tolist()
