def propose_candidate(seed=42, budget_s=990):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = 1500
    # initial sequence
    curr = np.ones(n).tolist()
    best = curr.copy()
    best_score = evaluate_sequence(best)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute LP direction
        res = compute_lp_step(curr)
        if res is None:
            # random perturb
            idx = np.random.randint(n)
            curr[idx] = max(0, curr[idx] + np.random.randn() * 0.01)
            continue
        g_norm, M = res
        # line search
        cand_best_seq = curr
        cand_best_score = best_score
        for t in [0.01,0.05,0.1,0.2,0.4]:
            new_seq = [(1 - t) * a + t * b for a,b in zip(curr, g_norm)]
            # ensure nonnegative; maybe round small negatives to zero
            # evaluate
            score = evaluate_sequence(new_seq)
            if score < cand_best_score:
                cand_best_score = score
                cand_best_seq = new_seq
        # accept if improvement
        if cand_best_score < best_score - 1e-9:
            best_score = cand_best_score
            best = cand_best_seq
            print(f"Iter {iteration}: improved score {best_score}")
        curr = cand_best_seq
    return best

"""
Propose a sequence of non‑negative heights that minimizes the given evaluation
function.

The algorithm is an accelerated version of the linear‑programming based
procedure described in the statement.

*   The current sequence `f` is always renormalised so that
    sum(f) = sqrt(2·n).  With this normalisation the current bound
    M = max(f * f) is exactly the value that appears in the objective
    2·n·M / (sum(f)^2).

*   For the fixed bound `M` we solve the linear program

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M    for every convolution index k
                    g_j ≥ 0

    The constraints are linear in `g`; the LP therefore produces a
    direction `g` that keeps the convolution bound while increasing the
    total sum of the coefficients as much as possible.

*   The LP solution is rescaled to the same normalisation (sum = sqrt(2·n))
    and mixed with the current sequence:

        f_new = (1‑t)·f + t·g          (0 < t ≤ 1)

    Several step sizes `t` are tried and the one giving the lowest
    `evaluate_sequence` value is kept.  Small `t` guarantees monotonic
    decrease of the bound; the line‑search permits larger advances when
    they are safe.

*   If the LP becomes infeasible (rare) a tiny random perturbation is
    applied to `f` to escape the stall.

The loop runs until the allocated time budget expires (default 990 s,
leaving a small safety margin).  The best sequence found during the run
is returned.

No external files are accessed; only NumPy and SciPy are used.
"""

import time
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def _normalise(seq: list[float]) -> np.ndarray:
    """
    Scale the sequence so that its L1‑norm equals sqrt(2·n).
    Returns a NumPy array.
    """
    n = len(seq)
    s = np.sum(seq)
    if s == 0:
        return np.zeros_like(seq, dtype=float)
    factor = np.sqrt(2.0 * n) / s
    return np.asarray(seq, dtype=float) * factor


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum value of the (full) convolution of seq with itself."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ g)_k == (f * g)[k] for all k.
    Shape of A is (2*n-1, n).  The matrix is dense – for n ≤ 2500 this
    comfortably fits in memory and is fast enough for the limited number
    of LP solves we perform.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # i runs over indices of f that contribute to output k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for each i, the coefficient of g_{k-i} is f_i
        # i runs from i_min … i_max inclusive
        # vectorised filling
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]
    return A


def _solve_convolution_lp(f_norm: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to  (f_norm * g)_k ≤ rhs   for all k
                    g_j ≥ 0
    Returns the optimal g as a NumPy array (not normalised), or None if
    the LP fails.
    """
    n = f_norm.shape[0]
    A = _build_conv_matrix(f_norm)                 # (2n‑1) × n
    # add the non‑negativity constraints g ≥ 0  →  -I·g ≤ 0
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.hstack([np.full(A.shape[0], rhs, dtype=float), np.zeros(n)])

    c = -np.ones(n, dtype=float)   # maximise Σ g_j  ⇔ minimise -Σ g_j
    # use the HiGHS interior‑point solver – fast and reliable
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return np.asarray(res.x, dtype=float)
    return None


def _lp_update_step(seq: list[float]) -> tuple[list[float] | None, float]:
    """
    Perform one LP‑based update.
    Returns a tuple (new_seq, M) where new_seq is the best candidate
    after a line‑search over several step sizes, or None if the LP
    failed.  M is the current max‑convolution of the *normalised* seq.
    """
    n = len(seq)
    f_norm = _normalise(seq)                # sum = sqrt(2n)
    M = _max_convolution(f_norm)            # the bound we must respect

    g_raw = _solve_convolution_lp(f_norm, M)
    if g_raw is None:
        return None, M

    # rescale g so that it also has sum = sqrt(2n)
    g_norm = _normalise(g_raw)

    # try a few mixing parameters and keep the best
    best_local = seq
    best_score = evaluate_sequence(seq)
    # candidate step sizes – larger steps speed up convergence when safe
    for t in (0.01, 0.05, 0.1, 0.2, 0.4):
        mixed = [(1.0 - t) * a + t * b for a, b in zip(seq, g_norm)]
        try:
            score = evaluate_sequence(mixed)
        except Exception:          # very unlikely because of the security checks
            continue
        if score < best_score:
            best_score = score
            best_local = mixed
    return best_local, M


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 990, **kwargs) -> list[float]:
    """
    Search for a sequence giving a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function will stop slightly before this
        limit to avoid forced termination.
    kwargs : dict
        Unused – kept for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best non‑negative height sequence discovered within the time
        budget.
    """
    np.random.seed(seed)
    deadline = time.time() + max(1.0, budget_s) - 5.0   # safety margin

    # -----------------------------------------------------------------
    # Choose a working length.  1500 is a good compromise between
    # solution quality and LP solve time.
    # -----------------------------------------------------------------
    n = 1500

    # -----------------------------------------------------------------
    # Start point: either the provided SOTA sequence (if available) or
    # a simple uniform sequence.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        start = list(height_sequence_1)[:n]
        if len(start) < n:
            start += [0.0] * (n - len(start))
    else:
        start = [1.0] * n

    # Normalise the start so that the first evaluation is meaningful.
    curr = list(_normalise(start))
    best_seq = curr.copy()
    best_score = evaluate_sequence(best_seq)

    it = 0
    # -----------------------------------------------------------------
    # Main optimisation loop – LP based updates plus occasional random
    # nudges when the LP becomes infeasible.
    # -----------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        # 1) LP direction
        new_seq, M = _lp_update_step(curr)

        if new_seq is None:                         # LP failed – random shake
            idx = np.random.randint(n)
            curr[idx] = max(0.0, curr[idx] + np.random.randn() * 0.01)
            continue

        # 2) Evaluate the candidate returned by the line‑search inside
        #    `_lp_update_step`.
        try:
            cur_score = evaluate_sequence(new_seq)
        except Exception:
            # Should never happen – but stay safe.
            cur_score = float("inf")

        # 3) Keep the best ever seen.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            print(f"Iter {it:5d} – new best score {best_score:.6f} (M≈{M:.4f})")

        # 4) Prepare for the next iteration.
        curr = new_seq

        # Occasionally (every 200 iterations) inject a tiny random perturbation
        # to avoid getting stuck in a flat region.
        if it % 200 == 0:
            perturb = np.random.randn(n) * 0.001
            curr = [(max(0.0, a + d)) for a, d in zip(curr, perturb)]

    # -----------------------------------------------------------------
    # Return the best sequence found.
    # -----------------------------------------------------------------
    return best_seq
