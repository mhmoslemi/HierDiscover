"""
Propose a high‑quality step‑function sequence.

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a))².
Scaling a sequence does not change the score, so only the *shape* matters.

For a geometric shape a_i = rⁱ (i = 0,…,n−1) we have
    max(conv) = max( a₀² , n·rⁿ⁻¹ )
and   sum(a) = (1−rⁿ)/(1−r).

Choosing r such that n·rⁿ⁻¹ = 1 (i.e. r = n^{−1/(n−1)}) balances the two
possible maximums and yields the smallest possible max(conv) for a given
length n.  The resulting score is

    score(n) = 2·n / ((1−rⁿ)/(1−r))²  ≈  2·(ln n)² / n,

which decreases as n grows.

The algorithm scans a few moderate lengths, estimates the score analytically,
picks the length with the smallest estimated score, builds the geometric
sequence for that length and returns it as a plain Python list.
All values lie in [0, 1], so they satisfy the non‑negativity and clipping
requirements of `evaluate_sequence`.
"""

import time
import numpy as np


def _estimated_score(n: int) -> float:
    """Analytical estimate of the evaluation score for the optimal geometric
    sequence of length n."""
    if n <= 1:
        return float("inf")
    # r = exp(-log(n)/(n-1))  ==  n^{ -1/(n-1) }
    log_n = np.log(n)
    log_r = -log_n / (n - 1)
    r = np.exp(log_r)

    # sum_{i=0}^{n-1} r^i = (1 - r^n) / (1 - r)
    r_n = np.exp(n * log_r)          # r**n
    one_minus_r = 1.0 - r
    if one_minus_r <= 0.0:
        return float("inf")
    sum_a = (1.0 - r_n) / one_minus_r

    # With the chosen r we have n·r^{n-1}=1, therefore max(conv)=1
    score = 2.0 * n / (sum_a * sum_a)
    return float(score)


def _geometric_sequence(n: int) -> list[float]:
    """Return the optimal geometric sequence of length n:
    a_i = r**i with r = n^{-1/(n-1)}."""
    log_n = np.log(n)
    log_r = -log_n / (n - 1)
    # a_i = exp(i * log_r)
    exponents = np.arange(n, dtype=np.float64) * log_r
    seq = np.exp(exponents)  # values are already in [0, 1]
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence that dramatically improves the
    evaluation metric.  The function scans a small set of lengths,
    picks the one with the lowest analytically estimated score, builds
    the corresponding geometric sequence and returns it.

    Parameters
    ----------
    seed : int
        Random seed (unused but kept for API compatibility).
    budget_s : float
        Time budget in seconds; the search stops early if the budget is
        almost exhausted.
    **kwargs
        Ignored – part of the required signature.

    Returns
    -------
    list[float]
        The chosen sequence of heights.
    """
    np.random.seed(seed)  # not used further, but respects the API
    start_time = time.time()

    # Lengths to consider – they are large enough to beat the target
    # while still allowing the final convolution to finish quickly.
    candidate_lengths = [2000, 5000, 8000, 12000, 20000]

    best_n = None
    best_score = float("inf")

    for n in candidate_lengths:
        # Ensure we stop before the budget expires.
        if time.time() - start_time > budget_s * 0.9:
            break

        est = _estimated_score(n)
        # Pick the smallest estimated score; break ties by smaller n
        if est < best_score:
            best_score = est
            best_n = n

    # Fall back to a modest length if something went wrong.
    if best_n is None:
        best_n = 2000
        best_score = _estimated_score(best_n)

    # Build the final sequence.
    sequence = _geometric_sequence(best_n)

    # Log the choice for debugging / transparency.
    print(
        f"Chosen length n={best_n}, estimated evaluation={best_score:.6f}"
    )
    return sequence
