            continue
        new_seq = seq.copy()
        new_seq[i] = new_i
        new_seq[j] = new_j
        # renormalize to sum 1 (but sum unchanged())
        # evaluate
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        # accept if better or with probability
        if score < current_score or rng.random() < np.exp((current_score - score) / temperature):
            seq = new_seq
            current_score = score
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"Iter {int(time.time()-start_time)} new best {score}")
        # temperature schedule
        temperature *= 0.99999

    print("Final best score:", best_score)
    return best_seq

import time
import numpy as np


def _linear_sequence(length: int) -> np.ndarray:
    """
    Return a normalised sequence a_i = i, i = 0..length‑1,
    scaled so that the sum equals 1.
    """
    # raw values are 0,1,2,…,length‑1
    raw = np.arange(length, dtype=np.float64)
    total = raw.sum()
    # Guard against accidental zero total (cannot happen for length>0)
    if total == 0.0:

    return raw / total
