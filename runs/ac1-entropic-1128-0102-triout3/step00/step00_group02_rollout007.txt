def project_onto_simplex(v, s):
    # v: array, returns w non-negative with sum s and minimal Euclidean distance to v.
    # Implementation: w = max(v - tau, 0)
    # where tau chosen such that sum w = s.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def smoothed_gradient(a, p):
    n = len(a)
    # compute convolution via FFT
    size = 1 << ((2*n-1)-1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    # compute weight
    w = p * np.power(conv, p-1, out=np.empty_like(conv))
    # correlation via FFT
    size2 = 1 << ((n + len(w) - 1)-1).bit_length()
    FA = np.fft.rfft(a, n=size2)
    FW = np.fft.rfft(w, n=size2)
    corr = np.fft.irfft(FA * np.conj(FW), n=size2)[:n + len(w) - 1]
    start = len(w) - 1
    grad = 2.0 * corr[start:start + n]
    return grad
```

def project_to_sum(v, target_sum):
    # v: np array, non-negative not guaranteed.
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total == 0:

        return np.full_like(v, target_sum / v.size)

"""
Propose a candidate step‐function heights sequence that aims to minimise
the evaluation function  2 * n * max(conv(a,a)) / (sum(a)**2).

The algorithm consists of two stages:

1. **Parameter sweep** – for several lengths we generate families of
   exponentially decaying sequences (including convex combinations with the
   uniform distribution).  These break the symmetry of the uniform
   sequence while keeping the largest entry small enough that a₀² does not
   dominate the convolution.  All candidates are evaluated and the best
   (lowest score) is kept.

2. **Smoothed gradient descent** – starting from the best candidate we
   minimise a smooth surrogate of the max‑convolution, namely
   Σ (conv(a,a)ᵖ) with p increasing during the run.  The gradient of this
   surrogate is computed efficiently with FFTs; each step is projected back
   onto the non‑negative simplex (fixed L¹‑norm).  Occasional random
   perturbations help escape shallow local minima.

The whole search runs for the allotted time (budget_s) and returns the
sequence with the smallest observed evaluation value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and renormalise to have the given L1 sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        # fallback: uniform distribution
        return np.full_like(v, target_sum / v.size, dtype=np.float64)
    return v * (target_sum / s)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a * a using a real FFT (returns length 2·len(a)-1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    c = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return c


def _corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation (a ⋆ w), i.e. element‑wise sum_i a_i * w_{i+k}.
    We need the values for k = 0 … len(a)-1.
    """
    n = a.size
    m = w.size
    size = _next_pow2(n + m - 1)
    FA = np.fft.rfft(a, n=size)
    FW = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(FA * np.conj(FW), n=size)[: n + m - 1]
    start = m - 1
    return corr_full[start : start + n]


def _smooth_gradient(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of the surrogate Σ (conv(a,a)ᵖ) w.r.t. a.
    The formula is 2 * (a ⋆ (p·conv(a,a)^{p-1})).
    """
    # convolution a*a
    conv = _conv_fft(a)                     # length 2n-1
    # weight w = p * conv^{p‑1}
    # conv is non‑negative, safe to raise to powers
    w = p * np.power(conv, p - 1, out=np.empty_like(conv))
    # correlation gives the gradient (up to factor 2)
    grad = 2.0 * _corr_fft(a, w)
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (floats) as a candidate solution.
    The routine respects the time budget and uses the strategy described
    above.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Grid search over exponential families
    # ------------------------------------------------------------------
    lengths = [501, 1001, 1501, 2001, 2401]          # diverse sizes
    alphas = np.linspace(0.001, 0.03, 30)            # exponential decay rates
    betas = [0.0, 0.2, 0.5, 0.8, 1.0]                # mixing with uniform
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        uniform = np.full(n, target_sum / n, dtype=np.float64)

        # evaluate the pure uniform sequence (it should give exactly 2.0)
        score = evaluate_sequence(uniform.tolist())
        if score < best_score:
            best_score = score
            best_seq = uniform.copy()

        indices = np.arange(n, dtype=np.float64)

        for alpha in alphas:
            # left‑hand exponential (mass concentrated near the beginning)
            exp_left = np.exp(-alpha * indices)

            # right‑hand exponential (mass concentrated near the end)
            exp_right = exp_left[::-1]

            for beta in betas:
                # convex combination with the uniform distribution
                cand_left = (1.0 - beta) * uniform + beta * exp_left
                cand_right = (1.0 - beta) * uniform + beta * exp_right

                # renormalise to the required L1 norm
                cand_left = _project_to_simplex(cand_left, target_sum)
                cand_right = _project_to_simplex(cand_right, target_sum)

                for cand in (cand_left, cand_right):
                    if time.time() >= deadline:
                        break
                    sc = evaluate_sequence(cand.tolist())
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
        # occasional progress report
        print(f"[grid] length={n} – best score so far: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed gradient descent starting from the best grid point
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of moderate size
        n = 1001
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    a = best_seq.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # schedule for the smoothing exponent p
    p_schedule = [2, 4, 8, 12, 16, 20]
    step_factor = 0.5       # base step magnitude (will be normalised by ||g||)

    it = 0
    while time.time() < deadline:
        # pick a smoothing exponent, slowly increasing as we iterate
        p = p_schedule[min(it // 200, len(p_schedule) - 1)]

        # compute gradient of the smooth surrogate
        g = _smooth_gradient(a, p)
        norm = np.linalg.norm(g)
        if norm < 1e-12:
            break

        # normalised step – the factor 0.5 empirically works well
        eta = step_factor / norm
        a_new = a - eta * g

        # project back onto the feasible simplex
        a = _project_to_simplex(a_new, target_sum)

        # occasional random jitter to escape plateaus
        if it % 100 == 99:
            jitter = 0.02 * target_sum * np.random.randn(n)
            a = _project_to_simplex(a + jitter, target_sum)

        # evaluate and keep the best ever seen
        if it % 10 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[gd] iter={it} – new best score: {best_score:.6f}")

        it += 1

    print(f"Final best score: {best_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
