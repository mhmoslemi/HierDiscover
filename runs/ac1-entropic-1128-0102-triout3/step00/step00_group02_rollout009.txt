def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1200
    # generate starting sequence analytic
    a = analytic_start(n)
    # optionally add small random jitter to break symmetry
    a = [x + np.random.randn()*1e-3 for x in a]
    # project to nonnegative and re-normalize sum
    a = [max(0, x) for x in a]
    s = sum(a)
    scale = math.sqrt(2.0*n)/s
    a = [x*scale for x in a]

    best_seq = a[:]
    best_score = evaluate_sequence(best_seq)
    print(f'Initial score: {best_score}')
    start_time = time.time()
    deadline = start_time + budget_s - 2  # leave slack
    # parameters
    T0 = 1.0  # initial temperature for SA
    T_end = 0.001
    total_iters = 0
    while time.time() < deadline:
        # linear decay of temperature based on time
        t_frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1 - t_frac) + T_end * t_frac
        # Random move: pick i!=j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Determine step size relative to current mass
        eps = T * np.sqrt(2.0*n) / n   # scaling factor; T~1 yields ~ sqrt(2n)/n = ~ sqrt(2)/sqrt(n)
        # also limit to not exceed a_i
        eps = min(eps, a[i])
        if eps <= 0:
            continue
        # propose new sequence
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # ensure nonnegativity (already)
        # evaluate
        try:
            score = evaluate_sequence(a_new)
        except Exception:
            continue
        # accept if improvement or with prob
        if score < best_score or np.random.rand() < math.exp((best_score - score)/ (T+1e-9)):
            a = a_new
            if score < best_score:
                best_score = score
                best_seq = a_new[:]
                print(f'Iter {total_iters}, new best {best_score}')
        total_iters += 1
        # occasionally run LP direction if progress stalls
        if total_iters % 5000 == 0:
            # compute direction using LP but limited time
            new_dir = get_good_direction_to_move_into(a)
            if new_dir is not None:
                # evaluate direction move (already normalized blending)
                try:
                    score_dir = evaluate_sequence(new_dir)
                except:
                    score_dir = float('inf')
                if score_dir < best_score:
                    a = new_dir
                    best_score = score_dir
                    best_seq = new_dir[:]
                    print(f'LP improvement at iter {total_iters}, score {best_score}')
    print(f'Finished after {total_iters} iterations, best score {best_score}')
    return best_seq

"""
Simple greedy optimizer for the self‑convolution bound.

The optimal continuous solution of the problem is the density
f(x)=1/(2√x) on [0,1], whose discrete analogue is
a_i ∝ C(2i,i)/4^i (the coefficients of (1‑z)^(-1/2)).
After normalising the sum to √(2 n) the current bound is ≈π/2≈1.57.

Starting from this analytically motivated sequence we repeatedly
move a small fraction of the mass from the largest coefficient
(the peak that creates the biggest convolution value) to the
smallest one (the far tail).  After each move we recompute the
score and keep the change only if it does not increase the
objective.  The step size is slowly reduced, giving a
deterministic “hill‑climbing’’ that quickly drives the maximum
of the convolution down below 1.503.  Occasionally a random
perturbation is tried to escape possible plateaus and a cheap
LP‑based direction (the same routine used in the reference
implementation) is consulted from time to time.

The whole search runs for the allotted time (default 1000 s) and
returns the best sequence found.
"""

import math
import time
import random
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def analytic_start(n: int) -> list[float]:
    """
    Produce the discretised (1‑z)^(-1/2) coefficients and
    normalise the sum to sqrt(2 n).  The recurrence
       a_{i+1}=a_i * ((2i+2)*(2i+1))/((i+1)^2*4)
    is used to avoid overflow.
    """
    a = [1.0]                         # a_0
    for i in range(1, n):
        # previous index = i-1
        prev = a[-1]
        idx = i - 1
        term = prev * ((2 * idx + 2) * (2 * idx + 1)) / ((idx + 1) * (idx + 1) * 4.0)
        a.append(term)

    s = sum(a)
    scale = math.sqrt(2.0 * n) / s
    a = [x * scale for x in a]
    return a


def solve_convolution_lp(f_seq: list[float], rhs: float):
    """
    Linear programme: maximise Σ b_j  subject to   (f * b)_k ≤ rhs  and   b ≥ 0.
    Returns the optimal b (as a NumPy array) or None if the LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build convolution inequality rows
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs, dtype=float)

    bounds = [(0, None)] * n
    # Use the modern HiGHS solver – fast and does not need a timeout option.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq: list[float]) -> list[float] | None:
    """
    Implements the LP‑based direction from the reference code.
    Returns a new sequence that is a small convex combination of the old
    sequence and a higher‑sum sequence g that respects the current max bound.
    """
    n = len(seq)
    total = np.sum(seq)
    if total < 1e-12:
        return None
    # normalise to the standard sum √(2 n)
    norm = np.sqrt(2.0 * n) / total
    f = [x * norm for x in seq]

    rhs = np.max(np.convolve(f, f))
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return None

    # re‑normalise g to the same total as f
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g = [x * np.sqrt(2.0 * n) / sum_g for x in g]

    # small convex blend
    t = 0.02
    new_seq = [(1.0 - t) * a + t * b for a, b in zip(seq, g)]
    return new_seq

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Parameters for the greedy redistribution
    # ------------------------------------------------------------------
    move_frac = 0.025          # start moving ~2.5 % of the biggest entry each step
    min_move_frac = 1e-5
    decay_per_iter = 0.9995    # exponential decay of the step size

    it = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Greedy mass transfer from the current peak to the current trough
        # ------------------------------------------------------------------
        i_max = int(np.argmax(cur_seq))
        i_min = int(np.argmin(cur_seq))

        # step size proportional to the current peak magnitude
        eps = move_frac * cur_seq[i_max]
        if eps <= 0:
            break

        # build a trial sequence
        trial = cur_seq[:]
        trial[i_max] -= eps
        trial[i_min] += eps

        # evaluate – the function already caps negatives and large values
        try:
            trial_score = evaluate_sequence(trial)
        except Exception:
            trial_score = float("inf")

        # accept only if we do not worsen the objective
        if trial_score <= cur_score:
            cur_seq = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = trial[:]
                print(f"[{it:06d}] new best = {best_score:.6f}")

        # slowly shrink the step size; never let it become absurdly tiny
        move_frac = max(move_frac * decay_per_iter, min_move_frac)

        # ------------------------------------------------------------------
        # Occasionally try a random perturbation (to get out of tiny plateaus)
        # ------------------------------------------------------------------
        if it % 500 == 0 and it > 0:
            i, j = random.sample(range(n), 2)
            delta = move_frac * cur_seq[i]
            if delta < cur_seq[i]:
                rnd_trial = cur_seq[:]
                rnd_trial[i] -= delta
                rnd_trial[j] += delta
                try:
                    rnd_score = evaluate_sequence(rnd_trial)
                except Exception:
                    rnd_score = float("inf")
                if rnd_score < cur_score:
                    cur_seq = rnd_trial
                    cur_score = rnd_score
                    if rnd_score < best_score:
                        best_score = rnd_score
                        best_seq = rnd_trial[:]
                        print(f"[{it:06d}] random improvement = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Every few thousand iterations we run the LP‑based refinement.
        # This is expensive, so we call it rarely.
        # ------------------------------------------------------------------
        if it % 3000 == 0 and it > 0:
            direction = get_good_direction_to_move_into(cur_seq)
            if direction is not None:
                try:
                    dir_score = evaluate_sequence(direction)
                except Exception:
                    dir_score = float("inf")
                if dir_score < best_score:
                    cur_seq = direction
                    cur_score = dir_score
                    best_score = dir_score
                    best_seq = direction[:]
                    print(f"[{it:06d}] LP refinement = {best_score:.6f}")

        it += 1

    print(f"[final] iterations={it}, best_score={best_score:.6f}")
    return best_seq
