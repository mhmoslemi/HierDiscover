"""
Propose a step‑function that lowers the evaluation score.

The evaluation is 2·n·max(f*f) / (sum(f))².
For a decreasing power‑law  a[i] = (i+1)^{-α}   the self‑convolution becomes
almost flat and the ratio  max(f*f)/(sum(f))² can be driven below the uniform
value (α=0 gives the classic bound 2).  Scanning a few α∈[0.45,0.8] is enough to
beat the current target 1.503 for the required length n=2401.

The function builds several candidates, evaluates them with the provided
``evaluate_sequence`` and returns the best one.
"""

import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a decreasing power‑law sequence a_i = (i+1)^{-α}.
    A short scan over α finds the exponent that gives the smallest
    evaluation score, then returns that sequence.

    Parameters
    ----------
    seed : int
        Random seed (unused, kept for compatibility with the framework).
    budget_s : float
        Time budget in seconds – this simple scan finishes well within the limit.
    **kwargs
        Ignored.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    # length that was used in the reference construction
    n = 2401

    # candidate exponents – dense enough to locate a good value quickly
    alphas = np.linspace(0.45, 0.80, 36)   # step ≈ 0.01

    best_seq = None
    best_score = float('inf')
    best_alpha = None

    # evaluate every candidate
    for alpha in alphas:
        # a_i = (i+1)^(-alpha)  (i starts at 0)
        seq = [(i + 1) ** (-alpha) for i in range(n)]

        # use the supplied evaluation routine
        score = evaluate_sequence(seq)

        print(f"alpha={alpha:.3f} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            best_alpha = alpha

    # report the winner
    print(f"Chosen α = {best_alpha:.3f} with score = {best_score:.6f}")

    # the list is already a plain Python list of floats, ready to be returned
    return best_seq
