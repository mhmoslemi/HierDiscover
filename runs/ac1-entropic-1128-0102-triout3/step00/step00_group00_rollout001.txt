def hill_climb_search(seq, time_limit, start_time):
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)
    seq = np.asarray(seq, dtype=np.float64)
    sum_seq = seq.sum()
    if sum_seq == 0:
        seq[:] = target_sum / n
    else:
        seq *= target_sum / sum_seq
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    # initial print
    print(f"Start hill climb: n={n}, init_score={best_score:.6f}")
    deadline = time.time() + time_limit
    iter_no = 0
    no_improve = 0
    while time.time() < deadline:
        iter_no += 1
        # compute convolution and locate max
        conv = max_conv_fft(seq)
        max_idx = int(np.argmax(conv))
        # find contribution products
        prod = contribution_for_max(seq, max_idx)
        # source index as high contribution mass
        if np.any(prod > 0):
            src = int(np.argmax(prod))
        else:
            src = np.random.randint(n)
        # dest index as low value (avoid src)
        dest = int(np.argmin(seq))
        if dest == src:
            dest = (dest + 1) % n
        # epsilon fraction of src value
        eps = seq[src] * np.random.uniform(1e-4, 0.02)
        if eps <= 0:
            continue
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[dest] += eps
        # renorm just in case of floating drift
        new_seq *= target_sum / new_seq.sum()
        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            no_improve = 0
            print(f"[{time.time() - start_time:.1f}s] iter {iter_no}: improved score={best_score:.6f}")
        else:
            # occasionally accept worse move to escape local minima
            no_improve += 1
            if no_improve > 2000:
                # random perturb whole vector a bit
                noise = np.random.randn(n) * (seq.mean() * 0.001)
                new_seq = seq + noise

                    seq = new_seq

"""
Propose a sequence of non‑negative heights that (hopefully) beats the
current bound 2.0.  The method is a simple “hill‑climbing” optimiser
that repeatedly moves a small amount of mass from a location that
contributes most to the current maximal convolution entry to a location
with a very small value.  The sum of the sequence is kept constant
(√(2 n)) so the evaluation function is invariant under scaling.
Multiple initial shapes (the provided SOTA sequence, a sine window,
a Gaussian window and a random vector) are tried; each gets a share of
the time budget.  Fast FFT‑based convolutions are used to keep the
iterations cheap.
"""

import numpy as np
import time
import random
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _convolution_fft(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution of seq with itself using an FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _max_conv_and_index(seq: np.ndarray):
    """Compute convolution, its maximum value and the index where it occurs."""
    conv = _convolution_fft(seq)
    idx = int(np.argmax(conv))
    return conv, conv[idx], idx


def _contributions_for_max(seq: np.ndarray, max_idx: int) -> np.ndarray:
    """
    For a given max index of the convolution, return an array where entry i
    contains a_i * a_{max_idx-i} (zero if the partner index is out of range).
    """
    n = seq.shape[0]
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    prod = np.zeros(n, dtype=seq.dtype)
    prod[mask] = seq[i[mask]] * seq[j[mask]]
    return prod


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale the vector so that its L1‑norm equals √(2 n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0:
        seq[:] = target / n
    else:
        seq *= target / s
    return seq


def _hill_climb_search(start_seq: np.ndarray, time_limit: float,
                       global_start: float):
    """
    Simple hill‑climbing / simulated‑annealing style optimisation.
    Returns the best sequence found and its evaluation score.
    """
    n = start_seq.shape[0]
    seq = _normalize(start_seq.astype(np.float64, copy=True))
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    print(f"[{time.time() - global_start:.1f}s] start hill‑climb: n={n},"
          f" init_score={best_score:.6f}")

    deadline = time.time() + time_limit
    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and its peak
        conv, _, max_idx = _max_conv_and_index(seq)

        # contributions of each position to that peak
        contrib = _contributions_for_max(seq, max_idx)

        # choose a source index that contributes heavily
        if np.any(contrib > 0):
            src = int(np.argmax(contrib))
        else:
            src = np.random.randint(n)

        # choose a destination that currently holds a very small value
        dest = int(np.argmin(seq))
        if dest == src:
            dest = (dest + 1) % n

        # move a modest fraction of the source mass
        eps = seq[src] * np.random.uniform(1e-4, 0.02)
        if eps <= 0:
            continue

        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[dest] += eps

        # renormalise (protect against tiny drift)
        new_seq = _normalize(new_seq)

        new_score = evaluate_sequence(new_seq.tolist())

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            no_improve = 0
            print(f"[{time.time() - global_start:.1f}s] iter {it}:"
                  f" improved to {best_score:.6f}")
        else:
            no_improve += 1

        # occasional random shake to escape plateaus
        if no_improve > 2500:
            noise = np.random.randn(n) * (seq.mean() * 0.001)
            shaken = seq + noise
            shaken = np.clip(shaken, 0, None)
            shaken = _normalize(shaken)
            shaken_score = evaluate_sequence(shaken.tolist())
            if shaken_score < best_score:
                best_score = shaken_score
                best_seq = shaken.copy()
                seq = shaken
                print(f"[{time.time() - global_start:.1f}s] random shake"
                      f" improved to {best_score:.6f}")
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a good height sequence within the given time budget.

    The routine builds several plausible initial shapes (the provided
    SOTA sequence, a sine window, a Gaussian window and a random vector)
    and runs a lightweight hill‑climbing optimisation on each,
    returning the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin so we don't overrun the limit
    deadline = start_time + budget_s - 3.0

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Assemble a list of starting sequences
    # ------------------------------------------------------------------
    candidates = []

    # 1) use the previously discovered construction, if available
    if "height_sequence_1" in globals():
        candidates.append(np.array(height_sequence_1, dtype=float))

    # Determine a base length (use the length of the first candidate if we have one)
    base_n = candidates[0].shape[0] if candidates else 3000
    base_n = max(base_n, 2000)          # ensure a reasonable size

    # 2) Sine window (positive on (0,π))
    sine = np.sin(np.pi * (np.arange(1, base_n + 1)) / (base_n + 1))
    candidates.append(sine)

    # 3) Gaussian‑like window
    x = np.arange(base_n)
    sigma = base_n / 8.0
    gauss = np.exp(-0.5 * ((x - (base_n - 1) / 2.0) / sigma) ** 2)
    candidates.append(gauss)

    # 4) Pure random positive vector
    candidates.append(np.random.rand(base_n))

    # ------------------------------------------------------------------
    # Optimise each candidate for a share of the remaining time
    # ------------------------------------------------------------------
    for init_seq in candidates:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # Give each new start at most 30 % of the remaining budget,
        # but not less than a few seconds.
        alloc = max(5.0, remaining * 0.30)

        best_seq, best_score = _hill_climb_search(
            init_seq, alloc, start_time
        )

        if best_score < global_best_score:
            global_best_score = best_score
            global_best_seq = best_seq.copy()
            print(
                f"[{time.time() - start_time:.1f}s] New global best:"
                f" {global_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – in the unlikely case that nothing was produced
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2500
        global_best_seq = np.full(
            n, math.sqrt(2.0 * n) / n, dtype=float
        )
        global_best_score = evaluate_sequence(global_best_seq.tolist())

    # Make absolutely sure the sum matches the required normalisation
    n = len(global_best_seq)
    target = math.sqrt(2.0 * n)
    global_best_seq = np.array(global_best_seq, dtype=float)
    if global_best_seq.sum() != 0:
        global_best_seq *= target / global_best_seq.sum()

    # Return as a plain Python list (as required by the evaluator)
    return global_best_seq.tolist()
