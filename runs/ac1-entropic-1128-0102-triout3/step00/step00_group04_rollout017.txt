import numpy as np, time, math
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum
    # Build constraints a_ub * b <= b_ub
    rows = []
    bounds = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)
    # nonneg constraints
    rows.append(-np.eye(n))
    bounds.extend([0.0]*n)
    A_ub = np.vstack(rows)
    b_ub = np.array(bounds)
    # set method maybe 'highs' default
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None
```

```python
def gradient_step(a, alpha, lr):
    conv = np.convolve(a, a)
    shift = conv.max()
    exp_vals = np.exp(alpha * (conv - shift))
    w = exp_vals / exp_vals.sum()
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    grad = conv_rev[2*len(a)-2 : len(a)-2 : -1]

    # Update direction (descend)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new, conv.max()
```

"""
Propose a candidate sequence that (hopefully) beats the uniform bound 2.0.

The idea is simple yet powerful:
* Generate many random non‑negative vectors whose entries sum to 1.
  (Dirichlet / Gamma draws give a wide variety of shapes.)
* Randomly permute the entries – the permutation changes the additive
  convolution a*a, because each sum‑index receives a different collection
  of products a_i·a_j.
* Evaluate the maximal entry of the self‑convolution; the score is
  2·n·max(a*a) / (sum(a)**2).  For the uniform vector this score is exactly 2.
* Keep the best vector seen so far and try to improve it locally by swapping
  pairs of positions (a cheap hill‑climbing step).
* The whole search runs until the time budget is exhausted (≈ 1000 s).

The algorithm is completely self‑contained: it only uses NumPy and the
provided ``evaluate_sequence`` for the final verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (exactly the same formula used in evaluate_sequence)
def _fast_evaluate(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative sequence."""
    total = np.sum(seq)
    if total < 0.01:          # matches the safety check in evaluate_sequence
        return np.inf
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    n = seq.size
    return float(2.0 * n * max_conv / (total * total))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine respects the time limit and returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # tiny safety margin

    # start from the obvious uniform sequence (score = 2.0)
    n = 800                                   # moderate length – fast enough
    best_seq = np.full(n, 1.0 / n, dtype=float)
    best_score = _fast_evaluate(best_seq)
    print(f"[init] uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main random‑search loop
    #
    # We sample Dirichlet vectors with a random concentration parameter.
    # After each sample we shuffle the order (the permutation is what really
    # matters for the convolution).  Whenever a new best is found we try a
    # short hill‑climbing phase swapping two positions.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1. draw a random Dirichlet vector
        alpha = np.random.uniform(0.2, 2.0)                 # controls sparsity
        gamma_samples = np.random.gamma(alpha, 1.0, size=n)
        seq = gamma_samples / gamma_samples.sum()          # normalised sum = 1

        # 2. random permutation – the ordering heavily influences the max
        np.random.shuffle(seq)

        # 3. evaluate
        score = _fast_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] new best {best_score:.6f} (t={time.time():.2f})")

            # --------------------------------------------------------------
            # 4. local hill‑climbing: try a few random swaps and keep them if
            #    they improve the score.
            # --------------------------------------------------------------
            improved = True
            while improved and time.time() < deadline:
                improved = False
                # a small fixed number of swap attempts per improvement round
                for _ in range(30):
                    i, j = np.random.randint(n, size=2)
                    cand = best_seq.copy()
                    cand[i], cand[j] = cand[j], cand[i]          # swap two entries
                    cand_score = _fast_evaluate(cand)
                    if cand_score < best_score:
                        best_seq = cand
                        best_score = cand_score
                        improved = True
                        # we break to restart the inner loop with the new candidate
                        break

    # ------------------------------------------------------------------
    # Final safety: make sure the returned list satisfies the security checks
    # of evaluate_sequence (non‑negative, not too small sum, values capped).
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:
        # use the provided function for ultimate verification (and for the
        # submission system it will be called again anyway)
        final_score = evaluate_sequence(final_list)
        print(f"[final] best score reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        # If something went wrong we still return the best we have.
        print("Warning: evaluate_sequence raised an exception:", exc)

    return final_list
