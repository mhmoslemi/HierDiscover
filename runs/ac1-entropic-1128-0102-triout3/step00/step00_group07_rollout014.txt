def evaluate(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # convolution via FFT
    size = 2*n - 1
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[:size]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a**2))

  n = len(a)
  i = np.random.randint(n)
  j = p - i
  if j <0 or j >= n or i==j:
    return None
  delta = min(a[i], a[j]) * np.random.rand() * 0.5 # up to half of smaller.
  # ensure delta positive small
  # List indices to increase
  k,l = np.random.choice(n, size=2, replace=False)

  # allocate delta to each:
  new_a = a.copy()

  new_a[j] -= delta

  new_a[l] += delta

"""
Propose a sequence of non‑negative heights that minimizes the
``evaluate_sequence`` objective.

The algorithm is a lightweight implementation of the method described
in the statement.  It repeatedly:

1. Normalises the current sequence to the target ℓ¹‑norm
   √(2·n) (the evaluation is scale‑invariant, this normalisation
   simply removes a degree of freedom).
2. Computes the maximal self‑convolution value M = ‖f∗f‖∞.
3. Solves a linear programme that maximises the ℓ¹‑norm of a new
   sequence g under the linear constraints ‖f∗g‖∞ ≤ M and g ≥ 0.
   This step finds a direction that can increase the total mass
   without increasing the worst convolution value.
4. Takes a small convex combination f←(1‑t)f + t·g (t≈0.02–0.05),
   re‑normalises, and accepts the new f if the evaluation improves.
5. If the LP fails or the move is not improving, a tiny random
   perturbation is performed to escape possible stalls.

The loop runs until the allotted time expires (default 1000 s).  The
best sequence encountered is returned.  The implementation uses FFT‑based
convolution for speed and SciPy’s high‑performance “highs” LP solver.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def _fft_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real‑FFT (fast for n up to a few‑thousand)."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv_full = np.fft.irfft(A * A, n=N)[:size]
    return conv_full


def _fast_evaluate(a: np.ndarray) -> float:
    """Same objective as ``evaluate_sequence`` but faster (no safety checks)."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:       # guard against division by zero (mirrors the original)
        return float("inf")
    conv = _fft_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _solve_lp(f: np.ndarray, bound: float, time_limit: float = 2.0):
    """
    Solve   max 1ᵀg   s.t.  (f∗g)(k) ≤ bound  for all k,  g ≥ 0.
    Returns the optimal g (as a NumPy array) or None if the LP fails.
    """
    n = f.shape[0]
    # Build the constraint matrix A_ub·g ≤ b_ub
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in [i_min, i_max] we have term f[i] * g[k-i]
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])          # non‑negativity: -g ≤ 0
    b_ub = np.hstack([np.full(2 * n - 1, bound), np.zeros(n)])

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n)

    # Use the high‑performance HiGHS solver; enforce a modest time limit.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),          # explicitly enforce g ≥ 0
            method="highs",
            options={"time_limit": time_limit, "presolve": True},
        )
    except Exception:               # any numerical problem → treat as failure
        return None

    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    The function runs for roughly ``budget_s`` seconds and returns the best
    sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # -----------------------------------------------------------------
    # Parameters (feel free to adjust)
    # -----------------------------------------------------------------
    n = 1800                       # length of the sequence – a good trade‑off
    target_norm = math.sqrt(2.0 * n)   # √(2·n) – the normalisation used in the paper
    blend_factor = 0.03            # initial mixing weight t
    min_blend = 0.004
    blend_decay = 0.94             # decay of t every few iterations

    # -----------------------------------------------------------------
    # Initial random feasible point (positive, normalised)
    # -----------------------------------------------------------------
    cur = np.random.rand(n)
    cur = cur / cur.sum() * target_norm
    cur_score = _fast_evaluate(cur)

    best_seq = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, score={cur_score:.6f}")

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- compute current bound M = max(f∗f) -----------------------
        conv_cur = _fft_convolution(cur)
        M = conv_cur.max()

        # ---- linear‑programming step --------------------------------
        g0 = _solve_lp(cur, M, time_limit=1.5)

        if g0 is None or np.allclose(g0, 0):
            # LP failed – fall back to a tiny random jitter
            i, j = np.random.randint(n, size=2)
            delta = cur[i] * 0.02
            cur[i] = max(0.0, cur[i] - delta)
            cur[j] += delta
            cur = cur / cur.sum() * target_norm
            continue

        # ---- normalise g0 to the target ℓ¹‑norm --------------------
        g = g0 * (target_norm / g0.sum())

        # ---- convex combination (small step) -----------------------
        new_seq = (1.0 - blend_factor) * cur + blend_factor * g
        # re‑normalise (tiny numerical drift may appear)
        new_seq = new_seq / new_seq.sum() * target_norm

        # ---- evaluate ------------------------------------------------
        new_score = _fast_evaluate(new_seq)

        # ---- accept if we improve -----------------------------------
        if new_score < cur_score:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                elapsed = time.time() - start_time
                print(f"[{it:05d}] new best {best_score:.6f} (t={elapsed:.1f}s)")

        # occasional stochastic acceptance to avoid tiny plateaus
        elif np.random.rand() < 1e-4:
            cur = new_seq
            cur_score = new_score

        # decay the blending factor slowly (helps fine‑tuning later)
        if it % 150 == 0:
            blend_factor = max(min_blend, blend_factor * blend_decay)

    # -----------------------------------------------------------------
    # Return the best sequence we have found (as a plain Python list)
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – in the unlikely case nothing was found

            return list(height_sequence_1)          # type: ignore
        except Exception:
            return [1.0]

    # Clip to the problem's hard limits (0 ≤ height ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
