"""
propose_candidate
-----------------
Search for a step‑function (non‑negative list of heights) that lowers the
evaluation metric

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution performed by ``np.convolve``.
The metric is scale‑invariant, therefore we can fix the L1‑norm of the
candidate to ``sqrt(2*n)`` without loss of generality.

The algorithm is a fast implementation of the iterative scheme described
in the statement:

1.  Initialise a *spiky* sequence : a single entry holds the whole mass
    (the L1‑norm) and all other entries are zero.
    This gives a very large convolution peak ``M`` (``M = 2*n``).

2.  For the current sequence ``f`` (always normalised to
    ``sqrt(2*n)``) compute ``M = max(f * f)``.

3.  Solve a linear programme

        maximise   sum(g_i)
        subject to  (f * g)_k ≤ M   for every shift k
                    g_i ≥ 0

    The constraints are linear in the unknown vector ``g`` because
    ``f`` is fixed.  The LP gives a feasible ``g`` whose L1‑norm is as
    large as possible while respecting the same convolution bound ``M``.

4.  Rescale ``g`` to the required norm ``sqrt(2*n)`` and replace a
    fraction ``t`` of ``f`` by this new vector:
        f ← (1‑t)·f + t·g .
    This convex combination strictly reduces the convolution peak unless
    ``f`` is already optimal.

5.  Repeat steps 2‑4 a few times (the bound drops quickly).  Keep the
    best sequence discovered (the one with the smallest evaluation
    value).

The LP is small enough for ``n ≤ 800`` to be solved many times within the
time budget.  After the LP‑driven phase a short hill‑climbing phase
applies random mass‑transfer moves to squeeze a little more improvement.

The whole search finishes well before the 1000‑second deadline and
returns the best sequence found.
"""

def max_convolution(seq: np.ndarray) -> float:
    """Return max element of the linear convolution seq * seq."""
    return float(np.max(np.convolve(seq, seq)))


def solve_convolution_lp(f_seq: np.ndarray, bound: float):
    """
    Solve the LP:
        maximise   sum(g_i)
        subject to  (f * g)_k ≤ bound   for all shifts k,
                    g_i ≥ 0.
    Returns the optimal g (as a NumPy array) or None if the LP fails.
    """
    n = f_seq.shape[0]
    # objective: minimise -sum(g)  (i.e. maximise sum(g))
    c = -np.ones(n)

    # Build the constraint matrix A_ub (size (2n-1) × n).
    # Row k corresponds to the inequality Σ_i f_i * g_{k-i} ≤ bound.
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for i in range(n):
        # f_i contributes to rows k = i + j  (j = 0 … n-1)
        # i.e. to columns j where 0 ≤ k-i < n  →  j = k-i
        # We fill the whole matrix by the straightforward double loop
        # – it is cheap for the modest n we use (≤ 800).
        for k in range(rows):
            j = k - i
            if 0 <= j < n:
                A[k, j] = f_seq[i]

    b = np.full(rows, bound, dtype=np.float64)

    # variable bounds: g_i ≥ 0, no explicit upper bound
    bounds = [(0, None) for _ in range(n)]

    # Use the HiGHS interior‑point LP solver (default in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    evaluation value than the current best bound (2.0).  The function runs
    for at most ``budget_s`` seconds and always returns a valid sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5          # keep a safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) LP‑driven refinement for several problem sizes
    # ------------------------------------------------------------------
    for n in (200, 400, 600, 800):
        if time.time() > deadline:
            break

        # target L1‑norm (the normalisation used in the statement)
        target_norm = math.sqrt(2.0 * n)

        # start from a single spike at position 0 (all mass concentrated)
        f = np.zeros(n, dtype=np.float64)
        f[0] = target_norm

        # initial score
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[Init] n={n}, spike score={cur_score:.6f}")

        # iterative improvement – a few dozen LP solves are enough
        t = 0.5                     # mixing ratio (how much of the new g we keep)
        max_iters = 40
        for it in range(max_iters):
            if time.time() > deadline:
                break

            M = max_convolution(f)          # current convolution peak
            g = solve_convolution_lp(f, M)
            if g is None:
                # LP failure – abort this size
                print(f"[LP] n={n} iteration {it} – LP failed")
                break

            # Rescale g to have the required L1‑norm
            sum_g = np.sum(g)
            if sum_g == 0:
                break
            g = g * (target_norm / sum_g)

            # Convex combination
            f = (1.0 - t) * f + t * g

            # Check new score
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[Update] n={n} it={it:02d} score={cur_score:.6f}")

            # Early stop if we have already crossed the target
            if best_score < 1.6:
                break

    # ------------------------------------------------------------------
    # 2) Light random hill‑climbing around the best found sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – should never happen, but create a tiny uniform vector
        best_seq = np.full(100, 1.0 / 100.0)

    seq = best_seq.copy()
    n = seq.shape[0]

    # small random perturbations (mass transfers)
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of mass from i to j
        eps = 0.02 * np.random.rand()          # up to 2 % of the source
        delta = eps * seq[i]
        if delta <= 0:
            continue
        seq[i] -= delta
        seq[j] += delta

        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[HillClimb] improved score={cur_score:.6f}")

        # occasionally reset step size or re‑randomise a tiny bit
        if np.random.rand() < 0.001:
            # tiny random shake to escape possible plateaus
            seq += 0.001 * np.random.randn(n)
            seq = np.clip(seq, 0.0, None)

    # Ensure the returned list satisfies the security checks
    final = [float(x) for x in best_seq.tolist()]
    return final
