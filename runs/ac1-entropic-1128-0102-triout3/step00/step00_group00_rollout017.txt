    best_seq = None

    # Function to evaluate and track best
    def try_seq(seq):
        nonlocal best_seq, best_score
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best score {best_score:.6f} length {len(seq)}")
        return score

    # generate some initial candidates (random epsilon lengths)
