```
def evaluate_sequence(sequence):
    # security checks omitted
    sequence = [float(x) for x in sequence]
    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]
    # Protect numbers too large
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

"""
Gradient–descent search for a step‑function that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works on the probability simplex (non‑negative entries, sum=1)
because the score is homogeneous of degree 0 – scaling does not change it.
It repeatedly flattens the self‑convolution by minimizing a smooth
approximation of the max (log‑sum‑exp) and projects back onto the simplex.
A few random jitters and occasional restarts help escape local minima.

The function `propose_candidate` returns the best sequence found within the
time budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _fft_len_for_conv(len_a: int, len_b: int) -> int:
    """Return a power‑of‑2 length that is safe for an FFT of two vectors."""
    needed = len_a + len_b - 1
    return 1 << (needed - 1).bit_length()


def _full_convolve_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of two real vectors using a real FFT.
    Returns the first len(x)+len(y)-1 entries (the full linear convolution).
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    z = np.fft.irfft(X * Y, fft_len)
    return z[: len(x) + len(y) - 1]


def _self_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution a*a (full length 2·n‑1) using the pre‑computed FFT length."""
    X = np.fft.rfft(a, fft_len)
    zz = np.fft.irfft(X * X, fft_len)
    n = a.shape[0]
    return zz[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Stable soft‑max of eps·conv (produces a probability vector)."""
    max_c = np.max(conv)
    # shift for numerical stability
    exp_vals = np.exp(eps * (conv - max_c))
    total = np.sum(exp_vals)
    if total == 0:
        # fallback to uniform – should not happen
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / total


def _gradient_softmax(a: np.ndarray, weights: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max (soft‑max) objective:
        g_i = 2 * Σ_j a_j * weights_{i+j}
    Computed efficiently with a single FFT.
    """
    n = a.shape[0]
    # reverse the weights for the correlation‑like convolution
    w_rev = weights[::-1]

    # convolution a * w_rev  (length ≈ 3n‑2)
    conv_full = _full_convolve_fft(a, w_rev, fft_len)   # length n+(2n‑1)-1 = 3n‑2

    # the part we need lives at indices n‑1 … 2n‑2 (inclusive) – exactly n numbers
    slice_part = conv_full[n - 1 : 2 * n - 1]          # shape (n,)
    # mapping i → 2n‑2‑i yields the reversed slice
    grad = 2.0 * slice_part[::-1]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1 (simplex projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:                      # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _score_from_vec(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the same value that `evaluate_sequence` would return for a vector
    that already satisfies the non‑negative, sum‑≥0.01 constraints.
    We assume sum(a) == 1, therefore the denominator disappears.
    """
    n = a.shape[0]
    conv = _self_convolve(a, fft_len)
    max_b = np.max(conv)
    return float(2.0 * n * max_b)   # denominator = (sum a)² = 1

# ------------------------------------------------------------

    score.  The routine runs for up to *budget_s* seconds and returns the
    best sequence it discovered (as a plain Python list of floats).
    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------------
    # 1. Initialise a starting vector
    # --------------------------------------------------------------------

        # Use the provided construction if it exists (may be long)
        start_vec = np.asarray(height_sequence_1, dtype=float)
        have_start = True
    except Exception:

    MIN_LEN = 800
    MAX_LEN = 3000

    else:

        target_len = np.random.randint(MIN_LEN, MAX_LEN + 1)
        seq = np.random.rand(target_len)

    # Normalise onto the simplex
    seq = _project_simplex(seq)
    n = seq.shape[0]

    # Pre‑computed FFT length that works for both self‑convolution
    # and the gradient convolution (needs 3·n‑2 points)
    fft_len = _fft_len_for_conv(n, 2 * n - 1)

    # --------------------------------------------------------------------
    # 2. Hyper‑parameters
    # --------------------------------------------------------------------
    eps = 1.0               # smoothing for soft‑max (starts loose)
    lr = 0.25               # learning rate
    eps_decay = 0.9995
    lr_decay = 0.999
    eps_min = 0.02
    lr_min = 5e-4

    # --------------------------------------------------------------------
    # 3. Tracking best solution
    # --------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = _score_from_vec(seq, fft_len)
    print(f"[init] length={n}, score={best_score:.6f}")

    it = 0
    no_improve_counter = 0

    # --------------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # --- compute current self‑convolution (full) --------------------
        conv = _self_convolve(seq, fft_len)

        # --- soft‑max weighted gradient of the max -----------------------
        weights = _softmax_weights(conv, eps)
        grad = _gradient_softmax(seq, weights, fft_len)

        # --- blend with the true sub‑gradient of the current max offset
        max_idx = int(np.argmax(conv))
        subgrad = np.zeros_like(seq)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # indices i in [lo,hi] contribute with partner j = max_idx-i
        subgrad[lo : hi + 1] = 2.0 * seq[max_idx - np.arange(lo, hi + 1)]
        grad = 0.7 * grad + 0.3 * subgrad   # simple convex combination

        # --- gradient step + projection onto the simplex -----------------
        seq_new = seq - lr * grad
        seq_new = _project_simplex(seq_new)

        # --- occasional random jitter to escape flat plateaus -----------
        if it % 300 == 0 and np.random.rand() < 0.5:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # +/-1 %
            seq_new = _project_simplex(seq_new + jitter)

        # --- replace current vector ---------------------------------------
        seq = seq_new

        # --- evaluate current score (fast internal version) -------------
        cur_score = _score_from_vec(seq, fft_len)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            no_improve_counter = 0
            print(f"[{it:05d}] new best = {best_score:.6f}")
        else:
            no_improve_counter += 1

        # --- simple restart if we stagnate for a while ------------------
        if no_improve_counter > 2000:
            # restart near the best solution with a fresh perturbation
            seq = best_seq.copy()
            seq += (np.random.rand(n) - 0.5) * 0.05   # 5 % noise
            seq = _project_simplex(seq)
            no_improve_counter = 0
            print(f"[{it:05d}] restart (stagnation)")

        # --- decay hyper‑parameters --------------------------------------
        eps = max(eps_min, eps * eps_decay)
        lr = max(lr_min, lr * lr_decay)

    # --------------------------------------------------------------------
    # 5. Return the best candidate found
    # --------------------------------------------------------------------
    print(f"[final] iterations={it}, best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
