"""
Simple simulated‑annealing / gradient descent optimiser.
The algorithm maintains a probability vector (sum = 1) and repeatedly
reduces the largest convolution entry by moving a small amount of mass
away from the indices that contribute to that entry.  The objective is
exactly the value returned by ``evaluate_sequence`` (scale‑invariant), so
the routine evaluates the candidate without calling the costly function
twice per iteration.  Random jitter and occasional restarts avoid
getting stuck in a local minimum.  The best sequence found before the
deadline is returned.
"""

import time
import math
import numpy as np

def _evaluate(seq):
    # Wrapper that uses the provided evaluate_sequence if available,
    # otherwise computes the same value directly (scale‑invariant).
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        # fallback: sum should be 1 after normalisation
        n = len(seq)
        conv = np.convolve(seq, seq)
        return float(2 * n * conv.max() / (np.sum(seq) ** 2))

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # keep a small safety margin

    # -------------------------------------------------------------
    # 1. Choose a length and initialise a probability vector.
    # -------------------------------------------------------------
    # If a previous construction is available we keep its length,
    # otherwise we fall back to a modest size that allows many iterations.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.shape[0]
        a = np.maximum(base, 0.0)
        if a.sum() < 1e-12:
            a = np.random.rand(n)
        a = a / a.sum()
        print(f"Initialising from height_sequence_1 (length {n})")
    except Exception:
        # No previous sequence – use a random start.
        n = 3000                     # manageable size, still large enough
        a = np.random.rand(n)
        a = a / a.sum()
        print(f"Initialising from random vector (length {n})")

    # -------------------------------------------------------------
    # 2. Helper to evaluate a candidate (fast version when sum=1)
    # -------------------------------------------------------------
    def fast_score(vec):
        # vec already normalised to sum = 1
        conv = np.convolve(vec, vec)
        return 2.0 * n * conv.max()

    # Initial best
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    # Initial evaluation via the official function (just to be sure)
    try:
        best_score = _evaluate(best_seq)
    except Exception:
        pass

    # -------------------------------------------------------------
    # 3. Main optimisation loop
    # -------------------------------------------------------------

    # A slowly decreasing step size; the factor 0.2 works well for the
    # typical magnitude of the entries (~1/n).
    step_base = 0.2

    while time.time() < deadline:
        iteration += 1

        # --- a) current convolution and the offending shift -----------------
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))

        # --- b) gradient of the maximal entry w.r.t. the vector ----------
        # For a given k, ∂(a*a)[k] / ∂a[i] = a[k-i]  (if index in range)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1, dtype=int)
        # g[i] = a[max_idx - i] for all i in the valid range
        grad = np.zeros_like(a)
        grad[idx] = a[max_idx - idx]

        # --- c) propose a new point ----------------------------------------
        step = step_base / math.sqrt(iteration + 1.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)

        # Small random jitter every few hundred iterations to escape plateaus.
        if iteration % 500 == 0:
            jitter = np.random.randn(n) * (step * 0.5)
            a_new += jitter
            a_new = np.maximum(a_new, 0.0)

        # Re‑normalise to keep the scale invariant property.
        s = a_new.sum()
        if s < 1e-12:
            # If we have collapsed to the zero vector, restart randomly.
            a_new = np.random.rand(n)
            s = a_new.sum()
        a_new /= s

        # --- d) evaluate the new candidate ---------------------------------
        cand_score = fast_score(a_new)

        # Update the global best if we improved.
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_new.copy()
            # Log progress – prints will be captured by the runner.
            print(f"Iter {iteration:6d} – new best score {best_score:.6f}")

        # probability to allow occasional uphill steps.
        cur_score = fast_score(a)
        if cand_score <= cur_score or np.random.rand() < 0.001:
            a = a_new

        if iteration % 5000 == 0 and np.random.rand() < 0.25:
            a = np.random.rand(n)
            a = a / a.sum()

    # 4. Final safety evaluation using the provided function.
    # -------------------------------------------------------------
    # Convert to a plain Python list – evaluate_sequence expects a list.
    final_seq = best_seq.tolist()

    except Exception:

        print(f"Finished – final fast score {best_score:.6f}")

"""
Simple gradient / simulated‑annealing optimiser for the convolution bound.

The routine works with a *probability vector* (sum = 1) because the
objective `evaluate_sequence` is invariant under scaling.
At each iteration we locate the shift `k` where the self‑convolution
`a*a` attains its maximum and move a small amount of mass away from the
indices that contribute to that entry.  This strictly reduces the
largest convolution value unless we are already at a flat distribution.
A tiny amount of random jitter and occasional full restarts avoid getting
trapped in local minima.  The best sequence found before the time budget
expires is returned as a plain Python list; it can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

def _evaluate(seq):
    """Wrapper that prefers the provided evaluate_sequence but falls back."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:                     # pragma: no cover – safety net
        n = len(seq)
        conv = np.convolve(seq, seq)
        return float(2 * n * conv.max() / (np.sum(seq) ** 2))
