# Helper: project a vector onto the probability simplex (sum = 1, ≥0)
def _project_to_simplex(v):
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]

    rho = rho[-1]                             # last positive index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Optional LP direction (fallback if gradient descent stalls)
def solve_convolution_lp(f_sequence, rhs):
    """Linear programme: maximise Σb  s.t.  (f * b) ≤ rhs , b ≥ 0."""
    n = len(f_sequence)
    c = -np.ones(n)                           # maximise sum → minimise –sum
    # Build convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  –b ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method='highs',
        options={'time_limit': 5.0, 'disp': False}
    )
    return result.x if result.success else None

    # otherwise start from a uniform vector.
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.size >= n:
            a = seq0[:n].copy()
        else:
            pad = np.random.rand(n - seq0.size)
            a = np.concatenate([seq0, pad])
    else:
        a = np.random.rand(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()                       # normalise to sum = 1

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # Stage 1 : projected gradient descent towards a flat convolution
    target_val = 1.0 / (2 * n - 1)             # uniform target for each b_k
    target_vec = np.full(2 * n - 1, target_val)

    lr = 0.5                                   # initial learning rate
    decay = 0.9999                             # geometric decay
    max_grad_iters = 20000

    for it in range(max_grad_iters):
        if time.time() >= deadline:
            break

        # convolution and residual
        b = np.convolve(a, a)                  # length 2n‑1
        residual = b - target_vec

        # gradient = 2 * conv(residual, a_rev) restricted to n entries
        grad_full = np.convolve(residual, a[::-1])
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # gradient step + projection onto simplex
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new)

        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[Grad] iter {it:5d}  score {best_score:.6f}")

        a = a_new
        lr *= decay
        if lr < 1e-7:
            break

    # ------------------------------------------------------------------
    # Stage 2 : simulated annealing (fine‑tune around the best found)
    # Temperature schedule – linear decay from 1.0 to 1e‑3
    anneal_start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - anneal_start
        total = deadline - anneal_start
        frac = min(1.0, elapsed / total)
        temp = (1.0 - frac) * 1.0 + frac * 1e-3

        # propose a small mass transfer
        cand = best_seq.copy()
        i, j = np.random.choice(n, size=2, replace=False)
        if cand[i] == 0.0:
            continue
        delta = cand[i] * (np.random.rand() * 0.08)   # up‑to 8 % of source
        cand[i] -= delta
        cand[j] += delta

        cand = _project_to_simplex(cand)               # keep sum = 1

        cand_score = evaluate_sequence(cand.tolist())

        # accept if better, otherwise with Metropolis probability
        if cand_score < best_score or np.random.rand() < np.exp(
                -(cand_score - best_score) / (temp + 1e-12)):
            best_seq = cand
            best_score = cand_score
            print(f"[Anneal] time {time.time() - start_time:5.1f}s  "
                  f"temp {temp:.4g}  score {best_score:.6f}")

    # ------------------------------------------------------------------
    # The evaluation function is homogeneous – scaling does not change the
    # score.  Clamp to the allowed range and return a plain Python list.
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    print(f"=== Finished === best score {best_score:.6f}")
    return final
