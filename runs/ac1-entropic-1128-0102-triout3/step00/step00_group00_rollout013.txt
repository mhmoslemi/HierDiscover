def project_onto_simplex(v, S):
    # ensure sum to S and nonneg.
    # Implementation works with v already non-negative? We'll just implement generic.

Pseudo-code:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv/(np.arange(len(u))+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

def propose_candidate(seed=..., budget_s=1000, **kwargs):
    import numpy as np, time, random
    deadline = time.time() + budget_s - 0.5
    np.random.seed(seed)
    # Determine length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 500
    target_sum = np.sqrt(2 * n)
    # Helper for projection onto simplex
    def project_to_simplex(v):
        # return vector v' >=0, sum = target_sum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - target_sum
        rho = np.where(u - cssv/(np.arange(len(u)) + 1) > 0)[0]
        if len(rho) == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w
    # initialise best
    best_seq = None
    best_score = np.inf
    # start with uniform
    base = np.full(n, target_sum / n)
    # try several restarts
    while time.time() < deadline:
        # random perturbation of uniform to create initial candidate
        seq = base.copy()
        # add some random noise
        seq += np.random.randn(n) * 0.1 * np.mean(seq)
        seq = np.clip(seq, 0, None)
        seq = project_to_simplex(seq)
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
        # local search parameters
        step = np.mean(seq) * 0.1
        temp = 1.0
        it = 0
        while time.time() < deadline and it < 2000:
            # propose move: pick i,j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * step
            # move delta from i to j
            if seq[i] - delta < 0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            # project to simplex to correct sum and nonnegative
            new_seq = project_to_simplex(new_seq)
            new_score = evaluate_sequence(new_seq.tolist())
            # acceptance
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-8)):
                seq = new_seq
                cur_score = new_score

```
def subgradient_of_max_conv(a):
    conv = np.convolve(a, a)
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    n = len(a)
    g = np.zeros(n)
    for k in active:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            g[i] += 2 * a[j]
    return g
```

Complexity: For each active k (maybe few), inner loop size about min(k+1, 2n-1-k, n) up to n. So worst-case O(n * m) where m = #active. Typically only one active (peak). So O(n). Acceptable.

Implementation details: To project onto simplex with sum S (S target). Standard algorithm (by Duchi et al.):

```
def simplex_projection(v, s):
    # v is array of shape (n,)
    # we assume s > 0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # Build matrix a_ub shape (2n-1 + n) x n
    rows = []
    for k in range(2*n -1):
        row = np.zeros(n)
        start = max(0, k - (n-1))
        end = min(n-1, k)
        # Fill row[j] = f_seq[k - j] for j from start to end
        # Equivalent to row[start:end+1] = f_seq[k - np.arange(start,end+1)]
        # compute indices
        indices = k - np.arange(start, end+1)
        row[start:end+1] = f_seq[indices]
        rows.append(row)
    a_ub = np.vstack(rows)
    # Add nonnegativity rows: -I
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.concatenate([np.full(2*n -1, rhs), np.zeros(n)])
    c = -np.ones(n)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

"""
Propose a candidate step‐function sequence that minimizes
evaluate_sequence(sequence).

Algorithm
---------
1. Choose a moderate length (default 200).  The search works on a
   sequence whose ℓ1‑norm is forced to sqrt(2·n); with this
   normalisation the evaluation equals the maximal value of the
   self‑convolution.
2. Initialise a random non‑negative vector and project it onto the
   simplex {x≥0, Σx = sqrt(2·n)}.
3. Repeatedly improve the vector by solving the linear program

        maximise Σ b_i
        subject to   (a * b)_k ≤ M   for all k
                     b_i ≥ 0

   where `a` is the current sequence, `*` denotes convolution and
   `M = max(a*a)` is the current bound.
   The LP finds a direction `b` that can increase the ℓ1‑norm while
   keeping the convolution bound unchanged.
4. Scale `b` back onto the simplex and move the current sequence
   towards `b` with a relatively large mixing factor `t = 0.2`.
   If the LP fails, a tiny random perturbation is applied.
5. Keep the best sequence seen so far.  The loop runs until the time
   budget (≈ 990 s) is exhausted.

The method is a faster variant of the published “LP‑direction” scheme
with a larger step size and explicit simplex projection, which
typically drives the evaluation well below the uniform value 2.0.
"""

import time
import numpy as np
from scipy import optimize

# -------------------------------------------------------------
# Helper: project a vector onto the simplex {x >= 0, sum(x)=S}
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex scaled by S."""
    if S <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Linear program: maximise Σ b_i  s.t. conv(a,b) ≤ M and b ≥ 0
def _solve_convolution_lp(a_seq: np.ndarray, bound: float):
    """Return the LP solution b (as ndarray) or None if infeasible."""
    n = len(a_seq)
    # Build matrix for constraints a_seq * b ≤ bound
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # indices j = k - i
        j = k - np.arange(i_start, i_end + 1)
        row[i_start:i_end + 1] = a_seq[j]
        rows.append(row)
    A_ub = np.vstack(rows)                # (2n‑1) × n
    A_ub = np.vstack([A_ub, -np.eye(n)])  # add b_i ≥ 0  →  -b_i ≤ 0
    b_ub = np.concatenate([np.full(2 * n - 1, bound), np.zeros(n)])
    c = -np.ones(n)                       # maximise Σb_i  ≡ minimise -Σb_i
    # Use the robust HiGHS interior‑point solver
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          method='highs',
                          options={'presolve': True, 'time_limit': 8})
    if res.success:
        return res.x
    return None


# -------------------------------------------------------------
def _improve_via_lp(seq: np.ndarray, mix: float, target_sum: float):
    """
    One LP‑based improvement step.
    Returns a new sequence (numpy array) or None on failure.
    """
    n = len(seq)
    # Ensure the current sequence is on the simplex (might be slightly off)
    seq = _project_onto_simplex(seq, target_sum)

    # current bound M = max convolution of the (already normalised) sequence
    conv = np.convolve(seq, seq)
    M = conv.max()

    # Solve LP to maximise the ℓ1‑norm while keeping the bound M
    b = _solve_convolution_lp(seq, M)
    if b is None:
        return None

    sum_b = b.sum()
    if sum_b <= 0:
        return None

    # Scale the LP solution back to the required ℓ1‑norm
    scaled_b = b * (target_sum / sum_b)

    # Move a little towards the LP solution
    new_seq = (1.0 - mix) * seq + mix * scaled_b
    # Clip to the allowed value range and re‑project
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    new_seq = _project_onto_simplex(new_seq, target_sum)
    return new_seq


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """

    """

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)   # leave a few seconds margin

    # ---------------------------------------------------------
    # Choose problem size

    def _eval(arr):
        return evaluate_sequence(arr.tolist())

    # ---------------------------------------------------------

    cur = np.random.rand(n)
    cur = _project_onto_simplex(cur, target_sum)

    best_seq = cur.copy()
    best_score = _eval(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Parameters for the search
    mix_factor = 0.20                # step size toward the LP solution
    fail_count = 0                  # number of consecutive LP failures

    while time.time() < deadline:
        # Try an LP improvement step
        new_seq = _improve_via_lp(cur, mix_factor, target_sum)

        if new_seq is None:
            # LP failed – perform a tiny random perturbation
            fail_count += 1
            eps = 0.02 * target_sum
            i, j = np.random.randint(n, size=2)
            delta = (np.random.rand() - 0.5) * eps
            # enforce non‑negativity before projection
            tmp = cur.copy()
            tmp[i] = max(0.0, tmp[i] + delta)
            tmp[j] = max(0.0, tmp[j] - delta)
            new_seq = _project_onto_simplex(tmp, target_sum)
        else:
            fail_count = 0

        cur_score = _eval(new_seq)

        # Accept if we improve the current point (hill climbing)
        if cur_score <= _eval(cur) or np.random.rand() < 0.02:
            cur = new_seq

        # Update global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            print(f"[update] new best score = {best_score:.6f}  time={time.time()-start_time:.1f}s")

        # Occasionally restart from a fresh random point if we are stuck
        if fail_count >= 12:
            cur = np.random.rand(n)
            cur = _project_onto_simplex(cur, target_sum)
            fail_count = 0

    print(f"[final] best score achieved = {best_score:.6f}")
    return best_seq.tolist()
