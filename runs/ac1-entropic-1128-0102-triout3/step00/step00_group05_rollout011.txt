def project_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""Search for a step‐function with low max self‑convolution.
We treat the coefficients as a probability vector p (sum=1). 
For a given p the evaluation equals 2·n·max(p∗p).  The algorithm
optimises p on the simplex by minimising a smooth approximation of
max(p∗p) (log‑sum‑exp).  In each iteration we
    * compute the self‑convolution,
    * build a softmax weight focusing on the current largest entries,
    * obtain the gradient of the smoothed objective via a second convolution,
    * take a projected gradient step onto the simplex,
    * keep the best sequence seen.
Multiple random restarts with different lengths are performed until the
time budget expires.
"""

import numpy as np
import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # based on Duchi et al., 2008
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_from_prob(p: np.ndarray) -> float:
    """Return 2·n·max(p∗p) for a probability vector p."""
    n = p.shape[0]
    conv = np.convolve(p, p)
    return 2.0 * n * float(conv.max())

    Several random restarts with different lengths are performed until the
    time budget expires.  The best sequence found is returned.
    """

    deadline = time.time() + budget_s - 2.0   # safety margin

        step0 = 0.5          # initial step size

            score = _score_from_prob(p)

                best_seq = (p * scale).tolist()

is invariant under scaling, it can be written as
    2·n·max(p * p) .
Hence we only have to minimise the peak of the self‑convolution of a
probability vector.

The algorithm performs projected gradient descent on the probability simplex.
A smooth approximation of the max, namely a log‑sum‑exp,
is used:
    L(p) = (1/α)·log Σ_k exp(α·(p * p)_k) .
Its gradient is a weighted sum of the gradients of the individual
convolution entries; the weights are the soft‑max of the current
convolution, concentrating on the largest entries.
The gradient can be computed by a second convolution, which is
efficient (FFT‑based for large n).

Multiple random restarts with different lengths are carried out until the
time budget expires.  The best sequence seen (scaled back to the required
ℓ₁‑norm √(2n)) is returned.
"""

import numpy as np
import time


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Based on Duchi et al., “Efficient Projections onto the L1‑Ball”.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_prob(p: np.ndarray) -> float:
    """
    Exact evaluation for a probability vector p:
    evaluate(sequence) = 2·n·max(p * p)
    """
    n = p.shape[0]
    conv = np.convolve(p, p)
    return 2.0 * n * float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of heights (non‑negative) that (hopefully) beats the current
    upper bound.  The routine runs a projected‑gradient descent on the simplex,
    minimising a smooth approximation of the peak of the self‑convolution.
    Several random restarts with different lengths are performed until the
    time budget expires; the best sequence found is returned.
    """
    np.random.seed(seed)
    # leave a small margin for the caller
    deadline = time.time() + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # lengths that give a good trade‑off between quality and speed
    candidate_lengths = [1500, 3000, 5000, 7000]

    while time.time() < deadline:
        # pick a length (larger lengths are tried less often)
        n = int(np.random.choice(candidate_lengths))

        # random start on the simplex (Dirichlet(1,…,1) via exponential sampling)
        p = np.random.exponential(1.0, size=n)
        p = p / p.sum()

        # hyper‑parameters (robust for a wide range of n)
        step0 = 0.5          # initial step size for the gradient step
        alpha0 = 5.0         # initial smoothing factor for the soft‑max
        max_iter = 2500

        for it in range(max_iter):
            if time.time() >= deadline:
                break

            # exact current score (no call to evaluate_sequence, much faster)
            score = _score_from_prob(p)

            if score < best_score:
                best_score = score
                scale = np.sqrt(2.0 * n)          # required ℓ₁‑norm of the raw heights
                best_seq = (p * scale).tolist()
                print(f"[{time.time():.1f}] n={n}, it={it}, new best = {best_score:.6f}")

                # stop early if we are already comfortably below the target
                if best_score < 1.5:
                    print("Target achieved – returning early.")
                    return best_seq

            # ----- gradient of a smooth approximation of the max -------------
            conv = np.convolve(p, p)
            max_conv = conv.max()
            # increase the temperature gradually, sharpening the approximation
            alpha = alpha0 + 0.01 * it
            # stable soft‑max on the convolution entries
            shift = conv - max_conv
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient g_i = 2 Σ_j w_{i+j} p_j
            p_rev = p[::-1]
            g_full = np.convolve(w, p_rev)               # length = 3n‑2
            start = n - 1
            g = 2.0 * g_full[start:start + n]

            # step size decaying like 1/√t
            step = step0 / np.sqrt(it + 1.0)

            # projected gradient descent step
            p = _project_simplex(p - step * g)

        # after a full run add a tiny random jitter to escape possible stagnation
        p += np.random.normal(scale=1e-4, size=n)
        p = _project_simplex(p)

    # fallback – should never be needed, but guarantees a valid return value
    if best_seq is None:
        n = 1000
        scale = np.sqrt(2.0 * n)
        best_seq = [scale / n] * n
    return best_seq
