def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Simulated annealing search for a non‑negative sequence that minimizes
    the evaluation function ``evaluate_sequence``.
    The algorithm works on a sequence whose entries sum to 1 (scale‑invariant)
    and repeatedly transfers a random amount of mass from one index to another,
    accepting improvements or uphill moves according to a temperature schedule.
    The best sequence found before the time budget expires is returned.
    """
    import numpy as np
    import time, random, math

    # configuration
    n = kwargs.get('length', 200)          # number of coefficients
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)         # honour the available length if possible
    # initialise a uniform distribution (sum = 1)
    seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = seq.copy()
    # local evaluation (identical to evaluate_sequence up to clipping)
    def local_eval(s):
        total = s.sum()
        if total < 1e-12:
            return float('inf')
        conv = np.convolve(s, s)
        return float(2.0 * n * conv.max() / (total * total))
    cur_val = local_eval(seq)
    best_val = cur_val

    deadline = time.time() + budget_s - 0.5
    # annealing schedule
    start_temp = 0.5
    end_temp = 1e-4
    # exponential cooling
    while time.time() < deadline:
        # linear interpolation of temperature based on elapsed fraction
        frac = (deadline - time.time()) / budget_s
        temp = end_temp + (start_temp - end_temp) * frac

        # propose a move
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:
            continue
        # amount to transfer, up to half of source
        max_transfer = seq[i] * 0.5
        if max_transfer <= 0:
            continue
        delta = random.random() * max_transfer
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = local_eval(new_seq)

        # acceptance rule
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12)):
            seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
        # occasionally perform a large random reshuffle to avoid stagnation
        if random.random() < 0.001:
            # perturb a random subset of indices
            perturb = np.random.rand(n)
            seq = np.maximum(seq + 0.05 * (perturb - 0.5), 0)
            seq /= seq.sum()
            cur_val = local_eval(seq)

    # final check with the official evaluator (to respect clipping etc.)
    # ensure the sequence is a plain Python list of floats
    result = best_seq.tolist()
    # optional: scale up to avoid sum being too small (<=0.01)
    total = sum(result)
    if total < 0.01:
        scale = 0.01 / total
        result = [x * scale for x in result]
    return result

from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    # Build A_ub for convolution constraints
    rows = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        b_ub.append(rhs)
    # non-negativity constraints added as -I <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None
```

import numpy as np
import time
import math
import random
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Simulated annealing + LP direction search
#
#  The objective of evaluate_sequence() is
#       score = 2 * n * max(conv(f,f)) / (sum(f)**2)
#  (scale‑invariant, lower is better).  We start from a uniform vector,
#  repeatedly solve a small linear program that finds a direction g which
#  increases the sum of the coefficients without violating the current
#  convolution bound, and mix the current vector with g using a line
#  search on the mixing weight t.  In parallel we run a cheap
#  simulated‑annealing “mass‑transfer’’ neighbourhood search to escape
#  possible plateaus.  The best sequence found before the time budget
#  expires is returned.
# ----------------------------------------------------------------------


def _solve_convolution_lp(f_vec, rhs):
    """
    Solve the linear program:
        maximise   sum(b_j)
        subject to  (f * b)_k <= rhs   for all k,
                    b_j >= 0
    The variable is b (same length as f).  Returns the optimal b as a
    NumPy array or None if the LP fails.
    """
    n = len(f_vec)
    # maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints: for each possible index k (0 … 2n‑2)
    #   Σ_i f_i * b_{k-i} <= rhs
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_vec[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints   -b_j <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # Use the robust HiGHS solver (available in recent SciPy)
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None


def _local_score(seq):
    """
    Fast replica of evaluate_sequence for non‑negative inputs.
    Returns the exact score (no clipping, no I/O).
    """
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * len(seq) * conv.max() / (total * total))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence that minimises `evaluate_sequence`.

    The algorithm mixes three ideas:
      * a deterministic LP step that expands the total sum without
        breaking the current maximal convolution bound,
      * a line‑search over the mixing factor,
      * a stochastic “mass‑transfer’’ neighbourhood move (simulated
        annealing) to explore the space.

    The best sequence encountered before the time budget expires is
    returned as a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    #  Parameters & basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # Length of the sequence – use the provided construction if present,
    # otherwise a modest size that keeps the optimisation inexpensive.
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = kwargs.get("length", 200)  # fallback length

    # Start from the uniform (sum = 1) – this yields the baseline score 2.
    cur = np.full(n, 1.0 / n, dtype=np.float64)
    best = cur.copy()
    cur_score = _local_score(cur)
    best_score = cur_score

    # Temperature schedule for the stochastic moves
    T_start = 0.5
    T_end = 1e-5

    # Tuning constants
    t_candidates = (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005)
    max_iter_without_lp = 0

    deadline = time.time() + budget_s - 0.2  # leave a tiny safety margin

    # ------------------------------------------------------------------
    #  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- simulated‑annealing mass transfer -----
        frac = (deadline - time.time()) / budget_s
        temperature = T_end + (T_start - T_end) * frac

        i, j = random.randrange(n), random.randrange(n)
        if i != j and cur[i] > 0.0:
            # move a random fraction of mass from i to j
            delta = random.random() * 0.5 * cur[i]   # at most 50 % of source
            trial = cur.copy()
            trial[i] -= delta
            trial[j] += delta

            trial_score = _local_score(trial)
            if trial_score < cur_score or random.random() < math.exp(
                -(trial_score - cur_score) / (temperature + 1e-12)
            ):
                cur = trial
                cur_score = trial_score
                if trial_score < best_score:
                    best, best_score = trial.copy(), trial_score

        # Occasionally restart the local noise to keep diversity
        if random.random() < 0.001:
            noise = (np.random.rand(n) - 0.5) * 0.01
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            cur_score = _local_score(cur)
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score

        # ----- deterministic LP step (every few iterations) -----
        max_iter_without_lp += 1
        if max_iter_without_lp >= 5:
            max_iter_without_lp = 0

            # Normalise the current vector to the conventional sum = sqrt(2n)
            total = cur.sum()
            sqrt_2n = math.sqrt(2.0 * n)
            f_norm = cur * (sqrt_2n / total)

            # Current maximal convolution value (the bound used in the LP)
            rhs = np.max(np.convolve(f_norm, f_norm))

            # Solve the LP for a direction g
            g = _solve_convolution_lp(f_norm, rhs)
            if g is not None:
                # Normalise g to the same sum = sqrt(2n)
                g_norm = g * (sqrt_2n / g.sum())

                # Try mixing with several step sizes and keep the best improvement
                improved = False
                for t in t_candidates:
                    mixed = (1.0 - t) * cur + t * g_norm
                    # Renormalise to keep the sum = sqrt(2n) (scale does not affect score)
                    mixed = mixed * (sqrt_2n / mixed.sum())
                    mixed_score = _local_score(mixed)
                    if mixed_score < cur_score - 1e-12:
                        cur, cur_score = mixed, mixed_score
                        improved = True
                        if mixed_score < best_score:
                            best, best_score = mixed.copy(), mixed_score
                        break   # accept the first successful t (larger steps tried first)
                # If none of the candidate t’s improved, fall back to a tiny step
                if not improved:
                    t = 0.01
                    mixed = (1.0 - t) * cur + t * g_norm
                    mixed = mixed * (sqrt_2n / mixed.sum())
                    mixed_score = _local_score(mixed)
                    if mixed_score < cur_score:
                        cur, cur_score = mixed, mixed_score
                        if mixed_score < best_score:
                            best, best_score = mixed.copy(), mixed_score

        # Small safety: occasionally re‑normalise to avoid numerical drift
        if (int(time.time()) % 30) == 0:
            cur = cur * (1.0 / cur.sum())
            cur_score = _local_score(cur)

    # ------------------------------------------------------------------
    #  Post‑processing – ensure the final sequence respects the
    #  minimal sum requirement of evaluate_sequence (>= 0.01)
    # ------------------------------------------------------------------
    result = best.tolist()
    if sum(result) < 0.01:
        scale = 0.01 / sum(result)
        result = [x * scale for x in result]

    return result
