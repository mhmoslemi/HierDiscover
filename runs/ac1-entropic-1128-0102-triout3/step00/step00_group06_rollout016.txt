"""
Optimiser for the step‑function evaluation problem.

The goal is to minimise

    E(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

for a non‑negative sequence `a`.  The function is homogeneous, therefore we keep
the ℓ₁–norm fixed to sqrt(2 n); under this normalisation the score is exactly
`max(conv(a,a))`.  Starting from a uniform sequence (which gives a score of 2)
we repeatedly solve a linear programme that finds a new sequence `g` with a
larger ℓ₁–norm while the mixed convolution `conv(a, g)` stays below the current
maximum.  After rescaling `g` to the required norm we either replace `a` by `g`
or take a convex combination.  This simple “LP‑move” dramatically reduces the
peak of the autocorrelation and therefore the evaluation value.

The optimiser runs for the allotted time (≈1000 s) and returns the best
sequence discovered.
"""

import time
import numpy as np
from scipy import optimize
from scipy.optimize import linprog


def _max_convolution(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)) using an FFT – fast for moderate lengths."""
    n = seq.size
    sz = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(seq, n=sz)
    conv = np.fft.irfft(ft * ft, n=sz)[: 2 * n - 1]
    return float(conv.max())


def _solve_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to  (f * b)[k] ≤ rhs   for all k = 0 … 2n‑2
                    b_j ≥ 0.
    Returns the optimal b (as a NumPy array) or None if the solver fails.
    """
    n = f.size
    # Build the constraint matrix A_ub x ≤ b_ub.
    # Row k corresponds to the convolution (f * b)[k].
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # rows[k, j] = f[k‑j]  if 0 ≤ k‑j < n else 0
        # we fill by scanning i (the index in f)
        for i, fi in enumerate(f):
            j = k - i
            if 0 <= j < n:
                row[j] = fi
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], rhs, dtype=float)

    # Linear programme: maximise 1·b  <=> minimise -1·b
    c = -np.ones(n, dtype=float)

    # Use the HiGHS solver (fast and robust).  No explicit time limit
    # – the outer routine controls the total budget.
    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None

        Time budget in seconds (the caller reserves ~10 s for the wrapper).

    -------
    list[float]
        The best sequence found (non‑negative, arbitrary length).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a working length.
    #     Larger lengths give more freedom but make the LP bigger.
    #     n = 800 is a sweet spot for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 800
    target_norm = np.sqrt(2.0 * n)          # required ℓ₁‑norm for this length

    # ------------------------------------------------------------------
    # 2.  Initialise – uniform (the known baseline) or a random start.
    # ------------------------------------------------------------------
    a = np.full(n, target_norm / n, dtype=float)   # uniform, sum = target_norm
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))   # noqa: F821

    # Occasionally switch to a random Dirichlet start to escape stagnation
    if np.random.rand() < 0.2:
        a = np.random.rand(n)
        a = a / a.sum() * target_norm
        cur_score = float(evaluate_sequence(a.tolist()))   # noqa: F821
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    t = 0.5                     # blending factor (large at start, shrinks later)

    while time.time() < deadline:
        iteration += 1

        # --- current peak of the autocorrelation --------------------------------
        cur_max = _max_convolution(a)          # this is the score under normalisation
        # ------------------------------------------------------------------------
        #   Solve the linear programme that finds a new vector with a larger
        #   ℓ₁‑norm while keeping the mixed convolution bounded by cur_max.
        # ------------------------------------------------------------------------
        g = _solve_lp(a, cur_max)

        if g is not None:
            # Rescale g so that its ℓ₁‑norm equals the target_norm.
            sum_g = g.sum()
            if sum_g == 0:
                # degenerate – fall back to a tiny random perturbation
                g = np.random.rand(n) * 1e-6
                sum_g = g.sum()
            g = g * (target_norm / sum_g)

            # Try the pure replacement first.
            new_seq = g
            new_score = float(evaluate_sequence(new_seq.tolist()))   # noqa: F821
            if new_score < best_score:
                best_score, best_seq = new_score, new_seq.copy()
                a = new_seq.copy()
                cur_max = _max_convolution(a)
                # On a good improvement we can afford a larger step next time
                t = min(0.9, t * 1.1)
                continue

            # If the pure replacement is not better, blend a little.
            blended = (1.0 - t) * a + t * g
            blended_score = float(evaluate_sequence(blended.tolist()))   # noqa: F821
            if blended_score < best_score:
                best_score, best_seq = blended_score, blended.copy()
                a = blended.copy()
                # shrink the blending factor slowly – we want finer moves later
                t = max(0.01, t * 0.95)
                continue

        # ------------------------------------------------------------------------
        #   If LP failed or gave no improvement, perform a cheap random tweak.
        # ------------------------------------------------------------------------
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 0.02 * target_norm   # ~2 % of total mass
        a[idx] = max(0.0, a[idx] + delta)

        # Re‑normalise to keep the required ℓ₁‑norm.
        a = a * (target_norm / a.sum())

        # Occasionally evaluate the current vector in case the random tweak helped.
        if iteration % 20 == 0:
            cur_score = float(evaluate_sequence(a.tolist()))   # noqa: F821
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                # reset the blending factor – we have a fresh improvement
                t = 0.5

        # Occasionally restart from the best sequence found so far (helps escape
        # small local plateaus).
        if iteration % 200 == 0 and np.random.rand() < 0.3:
            a = best_seq.copy()
            t = 0.5

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    # Convert to plain Python list (evaluate_sequence expects a list of floats)
    return best_seq.tolist()
