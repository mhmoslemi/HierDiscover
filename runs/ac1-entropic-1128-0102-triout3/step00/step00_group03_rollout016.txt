def construct_sidon_set(n, target_k=None):
    sums_used = set()
    sidon = []
    # Candidate indices 0..n-1
    candidates = list(range(n))
    random.shuffle(candidates)
    for i in candidates:
        conflict = False
        for j in sidon:
            s = i + j
            if s in sums_used:
                conflict = True
                break
        if not conflict:
            # also check self sum i+i not in sums_used (though must be distinct)
            if i + i in sums_used:
                continue
            # add
            for j in sidon:
                sums_used.add(i + j)
            sums_used.add(i + i)
            sidon.append(i)
            if target_k and len(sidon) >= target_k:
                break
    return sidon

"""
Search for a step‑function that beats the current upper bound.

Approach
========
* Build a *Sidon* (B2) set of indices – i.e. a set where each sum i+j
  appears at most once.  Between these indices we place the maximal
  allowed height (1000).  The Sidon property guarantees that the
  convolution of the high‑valued part never exceeds 2·10⁶ at any
  position.
* Fill the remaining positions with a low uniform height `w`.  The low
  part contributes many pairwise products, but thanks to the small
  magnitude the resulting convolution peak stays below the peak
  created by the high part.
* Randomly vary both the low height `w` (5 … 30) and the fraction of the
  remaining indices that receive this height.  For each candidate we
  evaluate the provided `evaluate_sequence` function and keep the best
  (smallest) score.
* The whole procedure runs until the time budget (≈1000 s) is exhausted
  and returns the best sequence found.

The algorithm is deliberately simple, fast and does not rely on any
external solvers – it works directly with NumPy and pure Python.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a Sidon set (no repeated sums) by a greedy random search
# ----------------------------------------------------------------------
def build_sidon_set(n, trials=20):
    """Return a list of indices forming a Sidon set (size ≈ √(2n))."""
    best_set = []
    # The theoretical maximum size is about sqrt(2*n); we try a few random orders.
    for _ in range(trials):
        order = np.random.permutation(n)
        used = np.zeros(2 * n - 1, dtype=bool)      # which sums i+j have been used
        cur = []
        for i in order:
            # check self‑sum
            if used[2 * i]:
                continue
            # check sums with already selected indices
            conflict = False
            for j in cur:
                if used[i + j]:
                    conflict = True
                    break
            if conflict:
                continue
            # add i to the Sidon set
            for j in cur:
                used[i + j] = True
            used[2 * i] = True
            cur.append(i)
        if len(cur) > len(best_set):
            best_set = cur
        # early exit if we already hit the known theoretical limit
        if len(best_set) >= int(np.sqrt(2 * n)) + 2:
            break
    return np.array(best_set, dtype=int)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(budget_s=1000, seed=42):
    """
    Search for a good coefficient sequence within the given time budget.

    Parameters
    ----------
    budget_s : float
        Approximate time budget in seconds (the caller reserves a few
        seconds extra for cleanup, so we stop a little early).
    seed : int
        Random seed for reproducibility.

    Returns
    -------
    list[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # safety margin

    # Length of the construction – use the length of the previous best if available
    try:
        n = len(height_sequence_1)   # supplied by the evaluation harness
    except Exception:
        n = 3401                    # fallback (the reference length)

    best_seq = None
    best_score = float("inf")

    # Parameters to sweep
    low_weights = [5, 10, 15, 20, 25, 30]
    fractions = [0.25, 0.5, 0.75, 1.0]

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) build a fresh Sidon set (high‑valued positions)
        # ------------------------------------------------------------------
        sidon_idx = build_sidon_set(n, trials=30)
        if len(sidon_idx) < 2:          # need at least two high points to get the 2·10⁶ peak
            continue

        # Positions that are still free
        all_idx = np.arange(n, dtype=int)
        free_idx = np.setdiff1d(all_idx, sidon_idx, assume_unique=True)

        # ------------------------------------------------------------------
        # 2) try a variety of low‑weight fillers
        # ------------------------------------------------------------------
        for w in low_weights:
            # Option A: use *all* free positions with weight w
            candidate = np.zeros(n, dtype=float)
            candidate[sidon_idx] = 1000.0
            candidate[free_idx] = float(w)

            score = evaluate_sequence(candidate.tolist())
            if score < best_score:
                best_score = score
                best_seq = candidate.tolist()
                print(
                    f"[{time.time():.1f}] New best (all free) score={best_score:.6f} "
                    f"| h={len(sidon_idx)} w={w}"
                )
                # Early stop if we already beat the target comfortably
                if best_score < 1.45:
                    return best_seq

            # Option B: use only a fraction of the free positions
            for frac in fractions:
                m = int(frac * free_idx.size)
                if m == 0:
                    continue
                low_idx = np.random.choice(free_idx, size=m, replace=False)
                candidate = np.zeros(n, dtype=float)
                candidate[sidon_idx] = 1000.0
                candidate[low_idx] = float(w)

                score = evaluate_sequence(candidate.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = candidate.tolist()
                    print(
                        f"[{time.time():.1f}] New best (frac={frac:.2f}) "
                        f"score={best_score:.6f} | h={len(sidon_idx)} w={w} m={m}"
                    )
                    if best_score < 1.45:
                        return best_seq

        # Occasionally try a different Sidon set even if the current one is large
        # (the outer loop repeats anyway).

    # If the loop exits without finding a very good candidate we still return
    # the best one we have.
    if best_seq is None:
        # Fallback: a uniform sequence (always yields score 2.0)
        best_seq = [1000.0] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback uniform sequence, score =", best_score)

    print("Best score after search:", best_score)
    return best_seq
