```
def proj_simplex(v):
    # Projects onto simplex sum(v)=1, v>=0
    # Implementation known: 
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 2000  # can set from height_sequence_1 length if provided else default
    # use starting sequence: if height_sequence_1 exists and its length >= n, use truncated else uniform.
    try:
        init_seq = list(height_sequence_1)[:n]
        if len(init_seq) < n:
            init_seq += [0.0] * (n - len(init_seq))
    except NameError:
        init_seq = None
    if init_seq is None or np.random.rand() < 0.3:
        seq = np.full(n, 1.0 / n)
    else:
        seq = np.array(init_seq, dtype=float)
        # make sure nonnegative and sum to 1
        seq = np.maximum(seq, 0)
        seq /= seq.sum()
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # precompute FFT size
    fft_size = 1
    while fft_size < 2*n:
        fft_size <<= 1
    
    def compute_max(seq):
        # compute convolution via FFT
        fa = np.fft.rfft(seq, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[:2*n-1]
        return conv.max()
    
    # SA parameters
    temperature = 1.0
    cooling = 0.99995  # each iteration multiply
    it = 0
    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if seq[i] <= 1e-12:
            continue
        # transfer amount
        delta = np.random.rand() * seq[i] * 0.5  # up to half of i's mass
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # compute max conv
        max_conv = compute_max(new_seq)
        new_score = 2.0 * n * max_conv  # sum=1 so evaluation = 2*n*max
        # acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            seq = new_seq
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {it}, new best score: {best_score:.6f}")
        # cooling
        temperature *= cooling
        it += 1
        # occasional random reset if no improvement for many steps
        if it % 50000 == 0:
            # reheat
            temperature = max(temperature, 0.05)
            # small random perturbation overall
            perturb = np.random.rand(n) * 0.001
            seq = seq + perturb
            seq = np.maximum(seq, 0)
            seq /= seq.sum()
    # convert best_seq to list of floats
    return best_seq.tolist()
```
