    eps_factor = 0.2 # fraction of average
    no_improve = 0

    while time.time() < deadline:
        # compute conv
        conv = np.convolve(cur_seq, cur_seq)
        # find max indices
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # compute contribution
        contrib = np.zeros_like(cur_seq)
        for i in range(len(cur_seq)):
            j = max_idx - i
            if 0 <= j < len(cur_seq):
                if i == j:
                    contrib[i] = cur_seq[i] * cur_seq[j]  # a_i^2
                else:
                    contrib[i] = 2 * cur_seq[i] * cur_seq[j]  # counted twice
        # choose source index i_src with highest contribution and positive mass
        src_candidates = np.where(cur_seq > 1e-12)[0]
        if len(src_candidates) == 0:
            break
        i_src = src_candidates[np.argmax(contrib[src_candidates])]
        # choose destination index i_dst with lowest contribution (could be same as src but skip)
        dst_candidates = np.setdiff1d(np.arange(len(cur_seq)), [i_src])
        i_dst = dst_candidates[np.argmin(contrib[dst_candidates])]
        # compute epsilon
        avg = cur_seq.mean()
        eps = min(cur_seq[i_src], max(1e-6, eps_factor * avg))
        # create new seq
        new_seq = cur_seq.copy()
        new_seq[i_src] -= eps
        new_seq[i_dst] += eps
        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                no_improve = 0
                print(f"New best score {best_score:.6f} at time {time.time()-start:.1f}s")
        else:
            # maybe accept with small probability to escape
            if np.random.rand() < 0.001:
                cur_seq = new_seq
                cur_score = new_score

"""
Propose a candidate step‑function using a simple hill‑climbing scheme.
The algorithm repeatedly moves a small amount of mass from indices that
contribute heavily to the current maximal convolution value to indices
that contribute little.  This flattens the autocorrelation and reduces
the objective 2·n·max(b)/(sum(a)^2).  Occasionally a direction obtained
by a linear‑programming step (the same LP used in the state‑of‑the‑art
method) is mixed in to accelerate progress.

The implementation:
* Starts from the provided `height_sequence_1` if it exists, otherwise a
  random non‑negative vector of length 1500‑3000.
* In each iteration:
    - Compute the convolution `b = a * a` and locate the lag `k_max`
      where `b` is maximal.
    - For that lag compute the contribution of each coefficient:
        contribution[i] = a[i] * a[k_max‑i]   (if the pair is in range).
    - Transfer a small amount `ε` from the index with the largest
      contribution to the index with the smallest one (keeping the total
      sum unchanged).
    - Accept the move if it improves the score; otherwise keep the old
      vector (with a tiny probability of acceptance to escape plateaus).
* Every few thousand iterations a LP direction is computed (`get_good_direction_to_move_into`);
  if it yields a better sequence we adopt it.
* The loop stops a few seconds before the time budget to guarantee a
  return value.

All helper functions are defined at top level for clarity.
"""

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximize    sum(b_i)
        subject to  conv(f_seq, b) ≤ rhs
                    b_i ≥ 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    # Objective: maximize sum(b)  ==> minimize -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints: for each lag k (0 .. 2n-2)
    #   Σ_i f_i * b_{k-i} ≤ rhs
    # This can be written as A_ub @ b ≤ rhs where
    #   A_ub[k, i] = f_{k-i}    (if 0 ≤ k-i < n) else 0
    A_ub = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        # indices i for which b_i appears with coefficient f_{k-i}
        i_vals = np.arange(n)
        j_vals = k - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        A_ub[k, i_vals[mask]] = f_seq[j_vals[mask]]

    b_ub = np.full(2 * n - 1, rhs, dtype=float)

    # Bounds enforce b_i ≥ 0
    bounds = [(0, None)] * n

    # Use the HiGHS solver (fast and robust)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.array(res.x, dtype=float)
    return None


def get_good_direction_to_move_into(seq):
    """
    Given a sequence `seq`, compute a direction using the LP described
    above, mix it with the current sequence and return the mixed sequence.
    The mixing factor `t` is small (0.05) to ensure a gentle update.
    Returns a list of floats or None if the LP fails.
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None
    # Normalise to the standard energy √(2n)
    scale = np.sqrt(2.0 * n) / sum_seq
    f_norm = np.array(seq, dtype=float) * scale

    # Upper bound for the convolution of the normalised sequence
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Bring `g` to the same normalisation as `f`
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    scale_g = np.sqrt(2.0 * n) / sum_g
    g_norm = g * scale_g

    t = 0.05  # small mixing weight
    new_seq = (1.0 - t) * np.array(seq, dtype=float) + t * g_norm
    return new_seq.tolist()

    deadline = start_time + budget_s - 2.0          # safety margin

    else:

        cur_seq = np.random.rand(length).astype(float)

    # Guarantee a non‑trivial sum

    cur_score = evaluate_sequence(cur_seq.tolist())
    best_seq = cur_seq.copy()
    best_score = cur_score

    # ------------------------------------------------------------------

            cur_score = evaluate_sequence(cur_seq.tolist())
            no_improve = 0

    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
