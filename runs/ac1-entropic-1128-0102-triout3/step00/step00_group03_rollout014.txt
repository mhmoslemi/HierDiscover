def _polynomial_sequence(n: int, p: int, decreasing: bool = True) -> list[float]:
    """Return a length‑n sequence a_i = (1 - i/(n-1))**p (decreasing)
    or a_i = (i/(n-1))**p (increasing)."""
    if n == 1:
        return [1.0]
    denom = float(n - 1)
    seq = []
    for i in range(n):
        x = i / denom
        if decreasing:
            val = 1.0 - x
        else:
            val = x
        seq.append(val ** p)
    return seq
