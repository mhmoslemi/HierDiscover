def simplex_projection(v):
    """Projects vector v onto probability simplex {x >= 0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)

    def project_simplex(v):
        """Projection onto simplex (nonnegative, sum=1)."""
        # from Duchi
        v = np.asarray(v, dtype=np.float64)
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, n+1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # fallback: uniform
            return np.ones_like(v) / n
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    def conv_max(a):
        # compute convolution using FFT for speed
        n = a.size
        m = 2*n - 1
        nfft = 1 << ((m - 1).bit_length() + 1)  # next power >= m
        A = np.fft.rfft(a, n=nfft)
        C = np.fft.irfft(A * A, n=nfft)[:m]
        return C.max()

    def obj_val(a):
        # sum assumed 1
        return 2 * a.size * conv_max(a)

    def gradient_of_max(a):
        # compute conv array via FFT? Simpler with numpy.convolve due to need full array.
        c = np.convolve(a, a)
        max_val = c.max()
        ks = np.flatnonzero(c == max_val)  # possible multiple
        n = a.size
        grad = np.zeros_like(a)
        for k in ks:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2 * a[j[mask]]
        # In case many ks, normalizing gradient perhaps divide by number of ks.
        return grad, max_val

"""
Propose a good step‑function height sequence.

The search proceeds in two stages.

1. **Parameterized families** – we generate many candidates of the form
       a_i ∝ (i+α)^(-p)      (or the symmetric sum of a left and a right copy)
   with random parameters p∈[0.45,0.55] and shift α∈[0.5,3.0].
   The vectors are projected onto the probability simplex (∑a_i = 1,
   a_i ≥ 0).  The objective
       2·n·max(conv(a,a))
   (equal to ``evaluate_sequence`` when the sum is 1) is cheap to compute
   with an FFT‑based convolution; the best few candidates are kept.

2. **Local refinement** – starting from a promising candidate we run a
   projected sub‑gradient descent on the true objective.
   The sub‑gradient of the max‑convolution is
       ∂max/∂a_i = 2·a_{k*‑i}   for the index k* where the maximum is attained.
   After each step the vector is projected back onto the simplex.
   Occasionally a small random jitter is applied to the current best
   solution to escape shallow local minima.

The whole routine respects the 1000 s budget, keeps the best sequence
found and returns it.  The algorithm is completely self‑contained and
uses only numpy and the already‑imported ``evaluate_sequence`` function.
"""

import numpy as np
import time
import random

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution (linear, not circular) and maximal entry
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] using FFT (linear convolution)."""
    n = a.size
    m = 2 * n - 1
    # next power of two for a clean FFT size
    nfft = 1 << ((m - 1).bit_length() + 1)
    A = np.fft.rfft(a, n=nfft)
    C = np.fft.irfft(A * A, n=nfft)[:m]
    return float(C.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (uses the full convolution once)
def _subgradient_of_max(a: np.ndarray):
    """Return (g, max_val) where g is a sub‑gradient of max(conv) at a."""
    n = a.size
    conv = np.convolve(a, a)           # full linear convolution
    max_val = conv.max()
    # all indices achieving the maximum (ties are possible)
    ks = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in ks:
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] += 2.0 * a[j[mask]]
    # averaging over the number of active indices stabilises the step size
    g /= max(1, ks.size)
    return g, max_val
