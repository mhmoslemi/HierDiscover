def project_simplex(v):
    """Project vector v onto simplex {x>=0, sum x=1}."""
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Project onto the probability simplex {x >=0, sum x = 1}."""
    # Implementation as per Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # All zero
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):

   import numpy as np, time

   np.random.seed(seed)

   deadline = time.time() + budget_s - 5

   # Determine n

   try:
       n = len(height_sequence_1)
       init_seq = np.array(height_sequence_1, dtype=np.float64)
   except NameError:
       # fallback
       n = 2000
       init_seq = np.full(n, 1.0 / n)

   # Normalize sum=1
   a = init_seq.copy()
   a = a / a.sum()

   best_seq = a.copy()
   best_score = evaluate_sequence(best_seq.tolist())

   # Parameters
   mu = 0.005  # smoothing
   eta0 = 0.5
   iteration = 0
   no_improve_counter = 0

   # Precompute FFT size
   M = 2 * n - 1
   L = 1 << ((n + M - 1).bit_length())  # power of two >= n+M-1

   while time.time() < deadline:
       iteration += 1

       # FFT of a
       A = np.fft.rfft(a, n=L)

       # Convolution (self) to get conv
       conv = np.fft.irfft(A * A, n=L)[:M]

       # Softmax weights
       # To improve numerical stability subtract max before exp
       conv_max = np.max(conv)
       exp_arg = (conv - conv_max) / mu
       w = np.exp(exp_arg)
       w /= np.sum(w)

       # Compute gradient via correlation
       w_rev = w[::-1]

       B = np.fft.rfft(w_rev, n=L)
       conv_rev = np.fft.irfft(A * B, n=L)[:n + M - 1]

       # indices for gradient
       idx = M - 1 - np.arange(n)  # shape (n,)
       g = conv_rev[idx]

       # Step size schedule
       eta = eta0 / np.sqrt(iteration)

       a_new = a - eta * g

       # Project onto simplex
       a = project_simplex(a_new)

       # Occasionally add small random perturbation to escape plateaus
       if iteration % 500 == 0:
           noise = np.random.normal(scale=0.001, size=n)
           a = project_simplex(a + noise)

       # Evaluate periodically (not each iteration to reduce compute)
       if iteration % 10 == 0:
           seq_list = a.tolist()
           score = evaluate_sequence(seq_list)
           if score < best_score:
               best_score = score
               best_seq = a.copy()
               no_improve_counter = 0
               print(f"Iter {iteration}: improved score {best_score:.6f}")
           else:
               no_improve_counter += 1

       # Early termination if improvement stalls long
       if time.time() + 0.1 >= deadline:
           break

   # Return best as list
   return best_seq.tolist()

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm works on a probability simplex (the coefficients are kept non‑negative
and sum to one, which does not affect the objective).  Starting from a uniform
distribution (or the supplied `height_sequence_1`), we iteratively minimise a
smooth approximation of the max‑convolution:

    max_k (a*a)[k]  ≈  μ·log∑_k exp( (a*a)[k] / μ )

The gradient of this smooth surrogate can be written in closed form as a
cross‑correlation of the current coefficient vector with the soft‑max weights.
All convolutions / correlations are performed with FFTs (O(N log N)) so that
many iterations fit easily into the 1000 s budget.

After each gradient step the vector is projected back onto the probability
simplex (Duchi et al. 2008).  Occasionally a small random perturbation is added
to avoid stagnation.  The best sequence seen so far (according to the provided
`evaluate_sequence`) is returned.

The method is completely self‑contained, uses only NumPy and runs on a single
CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1]          # the last true index (1‑based)
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: FFT based convolution (real inputs)
def _fft_convolve(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Convolution of real 1‑D arrays a and b using an FFT of length L."""
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = A * B
    c = np.fft.irfft(C, n=L)
    # the exact length needed is len(a)+len(b)-1
    return c[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish a few seconds
        before the limit, leaving room for the caller to retrieve the result).

    Returns
    -------
    list[float]
        The best sequence found during the search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a tiny safety margin

    # ------------------------------------------------------------------
    # 1) Choose length and initialise a probability vector.
    try:
        # Use the previously discovered construction if available.
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.size
    except Exception:                                 # fallback
        n = 2600
        init = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to sum = 1 (scaling does not change the objective)
    a = init.copy()
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    # ------------------------------------------------------------------
    # 2) Parameters of the optimiser
    mu = 0.005                 # smoothness for the soft‑max (smaller → tighter)
    eta0 = 0.6                 # initial step size (will decay as 1/√t)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Pre‑compute FFT length (must accommodate the longest convolution we use)
    M = 2 * n - 1                 # length of a*a
    L = 1 << ((n + M - 1).bit_length())   # next power of two ≥ n+M‑1

    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    while time.time() < deadline:
        iteration += 1

        # ----- convolution a * a  (used for the soft‑max weights) -----
        A_hat = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A_hat * A_hat, n=L)[:M]   # length 2n‑1

        # ----- soft‑max over the convolution values ------------------
        conv_max = conv.max()
        # subtract max for numerical stability
        w = np.exp((conv - conv_max) / mu)
        w /= w.sum()                                   # now Σ w = 1

        # ----- gradient = correlation of w with a --------------------
        # correlation g_i = Σ_j w_{i+j} a_j  (i = 0 … n‑1)
        w_rev = w[::-1]                               # reversed weights
        B_hat = np.fft.rfft(w_rev, n=L)
        corr_full = np.fft.irfft(A_hat * B_hat, n=L)[: n + M - 1]

        # indices that give the required correlation values
        idx = M - 1 - np.arange(n)                    # (M‑1‑i) for i=0..n‑1
        grad = corr_full[idx]

        # ----- projected gradient step --------------------------------
        eta = eta0 / np.sqrt(iteration)               # simple decay
        a_candidate = a - eta * grad
        a = _project_simplex(a_candidate)

        # ----- occasional random kick to escape plateaus -------------
        if iteration % 400 == 0:
            noise = np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a + noise)

        # ----- evaluate every few iterations (cheap enough) ----------
        if iteration % 10 == 0:
            seq = a.tolist()
            try:
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    no_improve = 0
                    print(f"[Iter {iteration:5d}] New best: {best_score:.6f}")
                else:
                    no_improve += 1
            except Exception:
                # any unexpected nan / inf handling – just skip this iteration
                pass

        # ----- early exit if we are about to run out of time ----------
        if time.time() + 0.05 >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
