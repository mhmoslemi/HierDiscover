def random_transfer(seq):
    n = len(seq)
    i = np.random.randint(n)
    if seq[i] <= 1e-12:
         # no mass to transfer; choose again
         return seq
    # choose amount to move
    eps = np.random.rand() * seq[i] * 0.5  # up to half
    j = np.random.randint(n)
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    return new_seq

def reduce_max_pair(seq):

    return new_seq

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy is two‑phased:

1. **Random sparse sampling** – generate many short sequences (n≈100‑500)
   whose mass is spread over a random subset of indices.  For a fixed total
   sum the expected auto‑convolution peak is about half of the uniform case,
   giving an evaluation close to 1.  The best of these candidates is kept.

2. **Local refinement** – starting from the best candidate, iteratively
   reduce the current convolution peak.  The algorithm identifies the index
   `k_max` where the convolution is maximal, finds the pair `(i, j)` that
   contributes the most to that peak and moves a small amount of mass from
   both `i` and `j` to other positions.  Simple random transfers are also
   tried.  A simulated‑annealing acceptance rule allows occasional uphill
   moves, helping to escape plateaus.

The whole search respects the 1000 s budget (a small safety margin is kept)
and returns the best sequence found.  All numbers are non‑negative and well
below the 0‑1000 clamp enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

def random_sparse_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a candidate of length ``n`` whose total mass is 1.0.
    The mass is placed uniformly on a random subset of size ``m`` where
    ``m`` is a random fraction of ``n`` (between 10% and 50%).
    """
    # Choose support size
    frac = rng.uniform(0.1, 0.5)
    m = max(2, int(n * frac))
    # Random distinct positions
    positions = rng.choice(n, size=m, replace=False)
    seq = np.zeros(n, dtype=float)
    seq[positions] = 1.0 / m                # sum == 1.0
    return seq


def random_transfer(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small random amount of mass from one index to another.
    The total sum stays unchanged.
    """
    n = len(seq)
    i = rng.integers(n)
    if seq[i] <= 1e-12:
        return seq.copy()
    # amount to move (up to 40 % of the source)
    eps = rng.random() * seq[i] * 0.4
    j = rng.integers(n)
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    return new_seq


def reduce_max_pair(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the pair of indices (i, j) that contributes most to the current
    convolution maximum and move a small fraction of their mass to two random
    other positions.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))                     # index of the current peak

    # valid i such that 0 <= i < n and 0 <= j = k-i < n
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    jdx = k - idx                                 # mirror indices

    # contribution of each pair to the peak
    contrib = seq[idx] * seq[jdx]
    max_pair = np.argmax(contrib)
    i = int(idx[max_pair])
    j = int(jdx[max_pair])

    # move a modest fraction from both ends
    eps = min(seq[i], seq[j]) * 0.15
    if eps < 1e-12:
        return seq.copy()

    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] -= eps
    total = 2.0 * eps

    # pick two *different* targets not equal to i or j
    candidates = list(set(range(n)) - {i, j})
    if len(candidates) >= 2:
        t1, t2 = rng.choice(candidates, size=2, replace=False)
        new_seq[t1] += total * 0.5
        new_seq[t2] += total * 0.5
    else:                       # a very unlikely fallback
        new_seq[i] += eps
        new_seq[j] += eps
    return new_seq


def evaluate(seq: np.ndarray) -> float:
    """
    Thin wrapper around the provided evaluate_sequence (already imported).
    """
    return evaluate_sequence(seq.tolist())   # type: ignore[name-defined]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) achieves a score
    below the target 1.503 within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – random sparse candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # a few different lengths give the algorithm a chance to discover
    # a good trade‑off between support size and number of convolution bins
    length_choices = [96, 128, 192, 256, 384, 512]

    # keep trying until we run out of time or achieve the target score
    while time.time() < deadline:
        n = int(rng.choice(length_choices))
        cand = random_sparse_candidate(n, rng)
        score = evaluate(cand)

        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] New best "
                f"score={best_score:.6f} (n={n})"
            )
            # Early exit if we already beat the target
            if best_score < 1.503:
                break

    # If for any reason we have not found a feasible candidate, fall back
    # to a simple uniform sequence (this always yields a score of 2.0).
    if best_seq is None:
        n = 128
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – local refinement (hill‑climbing + simulated annealing)
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    current_score = best_score

    iteration = 0
    while time.time() < deadline:
        # Decide which neighbourhood move to use
        if rng.random() < 0.7:
            new_seq = reduce_max_pair(seq, rng)
        else:
            new_seq = random_transfer(seq, rng)

        # Occasionally throw in a pure random perturbation to diversify
        if rng.random() < 0.02:
            new_seq = random_sparse_candidate(len(seq), rng)

        new_score = evaluate(new_seq)

        # Record improvement
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Improved to "
                f"{best_score:.6f} (iter {iteration})"
            )
            if best_score < 1.503:
                # Good enough, stop early
                break

        # Accept if it improves the current trajectory, otherwise accept
        # with a probability that slowly cools down (simulated annealing).
        if new_score < current_score:
            seq = new_seq
            current_score = new_score
        else:
            # temperature proportional to remaining fractional time
            remaining = max(0.0, deadline - time.time())
            temperature = (remaining / (budget_s * 0.9)) + 1e-12
            prob = np.exp(-(new_score - current_score) / temperature)
            if rng.random() < prob:
                seq = new_seq
                current_score = new_score

        iteration += 1

    # Convert to plain Python list as required by the framework
    return best_seq.tolist()
