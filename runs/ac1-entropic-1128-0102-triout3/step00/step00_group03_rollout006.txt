Implement project_to_simplex using algorithm:

def project_to_simplex(v, s):
    # ensure sum v = s, argmin ||w - v||_2 s.t. w >=0, sum w = s.
    # Implementation from Duchi et al.
    n = len(v)
    # shift by sorting descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    """Project vector v onto the simplex {x: x>=0, sum(x)=s}."""
    # Implementation from Duchi et al.
    n = v.shape[0]
    if s < 0:
        raise ValueError("Radius s must be non-negative")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]  # last index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence for the minimisation problem.

The algorithm is a projected‑gradient descent on the objective
    f(a) = max_k (a * a)[k]   (the peak of the self‑convolution)
with the linear constraints  a_i ≥ 0  and  Σ a_i = √(2 n).

Key ideas
---------
* Initialise a with the known construction (if available) plus a small
  random perturbation, or with a uniform distribution plus noise.
* In each iteration:
    - Compute the linear convolution a*a.
    - Locate the index k where the convolution attains its maximum.
    - The sub‑gradient of the peak at k is g_i = a_i + a_{k-i}
      (a_{k-i}=0 when k-i is out of range).  This points to the
      coordinates that enlarge the peak.
    - Take a step opposite to the gradient and project the result back
      onto the simplex {x ≥ 0, Σx = √(2 n)}.  The projection automatically
      shifts mass from “high‑gradient’’ coordinates to “low‑gradient’’
      ones, flattening the convolution.
* The step size is annealed: large at the beginning, very small near the
  deadline, allowing coarse adjustments followed by fine tuning.
* Several independent restarts are performed until the time budget
  expires; the best sequence found over all restarts is returned.
* For speed the evaluation of a candidate is computed directly from the
  convolution (the same formula used by `evaluate_sequence`).

The method works purely with NumPy, needs no external optimisation
solvers and stays well within the 1 CPU limit.
"""

import time
import math
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector v onto the simplex {x : x_i ≥ 0, Σ x_i = s}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)

    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_at_peak(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the peak convolution (at index k) w.r.t. a.
    For each i:  g_i = a_i + a_{k-i}  (the second term is omitted when k-i is out of range).
    """
    n = a.shape[0]
    idx = np.arange(n)
    other = k - idx
    mask = (other >= 0) & (other < n)

    g = a.copy()
    g[mask] += a[other[mask]]
    return g


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function height sequence.

    Returns a list of non‑negative floats.  The function runs for at most
    `budget_s` seconds and tries to beat the current bound of 2.0.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # keep a small safety margin

    # ------------------------------------------------------------------
    # problem dimensions (use the length of the known construction when present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 3401          # a safe default similar to the reference run
        base_seq = None

    S_target = math.sqrt(2.0 * n)          # Σ a_i should equal this value

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    restarts = 0

    # ------------------------------------------------------------------
    # main optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restarts += 1

        # -------- initialiser -------------------------------------------------
        if base_seq is not None and restarts == 1:
            # start from the known construction, add a tiny perturbation
            a = base_seq.copy()
            a += np.random.randn(n) * (S_target / n) * 0.05
        else:
            # uniform + moderate random noise
            a = np.full(n, S_target / n)
            a += np.random.randn(n) * (S_target / n) * 0.5

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _simplex_projection(a, S_target)

        # initial score (computed analytically, avoids the expensive checks)
        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart {restarts:3d} "
                  f"initial score {best_score:.6f}")

        # -------- inner projected‑gradient descent -----------------------------
        inner_iter = 0
        while time.time() < deadline:
            inner_iter += 1

            # compute convolution and locate the peak
            conv = np.convolve(a, a)
            k_peak = int(np.argmax(conv))
            max_conv = conv[k_peak]

            # gradient (sub‑gradient of the peak)
            g = _gradient_at_peak(a, k_peak)

            # step‑size schedule: large at the start, tiny near the end
            elapsed = time.time() - start_time
            frac_done = elapsed / (deadline - start_time)          # 0 … 1
            # linearly decay from 0.5 to 0.001
            step_factor = 0.5 * (1.0 - frac_done) + 0.001 * frac_done

            # normalise step by the gradient magnitude
            g_norm = np.linalg.norm(g)
            if g_norm == 0.0:
                break
            # the factor “max_conv / g_norm” puts the update in the same scale
            step = step_factor * max_conv / (g_norm + 1e-12)

            # gradient descent step and projection back onto the simplex
            a_new = a - step * g
            a_new = _simplex_projection(a_new, S_target)

            # quick analytic evaluation (identical to evaluate_sequence for feasible vectors)
            conv_new = np.convolve(a_new, a_new)
            max_conv_new = conv_new.max()
            new_score = 2.0 * n * max_conv_new / (a_new.sum() ** 2)

            # accept if we improve, otherwise accept with a tiny probability
            if new_score < cur_score or np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] restart {restarts:3d} "
                          f"iter {inner_iter:6d} new best {best_score:.6f}")

            # occasional random “kick” to avoid stagnation
            if inner_iter % 500 == 0:
                i = np.random.randint(n)
                delta = np.random.randn() * step_factor * (S_target / n) * 0.2
                a[i] = max(0.0, a[i] + delta)
                a = _simplex_projection(a, S_target)
                cur_score = 2.0 * n * np.convolve(a, a).max() / (a.sum() ** 2)

            # stop early if we already passed a reasonable target
            if best_score <= 1.55:
                break

        # global early exit
        if best_score <= 1.55:
            break

    # ------------------------------------------------------------------
    # fallback (should never happen, but keeps the interface safe)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, S_target / n)

    # Return a plain Python list as required by the checker
    return best_seq.tolist()
