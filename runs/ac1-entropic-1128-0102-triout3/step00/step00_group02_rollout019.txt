"""
propose_candidate
-----------------

A stochastic hill‑climbing / simulated‑annealing search for a step‑function
(sequence of non‑negative heights) that minimises the evaluation metric

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is a linear convolution.  Because the objective is invariant to
uniform scaling we keep the total mass of the sequence equal to one
(at every iteration we renormalise).  The search works as follows:

1. **Initialisation** – start from a uniform distribution on `n` points.
   The length `n` is chosen as 2000 (a good trade‑off between speed and
   the theoretical lower bound 2 · n·max/​sum² ≈ 1).

2. **Perturbations** – at each step a small amount of mass `δ` is moved
   from one index to another (maintaining non‑negativity).  Two kinds of
   moves are performed:
   * a *random* move (both indices are chosen uniformly);
   * a *targeted* move that tries to reduce the current worst convolution
     entry: the index `k` where the convolution attains its maximum is
     identified, then a source index participating in that maximum is
     selected and a small amount of its mass is transferred to a random
     destination.

3. **Acceptance** – the new sequence is accepted if it improves the score.
   Otherwise it is kept with a probability that decreases exponentially
   with the score increase (`temperature`), implementing a simple simulated‑
   annealing schedule.  The temperature is multiplied by a constant decay
   factor each iteration.

4. **Book‑keeping** – the best sequence found so far (lowest score) is
   stored and finally returned.

The algorithm runs until the supplied time budget is exhausted (leaving a
small safety margin).  All operations are pure NumPy and therefore fast
enough to perform many thousands of iterations within the 1000 s limit.
The returned sequence is guaranteed to contain only non‑negative floats
and to respect the constraints of `evaluate_sequence`.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a sequence of step heights with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default: 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------
    # Length of the candidate sequence (larger values give a smaller
    # theoretical lower bound, but also increase the cost of a convolution).
    n = 2000

    # Start from a uniform distribution (sum = 1).
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Keep the best solution.
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # Simulated‑annealing parameters.
    temperature = 0.05          # initial acceptance temperature
    decay = 0.9999              # exponential decay per iteration
    step_scale = 0.02           # typical relative mass moved per step

    # Time management.
    deadline = time.time() + budget_s - 1.0   # leave a second margin

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Choose a perturbation.
        # ------------------------------------------------------------------
        if rng.random() < 0.7:
            # Targeted move: reduce the current maximal convolution entry.
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))

            # Indices that actually contribute to the max entry.
            # i contributes if 0 <= i < n and 0 <= k_max-i < n.
            possible_i = np.arange(max(0, k_max - (n - 1)), min(n - 1, k_max) + 1)
            if possible_i.size == 0:
                # fallback to a random move
                src = rng.integers(n)
                dst = rng.integers(n)
            else:
                src = rng.choice(possible_i)
                dst = rng.integers(n)
        else:
            # Pure random move.
            src = rng.integers(n)
            dst = rng.integers(n)

        if src == dst:
            continue

        # Amount of mass to move: a small fraction of the source weight.
        delta = step_scale * a[src] * rng.random()
        if delta <= 0.0:
            continue

        # Ensure we do not create negative entries.
        if a[src] < delta:
            delta = a[src]

        # Apply the move on a temporary copy.
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # Renormalise to keep total mass = 1 (protects against round‑off).
        total = a_new.sum()
        if total <= 0.0:
            continue
        a_new /= total

        # --------------------------------------------------------------
        # Evaluate the new candidate.
        # --------------------------------------------------------------
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # Any numerical issue – reject this move.
            continue

        # --------------------------------------------------------------
        # Acceptance criterion.
        # --------------------------------------------------------------
        accept = False
        if score < best_score:
            # New best solution.
            best_score = score
            best_a = a_new.copy()
            accept = True
        else:
            # Simulated‑annealing acceptance.
            prob = np.exp(-(score - best_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new

        # --------------------------------------------------------------
        # Temperature schedule.
        # --------------------------------------------------------------
        temperature *= decay

        # Periodically report progress (optional, harmless for the judge).
        if iteration % 10000 == 0:
            print(f"Iter {iteration:,d} – best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    print(f"Finished after {iteration:,d} iterations – final best_score={best_score:.6f}")
    return best_a.tolist()
