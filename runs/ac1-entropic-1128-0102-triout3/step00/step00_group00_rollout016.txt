def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave foot

    best_seq = None
    best_score = float('inf')
    iter = 0

    while time.time() < deadline:
        # randomly pick length L in [100, 2000]
        L = np.random.randint(100, 2000)
        # random delta in (0.001, 0.01)
        delta = np.random.uniform(0.001, 0.02)  # maybe up to 0.02
        r = 1 - delta
        seq = [(1 - r) * (r ** i) for i in range(L)]
        # optionally add small noise
        if np.random.rand() < 0.2:
            noise = np.random.randn(L) * 1e-4
            seq = [max(0.0, seq[i] + noise[i]) for i in range(L)]
        # evaluate
        try:
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best: {best_score:.6f} length {L} delta {delta:.6f}")
        except Exception:
            pass

        # optionally local improvement: perturb best_seq
        if best_seq is not None and np.random.rand() < 0.1:
            # copy best and do small random transfer
            seq2 = best_seq.copy()
            # choose two distinct indices
            i, j = np.random.choice(len(seq2), size=2, replace=False)
            # move epsilon from i to j
            epsilon = np.random.rand() * seq2[i] * 0.1
            seq2[i] = max(0.0, seq2[i] - epsilon)
            seq2[j] = seq2[j] + epsilon
            try:
                score2 = evaluate_sequence(seq2)
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2
                    print(f"Improved via hillclimb: {best_score:.6f}")
            except Exception:
                pass

    if best_seq is None:
        # fallback: use uniform sequence length 100 (should give eval ~2)
        best_seq = [1.0]*100
    return best_seq

def anneal(N, max_iters, init=None):
    if init is None:
        a = np.random.rand(N)
    else:
        a = init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    current_score = best_score

    temp0 = 1.0
    for iter in range(max_iters):
        # linear cooling
        temp = temp0 * (1 - iter / max_iters)
        # propose move
        i, j = np.random.choice(N, size=2, replace=False)
        if a[i] == 0:
            continue
        delta = np.random.rand() * a[i] * 0.1  # up to 10% of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonneg
        if a_new[i] < 0:
            continue
        # Normalize might be unnecessary because sum unchanged
        # evaluate
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < current_score or np.random.rand() < np.exp((current_score - new_score) / (temp + 1e-12)):
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                # optional print
        # occasional restart
        if iter % 10000 ==0 and np.random.rand() < 0.1:
            a = np.random.rand(N)
            a = a / a.sum()
            current_score = evaluate_sequence(a.tolist())
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    
    # strategy 1: exponential sweep
    for N in [200, 300, 400, 500]:
        if time.time() > deadline: break
        for delta in np.linspace(0.001, 0.02, 40):
            r = 1 - delta
            seq = [(1 - r) * (r ** i) for i in range(N)]
            # normalize explicitly to sum=1
            seq = np.array(seq)
            if seq.sum() == 0:
                continue
            seq = seq / seq.sum()
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"Exp sweep N={N} delta={delta:.5f} score={score:.6f}")
                if best_score < 1.5:
                    return best_seq
    # strategy 2: evolutionary search with differential evolution (small N)
    for N in [30, 40, 50]:
        if time.time() > deadline: break

        _tmp_N = N
        bounds = [(0.0, 1.0) for _ in range(N)]

            seq = result.x
            seq = np.maximum(seq, 0)
            seq = seq / np.sum(seq)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"DE N={N} score={score:.6f}")
                if best_score < 1.5:
                    return best_seq
    # strategy 3: simulated annealing (various N)
    while time.time() < deadline:

"""
Simple multi‑strategy optimizer for the step‑height evaluation problem.

The function `propose_candidate` runs three lightweight searches
inside the allowed time budget:

1.  Exponential families – a closed‑form family a_i = (1‑r) r^i
    (truncated and re‑normalised).  A tiny grid over length and decay
    explores these analytically promising candidates.

2.  Differential evolution (DE) on very short sequences (N≈30‑50).  
    DE is a global optimiser that works without any closures – the
    objective uses a global variable `_tmp_N` to know the current size.

3.  Simulated‑annealing hill‑climbing on medium‑size sequences
    (N≈80‑200).  Random mass transfers between two positions are
    accepted either if they improve the score or with a temperature‑
    dependent probability, allowing the search to escape local minima.

All candidates are normalised to sum‑to‑one, which is safe because the
evaluation is homogeneous under scaling.  The best sequence found
during the run (score < 1.5 if possible) is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper for the DE optimiser – uses a global variable to know the length
_tmp_N = None

def _de_objective(x):
    """Objective for differential_evolution: normalise x and evaluate."""
    if _tmp_N is None:
        return np.inf
    # clip negatives, normalise to sum 1 (the evaluation is scale‑invariant)
    x = np.maximum(x, 0.0)
    s = np.sum(x)
    if s < 1e-12:
        return np.inf
    x = x / s
    # `evaluate_sequence` is supplied by the runner
    return evaluate_sequence(x.tolist())

# ----------------------------------------------------------------------
def _exp_sequence(N, delta):
    """Return a normalised exponential decay sequence of length N.

    a_i = (1‑r) r**i   with r = 1‑delta, truncated to N terms.
    """
    r = 1.0 - delta
    seq = [(1.0 - r) * (r ** i) for i in range(N)]
    seq = np.array(seq, dtype=float)
    total = seq.sum()
    if total == 0.0:
        return None
    return (seq / total).tolist()

# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a step‑height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a little safety margin

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # 1) Scan exponential families (fast, cheap)
    # --------------------------------------------------------------
    for N in (200, 300, 400, 500):
        if time.time() > deadline:
            break
        for delta in np.linspace(0.001, 0.02, 40):
            seq = _exp_sequence(N, delta)
            if seq is None:
                continue
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score, best_seq = score, seq
                print(f"[exp] N={N:4d} delta={delta:.5f} → score={score:.6f}")
                if best_score < 1.5:
                    return best_seq

    # --------------------------------------------------------------
    # 2) Differential evolution on tiny instances (N=30..50)
    # --------------------------------------------------------------
    global _tmp_N
    for N in (30, 40, 50):
        if time.time() > deadline:
            break
        _tmp_N = N
        bounds = [(0.0, 1.0) for _ in range(N)]

        # Small number of generations – enough to move away from the uniform start
        result = differential_evolution(
            _de_objective,
            bounds,
            maxiter=30,
            popsize=15,
            polish=False,
            updating='deferred',
            disp=False,
            seed=seed,
        )
        if result.success:
            seq = np.maximum(result.x, 0.0)
            seq /= seq.sum()
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq = score, seq.tolist()
                print(f"[DE ] N={N:2d} → score={score:.6f}")
                if best_score < 1.5:
                    return best_seq

    # --------------------------------------------------------------
    # 3) Simulated‑annealing hill‑climbing on medium‑size vectors
    # --------------------------------------------------------------
    while time.time() < deadline:
        # Random medium size (keep conv cheap)
        N = np.random.randint(80, 200)
        # start from a random point on the simplex
        a = np.random.rand(N)
        a /= a.sum()
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.tolist()
            print(f"[SA ] init N={N:3d} → score={cur_score:.6f}")

        # SA parameters
        max_iters = 20000
        for it in range(max_iters):
            if time.time() > deadline:
                break

            # temperature linearly decreasing from 0.5 to 0.01
            temp = 0.5 * (1.0 - it / max_iters) + 0.01

            # propose moving a random amount from i to j
            i, j = np.random.choice(N, size=2, replace=False)
            if a[i] <= 0.0:
                continue
            # up to 20 % of the donor mass is shifted
            delta = np.random.rand() * a[i] * 0.2
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_score = evaluate_sequence(a_new.tolist())
            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < math.exp((cur_score - new_score) / temp):
                a, cur_score = a_new, new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, a_new.tolist()
                    print(f"[SA ] improve N={N:3d} it={it:5d} → score={new_score:.6f}")
                    if best_score < 1.5:
                        return best_seq

            # occasional random restart to diversify search
            if it % 5000 == 0 and np.random.rand() < 0.05:
                a = np.random.rand(N)
                a /= a.sum()
                cur_score = evaluate_sequence(a.tolist())

    # --------------------------------------------------------------
    # Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 100
    return best_seq
