"""Propose a low‐evaluation step‑function using a Gaussian‑shaped sequence.

The evaluation function is scale‑invariant:
    score = 2·n·max(conv(s,s)) / (sum(s))**2 .
For a continuous Gaussian f(x) with variance σ² and ∫f = S,
    max(conv) = S²/(2σ√π)  ⇒  score ≈ n/(σ√π).

Hence, fixing a relative width σ = α·n (α∈(0,0.5]) gives
    score ≈ 1/(α·√π) .
Choosing α≈0.45 yields a score near 1.25, well below the target 1.503.
The algorithm therefore searches over lengths n and relative widths α,
constructs the corresponding discrete Gaussian, rescales it to the
required L¹‑norm (√(2n)), evaluates the sequence and keeps the best one.
A small random‑perturbation phase is added to possibly improve the result
within the remaining time budget.

def _gaussian_sequence(n: int, alpha: float) -> list[float]:
    """Return a symmetric Gaussian sequence of length n.

    The Gaussian has standard deviation sigma = alpha * n.
    The sequence is scaled so that its L1‑norm equals sqrt(2·n),
    which keeps the values far below the 1000‑cap used by evaluate_sequence.
    """
    sigma = max(alpha * n, 1e-9)
    # centre the Gaussian
    x = np.arange(n) - (n - 1) / 2.0
    raw = np.exp(-0.5 * (x / sigma) ** 2)
    # rescale to the required sum
    target_sum = math.sqrt(2.0 * n)
    scale = target_sum / raw.sum()
    seq = (raw * scale).tolist()
    return seq

    # ------------------------------------------------------------------
    # lengths to explore – they are modest enough for fast convolution
    n_vals = list(range(200, 2001, 200)) + [2500, 3000, 3500, 4000]
    # relative widths – α must stay below 0.5 to keep the Gaussian inside the interval
    alphas = np.linspace(0.30, 0.49, 10)

    for n in n_vals:
        for alpha in alphas:
            seq = _gaussian_sequence(n, alpha)
            try:
                sc = evaluate_sequence(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq
                    print(f"[grid] n={n:4d} α={alpha:.3f} → score={sc:.6f}")
                # early stop if we have already beaten the target by a comfortable margin
                if best_score <= 1.20:
                    break
            except Exception:
                pass
        if best_score <= 1.20:
            break

    # ------------------------------------------------------------------
    # random local refinements (perturb the current best)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if best_seq is None:
            # fall back to a random start if everything failed
            n = random.randint(200, 2000)
            seq = np.random.rand(n) * 0.1
        else:
            seq = best_seq.copy()
            n = len(seq)

        # pick a few random indices and perturb them
        for _ in range(3):
            idx = random.randint(0, n - 1)
            # scale perturbation to the current magnitude
            magnitude = max(seq) if max(seq) > 0 else 1.0
            delta = (np.random.randn() * 0.02) * magnitude
            seq[idx] = max(0.0, seq[idx] + delta)

        # renormalise the L1‑norm to sqrt(2·n) (the scaling does not affect the score)
        target = math.sqrt(2.0 * n)
        total = sum(seq)
        if total > 0:
            seq = [x * target / total for x in seq]

        try:
            sc = evaluate_sequence(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(f"[perturb] n={n:4d} → score={sc:.6f}")
                # stop early if we are already well below the target
                if best_score <= 1.20:
                    break
        except Exception:
            pass

    print(f"[final] best score = {best_score:.6f} (length={len(best_seq) if best_seq else 0})")
    return best_seq
