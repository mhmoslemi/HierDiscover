def solve_convolution_lp(f, rhs):
    n = len(f)
    # variables b_j ≥ 0
    # constraints: A b ≤ rhs, where A_{k,j} = f_{k-j} if valid else 0.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                # term f_i * b_j
                A[k, j] = f[i]
    # add non-negativity via bounds
    c = -np.ones(n) # maximize sum b
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, rhs), bounds=bounds, method='highs')
    return result.x if result.success else None

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm is a simple iterative improvement that alternates between
a linear‑programming direction (as in the paper) and small random
perturbations.  For each iteration we:

1. Normalise the current sequence so that its ℓ¹–norm equals √(2·n).
2. Compute the current convolution bound M = max(conv(f,f)).
3. Solve a linear program that finds a non‑negative sequence g with
   maximal total mass while keeping conv(f,g) ≤ M.
4. Blend the current sequence with g using several step sizes and keep
   the best improvement (a cheap line‑search).
5. If the LP fails, perform a random mass‑transfer between two positions.
6. Keep track of the best sequence ever seen.

The loop runs until the time budget (≈ 1000 s) is exhausted, after which
the best sequence found is returned.  The method works for any length
and usually finds a sequence with an evaluation score well below the
initial uniform bound of 2.0.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP
        maximise   sum(b)
        subject to conv(f_seq, b) ≤ rhs
                   b ≥ 0
    Returns the optimal b (numpy array) or None if infeasible.
    """
    n = len(f_seq)
    # Build the constraint matrix A where each row corresponds to a shift k
    # and A[k, j] = f[i] with i + j = k (i.e. i = k - j).
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # only i = 0..n-1 that give a valid j = k-i are needed
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs)

    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return np.maximum(res.x, 0.0)   # numerical safety
    return None
