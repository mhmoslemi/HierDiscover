def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep margin

    # choose length
    n = 2000  # maybe 2500?
    # maybe adapt n based on time left? Keep 2000.
    # initialize with uniform distribution
    a = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"Initial score: {best_score}")

    # hyperparameters
    eta = 0.5
    eta_decay = 0.9995  # each iteration
    epsilon = 1e-6  # tolerance for near-maximum
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(a, a)
        max_val = conv.max()
        # optionally compute near-maximum set
        # Use tolerance relative to max
        tol = max_val * 0.001  # 0.1% tolerance
        near_mask = conv >= max_val - tol
        # compute weight vector w: 1 for near points, else 0
        w = near_mask.astype(np.float64)
        if w.sum() == 0:  # fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        # compute gradient via convolution: grad = 2 * conv_rev_a[w]
        # As derived: grad_i = 2 * sum_j a_j * w_{i+j}
        # Use convolution of a_rev and w
        a_rev = a[::-1]
        conv_aw = np.convolve(a_rev, w)
        grad = 2.0 * conv_aw[n-1 : n-1 + n]  # length n slice

        # apply exponentiated gradient step
        # Here we want to move opposite to grad to reduce max.
        step = eta * grad
        # clip step size to avoid underflow/overflow
        # exponentiate negative step
        a = a * np.exp(-step)
        # re-normalize to sum=1
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            # reinitialize if degenerated
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        a = a / sum_a

        # optional small random perturbation to avoid stagnation
        if iteration % 50 == 0:
            # Add Gaussian noise
            noise = np.random.randn(n) * 0.001
            a = a + noise
            a = np.clip(a, 0, None)
            a = a / a.sum()

        # evaluate
        current_score = evaluate_sequence(a.tolist())
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            print(f"Iter {iteration}: new best score {best_score}")

        # decay eta
        eta *= eta_decay
        iteration += 1

    print(f"Final best score: {best_score}")
    return best_seq.tolist()

def full_convolution_fft(x, y):
    n = len(x) + len(y) -1

        A = np.zeros((2*n-1, n))

Propose a candidate sequence for the self‑convolution evaluation problem.

The algorithm works directly on a probability vector `a` (non‑negative, sum = 1)
because the evaluation function is scale‑invariant.
We minimise the L2 distance between the full self‑convolution `c = a*a`
and the uniform target `c_target = 1/(2n‑1)` (the average value of any
convolution with unit L1‑norm).  The gradient of the loss
    L = ||c - c_target||_2^2
is
    ∇L_i = 2 * Σ_j a_j * (c_{i+j} - c_target) .
The gradient can be computed efficiently by a single FFT‑based convolution.
An additive projected gradient step (with a tiny positivity floor) is applied,
followed by a renormalisation to keep the vector on the simplex.
Occasional small random jitter helps to escape plateaus.
The best sequence seen during the allotted time is returned.
"""

# Helper: full (linear) convolution using a power‑of‑two FFT.
def _full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.size + y.size - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]                               # exact length


# ----------------------------------------------------------------------
# Gradient of the squared‑error to the flat target.
def _grad_flatten(a: np.ndarray, target: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute gradient of  ||a*a - target||_2^2  w.r.t. a and also return the
    current convolution a*a.
    """
    conv = _full_convolve_fft(a, a)               # length 2n‑1
    residual = conv - target                      # same length
    # gradient = 2 * (a_rev  * residual)  over the central n entries
    a_rev = a[::-1]
    tmp = _full_convolve_fft(a_rev, residual)    # length 3n‑2
    n = a.size
    grad = 2.0 * tmp[n - 1 : n - 1 + n]           # slice of length n
    return grad, conv
