    N = 1 << (n-1).bit_length()

"""
Simulated annealing + directed mass transfer.

The algorithm works in the probability simplex:
* Start from the uniform step‑function (or the provided ``height_sequence_1``)
  with a tiny random perturbation to break symmetry.
* At each iteration we propose moving a small amount of mass ε from an index i
  to another index j (keeping the total sum unchanged).
* The acceptance rule is a simple simulated‑annealing Metropolis test:
  improvements are always kept, worse moves are accepted with probability
  exp(−Δ/ T) where the temperature T is slowly cooled.
* Every few iterations we compute a sub‑gradient of the current maximal
  convolution value (the indices where the convolution attains its maximum)
  and use it to bias the choice of i (high‑gradient, i.e. contributes a lot
  to the peak) and j (low‑gradient, contributes little).  This drives the
  mass away from the “peaky’’ parts of the autocorrelation.
* The convolution is evaluated by FFT (O(N log N)) which is fast enough
  for sequences of length ~6000.  The best sequence found before the
  time budget expires is returned.

The method does not rely on the linear‑programming direction used by the
previous SOTA code; it explores the space more globally and can escape the
uniform fixed point, often achieving a score below the target 1.503.
"""

import numpy as np
import time
import math

def _full_convolution_fft(x: np.ndarray, N_fft: int) -> np.ndarray:
    """Return the full linear convolution x*x using FFT (real‐valued)."""
    fx = np.fft.rfft(x, n=N_fft)
    # multiplication in frequency domain (real → real)
    conv = np.fft.irfft(fx * fx, n=N_fft)[: 2 * len(x) - 1]
    return conv


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  Returns the best sequence found as a list of
    floats before the time budget runs out.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence
    # ------------------------------------------------------------------
    try:
        # Use the provided construction if it exists.
        init = np.array(height_sequence_1, dtype=float)
        n = init.size
    except Exception:                         # fallback: uniform
        n = 6000
        init = np.full(n, 1.0 / n, dtype=float)

    # tiny random noise to break the perfect symmetry of the uniform vector
    init += np.random.randn(n) * 1e-4
    init = np.clip(init, 0.0, None)
    a = init / init.sum()                     # keep it on the simplex

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT size (next power of two)
    # ------------------------------------------------------------------
    N_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Helper to compute the objective quickly (no call to evaluate_sequence)
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray, conv_cache: np.ndarray = None) -> float:
        """Return 2*n*max(conv)/sum(vec)^2 .  conv may be supplied to save work."""
        if conv_cache is None:
            conv_cache = _full_convolution_fft(vec, N_fft)
        max_c = conv_cache.max()
        s = vec.sum()
        return 2.0 * n * max_c / (s * s)

    # initial convolution and score
    cur_conv = _full_convolution_fft(a, N_fft)
    cur_score = _score(a, cur_conv)

    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated annealing parameters
    # ------------------------------------------------------------------
    T = 0.02                     # initial temperature
    T_min = 1e-7
    cooling = 0.9995

    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        #   5a. Choose i (source) and j (destination) for a tiny mass transfer
        # ------------------------------------------------------------------
        # Every few steps we bias the choice using a cheap gradient estimate.
        if iteration % 7 == 0:
            # indices where the convolution attains its current maximum
            max_val = cur_conv.max()
            mask = (cur_conv >= max_val - 1e-12).astype(float)

            # sub‑gradient of the max‐convolution w.r.t. a:
            #   g_i = 2 * Σ_{k in max set} a_{k-i}
            #   which is the convolution of mask with the reversed a.
            grad_full = np.convolve(mask, a[::-1])
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

            # pick a “bad’’ index (large contribution) and a “good’’ one
            top_k = max(1, n // 30)          # ~3 % of positions
            bad_idx = np.random.choice(np.argpartition(-grad, top_k)[:top_k])
            good_idx = np.random.choice(np.argpartition(grad, top_k)[:top_k])
            i, j = int(bad_idx), int(good_idx)
            if i == j:
                i, j = np.random.choice(n, 2, replace=False)
        else:
            i, j = np.random.choice(n, 2, replace=False)

        # ------------------------------------------------------------------
        #   5b. Propose a move (transfer a small amount of mass)
        # ------------------------------------------------------------------
        max_eps = a[i]
        if max_eps <= 0.0:
            continue
        # transfer up to half of the source mass, scaled by a random factor
        eps = max_eps * np.random.rand() * 0.5
        if eps == 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # ------------------------------------------------------------------
        #   5c. Evaluate the proposal (fast FFT based)
        # ------------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, N_fft)
        new_score = _score(a_new, conv_new)

        # ------------------------------------------------------------------
        #   5d. Metropolis acceptance
        # ------------------------------------------------------------------
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability to accept a worsening move
            if T > 0:
                prob = math.exp(-delta / T)
                accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_conv = conv_new
            cur_score = new_score

            # keep the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                last_improve_iter = iteration
                print(f"[iter {iteration}] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        #   5e. Temperature schedule
        # ------------------------------------------------------------------
        T = max(T_min, T * cooling)

        # ------------------------------------------------------------------
        #   5f. Occasional random restart if no improvement for a while
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 5000 and best_score > 1.8:
            # fresh random start
            a = np.random.rand(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_conv = _full_convolution_fft(a, N_fft)
            cur_score = _score(a, cur_conv)
            last_improve_iter = iteration
            print(f"[restart] iter {iteration}  score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[finish] best score = {best_score:.6f} after {iteration} iterations")
    return best_a.tolist()
