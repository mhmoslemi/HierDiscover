"""
Simple hill‑climbing search for a non‑negative step function that minimises
evaluate_sequence().

The algorithm:
1. Choose a handful of sensible initial shapes (uniform, random exponential,
   sinusoid, Hann window, Gaussian, and the provided height_sequence_1).
2. Normalise each candidate so that its L1‑norm equals sqrt(2·n); with this
   normalisation evaluate_sequence() returns exactly the maximal convolution
   entry, i.e. 2·n·max(b)/sum² == max(b).
3. For a fixed candidate run an iterative improvement loop until the allocated
   time expires:
   * Compute the self‑convolution (FFT based, O(n log n)).
   * Identify the (few) indices where the convolution is near its maximum.
   * Approximate a sub‑gradient of the maximal entry:
       g_i = Σ_{k∈max} 2·a_{k‑i}   (only indices that actually contribute).
   * Move a small amount of mass from the index with the largest gradient
     (i.e. contributing most to the peak) to the index with the smallest
     gradient, keep the total sum constant by a renormalisation.
   * Accept the move if the score improves; otherwise accept it with a tiny
     probability that decays with the remaining time (simulated‑annealing style).
   * The step‑size gradually shrinks.
4. Keep the best sequence seen over all initial shapes and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast self–convolution using the FFT (real → real)
def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length power‑of‑2."""
    n = a.size
    # next power of two ≥ 2·n  (full convolution length = 2·n‑1)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[: 2 * n - 1]
    # numerical noise may produce tiny negative values – clip safely
    return np.real(conv)


# ----------------------------------------------------------------------
# Gradient (sub‑gradient of the maximal convolution entry)
def _gradient_at_max(a: np.ndarray, conv: np.ndarray, thresh: float = 0.99) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) w.r.t. a.
    Only the entries where conv is within `thresh` of the maximum are used.
    """
    n = a.size
    max_val = conv.max()
    # indices we care about (usually a handful)
    idx = np.where(conv >= max_val * thresh)[0]
    grad = np.zeros_like(a)

    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]          # derivative of a_i a_j (both orders)

    return grad


# ----------------------------------------------------------------------
# Propose a small redistribution move based on the gradient
def _propose_move(a: np.ndarray, grad: np.ndarray, step_frac: float = 0.05) -> np.ndarray:
    """
    Return a new vector where a small amount of mass is transferred
    from the index with the largest gradient to the one with the smallest.
    """
    n = a.size
    i_reduce = int(np.argmax(grad))
    i_inc = int(np.argmin(grad))

    # Ensure we have two different positions
    if i_reduce == i_inc:
        i_reduce = np.random.randint(n)
        i_inc = np.random.randint(n)

    # Transfer a modest fraction of the mass at i_reduce
    delta = step_frac * a[i_reduce]
    delta = min(delta, a[i_reduce])           # never go negative

    if delta <= 0.0:                         # all‑zero case – add a tiny bump
        delta = step_frac * a.mean()

    new_a = a.copy()
    new_a[i_reduce] -= delta
    new_a[i_inc] += delta
    return new_a


# ----------------------------------------------------------------------
def _hill_climb(initial: np.ndarray,
                target_sum: float,
                budget: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple stochastic hill‑climbing search starting from `initial`.
    Returns the best sequence found and its score.
    """
    cur = initial.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score

    step_frac = 0.08      # initial mass‑transfer fraction
    thresh = 0.99         # how close to the current max we consider “active”

    iteration = 0
    while time.time() < start_time + budget:
        # --- convolution and gradient ------------------------------------
        conv = _self_convolve(cur)
        max_val = conv.max()
        grad = _gradient_at_max(cur, conv, thresh)

        # --- propose a move ---------------------------------------------
        cand = _propose_move(cur, grad, step_frac)

        # normalise to the fixed L1‑norm required by the evaluator
        s = cand.sum()
        if s > 0.0:
            cand *= (target_sum / s)
        else:                                   # pathological – re‑initialise
            cand = np.maximum(0.0, np.random.rand(cur.size))
            cand *= (target_sum / cand.sum())

        cand_score = evaluate_sequence(cand.tolist())

        # --- acceptance --------------------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f"[{time.time():.1f}] New best: {best_score:.6f}")
        else:
            # occasional uphill moves (simulated annealing style)
            remaining = start_time + budget - time.time()
            temperature = max(1e-8, remaining / budget)   # linearly decays
            prob = np.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                cur, cur_score = cand, cand_score

        # --- schedule ----------------------------------------------------
        iteration += 1
        if iteration % 100 == 0:
            step_frac = max(0.003, step_frac * 0.9)  # slowly shrink step

    return best, best_score

        now = time.time()
        remaining = deadline - now
        if remaining <= 0.0:
            break
        budget = min(per_candidate_budget, remaining)

        print(f"[{now:.1f}] Starting hill‑climb for candidate {idx} "
              f"(budget {budget:.1f}s)")

        seq, score = _hill_climb(cand, target_sum, budget, now)

        if score < best_score:
            best_score = score
            best_overall = seq.copy()
            print(f"*** Global best updated: {best_score:.6f}")

        # Early exit if we have already beaten the target bound
        if best_score <= 1.503:
            print("Target bound reached – stopping early.")
            break

    # If for some reason nothing was found, fall back to uniform.
    if best_overall is None:
        best_overall = uniform

    return best_overall.tolist()
