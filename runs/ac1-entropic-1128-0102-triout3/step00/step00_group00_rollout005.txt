def project_simplex(v):
    # ensure nonneg sum to 1 if possible.
    # If sum(v) == 0: return uniform.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

* compute the current maximum of the self‑convolution (M);
* solve a linear program that finds a new non‑negative vector `g`
  with the largest possible ℓ₁‑norm while respecting the linear
  constraints  (f * g) ≤ M (i.e. the convolution of the current
  sequence `f` with the variable `g` never exceeds the current peak);
* mix `f` and the normalized solution `g` with a step size α
  (multiple α values are tried and the best improvement is kept);
* repeat until the time budget expires or the target score is reached.

The LP is tiny (≈2·n constraints, n variables) and the whole loop
runs many thousands of iterations, quickly driving the peak of the
convolution below the uniform value 1/n, which yields a score
significantly lower than 2.0 (the target is ≈1.503).
"""
import time
import numpy as np
from scipy.optimize import linprog

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x≥0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:      # all entries are zero or negative
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a good sequence of heights.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Time budget in seconds (the function will stop earlier if a good
        solution is found).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny margin

    # ---------------------------------------------------------
    # Problem dimensions (feel free to adjust)
    n = 200                      # length of the sequence
    # initial uniform distribution (sum == 1)
    f = np.full(n, 1.0 / n)

    # keep track of the best sequence seen
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # some predefined step sizes (α) to try each iteration
    alphas = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]

    # ---------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # current convolution and its maximum
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # ----- build the linear program ---------------------------------
        # constraints: (f * g) ≤ M   for each shift k = 0 .. 2n-2
        # This is a Toeplitz‑like matrix: row k has coefficient f[i] for g[k-i]
        rows = 2 * n - 1
        A_ub = np.zeros((rows, n), dtype=float)
        for j in range(n):
            # column j (variable g_j) contributes f[i] to rows i+j
            A_ub[j:j + n, j] = f  # vectorised fill

        b_ub = np.full(rows, M, dtype=float)

        # solve: maximize sum(g)  ⇒ minimize -sum(g)
        c = -np.ones(n, dtype=float)
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"time_limit": 1.0, "presolve": True},
        )

        if not res.success:
            # LP failed – perform a tiny random move to shake things up
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.001 * f[i]
            f[i] = max(0.0, f[i] - delta)
            f[j] += delta
            f = _project_simplex(f)
            continue

        g = res.x
        # Normalise g to a probability vector (sum == 1)
        s = g.sum()
        if s <= 0:
            # degenerate solution – random restart
            f = np.random.dirichlet(np.ones(n))
            continue
        g = g / s

        # ----- try several mixing factors and keep the best improvement ---
        improved = False
        for alpha in alphas:
            new_f = (1.0 - alpha) * f + alpha * g
            # numerical noise may produce slight negative entries
            new_f = _project_simplex(new_f)

            score = evaluate_sequence(new_f.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_f.copy()
                f = new_f   # continue from the improved point
                improved = True
                print(
                    f"[iter] time={time.time():.1f}, alpha={alpha:.2f}, "
                    f"score={best_score:.6f}"
                )
                break   # accept the first α that improves

        if not improved:
            # No α gave improvement – make a small random perturbation
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.005 * f[i]
            tentative = f.copy()
            tentative[i] = max(0.0, tentative[i] - delta)
            tentative[j] += delta
            tentative = _project_simplex(tentative)
            score = evaluate_sequence(tentative.tolist())
            if score < best_score:
                best_score = score
                best_seq = tentative.copy()
                f = tentative
                print(
                    f"[perturb] time={time.time():.1f}, score={best_score:.6f}"
                )

        # early exit if we beat the target
        if best_score <= 1.503:
            print(f"[done] target reached with score {best_score:.6f}")
            break

    # final sanity – make sure the result satisfies the basic constraints
    final_seq = best_seq.tolist()
    return final_seq
