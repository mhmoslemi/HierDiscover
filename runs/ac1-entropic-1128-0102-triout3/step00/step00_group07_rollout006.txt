def project_to_simplex(v):
    # https://github.com/...
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simple_projection(v):

    v = np.asarray(v)
    # in case sum already 1 and all nonneg: return
    # Use standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]

        return np.full_like(v, 1.0 / len(v))

    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""Hill‑climbing on the simplex with a sub‑gradient step.

The algorithm keeps a probability vector a (non‑negative, sum 1) and
iteratively reduces the maximum entry of the linear convolution a*a.
At each iteration the sub‑gradient of the ‖a*a‖_∞ functional is
computed, a small step is taken opposite to the gradient and the
resulting vector is projected back onto the probability simplex.
Occasional random mass‑transfer moves are mixed in to escape local
plateaus.  Several random restarts with different lengths are performed
until the time budget expires; the best sequence found (converted back
to a plain list) is returned.
"""

import numpy as np
import time
import random

def _project_to_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    if np.all(v <= 0):
        # Uniform distribution is a safe fallback
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1   # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_of(a):
    """Exact value of evaluate_sequence for a (no clipping)."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))

def _subgradient_step(a, step):
    """Take a projected sub‑gradient step aiming to reduce ‖a*a‖_∞."""
    n = a.size
    b = np.convolve(a, a)
    max_b = b.max()
    ks = np.where(np.isclose(b, max_b, atol=1e-12))[0]

    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        grad[i_idx] += 2.0 * a[j_idx]

    a_new = a - step * grad
    a_new = _project_to_simplex(a_new)
    return a_new

def _random_transfer_step(a, eps=0.05):
    """Move a small amount of probability mass from one index to another."""
    n = a.size
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] == 0:
        return a.copy()
    # epsilon is a relative amount of the source bucket
    delta = eps * a[i] * np.random.rand()
    delta = min(delta, a[i])
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # projection is not needed – sum stays constant and non‑negativity holds
    return a_new

    Returns

    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    end_time = time.time() + budget_s - 5.0   # a safety margin

    best_seq = None
    best_score = float('inf')

    # If a previously discovered sequence exists, use it as a warm start
    try:
        import __main__
        if hasattr(__main__, 'height_sequence_1'):
            init = np.array(__main__.height_sequence_1, dtype=float)

                best_score = _score_of(best_seq)
                print(f"Start from provided height_sequence_1, score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts)
    # ------------------------------------------------------------------
    while time.time() < end_time:
        # Choose a moderate length – longer sequences give a lower theoretical bound
        n = np.random.randint(50, 300)

        # Initialise with a random probability vector
        a = np.random.rand(n)
        a = _project_to_simplex(a)

        cur_score = _score_of(a)

        iteration = 0

                if new_score < best_score:

                    best_seq = a.copy()
                    print(f"New global best: {best_score:.6f} (n={n}, it={iteration})")

            remaining = end_time - time.time()

            if iteration % 400 == 0:

        # Small pause between restarts – optional
        # (the loop will automatically start a new random length if time permits)

    # ------------------------------------------------------------------

    if best_seq is None:

        best_seq = np.array([1.0])

    print(f"Finished search – best score = {best_score:.6f}")
    # Make sure the result respects the bounds imposed by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()

The code keeps a non‑negative vector `a` with unit L1‑norm (a probability
distribution).  The objective of `evaluate_sequence` for such a vector is

take a small step opposite to the gradient and project back onto the simplex.
Random “mass‑transfer” moves are mixed in to escape local plateaus, and a
simple simulated‑annealing acceptance rule occasionally keeps a worsening

time budget expires; the best sequence seen is returned.

The algorithm does **not** rely on the provided LP routine and therefore

    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # uniform fallback (should not happen)
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1          # zero‑based index of last positive
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of(a):
    """Exact value of evaluate_sequence for a non‑negative vector a."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))


def _subgradient_step(a, step):
    """
    One projected sub‑gradient step aiming to lower max(conv(a,a)).
    `step` is a positive scalar controlling the step length.
    """
    n = a.size
    b = np.convolve(a, a)
    max_b = b.max()
    ks = np.where(np.isclose(b, max_b, atol=1e-12))[0]

    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        # derivative of b_k w.r.t. a_i is 2 * a_{k-i}
        grad[i_idx] += 2.0 * a[j_idx]

    a_new = a - step * grad
    a_new = _project_to_simplex(a_new)
    return a_new


def _random_transfer_step(a, eps=0.03):
    """
    Randomly move a small amount of probability mass from one index to another.
    `eps` controls the maximal fraction of the source entry that can be moved.
    """
    n = a.size
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] == 0.0:
        return a.copy()
    delta = eps * a[i] * np.random.rand()
    delta = min(delta, a[i])
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return a_new

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` value.  The method performs several random restarts,
    each consisting of a few thousand projected sub‑gradient / random‑move
    steps, and keeps the best sequence found before the time budget expires.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget (seconds).  The function returns a bit earlier to avoid
        being killed.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    # give a tiny safety margin so the caller can still retrieve the result
    end_time = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Warm‑start from a previously supplied construction if it exists
    # ------------------------------------------------------------------
    try:
        if 'height_sequence_1' in globals():
            init = np.asarray(globals()['height_sequence_1'], dtype=float)
            if init.size > 0:
                init = _project_to_simplex(init)
                best_seq = init.copy()
                best_score = _score_of(best_seq)
                print(f"[init] Using height_sequence_1, score = {best_score:.6f}")
    except Exception:
        # any problem – just ignore and continue with random starts
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts)
    # ------------------------------------------------------------------
    while time.time() < end_time:
        # pick a length – longer vectors give a lower theoretical bound
        n = np.random.randint(50, 300)

        # start from a random probability vector
        a = np.random.rand(n)
        a = _project_to_simplex(a)

        cur_score = _score_of(a)

        iteration = 0
        # inner optimisation for the current restart
        while time.time() < end_time and iteration < 5000:
            iteration += 1
            # decreasing step size (1/sqrt(t) schedule)
            step = 0.8 / np.sqrt(iteration)

            # predominantly follow the sub‑gradient direction
            if np.random.rand() < 0.75:
                a_new = _subgradient_step(a, step)
            else:
                a_new = _random_transfer_step(a, eps=0.03)

            new_score = _score_of(a_new)

            # accept if we improve
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(
                        f"[update] New best = {best_score:.6f} (n={n}, it={iteration})"
                    )
                continue

            # occasional uphill moves – simple simulated annealing
            remaining = end_time - time.time()
            temperature = max(1e-4, remaining / budget_s)
            prob = np.exp(-(new_score - cur_score) / temperature)
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

            # re‑project every few hundred steps to keep numerical stability
            if iteration % 400 == 0:
                a = _project_to_simplex(a)

        # end of one restart – the outer loop will start another length
        # if time permits

    # ------------------------------------------------------------------
    # Return the best sequence found (fallback to a trivial one if needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([1.0])
        best_score = _score_of(best_seq)

    print(f"[final] Best score achieved = {best_score:.6f}")

    # Clip values to the limits enforced by evaluate_sequence (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
