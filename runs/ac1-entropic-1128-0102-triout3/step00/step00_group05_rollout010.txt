def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_sequence = None
    best_score = float('inf')
    while time.time() < deadline:
        # randomly choose n in some range
        n = np.random.randint(200, 2000)   # could be 2000
        # initialize a: random positive numbers
        a = np.random.rand(n)
        # normalize to sqrt(2n)
        target_sum = np.sqrt(2 * n)
        a = a * target_sum / np.sum(a)
        # gradient descent iterations, limited by time or fixed steps
        for it in range(200):
            b = np.convolve(a, a)
            max_idx = np.argmax(b)
            max_b = b[max_idx]
            # compute gradient
            g = np.zeros_like(a)
            # For each i, if k_max-i in range, add 2*a[k_max-i]
            # Actually g[i] = 2*a[k_max - i] if valid
            lo = max_idx - (n-1)
            hi = max_idx
            # i between max(0, lo) to min(n-1, hi)
            start = max(0, lo)
            end = min(n-1, hi)
            # vectorized
            i_arr = np.arange(start, end+1)
            j_arr = max_idx - i_arr
            g[i_arr] = 2 * a[j_arr]
            # step size schedule
            eta = 0.1 / (np.sqrt(it+1))   # reduce step size over iterations
            a_new = a - eta * g
            a_new = np.clip(a_new, 0, None)
            # renorm

            if sum_a == 0:
                a_new = np.full_like(a_new, target_sum/n)
            else:
                a_new = a_new * target_sum / sum_a
            a = a_new
        # after local descent compute score via evaluate_sequence (just call)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score

"""
Propose a step‑function (non‑negative sequence) that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm starts from a deterministic analytic shape that is known
to give a much lower bound than the naive constant sequence:

    a_i ∝ sqrt(i+1+δ) – sqrt(i+δ)      (δ ≥ 0)

For δ=0 this is the discrete derivative of √x; for δ>0 the large
coefficients at the left edge are reduced and the maximal convolution
value drops from 2 to ≈π/2 ≈ 1.57.  The sequence is scaled so that
∑a = √(2n) (the same normalisation used by the reference code).

From this analytic starting point a simple stochastic hill‑climbing
search is performed:

* a random index is perturbed by a small Gaussian step,
* the whole vector is re‑normalised to the required L¹‑norm,
* the new sequence is accepted only if the evaluation score improves,
* the step size slowly decays.

Multiple restarts with different lengths (n) and different δ values are
tried until the time budget expires.  The best sequence found is
returned.

The search is lightweight (O(n) work per iteration thanks to NumPy’s
fast convolution) and fits comfortably inside the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top level, no closures
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """The normalisation used in the reference code:  sum = sqrt(2*n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence of the form
        a_i = √(i+1+δ) – √(i+δ)
    scaled so that its L¹‑norm equals sqrt(2*n).
    """
    i = np.arange(n, dtype=float)
    # discrete derivative of sqrt shifted by delta
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    # rescale to the required L¹ norm
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrap the provided evaluate_sequence (expects a Python list)."""
    # evaluate_sequence is already available in the execution environment
    return evaluate_sequence(seq.tolist())


def _convolution_max(seq: np.ndarray) -> float:
    """Fast computation of max(conv(seq,seq)); used for cheap checks."""
    return np.max(np.convolve(seq, seq))


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a simple stochastic hill‑climbing optimisation on `seq`.
    Returns the best sequence encountered and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # initial step size – a fraction of the mean entry
    step_sigma = max(1e-8, seq.mean() * 0.2)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # propose a single‑coordinate Gaussian perturbation
        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma
        old_val = seq[idx]
        new_val = old_val + delta
        if new_val < 0.0:
            new_val = 0.0
        seq[idx] = new_val

        # renormalise to keep the L¹ norm fixed
        seq *= _target_sum(n) / seq.sum()

        # evaluate the candidate
        cand_score = _evaluate(seq)

        if cand_score < best_score:
            # improvement – keep it
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # reject – revert the change
            seq[idx] = old_val
            # (renorm is unnecessary because we restored the old value,
            #  but numerical drift can appear – re‑normalise anyway)
            seq *= _target_sum(n) / seq.sum()

        # slowly decay the step size
        step_sigma *= 0.9995

    return best_seq, best_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    Multiple random restarts with different lengths and δ‑parameters are
    performed; each restart is refined by hill‑climbing.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer loop
    deadline = start_time + max(0.0, budget_s - 5.0)

    # best result seen so far
    best_seq = None
    best_score = float("inf")

    # lengths we will try – a few hundred up to a few thousand elements
    possible_lengths = list(range(300, 2001, 100))

    # main outer loop: keep launching new restarts until time runs out
    while time.time() < deadline:
        n = int(np.random.choice(possible_lengths))
        # pick a random δ ≥ 0 (δ near 0 gives edge‑dominated max,
        # larger δ reduces the edge but slightly raises the interior)
        delta = np.random.rand() * 0.8  # uniform in [0,0.8)

        # initialise sequence from the analytic form
        seq = _analytic_sqrt_sequence(n, delta=delta)

        # quick sanity check – the analytic start is already better than 2
        init_score = _evaluate(seq)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d} delta={delta:.3f} score={init_score:.6f}")

        # run a limited hill‑climb from this start
        # we give each restart a modest CPU budget (≈0.2 s)
        inner_deadline = time.time() + 0.2
        seq_opt, score_opt = _hill_climb(seq,
                                         max_iter=2000,
                                         deadline=inner_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d} delta={delta:.3f} score={score_opt:.6f}")

        # optional early exit if we already beat the target
        if best_score < 1.55:   # a bit above the target 1.503, still useful
            # continue a little longer to see if we can push even lower
            pass

    # If nothing was found (should never happen), fall back to a trivial sequence.
    if best_seq is None:
        best_seq = np.ones(10) * (_target_sum(10) / 10.0)

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
