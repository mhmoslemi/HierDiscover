def project_simplex(v):
    """Project vector v onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

"""
Optimization of a step‑function heights sequence.
The objective          = 2·n·max(conv(p,p))/ (sum(p))^2
Since sum(p) cancels when we work with a probability distribution,
the problem becomes: minimise the peak of the self‑convolution
of a probability vector p (p_i ≥ 0, Σp_i = 1).  This is a convex
program:
    minimise t
    s.t.   p_i ≥ 0, Σp_i = 1,
           (p * p)_k = Σ_i p_i p_{k‑i} ≤ t   for every k.
For a moderate length (n≈200) the quadratic constraints are cheap.
We first try a direct convex formulation with CVXPY (SCS solver);
if CVXPY is unavailable or the solver does not finish in time we
fallback to a simple projected sub‑gradient method.
The best point found is returned as the list of heights.
"""

import time
import numpy as np

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the rho
    rho = ind[u - cssv / ind > 0]
    if rho.size == 0:
        # all entries will be zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def build_quadratic_matrices(n):

Optimization of a step‑function heights sequence.

The evaluation function can be rewritten as  

    evaluate(sequence) = 2·n·max_k (p * p)_k ,

where  n = len(sequence)  and  p_i = a_i / Σ a_i .  Hence the
objective depends only on the *probability* vector p (p_i ≥ 0,
Σ p_i = 1) and we must minimise the peak of its self‑convolution.
This is a convex problem:

    minimise   t
    subject to p_i ≥ 0,   Σ p_i = 1,
               Σ_i p_i p_{k‑i} ≤ t   for every k = 0,…,2n‑2 .

For a moderate length (n≈200) the quadratic constraints are cheap.
We first try a direct convex formulation with **CVXPY** (SCS solver);
if CVXPY is unavailable or the solver does not finish in time we fall

# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v):
    """Return the Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted in descending order
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the largest index ρ satisfying  u_ρ  >  (cumulative_sum_ρ - 1) / ρ
    rho = ind[u - cssv / ind > 0]
