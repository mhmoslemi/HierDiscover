def simplex_proj(v):
    """Project vector v onto the probability simplex (nonnegative and sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all <=0, then return uniform?
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = kwargs.get('n', 1200) # maybe parameter.

    # Choose random start: Dirichlet
    a = np.random.dirichlet(np.ones(n)).astype(np.float64)  # sum=1

    best_a = a.copy()
    best_score = compute_score(a)

    # Additional start from height_sequence_1 if available
    if 'height_sequence_1' in globals():
        tmp = np.array(height_sequence_1, dtype=float)
        nm = len(tmp)
        # Normalize to sum 1
        tmp = tmp / tmp.sum()
        if compute_score(tmp) < best_score:
            best_a = tmp.copy()
            best_score = compute_score(tmp)
            a = tmp.copy()
    
    beta = 5.0
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # compute conv and weights
        conv = np.convolve(a, a)
        # w weights
        max_conv = conv.max()
        # compute softmax weights
        w = np.exp(beta * (conv - max_conv))
        w = w / w.sum()
        # compute gradient (correlation)
        grad = correlation(w, a)  # length n
        # try line search steps
        improved = False
        for step in (5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = simplex_proj(a - step * grad)
            cand_score = compute_score(cand)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            # occasional perturbation
            if no_improve % 50 == 0:
                # add random noise & project
                noise_scale = 0.05 / (no_improve // 50)
                cand = simplex_proj(a + noise_scale * np.random.randn(n))
                cand_score = compute_score(cand)
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    a = cand
                # maybe shrink beta gradually to explore
                beta = max(5.0, beta * 0.9)
            else:
                # random perturb tiny
                i, j = np.random.randint(n, size=2)
                delta = min(a[i], 0.01) * np.random.rand()
                a[i] -= delta
                a[j] += delta
                a = simplex_proj(a)  # ensure nonneg sum 1
        iteration += 1
        # periodic print
        if iteration % 200 == 0:
            print(f"Iter {iteration}, best_score={best_score:.6f}, beta={beta:.2f}, n={n}")

        # if best_score already < target 1.503, break early?
        if best_score < 1.503:
            print("Target achieved, breaking early")
            break

    # Return as list
    return best_a.tolist()

    deadline = time.time() + budget_s - 5

            a = seq[:n]
            a = a / a.sum()
        else:
            # pad with uniform

            a = np.concatenate([seq, padding])

"""
propose_candidate:  Stochastic hill‑climbing search for a step‑function
with low self‑convolution peak.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is homogeneous of degree 0 we may (and do) normalise the
sequence to sum = 1.  For a uniform sequence the score equals 2, independent
of the length.  To obtain a value below 2 we must make the maximum entry of
the convolution smaller than 1/n.  This can be achieved by moving probability
mass away from the centre of the sequence – the centre receives the most
contributions (many pairs sum to the same index) – while keeping the total
mass constant.

The algorithm starts from a uniform (or optionally a provided) sequence,
applies random multiplicative perturbations, projects back onto the probability
simplex and accepts the change only if the score improves.  When the search
gets stuck a fresh random Dirichlet start is performed.  Over the allowed
budget (≈ 1000 s) the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        v_i >= 0  and  sum(v) = 1
    Uses the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of(seq: np.ndarray) -> float:
    """
    Compute the evaluation score for a normalised sequence (sum == 1).
    Directly implements the expression used in evaluate_sequence.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 600,
                     **kwargs) -> list[float]:
    """
    Stochastic optimisation for the step‑function coefficients.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before this time).
    n : int, optional
        Length of the sequence to optimise.  The default (600) gives a good
        trade‑off between speed and quality.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) discovered within the time
        budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a few seconds margin

    # ------------------------------------------------------------------
    # Initialise a starting point
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # try to reuse the known construction; cut / pad to the desired length
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.size >= n:
            a = base[:n].copy()
        else:
            pad = np.zeros(n - base.size, dtype=np.float64)
            a = np.concatenate([base, pad])
        # Normalise to a probability vector
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
    else:
        # pure uniform start
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Keep the best solution seen so far
    best_a = a.copy()
    best_score = _score_of(best_a)
    cur_score = best_score

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    step = 0.2                     # multiplicative perturbation magnitude
    no_improve = 0                 # counter for stagnation
    iter_cnt = 0
    print(f"Starting optimisation: length={n}, initial score={best_score:.6f}")

    while time.time() < deadline:
        # ----------------------------------------------------------------
        # Propose a random multiplicative perturbation (preserving the sum)
        # ----------------------------------------------------------------
        # random factors in [1-step, 1+step]
        factors = 1.0 + step * np.random.uniform(-1.0, 1.0, size=n)
        cand = a * factors
        cand = np.clip(cand, 0.0, None)          # enforce non‑negativity
        s = cand.sum()
        if s == 0.0:
            # extremely unlikely, just skip this iteration
            continue
        cand = cand / s                           # project back onto the simplex

        # ----------------------------------------------------------------
        # Evaluate the candidate
        # ----------------------------------------------------------------
        cand_score = _score_of(cand)

        # Accept if better (strict improvement)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"Iter {iter_cnt:6d} – new best score = {best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------------
        # Occasionally shrink the perturbation step and restart if stuck
        # ----------------------------------------------------------------
        if iter_cnt % 2000 == 0 and iter_cnt > 0:
            step = max(0.02, step * 0.9)   # become more conservative over time

        if no_improve > 5000:
            # re‑initialise from a fresh Dirichlet sample
            a = np.random.dirichlet(np.ones(n))
            cur_score = _score_of(a)
            no_improve = 0
            # occasional print
            print(f"Iter {iter_cnt:6d} – restart, score = {cur_score:.6f}")

        iter_cnt += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain python list)
    # ------------------------------------------------------------------
    print(f"Finished after {iter_cnt} iterations. Best score = {best_score:.6f}")
    # Convert to plain list; evaluate_sequence will clamp negatives and
    # enforce the [0,1000] range automatically.
    return best_a.tolist()
