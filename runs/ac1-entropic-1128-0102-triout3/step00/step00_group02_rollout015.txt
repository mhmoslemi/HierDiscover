"""
Propose a good step‑function for the evaluation metric.

The algorithm searches over a simple 2‑parameter family of non‑negative
sequences that are known to give low autocorrelation peaks:

    a_i(p) = ((i+½)·(n‑½‑i))^(‑p) ,   i = 0,…,n‑1 ,   0 ≤ p < ½

For p = 0 the sequence is uniform (objective = 2).  As p increases the
mass moves towards the ends of the interval; the resulting convolution
becomes flatter and the objective drops below 2.  The asymptotic value
of the objective depends only on the exponent p, not on n, so we can
search a few reasonable lengths (≈ 500–2400) and a grid of p‑values.
The best candidate from this grid is then refined with a tiny random
hill‑climb to escape local plateaus.

The whole search is confined to the supplied time budget (default 1000 s)
and respects the security checks of ``evaluate_sequence`` (non‑negative,
no infinities, etc.).  The returned list is the best sequence found.
"""

import numpy as np
import time
import random

# ----------------------------------------------------------------------
# helper: generate the beta‑like family described above
def beta_sequence(n: int, p: float) -> list[float]:
    """
    Return a sequence of length n with shape
        a_i = ((i + 0.5) * (n - 0.5 - i)) ** (-p)
    for 0 ≤ p < 0.5.  For p == 0 the sequence is uniform (all ones).
    """
    if p == 0.0:
        return np.ones(n, dtype=float).tolist()
    i = np.arange(n, dtype=float)
    # denominator never reaches zero because of the 0.5 offset
    denom = (i + 0.5) * (n - 0.5 - i)
    seq = np.power(denom, -p)
    return seq.tolist()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimizes ``evaluate_sequence``.
    The function respects the 1000 s time limit and returns the best
    sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    # time bookkeeping
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. try the previously known construction (if it exists)
    try:
        candidate = list(height_sequence_1)          # type: ignore
        sc = evaluate_sequence(candidate)
        if sc < best_score:
            best_seq, best_score = candidate, sc
            print(f"[init] SOTA sequence score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. systematic grid search over length and exponent p
    #    (the objective is almost independent of n for large n,
    #     but a few different n give us a safety margin)
    lengths = [500, 800, 1200, 1600, 2000, 2400]
    p_grid = np.linspace(0.0, 0.49, 25)   # include the uniform case p = 0

    for n in lengths:
        if time.time() > deadline:
            break
        for p in p_grid:
            if time.time() > deadline:
                break
            seq = beta_sequence(n, float(p))
            # evaluate – any exception means the sequence was rejected
            try:
                sc = evaluate_sequence(seq)
            except Exception:
                continue
            if sc < best_score:
                best_score, best_seq = sc, seq
                print(f"[grid] n={n:4d}, p={p: .3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. local refinement: tiny random perturbations (hill‑climbing /
    #    simulated‑annealing style).  The scaling of the whole vector
    #    does not affect the objective, therefore we only alter a few
    #    entries.
    if best_seq is None:
        # fallback – produce a uniform sequence of moderate length
        n = 1200
        best_seq = np.ones(n, dtype=float).tolist()
        best_score = evaluate_sequence(best_seq)

    cur_seq = best_seq[:]          # current working copy
    cur_score = best_score
    n_cur = len(cur_seq)

    # parameters for the random walk
    perturb_scale = 0.02          # relative standard deviation of a change
    accept_prob = 0.001           # probability to accept a worsening move

    while time.time() < deadline:
        # pick a random index and propose a relative perturbation
        i = np.random.randint(n_cur)
        delta = (np.random.randn() * perturb_scale) * max(1.0, cur_seq[i])
        new_val = max(0.0, cur_seq[i] + delta)   # stay non‑negative
        new_seq = cur_seq.copy()
        new_seq[i] = new_val

        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        if new_score < cur_score:
            # improvement – accept and remember globally if it is the best
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                print(f"[refine] improved to {new_score:.6f}")
        else:
            # occasional uphill moves to avoid getting stuck
            if np.random.rand() < accept_prob:
                cur_seq, cur_score = new_seq, new_score

    print(f"[final] best score = {best_score:.6f} (length={len(best_seq)})")
    return best_seq
