def gradient_step(a, beta, lr):
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    # compute convolution/r array
    A = np.fft.rfft(a, m)
    conv = np.fft.irfft(A * A, m)
    r = conv[:2 * n - 1]
    max_r = r.max()
    # weights
    w = np.exp(beta * (r - max_r))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute gradient via conv with w and reversed
    W = np.fft.rfft(w, m)
    grad1 = np.fft.irfft(A * W, m)
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, m)
    grad2 = np.fft.irfft(A * W_rev, m)
    # extract relevant positions
    g = grad1[n-1:n-1+n] + grad2[n-1:n-1+n]
    # gradient descent step
    a_new = a - lr * g
    # projection onto simplex
    a_new = np.maximum(a_new, 0.0)
    sum_a = a_new.sum()
    if sum_a == 0:
        # fallback to random
        a_new = np.random.random(n)
        a_new /= a_new.sum()
    else:
        a_new /= sum_a
    return a_new, r.max()

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of `seq` with itself, returned as a 1‑D array of length
    2*len(seq)-1.  The computation is performed by FFT using the supplied FFT
    length (a power of two).
    """
    a_hat = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(a_hat * a_hat, fft_len)
    return conv[: 2 * len(seq) - 1]


def _max_conv(seq: np.ndarray, fft_len: int) -> float:
    """
    Return only the maximum value of the self‑convolution of `seq`.
    """
    return _conv_full(seq, fft_len).max()


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Simple implementation that works sufficiently well for the present
    optimisation: clip negatives and renormalise.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform vector
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _gradient_step(
    a: np.ndarray,
    fft_len: int,
    beta: float,
    lr: float,
) -> (np.ndarray, float):
    """
    One (smoothed) gradient‑descent step on the objective
    `max(conv(a, a))`.  The gradient of the smooth approximation
    `logsumexp(beta * r)` (r = self‑convolution) is computed via FFT.

    Returns
    -------
    a_new : np.ndarray
        Updated coefficient vector (projected onto the simplex).
    max_r : float
        The new maximal convolution value (used for the score).
    """
    n = a.size

    # ----- convolution and smooth weights -----
    a_hat = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(a_hat * a_hat, fft_len)
    r = conv[: 2 * n - 1]                 # full linear convolution
    max_r = float(r.max())

    # Soft‑max weights concentrate on the largest entries of r.
    w = np.exp(beta * (r - max_r))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- gradient via two convolutions (w and its reverse) -----
    w_hat = np.fft.rfft(w, fft_len)
    grad1 = np.fft.irfft(a_hat * w_hat, fft_len)

    w_rev = w[::-1]                       # w_{-s}
    w_rev_hat = np.fft.rfft(w_rev, fft_len)
    grad2 = np.fft.irfft(a_hat * w_rev_hat, fft_len)

    # combine the two parts; the relevant slice is centred at index n‑1
    g = grad1[n - 1 : n - 1 + n] + grad2[n - 1 : n - 1 + n]

    # ----- descent step and projection -----
    a_new = a - lr * g
    a_new = _project_to_simplex(a_new)

    # recompute the maximal convolution for the new point (cheap, reuse fft_len)
    max_r_new = _max_conv(a_new, fft_len)

    return a_new, max_r_new


def _random_simplex(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1) sample)."""
    v = rng.exponential(scale=1.0, size=n)
    return v / v.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a good coefficient vector inside the 1000 s time budget.
    The routine mixes a simple simulated‑annealing move set with occasional
    gradient‑descent refinements.  It returns the best sequence found as a
    plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # If a previous construction `height_sequence_1` is available we reuse its
    # length (it is usually around 3000).  Otherwise we use a medium size that
    # keeps the FFT cheap.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        n = 2000
    if n < 500:
        n = 500

    # pre‑computed FFT length (next power of two for 2n‑1)
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Prepare a few different starting points.
    # --------------------------------------------------------------
    starts = []

    # 1. uniform distribution (baseline)
    starts.append(np.full(n, 1.0 / n))

    # 2. random Dirichlet point
    starts.append(_random_simplex(rng, n))

    # 3. monotone increasing shape (roughly linear)
    lin = np.linspace(0.0, 1.0, n)
    starts.append(lin / lin.sum())

    # 4. use the previous SOTA construction if it exists
    try:
        hs = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if hs.size != n:
            if hs.size > n:
                hs = hs[:n]
            else:
                pad = np.zeros(n)
                pad[: hs.size] = hs
                hs = pad
        hs = _project_to_simplex(hs)
        starts.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Global trackers for the best solution seen so far.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Helper for fast score from maximal convolution value
    def _score_from_maxconv(max_conv):
        return 2.0 * n * max_conv  # sum(a) == 1 after projection

    # ------------------------------------------------------------------
    # Main optimisation loop.  For each start we perform a long SA walk,
    # occasionally applying a gradient step to accelerate convergence.
    # ------------------------------------------------------------------
    for start_idx, a in enumerate(starts):
        # ensure the start lies on the simplex
        a = _project_to_simplex(a.copy())

        # initial score
        cur_max = _max_conv(a, fft_len)
        cur_score = _score_from_maxconv(cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Init {start_idx}] score = {best_score:.6f}")

        it = 0
        # temperature schedule is linear in remaining time
        total_time = deadline - time.time()
        while time.time() < deadline:
            remaining = deadline - time.time()
            if remaining <= 0:
                break

            # Linear cooling: from 0.2 → 0 as time runs out
            temp = 0.2 * (remaining / total_time)

            # ------------------------------------------------------------------
            # Decide between a cheap random move and an expensive gradient step.
            # Gradient steps are performed with decreasing probability to spend
            # more time on simple moves when the temperature is high.
            # ------------------------------------------------------------------
            if rng.random() < 0.02:  # gradient step approximately every 50th iteration
                # smoothing parameter for the soft‑max.  Larger values give a
                # gradient that points more directly at the current maximum.
                beta = 30.0
                # learning rate shrinks with temperature
                lr = max(1e-4, 5.0 * temp)
                a_new, new_max = _gradient_step(a, fft_len, beta, lr)
                new_score = _score_from_maxconv(new_max)
                accept = new_score < cur_score or rng.random() < math.exp(
                    -(new_score - cur_score) / max(temp, 1e-12)
                )
                if accept:
                    a = a_new
                    cur_score, cur_max = new_score, new_max
            else:
                # ----------- cheap Monte‑Carlo move -------------
                # pick two distinct indices, move a small amount of mass from i to j
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                if a[i] == 0.0:
                    continue
                # the transferred mass is a random fraction of a[i]
                frac = rng.random() * 0.1 * temp   # scale with temperature
                delta = frac * a[i]
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_to_simplex(a_candidate)

                cand_max = _max_conv(a_candidate, fft_len)
                cand_score = _score_from_maxconv(cand_max)

                # Metropolis acceptance
                if cand_score < cur_score or rng.random() < math.exp(
                    -(cand_score - cur_score) / max(temp, 1e-12)
                ):
                    a = a_candidate
                    cur_score, cur_max = cand_score, cand_max

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Iter {it:07d}] new best = {best_score:.6f}")

            it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list of floats).
    # ------------------------------------------------------------------
    if best_seq is None:
        # Extremely unlikely – fallback to a uniform vector.
        best_seq = np.full(n, 1.0 / n)

    # Clip tiny negative rounding errors (should not happen) and enforce bounds.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Final normalisation (security check in evaluate_sequence will also do this)
    if best_seq.sum() == 0.0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / best_seq.sum()

    # optional sanity print using the provided evaluator (costly, but once)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[Final] evaluated score = {final_score:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
