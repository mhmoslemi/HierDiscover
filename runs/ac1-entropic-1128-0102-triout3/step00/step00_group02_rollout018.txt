        # renormalize to unit sum
        s = a_candidate.sum()

            a_candidate = np.abs(np.random.randn(n))
            s = a_candidate.sum()
        a_candidate = a_candidate / s
        
        score = evaluate_sequence(a_candidate.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_candidate.copy()
            print(f"Improved score to {best_score:.6f} at time {time.time():.2f}")
            # optionally increase step
            step *= 1.05
        else:
            # reduce step
            step *= 0.95
        
        a = a_candidate
        
        # occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.01:
            # add small noise
            noise = np.random.randn(n) * 0.01
            a = np.maximum(0, a + noise)
            a = a / a.sum()
    
    return best_seq.tolist()

def refine(seq, deadline):
    # projected gradient descent as earlier
    n = len(seq)
    step = 0.1
    beta = 10.0
    while time.time() < deadline:
        b = np.convolve(seq, seq)
        max_b = b.max()

"""
Propose a sequence of non‑negative heights that gives a low value for
evaluate_sequence.  The construction is based on a *Sidon* (B₂) set:
choose a modest number k ≈ √n of “large’’ coefficients placed at
indices whose pairwise sums are all distinct.  All remaining positions
receive a small common value.  With the sums of the large and small
parts balanced (2·A² ≈ (n‑k)·B²) the maximal entry of the self‑convolution
is roughly the average, yielding a score well below the target 1.5.

The algorithm:
  1. Fix the length n (use the length of the known construction if
     available, otherwise 2401).
  2. Build a Sidon set S⊂{0,…,n‑1} of size k≈√n by a greedy random
     procedure that forbids any repeated sum i+j (including i=j).
  3. Set the total sum to 1 (the objective is scale‑invariant).  Solve
     2·A² = (n‑k)·B² together with k·A + (n‑k)·B = 1, which gives

          A = 1 / (k + √(2·(n‑k))) ,
          B = (1 – k·A) / (n‑k) .

  4. Fill the sequence with B and replace the indices in S by A.
  5. (Optional) a few projected‑gradient steps are applied to polish
     the result.
  6. Keep the best sequence seen before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a Sidon (B2) set
def build_sidon_set(n: int, target_size: int) -> list[int]:
    """
    Return a list of indices forming a Sidon set (all i+j are distinct).
    The algorithm is greedy, trying indices in random order.
    """
    used = np.zeros(2 * n - 1, dtype=bool)   # used[sum] == True if sum already taken
    chosen = []
    order = np.random.permutation(n)
    for i in order:
        # check whether any sum i+j already exists
        conflict = False
        for j in chosen:
            if used[i + j]:
                conflict = True
                break
        if conflict or used[2 * i]:   # also forbid a duplicate square sum
            continue
        # accept i
        for j in chosen:
            used[i + j] = True
        used[2 * i] = True
        chosen.append(i)
        if len(chosen) >= target_size:
            break
    return chosen


# ----------------------------------------------------------------------
# Optional refinement: projected gradient descent on a smoothed max
def refine_sequence(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform a few gradient steps that try to lower the maximal
    convolution entry while keeping the vector non‑negative and
    normalised (sum = 1).  Stops when time runs out.
    """
    n = len(seq)
    step = 0.05
    beta = 12.0          # sharpness of the soft‑max
    while time.time() < deadline:
        b = np.convolve(seq, seq)
        max_b = b.max()
        # soft‑max weights concentrated on the large entries of b
        w = np.exp(beta * (b - max_b))
        w /= w.sum()
        # gradient of the smoothed max w.r.t. seq
        g = np.correlate(w, seq, mode="valid")          # length n
        # move opposite to the gradient, but keep the sum unchanged
        direction = -(g - g.mean())
        new_seq = seq + step * direction
        new_seq = np.clip(new_seq, 0.0, None)
        s = new_seq.sum()
        if s == 0.0:
            break
        new_seq /= s
        # accept only if the score improves
        old_score = evaluate_sequence(seq.tolist())
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < old_score:
            seq = new_seq
            step = min(step * 1.1, 0.5)   # grow a little when successful
        else:
            step *= 0.7
            if step < 1e-5:
                break
    return seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 950, **kwargs) -> list[float]:
    """
    Construct a step‑function sequence using a Sidon set of large
    coefficients.  The method is deterministic up to the provided
    seed and finishes well before the 1000‑second limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little margin

    # Length of the problem – use the known construction when available
    n = len(globals()["height_sequence_1"]) if "height_sequence_1" in globals() else 2401
    print(f"Target length = {n}")

    best_seq = None
    best_score = float("inf")

    # Try a few random Sidon sets (different seeds) and keep the best
    attempts = 0
    while time.time() < deadline and attempts < 20:
        attempts += 1

        # Desired size of the Sidon set – roughly sqrt(n)
        target_k = int(np.sqrt(n))
        # Build the set (may be slightly smaller than target_k)
        sidon = build_sidon_set(n, target_k)
        k = len(sidon)
        if k < 2:
            continue
        m = n - k

        # Balanced amplitudes (total sum = 1)
        denom = k + np.sqrt(2.0 * m)
        A = 1.0 / denom
        B = (1.0 - k * A) / m

        # safety check
        if B < 0.0:
            continue

        seq = np.full(n, B, dtype=float)
        seq[sidon] = A

        # Small polishing step (optional but cheap)
        seq = refine_sequence(seq, deadline)

        score = evaluate_sequence(seq.tolist())
        print(f"Attempt {attempts:02d}: k={k}, A={A:.6f}, B={B:.6f}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  ** New best score {best_score:.6f}")

    # Fallback – uniform sequence (should never be needed)
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback to uniform sequence,", best_score)

    print("Final best score:", best_score)
    return best_seq.tolist()
