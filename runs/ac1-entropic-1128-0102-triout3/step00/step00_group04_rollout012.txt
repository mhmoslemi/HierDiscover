def solve_convolution_lp(f, M):
    n = len(f)
    # Build A_ub as sparse
    # For each k: we need row vector of length n: row[j] = f[k - j] if 0 <= k - j < n else 0.
    # We'll build lists of (data,row_ind, col_ind)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i ranges max(0,k-(n-1)) to min(n-1,k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # For each i, j = k - i
        # f[i] is coefficient for variable g[j]? Wait earlier row[j] = f[i] where i = k - j?
        # Let's derive: conv(f,g)[k] = sum_i f[i] * g[k - i] (with g out of range zero). So coefficient for g[j] is f[k - j] if 0 <= k - j < n.
        # So for each i index of f, we add f[i] * g[k-i]: So variable index j = k - i. So row[j] = f[i].
        # So loop over i in range.
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # Now add nonneg constraints: -I * g <= 0
    for i in range(n):
        rows.append(2*n-1 + i)
        cols.append(i)
        data.append(-1.0)

    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1 + n, n))
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])

    c = -np.ones(n)  # maximize sum => minimize negative sum
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    except Exception as e:
        return None
    if res.success:
        return res.x
    else:
        return None

import numpy as np
import time
from scipy.optimize import linprog

def solve_convolution_lp(f, M):
    n = len(f)
    # Build linear constraints for g: f * g <= M
    # Use dense A_ub: shape (2*n-1 + n, n)
    rows = 2*n - 1 + n
    A = np.zeros((rows, n), dtype=float)
    # convolution constraints
    for k in range(2*n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, j] = f[i]
    # nonnegativity constraints: -g_i <= 0
    A[2*n-1:, :] = -np.eye(n)
    b = np.concatenate([np.full(2*n-1, M), np.zeros(n)])

    c = -np.ones(n)   # maximize sum(g)
    # Use high performance method
    options = {'disp': False, 'time_limit': 2.0}
    try:
        res = linprog(c, A_ub=A, b_ub=b, method='highs', options=options)
    except Exception:
        return None
    if res.success:
        return res.x
    return None

def propose_candidate(seed=12345, budget_s=990):
    """
    Iteratively improve a non‑negative step function using a linear‑programming
    direction.  Each iteration solves a LP that maximises the sum of a new
    candidate g while keeping the convolution bound unchanged, then takes a
    line‑search over a few step sizes.  Random perturbations are added when the
    LP fails to keep the search alive.  The best sequence seen within the time
    budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # start from a random normalised sequence; also try the global height_sequence_1 if present
    if 'height_sequence_1' in globals():
        start = np.array(height_sequence_1, dtype=float)
        n_start = len(start)
        if np.sum(start) > 0:
            start = start * (np.sqrt(2*n_start) / np.sum(start))
        else:
            start = np.random.rand(n_start)
            start = start * (np.sqrt(2*n_start) / np.sum(start))
        best_seq = start.copy()
        best_score = evaluate_sequence(best_seq.tolist())
    else:
        best_seq = None
        best_score = np.inf

    # auxiliary function to initialise a random normalised vector of length n
    def random_normalised(n):
        v = np.random.rand(n) + 1e-6
        v = v * (np.sqrt(2*n) / np.sum(v))
        return v

    # initial length (will be changed on restart)
    n = np.random.randint(300, 800)
    seq = random_normalised(n)

    # if we have a pre‑existing best use it
    if best_seq is not None:
        # compare scores and keep the better one as global best
        cur_score = evaluate_sequence(seq.tolist())
        if cur_score < best_score:
            best_seq = seq.copy()
            best_score = cur_score
    else:
        best_seq = seq.copy()
        best_score = evaluate_sequence(seq.tolist())

    last_improve = time.time()
    # main optimisation loop
    while time.time() < deadline:
        # normalise current sequence (kept normalised most of the time)
        total = np.sum(seq)
        if total == 0:
            seq = random_normalised(n)
            continue
        seq = seq * (np.sqrt(2*n) / total)

        # compute bound M = max(f * f)
        conv_ff = np.convolve(seq, seq)
        M = float(np.max(conv_ff))

        # solve LP: maximize sum(g) subject to f * g <= M, g >= 0
        g = solve_convolution_lp(seq, M)

        if g is not None:
            # scale g to the normalisation used for f
            g_sum = np.sum(g)
            if g_sum == 0:
                # fallback to perturbation
                idx = np.random.randint(n)
                seq[idx] = max(0.0, seq[idx] + np.random.randn()*0.01)
                continue
            g = g * (np.sqrt(2*n) / g_sum)

            # line‑search over a few step sizes
            improved = False
            for t in (0.01, 0.05, 0.1, 0.2, 0.5):
                cand = (1.0 - t) * seq + t * g
                cand = np.maximum(cand, 0.0)
                cand = cand * (np.sqrt(2*n) / np.sum(cand))
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    seq = cand.copy()
                    improved = True
                    last_improve = time.time()
                    print(f"[{time.time():.2f}] new best {best_score:.6f} (t={t:.2f}, n={n})")
                    break
            if not improved:
                # small random perturbation to escape potential plateaus
                idx = np.random.randint(n)
                seq[idx] = max(0.0, seq[idx] + np.random.randn()*0.02)
                continue
        else:
            # LP failed – inject a tiny random move
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn()*0.05)

        # occasional restart if we have been idle for a while
        if time.time() - last_improve > 30.0:
            n = np.random.randint(300, 800)
            seq = random_normalised(n)
            last_improve = time.time()
            print(f"[{time.time():.2f}] restart with new length {n}")

    # return the best sequence seen (as a plain Python list)
    return best_seq.tolist()

import numpy as np
import time
from scipy.optimize import linprog

def solve_convolution_lp(f, M):
    n = len(f)
    # Build constraint matrix for g: f * g <= M
    # dense matrix version for simplicity (n <= 800)
    rows = 2 * n - 1 + n
    A = np.zeros((rows, n))
    # convolution constraints
    for k in range(2 * n - 1):
        # i runs where j=k-i inside [0,n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]
    # nonnegativity constraints: -g_i <= 0
    A[2 * n - 1:, :] = -np.eye(n)
    b = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    # maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, method='highs')
    except Exception:
        return None
    if res.success:
        return res.x
    return None

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f, M):
    """
    Solve the linear programme

        maximise   sum(g)
        subject to conv(f, g) <= M   (for all shifts)
                    g >= 0

    where ``f`` is a non‑negative vector of length n and ``M`` is a scalar.
    The programme is linear because for fixed ``f`` the convolution constraints
    are linear in the unknown vector ``g``.
    """
    n = len(f)

    # number of inequality rows: (2n‑1) convolution rows + n non‑negativity rows
    rows = 2 * n - 1 + n
    A = np.zeros((rows, n), dtype=float)

    # ----- convolution constraints  (f * g)[k] <= M  -----
    # (f * g)[k] = sum_i f[i] * g[k-i]   with g out of range = 0
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs over indices of f that actually contribute to this shift
        for i in range(i_min, i_max + 1):
            j = k - i                # the index of g that multiplies f[i]
            A[k, j] = f[i]

    # ----- non‑negativity:   -g_i <= 0  -----
    A[2 * n - 1 :, :] = -np.eye(n)

    b = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n)

    # Use the fast HiGHS driver (available in recent SciPy versions)
    try:
        res = linprog(c, A_ub=A, b_ub=b, method="highs")
    except Exception:                # safety net for unexpected solver errors
        return None

    if res.success:
        return res.x
    return None

         the full replacement t=1) and keeps the best improvement,
      5. injects a tiny random perturbation when the LP fails,
