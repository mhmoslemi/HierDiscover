    Generate a step function based on discrete arcsine distribution.

    improve the score.

    n = kwargs.get('length', 2401)  # default length
    # Build raw arcsine weights
    raw = [1.0 / math.sqrt((i + 0.5) * (n - 0.5 - i)) for i in range(n)]
    s = sum(raw)
    seq = [x / s for x in raw]   # L1 = 1
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    print(f"Init score (arcsine) = {best_score:.6f}")
    deadline = time.time() + budget_s - 0.5  # keep small margin
    random.seed(seed)
    # Small greedy stochastic refinement
    while time.time() < deadline:
        # pick two indices, move epsilon mass
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:

        eps = seq[i] * 0.02  # move 2% of i's mass
        if eps <= 0:

        new_seq = seq[:]
        new_seq[i] -= eps

        # keep non‑negative (should be)

            best_seq = new_seq[:]
