"""
candidate_proposer.py

Search for a non‑negative step‑function (list of heights) that minimizes

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)

The algorithm proceeds in three phases

1.  Deterministic exploration of a few simple parametric families
    (geometric, exponential, Gaussian and power‑law).  The best result
    from these families is kept as a starting point.

2.  A stochastic Dirichlet / random‑gamma sampling stage that draws many
    points on the probability simplex.  Every sample is evaluated and the
    current best is updated.

3.  Light‑weight hill‑climbing: occasional small moves that transfer a
    tiny amount of mass from one index to another.  If the move improves
    the score it is accepted, otherwise it is discarded.

All sequences are normalised to have sum = 1 (the objective is
scale‑invariant, therefore this normalisation does not affect the result)
and are projected back onto the simplex after each perturbation.

The loop runs until the allotted time (budget_s) is exhausted and finally
returns the best sequence found.
"""

import time
import math
import numpy as np

# The judge injects ``evaluate_sequence`` into the global namespace.
# We only call it – never re‑implement it.


def _project_onto_simplex(v):
    """
    Projects a vector onto the probability simplex {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm from Duchi et al. (ICML 2008).
    """
    n = len(v)
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _eval(seq):
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq)


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) gives a low
    value of ``evaluate_sequence``.  The routine respects the supplied time
    budget; as soon as the deadline is reached it returns the best
    sequence discovered so far.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the environment
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) pick a length – a moderate size gives a good trade‑off between
    #    expressiveness and speed.  The theoretical bound is independent of
    #    the exact n, so 500 is enough for a quick search.
    # ------------------------------------------------------------------
    n = 500

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2) systematic sweep over a few classic families
    # ------------------------------------------------------------------

    # ---- geometric (a_i ∝ r^i) ---------------------------------------
    for c in np.linspace(0.5, 2.0, 16):          # c = (1‑r) * n
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        raw = [(1.0 - r) * (r ** i) for i in range(n)]
        s = sum(raw)
        if s == 0.0:
            continue
        seq = [x / s for x in raw]
        score = _eval(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[geom] c={c:.3f}  score={score:.6f}")

    # ---- simple exponential decay (a_i ∝ exp(-β i)) ------------------
    for beta in np.linspace(0.001, 0.02, 20):
        raw = [math.exp(-beta * i) for i in range(n)]
        s = sum(raw)
        seq = [x / s for x in raw]
        score = _eval(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[exp ] beta={beta:.5f}  score={score:.6f}")

    # ---- Gaussian‑like (a_i ∝ exp(-β (i‑mid)^2)) --------------------
    mid = (n - 1) / 2.0
    for beta in np.linspace(0.0001, 0.001, 15):
        raw = [math.exp(-beta * (i - mid) ** 2) for i in range(n)]
        s = sum(raw)
        seq = [x / s for x in raw]
        score = _eval(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[gaus] beta={beta:.6f}  score={score:.6f}")

    # ---- power‑law (a_i ∝ (i+1)^‑α) ---------------------------------
    for alpha in np.linspace(0.5, 3.0, 14):
        raw = [1.0 / ((i + 1) ** alpha) for i in range(n)]
        s = sum(raw)
        seq = [x / s for x in raw]
        score = _eval(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[pow ] α={alpha:.3f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stochastic Dirichlet sampling + tiny hill‑climbing moves
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- draw a random point on the simplex --------------------
        # we use Gamma draws with position‑dependent shape parameters
        scale_len = np.random.uniform(0.1, 0.5) * n
        shape = np.exp(-np.arange(n) / scale_len)          # decays toward the end
        shape = shape * (0.5 + np.random.rand() * 1.5)     # add some randomness
        sample = np.random.gamma(shape, 1.0)               # element‑wise Gamma
        seq = (sample / sample.sum()).tolist()

        score = _eval(seq)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[rand] new best  score={score:.6f}")

        # ----- occasionally perturb the current best -----------------
        if best_seq is not None and np.random.rand() < 0.08:
            a = best_seq.copy()
            i, j = np.random.choice(n, 2, replace=False)
            # move a small random amount from i to j
            delta = min(a[i], 0.02 * np.random.rand())
            a[i] -= delta
            a[j] += delta
            a = _project_onto_simplex(np.array(a)).tolist()
            score = _eval(a)
            if score < best_score:
                best_score, best_seq = score, a
                print(f"[pert] improved    score={score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    # ------------------------------------------------------------------
    print(f"Finished search – best score = {best_score:.6f}")
    return best_seq
