def propose_candidate(seed=42, budget_s=1000, **kwargs):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    n = 7937  # length (we could also random choose near this)
    best_seq = None
    best_score = float('inf')

    # try various exponents
    for exponent in [0, 1, 2, 3, 4, 5, 6, 8, 10, 12]:
        # generate decreasing power shape
        idx = np.arange(n)

        seq = raw / total
        # ensure max <= 1000; after normalization max will be small
        # evaluate
        score = evaluate_sequence(seq.tolist())
        print(f"Exponent {exponent}, score {score}")
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
        # early exit if time near
        if time.time() >= deadline:
            break

"""
Propose a very low‑score sequence.

The evaluation function is:
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Scaling a does not change the score, therefore we can work with a
probability distribution (sum = 1).  For a decreasing power‑law
   a_i ∝ (n - i)^{p}    (0 ≤ i < n)

the convolution at shift n‑1 (the usually dominant term) is
   Σ a_i a_{n-1-i} ≈ (p+1)^2 * B(p+1, p+1) / n,
where B is the beta function.  For p = 1 this gives the classical
uniform bound 2.0, while p = 2 already improves it to ≈0.6 and
p = 3 to ≈0.23.  For moderate p (≈3–5) the maximal convolution moves
to the first shift (a_0²) and the resulting score is about
2·(p+1)² / n – still far below the target 1.503.

The algorithm simply tries a handful of exponents (and a few
exponential decays) on the given length, keeps the best sequence,
and returns it.  All values are normalised to sum to 1, so they are
well under the 1000‑clipping limit.
"""

import time
import numpy as np

def _make_power_sequence(n: int, p: int) -> list[float]:
    """Return a decreasing power‑law sequence of length n,
    normalised so that its sum equals 1."""
    # raw values (n-i)^p  – use float to avoid integer overflow
    idx = np.arange(n, dtype=np.float64)
    raw = (n - idx) ** p
    total = raw.sum()
    # normalise; after normalisation every entry is ≤ (p+1)/n << 1000
    prob = raw / total
    return prob.tolist()


def _make_exp_sequence(n: int, lam: float) -> list[float]:
    """Return an exponential decay sequence a_i = exp(-lam*i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-lam * idx)
    prob = raw / raw.sum()
    return prob.tolist()
