def projection_simplex(v):
    # v: np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_max(a):
    n = a.shape[0]

def compute_conv(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return conv[:2*n-1]

def project_simplex(v):
    # returns projection onto {x >= 0, sum x = 1}
    # reference: Duchi et al. 2008
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    # algorithm
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

    f(a) = max_k  Σ_i a_i a_{k‑i}

    (the gradient of that active term is 2·a_{k‑i}),
  * project back onto the probability simplex,
  * keep the best sequence seen.

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond].max() - 1          # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray):
    """
    Compute the full linear convolution a∗a using an FFT,
    return (max_value, argmax_index, full_convolution_array).
    """
    n = a.shape[0]

    conv = np.maximum(conv, 0.0)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv

        # small random jitter every few iterations to escape flat regions
        if iteration % 100 == 0:
            a += np.random.normal(scale=5e-4, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        iteration += 1

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
```
