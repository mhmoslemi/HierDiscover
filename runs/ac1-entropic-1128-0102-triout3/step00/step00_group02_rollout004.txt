```
    # v: array, sum to z
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\Phi(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the non‑negative vector **a** with itself.  
If we keep the ℓ₁‑norm fixed (`∑a_i = √(2n)`) the denominator is constant and the
objective reduces to the maximal entry of the convolution.  
Hence we have the following optimisation problem

```
minimise   max_k  (a * a)[k]
subject to a_i ≥ 0 ,   ∑ a_i = √(2n) .
```

The problem is non‑convex, but a smooth approximation of the *max* function gives us a
well‑behaved gradient that can be used in a projected (sub‑)gradient descent scheme.
A soft‑max with a temperature τ approximates the maximum:

\[
F_τ(a)=\tau\log\Bigl(\sum_k e^{(a*a)[k]/τ}\Bigr)
\qquad\text{and}\qquad
∇F_τ(a)=2\; (w * a^{\text{rev}}),\;
w_k=\frac{e^{(a*a)[k]/τ}}{\sum_j e^{(a*a)[j]/τ}} .
\]

`a^{rev}` is the reversed vector.  
The update step is

```
a ←  Π_S ( a – η ∇F_τ(a) )
```

where `Π_S` is the Euclidean projection onto the simplex
\(\{x≥0,\;∑x = √(2n)\}\) and `η` is a small stepsize.
After each step we evaluate the true objective (the maximum convolution entry)
and keep the best vector seen so far.

**Key ingredients**

* **FFT based convolutions** – O(n log n) time, crucial for n≈2400.  
* **Simplex projection** – guarantees feasibility after every update.  
* **Annealing schedule** for τ (starts ≈1, ends ≈0.01) to gradually sharpen the
  approximation of the max.  
* **Random restarts / perturbations** – occasional Dirichlet initialisations
  avoid being trapped in poor local minima.  
* **Time‑bounded loop** – stops automatically after the allotted 1000 s.

The algorithm is completely self‑contained, uses only `numpy` and the already
available `evaluate_sequence`, and prints progress information.

```python
"""
Projected gradient descent with soft‑max smoothing to minimise the
maximum entry of the self‑convolution of a non‑negative step function.
The routine works for any length; if a global variable `height_sequence_1`
exists its length is used (the current best length is 2401).
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Compute the linear convolution a * a using FFT, returning the
    first (2*len(a)-1) entries (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    grad = 2 * (w * a_rev)   where w_k ∝ exp(conv_k / τ)
    """
    # soft‑max weights
    w = np.exp(conv / tau)
    w /= w.sum()

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    # extract the n entries that correspond to the gradient
    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0,  Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are forced to zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Uses a projected gradient descent with a soft‑max approximation of the
    max‑convolution.  Returns the best sequence found before the time budget
    expires.
    """
    # -----------------------------------------------------------------
    # configuration & initialisation
    # -----------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a safety margin

    # use the length of the existing construction if available,
    # otherwise fall back to a reasonably large size.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    S = np.sqrt(2.0 * n)                    # required ℓ₁‑norm
    nfft = _next_pow2(2 * n - 1)            # FFT length (power of two)

    # start from the uniform sequence (the known baseline)
    a = np.full(n, S / n, dtype=np.float64)

    # bookkeeping for the best solution discovered
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # annealing / stepsize schedule
    tau = 1.0               # initial temperature for soft‑max
    tau_min = 0.01
    total_iters_est = 5000          # reasonable number of outer iterations
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    base_eta = 0.3 * S / n          # base stepsize (scaled with problem size)
    no_improve_counter = 0
    max_no_improve_before_restart = 1500

    iteration = 0
    while time.time() < deadline:
        # -------------------------------------------------------------
        # 1. evaluate current objective (max of the convolution)
        # -------------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                     # equal to evaluate_sequence when sum=S

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:05d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations.")
            break

        # -------------------------------------------------------------
        # 2. gradient of soft‑max approximation of the max‑convolution
        # -------------------------------------------------------------
        grad = _grad_softmax(a, conv, tau, nfft)

        # -------------------------------------------------------------
        # 3. descent step (projected onto the ℓ₁‑simplex)
        # -------------------------------------------------------------
        # stepsize slowly shrinks; we also normalise by the gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            eta = base_eta / (1.0 + iteration * 1e-4)
            step = eta * grad / grad_norm
        else:
            step = np.zeros_like(a)

        a_candidate = _project_to_simplex(a - step, S)

        # -------------------------------------------------------------
        # 4. accept if the true objective improves, otherwise keep a.
        #    Small random perturbations are added occasionally to escape
        #    shallow plateaux.
        # -------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        if cand_max < cur_max:
            a = a_candidate
        else:
            # occasional random jitter (1 % of iterations)
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (eta * 0.5)
                a = _project_to_simplex(a + jitter, S)

        # -------------------------------------------------------------
        # 5. temperature annealing
        # -------------------------------------------------------------
        tau = max(tau * tau_decay, tau_min)

        # -------------------------------------------------------------
        # 6. occasional restart if no improvement for a long time
        # -------------------------------------------------------------
        if no_improve_counter >= max_no_improve_before_restart:
            # draw a fresh random point on the simplex (Dirichlet)
            alpha = np.full(n, 0.3)        # moderate concentration
            a = np.random.dirichlet(alpha) * S
            no_improve_counter = 0
            print(f"[{iteration:05d}] Restarted from a random simplex point.")

        iteration += 1

    # final verification (in case rounding changed the ℓ₁‑norm)
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
```
