def subgradient_step(seq, step):
    # compute conv, find index with max
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    # compute gradient vector
    grad = np.zeros_like(seq)
    # For each j where 0<=k-j<n
    start = max(0, k - (len(seq) - 1))
    end = min(k, len(seq)-1)
    # for j in range(start, end+1):
    #    grad[j] = 2 * seq[k-j]
    grad[start:end+1] = 2 * seq[k - np.arange(start, end+1)]
    # update
    new_seq = seq - step * grad
    # enforce nonnegativity
    new_seq = np.maximum(new_seq, 0.0)
    # renormalize to sum 1 (or to sqrt(2n) factor used by evaluate)
    sum_seq = new_seq.sum()
    if sum_seq < 1e-12:
        new_seq = seq
        sum_seq = new_seq.sum()
    new_seq = new_seq / sum_seq  # sum=1
    return new_seq
```

def simplex_projection(v):
    # from https://arxiv.org/abs/1101.6081
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_step(seq, eta):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())
    # gradient of max conv = 2 * reversed seq at offset
    grad = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end+1)
    grad[idx] = 2 * seq[k - idx]
    new_seq = seq - eta * grad
    # project onto simplex
    new_seq = project_simplex(new_seq)
    return new_seq, conv.max(), k

"""
Propose a candidate step‑function sequence.

The algorithm searches for a non‑negative sequence a that minimizes

        2 * n * max(a * a) / (sum(a)**2)

where * denotes the discrete convolution.  The objective is invariant under
scaling, therefore we keep the ℓ₁‑norm of the sequence equal to one.
The search proceeds in three stages:

1. **Parametric families** – For several lengths ``n`` we try simple
   monotone families (power‑law decay ``a_i ∝ (i+1)^{-β}``) and keep the best
   candidate from a coarse grid of the exponent β.

2. **Sub‑gradient descent** – Starting from the best candidates we perform a
   projected sub‑gradient optimisation of the maximal convolution entry.
   The sub‑gradient of ``max_k (a*a)_k`` at the active index k∗ is
   ``g_i = 2·a_{k∗−i}`` (zero if the index is out of range).  After a step
   we project the vector onto the probability simplex (non‑negative, sum=1)
   and keep the best sequence seen.

3. **Random jitter** – Occasionally a small random perturbation is added to
   escape shallow local minima.

All operations are pure NumPy and run well within the 1000 s budget.
If no improvement is found the previously known construction
``height_sequence_1`` is returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0, sum x_i = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1        # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(seq: np.ndarray) -> float:
    """
    Compute the evaluation function for a sequence whose ℓ₁‑norm is 1.
    The function evaluate_sequence is invariant under scaling, therefore
    we can compute the score directly: 2 * n * max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _subgradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective
    max_k (seq * seq)_k.  The step size eta is a small positive scalar.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))                # active index
    # gradient of the active convolution entry
    grad = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1)
    grad[idx] = 2.0 * seq[k - idx]          # 2·a_{k−i}
    # take a step and project back onto the simplex
    new_seq = seq - eta * grad
    new_seq = _project_simplex(new_seq)
    # keep the sequence monotone decreasing (helps the antisymmetric
    # perturbations that lower the central convolution term)
    new_seq = np.sort(new_seq)[::-1]
    return new_seq


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """Generate a monotone decreasing power‑law sequence a_i ∝ (i+1)^{-β}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq = seq / seq.sum()
    return seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a sequence) that attempts to
    minimise the provided ``evaluate_sequence`` function.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a tiny margin

    # initialise from the best known construction if it exists
    best_seq = list(height_sequence_1) if 'height_sequence_1' in globals() else None
    best_score = float('inf')
    if best_seq is not None:
        try:
            best_score = evaluate_sequence(best_seq)
        except Exception:
            best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) Coarse grid search over power‑law families for several lengths
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1600, 2800]      # sizes that fit comfortably
    for n in lengths:
        if time.time() >= deadline:
            break

        # grid of exponents – wider range for larger n
        betas = np.linspace(0.0, 3.0, 61)      # 0 … 3 in steps of 0.05
        local_best_seq = None
        local_best_score = float('inf')

        for beta in betas:
            seq = _powerlaw_sequence(n, beta)
            score = _compute_score(seq)
            if score < local_best_score:
                local_best_score = score
                local_best_seq = seq

        # keep global best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.tolist()
            print(f"[grid] New global best {best_score:.6f} (n={n}, beta≈{beta:.3f})")

        # --------------------------------------------------------------
        # 2) Refine the locally best candidate with sub‑gradient descent
        # --------------------------------------------------------------
        seq = local_best_seq.copy()
        eta = 0.5 / n            # initial step size (tuned for stability)

        # run a limited number of optimisation steps; stop early if out of time
        for it in range(15000):
            if time.time() >= deadline:
                break

            # evaluate current candidate
            cur_score = _compute_score(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.tolist()
                print(f"[sgd] New global best {best_score:.6f} (n={n}, it={it})")

            # take a projected sub‑gradient step
            new_seq = _subgradient_step(seq, eta)

            # accept the step if it does not worsen the objective,
            # otherwise keep the old sequence (simple greedy acceptance)
            new_score = _compute_score(new_seq)
            if new_score <= cur_score:
                seq = new_seq
                cur_score = new_score
            else:
                # occasional stochastic acceptance to escape flat plateaus
                if np.random.rand() < 0.01:
                    seq = new_seq

            # decay step size slowly
            eta *= 0.9998

            # random jitter every few hundred iterations (helps to break symmetry)
            if (it + 1) % 250 == 0:
                jitter = (np.random.rand(n) - 0.5) * eta * 0.1
                seq = _project_simplex(np.maximum(seq + jitter, 0.0))

    # ------------------------------------------------------------------
    # Fallback: if nothing better was discovered, return the provided one.
    # ------------------------------------------------------------------
    if best_seq is None:
        # As a last resort, return a uniform sequence of moderate length.
        n = 500
        seq = np.full(n, 1.0 / n)
        best_seq = seq.tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
