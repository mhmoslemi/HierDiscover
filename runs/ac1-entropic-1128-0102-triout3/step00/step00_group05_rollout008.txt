    n = len(a)
    # compute conv
    fa = np.fft.rfft(a, n=2*n-1)

import numpy as np
import time, math, random

def compute_score(a):
    n = a.shape[0]
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    m = 2*n - 1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S*S)

def project_to_simplex(v, z):
    # v >= 0? accept any.
    if np.sum(v) == z and np.all(v >= 0):
        return v
    # from https://doi.org/10.1.. etc
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_weights(conv, beta):
    maxc = np.max(conv)
    # prevent overflow
    exp_term = np.exp(beta * (conv - maxc))
    total = np.sum(exp_term)
    if total == 0:
        # fallback uniform
        return np.full_like(conv, 1.0/conv.size)
    return exp_term / total

def gradient_step(a, beta, step_size, target_sum):
    n = a.shape[0]
    m = 2*n - 1
    # FFT to compute convolution of a with itself
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    w = softmax_weights(conv, beta)  # shape m
    # gradient = 2 * conv_valid(a, w)
    grad = 2.0 * np.convolve(a, w, mode='valid')
    # update
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    # project onto simplex to keep sum = target_sum
    a_new = project_to_simplex(a_new, target_sum)
    return a_new, np.max(conv)

The algorithm builds several candidate sequences (uniform, sine, semi‑circle and
Dirichlet samples), then repeatedly reduces the maximum self‑convolution
using a smooth approximation of the max (soft‑max) together with a projection
onto the simplex (fixed non‑negative sum).  The best sequence found before the

import time

# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex {x >= 0, sum x = z}.

    Reference: Duchi et al., "Efficient Projections onto the L1-Ball for
    Learning in High Dimensions".
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    # shift to guarantee non‑negativity after projection
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - z
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case – return a feasible point
        return np.maximum(v, 0.0)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (without the security wrapper)."""
    n = a.shape[0]
    S = np.sum(a)
    if S < 0.01:
        return float("inf")
    m = 2 * n - 1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S * S)


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution vector."""
    maxc = np.max(conv)
    # subtract max to avoid overflow
    exp_term = np.exp(beta * (conv - maxc))
    total = np.sum(exp_term)
    if total == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / total

    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new, target_sum)
    return a_new, np.max(conv)
