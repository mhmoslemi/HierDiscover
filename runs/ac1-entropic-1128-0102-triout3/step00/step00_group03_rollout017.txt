```python
import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, n=2000, **kwargs):
    """
    Gradient-descent (exponentiated) search for a non‑negative sequence that minimizes
    the evaluation function `evaluate_sequence`.

    The algorithm minimises a smooth approximation of the `max` in the evaluation
    (`log‑sum‑exp`) and updates the sequence with an exponentiated‑gradient step
    while projecting onto the simplex (non‑negative entries summed to one).  Several
    random initialisations are tried within the time budget and the best sequence
    is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    best_score = float('inf')
    best_seq = None

    # hyper‑parameters
    total_trials = 5   # number of random restarts
    for trial in range(total_trials):
        # random initial vector on the simplex
        a = np.random.rand(n)
        a = a / a.sum()
        # occasional use of uniform start
        if trial == 0:
            a = np.full(n, 1.0 / n)

        # schedule for smooth‑max temperature τ and learning rate η
        tau = 0.5
        eta = 0.5
        iter_count = 0

        while time.time() < deadline:
            # full convolution
            b = np.convolve(a, a)

            # evaluate periodically
            if iter_count % 20 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"Trial {trial}, iter {iter_count}, new best {best_score:.6f}")

            # smooth‑max weights (log‑sum‑exp trick for stability)
            scaled = tau * b
            max_scaled = np.max(scaled)
            exp_scaled = np.exp(scaled - max_scaled)
            w = exp_scaled / exp_scaled.sum()   # length 2n‑1, sum = 1

            # gradient of the smooth max with respect to a
            a_rev = a[::-1]
            conv_rev = np.convolve(a_rev, w)          # length n + (2n-1) - 1 = 3n-2
            T = conv_rev[n-1:2*n-1]                   # shape (n,)

            # sliding sum of w over a window of length n
            cum = np.cumsum(w, dtype=np.float64)
            W = cum[n-1:2*n-1] - np.concatenate(([0.0], cum[:n-1]))
            # alternative: np.concatenate

            grad = T + a * W

            # exponentiated‑gradient step
            a = a * np.exp(-eta * grad)
            # renormalise to the simplex
            a_sum = a.sum()
            if a_sum == 0 or np.isnan(a_sum):
                # re‑initialise if we hit numerical trouble
                a = np.random.rand(n)
                a = a / a.sum()
                continue
            a = a / a_sum

            # update schedules
            iter_count += 1
            if iter_count % 200 == 0:
                # slowly increase τ (making the approximation sharper) and
                # decay the learning rate.
                tau = min(tau * 1.05, 5.0)
                eta = max(eta * 0.95, 0.01)

    # fallback: if nothing succeeded use a simple uniform sequence
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    return best_seq.tolist()
```

            w = exp_scaled / np.sum(exp_scaled)  # len = 2*n-1

            # gradient
            a_rev = a[::-1]
            conv_rev = np.convolve(a_rev, w)  # length 3*n-2
            T = conv_rev[n-1:2*n-1]   # shape (n,)
            # sliding window sum of w over length n
            cum = np.concatenate(([0.0], np.cumsum(w, dtype=np.float64)))
            W = cum[n:2*n] - cum[0:n]   # length n
            grad = T + a * W

            # exponentiated gradient step
            a = a * np.exp(-eta * grad)

            # renormalize to simplex
            s = a.sum()
            if s == 0 or not np.isfinite(s):
                # reinitialize
                a = np.full(n, 1.0 / n)
                continue
            a = a / s

            iter_count += 1
            # schedule updates
            if iter_count % 200 == 0:
                # increase temperature, decrease step size
                tau = min(tau * 1.1, 5.0)
                eta = max(eta * 0.9, 0.01)

    # fallback: return uniform of length 3401
    if best_seq is None:
        n = 3401
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()

"""
Propose candidate sequence for the step‑function inequality.

The function performs a constrained optimisation on the simplex
(the non‑negative entries that sum to 1).  The objective is to
minimise the quantity returned by ``evaluate_sequence``:

    C(a) = 2·n·max_k (a * a)_k / (sum a)^2

Because the expression is scale‑invariant we can keep ``sum a = 1``.
We approximate the non‑smooth ``max`` by a smooth “log‑sum‑exp’’,
obtain its gradient analytically (using only a few FFT‑free
convolutions), and apply an exponentiated‑gradient update which
preserves non‑negativity.  After each update we renormalise the
vector onto the simplex.

The algorithm runs several independent restarts with different
lengths (short runs first, a longer run with the target length
≈3400 at the end).  Whenever a better sequence is found the global
best is stored and finally returned.

The method is completely self‑contained, uses only numpy and the
provided ``evaluate_sequence`` function and respects the 1000‑second
budget.
"""

import time
import numpy as np

def _smooth_max_weights(conv_vals: np.ndarray, tau: float) -> np.ndarray:
    """Return soft‑max weights w_k = exp(tau·conv_k) / Σ exp(tau·conv)."""
    scaled = tau * conv_vals
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)  # stability
    return exp_scaled / np.sum(exp_scaled)


def _gradient_of_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. the vector ``a``.
    a : shape (n,)
    w : shape (2n-1,) – soft‑max weights for each convolution index
    Returns gradient of shape (n,).
    """
    n = a.shape[0]

    # term T_i = Σ_j a_j · w_{i+j}
    a_rev = a[::-1]
    conv_rev = np.convolve(a_rev, w)          # length = 3n-2
    T = conv_rev[n - 1 : 2 * n - 1]           # slice of length n

    # sliding sums W_i = Σ_{k=i}^{i+n-1} w_k
    cum = np.concatenate(([0.0], np.cumsum(w, dtype=np.float64)))
    W = cum[n:2 * n] - cum[0:n]               # length n

    return T + a * W


def _exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient step followed by projection onto the simplex."""
    # multiplicative update (keeps positivity)
    a_new = a * np.exp(-eta * grad)

    # re‑normalise to sum = 1 (simplex projection)
    s = a_new.sum()
    if s <= 0 or not np.isfinite(s):
        # numerical trouble – fall back to uniform vector
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height sequence that minimises
    ``evaluate_sequence``.  Returns a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a few seconds margin

    best_score = float("inf")
    best_seq = None

    # Helper to evaluate and update the global best
    def _maybe_update(candidate):
        nonlocal best_score, best_seq
        score = evaluate_sequence(candidate.tolist())
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"New best: {best_score:.6f} (len={candidate.shape[0]})")
        return score

    # Length schedule – start with short vectors (fast iterations)
    lengths = [500, 1000, 2000, 3401]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Number of restarts for this length
        restarts = 3 if n < 2000 else 2

        for restart in range(restarts):
            if time.time() >= deadline:
                break

            # Initialise: uniform is a good baseline; occasional random start
            if restart == 0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                # Dirichlet‑like random start (still sums to 1)
                a = np.random.rand(n)
                a /= a.sum()

            # Hyper‑parameters (they evolve during the run)
            tau = 0.5          # smooth‑max temperature
            eta = 0.7          # step size
            it = 0

            while time.time() < deadline:
                # Full convolution a*a (size 2n‑1)
                conv = np.convolve(a, a)

                # Periodic evaluation – cheap enough to do every few steps
                if it % 30 == 0:
                    _maybe_update(a)
                    # stop early if the target is already surpassed
                    if best_score <= 1.503:
                        return best_seq.tolist()

                # Smooth‑max weights for the current convolution
                w = _smooth_max_weights(conv, tau)

                # Gradient of the smooth max w.r.t. a
                grad = _gradient_of_smooth_max(a, w)

                # Exponential‑gradient step + simplex projection
                a = _exponential_update(a, grad, eta)

                # Schedule adaptation
                it += 1
                if it % 200 == 0:
                    tau = min(tau * 1.12, 5.0)      # sharpen the max approximation
                    eta = max(eta * 0.88, 0.02)    # slowly decay step size

    # If nothing better than the uniform baseline was found, fall back to it
    if best_seq is None:
        n = 3401
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    return best_seq.tolist()
