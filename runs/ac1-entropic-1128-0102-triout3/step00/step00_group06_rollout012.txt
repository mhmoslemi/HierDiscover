def search_for_n(n, budget):
    # returns best seq found within budget seconds
    start = time.time()
    # random initial
    a = np.random.rand(n)
    a /= a.sum()
    best_a = a.copy()
    best_score = ratio(a)
    temperature = 0.5
    while time.time() - start < budget:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose shift eps
        max_shift = min(a[i], 0.1) # avoid too big
        eps = (np.random.rand()*2 - 1) * max_shift
        # Actually we must keep sum constant: we will move eps from i to j
        eps = np.clip(eps, -a[i], a[j])  # ensure a[i]-eps>=0, a[j]+eps>=0
        if eps == 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        score = ratio(a_new)
        # acceptance
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        # Sim anneal acceptance
        if score < ratio(a) or np.exp((ratio(a)-score)/temperature) > np.random.rand():
            a = a_new
        # update temperature slowly
        temperature *= 0.999999 # or linear schedule
    return best_a, best_score

"""
Simple global optimiser for the "step‑function" challenge.

The evaluation function is homogeneous: scaling the whole vector does not
change the score, therefore we normalise every candidate to have L1–norm 1.
The score then reduces to 2·n·max(conv(a,a)).

The optimiser works in three stages:

1. Random sampling of many short vectors (n∈[2,15]) to get a decent baseline.
2. Nelder–Mead optimisation on the log‑variables (to keep positivity) for a
   few restarts per length.  The objective is exactly the score (with the
   normalisation already built in).
3. A tiny deterministic hill‑climb that repeatedly moves a small amount of
   mass from the index contributing most to the current maximal convolution
   entry to an index that does not contribute to that entry.  The move is
   accepted if it improves the score (occasionally we accept a worsening
   move to escape tiny plateaus).

The whole routine stops after the allotted budget (default 1000 s) and
returns the best sequence found.
"""

import time
import random
import numpy as np
from scipy.optimize import minimize, linprog  # linprog is kept for compatibility

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Score for a normalised (sum=1) non‑negative vector."""
    conv = np.convolve(a, a)
    return 2.0 * len(a) * conv.max() / (a.sum() ** 2)


def random_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random positive vector of length n, normalised to have sum 1."""
    v = rng.random(n)
    v /= v.sum()
    return v


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Geometric progression a_i ∝ r**i, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    v = r ** i
    v /= v.sum()
    return v


def hill_climb(a0: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Very cheap deterministic hill‑climb.
    Moves a tiny amount of mass from the index contributing most to the
    current maximal convolution entry to an index that does not contribute.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = fast_score(a)

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))               # index with maximal value
        # contribution of each coordinate to conv[k]
        contrib = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    contrib[i] = a[i] * a[i]              # diagonal term
                else:
                    contrib[i] = 2.0 * a[i] * a[j]        # both (i,j) and (j,i)

        i_max = int(np.argmax(contrib))                       # biggest offender
        # indices that do NOT appear in this convolution sum
        zero_idx = [i for i in range(n) if not (0 <= k - i < n)]
        if zero_idx:
            i_min = random.choice(zero_idx)
        else:
            i_min = int(np.argmin(a))                         # fallback

        # propose a tiny transfer of mass
        eps = min(a[i_max], 0.05) * random.random()
        if eps <= 0.0:
            break
        a_new = a.copy()
        a_new[i_max] -= eps
        a_new[i_min] += eps
        # renormalise (should stay very close to 1)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        sc_new = fast_score(a_new)
        sc_cur = fast_score(a)

        # accept if improvement, otherwise with a tiny probability
        if sc_new < sc_cur or random.random() < 0.01:
            a = a_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                # early exit if we already beat the target
                if best_sc < 1.503:
                    break
    return best_a, best_sc


def optimise_length_n(n: int, rng: np.random.Generator,
                     global_deadline: float) -> (np.ndarray, float):
    """
    For a fixed length n perform a few random restarts of the Nelder–Mead
    optimiser (working on log‑variables to enforce positivity) and
    finish each restart with a short hill‑climb.
    """
    best_vec = None
    best_sc = np.inf

    # number of restarts is limited by the remaining time
    while time.time() < global_deadline - 0.2:
        # ---- random start -------------------------------------------------
        a0 = random_sequence(n, rng)
        # work in log‑space to keep variables strictly positive
        z0 = np.log(a0 + 1e-12)

        # ---- objective (score) --------------------------------------------
        def obj(z):
            a = np.exp(z)
            a = a / a.sum()
            conv = np.convolve(a, a)
            return 2.0 * n * conv.max()      # sum(a)==1 → denominator = 1

        # ---- Nelder–Mead ---------------------------------------------------
        # limit the number of function evaluations to keep the runtime low
        max_fevals = 1500
        res = minimize(obj, z0, method='Nelder-Mead',
                       options={'maxfev': max_fevals, 'disp': False,
                                'xatol': 1e-12, 'fatol': 1e-12})
        if not res.success:
            continue

        a_opt = np.exp(res.x)
        a_opt = a_opt / a_opt.sum()
        sc_opt = fast_score(a_opt)

        # ---- short hill‑climb ---------------------------------------------
        sub_deadline = min(global_deadline, time.time() + 0.2)
        a_hc, sc_hc = hill_climb(a_opt, sub_deadline)

        # keep the best of the two
        if sc_hc < sc_opt:
            a_opt, sc_opt = a_hc, sc_hc

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_vec = a_opt.copy()
            print(f"   new local best for n={n}: score={best_sc:.6f}")

        # stop early if we already beat the target
        if best_sc < 1.503:
            break

    return best_vec, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a low evaluation
    score.  The routine respects the global time budget, tries a few
    different lengths, performs Nelder–Mead optimisation with random restarts
    and finishes each restart with a tiny hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # global best
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) quick random sampling – gives a safe fallback
    # ------------------------------------------------------------------
    while time.time() + 0.05 < deadline:
        n = rng.integers(2, 16)                # very short vectors are cheap
        a = random_sequence(n, rng)
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"random sample n={n} -> score={sc:.6f}")

        # stop if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2) systematic optimisation for several lengths
    # ------------------------------------------------------------------
    # we try lengths from 2 up to 15 (larger lengths give finer granularity)
    lengths = list(range(2, 16))
    rng.shuffle(lengths)

    for n in lengths:
        if time.time() >= deadline:
            break
        vec, sc = optimise_length_n(n, rng, deadline)
        if vec is not None and sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(f"global best updated (n={n}) -> score={sc:.6f}")

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3) final safety net – if nothing succeeded, return the known 2‑element optimum
    # ------------------------------------------------------------------
    if best_seq is None or best_score == np.inf:
        # known optimal 2‑element construction (score ≈ 1.777)
        best_seq = np.array([2.0 / 3.0, 1.0 / 3.0])
        best_score = fast_score(best_seq)

    print(f"final best score = {best_score:.6f}  (length {len(best_seq)})")
    return best_seq.tolist()
