def propose_candidate(seed=0, budget_s=990, **kwargs):
    import numpy as np, time, math, random

    np.random.seed(seed)
    n = 8000  # length
    # initial sequence: random positive distribution, then scale so sum = 1 (any positive sum)
    a = np.random.exponential(scale=1.0, size=n)
    # optionally make symmetric
    a = (a + a[::-1]) / 2  # ensure symmetry for easier flattening (convolution symmetric).
    sum_target = a.sum()
    # baseline evaluation
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score: {best_score:.6f}')
    start = time.time()
    # Determine FFT size
    L = 1 << ((2*n-1).bit_length())
    # parameters
    temperature = 1.0
    decay = 0.9995
    step_factor = 0.02  # fraction of mean
    min_step = 1e-6
    iter_count = 0
    while time.time() - start < budget_s - 5:
        # Compute convolution via FFT
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        # compute gradient vector
        g = np.zeros_like(a)
        i_min = max(0, k_max - (n-1))
        i_max = min(n-1, k_max)
        idx = np.arange(i_min, i_max+1)
        jdx = k_max - idx
        g[idx] = 2.0 * a[jdx]
        # Step size adapt
        step = step_factor * np.mean(a)
        if step < min_step:
            step = min_step
        a_new = a - step * g
        # Clip non-negative
        a_new = np.maximum(a_new, 0.0)
        # Rescale to preserve sum_target
        s_new = a_new.sum()
        if s_new > 0:
            a_new *= (sum_target / s_new)
        # Evaluate
        score = evaluate_sequence(a_new.tolist())
        # Accept if better or with temperature
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f'Iter {iter_count}: new best {best_score:.6f}')
        # Accept move according to SA
        if score < best_score or math.exp((best_score - score) / max(temperature, 1e-8)) > np.random.rand():
            a = a_new
        else:
            # occasional random jitter
            i, j = np.random.randint(n, size=2)
            delta = step * np.random.rand() * a[i]
            a[i] = max(0.0, a[i] - delta)
            a[j] += delta

        temperature *= decay
        # occasional random reset small
        if np.random.rand() < 0.001:
            a = np.random.exponential(scale=1.0, size=n)

            s_target = a.sum()

        iter_count += 1
    # final best sequence as list
    print(f'Finished after {iter_count} iterations. Best score: {best_score:.6f}')
    return best_seq.tolist()

"""
Simple simulated‑annealing / gradient approach to flatten the self‑convolution
of a non‑negative step function.

The algorithm works with a fixed length `n` (≈ 8000, similar to the current
best construction).  Starting from a random positive vector (optionally
perturbed from the provided `height_sequence_1`) we repeatedly

1. compute the linear convolution a*a with an FFT (O(n log n));
2. locate the shift `k_max` where the convolution attains its maximum;
3. build the gradient of that peak – for the peak at shift `k`
   the gradient component is `g[i] = 2 * a[k‑i]` when `0 ≤ k‑i < n`,
   otherwise `0`;
4. move a small amount opposite to the gradient, clip negatives and
   rescale to keep the total sum constant (the objective is scale‑invariant);
5. optionally add a tiny “edge‑bias” that slowly pushes mass toward the ends;
6. accept the new point if it improves the score or with a
   Metropolis probability (simulated‑annealing) and keep the best
   sequence seen so far.

The whole loop runs until the supplied time budget expires (≈ 1000 s) and
returns the best sequence found.  The score is exactly the value returned
by `evaluate_sequence`, but it is recomputed from the already available
convolution to avoid a second FFT call.

The method is deliberately different from the LP‑based approach used in the
reference implementation and is designed to explore non‑uniform, edge‑heavy
distributions that can lower the peak of the autocorrelation.
"""

import time
import math
import numpy as np

# Helper: length of the zero‑padded FFT (next power of two ≥ 2n‑1)
def _fft_len(n: int) -> int:
    return 1 << ((2 * n - 1).bit_length())

# Linear (non‑circular) convolution via FFT, returns the real part only.
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)
    return c[: 2 * len(a) - 1]

# Compute the evaluation score from the convolution (no extra FFT).
def _score_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    # protect against division by zero – the caller ensures sum_a ≥ 0.01
    max_c = np.max(conv)
    return 2.0 * n * max_c / (sum_a * sum_a)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine returns a plain Python list.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Set problem size (use the same order of magnitude as the current
    #     best construction).  The exact length is not critical – larger
    #     n gives a lower theoretical bound 2n/(2n‑1) ≈ 1 but also costs
    #     more time per iteration.
    # ------------------------------------------------------------------
    n = 8000                               # length of the step function
    L = _fft_len(n)                        # FFT length (power of two)

    # ------------------------------------------------------------------
    # 2.  Initialise the sequence.
    #     If a previous construction exists we perturb it a little;
    #     otherwise we start from a random exponential vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        # keep the original length if it differs from our target length
        if len(a) != n:
            # simple truncation / padding to the desired length
            if len(a) > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - len(a)), "constant")
        # tiny random noise to break symmetry
        a += np.random.normal(scale=0.005, size=n)
    else:
        a = np.random.exponential(scale=1.0, size=n)

    # enforce non‑negativity and a reasonable scale (the objective is
    # homogeneous – any scaling leaves the score unchanged)
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    total_sum = a.sum()
    if total_sum < 0.01:                         # safety guard
        a += 0.01 / n
        total_sum = a.sum()

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    best_a = a.copy()
    # compute an initial score (use the fast helper)
    conv = _conv_fft(a, L)
    best_score = _score_from_conv(conv, total_sum, n)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # keep a few seconds margin

    # annealing parameters
    temperature = 1.0
    decay = 0.9995
    # step size – a small fraction of the current mean value
    step_factor = 0.02
    min_step = 1e-8
    # edge‑bias factor (gradually pushes mass to the ends)
    edge_bias = 0.001

    iteration = 0
    while time.time() < deadline:
        # ----- convolution and peak location --------------------------------
        conv = _conv_fft(a, L)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # ----- gradient of the peak (linear w.r.t. a) -----------------------
        g = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idx = np.arange(i_min, i_max + 1)
        jdx = k_max - idx
        g[idx] = 2.0 * a[jdx]

        # ----- take a step opposite to the gradient -----------------------
        step = step_factor * np.mean(a)
        if step < min_step:
            step = min_step
        a_proposed = a - step * g
        a_proposed = np.maximum(a_proposed, 0.0)

        # ----- gentle edge‑bias (U‑shaped encouragement) -----------------
        if edge_bias > 0:
            positions = np.arange(n)
            mid = (n - 1) / 2.0
            # quadratic bias: larger at the two ends, 1 in the centre
            bias = 1.0 + edge_bias * ((positions - mid) / mid) ** 2
            a_proposed *= bias

        # ----- renormalise to keep the total L1 norm constant -------------
        s_new = a_proposed.sum()
        if s_new > 0:
            a_proposed *= (total_sum / s_new)

        # ----- respect the hard caps imposed by evaluate_sequence -----------
        a_proposed = np.clip(a_proposed, 0.0, 1000.0)

        # ----- compute the new score (no extra FFT) -------------------------
        conv_new = _conv_fft(a_proposed, L)
        new_score = _score_from_conv(conv_new, total_sum, n)

        # ----- keep the globally best sequence -----------------------------
        if new_score < best_score:
            best_score = new_score
            best_a = a_proposed.copy()
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # ----- simulated‑annealing acceptance --------------------------------
        accept = False
        if new_score < best_score:                    # improvement
            accept = True
        else:
            # Metropolis criterion (temperature slowly decays)
            prob = math.exp((best_score - new_score) / max(temperature, 1e-12))
            accept = prob > np.random.rand()

        if accept:
            a = a_proposed
        else:
            # occasional random jitter to escape shallow plateaus
            if np.random.rand() < 0.001:
                i, j = np.random.randint(n, size=2)
                delta = step * np.random.rand() * a[i]
                a[i] = max(0.0, a[i] - delta)
                a[j] += delta

        # ----- temperature schedule -----------------------------------------
        temperature *= decay

        # ----- very rare full restart (helps avoid dead‑ends) ---------------
        if np.random.rand() < 0.0005:
            a = np.random.exponential(scale=1.0, size=n)
            a = np.clip(a, 0.0, 1000.0)
            a *= (total_sum / a.sum())
            print(f"[{iteration:6d}] Random restart performed")

        iteration += 1

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain list of floats)
    # ------------------------------------------------------------------
    print(f"Optimization finished after {iteration} iterations.")
    print(f"Best score obtained: {best_score:.6f}")
    # Ensure the final list respects the input checks of evaluate_sequence
    best_seq = [float(x) for x in best_a.tolist()]
    return best_seq
