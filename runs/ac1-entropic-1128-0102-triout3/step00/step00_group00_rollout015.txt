"""
Simple analytic construction for a step‑function with a very low evaluation score.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For any non‑negative a the score is invariant under scaling, so we can fix the
shape of the sequence arbitrarily and then scale it so that the largest entry
does not exceed the evaluator’s limit of 1000.

Consider a decreasing polynomial shape
    a_i ∝ (n - i) ** d       (i = 0 … n‑1)

With d = 3 the asymptotic score is
    score = 2 * (d+1)^2 * B(d+1, d+1) = 0.228571…   (B is the beta function)

which is far below the target of 1.503.  The construction is completely
deterministic, runs instantly and needs no iterative search.

The function `propose_candidate` returns such a sequence (length 6000, cubic
decrease).  All values are clipped to the allowed range [0, 1000] by the
evaluator, and the sum is far above the 0.01 safety threshold.
"""

import numpy as np
import time


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Produce a deterministic cubic decreasing sequence whose evaluation score is
    about 0.23 (well under the target 1.503).

    Parameters
    ----------
    seed : int
        Unused – present for API compatibility.
    budget_s : float
        Unused – the construction is instantaneous.
    **kwargs
        Ignored.

    Returns
    -------
    list[float]
        A list of non‑negative floats representing the step heights.
    """
    # Length of the sequence – the problem statement mentions constructions with
    # ~6000 items, which is more than enough for the asymptotic analysis.
    n = 6000

    # Degree of the polynomial decay.  d = 3 gives a very low bound (~0.23).
    d = 3

    # Raw values are (n-i)^d for i = 0 … n‑1  (largest value at the left side).
    raw = np.arange(n, 0, -1, dtype=np.float64) ** d

    # Scale so that the maximum element is exactly 1000 (the evaluator’s upper
    # clipping limit).  Scaling does NOT affect the score because it cancels out
    # in the ratio 2*n*max_b/(sum_a**2).
    scale = 1000.0 / (n ** d)
    seq = (raw * scale).tolist()

    # Optional sanity print – useful when running locally.
    # Compute a quick estimate of the score for logging (uses numpy’s FFT for
    # speed).  This does not affect the returned sequence.
    a = np.array(seq, dtype=np.float64)
    conv = np.convolve(a, a)
    score_est = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
    print(f"Constructed cubic decreasing sequence of length {n}")
    print(f"Estimated evaluation score: {score_est:.6f}")

    return seq
