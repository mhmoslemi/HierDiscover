def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5
    n = 500   # moderate length
    # initial distribution: descending linearly
    a = np.arange(n, 0, -1, dtype=float)
    a = a / a.sum()
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    temperature = 1.0
    iteration = 0
    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to transfer
        eps = np.random.exponential(scale=0.01)  # small
        eps = min(eps, a[i])  # cannot go negative
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # maybe re-normalize only if needed? sum still 1.
        # ensure non-negative automatically
        # optional small noise to ensure sum.
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        # accept if better or with prob exp(-(score - cur_score)/temperature)
        cur_score = evaluate_sequence(a.tolist())
        delta = score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            a = a_new
            cur_score = score
            if score < best_score:
                best_score = score
                best = a.copy()
                print(f"Iter {iteration}: New best {best_score}")
        # cooling
        iteration += 1
        temperature *= 0.9999  # or schedule
    return best.tolist()
