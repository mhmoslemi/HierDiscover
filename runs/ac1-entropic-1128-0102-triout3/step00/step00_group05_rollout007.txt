def proj_simplex(v):
    # returns projection onto simplex sum(v)=1, v>=0
    n=len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # Implementation from literature (Condon, Duchi)
    # Handles v as 1-D numpy array.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find position where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0  # all zero? but sum must be 1; but if all negative, result is zero vector? Should not happen.
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_self_fft(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length()) # next power of two
    # real FFT
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:size]
    return conv

def compute_avg_shifts(a, max_idxs):
    n = len(a)
    avg = np.zeros(n)
    for K in max_idxs:
        # compute idx = K - np.arange(n)
        idx = K - np.arange(n)
        # mask boundaries
        mask = (idx >= 0) & (idx < n)
        avg[mask] += a[idx[mask]]
    avg /= len(max_idxs)
    return avg

def project_to_simplex(v):
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative; return uniform? but we want sum=1, nonnegative
        # return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    fa = np.fft.rfft(a, n=N)

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all defined at top level as required)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i >= 0 , sum_i x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full convolution of a with itself using an FFT whose length is the
    next power of two (fast and numerically stable).
    """
    n = a.shape[0]
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:size]
    return conv


def avg_shifts_from_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    For each position i compute the average of a_{K-i} over all
    K in max_idxs (values outside the interval are taken as zero).
    """
    n = a.shape[0]
    avg = np.zeros_like(a)
    for K in max_idxs:
        i_start = max(0, K - (n - 1))
        i_end = min(n - 1, K)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        idx = K - i_range                     # guaranteed 0 <= idx < n
        avg[i_range] += a[idx]
    avg /= len(max_idxs)
    return avg


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided ``evaluate_sequence`` function.

    The algorithm runs a projected sub‑gradient descent on the objective

        F(a) = 2 * n * max( a * a ) / ( sum(a)^2 )

    where ``a * a`` denotes the discrete (full) convolution of the vector
    with itself.  Each iterate is projected onto the probability simplex
    (non‑negative, sum = 1) to keep the objective scale‑invariant.
    Several random restarts are performed until the time budget expires.
    The best sequence encountered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a safety margin

    best_score = np.inf
    best_seq = None

    # Length of the candidate.  4000 gives a 8191‑long convolution,
    # the next power of two is 8192 – cheap for FFT.
    n = 4000

    # ------------------------------------------------------------------
    # Helper for a cheap bound based on max convolution value (sum is 1)
    # ------------------------------------------------------------------
    def cheap_score(max_val, total_sum):
        return 2.0 * n * max_val / (total_sum * total_sum)

    # ------------------------------------------------------------------
    # Main outer loop – a few random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- initialise a new restart ---------------------------------
        if np.random.rand() < 0.5:
            # purely random point on the simplex (exponential = Dirichlet(1))
            a = np.random.exponential(scale=1.0, size=n)
        else:
            # decreasing linear profile plus a tiny random jitter
            a = np.linspace(1.0, 0.1, n) + 0.01 * np.random.rand(n)
        a = a / a.sum()                       # exact projection onto simplex

        # momentum variables (simple exponential moving average of the sub‑gradient)
        velocity = np.zeros_like(a)
        beta = 0.9

        step = 0.5          # initial step size
        min_step = 1e-4

        # ----- inner optimisation loop ------------------------------------
        for it in range(5000):
            if time.time() >= deadline:
                break

            conv = conv_self_fft(a)
            max_val = conv.max()
            # treat all entries within a tiny tolerance as active
            eps = 1e-12 * max_val
            max_idxs = np.flatnonzero(conv >= max_val - eps)

            # sub‑gradient of the max‑convolution term
            avg = avg_shifts_from_max(a, max_idxs)          # shape (n,)
            # gradient of the full objective (sum(a) == 1 after projection)
            grad = 4.0 * n * (avg - max_val)

            # momentum update + projected gradient step
            velocity = beta * velocity + (1.0 - beta) * grad
            a_new = a - step * velocity
            a = project_to_simplex(a_new)

            # gradually shrink the step size
            step = max(step * 0.998, min_step)

            # ---- periodic evaluation of the true objective ----------------
            if it % 20 == 0:
                # the official evaluator does the final clipping/normalisation,
                # we call it to stay perfectly consistent
                try:
                    true_score = evaluate_sequence(a.tolist())
                except Exception:                     # very unlikely
                    true_score = cheap_score(max_val, a.sum())

                if true_score < best_score:
                    best_score = true_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.1f}s] "
                          f"new best = {best_score:.6f} (iter {it}, n={n})")

                # early termination if we already beat the target
                if best_score < 1.503:
                    deadline = time.time()   # break outer loop as well
                    break

        if best_score < 1.503:
            break   # stop after reaching the desired bound

    # fallback – uniform distribution (always feasible and yields score = 2)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list as required by the harness
    return best_seq.tolist()
