```
def simplex_projection(v, s):
    """Project v onto the simplex of sum s and nonnegative."""
    # shift by tau such that w_i = max(v_i - tau, 0) and sum w = s.
    # Implementation using sorting.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def subgradient(seq, max_k):
    n = len(seq)
    idx = np.arange(n)
    j = max_k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    g[mask] = 2 * seq[j[mask]]
    return g
```

Now simplex projection:

```
def project_to_simplex(v, s):
    # based on algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        tau = 0
    else:
        rho = ind[cond].max()
        tau = cssv[cond].max() / rho
    w = np.maximum(v - tau, 0)
    return w
```

            for alpha in alphas:

                a = 1.0 + alpha * p

Simple search for a step‑function with a low max‑self‑convolution.

The evaluation function is homogeneous:
score = 2 * n * max(a*a) / (sum(a)**2)

We therefore search for zero‑mean patterns p with values in [-1,1] and
form candidate sequences

    a = 1.0 + α * p        (α ≤ 1 ensures a_i ≥ 0)

The pattern is centred (mean zero) so sum(a) = n, which keeps the
objective simple.  Two families of patterns are tried:

* linear_u    – a U‑shaped linear pattern (high at the ends, low in the centre);
* cosine_u    – a cosine wave with one full period (high at the ends,
                low in the centre).

For each length n we sweep α ∈ [0,0.99] and keep the best sequence.
After the sweep we perform a tiny stochastic refinement that moves a

def _fft_len(n: int) -> int:
    """Next power of two ≥ 2*n‑1 (suitable length for the FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _convolution_max(seq: np.ndarray, fft_len: int) -> float:
    """Return max value of the linear convolution seq * seq using FFT."""
    # rfft/irfft are efficient for real sequences.
    fft_seq = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fft_seq * fft_seq, n=fft_len)[: 2 * len(seq) - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact evaluation used by the target function (avoids evaluate_sequence)."""
    n = len(seq)
    fft_len = _fft_len(n)
    max_b = _convolution_max(seq, fft_len)
    sum_a = seq.sum()
    # protect against near‑zero sum – matches evaluate_sequence behaviour
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _linear_u_pattern(n: int) -> np.ndarray:
    """
    U‑shaped linear pattern, values in [-1, 1] and zero mean:
        p_i = (2*|i - (n-1)/2|/(n-1)) - 1   (original shape)
    Then we subtract the mean to enforce Σp_i = 0.
    """
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    p = (2.0 * np.abs(i - mid) / mid) - 1.0
    p -= p.mean()
    return p


def _cosine_u_pattern(n: int) -> np.ndarray:
    """
    Cosine pattern with one full period (high at both ends, low in centre).
    The raw cosine already has almost zero mean; we centre it explicitly.
    """
    i = np.arange(n, dtype=np.float64)
    p = np.cos(2.0 * np.pi * i / (n - 1))
    p -= p.mean()
    return p
