def simplex_projection(v, s=1.0):
    # v is numpy array, may contain negative values
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all false, set zero
        rho = 0
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

            a = a_new
            # if also improvement over best
            if eval_new < best_score:

import time, numpy as np

def project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x >= 0, sum x = s}."""
    # Shift for numerical stability
    n = v.shape[0]
    if s < 0:
        raise ValueError('Target sum must be non‑negative')
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def evaluate_fast(a):
    """Fast copy of evaluate_sequence for internal use (sum already >0)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2 * len(a) * b.max() / (s * s))

"""
search_step_function.py

A fast heuristic for minimising the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The idea is to flatten the autocorrelation (the convolution of the
vector with itself).  For a vector a with sum = 1 the average value of
the convolution is 1 / (2n‑1).  If the convolution were perfectly flat,
the score would be 2·n/(2n‑1) ≈ 1.  Starting from a random point on the
probability simplex we perform projected gradient descent on the
quadratic loss

    ½‖conv(a, a) – μ‖²   with μ = mean(conv(a, a))

The gradient of this loss is

    g = 2 * ( (conv(a, a) – μ) * a_rev )

where * denotes a full convolution and a_rev is a reversed copy of a.
After each step we project back onto the simplex (non‑negative entries,
sum = 1).  The step size is adapted, and occasional random
perturbations are applied when the algorithm stalls.  The best vector
found before the time budget expires is returned.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluate_fast(a: np.ndarray) -> float:
    """Straight‑forward evaluation; identical to evaluate_sequence but cheap."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2 * len(a) * b.max() / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.  The algorithm
    runs projected gradient descent on the variance of the convolution and
    returns the best sequence found before the time budget runs out.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # keep a tiny safety margin

    # --- problem size -------------------------------------------------
    # Use a moderate length (fast to evaluate, yet large enough to get a
    # meaningful improvement over the uniform distribution).
    if 'height_sequence_1' in globals():
        max_len = len(height_sequence_1)
    else:
        max_len = 300
    n = min(300, max_len)          # 300 works comfortably within the time limit

    # --- initialise a random point on the simplex --------------------
    a = np.full(n, 1.0 / n)                 # start from uniform
    a += 0.02 * rng.standard_normal(n)      # add a small perturbation
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                 # enforce sum = 1

    best_a = a.copy()
    best_score = _evaluate_fast(best_a)
    print(f"[init] n = {n}, score = {best_score:.6f}")

    # --- optimisation parameters --------------------------------------
    step = 0.5                # initial gradient step size
    min_step = 1e-8
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # 1) compute convolution and its mean
        conv_full = np.convolve(a, a)        # length 2n‑1
        mu = conv_full.mean()
        err = conv_full - mu                 # deviation from the flat target

        # 2) gradient of ½‖err‖² w.r.t. a  = 2 * (err * a_rev)
        #    where * is full convolution
        grad_full = np.convolve(err, a[::-1])          # length 3n‑2
        grad = 2.0 * grad_full[(len(err) - 1):(len(err) - 1 + n)]

        # 3) tentative update and projection onto the simplex
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        # 4) evaluate candidate
        cand_score = _evaluate_fast(a_candidate)
        cur_score = _evaluate_fast(a)

        if cand_score < cur_score:
            # accept improvement
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[{iteration:5d}] new best score = {best_score:.6f}")
            # slightly increase step size after a success
            step = min(1.0, step * 1.02)
            no_improve = 0
        else:
            # reject – shrink step size
            step = max(min_step, step * 0.5)
            no_improve += 1

        # occasional random perturbation if we seem stuck
        if no_improve >= 80:
            pert = 0.05 * rng.standard_normal(n)
            a = a + pert
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            step = 0.5
            no_improve = 0

        iteration += 1

    # ----- final polishing (very small step) -------------------------
    for _ in range(5):
        conv_full = np.convolve(best_a, best_a)
        mu = conv_full.mean()
        err = conv_full - mu
        grad = 2.0 * np.convolve(err, best_a[::-1])[(len(err) - 1):(len(err) - 1 + n)]
        best_a = best_a - 0.1 * grad
        best_a = np.maximum(best_a, 0.0)
        best_a = _project_simplex(best_a)

    # return the best sequence as a plain Python list
    return best_a.tolist()
