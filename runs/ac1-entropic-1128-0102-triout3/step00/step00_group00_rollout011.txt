def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random, math
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    # choose length maybe random
    n = np.random.randint(30, 200)  # moderate size
    # initial sequence: random Dirichlet
    a = np.random.exponential(size=n)
    a = a / a.sum()
    best_a = a.copy()
    best_val = 2 * n * np.convolve(a, a).max()
    
    # parameters
    eta = 0.1
    decay = 0.9995
    min_eta = 1e-5
    
    # main loop
    while time.time() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        kmax = b.argmax()
        # gradient of max_b w.r.t a:
        grad = np.zeros_like(a)
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2 * a[j[mask]]  # derivative of b_k w.r.t a_i
        # step opposite to gradient
        a_new = a - eta * grad
        # ensure nonnegative
        a_new = np.maximum(a_new, 0.0)
        # if all zero, reinitialize
        if a_new.sum() == 0:
            a_new = np.random.exponential(size=n)
        # renormalize sum to 1
        a_new = a_new / a_new.sum()
        a = a_new
        
        # compute new objective quickly
        b = np.convolve(a, a)
        cur_val = 2 * n * b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # optionally print progress
            print(f"Iter improvement: n={n}, val={best_val:.6f}")
        
        # adapt step size
        eta = max(min_eta, eta * decay)
        
        # occasional random restart for exploration
        if random.random() < 0.001:
            a = np.random.exponential(size=n)
            a = a / a.sum()
    
    # return best sequence as list of floats (convert to python list)
    return best_a.tolist()

    # function to compute score
    def score(seq):
        b = np.convolve(seq, seq)
        return 2 * len(seq) * b.max() / (np.sum(seq) ** 2)

    while time.time() < deadline:
        # random length between 30 and 120
        n = random.randint(30, 120)
        # random init via Dirichlet (exponential)
        a = np.random.exponential(scale=1.0, size=n)
        a = a / a.sum()
        cur_score = score(a)
        # local search
        # parameters
        step = 0.1
        temp = 0.1
        # annealing schedule
        it = 0
        max_iter = 2000
        while it < max_iter and time.time() < deadline:
            # compute convolution and max index
            b = np.convolve(a, a)
            max_b = b.max()
            kmax = b.argmax()
            # compute gradient of max_b w.r.t a
            grad = np.zeros_like(a)

def reduce_max(a):
    b = np.convolve(a,a)
    kmax = np.argmax(b)
    contrib = np.zeros_like(a)
    idx = np.arange(len(a))
    j = kmax - idx
    mask = (j>=0) & (j<len(a))
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    total = contrib.sum()  # equals b[kmax]
    # compute reduction factor
    alpha = 0.1
    reduction = alpha * contrib / total  # fraction
    a_new = a * (1 - reduction)
    # ensure non-negative
    a_new = np.maximum(a_new, 0)
    # renorm
    a_new = a_new / a_new.sum()
    return a_new
```

"""
Propose a sequence that (hopefully) beats the uniform bound 2.0.

Algorithm
---------
* The evaluation 2 · n · max(conv(a,a)) / (sum(a)²) is homogeneous:
  scaling of the vector does not change its value.  We therefore
  keep the sum of the coefficients equal to 1 during the whole search.
* Starting from a random point on the simplex we repeatedly
  identify the index `k_max` where the convolution `a*a` attains its
  maximum.  The contribution of the i‑th coefficient to this peak is
  `c_i = a_i * a_{k_max-i}` (zero when the index is out of range).
* To lower the peak we shrink the coefficients proportionally to their
  contribution and then renormalise – this “pushes mass away’’ from the
  part of the sequence that currently dominates the autocorrelation.
* The update is embedded in a simple simulated‑annealing loop:
  - always accept a move that improves the score,
  - otherwise accept with a probability that decays with a temperature.
  - Occasionally add a tiny random perturbation to avoid getting stuck.
* Several restarts with different lengths (`n` from 30 to 120) are tried
  within the allotted time.  The best sequence found over all runs is
  returned.

The method is cheap (O(n²) per iteration) for the modest sizes we use
and works without any external optimisation libraries.
"""

import time
import random
import math
import numpy as np

def _conv_max(a: np.ndarray) -> float:
    """Return the maximal value of the self‑convolution of a."""
    b = np.convolve(a, a)
    return float(b.max())

def _score(a: np.ndarray) -> float:
    """Exact objective value used by evaluate_sequence (sum==1)."""
    n = a.shape[0]
    max_b = _conv_max(a)
    # sum(a) == 1  → denominator is 1
    return 2.0 * n * max_b

def _improve_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One deterministic improvement step.
    Reduce coefficients proportionally to their contribution to the current
    maximal convolution entry.
    """
    n = a.shape[0]
    # convolution and index of the maximal entry
    b = np.convolve(a, a)
    kmax = int(b.argmax())
    # contribution of each coefficient to b[kmax]
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    max_b = float(b.max())
    if max_b == 0.0:        # degenerate case – re‑randomise
        return np.random.exponential(size=n)

    # shrink proportionally to the contribution
    factor = 1.0 - eta * (contrib / max_b)
    a_new = a * factor
    a_new = np.maximum(a_new, 0.0)

    # renormalise to keep the sum equal to 1 (the objective is scale‑invariant)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.random.exponential(size=n)
    else:
        a_new /= s
    return a_new


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **_kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence found.
    """

    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_val = float("inf")

    # Helper to update the current best
    def _maybe_update(cur, a):
        nonlocal best_seq, best_val
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New best: {best_val:.6f}")

    # ------- main search loop -------------------------------------------------
    while time.time() < deadline:
        # Randomly pick a length for this restart (small enough for fast eval)
        n = random.randint(30, 120)

        # Initialise a random point on the simplex (Dirichlet via exponential)
        a = np.random.exponential(scale=1.0, size=n).astype(np.float64)
        a /= a.sum()

        cur_val = _score(a)
        _maybe_update(cur_val, a)

        # Simulated‑annealing parameters
        eta = 0.6               # initial step size for deterministic shrinkage
        temp = 0.3              # temperature for occasional uphill moves
        decay_eta = 0.9995
        decay_temp = 0.9998
        min_eta = 5e-4
        min_temp = 5e-4
        it = 0
        max_iter = 20000        # a generous per‑restart budget

        while it < max_iter and time.time() < deadline:
            # Deterministic improvement step
            a_new = _improve_step(a, eta)

            # Small random perturbation (helps escape plateaus)
            if random.random() < 0.01:
                perturb = np.random.normal(scale=eta * 0.05, size=n)
                a_new += perturb
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s == 0.0:
                    a_new = np.random.exponential(size=n)
                else:
                    a_new /= s

            new_val = _score(a_new)

            # Acceptance rule (SA style)
            if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
                a = a_new
                cur_val = new_val
                _maybe_update(cur_val, a)

            # Decay schedules
            eta = max(min_eta, eta * decay_eta)
            temp = max(min_temp, temp * decay_temp)

            it += 1

        # Occasionally start a completely fresh random walk
        if random.random() < 0.02:
            continue   # the outer while will start a new length

    # Fallback – should never happen, but ensures a valid return value
    if best_seq is None:
        best_seq = np.ones(100, dtype=np.float64) / 100.0

    # Convert to a plain Python list of floats for the caller
    return best_seq.tolist()
