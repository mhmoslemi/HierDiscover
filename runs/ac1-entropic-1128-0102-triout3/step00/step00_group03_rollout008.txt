def _local_search(a_initial, deadline):

    a = a_initial.copy()
    n = len(a)
    b = np.convolve(a, a)
    best_a = a.copy()
    best_score = 2 * n * np.max(b)
    curr_score = best_score
    step_factor = 0.2
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # find peak
        k_max = int(np.argmax(b))
        # source index with biggest contribution
        i_low = max(0, k_max - (n-1))
        i_high = min(n-1, k_max)
        idx_range = np.arange(i_low, i_high+1)
        contrib = a[idx_range] * a[k_max - idx_range]
        # If all zero (unlikely) pick random source
        source_i = idx_range[np.argmax(contrib)]

        threshold = np.percentile(a, 20)

            target_j = np.random.randint(n)
        else:
            target_j = np.random.choice(low_idxs)

"""
Propose a candidate step‑function that minimizes the evaluation score

    score = 2 * n * max(convolve(a, a)) / (sum(a)**2)

The algorithm is a simple greedy/annealing search that tries to flatten the
autocorrelation (the convolution of the sequence with itself).  Starting from
a reasonable seed (the provided `height_sequence_1`, a uniform sequence, or a
random one) it repeatedly:

1.   Finds the index `k_max` where the convolution attains its maximum.
2.   Chooses a source index `i` that contributes most to this peak
     (maximising a[i] * a[k_max-i]).
3.   Chooses a target index `j` that currently carries little mass
     (below the 20‑th percentile of the values).
4.   Transfers a small amount `eps` from `i` to `j`.  The convolution is updated
     incrementally (O(n) work) using the exact formula for the effect of a
     change in a single coefficient, including the second‑order `eps**2`
     terms.  If the move lowers the score it is kept, otherwise it is rejected.
5.   Every few thousand accepted moves the step size is reduced and the full
     convolution is recomputed to avoid numerical drift.

The search runs until the allocated time budget (default 1000 s) expires and
returns the best sequence found.  All numbers are kept non‑negative and the
sum is normalised to 1, which does not affect the objective because it is
scale‑invariant.
"""

import time
import random
import math
import numpy as np

def _incremental_update(b, a, src, tgt, eps):
    """
    Return a copy of `b` after moving `eps` mass from index `src` to `tgt`
    (both in the array `a`).  The update follows the exact algebraic change
    of the convolution, i.e. includes the quadratic `eps**2` terms.
    """
    n = len(a)
    b_new = b.copy()

    # Linear part: -2*eps*a for the source, +2*eps*a for the target
    b_new[src:src + n] -= 2.0 * eps * a
    b_new[tgt:tgt + n] += 2.0 * eps * a

    # Quadratic corrections
    k2s = 2 * src
    if k2s < len(b_new):
        b_new[k2s] += eps * eps
    k2t = 2 * tgt
    if k2t < len(b_new):
        b_new[k2t] += eps * eps
    kst = src + tgt
    if kst < len(b_new):
        b_new[kst] -= 2.0 * eps * eps

    return b_new


def _local_search(start_a, deadline):
    """
    Perform the greedy/annealing search starting from `start_a`
    (a non‑negative 1‑D NumPy array whose sum is 1).  The function stops
    when the wall‑clock time reaches `deadline` and returns the best
    sequence found together with its score.
    """
    a = start_a.copy()
    n = len(a)

    # Initial convolution and score
    b = np.convolve(a, a)
    best_a = a.copy()
    best_score = 2.0 * n * np.max(b)
    cur_score = best_score

    # Hyper‑parameters
    step_factor = 0.20          # maximal fraction of a[src] that can be moved
    min_step_factor = 1e-6
    iter_cnt = 0
    recompute_interval = 2000   # recompute full convolution every N accepted moves

    while time.time() < deadline:
        iter_cnt += 1

        # 1. locate the current peak of the convolution
        k_max = int(np.argmax(b))

        # 2. pick a source index that contributes most to this peak
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_range = np.arange(i_low, i_high + 1, dtype=int)
        contrib = a[idx_range] * a[k_max - idx_range]
        src = idx_range[np.argmax(contrib)]

        # 3. pick a target index that has a relatively small coefficient
        threshold = np.percentile(a, 20)          # 20‑th percentile
        low_candidates = np.where(a <= threshold)[0]
        if low_candidates.size == 0:
            tgt = random.randrange(n)
        else:
            tgt = random.choice(low_candidates)
        if tgt == src:                           # ensure different positions
            tgt = (tgt + 1) % n

        # 4. decide how much mass to move
        max_transfer = a[src]
        if max_transfer <= 0.0:
            continue
        eps = max_transfer * step_factor * random.random()
        if eps <= 0.0:
            continue

        # 5. compute the new convolution quickly
        b_new = _incremental_update(b, a, src, tgt, eps)

        new_score = 2.0 * n * np.max(b_new)

        # 6. accept the move only if it improves the objective
        if new_score < cur_score - 1e-14:
            # commit the change
            a[src] -= eps
            a[tgt] += eps
            b = b_new
            cur_score = new_score

            # record best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] iter {iter_cnt}: new best score = {best_score:.6f}")

            # periodic housekeeping
            if iter_cnt % recompute_interval == 0:
                b = np.convolve(a, a)          # avoid drift from ignored higher‑order terms
                cur_score = 2.0 * n * np.max(b)

                # slowly shrink the step size
                step_factor = max(min_step_factor, step_factor * 0.90)

        # optional: allow tiny uphill moves (simulated annealing)
        else:
            # simple temperature schedule based on iteration count
            temperature = max(0.0, (50000 - iter_cnt) / 50000.0)
            if temperature > 0.0 and random.random() < math.exp(-(new_score - cur_score) / temperature):
                a[src] -= eps
                a[tgt] += eps
                b = b_new
                cur_score = new_score

        # early exit if we have already reached the target bound
        if best_score <= 1.503:  # target from the statement
            break

    # final sanity recompute
    final_score = 2.0 * n * np.max(np.convolve(best_a, best_a))
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Entry point required by the evaluator.  Returns a list of non‑negative
    heights that (hopefully) gives a low evaluation score.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # Determine a suitable length `n`.  If the global `height_sequence_1`
    # exists we reuse its length, otherwise we fall back to the default used
    # in the problem statement (3401).
    try:
        base = np.array(height_sequence_1, dtype=float)
        n = len(base)
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base = np.full(n, 1.0 / n)
        else:
            base = base / base.sum()
        seeds = [base]
    except NameError:
        n = 3401
        seeds = []

    # add a uniform start and a random start
    seeds.append(np.full(n, 1.0 / n))
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    seeds.append(rnd)

    best_overall_a = None
    best_overall_score = float('inf')

    # Run a separate local search for each seed, sharing the global time budget
    for start_a in seeds:
        # make sure we still have time left
        if time.time() >= deadline:
            break
        a, score = _local_search(start_a, deadline)
        if score < best_overall_score:
            best_overall_score = score
            best_overall_a = a
            print(f"--- New global best score = {best_overall_score:.6f}")

    # Fall back to a uniform sequence if something went wrong
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # Convert to plain Python list as required by the grader
    return best_overall_a.tolist()
