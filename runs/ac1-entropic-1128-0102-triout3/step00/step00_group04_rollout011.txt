def simplex_projection(v):
    """Project v onto simplex {x>=0, sum x = 1}."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
