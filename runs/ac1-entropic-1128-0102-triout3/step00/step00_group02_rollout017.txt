def project_simplex(v):
    """Projection of vector v onto the probability simplex (sum=1, nonnegative)."""
    # Using algorithm from Chen & Ye.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
