def propose_candidate(budget_s=1000, seed=42):
    np.random.seed(seed)
    start_time = time.time()
    best_seq = None
    best_score = np.inf
    # Define some candidate lengths; we can try small to medium: between 500 and 3000 step 500.
    lengths = [500, 1000, 1500, 2000, 2500, 3000]
    betas = [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]
    for n in lengths:
        for beta in betas:
            # compute w_i
            i = np.arange(n, dtype=float)
            w = ( (i + 0.5) * (n - 0.5 - i) ) ** (-beta)
            # Avoid overflow at boundaries:
            w[ (i == 0) | (i == n-1) ] = (0.5 * (n - 0.5)) ** (-beta) # but formula works.
            sum_w = w.sum()
            scaling = np.sqrt(2 * n) / sum_w
            seq = scaling * w
            # Clip to allowed range
            seq = np.clip(seq, 0, 1000.0)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best: n={n}, beta={beta}, score={score}")
            # short check time
            if time.time() - start_time > budget_s - 5:
                break
        if time.time() - start_time > budget_s - 5:
            break

    # Optional refine via simple local gradient descent:
    # Start from best_seq and perform simulated annealing for remainder of time.
    while time.time() - start_time < budget_s - 2:
        # Choose random index to modify slightly
        seq = best_seq.copy()
        idx = np.random.randint(len(seq))
        delta = np.random.randn() * 0.05 * seq[idx]  # small relative change
        seq[idx] = max(0.0, seq[idx] + delta)
        # re-normalize to maintain sum sqrt(2n)
        sum_seq = sum(seq)
        target_sum = np.sqrt(2 * len(seq))
        factor = target_sum / sum_seq if sum_seq > 0 else 0
        seq = [x * factor for x in seq]
        # Clip
        seq = [min(1000.0, max(0.0, x)) for x in seq]

        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Improved via perturbation: score={score}")
    return best_seq
