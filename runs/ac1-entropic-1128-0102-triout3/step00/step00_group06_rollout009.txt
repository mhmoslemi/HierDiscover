Implementation:

def project_to_simplex(v, s):
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - s
   rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
   theta = cssv[rho] / (rho+1)
   w = np.maximum(v - theta, 0)
   return w

Return w.

def improve_sequence(a, budget_s=...):
    start_time = time.time()
    deadline = start_time + budget_s - 5
    best_a = a.copy()
    best_score = evaluate_sequence(best_a)
    n = len(a)
    sqrt_total = math.sqrt(2*n)
    # Ensure sum normalized
    a = np.array(a, dtype=float)
    # Normalize onto simplex
    a = project_to_simplex(a, sqrt_total)

    step = sqrt_total * 0.1  # initial step size maybe relative?

    while time.time() < deadline:
        b = np.convolve(a, a)
        max_b = b.max()
        max_idx = b.argmax()
        # gradient
        g = np.zeros_like(a)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                if i == j:
                    g[i] = a[i]
                else:
                    g[i] = 2 * a[j]
        # gradient descent step
        # compute step size: we need small enough to maintain positivity after projection.
        eta = step / (np.linalg.norm(g) + 1e-12)  # scaling
        a_new = a - eta * g
        # Ensure nonnegatives after projection
        a_new = project_to_simplex(a_new, sqrt_total)
        # Evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # also print progress

def project_simplex(v, s=1):
    """Project onto simplex {x >= 0, sum x = s}."""
    n = len(v)
    if s < 0:
        raise ValueError
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        iter += 1
    return best_local

"""
Propose a better step–function sequence.

The evaluation metric simplifies (after normalising the L1‑norm) to the
maximum entry of the self‑convolution `a*a`.  The goal is therefore to
make that peak as low as possible.

We use a simple projected sub‑gradient descent on the function
    f(a) = max_k (a*a)[k]

starting from several diverse initialisations (random Dirichlet,
linearly decreasing, the historic sequence, …).  In each iteration we
  * compute the convolution (FFT based, O(n log n)),
  * locate the index(es) where the maximum is attained,
  * build a sub‑gradient of the maximum (for a fixed `k`,
    ∂(a*a)[k]/∂a_i = a_{k-i} + a_{k-i}  (i≠k-i)  or a_i  (i=k-i)),
  * take a small step opposite to the sub‑gradient,
  * re‑project onto the simplex {a_i≥0, Σa_i = √(2n)}.

Occasionally a tiny random jitter is added to escape flat plateaus.
The best sequence found over all restarts (and a final refinement) is
returned.  The routine respects the 1000 s time limit and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x≥0 , Σx = s}
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution using FFT (real valued)
def _convolve_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient step for the current sequence
def _subgradient_step(a: np.ndarray, target_sum: float, step: float):
    """One projected sub‑gradient update; returns new a and the current max."""
    n = a.shape[0]
    conv = _convolve_self(a)
    max_val = conv.max()
    # indices where the maximum is attained (tolerance to handle ties)
    active = np.where(np.abs(conv - max_val) < 1e-12)[0]

    # build sub‑gradient (average of gradients of all active ks)
    g = np.zeros_like(a)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if i == j:
                g[i] += a[i]                 # derivative of a_i * a_i
            else:
                g[i] += 2.0 * a[j]           # a_i a_j appears twice (i,j) & (j,i)
    g /= float(len(active))

    # gradient descent step and projection onto the feasible simplex
    a_new = a - step * g
    a_new = _project_simplex(a_new, target_sum)
    return a_new, max_val


# ----------------------------------------------------------------------
# Local optimisation starting from a given vector
def _local_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected sub‑gradient descent with occasional jitter.
    Returns the best sequence found before ``deadline``.
    """
    a = _project_simplex(a0, target_sum)
    best_a = a.copy()
    # Since Σa = √(2n) we have (Σa)^2 = 2n, therefore the evaluation
    # score equals the maximal convolution entry.
    best_score = float(_convolve_self(a).max())

    # step size – start relatively aggressive, then adapt
    step = max(0.01, target_sum * 0.05)

    it = 0
    no_improve = 0
    while time.time() < deadline:
        a, cur_max = _subgradient_step(a, target_sum, step)

        # cheap local score (no clipping needed – values stay non‑negative)
        cur_score = cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            step *= 1.02                     # a tiny acceleration
            no_improve = 0
            print(f"[{time.time():.1f}]  iter {it:<6d}  new best {best_score:.6f}")
        else:
            step *= 0.98                     # shrink when no improvement
            no_improve += 1

        # every few hundred iterations add a tiny random jitter to escape
        # possible plateaus (especially important when we start from a uniform
        # distribution where the sub‑gradient is a constant vector).
        if it % 500 == 0 and it > 0:
            jitter = np.random.randn(a.shape[0]) * step * 0.1
            a = _project_simplex(a + jitter, target_sum)

        it += 1

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 950, **kwargs):

        n = 2807

    # ------------------------------------------------------------------

    if best_overall is None:

    # Return a plain Python list as required by the evaluation harness.
    print(f"\n=== Finished. Best score = {best_score:.6f} ===")
    return best_overall.tolist()
