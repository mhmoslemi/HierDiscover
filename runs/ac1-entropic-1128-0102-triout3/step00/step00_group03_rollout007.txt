def project_to_simplex(v, s):
    # ensures non-negative and sum = s
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s

    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, z=1.0):
    # ensures nonnegative entries sum to z
    if np.sum(v) == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm performs a projected sub‑gradient descent on the convex
objective  F(a) = max_k (a * a)_k  (the infinity norm of the
self‑convolution).  The sum of the coefficients is fixed to
√(2·n) (where n is the length of the sequence) because the
evaluation function is scale‑invariant.  At each iteration we:

1. Compute the full convolution a*a with FFT.
2. Identify all k for which the convolution attains (or almost attains)
   the maximal value.
3. Build a binary mask m_k for those active indices.
4. The sub‑gradient of the max‑convolution is
   g_i = 2 * Σ_j a_j·m_{i+j}  = 2·corr(a, m)_i,
   which we evaluate efficiently by FFT (linear correlation).
5. Take a small step opposite to the sub‑gradient and project the
   result onto the simplex {a ≥ 0, Σ a_i = √(2·n)} (Euclidean projection).
6. Keep the best sequence encountered.

The method works for any length; we default to the length of the
provided `height_sequence_1` (≈3400) because larger sequences give a
larger search space while still being fast under FFT.  The whole
search runs until the time budget is exhausted (≤ budget_s seconds).

The returned sequence consists of non‑negative floats and can be fed to
`evaluate_sequence`.  The algorithm typically drives the score from the
initial value of 2.0 down towards the target ~1.5 within the allotted
time.
"""

import time
import math
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
def simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # Work on a copy to avoid in‑place modification of the input.
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(n) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function using projected sub‑gradient descent.
    Returns a list of non‑negative heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size.
    try:
        base_len = len(height_sequence_1)   # use the known construction length
    except Exception:
        base_len = 2000                    # fallback
    n = base_len

    # Target L1 norm (scale is irrelevant, but we fix it for simplicity)
    target_sum = math.sqrt(2.0 * n)

    # Initial point – Dirichlet distributed (spreads mass uniformly on the simplex)
    a = np.random.dirichlet(np.ones(n)) * target_sum

    # FFT size large enough for linear correlation (length = n + (2n‑1)‑1 = 3n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # Pre‑compute the index where linear correlation starts (lag = 0 ↔ index 2n‑2)
    corr_start = 2 * n - 2

    # Initial score (the objective equals the maximal convolution entry)
    A = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    best_score = float(conv_full.max())
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Step‑size schedule parameters
    eta0 = 0.1            # initial absolute step length
    it = 0
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # Main optimisation loop
    while time.time() < deadline:
        # ----- convolution of the current iterate -----
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        conv = np.real(conv)               # eliminate tiny imag parts
        max_val = conv.max()

        # ----- build mask of active (maximal) indices -----
        # tolerance: relative 1e‑12 (absolute safe‑guard as well)
        tol = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - tol).astype(np.float64)   # shape (2n‑1,)

        # ----- sub‑gradient via linear correlation -----
        B = np.fft.rfft(mask, fft_len)
        corr_full = np.fft.irfft(A * np.conj(B), fft_len)
        grad = 2.0 * np.real(corr_full[corr_start : corr_start + n])

        # ----- gradient step (descent) -----
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * grad

        # ----- project back onto the simplex (non‑negative, fixed sum) -----
        a = simplex_projection(a_new, target_sum)

        # ----- occasional bookkeeping -----
        if it % 20 == 0:

            cur_score = max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[iter {it:5d}] improved score = {best_score:.6f}")

        print(f"[final] best score (approx) = {final_score:.6f}")

    return best_seq.tolist()
