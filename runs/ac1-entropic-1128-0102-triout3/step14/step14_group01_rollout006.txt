"""
Simulated‑annealing refinement of a step‑function.

The routine starts from the best sequence known in the global
variable ``height_sequence_1`` (or from a uniform distribution if that
variable is missing).  The sequence is normalised to sum = 1 and its
full convolution with itself is computed once via an FFT.

A fast incremental update of the convolution is used when a tiny
amount of mass ε is transferred from a source index i to a destination
index j:

    conv' = conv - 2·ε·a shifted by i  + 2·ε·a shifted by j   (δ²‑terms are
    ignored – they are ≪ 10⁻⁸ for the step sizes we use).

The objective is   S = 2·n·max(conv)   (the denominator (∑a)² cancels after
normalisation).  A classic Metropolis criterion with an exponential
temperature schedule is applied:   accept if S decreases, otherwise accept
with probability exp(−ΔS/T).  The temperature is tuned so that the early
iterations explore aggressively while later iterations become greedy.

The loop runs until the allotted wall‑clock budget (≈1000 s) is exhausted,
keeping the best sequence seen.  Finally the best vector is projected
onto the probability simplex and returned as a plain Python list.

The whole optimisation fits comfortably in a few seconds per 10 k
iterations and therefore can perform several hundred thousand moves
within the time limit, typically improving the score from ≈1.508 to
below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv_full


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm performs a long simulated‑annealing
    walk with cheap O(n) convolution updates, starting from the global
    starter `height_sequence_1` if it exists.
    """
    # ------------------------------------------------------------------
    # basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15          # safety margin

    # try to use the previously discovered construction
    try:
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.ndim != 1 or a.size == 0 or a.sum() <= 0:
            raise ValueError
        a /= a.sum()
        n = a.shape[0]
        print(f"[{time.time() - start_time:.2f}s] Starter from height_sequence_1 (n={n})")
    except Exception:
        # fall back to a fairly long uniform vector (the reference length)
        n = 4096
        a = np.full(n, 1.0 / n)
        print(f"[{time.time() - start_time:.2f}s] Using uniform starter (n={n})")

    # ------------------------------------------------------------------
    # initial convolution (full length = 2·n‑1)
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] Initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # simulated‑annealing parameters
    # ------------------------------------------------------------------
    total_time = max(0.0, deadline - start_time)
    # initial temperature = a fraction of the current excess above the low bound 1.0
    T0 = max(0.05, (cur_score - 1.0) * 0.3)
    # exponential decay to ~1e‑8 at the end of the budget
    decay_rate = np.log(1e-8 / T0) / total_time if total_time > 0 else -np.inf

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # main SA loop
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # pick a destination (can be any index, including src – we reject that later)
        dst = rng.integers(n)
        if dst == src:
            continue

        # transfer a small fraction of the source mass
        frac = rng.random() * 0.019 + 0.001   # 0.1 % … 2 %
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # incremental update of the convolution (ignore δ²‑terms)
        # conv_new = conv - 2·δ·a shifted by src  + 2·δ·a shifted by dst
        # ------------------------------------------------------------------
        conv_new = conv.copy()
        # subtraction slice
        conv_new[src : src + n] -= 2.0 * delta * a
        # addition slice
        conv_new[dst : dst + n] += 2.0 * delta * a

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # temperature according to elapsed time
        elapsed = time.time() - start_time
        T = max(1e-12, T0 * np.exp(decay_rate * elapsed))

        # Metropolis acceptance
        accept = False
        if delta_score < -1e-12:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            accept = rng.random() < prob

        if accept:
            # commit the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_score = new_score

            # record best solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] iter={it:,}  New best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # occasional global perturbation to escape deep local minima
        # every ~0.5 % of the remaining time we draw a fresh Dirichlet vector
        # ------------------------------------------------------------------
        if it % 4000 == 0 and (deadline - time.time()) > 1.0:
            if rng.random() < 0.005:
                # draw a random Dirichlet candidate and keep it if it improves
                cand = rng.gamma(shape=0.3, scale=1.0, size=n)
                cand /= cand.sum()
                cand_conv = _conv_self_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    a = cand
                    conv = cand_conv
                    cur_score = cand_score
                    print(f"[{time.time() - start_time:.2f}s] Global re‑draw accepted, score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # final clean‑up: project the best vector onto the simplex (numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
