"""
Improved step‑function optimiser for the max‑convolution challenge.

Strategy
---------
1. Initialise with a strong candidate (global starter if present, otherwise
   the uniform distribution).
2. Exhaustively sweep several parametric families (geometric, power,
   beta, stretched‑exponential, logistic) with finer grids than before.
3. Generate many random Dirichlet draws with a broad range of concentration
   parameters.
4. For every promising candidate run a short Adam‑optimisation of the
   smooth‑max surrogate (temperature schedule 1 → 1e6).  The best result
   of all runs is kept.
5. Refine the current best sequence by solving a linear programme:
   maximise Σg_j subject to   (f * g)_k ≤ M   (where M = max(f * f))
   and g≥0.  This moves mass away from positions that dominate the
   convolution.  After scaling g back to unit sum we mix it with f.
   A few such LP‑updates dramatically lower the max‑convolution.
6. Finish with an aggressive hill‑climb / simulated‑annealing phase that
   transfers tiny amounts of mass from the most “harmful’’ indices to the
   most “helpful’’ ones.
7. Return the best sequence found (normalised to sum = 1).

The implementation uses FFT‑based convolutions (O(n log n)), a fast simplex
projection, and SciPy’s HiGHS LP solver.  All stages respect the 1000 s
budget and print intermediate scores for monitoring.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑v=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)_k .
    For indices attaining the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during search: 2·n·max(conv(seq,seq)) (∑seq=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # --- uniform baseline ------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # --------------------------------------------------- geometric ----------
    eps_grid = np.logspace(-7, -1, 300)                # finer at the low end
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------- power ----------
    p_grid = np.linspace(-3.0, 3.0, 151)

        s = pow_seq.sum()
        if s == 0:

        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # --------------------------------------------------- beta ----------
    alphas = np.linspace(0.05, 10.0, 81)   # extended range
    betas = np.linspace(0.05, 10.0, 81)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # -------------------------------------------- stretched‑exponential
    scales = np.linspace(0.2, 1.0, 9)
    p_vals = np.linspace(0.1, 6.0, 51)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    # --------------------------------------------------- logistic ----------
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.005, 0.25, 13)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.3f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (wide concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 800
    log_c_low, log_c_high = np.log(0.01), np.log(20.0)
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(log_c_low, log_c_high)
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.06
    tau = 1.0               # start with a smooth approximation
    tau_factor = 2.0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)               # length = 3·n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature gradually
        if it % 1500 == 0:
            tau = min(tau * tau_factor, 1e8)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Linear‑programming refinement (convex‑but‑linear sub‑problem)
# ----------------------------------------------------------------------
def _build_conv_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """Build the matrix A where (A·g)_k = (f * g)_k (full convolution)."""
    n = f.shape[0]
    # rows = i + j, cols = j
    row = np.add.outer(np.arange(n), np.arange(n)).ravel()
    col = np.repeat(np.arange(n), n)
    data = np.tile(f, n)               # repeat f for each column
    M = 2 * n - 1
    return sparse.csr_matrix((data, (row, col)), shape=(M, n))


def _lp_max_sum_under_M(f: np.ndarray, Mbound: float) -> np.ndarray | None:
    """Solve  max Σ g_j   s.t.  f∗g ≤ Mbound, g≥0.  Returns the optimal g or None."""
    n = f.shape[0]
    A = _build_conv_constraint_matrix(f)
    b = np.full(A.shape[0], Mbound, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σg  ⇔ minimise −Σg
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x


def _lp_refinement(f: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Apply a few LP‑updates to improve the current best sequence."""
    a = f.copy()
    n = a.shape[0]
    while time.time() < deadline - 5.0:          # keep a small safety margin
        conv = _conv_self(a)
        M = conv.max()
        g = _lp_max_sum_under_M(a, M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                     # renormalise to unit sum
        # aggressive mixing – a full move often reduces the bound
        t = 0.7
        new_a = (1.0 - t) * a + t * g
        new_conv = _conv_self(new_a)
        new_M = new_conv.max()
        new_score = 2.0 * n * new_M
        if new_score < best_score - 1e-12:
            a, best_score, best_seq = new_a, new_score, new_a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP refined score={new_score:.6f}")
        else:
            # no improvement – shrink the step size and try once more
            t = 0.3
            new_a = (1.0 - t) * a + t * g
            new_conv = _conv_self(new_a)
            new_M = new_conv.max()
            new_score = 2.0 * n * new_M
            if new_score < best_score - 1e-12:
                a, best_score, best_seq = new_a, new_score, new_a.copy()
                print(f"[{time.time() - start_time:.2f}s] LP refined (small step) score={new_score:.6f}")
            else:
                # cannot make progress – terminate LP phase
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule (log‑linear decay)
    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer a tiny fraction (up to 0.8 % of source mass)
        frac = rng.random() * 0.008 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine follows a multi‑stage pipeline:
    parametric sweep → random Dirichlet → Adam smoothing → LP refinement
    → final hill‑climb, respecting the given time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (keep the classic size of the reference run)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise (use a global starter if present)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (budget‑aware)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # reserve ~5 min for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation of smooth‑max
    # --------------------------------------------------------------
    hill_time = 120.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Linear‑programming refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:            # leave ~1 min for final climb
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        best_seq, best_score = _lp_refinement(
            best_seq, start_time, deadline - 60.0, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After LP best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final directed hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise to sum = 1 and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
