"""
Hybrid optimisation for the max‑auto‑correlation step‑function problem.

The search consists of three stages:
1️⃣  Fast parametric sweeps (geometric / power / full (α,β)‑Beta families) to obtain
    a decent starting vector.
2️⃣  A long “hybrid” descent on the probability simplex:
        – the objective is a convex combination of
          • the true sub‑gradient of max‑conv (λ≈1 at the end)
          • the L2‑gradient of the deviation from a perfectly flat convolution
        – the vector is represented as a soft‑max of unconstrained logits φ,
          and Adam is applied to φ.
        – λ grows linearly from 0 to 1 as time passes, letting the method first
          flatten the whole convolution and later focus on the peak.
3️⃣  A final directed hill‑climb that moves tiny chunks of mass from the
    index with the largest positive sub‑gradient to the index with the most
    negative one (with a simple line‑search on the transferred amount).

All steps respect the 1000 s time budget, print intermediate scores and finally
return a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax(phi: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector (sum = 1)."""
    phi_max = np.max(phi)
    ex = np.exp(phi - phi_max)
    return ex / ex.sum()


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _grad_l2(a: np.ndarray, conv: np.ndarray, uniform_target: np.ndarray) -> np.ndarray:
    """
    Gradient of the L2 loss ‖conv(a,a) – uniform‖².
    Returned gradient has the same shape as a.
    """
    n = a.shape[0]
    err = conv - uniform_target
    a_rev = a[::-1]
    # full convolution of err (length 2n‑1) with a_rev (length n)
    tmp = _conv_fft(err, a_rev)                # length = 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]        # extract the central n entries
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (very cheap, give a decent start)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 120)       # fewer points than before → faster
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    """Full (α,β) grid – a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    alphas = np.linspace(0.2, 5.0, 26)   # 26×26 ≈ 676 combos → still cheap
    betas = alphas

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Hybrid gradient descent (Adam on logits + L2 flattening)
# ----------------------------------------------------------------------
def _hybrid_descent(start_a, deadline, best_seq, best_score):
    """
    Perform a long Adam descent on the logits φ such that a = softmax(φ).
    The loss is   λ·max(conv) + (1‑λ)·‖conv‑uniform‖² .
    λ grows linearly from 0 to 1 during the run.
    """
    n = start_a.shape[0]
    # initialise logits from the start vector
    phi = np.log(start_a + 1e-12)

    # Adam state
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.08

    m = np.zeros_like(phi)
    v = np.zeros_like(phi)

    # constant uniform target for the convolution
    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    it = 0
    total_allowed = deadline - start_time
    while time.time() < deadline:
        it += 1

        # ---- current point -------------------------------------------------
        a = _softmax(phi)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it:,} score={cur_score:.6f}")

        # ---- schedule λ (linear growth) ------------------------------------
        elapsed = time.time() - start_time
        lam = min(1.0, elapsed / total_allowed)   # λ∈[0,1]

        # ---- gradients ------------------------------------------------------
        a_rev = a[::-1]

        #   L2 part (flattening)
        grad_l2 = _grad_l2(a, conv, uniform_target)

        #   max‑conv part (exact sub‑gradient)
        grad_max = _subgradient_maxconv(a, conv)

        #   combine
        grad_a = lam * grad_max + (1.0 - lam) * grad_l2

        # ---- transform to logits ------------------------------------------------
        #   dL/dφ = a * (dL/da – ⟨a, dL/da⟩)
        inner = np.dot(a, grad_a)
        grad_phi = a * (grad_a - inner)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_phi
        v = beta2 * v + (1.0 - beta2) * (grad_phi ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        phi -= step

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb (a slightly more aggressive version)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # random perturbation when nothing obvious
            i1, i2 = rng.choice(n, size=2, replace=False)
            if a[i1] > 1e-15:
                src, dst = i1, i2
            else:
                continue

        # try a few candidate transfers (log‑space)
        max_transfer = a[src]
        epsilons = max_transfer * np.array([0.5, 0.2, 0.05, 0.01, 0.001])
        improved = False
        for eps in epsilons:
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            cur_score = float(2.0 * n * _conv_self(a_prop).max())
            if cur_score < best_score:
                a = a_prop
                best_a = a.copy()
                best_score = cur_score
                no_improve = 0
                improved = True
                print(f"[{time.time() - start_time:.2f}s] Hill climb score={best_score:.6f}")
                break
        if not improved:
            no_improve += 1
            if no_improve > 4000:
                # occasional random reshuffle to escape flat plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps = a[i1] * (rng.random() * 0.01 + 1e-6)
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising
        2·n·max(conv(a,a)) / (sum(a)²)

    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25      # safety margin

    # ------------------------------------------------------------------
    # Problem size (keep n moderate – 4096 works well with the time budget)
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if it exists use it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈20 % of total time)
    # ------------------------------------------------------------------
    phase_deadline = start_time + min(200.0, budget_s * 0.2)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Hybrid descent (the heavyweight part – ≈70 % of budget)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 200.0            # keep ~200 s for final hill‑climb
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hybrid Adam‑descent")
        best_seq, best_score = _hybrid_descent(
            best_seq, smooth_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final directed hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
