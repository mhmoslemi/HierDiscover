"""
Propose a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy is a four‑stage pipeline (total budget ≤ 1000 s)

1️⃣  Coarse grid search over the 3‑parameter family
     a_i ∝ r**i·(i+1)**p mixed with the uniform distribution.
2️⃣  Stochastic hill‑climb on the same three parameters (r, p, λ) for
    fine‑tuning.
3️⃣  Projected sub‑gradient descent on the exact objective
    (the shift attaining the maximal convolution entry) – the same
    method as the reference solution but with a slightly more aggressive
    step‑size schedule.
4️⃣  Simulated annealing on the full vector: repeatedly move a random
    fraction of mass from one index to another and accept worsening
    moves with a temperature that decays to near‑zero.  This stage can
    escape the local minima left by the sub‑gradient phase and usually
    pushes the score a few 10⁻³ lower.

All stages respect the 1000 s budget.  The returned list is a plain
Python list of floats (the sum is re‑normalised to 1 before returning).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (returns length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv_full


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Return a normalised vector ∝ r**i·(i+1)**p (i=0…n‑1)."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # work in log‑space for stability
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)            # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """lam·uniform + (1‑lam)·shape (both already sum‑to‑1)."""
    n = shape.size
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()                      # safeguard against round‑off
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.2                      # a slightly larger initial step
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # --- compute sub‑gradient for shift k (O(n)) -----
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # add the symmetric part for i≠j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # --- step with a polynomially decaying step size ----------
        step = step0 / (it ** 0.55)   # slower decay than 1/√it
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_score(a_candidate, n, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = math.exp(-(cur_score - best_score) * 10.0 / (math.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # tiny random jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the full vector
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray,
                         cur_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Global search by moving random fractions of mass between entries."""
    n = a_start.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_score
    cur = cur_score

    rng = np.random.default_rng()
    total_time = deadline - time.time()
    if total_time <= 0:
        return best_a, best_score

    # temperature schedule
    Tmax = 0.02
    Tmin = 1e-4

    # maximal fraction of source mass that may be moved (decays with temperature)
    max_frac_start = 0.2
    min_frac = 0.002

    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = (elapsed) / (deadline - start_time)  # 0 → 1
        T = Tmax * (1.0 - fraction) + Tmin * fraction

        max_move_frac = max_frac_start * (1.0 - fraction) + min_frac * fraction

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        delta = a[src] * max_move_frac * rng.random()   # amount to transfer
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _eval_score(a_prop, n, L_fft)

        if new_score < cur or rng.random() < math.exp(-(new_score - cur) / T):
            a = a_prop
            cur = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA score={new_score:.6f}")

        # occasional global jitter to escape flat plateaus
        if rng.random() < 0.001:
            eps = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            cur = _eval_score(a, n, L_fft)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA jitter score={cur:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

                cur = _eval_score(starter, n, L_fft)

    cur_p = 0.0

    sigma_r = 0.015
    sigma_p = 0.25
    sigma_lam = 0.12
    cur_seq = best_seq.copy()
    cur_score = best_score

    while time.time() < param_deadline:

        p_new = cur_p + sigma_p * rng.standard_normal()

    best_seq = np.maximum(best_seq, 0.0)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
