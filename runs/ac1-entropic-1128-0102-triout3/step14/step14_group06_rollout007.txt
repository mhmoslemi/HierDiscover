def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose large N based on budget
    total_time = deadline - start_time
    if total_time > 800:
        N = 65536
    elif total_time > 400:
        N = 32768
    else:
        N = 16384

    # Initial best: uniform
    best_vec = np.full(N, 1.0 / N)
    best_score = _objective(best_vec)
    print(f"initial uniform score={best_score}")

    # coarse search phase using small N

    # Determine N_small
    N_small = max(2048, min(8192, N // 8))
    # allocate time for coarse search: upto 25% of total
    coarse_deadline = start_time + total_time * 0.25
    while time.time() < coarse_deadline:
        # sample random shape generator
        shape_type = rng.choice(['beta', 'beta_asym', 'ramp', 'dirichlet'])
        if shape_type == 'beta':
            alpha = rng.uniform(0.2, 2.5)
            vec_small = _make_beta_shape(N_small, alpha)
        elif shape_type == 'beta_asym':
            a = rng.uniform(0.2, 2.5)
            b = rng.uniform(0.2, 2.5)
            vec_small = _make_beta_shape_asym(N_small, a, b)
        elif shape_type == 'ramp':
            p = rng.uniform(0.1, 5.0)
            plateau = rng.uniform(0.0, 0.5)
            vec_small = _make_shape(N_small, p, plateau)
        else: # dirichlet
            conc = rng.uniform(0.1, 1.0)
            vec_small = rng.gamma(conc, 1.0, size=N_small)
            s = vec_small.sum()
            if s > 0:
                vec_small /= s

        # upscale to N
        vec_large = _interpolate(vec_small, N)  # implement using np.interp
        # ensure normalized
        vec_large = np.maximum(vec_large, 0)
        s = vec_large.sum()
        if s > 0:
            vec_large /= s

        # run short exponentiated gradient descent on this candidate to refine quickly
        # use limited iteration count, e.g., 500 steps
        vec_refined, score_refined = _exp_subgradient_descent_limited(vec_large, limit_iters=800, start_time=start_time, deadline=coarse_deadline)
        # update best
        if score_refined < best_score:
            best_score = score_refined
            best_vec = vec_refined.copy()
            print(f"new best from coarse {score_refined:.6f}")

    # main descent phase (full)

"""
Step‑function optimiser – hierarchical search + long‑run exponentiated descent.

1) Choose a problem size N (larger N gives finer discretisation).
2) Run a fast “coarse” search on a much smaller grid N_small.
   – Randomly generate a large variety of parametric shapes
     (symmetric/asymmetric Beta, symmetric power‑ramps, monotone ramps,
     Dirichlet‑samples).
   – Upsample each shape to size N, normalise to sum = 1 and evaluate the
     exact objective 2·N·max(conv(a,a))/sum(a)².
   – Keep the few best candidates (top‑k pool).

3) Initialise the long‑run optimiser with the best coarse candidate.
   – Exponentiated sub‑gradient descent on the exact ℓ∞ objective.
     The active set (indices achieving the current maximum) provides a
     sub‑gradient; multiplicative updates keep the vector non‑negative.
   – Step size decays as η = η₀ / sqrt(t) to allow fine‑grained convergence.
   – The best vector seen during the run is stored.

4) If a few seconds remain, a deterministic hill‑climbing pass swaps a
   small amount of mass between two random positions, accepting only
   improvements (no temperature).

The final vector is projected onto the simplex, converted to a plain
Python list and returned.  All numbers are non‑negative; the optimiser
uses only NumPy and the FFT‑based convolution for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a size‑k pool sorted by increasing score."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


def _interpolate(vec_small: np.ndarray, N_target: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length N_target."""
    n_small = vec_small.shape[0]
    if n_small == N_target:
        return vec_small.copy()
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_target = np.linspace(0.0, 1.0, N_target, endpoint=False)
    return np.interp(x_target, x_small, vec_small)


# ----------------------------------------------------------------------
# Analytic shape generators (all produce a probability vector)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with optional flat plateau.
    Normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density, normalised to sum = 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_linear_ramp(n: int, p: float) -> np.ndarray:
    """Monotone increasing power‑law ramp, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** p
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _random_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random shape from a diverse family of parameterised distributions."""
    shape_type = rng.choice(['beta', 'beta_asym', 'ramp', 'lin', 'dirichlet'])
    if shape_type == 'beta':
        alpha = rng.uniform(0.2, 2.5)
        return _make_beta_shape(n, alpha)
    if shape_type == 'beta_asym':
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        return _make_beta_shape_asym(n, a, b)
    if shape_type == 'ramp':
        p = rng.uniform(0.1, 5.0)
        plateau = rng.uniform(0.0, 0.5)
        return _make_shape(n, p, plateau)
    if shape_type == 'lin':
        p = rng.uniform(0.1, 5.0)
        return _make_linear_ramp(n, p)
    # dirichlet‑like sample (sparse)
    conc = rng.uniform(0.05, 1.0)
    vec = rng.gamma(conc, 1.0, size=n)
    total = vec.sum()
    if total > 0:
        vec /= total
    else:
        vec = np.full(n, 1.0 / n)
    return vec

# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on the exact max‑convolution objective.
    Multiplicative updates keep the vector non‑negative; a final simplex
    projection preserves the probability‑vector constraint.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # initial step‑size
    eta0 = 0.6

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (all indices essentially achieving the maximum)
        tol = max(1e-14, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # uniform weight over active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)   (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size decreasing like 1/√t  (more stable for fine tuning)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip to avoid overflow/underflow)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


def _local_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a tiny amount of mass from a random
    source index to a random destination and keep the change only if the
    objective improves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _objective(a)
    best_a = a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # take a small fraction (≤1 %) of the source mass
        delta = a[src] * 0.01 * rng.random()
        if delta <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        cur_score = _objective(a_prop)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] HillClimb improved {cur_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Returns a plain Python list of non‑negative floats (the heights).
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger N gives a finer discretisation.
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # Coarse search on a small grid (≈20 % of total time)
    # ------------------------------------------------------------------
    frac_coarse = 0.20
    coarse_deadline = start_time + remaining * frac_coarse
    N_small = max(2048, min(8192, N // 8))

    # Top‑k pool for the coarse phase
    TOP_K = 5
    top_pool = []   # each entry: (score, vector)

    # always insert the uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(top_pool, _objective(uniform), uniform, TOP_K)

    # optionally add the globally supplied starter (if it exists)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # up‑sample / repeat to match required size
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    starter_up = _interpolate(starter, N)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _push_top(top_pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # Randomly generate many shapes on the small grid, evaluate after up‑sampling.
    while time.time() < coarse_deadline:
        vec_small = _random_shape(rng, N_small)
        vec_large = _interpolate(vec_small, N)
        vec_large = np.maximum(vec_large, 0.0)
        total = vec_large.sum()
        if total <= 0.0:
            continue
        vec_large /= total
        sc = _objective(vec_large)
        _push_top(top_pool, sc, vec_large, TOP_K)

    # Pick the best among the coarse pool as the warm‑start for the fine phase
    best_score, best_vec = top_pool[0][0], top_pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Coarse best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation – long‑run exponentiated sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        best_vec, best_score = _exp_subgradient_descent(best_vec, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After ExpGD best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic hill‑climbing polish (if a few seconds left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_vec, best_score = _local_hill_climb(best_vec, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] After HillClimb best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
