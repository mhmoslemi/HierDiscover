def _make_beta_shape(n, alpha, beta):
    i = np.arange(n) + 0.5
    x = i / n
    log_shape = (alpha-1)*np.log(x) + (beta-1)*np.log(1-x)
    # For numerical stability offset
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

"""
Improved optimizer for the step‑function evaluation problem.

The algorithm consists of four stages:

1️⃣  **Rich parametric sweep** – search a larger family of shapes:
    * Geometric × power (the previous baseline).
    * Symmetrised (reversed) version of the geometric shape.
    * Beta‑distribution shapes (α,β) on the unit interval.
    * All shapes are optionally mixed with the uniform distribution.
    The best candidate from this exhaustive sweep is kept as a warm‑start.

2️⃣  **Random Dirichlet seeds** – generate many random probability vectors
    (γ‑distributed entries) and keep a few of the best ones.  This adds
    diversity and helps escape the bias of the parametric families.

3️⃣  **Projected momentum‑gradient descent** – starting from every retained
    candidate we run a fast sub‑gradient ascent on the true objective.
    The gradient is taken with respect to the *currently maximal* convolution
    entry; a heavy‑ball momentum term and an Adam‑style adaptive step size
    accelerate convergence.  The loop runs until the allotted time expires,
    accepting occasional uphill moves (simulated‑annealing flavour) and
    injecting small random jitter every few thousand iterations.

4️⃣  **Fine stochastic hill‑climb** – a very cheap mass‑transfer move
    (move a tiny fraction of probability mass from a random source index
    to a random destination) is repeatedly applied.  Improvements are kept,
    and the loop finishes just before the deadline.

All stages respect the 1000 s budget; the final vector is projected onto the
probability simplex and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT (length 2·len(a)‑1).
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes Σa ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def _make_geo_power_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Geometric × power shape:
        a_i ∝ r**i * (i+1)**p    (i = 0 … n‑1)
    Returned vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)               # stabilise exponentials
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Beta‑distribution shape on the discrete grid {½/n, …, (n‑½)/n}:
        a_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1},
    where  x_i = (i+0.5)/n .
    Returned vector is normalised to sum = 1.
    """
    # centre points to avoid the singularities at 0 and 1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination lam·uniform + (1‑lam)·shape.
    Both operands already sum to 1, the result is re‑normalised for safety.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Projected momentum‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _momentum_descent(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Fast descent on 2·n·max(conv(a,a)).
    Uses a heavy‑ball momentum term and an Adam‑style adaptive step size.
    Returns the best vector encountered and its score.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam‑style accumulators
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = 0.30                  # base step before decay
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- compute maximal convolution entry ----------
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        # collect all (or most) shifts that achieve the max up to a tiny tolerance
        # This helps when several shifts share the same peak.
        max_shifts = np.where(conv >= max_val - 1e-15)[0]

        # ---------- gradient of the (smoothed) max ----------
        g = np.zeros_like(a)
        for k in max_shifts:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals

            # contribution a_j to ∂/∂a_i
            g[i_vals] += a[j_vals]

            # symmetric contribution for i ≠ j
            mask = i_vals != j_vals
            if mask.any():
                g[j_vals[mask]] += a[i_vals[mask]]

        # ---------- Adam‑style update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays with sqrt(iter) – helps stability
        step = step0 / np.sqrt(it)
        a_candidate = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] momentum it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # occasional simulated‑annealing accept to avoid being stuck
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if it % 3000 == 0:
            eps_vec = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

            2 * n * max(conv(a, a)) / (sum(a)**2)

    rng = np.random.default_rng(seed)

    deadline = start_time + budget_s - 0.2

    L_fft = _next_pow2(2 * n - 1)

    # Baseline: uniform distribution (score = 2)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a global starter if one is supplied (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:

            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Rich parametric sweep (geometric, reversed, beta)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 150.0

    p_grid = np.linspace(-5.0, 5.0, 15)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_geo_power_shape(n, r, p)
            # try the shape itself and its reversed version
            for base_shape in (shape, shape[::-1]):
                for lam in lam_grid:
                    cand = _mix_with_uniform(base_shape, lam)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} "
                              f"p={p:.2f} rev={base_shape is shape[::-1]} λ={lam:.2f} "
                              f"score={sc:.6f}")

    # beta distributions
    alpha_grid = np.linspace(0.2, 5.0, 10)
    beta_grid = np.linspace(0.2, 5.0, 10)
    for alpha in alpha_grid:
        if time.time() >= sweep_deadline:
            break
        for beta in beta_grid:
            shape = _make_beta_shape(n, alpha, beta)
            for base_shape in (shape, shape[::-1]):
                for lam in lam_grid:
                    cand = _mix_with_uniform(base_shape, lam)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc

                              f"score={sc:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # allocate the bulk of the remaining budget, leaving ~80 s for the final hill‑climb
    grad_deadline = deadline - 80.0
    for idx, start_vec in enumerate(top_seeds):
        if time.time() >= grad_deadline:
            break
        cur_score = _eval_fast(start_vec, L_fft)
        print(f"[{time.time() - start_time:.2f}s] descent start {idx+1}/{len(top_seeds)} "

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

                  f"score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb (tiny mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        best_a = a.copy()
        best_sc = best_score
        # pre‑compute FFT length once more (might have changed if n changed)
        L_fft = _next_pow2(2 * n - 1)
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue

            eps = a[src] * rng.random() * 4e-3

            a_prop[src] -= eps

            # project not needed – the operation conserves the sum and keeps non‑negativity
            sc = _eval_fast(a_prop, L_fft)
            if sc < best_sc:

                best_sc = sc
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={best_sc:.6f}")

        best_seq = best_a
        best_score = best_sc

    # --------------------------------------------------------------
    # Normalise finally (ensure sum = 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
