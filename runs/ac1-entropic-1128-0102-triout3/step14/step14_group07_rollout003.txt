"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a three‑phase local search:

1. **Diversified seeding** – a small pool of analytic shapes (uniform,
   power ramps, Beta‑densities, sine‑shapes) and many random Dirichlet
   vectors is built.  The best K candidates (by the exact objective)
   are kept.

2. **Heavy‑weight continuous optimisation** – the current best candidate
   is refined with two gradient‑based stages:
      * *Annealed smooth‑max Adam* (the smooth approximation of the
        “max” is gradually cooled to the true max).  This stage gets the
        lion’s share of the time budget.
      * *Exact sub‑gradient descent* on the true max‑objective, using the
        binary mask of the active convolution entries.  A simple
        decreasing step‑size and Euclidean projection onto the simplex
        are employed.

3. **Fine‑grained simulated‑annealing polish** – a short SA run that
   moves a small fraction of mass between random indices, accepting
   uphill moves with a temperature schedule.

All phases respect the 1000 s budget, use only NumPy, and keep the vector
on the probability simplex (non‑negative, sums to 1).  The final result
is returned as a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two not smaller than x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution a*a (real FFT, length 2·len(a)-1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (full) self‑convolution of a using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective (no scaling) for a vector that already sums to 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sine_shape(n: int) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·(i+0.5)/n)."""
    idx = (np.arange(n, dtype=float) + 0.5) / n
    shape = np.sin(np.pi * idx)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_shape(n: int) -> np.ndarray:
    """Cosine bump shape (non‑negative): a_i ∝ max(0, cos(π·(i+0.5)/n))."""
    idx = (np.arange(n, dtype=float) + 0.5) / n
    shape = np.cos(np.pi * idx)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Keep the K best candidates (sorted by score, lower better)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the pool if it belongs to the top‑k."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (temperature anneals to the true max)
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv).
    Returns the best (vector,score) seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule
    T_start = 0.20        # a little lower than the previous 0.30
    T_end = 1e-7
    step0 = 0.18          # a modest initial learning rate

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # full convolution and max
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # cooling
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # smooth‑max weights (stable softmax)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max = 2 * (w ⋆ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(conv(a,a)).
    Uses the binary mask of active constraints.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and active mask
        conv = _conv_self(a)
        M = conv.max()
        # tolerance to catch ties caused by floating point noise
        mask = (conv >= M - 1e-12).astype(float)
        mask_sum = mask.sum()
        if mask_sum > 0:
            mask /= mask_sum
        else:
            mask = np.full_like(conv, 1.0 / conv.size)

        # gradient of max = 2 * correlation(mask, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(mask, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # adaptive step (simple diminishing schedule)
        step = step0 / math.sqrt(it)

        a -= step * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} "
                  f"new best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """Very simple SA that moves a small fraction of mass between two indices."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # source chosen proportionally to current weight (more likely to move mass from big piles)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (sum ≈ 1).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose discretisation size
    # --------------------------------------------------------------
    # 65536 offers a good trade‑off between resolution and iteration speed.
    N = 65536

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 5
    pool = []   # (score, vector)

    # Uniform baseline
    _push_top(pool, _objective(_make_uniform(N)), _make_uniform(N), TOP_K)

    # Power‑ramps
    p_grid = np.linspace(0.2, 2.0, 9)          # 0.2 … 2.0
    plate_grid = np.linspace(0.0, 0.5, 7)     # 0 … 0.5
    for p in p_grid:
        for plat in plate_grid:
            a = _make_power_ramp(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # Beta shapes (fine sweep for α < 1)
    for alpha in np.arange(0.01, 0.31, 0.01):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _objective(a), a, TOP_K)

    # Sine / cosine shapes
    _push_top(pool, _objective(_make_sine_shape(N)), _make_sine_shape(N), TOP_K)
    _push_top(pool, _objective(_make_cosine_shape(N)), _make_cosine_shape(N), TOP_K)

    # Random Dirichlet vectors with a range of concentration parameters
    concs = [0.005, 0.02, 0.08, 0.2, 0.6, 2.0]
    for conc in concs:
        for _ in range(10):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)

    # Incorporate any previously discovered sequence if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # simple linear interpolation to required size
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Heavy continuous optimisation on the current best
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Allocate ~75 % of the remaining budget to Adam
        adam_deadline = time.time() + 0.75 * remaining
        best_seq, best_score = _adam_smooth_max(best_seq, adam_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Exact sub‑gradient descent (fine tuning)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sub_deadline = time.time() + 0.70 * remaining
        best_seq, best_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Short SA polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.02)
        print(f"[{time.time() - start_time:.2f}s] After SA best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
