def proj_simplex(v):
    # ensure v is numpy array
    # returns projection onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    # projecting onto simplex sum=1, v can have negative.
    # Based on Duchi algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]

        return np.full(n, 1.0 / n)

   random mixtures, power‑law shapes) and keep the best few.
2. Refine the current best candidate with a *smoothed* exponentiated‑gradient
   descent on the probability simplex.  Instead of using only the active
   (max‑valued) convolution entries, we form a softmax over all
   convolution values (temperature = ε) and use the resulting weights to
   compute a smooth sub‑gradient.  The update is multiplicative and the
   step size decays as η₀/√t; ε is slowly annealed.
3. Finish with a short simulated‑annealing polish that swaps tiny packets
   of mass between positions.
4. Return the best vector found (normalised to sum = 1).  The routine respects
   the supplied time budget (default 1000 s) and prints progress.

The combination of a smoother gradient and a longer‑range search over
analytic families typically pushes the score below the previous 1.5078,
aiming for the target ≈ 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real FFT, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (both real) using a real FFT."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes (used for the initial pool)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_powerlaw_shape(n: int, p: float, delta: float = 0.5) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p}."""
    idx = np.arange(n, dtype=float)
    vals = ((idx + delta) * (n - 1 - idx + delta)) ** (-p)
    total = vals.sum()
    if total > 0.0:
        vals /= total
    else:
        vals = np.full(n, 1.0 / n)
    return vals


def _interpolate_to(src: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to length n."""
    m = src.shape[0]
    if m == n:
        return src.copy()
    x_old = (np.arange(m) + 0.5) / m
    x_new = (np.arange(n) + 0.5) / n
    return np.interp(x_new, x_old, src)


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 eta0: float = 0.5,
                                 eps_start: float = 12.0,
                                 eps_min: float = 0.4,
                                 eps_decay_rate: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Gradient descent with a soft‑max weighting of the convolution entries.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps = eps_start
    eps_update_period = 200          # update ε every this many iterations

    # tiny floor to keep all coordinates strictly positive
    floor = 1e-14

    while time.time() < deadline:
        it += 1

        # ----- compute convolution and softmax weights -----
        conv = _conv_self(a)
        max_conv = conv.max()

        # softmax over convolution values (temperature = eps)
        # shift by max_conv for numerical stability
        w = np.exp(eps * (conv - max_conv))
        w_sum = w.sum()
        w_norm = w / w_sum

        # ----- gradient of the smooth max -----
        # grad = 2 * (w_norm * a) convolved with reversed a
        a_rev = a[::-1]
        conv_full = _conv_full(w_norm, a_rev)          # length = 3n‑2
        # contribution from the self‑pair (k = 2i) must be subtracted once
        self_term = w_norm[2 * np.arange(n)] * a
        grad = 2.0 * conv_full[n - 1:n - 1 + n] - self_term

        # ----- multiplicative update (exponentiated gradient) -----
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # enforce positivity floor and re‑normalise onto the simplex
        a = np.maximum(a, floor)
        a = _proj_simplex(a)

        # ----- keep best solution -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Smooth‑EG iter={it:,} eps={eps:.3f} score={cur_score:.6f}")

        # ----- anneal the temperature ε (make the max sharper) -----
        if it % eps_update_period == 0:
            eps = max(eps_min, eps * eps_decay_rate)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (unchanged)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # cooling schedule: T goes from 0.02 to 1e‑6
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)           # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5   # up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function within a time budget.
    Returns a list of non‑negative floats (the heights) whose sum is 1.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose length (power‑of‑two for fast FFT)
    # --------------------------------------------------------------
    n = 32768  # 2¹⁵ – a good compromise between quality and speed

    # --------------------------------------------------------------
    # 2) Build an initial pool of promising candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)
    pool_size = 12           # keep only the best few for later refinement

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # Random beta‑distribution shapes
    for _ in range(80):
        a = 10 ** rng.uniform(-2.0, 0.7)   # roughly 0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # Random mixtures of several beta shapes
    for _ in range(60):
        components = []
        total_w = 0.0
        for _ in range(rng.integers(2, 5)):  # 2‑4 components
            a = 10 ** rng.uniform(-2.0, 0.7)
            b = 10 ** rng.uniform(-2.0, 0.7)
            w = rng.random()
            total_w += w
            components.append((w, _make_beta_shape(n, a, b)))
        # combine
        v = sum(w / total_w * shp for w, shp in components)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # Power‑law family (U‑shaped)
    for p in np.linspace(0.15, 1.8, 20):
        v = _make_powerlaw_shape(n, p, delta=0.5)
        pool.append((_objective(v), v.copy()))

    # Optional global starter (height_sequence_1) – upscale if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = _interpolate_to(starter, n)
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                pool.append((sc, starter.copy()))
    except Exception:
        pass

    # Keep only the best `pool_size` candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:pool_size]

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Smoothed exponentiated‑gradient descent on the current best vector
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:  # ensure we have enough time for a meaningful run
        eg_deadline = time.time() + 0.70 * remaining
        best_seq, best_score = _smooth_exp_gradient_descent(
            best_seq,
            eg_deadline,
            start_time,
            eta0=0.5,
            eps_start=12.0,
            eps_min=0.3,
            eps_decay_rate=0.992,
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth‑EG best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
