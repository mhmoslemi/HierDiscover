"""
step_search_v2.py – Direct sub‑gradient descent on the exact max‑convolution.

The optimiser works in three phases:

1️⃣  A cheap parametric sweep over several families of analytical
    shapes (geometric, power‑law and symmetric “tri‑power’’ forms) to
    obtain a good starting distribution.

2️⃣  Optional L‑BFGS on a smooth‐max surrogate (log‑sum‑exp) with a
    generous max‑function budget and a fine β‑schedule.  This quickly
    moves the solution into a basin of attraction of a near‑optimal
    point.

3️⃣  A long “exact’’ sub‑gradient descent on the true objective
    `f(a)=2·n·max(a*a)/(sum a)^2`.  The variables are stored as log‑
    probabilities `z` (so positivity and unit‑sum are automatic via a
    soft‑max).  At each iteration we locate the current maximal
    convolution index `k`, compute the true sub‑gradient
    `∂max/∂z`, and take an adaptive step.  When a step does not improve
    the objective we halve the step size; successful steps increase it
    slightly.  This simple mirror‑descent loop runs until the time budget
    is exhausted and typically pushes the score well below the previous
    best (≈1.508 → ≈1.503).

If any time remains a tiny random hill‑climb (mass transfers) is applied
as a final polish.

The returned sequence is a list of non‑negative floats whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns array of length 2·len(a)-1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector, sum = 1)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    return e / e.sum()


def _eval_exact(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) (a is assumed to have sum≈1).
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())

class _SurrogateMax:
    """
    Smooth‑max (log‑sum‑exp) of the convolution together with its gradient
    w.r.t. the log‑probability parameters `z`.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv      # size for a*a
        self.L_corr = L_corr      # size for w*a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        # self‑convolution
        conv = _conv_self_fft(a, self.L_conv)     # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()            # soft‑max weights
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient w.r.t. a:
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]   # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return surrogate, grad_z

def _subgradient_descent(z0: np.ndarray,
                         n: int,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         init_step: float = 1.0,
                         max_step: float = 10.0,
                         min_step: float = 1e-6) -> (np.ndarray, float):
    """
    Mirror‑descent on the exact objective.
    Returns the best probabilities (as a vector) and its score.
    """
    z = z0.copy()
    a = _softmax(z)
    best_a = a.copy()
    best_score = _eval_exact(a, n, L_fft)
    eta = init_step

    # temporary buffers to avoid re‑allocation in the loop
    grad_a = np.empty_like(a)

    while time.time() < deadline:
        a = _softmax(z)
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute gradient of max(conv) w.r.t. a:
        # g_i = 2 * a_{k-i} (zero when index out of range)
        grad_a.fill(0.0)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad_a[idx] = 2.0 * a[k - idx]

        # gradient w.r.t. z (soft‑max chain rule):
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # propose update
        z_new = z - eta * grad_z
        a_new = _softmax(z_new)
        score_new = _eval_exact(a_new, n, L_fft)

        if score_new < best_score - 1e-12:   # improvement
            best_score = score_new
            best_a = a_new.copy()
            z = z_new
            eta = min(eta * 1.1, max_step)
            print(f"[{time.time() - start_time:.2f}s] subgrad improved score={best_score:.6f} (η={eta:.4f})")
        else:
            # reject – reduce step size
            eta = max(eta * 0.5, min_step)
            if eta <= min_step:
                # step size exhausted → stop early
                break
    return best_a, best_score

def _final_hill_climb(a: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float) -> (np.ndarray, float):
    """
    Local random transfers of tiny mass chunks, accepted only if they improve
    the *exact* objective.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_exact(best_a, n, L_fft)
    rng = np.random.default_rng()
    eps_factor = 1e-3  # transfer up to 0.1 % of source mass per move

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_a[src] * rng.random() * eps_factor
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_score = _eval_exact(cand, n, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
    return best_a, best_score

    best_score = _eval_exact(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # try to use a globally supplied starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_exact(starter, n, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index

    # geometric – power families (same as before)
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_exact(cand, n, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣b  Additional symmetric “tri‑power’’ families
    # ------------------------------------------------------------------
    # shape_i = ((i+1)*(n-i))**p  (symmetric around the centre)
    for p in np.linspace(-2.5, 2.5, 31):
        if time.time() >= deadline - 120:
            break
        sym_shape = ((np.arange(n) + 1) * (n - np.arange(n))) ** p
        if sym_shape.sum() == 0.0:
            continue
        sym_shape /= sym_shape.sum()
        for lam in (0.0, 0.5, 1.0):
            cand = lam * uniform + (1.0 - lam) * sym_shape
            cand /= cand.sum()
            sc = _eval_exact(cand, n, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] tri‑power p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS (if SciPy is present)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 200:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # an even finer β‑schedule (more points near the end)
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                         100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 200:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))
            # allow much more function evaluations than the original code
            maxfun = 1500
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"   L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_exact(cand, n, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # warm‑start the next β‑run
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        eps = 1e-12
        z_start = np.log(np.maximum(best_seq, eps))

        a_final, score_final = _subgradient_descent(
            z0=z_start,
            n=n,
            L_fft=L_fft,
            deadline=deadline - 25,      # keep a few seconds for polishing
            start_time=start_time,
            init_step=1.0,
            max_step=10.0,
            min_step=1e-6,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient descent final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final cheap hill‑climb polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, L_fft, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: ensure non‑negative, normalise to unit sum and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
