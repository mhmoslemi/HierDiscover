"""
Multi‑phase optimizer for the self‑convolution L∞ problem.

1. Build a modest pool of analytic candidates (uniform, beta‑shapes,
   random mixtures) and keep the best few.
2. Refine the current best vector with a few iterations of exponentiated‑
   gradient descent on the probability simplex.
3. Perform a Frank‑Wolfe–style improvement: for the current vector f,
   solve a small linear program (maximise Σg_i under the linearised
   constraints (f * g)_k ≤ M, where M = max(f * f)).  The LP direction g
   is normalised to a probability vector and mixed with f; the optimal
   mixing coefficient t is found analytically on a short grid.
   The step is repeated while it yields a strict decrease of M.
4. A short simulated‑annealing polish swaps tiny packets of mass
   between positions.

All phases respect the supplied time budget (default 1000 s) and return
a non‑negative list of floats (a discrete step function).  The score
is printed during the run; the final objective is
    2·n·max(conv(f,f))  (the sum of f is normalised to 1).
"""

import time
import math
import numpy as np

# optional SciPy – used for the LP phase
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (both length n) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale invariant – sum of a is forced to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    Continuous pdf ∝ x^{a‑1}·(1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.7,
                          max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """A short EG descent – multiplicative updates with KL mirror map."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set: all entries within a tiny relative tolerance
        ks = np.where(conv >= max_val - max_val * 1e-12)[0]

        # sub‑gradient (average of active gradients)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray,
                conv_f_f: np.ndarray,
                M: float,
                n: int,
                deadline: float,
                start_time: float,
                max_steps: int = 50) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Given f (sum = 1) and its self‑convolution max M,
    repeatedly:
        * take the indices with the largest convolution values,
        * solve max Σ g_i  s.t. (f·g)_k ≤ M for those k, g≥0,
        * mix f←(1‑t)f + t g with the best t on a short grid,
    until no improvement or time runs out.
    Returns the updated f, its convolution and the new M.
    """
    if not _SCIPY_OK:
        return f, conv_f_f, M

    # constant grid of mixing parameters
    t_grid = np.array([0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01])

    for step in range(max_steps):
        if time.time() >= deadline - 0.5:
            break

        # ----- active set (largest values of the current convolution) -----
        topk = min(200, conv_f_f.size)                     # keep the LP small
        idx = np.argpartition(-conv_f_f, topk - 1)[:topk]   # unsorted top‑k

        # ----- build sparse constraint matrix A (size |idx| × n) -----
        rows = []
        cols = []
        data = []
        for row_i, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1, dtype=int)
            rows.extend([row_i] * len(js))
            cols.extend(js.tolist())
            vals = f[k - js]                # coefficients f_{k‑j}
            data.extend(vals.tolist())

        A = csr_matrix((data, (rows, cols)), shape=(len(idx), n))
        b = np.full(len(idx), M)
        c = -np.ones(n)                     # maximise Σ g_i  ⇒ minimise -Σ g_i
        bounds = [(0.0, None)] * n

        # ----- solve the LP -----
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort the LP phase
            break

        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-12:
            break
        g /= g.sum()        # keep it on the simplex

        # ----- compute the three convolutions needed for the mixing test -----
        conv_f_g = _conv_cross(f, g)
        conv_g_g = _conv_self(g)

        # ----- evaluate mixing coefficients on the grid -----
        t = t_grid[:, None]                     # shape (T,1)
        combined = ((1 - t) ** 2) * conv_f_f + \
                   2 * t * (1 - t) * conv_f_g + \
                   (t ** 2) * conv_g_g
        new_M_vals = combined.max(axis=1)
        best_i = int(np.argmin(new_M_vals))
        best_t = t_grid[best_i]
        best_M = float(new_M_vals[best_i])

        if best_M < M * (1 - 1e-12):
            # accept the step
            f = (1.0 - best_t) * f + best_t * g
            conv_f_f = ((1 - best_t) ** 2) * conv_f_f + \
                       2 * best_t * (1 - best_t) * conv_f_g + \
                       (best_t ** 2) * conv_g_g
            M = best_M
            print(f"[{time.time() - start_time:.2f}s] LP step {step+1:3d}: "
                  f"t={best_t:.3f}  M={M:.6e}  score={2*n*M:.6f}")
        else:
            # no improvement – stop the LP phase
            break

    return f, conv_f_f, M


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Tiny‑packet swaps with a Metropolis rule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400_000               # generous but still fast

    for it in range(1, max_steps + 1):
        if time.time() >= deadline - 0.1:
            break

        # exponential cooling from 0.02 down to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # source biased by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain list of floats (the heights of the step function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length (small enough for fast FFTs)
    # --------------------------------------------------------------
    n = 4096   # 2¹¹ – gives a good trade‑off between flexibility and speed

    # --------------------------------------------------------------
    # 2) Build a small pool of promising analytic candidates
    # --------------------------------------------------------------
    pool = []          # (score, vector)
    pool_size = 7

    # uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # arcsine (beta with a=b=0.5) – often a strong candidate
    arcsine = _make_beta_shape(n, 0.5, 0.5)
    pool.append((_objective(arcsine), arcsine.copy()))

    # random beta shapes (parameters drawn log‑uniformly)
    for _ in range(80):
        a = 10 ** rng.uniform(-2.0, 0.7)   # about 0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # random convex combinations of two beta shapes
    for _ in range(40):
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1.0 - w) * _make_beta_shape(n, a2, b2)
        v /= v.sum()

                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                pool.append((sc, starter.copy()))
    except Exception:
        pass

    # keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:pool_size]

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Quick EG refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:

        best_seq, best_score = _exp_gradient_descent(best_seq,
                                                     eg_deadline,
                                                     start_time,
                                                     eta0=0.6)
        print(f"[{time.time() - start_time:.2f}s] After EG best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Frank‑Wolfe / LP improvement phase
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and _SCIPY_OK:
        # pre‑compute convolution of the current best
        conv_best = _conv_self(best_seq)
        M_cur = conv_best.max()
        best_seq, conv_best, M_cur = _lp_improve(best_seq,
                                                conv_best,
                                                M_cur,
                                                n,
                                                deadline,
                                                start_time,
                                                max_steps=70)
        best_score = 2.0 * n * M_cur
        print(f"[{time.time() - start_time:.2f}s] After LP best score = {best_score:.6f}")

    # 5) Light annealing polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:

    # 6) Normalise and convert to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")
