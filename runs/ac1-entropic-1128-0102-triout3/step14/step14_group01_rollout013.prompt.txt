def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]

def _adam_optimize(a0: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    τ (temperature) is increased geometrically – the surrogate turns into
    the true max as optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)               # for self‑convolution
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04                                   # base learning rate
    tau = 20.0                                  # start temperature
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        # current score
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max weights over the convolution vector
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w_raw / w_sum

        # gradient of smooth‑max: 2·(w * a_rev) restricted to n entries
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)            # length = 3·n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature gradually
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score

def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that moves a small mass from the most
    positively‑gradient index to the most negative one.
    Occasionally accepts worsening moves according to a simple
    temperature schedule (simulated annealing flavour).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02  # initial temperature (relative to score)
    # decay factor so T → ~0 at the deadline
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        # sub‑gradient of the (non‑smooth) max‑convolution
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # candidate source (must have mass) – pick index with largest
        # positive sub‑gradient
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – most negative sub‑gradient
        dst = np.argmin(grad)

        if src == dst:
            # fallback to a random transfer
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer a tiny fraction (up to 0.5 % of source mass)
        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score
