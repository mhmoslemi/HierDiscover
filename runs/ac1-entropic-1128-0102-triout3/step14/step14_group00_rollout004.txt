"""
search_step_function.py

A time‑limited optimiser for the “self‑convolution” objective

    2·n·max(conv(a,a)) / (sum(a)**2)

All candidate vectors are probability distributions (non‑negative,
summing to one).  The algorithm consists of three stages:

1️⃣  **Analytic initial pool** – a dense grid of Beta‑distributions,
    a simple power‑law family and a few random Dirichlet draws.
    The best few are kept.

2️⃣  **Projected Gradient Descent (PGD)** on the simplex.
    At each iteration the active set of convolution indices attaining the
    current maximum is built, a sub‑gradient of the L∞‑norm is formed and
    an additive step is taken.  A back‑tracking line‑search together with a
    projection onto the probability simplex guarantees feasibility and
    fast descent.

3️⃣  **Light simulated annealing** – a short Metropolis‑style polish that
    swaps tiny packets of mass between positions.

The routine respects the supplied time budget (default 1000 s) and
returns a plain `list[float]` that can be fed directly to
`evaluate_sequence`.  The implementation is self‑contained, uses only
NumPy and the standard library and runs on a single CPU core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised Beta distribution on {0,…,n‑1}, normalised to sum = 1.
    Continuous pdf ∝ x^{a‑1}(1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_shape(n: int, exponent: float) -> np.ndarray:
    """
    Shape proportional to ((i+1)*(n-i))^{exponent}, uniform when exponent=0.
    Gives a U‑shape for negative exponents and a mountain‑shape for positive.
    """
    i = np.arange(n)
    shape = ((i + 1) * (n - i)) ** exponent
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected Gradient Descent on the simplex (sub‑gradient of L∞‑norm)
# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 max_no_improve: int = 800) -> tuple[np.ndarray, float]:
    """
    Additive (projected) gradient descent on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    no_improve = 0
    it = 0

    while time.time() < deadline and no_improve < max_no_improve:
        it += 1

        # --- compute active set of indices attaining the current maximum ---
        conv = _conv_self(a)
        max_val = conv.max()
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        # --- sub‑gradient of the L∞‑norm (average over active set) ---
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # descent direction = –gradient
        d = -grad

        # --- back‑tracking line search (max step 1.0) ---
        step = 1.0
        improved = False
        for _ in range(25):
            a_prop = a + step * d
            a_prop = _project_simplex(a_prop)
            new_score = _objective(a_prop)
            if new_score < cur_score - 1e-14:
                a = a_prop
                cur_score = new_score
                improved = True
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] PGD iter={it:,}  score={new_score:.6f}")
                break
            step *= 0.5

        if not improved:
            no_improve += 1

        # safety break – leave a few seconds for the next phase
        if time.time() > deadline - 5.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the previous implementation)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from 2·10⁻² down to 10⁻⁶
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # bias towards big entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine builds a pool of analytical candidates, refines the
    best ones by projected gradient descent on the simplex and finishes
    with a brief simulated‑annealing polish.
    The returned list contains non‑negative floats and respects the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length (good trade‑off for FFT speed)
    # --------------------------------------------------------------
    n = 16384   # 2¹⁴

    # --------------------------------------------------------------
    # 2) Build an initial pool of promising analytic candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # Grid of Beta‑shapes (log‑spaced parameters)
    a_vals = np.logspace(-2.0, 0.7, 15)   # ≈ 0.01 … 5
    b_vals = a_vals
    for a in a_vals:
        for b in b_vals:
            if time.time() >= deadline:
                break
            v = _make_beta_shape(n, a, b)
            pool.append((_objective(v), v.copy()))

    # Power‑law family (U‑shapes for negative exponents)
    exponents = np.linspace(-1.0, 1.5, 13)   # from U‑shaped to peaked
    for e in exponents:
        if time.time() >= deadline:
            break
        v = _make_power_shape(n, e)
        pool.append((_objective(v), v.copy()))

    # Random Dirichlet‑type vectors (simple random numbers)
    for _ in range(30):
        if time.time() >= deadline:
            break
        v = rng.random(n)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # Random mixtures of two Beta shapes
    for _ in range(40):
        if time.time() >= deadline:
            break
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1.0 - w) * _make_beta_shape(n, a2, b2)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # Optional global starter (if a previous best is available)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Resample to the working size if necessary
            if starter.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, starter.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                pool.append((_objective(starter), starter.copy()))
    except Exception:
        pass

    # Keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    top_k = 6
    pool = pool[:top_k]

    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time()-start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Projected Gradient Descent on the current best candidates
    # --------------------------------------------------------------
    for idx, (_, init_seq) in enumerate(pool):
        if time.time() >= deadline - 20.0:   # keep a little room for polishing
            break
        seq, sc = _pgd_descent(init_seq, deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] New best after PGD #{idx} = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Optional up‑sampling & a second PGD pass (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        # Double the resolution by linear interpolation
        n2 = best_seq.shape[0] * 2
        xs_old = np.linspace(0.0, 1.0, best_seq.shape[0])
        xs_new = np.linspace(0.0, 1.0, n2)
        up_seq = np.interp(xs_new, xs_old, best_seq)
        up_seq = np.maximum(up_seq, 0.0)
        up_seq /= up_seq.sum()
        seq, sc = _pgd_descent(up_seq, deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] New best after up‑scaled PGD = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time()-start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time()-start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
