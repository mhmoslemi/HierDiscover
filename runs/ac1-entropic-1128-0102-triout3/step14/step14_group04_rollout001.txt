"""
Improved optimisation for the step‑function objective.

We search for a non‑negative vector a (∑a = 1) that minimises

    2·n·max(conv(a,a))

where conv is the linear self‑convolution.  The programme proceeds as follows:

1.  Build a diverse initial pool of analytic windows (Hann, Kaiser, β‑,
    power‑ramp, Gaussian) together with many random Dirichlet draws.
    Keep the best few candidates (by the exact objective).

2.  Repeatedly improve the current best vector by a **linear‑programming
    “mass‑filling’’ step**:
        maximise   ∑b            (b ≥ 0)
        subject to  a * b ≤ M   (where M = max(conv(a,a))).

    The LP is tiny (n ≤ 1024) and can be solved with `scipy.optimize.linprog`.
    The resulting b is scaled to the same ℓ₁‑norm as a; if the objective
    improves we accept it.

3.  Perform a short **projected Adam‑style sub‑gradient descent** on the
    (smoothed) max‑convolution, keeping the iterate on the probability simplex.

4.  Finish with a light **simulated‑annealing** polish that swaps tiny packets
    of mass between positions.

The whole search adapts to the remaining time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective for vectors normalised to sum = 1
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of windows (all are normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        # fallback – simple Bessel approximation
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two symmetric power ramps with a flat central plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    # enforce even plateau length (helps symmetry)
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad if rounding produced a short vector
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Adam‑style momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    Acts on the (smoothed) max‑convolution.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # denominator = 1 (∑a = 1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        # Active set
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        # Sub‑gradient of the (averaged) max‑conv
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_i = np.arange(lo, hi + 1, dtype=int)
            idx_j = k - idx_i
            grad[idx_i] += a[idx_j]
        grad = 2.0 * grad / max(1, len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of mass between positions,
    Metropolis acceptance with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Exponential cooling schedule
        T = max(1e-5, 0.1 * (0.01 / 0.1) ** (it / max_steps))

        src = rng.choice(n, p=a)          # source prefers large entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise   ∑b
            subject to a * b ≤ M ,  b ≥ 0

    Returns the optimiser b (not yet normalised).
    """
    from scipy.optimize import linprog

    n = a.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A (rows × n) such that (A @ b)[k] = (a * b)[k]
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j + n, j] = a  # fill the j‑th column with a, shifted down by j

    # Linear programme: minimise -∑b  (=> maximise ∑b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    # Solve with the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        raise RuntimeError("LP failed: " + (res.message or "unknown"))
    b = np.maximum(res.x, 0.0)
    return b


# ----------------------------------------------------------------------
# Helper to down‑sample a longer vector to a shorter length (simple averaging)
# ----------------------------------------------------------------------
def _downsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Average‑pool `vec` to length `new_n` (preserves total mass)."""
    old_n = vec.shape[0]
    factor = old_n // new_n
    truncated = vec[: factor * new_n]
    reshaped = truncated.reshape(new_n, factor)
    new_vec = reshaped.mean(axis=1)
    # renormalise to sum = 1
    s = new_vec.sum()
    if s > 0.0:
        new_vec /= s
    else:
        new_vec = np.full(new_n, 1.0 / new_n)
    return new_vec
