def _exp_subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # parameters
    eta0 = 2.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # active set
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        # subgradient
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        # correlation via FFT
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step-size schedule
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # projection onto simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score

"""
Step‑function optimiser (improved).

The algorithm works in two stages.
  1) A fast exploration on a moderate size (≈8‑16 k) uses a very
     diverse family of analytic shapes (Beta, power‑ramps, exponentials,
     random Dirichlet mixtures, random convex combinations of shapes) and
     keeps the TOP_K best candidates.
  2) The current best candidate is refined on the final problem size.
     First a (fast) exponentiated sub‑gradient descent on the exact
     ℓ∞‑objective reduces the maximal self‑convolution, then an Adam‑based
     smooth‑max descent (soft‑max surrogate) polishes the solution.
If time permits a tiny hill‑climbing phase performs cheap local swaps.

All steps respect the required format (non‑negative list, no scaling
outside the allowed range) and run within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “power‑ramp” with optional flat plateau; normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # pad / trim to exact length
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}; normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}; normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, rate: float) -> np.ndarray:
    """Decreasing exponential a_i ∝ exp(-rate·i) (normalised)."""
    shape = np.exp(-rate * np.arange(n, dtype=float))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 2.0        # start step size – a bit larger than before
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (within tiny tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # correlation = (w * a_rev) in the time domain
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule (slow decay)
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise onto the simplex (exact sum = 1)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max (soft‑max) surrogate of the ℓ∞‑objective.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05
    T0 = 0.02          # initial temperature for soft‑max
    T_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature moves linearly towards T_end
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights (log‑sum‑exp trick for stability)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * (w * a_rev)   (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


def _hill_climb(seq: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny amount of mass from one index to another.
    """
    a = seq.copy()
    n = a.shape[0]
    best_score = _objective(a)
    rng = np.random.default_rng(42)
    delta_factor = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        max_move = a[i] * delta_factor
        if max_move <= 0:
            continue
        move = max_move * rng.random()
        a_new = a.copy()
        a_new[i] -= move
        a_new[j] += move
        if a_new[i] < -1e-12:
            continue
        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            a = a_new
            delta_factor = max(delta_factor * 0.99, 0.001)
            print(f"[{time.time() - start_time:.2f}s] HillClimb iter={it:,} best={cur_score:.6f}")
    return a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    The routine builds a large but cheap candidate pool at a moderate size,
    refines the best element with exponentiated sub‑gradient descent,
    upsamples to the final size and finishes with Adam‑based smoothing and
    a tiny hill‑climbing pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a modest size for the broad search (speed vs resolution)
    # --------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 600:
        N_small = 16384
    elif remaining_total > 300:
        N_small = 8192
    else:
        N_small = 4096

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool of candidates
    # --------------------------------------------------------------
    TOP_K = 15
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        vec = np.maximum(vec, 0)
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        _push_top(top, _objective(vec), vec, TOP_K)

    # basic baselines
    _add(np.full(N_small, 1.0 / N_small))

    # symmetric Beta (log‑spaced alphas, very wide range)
    for a in np.logspace(-2, 0.7, 30):           # 0.01 … ≈5
        _add(_make_beta_shape(N_small, a))

    # asymmetric Beta (coarse grid)
    for a in np.linspace(0.2, 5.0, 8):
        for b in np.linspace(0.2, 5.0, 8):
            _add(_make_beta_shape_asym(N_small, a, b))

    # power‑ramp shapes
    for p in np.linspace(0.1, 5.0, 12):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add(_make_shape(N_small, p, plateau))

    # exponential decays
    for rate in np.logspace(-3, -0.1, 12):
        _add(_make_exponential(N_small, rate))

    # random mixtures of two symmetric Beta shapes
    rng = np.random.default_rng(seed + 2023)
    for _ in range(60):
        a1 = 10 ** rng.uniform(-2, 0.7)
        a2 = 10 ** rng.uniform(-2, 0.7)
        w = rng.random()
        shape = w * _make_beta_shape(N_small, a1) + (1 - w) * _make_beta_shape(N_small, a2)
        _add(shape)

    # random mixtures of three symmetric Beta shapes

        weights = rng.dirichlet(np.ones(3))

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")
