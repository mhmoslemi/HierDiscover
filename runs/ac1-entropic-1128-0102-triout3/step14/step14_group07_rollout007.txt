"""
propose_candidate
-----------------

Search for a sequence of non‑negative heights (step‑function values) that
minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full discrete convolution.  The optimizer works in
four stages:

1️⃣ **Coarse exploration on a small grid** – we generate a diverse
   family of analytic shapes (uniform, power‑ramps, asymmetric
   Beta‑densities and random Dirichlet draws) on a modest size
   ``n_small`` (≈ 8192).  The best ``TOP_K`` candidates are retained.

2️⃣ **Projected Adam on the exact max‑convolution** – for each retained
   candidate we run an Adam optimiser that uses the true sub‑gradient of
   the max‑convolution (the gradient is a correlation of the active‑set
   mask with the reversed vector).  The iterates are projected onto the
   probability simplex after each step.  The optimisation is performed
   first on the small grid (many cheap iterations) and the final best
   candidate is interpolated to the full size ``N`` (≈ 65536) for a short
   second‑stage refinement.

3️⃣ **Aggressive simulated‑annealing polish** – a mass‑move SA (transfer
   a small fraction of probability from one index to another) is run
   for the remaining time, accepting worsening moves with a temperature
   schedule.  This helps escape the final few local minima.

4️⃣ **Normalisation & return** – the resulting vector is forced onto the
   simplex, clipped to non‑negative values, and converted to a plain
   Python list.

All heavy work uses only NumPy and runs on a single CPU within the
1000 s budget.  The code prints progress information (objective value,
iteration counters, timestamps) so the search behaviour can be followed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (sum already normalised to 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric shape generators
# ----------------------------------------------------------------------
def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    # avoid exact 0/1 for extreme parameters
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted pool of the k lowest scores."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Adam optimiser based on the exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_subgrad(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising 2·n·max(conv(a,a)).
    Uses the true sub‑gradient (average over all active indices).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution and current objective
        conv = _conv_self(a)
        max_c = conv.max()
        cur_score = float(2.0 * n * max_c)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                f"score={cur_score:.6f}"
            )

        # active‑set mask (all indices within a tiny tolerance of the maximum)
        tol = max_c * 1e-12 + 1e-16
        mask = (conv >= max_c - tol).astype(float)
        mask_sum = mask.sum()
        if mask_sum == 0:
            mask = np.full_like(conv, 1.0 / conv.size)
            mask_sum = conv.size
        w = mask / mask_sum

        # gradient = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.2 / math.sqrt(it)   # aggressive early steps, then decay
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
    max_move_frac: float = 0.04,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # pick source proportional to its weight (more mass → larger moves)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                    f"new best={new_score:.6f}"
                )
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a Python list of non‑negative floats (the heights).
    """
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # --------------------------------------------------------------
    # Choose problem sizes
    # --------------------------------------------------------------
    # Small size for the wide exploratory phase (fast FFTs)
    n_small = 8192

    # Final size – use the largest power‑of‑two that fits comfortably in time
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Coarse exploration on the small grid
    # --------------------------------------------------------------
    TOP_K = 30
    pool = []   # list of (score, vector)

    # uniform
    uniform = np.full(n_small, 1.0 / n_small)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 3.0, 7):
        for plateau in np.linspace(0.0, 0.5, 5):
            a = _make_power_ramp(n_small, p, plateau)
            _push_top(pool, _objective(a), a, TOP_K)

    # asymmetric Beta shapes (grid + random sampling)
    alphas = np.logspace(-1.3, 0.0, 6)      # ~0.05 … 1
    betas = np.logspace(-0.3, 0.7, 6)      # ~0.5 … 5
    for a_ in alphas:
        for b_ in betas:
            a = _make_beta_shape(n_small, a_, b_)
            _push_top(pool, _objective(a), a, TOP_K)

    # random asymmetric Beta draws
    for _ in range(50):
        a_ = 10 ** rng.uniform(-2.0, 0.0)      # α ∈ [0.01,1]
        b_ = 10 ** rng.uniform(-0.5, 0.8)     # β ∈ [0.3,6.3]
        a = _make_beta_shape(n_small, a_, b_)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet vectors with low concentration (highly spiky)
    concs = [0.01, 0.03, 0.07, 0.12, 0.25]
    for conc in concs:
        for _ in range(30):
            if time.time() >= deadline:
                break
            draws = rng.gamma(conc, 1.0, size=n_small)
            a = draws / draws.sum()
            _push_top(pool, _objective(a), a, TOP_K)

    # --------------------------------------------------------------
    # 2) Refine the best candidates on the small grid
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])   # best first
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    remaining = deadline - time.time()
    # give about half of the remaining budget to the refinement on the
    # small grid (many cheap iterations)
    refine_deadline = time.time() + remaining * 0.45

    for idx, (_, vec) in enumerate(pool[:5]):          # only the top few
        if time.time() >= refine_deadline:
            break
        refined, score = _adam_subgrad(vec, refine_deadline, start_time)
        if score < best_score:
            best_score, best_seq = score, refined.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Refinement #{idx} improved best to {score:.6f}"
            )

    # --------------------------------------------------------------
    # 3) Upsample to the final size and finish with high‑resolution optimisation
    # --------------------------------------------------------------
    # linear interpolation to the final length N
    x_old = np.linspace(0, 1, n_small, endpoint=False)
    x_new = np.linspace(0, 1, N, endpoint=False)
    best_seq = np.interp(x_new, x_old, best_seq)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum()

    # one more short Adam‑run on the full size
    remaining = deadline - time.time()
    if remaining > 30.0:
        full_deadline = time.time() + remaining * 0.6
        best_seq, best_score = _adam_subgrad(best_seq, full_deadline, start_time)

    # --------------------------------------------------------------
    # 4) Optional simulated‑annealing polish
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, start_time, rng, max_move_frac=0.05
        )

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
