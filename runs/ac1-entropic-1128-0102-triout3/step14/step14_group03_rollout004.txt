The algorithm searches for a non‑negative vector `a` (the step‑function heights) that minimizes  

\[
F(a)=\frac{2\,n\;\max\bigl((a*a)\bigr)}{(\sum_i a_i)^2},
\]

where `*` is the linear convolution.  
Because the expression is homogeneous of degree 0 we may fix `∑a_i = 1`.  
The goal is therefore to make the self‑convolution as flat as possible – i.e. to push every entry of `a*a` towards the average value `1/(2n‑1)`.

The search consists of three phases, each respecting the global 1000 s budget.

1. **Warm‑start** – use the globally supplied `height_sequence_1` (or the uniform vector) and resize it to the working size `n = 4096`.  
2. **Smooth‑gradient descent** – minimise a *smooth* surrogate of the max‑operator  

   \[
   \hat F_\beta(a)=\frac{2\,n}{(\sum a)^2}\;
   \frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr)
   \]

   for an increasing temperature `β`.  
   The gradient of the soft‑max term can be written as a single convolution,
   so a full iteration needs only two FFTs (one for the convolution, one for the weighted gradient).  
   An Adam optimiser together with a Euclidean projection onto the probability simplex
   (non‑negative entries that sum to 1) drives the iterates.
3. **Exact sub‑gradient polishing** – once the smooth phase has exhausted its
   budget, a classic sub‑gradient descent on the true max‑convolution (the
   method used in the reference solution) is run for the remaining seconds.

All intermediate best vectors are kept; the final result is normalised and
returned as a plain Python list.

The whole code follows the required API (`propose_candidate`) and prints
progress information.

```python
"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  Warm‑start from the globally supplied `height_sequence_1` (or uniform).

2️⃣  Smooth gradient descent on a soft‑max surrogate of the max‑convolution.
    The surrogate is controlled by a temperature β that is increased
    during the run.  The gradient can be expressed as a single convolution,
    thus each iteration needs two FFTs (O(n log n)).  An Adam optimiser
    together with Euclidean projection onto the probability simplex guides
    the search.

3️⃣  Final polishing with the exact sub‑gradient descent (the method from the
    reference solution).  The remaining time budget is used here.

All phases respect the global 1000 s budget.  The best sequence found is
returned as a list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, sum w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate
# ----------------------------------------------------------------------


def _weighted_gradient(a: np.ndarray, w: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Gradient of   g(a) = Σ_k w_k * (a*a)_k   (w is a probability vector).
    The result is 2 * N * a where N[i,j] = w_{i+j}.
    It can be computed via one convolution.
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    if L is None:
        L = _next_pow2(n + w_rev.size - 1)   # length for a * w_rev
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)
    # slice the part that corresponds to indices n‑1 … 2n‑2
    start = n - 1
    end = start + n
    slice_n = conv[start:end]
    # reverse to obtain (N a)
    grad = 2.0 * slice_n[::-1]
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (as in the reference solution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional uphill move
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

    Search for a non‑negative step function that minimises the evaluation
    function.  The routine runs three phases (warm‑start, smooth gradient,
    exact sub‑gradient) and respects the 1000 s budget.  The best sequence is
    returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm‑start: try to reuse the globally supplied construction
    # ------------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.ndim == 1 and seq0.size > 0:
            seq0 = np.maximum(seq0, 0.0)
            if seq0.sum() == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                # Resize to the required length by simple linear interpolation
                if seq0.shape[0] != n:
                    x_old = np.linspace(0.0, 1.0, seq0.shape[0])
                    x_new = np.linspace(0.0, 1.0, n)
                    seq0 = np.interp(x_new, x_old, seq0)
                a = seq0 / seq0.sum()
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Guarantee correct shape and normalisation
    a = a[:n]                     # truncate if somehow longer
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a.sum()

    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    print(f"[{time.time() - start_time:.2f}s] initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth gradient descent on a soft‑max surrogate
    # ------------------------------------------------------------------
    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.12          # learning rate before Adam scaling
    it = 0

    # Temperature schedule for the soft‑max
    beta_start = 0.5
    beta_target = 25.0
    beta_ramp_iters = 50000   # number of iterations over which β is increased

    # Allocate the FFT length for the weighted gradient once (re‑used every iter)
    L_grad = _next_pow2(3 * n - 2)

    # Main loop – stops when the remaining budget is insufficient for the
    # final polishing phase.
    while time.time() < deadline - 40.0:   # keep ~40 s for the sub‑gradient step
        it += 1

        # 1) self‑convolution (needed for both the surrogate value and the weights)
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()

        # 2) temperature β (linear increase, then clamp)
        beta = beta_start + (beta_target - beta_start) * min(1.0, it / beta_ramp_iters)

        # 3) soft‑max weights (stabilised by subtracting the maximum)
        exp_term = np.exp(beta * (conv - max_conv))
        w = exp_term / exp_term.sum()          # w is a probability vector, length 2n‑1

        # 4) gradient of the weighted sum Σ_k w_k (a*a)_k
        g = _weighted_gradient(a, w, L=L_grad)

        # 5) Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # 6) evaluate (fast version – exact because Σa = 1)
        score = _eval_fast(a_new, L_fft)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} β={beta:.2f} "
                  f"new best {score:.6f}")

        # Accept the new point (the soft‑max surrogate is smooth, so we keep it)
        a = a_new

        # ------------------------------------------------------------------
        # Small random jitter every few thousand steps – helps escape plateaus
        # ------------------------------------------------------------------
        if it % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

        # Occasionally reset to the best point to avoid drifting too far
        if it % 8000 == 0:
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient polishing (uses the reference routine)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 5.0
        a_refined, sc_refined = _subgradient_descent(best_a, sub_deadline,
                                                     start_time, best_score)
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final sub‑grad score={sc_refined:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
```
