Improved step‑function search for minimise

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm follows a richer multi‑stage pipeline than the reference
implementation:

1. **Broad parametric sweep** – geometric, power and a dense 2‑D
   Beta‑grid (α,β∈[0.2,10]) plus a sinusoidal family.  Each candidate is
   normalised to the simplex and its score recorded.

2. **Massive random Dirichlet initialisers** – 600 draws covering a very
   wide range of concentration parameters (10⁻³ … 10²).  The ten best
   candidates are kept.

3. **Multi‑start Adam with smooth‑max** – the best 5 candidates from step 2
   are each refined with an Adam optimiser that gradually turns the
   smooth‑max into the true max.  The runs share the remaining time
   equally.

4. **Aggressive hill‑climb** – for every Adam‑refined vector we run a
   directed hill‑climb that transfers a *larger* fraction of mass (up to
   2 %) along the sub‑gradient direction, interleaved with occasional
   random “re‑smooth’’ moves (convolution with a tiny Gaussian kernel).

5. **Final polishing** – a short Adam run on the best vector found so
   far, followed by one more hill‑climb.

All operations respect the 1000 s budget; the routine aborts early if the
target score ≈ 1.503 is reached.  The returned list is a plain Python list
of non‑negative floats that sums to 1 (the optimiser works on the simplex
via Euclidean projection).  The code uses only NumPy and the provided
``evaluate_sequence`` for final validation.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# --------------------------------------------------------------
# Candidate bookkeeping
# --------------------------------------------------------------
def _update_best(seq: np.ndarray, score: float,
                 best_list: list, max_len: int = 10):
    """Insert (score,seq) into a sorted best‑list of length ≤ max_len."""
    # keep list sorted ascending (lower score = better)
    best_list.append((score, seq.copy()))
    best_list.sort(key=lambda x: x[0])
    # truncate
    if len(best_list) > max_len:
        best_list.pop()


# --------------------------------------------------------------
# 1️⃣  Parametric families
# --------------------------------------------------------------
def _sweep_geometric(n, best_list, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-6, -0.1, 350)    # finer near zero
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 13):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            _update_best(seq, sc, best_list)
    return


def _sweep_power(n, best_list, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-6.0, 6.0, 241):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        _update_best(seq, sc, best_list)
    return


def _sweep_beta(n, best_list, deadline):
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    alphas = np.linspace(0.2, 10.0, 41)   # dense grid
    betas = alphas
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * log_rev
            # numerical stability
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            _update_best(seq, sc, best_list)
    return


def _sweep_sinusoidal(n, best_list, deadline):
    """a_i ∝ sin(π·(i+0.5)/n)^p  (zero at ends for any p>0)."""
    idx = np.arange(n, dtype=np.float64)
    base = np.sin(np.pi * (idx + 0.5) / n)
    for p in np.linspace(0.5, 6.0, 112):
        if time.time() >= deadline:
            break
        seq = np.power(base, p)
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        _update_best(seq, sc, best_list)
    return


# --------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# --------------------------------------------------------------
def _random_dirichlet_trials(n, best_list, deadline, trials=600):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # Random concentration: log‑uniform from 1e‑3 to 1e2
        logc = rng.uniform(-3.0, 2.0)
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        _update_best(a, sc, best_list)
    return


# --------------------------------------------------------------
# 3️⃣  Smooth‑max Adam optimisation (multi‑start)
# --------------------------------------------------------------
def _adam_smooth_max_descent(a0, deadline, best_list, max_iters=200000):
    """Adam with a gradually sharpening smooth‑max."""
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.04

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 12.0                # start temperature (softness)
    tau_factor = 1.4
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        _update_best(a, cur_score, best_list)

        # smooth approximation of max
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of smooth max: 2 * (w ⋆ rev(a))
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)      # length = 3n‑2 (full)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature every few thousand steps
        if it % 3000 == 0:
            tau = min(tau * tau_factor, 1e9)

        # early exit if we already reached the target
        if cur_score <= 1.503:
            break
    return


# --------------------------------------------------------------
# 4️⃣  Aggressive hill‑climb with occasional smoothing
# --------------------------------------------------------------
def _hill_climb(start_a, deadline, best_list, max_noimp=8000):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # choose source (largest positive gradient) and destination (most negative)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # sometimes perturb randomly to avoid stagnation
        if src == dst or rng.random() < 0.02:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # transfer a *significant* fraction (up to 2 %) of source mass
        frac = rng.random() * 0.02 + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)   # maintains feasibility

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score - 1e-7:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            _update_best(a, cur_score, best_list)
        else:
            no_improve += 1
            if no_improve >= max_noimp:
                # apply a small smoothing move to escape plateaus
                kernel = np.array([0.25, 0.5, 0.25])
                smooth = np.convolve(a, kernel, mode='same')
                a = _project_simplex(np.maximum(smooth, 0.0))
                no_improve = 0

        if best_score <= 1.503:
            break
    return best_a, best_score

# Wrapper: propose_candidate
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The function respects a 1000 s budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # problem size
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # initialise best‑list (sorted by score, best first)
    best_list = []
    init_score = _objective_score(uniform)
    _update_best(uniform, init_score, best_list)

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if provided externally
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                _update_best(starter, sc, best_list)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (broad)
    # --------------------------------------------------------------
    phase_deadline = deadline - 260.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting parametric sweeps")
        _sweep_geometric(n, best_list, phase_deadline)
        _sweep_power(n, best_list, phase_deadline)
        _sweep_beta(n, best_list, phase_deadline)
        _sweep_sinusoidal(n, best_list, phase_deadline)
        print(f"[{time.time() - start_time:.2f}s] Parametric sweep best score={best_list[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Random Dirichlet trials")
        _random_dirichlet_trials(n, best_list, phase_deadline, trials=800)
        print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score={best_list[0][0]:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Multi‑start Adam (smooth‑max)
    # --------------------------------------------------------------
    # take the top‑5 distinct candidates (avoid duplicates)
    top_candidates = []
    seen_hashes = set()
    for score, seq in best_list:
        h = hash(seq.tobytes())
        if h not in seen_hashes:
            top_candidates.append(seq)
            seen_hashes.add(h)
        if len(top_candidates) >= 5:
            break

    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        per_run = remaining / len(top_candidates)
        print(f"[{time.time() - start_time:.2f}s] Multi‑start Adam ({len(top_candidates)} runs, ~{per_run:.1f}s each)")
        for i, seq in enumerate(top_candidates, 1):
            run_deadline = time.time() + per_run - 0.2
            _adam_smooth_max_descent(seq, run_deadline, best_list)
            print(f"[{time.time() - start_time:.2f}s] Adam run {i}/{len(top_candidates)} done")
            if best_list[0][0] <= 1.503:
                break

    # --------------------------------------------------------------
    # 4️⃣  Aggressive hill‑climb on the current best vector
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_vec = best_list[0][1]
        best_vec, best_score = _hill_climb(best_vec, deadline, best_list)
        _update_best(best_vec, best_score, best_list)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final polishing (short Adam + hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline - 10.0
        if time.time() < final_deadline:
            print(f"[{time.time() - start_time:.2f}s] Final short Adam refinement")
            _adam_smooth_max_descent(best_list[0][1], final_deadline, best_list)
        # final hill‑climb with tighter tolerance
        if time.time() < deadline:
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb")
            _hill_climb(best_list[0][1], deadline, best_list, max_noimp=12000)

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # --------------------------------------------------------------
    final_seq = best_list[0][1]
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total > 0:
        final_seq = final_seq / total
    else:
        final_seq = uniform.copy()

    # sanity check with the provided evaluator (optional, does not affect return)
    try:
        from __main__ import evaluate_sequence  # may already be in globals
    except Exception:
        evaluate_sequence = None
    if evaluate_sequence is not None:
        val = evaluate_sequence([float(x) for x in final_seq])
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence reports {val:.6f}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_list[0][0]:.6f}")
    return [float(x) for x in final_seq]
