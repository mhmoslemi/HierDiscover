```
def _mirror_descent(a0, deadline, start_time, eta0=0.6):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    eps_floor = 1e-14
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        tol = M * 1e-12
        ks = np.where(conv >= M - tol)[0]
        grad = np.zeros_like(a)
        if len(ks) == 1:
            k = ks[0]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] = a[k - idx]
        else:
            for k in ks:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                grad[idx] += a[k - idx]
            grad /= len(ks)
        # subtract M*a (the term from the derivative of denominator)
        grad -= M * a
        # step size schedule
        eta = eta0 / math.sqrt(it)
        # multiplicative update (mirror descent on simplex with KL divergence)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
        score = _objective(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror iter={it:,}  score={score:.6f}")
    return best_a, best_score
```

"""Simplified two‑phase optimizer for the step‑function bound.

The first phase builds a short pool of analytic candidates (uniform,
beta‑shapes, a global starter if provided) and keeps the few best.
The second phase performs a mirror‑descent (exponentiated‑gradient)
optimisation on the probability simplex using the exact sub‑gradient
of the objective 2·n·max(conv(a,a))/sum(a)².  The sub‑gradient is
computed from the active set of convolution positions attaining the
maximum and includes the denominator term (−M·a) which yields a more
accurate descent direction than the plain EG used previously.  The
method runs until the time budget expires and returns the best vector
found (as a plain list of floats)."""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))

def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _interpolate(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `arr` to length `new_n` by linear interpolation, keep sum=1."""
    old_n = arr.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    new_arr = np.interp(new_x, old_x, arr)
    s = new_arr.sum()
    if s > 0.0:
        new_arr /= s
    else:
        new_arr = np.full(new_n, 1.0 / new_n)
    return new_arr

def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated‑gradient) descent on the simplex.
    Uses the true sub‑gradient of the objective:
        g = avg_{k∈Active} a_{k‑i}  –  M·a_i .
    The update is a_i ← a_i·exp(−η·g_i) followed by renormalisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    eps_floor = 1e-14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # active set (indices achieving the maximum up to tiny tol)
        tol = M * 1e-12
        ks = np.where(conv >= M - tol)[0]

        # compute average contribution of a_{k‑i}
        avg_contrib = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            avg_contrib[idx] += a[k - idx]
        avg_contrib /= float(len(ks))

        # sub‑gradient of the objective (ignoring the constant factor 4·n)
        grad = avg_contrib - M * a

        # step size schedule (η_t = η0 / sqrt(t))
        eta = eta0 / math.sqrt(it)

        # multiplicative (mirror) update
        a = a * np.exp(-eta * grad)

        # enforce a tiny floor then renormalise
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing polishing (as in the previous
    baseline).  Tiny packets of mass are moved between random positions;
    a Metropolis rule with an exponential cooling schedule decides acceptance.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # temperature schedule: from 0.02 down to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    The routine builds a modest pool of analytic shapes, refines the
    best one with mirror‑descent on the simplex and finishes with a
    short simulated‑annealing polish.  It respects the supplied time
    budget (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length (power of two for fast FFTs)
    # ------------------------------------------------------------------
    n = 32768   # 2¹⁵ – a good trade‑off between resolution and speed

    # ------------------------------------------------------------------
    # 2) Assemble a small candidate pool
    # ------------------------------------------------------------------
    pool = []          # list of (score, vector)
    pool_size = 7

    # a) global starter, if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # crop / pad to length n
            if starter.size != n:
                starter = _interpolate(starter, n)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                pool.append((sc, starter.copy()))
    except Exception:
        pass

    # b) uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # c) a few random Beta‑distribution shapes
    for _ in range(20):
        a = 10 ** rng.uniform(-2.0, 0.7)   # roughly 0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # d) random Dirichlet‑type vectors (sparse / dense)
    for _ in range(10):
        # Dirichlet with concentration parameter drawn log‑uniformly
        conc = 10 ** rng.uniform(-2.0, 0.5)   # 0.01 … 3
        v = rng.gamma(conc, 1.0, size=n)
        v = np.maximum(v, 0.0)
        if v.sum() > 0.0:
            v /= v.sum()
            pool.append((_objective(v), v.copy()))

    # keep only the best `pool_size` candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:pool_size]

    best_score, best_seq = pool[0]

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement (allow several restarts if time permits)
    # ------------------------------------------------------------------
    # allocate at most 70 % of the remaining time to the exploratory stage
    remaining = deadline - time.time()
    exploratory_deadline = time.time() + 0.70 * remaining

    # run a short mirror‑descent from each pooled candidate (time‑sliced)
    for idx, (sc, vec) in enumerate(pool):
        now = time.time()
        if now >= exploratory_deadline:
            break
        # give each run at most 30 s (or less if remaining is small)
        run_deadline = min(now + 30.0, exploratory_deadline)
        a_opt, sc_opt = _mirror_descent(vec, run_deadline, start_time, eta0=0.8)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after MD from pool {idx} = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final intensive refinement on the current best vector
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:      # keep a few seconds for polishing
        final_deadline = deadline - 5.0
        best_seq, best_score = _mirror_descent(best_seq,
                                                final_deadline,
                                                start_time,
                                                eta0=0.9)
        print(f"[{time.time() - start_time:.2f}s] After final MD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _interpolate(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `arr` to length `new_n` by linear interpolation, preserve sum=1."""
    old_n = arr.shape[0]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_arr = np.interp(new_x, old_x, arr)
    s = new_arr.sum()
    if s > 0.0:
        new_arr /= s
    else:
        new_arr = np.full(new_n, 1.0 / new_n)
    return new_arr


# ----------------------------------------------------------------------
# Mirror‑descent on the simplex (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated‑gradient) descent on the simplex.
    Sub‑gradient:  g_i = avg_{k∈Active} a_{k‑i} – M·a_i.
    Update: a_i ← a_i·exp(−η·g_i) followed by renormalisation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    eps_floor = 1e-14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        tol = M * 1e-12
        ks = np.where(conv >= M - tol)[0]

        # average contribution of a_{k‑i} over the active set
        avg_contrib = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            avg_contrib[idx] += a[k - idx]
        avg_contrib /= float(len(ks))

        # sub‑gradient (ignoring the constant factor 4·n)
        grad = avg_contrib - M * a

        # step‑size schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # enforce positivity and renormalise
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (kept from the previous baseline)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly move tiny packets of mass; accept with Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Builds a small pool of analytic candidates, refines the best one
    with mirror‑descent on the simplex and finishes with a short
    simulated‑annealing polish.  The routine respects the supplied time
    budget (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length for fast FFTs
    # --------------------------------------------------------------
    n = 32768   # 2¹⁵ – reasonable speed / resolution trade‑off

    # --------------------------------------------------------------
    # 2) Build a short pool of promising candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)
    pool_size = 7

    # Global starter, if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                starter = _interpolate(starter, n)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                pool.append((sc, starter.copy()))
    except Exception:
        pass

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # Random Beta‑distribution shapes
    for _ in range(20):
        a = 10 ** rng.uniform(-2.0, 0.7)   # roughly 0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # Random Dirichlet‑type vectors (sparse / dense)
    for _ in range(10):
        conc = 10 ** rng.uniform(-2.0, 0.5)   # 0.01 … 3
        v = rng.gamma(conc, 1.0, size=n)
        v = np.maximum(v, 0.0)
        if v.sum() > 0.0:
            v /= v.sum()
            pool.append((_objective(v), v.copy()))

    # Keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:pool_size]

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Mirror‑descent refinement (allow several short runs)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    exploratory_deadline = time.time() + 0.70 * remaining

    for idx, (sc, vec) in enumerate(pool):
        now = time.time()
        if now >= exploratory_deadline:
            break
        run_deadline = min(now + 30.0, exploratory_deadline)
        a_opt, sc_opt = _mirror_descent(vec, run_deadline, start_time, eta0=0.8)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after MD from pool {idx} = {sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Final intensive refinement on the current best vector
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:      # leave a few seconds for polishing
        final_deadline = deadline - 5.0
        best_seq, best_score = _mirror_descent(best_seq,
                                                final_deadline,
                                                start_time,
                                                eta0=0.9)
        print(f"[{time.time() - start_time:.2f}s] After final MD best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
