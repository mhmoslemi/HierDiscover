The evaluator computes  

\[
J(a)=\frac{2n\;\max_k (a*a)_k}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

The objective is scale‑invariant, therefore we can always enforce  

\[
\sum_i a_i = 1 .
\]

The problem

\[
\min_{a\ge0,\;\sum a_i=1}\max_k (a*a)_k
\]

is **convex** after the change of variables \(x_i=\log a_i\) :

* \(\displaystyle a_i=\exp(x_i)>0\)  
* \(\displaystyle \sum_i\exp(x_i)=1\) (convex)  
* for every shift \(k\)

  \[
  (a*a)_k=\sum_{i}\exp(x_i+x_{k-i})\le M ,
  \]

  a sum of exponentials of affine functions – also convex.

Hence we can solve a small version of the problem (few hundred variables) with a standard convex
solver (CVXPY + SCS).  
The optimal small‑scale shape is then linearly interpolated to the required length
\(n=4096\).  This gives a very good starting point for a fine‑grained
continuous optimisation.

The remaining optimisation uses the same four‑stage pipeline as the reference implementation,
but with a few improvements:

1. **Parametric sweep** – shortened, only the most promising families are scanned.
2. **Random Dirichlet draws** – many more draws, a wider concentration range.
3. **Smooth‑max Adam** – starts with a low temperature (smooth surrogate) and raises it slowly; the learning rate is adapted.
4. **Hill‑climb / simulated‑annealing** – allows larger mass transfers (up to 2 % of the source) and uses a simple Metropolis acceptance rule.

All stages respect the 1000 s time budget; the best sequence found is returned
as a plain Python list.

```python
"""
Step‑function optimiser for the “max‑convolution” challenge.

Strategy
--------
1.  Solve the convex formulation (log‑variables) for a modest size
    (n_small = 200).  The global optimum of the relaxed problem is obtained
    with CVXPY + SCS.
2.  Upsample the optimal small solution to the required length n=4096
    (linear interpolation) and normalise it.
3.  Run a short parametric sweep (geometric, power, beta, stretched‑exp,
    logistic) to possibly improve the start.
4.  Perform many random Dirichlet initialisations.
5.  Optimise a smooth‑max surrogate with Adam (temperature τ starts at 5,
    grows geometrically, learning rate is annealed).
6.  Final directed hill‑climb / simulated‑annealing that moves up to 2 % of
    the source mass per step.
7.  Return the best sequence (sum = 1, all entries ≥ 0).

All helper functions are top‑level; the routine respects the 1000 s budget
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """
    Score used by the optimiser: 2·n·max(conv(seq,seq))   (sum(seq)=1).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, stretched‑exp, logistic)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """A lightweight sweep over a few classical families."""
    idx = np.arange(n, dtype=np.float64)

    # uniform (baseline)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform  score={sc:.6f}")

    # geometric (decaying)
    for eps in np.logspace(-5, -1, 80):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        # mixture with uniform helps to avoid very sharp spikes
        for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} λ={lam:.2f} score={sc:.6f}")

    # power law
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # beta‑type (symmetric)
    alphas = np.linspace(0.1, 4.0, 31)
    betas = alphas  # enforce symmetry to keep the search cheap
    for a in alphas:
        if time.time() >= deadline:
            break
        for b in betas:
            if time.time() >= deadline:
                break
            # log‑space construction for numerical stability
            log_seq = (a - 1.0) * np.log(idx + 1.0) + (b - 1.0) * np.log(np.maximum(n - idx, 1.0))
            seq = np.exp(log_seq - np.max(log_seq))
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")

    # stretched‑exponential
    for scale in np.linspace(0.3, 1.0, 7):
        if time.time() >= deadline:
            break
        for p in np.linspace(0.4, 5.0, 31):
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} s={scale:.2f} score={sc:.6f}")

    # logistic (sigmoid‑shaped)
    for mu_f in np.linspace(0.2, 0.8, 9):
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in np.linspace(0.02, 0.15, 7):
            if time.time() >= deadline:
                break
            s_val = s_f * n
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic μ={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (wide concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 1500                     # many draws – cheap to evaluate
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # draw a concentration parameter from a log‑uniform interval
        log_c = rng.uniform(np.log(0.02), np.log(15.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f}  score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Smooth‑max minimisation using Adam.
    τ (temperature) starts low (smooth) and grows geometrically.
    Learning rate decays slowly.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.06               # initial step size
    tau = 5.0               # start temperature (smooth surrogate)
    tau_factor = 1.02       # slow increase
    it = 0

    while time.time() < deadline:
        it += 1

        # current (exact) score – useful for early stopping
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,}  score={cur_score:.6f}")

        # ---- smooth‑max weights ------------------------------------------------
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # ---- gradient of the smooth surrogate ----------------------------------
        a_rev = a[::-1]
        # convolution w * a_rev gives the needed sums
        conv_w = _conv_fft(w, a_rev)          # length 3·n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]  # keep the central n components

        # ---- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # ---- temperature & learning‑rate schedules -----------------------------
        if it % 500 == 0:
            tau = min(tau * tau_factor, 1e9)          # make the surrogate sharper
            lr = lr * 0.995                           # gentle decay

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that moves a small (but now up to 2 %)
    fraction of mass from the index with the largest positive sub‑gradient
    to the one with the most negative gradient.  A simple Metropolis rule
    allows occasional uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule (decays to ~0)
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.03
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source: index with positive gradient and non‑negligible mass
        src_candidates = np.where((a > 1e-12) & (grad > 0))[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]

        # destination: most negative gradient (could be the same as src)
        dst = np.argmin(grad)

        if src == dst:
            # fallback to a random pair
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer up to 2 % of the source mass
        frac = rng.random() * 0.019 + 0.001   # 0.1 % … 2 %
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        delta = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        accept = False
        if delta < -1e-12:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Convex solve for a small problem (log‑variables)
# ----------------------------------------------------------------------
def _solve_convex_small(n_small: int, deadline: float) -> np.ndarray | None:
    """
    Solve   min M
            s.t. Σ exp(x_i) = 1,
                 Σ exp(x_i + x_{k-i}) ≤ M   for all k,
                 x ∈ ℝⁿ .
    The solution a_i = exp(x_i) is globally optimal for the given n_small.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print(f"[{time.time() - start_time:.2f}s] CVXPY not available: {e}")
        return None

    print(f"[{time.time() - start_time:.2f}s] Solving convex problem (n={n_small}) …")
    x = cp.Variable(n_small)
    a_exp = cp.exp(x)                 # a_i = e^{x_i}
    M = cp.Variable()

    constraints = [
        cp.sum(a_exp) == 1,
        M >= 0
    ]

    # build the convolution–type constraints
    for k in range(2 * n_small - 1):
        lo = max(0, k - (n_small - 1))
        hi = min(n_small - 1, k)
        exps = [cp.exp(x[i] + x[k - i]) for i in range(lo, hi + 1)]
        constraints.append(cp.sum(cp.hstack(exps)) <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)

    # give the solver a modest time‑budget
    max_iters = 4000
    try:
        prob.solve(solver=cp.SCS,
                   max_iters=max_iters,
                   eps=1e-4,
                   warm_start=True,
                   verbose=False)
    except Exception as e:
        print(f"[{time.time() - start_time:.2f}s] CVXPY solve failed: {e}")
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        print(f"[{time.time() - start_time:.2f}s] CVXPY status: {prob.status}")
        return None

    a = np.exp(x.value)
    if a.sum() == 0 or not np.all(np.isfinite(a)):
        return None
    a /= a.sum()
    return a


def _upsample_to_n(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from len(a_small) to n_large (keeps positivity)."""
    xs = np.linspace(0.0, 1.0, a_small.size)
    xl = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(xl, xs, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n_large = 4096
    L_fft_large = _next_pow2(2 * n_large - 1)

    # ------------------------------------------------------------------
    # Initialise with uniform distribution (or a provided starter)
    # ------------------------------------------------------------------
    uniform = np.full(n_large, 1.0 / n_large)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline  score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter  score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 5️⃣ Convex optimisation for a small instance (if time permits)
    # ------------------------------------------------------------------
    n_small = 200
    small_deadline = deadline - 250.0           # reserve ~250 s for later stages
    if time.time() < small_deadline:
        a_small = _solve_convex_small(n_small, small_deadline)
        if a_small is not None:
            a_large = _upsample_to_n(a_small, n_large)
            sc = _objective_score(a_large)
            if sc < best_score:
                best_score, best_seq = sc, a_large.copy()
                print(f"[{time.time() - start_time:.2f}s] Convex‑upsample score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Light parametric sweep (shortened to stay within budget)
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 200.0            # keep ~200 s for later stages
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_families(
            n_large, best_seq, best_score, sweep_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # ------------------------------------------------------------------
    dirichlet_deadline = deadline - 130.0
    if time.time() < dirichlet_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n_large, best_seq, best_score, dirichlet_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation of the smooth‑max surrogate
    # ------------------------------------------------------------------
    hill_time = 130.0
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, adam_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb / simulated‑annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
