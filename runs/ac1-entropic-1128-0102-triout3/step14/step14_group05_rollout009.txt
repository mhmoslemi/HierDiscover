def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

step_search_v2.py – Multi‑phase optimisation for the step‑function challenge.

The algorithm works under a strict 1000 s wall‑clock budget and proceeds
through four independent phases:

  1️⃣  Global random exploration:  Dirichlet‑draws with a wide range of
      concentrations plus a dense parametric sweep over geometric‑power

  2️⃣  Smooth‑max surrogate optimisation:  the same ``_SurrogateMax``

  3️⃣  Simulated‑annealing hill‑climb on the **exact** objective.
      Small‑mass transfers between random coordinates are accepted
      according to a decreasing temperature, allowing the search to

  4️⃣  Final projection onto the probability simplex and return of the

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (full length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)
    assuming ``a`` is non‑negative.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:                     # protection – should never happen
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                  # fallback – should never be needed
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable that returns (value, gradient) of the smooth‑max surrogate
    of ``max(conv(a, a))`` where ``a = softmax(z)``.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # FFT length for a*a
        self.L_corr = L_corr          # FFT length for w * a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        # -------- self‑convolution ---------------------------------
        conv = _conv_self_fft(a, self.L_conv)   # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()          # smooth‑max weights (len 2n‑1)

        # surrogate value = max_c + (1/β)·log Σ exp(β·(conv−max_c))
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # -------- gradient w.r.t. a ---------------------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        # slice (n‑1 … 0) and reverse -> gradient of max(conv) w.r.t a
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]  # shape (n,)

        # -------- chain rule through soft‑max -----------------------
        dot = np.dot(grad_a, a)                 # scalar ⟨grad_a , a⟩
        grad_z = a * (grad_a - dot)             # ∂value/∂z_i
        return surrogate, grad_z

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("vector expected")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
