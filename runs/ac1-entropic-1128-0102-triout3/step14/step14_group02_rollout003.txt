assistantfinal**Solution Overview**

The task is to return a non‑negative vector `a` (the *heights* of a step function) that minimises  

\[
\mathrm{score}(a)=\frac{2n\,\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear self‑convolution and `n=len(a)`.  
The score is *scale‑invariant*; we therefore keep every candidate on the probability
simplex (`∑a = 1`).  

The programme below follows a **four‑phase** search strategy that fits into the
1000 s budget:

1. **Random Dirichlet initialisers** – a very cheap way of exploring the simplex.
2. **Multiple Adam‑optimisations** of a *smooth‑max* surrogate  
   \(\operatorname{softmax}_\tau (c)=\frac1\tau\log\!\sum_i e^{\tau c_i}\)  
   of the maximum of the convolution vector `c`.  The surrogate becomes the true
   max as the temperature `τ` grows, so we start with a low temperature (highly
   smooth) and increase it geometrically.
3. **Stochastic hill‑climbing / simulated‑annealing** – random mass transfers
   between two indices are accepted if they improve the true objective,
   otherwise with a Metropolis probability that decreases with a temperature
   schedule.  This phase is able to escape the plateaus where the gradient based
   methods stall.
4. **Final projection** – the best vector found is projected onto the simplex,
   converted to a plain Python list and returned.

All heavy operations (convolutions, gradients) use the FFT for `O(n log n)`
complexity; a vector length of `n = 4096` gives a good trade‑off between
expressiveness and speed.  The code prints intermediate scores together with the
elapsed time, so the progress can be monitored while the search runs.

```python
"""
A fast 4‑stage optimiser for the “max‑convolution” score.

1. many random Dirichlet draws (cheap global exploration);
2. several restarts of Adam on a smooth‑max surrogate of the max‑convolution;
3. stochastic hill‑climbing / simulated annealing to break plateaus;
4. final simplex projection and return.

The implementation works for any n (default 4096) and stays inside the 1000 s
budget.
"""
import time
import numpy as np

# --------------------------------------------------------------
# 0.  Global timing (used by the helper functions)
# --------------------------------------------------------------
start_time = None   # will be set inside propose_candidate()


# --------------------------------------------------------------
# 1.  Low‑level linear‑algebra helpers
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (FFT implementation)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x≥0, ∑x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# 2.  Objective helpers
# --------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """
    Exact objective (sum(a)=1 is assumed).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
        s(a) = (1/tau)·log Σ_i exp(tau·c_i)
    where c = a*a (full linear convolution).
    The gradient is 2·(w ⊛ rev(a)) on the original window,
    with w being the soft‑max weights of the entries of c.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum

    # convolution of the weights with the reversed a
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev)                     # length = 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad


# --------------------------------------------------------------
# 3.  Adam optimiser on the smooth‑max surrogate
# --------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float) -> tuple:
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature τ starts small (smooth) and is multiplied by tau_factor
    every 1000 iterations, steering the surrogate towards the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.1               # a slightly larger base step
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 0.5                 # start very smooth
    tau_factor = 1.2
    it = 0

    while time.time() < deadline:
        it += 1

        # exact score (for bookkeeping)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}"
            )

        # gradient of the smooth‑max surrogate
        grad = _smooth_max_gradient(a, tau)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)        # keep non‑negative
        a_new = _project_simplex(a_new)       # enforce sum‑to‑1

        a = a_new

        # increase smoothness every 1000 steps
        if it % 1000 == 0:
            tau = min(tau * tau_factor, 1e4)

    return best_seq, best_score


# --------------------------------------------------------------
# 4.  Stochastic hill‑climbing / simulated annealing
# --------------------------------------------------------------
def _stochastic_hill_climb(a_start: np.ndarray, deadline: float,
                           best_seq: np.ndarray, best_score: float) -> tuple:
    """
    Random mass‑transfer moves accepted according to a Metropolis rule.
    The temperature decays geometrically, allowing early uphill jumps and later
    greedy refinement.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # initialise temperature from the current gap to the trivial bound (2)
    T = max(best_score - 1.0, 0.1) * 0.2
    decay = 0.9997
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1

        # choose a source index with some mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # destination can be any other index
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # transfer a random fraction of the source mass
        max_frac = min(0.05, T + 1e-6)          # fraction scales with temperature
        delta = a[src] * rng.uniform(0, max_frac)
        if delta <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # numerical safety
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA improvement score={best_score:.6f}"
                )
        else:
            # accept uphill move with Metropolis probability
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_prop
                cur_score = new_score

        # temperature decay
        T *= decay
        if it % 10000 == 0:
            # occasional reheating to avoid complete stagnation
            T = max(T, 0.001)

    return best_seq, best_score


# --------------------------------------------------------------
# 5.  Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation score.
    The method consists of:
      * random Dirichlet sampling,
      * several Adam‑optimisations on a smooth‑max surrogate,
      * a final stochastic hill‑climb (simulated annealing).
    Returns a plain list of floats ready for `evaluate_sequence`.
    """
    global start_time
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size (4096 is a good compromise between expressiveness
    # and speed; the code works for other n as well)
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(
        f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # optional global starter (if the notebook supplies one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}"
                    )
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet draws (the cheap, broad exploration)
    # ------------------------------------------------------------------
    dirichlet_deadline = start_time + 200.0      # spend ~200 s here
    while time.time() < dirichlet_deadline:
        # concentration drawn on a log‑scale → both sparse and dense vectors
        logc = rng.uniform(-3.0, 0.8)            # c ∈ [0.001, 6.3]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Multiple Adam restarts (smooth‑max continuation)
    # ------------------------------------------------------------------
    # allocate the remaining time, reserving ~250 s for the final SA phase
    remaining = start_time + budget_s - 250.0 - time.time()
    max_restarts = 6
    for r in range(max_restarts):
        if time.time() >= start_time + budget_s - 250.0:
            break
        # initialise either from the current best or a fresh Dirichlet draw
        if r == 0:
            a0 = best_seq.copy()
        else:
            # a fresh random point
            logc = rng.uniform(-2.5, 0.5)
            c = 10 ** logc
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
        # give each restart roughly equal time
        run_deadline = time.time() + remaining / max_restarts
        best_seq, best_score = _adam_optimize(
            a0, run_deadline, best_seq, best_score
        )
        # recompute remaining budget
        remaining = start_time + budget_s - 250.0 - time.time()

    # ------------------------------------------------------------------
    # 3️⃣  Stochastic hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    sa_deadline = start_time + budget_s - 0.05   # tiny safety margin
    if time.time() < sa_deadline:
        print(
            f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing hill‑climb"
        )
        best_seq, best_score = _stochastic_hill_climb(
            best_seq, sa_deadline, best_seq, best_score
        )
        print(
            f"[{time.time() - start_time:.2f}s] Final SA score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
```
