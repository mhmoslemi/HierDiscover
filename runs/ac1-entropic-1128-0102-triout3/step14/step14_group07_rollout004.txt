    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

        # tolerance relative

        idxs = np.where(conv >= maxv - tol)[0]

"""
Mirror‑descent optimisation for the max‑convolution bound.

The algorithm works on a modest discretisation (n≈8‑k) to allow many
iterations within the 1000 s budget.

1. Build a small pool of analytic shapes (uniform, Beta(α) and a few
   random Dirichlet vectors).  Keep the one with the lowest objective.
2. Run a projected *exponentiated‑gradient* (mirror descent on the
   simplex) that directly minimises the nonsmooth objective
   f(a)=maxₖ (a*a)[k].  At each iteration we:
   * compute the full convolution using a real FFT,
   * locate a shift attaining the current maximum,
   * form the sub‑gradient gₖ (which is just a shifted copy of a),
   * update a ← a·exp(−η·gₖ) and renormalise.
   Step size η decays as η₀/√t.
   The best vector seen so far is recorded.
3. When time permits a brief simulated‑annealing polish (mass swaps)
   is performed to squeeze a little more out of the result.
4. The final vector is normalised to sum‑to‑one and returned as a plain
   Python list.  All entries are non‑negative, so the provided
   ``evaluate_sequence`` will accept it.

The mirror‑descent step works directly on the original nonsmooth
objective (no temperature smoothing) and therefore converges faster
than the previous Adam‑based smoother‑max approach.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full convolution a * a using a real FFT.
    Returns a length 2·len(a)‑1 array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(a,a)) for a simplex‑normalised vector.
    Since we keep a.sum() == 1, this equals the value returned by
    ``evaluate_sequence``.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families for initialisation
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1},
    normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the simplex
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on f(a)=max_k conv_k(a).
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and maximal shift
        conv = _conv_self(a)
        max_val = conv.max()
        # gather all shifts attaining the maximum (within tiny tolerance)
        tol = max_val * 1e-12 + 1e-15
        idxs = np.where(conv >= max_val - tol)[0]
        k = rng.choice(idxs)

        # sub‑gradient for shift k :  g_i = 2·a_{k-i}  (zero outside range)
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i = np.arange(low, high + 1)
        g[i] = 2.0 * a[k - i]

        # step size decay
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (mirror descent w.r.t. KL)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # keep best
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] MD iter={it:,} "
                  f"best={cur_score:.6f}")

        # occasional tiny stochastic perturbation to avoid stagnation
        if it % 5000 == 0 and rng.random() < 0.15:
            jitter = rng.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            total = a.sum()
            if total > 0:
                a /= total
            else:
                a = np.full(n, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that repeatedly moves a random fraction of mass from one
    index to another.  Accepts moves that improve the objective or that
    are accepted by a temperature schedule.
    """
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)                 # source weighted by current mass
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (sums to 1).
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 123456)

    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a modest discretisation size – small enough for many
    # iterations, large enough to give a smooth shape.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 8192
    elif remaining > 400:
        n = 4096
    else:
        n = 2048

    # ------------------------------------------------------------------
    # 1) generate a handful of diverse seed vectors and keep the best
    # ------------------------------------------------------------------
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # a few symmetric Beta densities (U‑shaped for α<1, bell for α>1)
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]:
        seeds.append(_make_beta_shape(n, alpha))

    # random Dirichlet with a spread of concentration parameters
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        a = rng.gamma(conc, 1.0, size=n)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # incorporate the global starter if it exists (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample / interpolate to required size
            if starter.size != n:
                if n % starter.size == 0:
                    factor = n // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, n, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                seeds.append(starter_up)
    except Exception:
        pass

    # pick the currently best seed (lowest objective)
    best_score = float("inf")
    best_vec = None
    for vec in seeds:
        scr = _objective(vec)
        if scr < best_score:
            best_score = scr
            best_vec = vec.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Mirror‑descent refinement (main workhorse)
    # ------------------------------------------------------------------
    # allocate the bulk of the remaining time to this phase
    md_deadline = deadline - 5.0  # reserve a few seconds for polishing
    refined_vec, refined_score = _mirror_descent(best_vec,
                                                 md_deadline,
                                                 start_time,
                                                 rng,
                                                 eta0=1.0)
    if refined_score < best_score:
        best_score, best_vec = refined_score, refined_vec.copy()
        print(f"[{time.time() - start_time:.2f}s] Mirror‑descent improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional short simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        best_vec, best_score = _simulated_annealing(best_vec,
                                                    deadline,
                                                    start_time,
                                                    rng,
                                                    max_move_frac=0.02)

    # ------------------------------------------------------------------
    # 4) final normalisation and conversion to Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(n, 1.0 / n)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
