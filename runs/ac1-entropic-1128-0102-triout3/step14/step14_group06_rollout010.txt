"""
Step–function optimiser (≈ 1 000 s budget).

The optimiser works in three stages:

1️⃣  **Initial diverse pool** – deterministic shapes (uniform,
    symmetric/asymmetric Beta, power‑ramps, Gaussian and exponential
    windows) and a modest number of Dirichlet draws.  The best `TOP_K`
    candidates are kept.

2️⃣  **Random‑restart fast sub‑gradient phase** – many short
    exponentiated sub‑gradient runs (≈ 300 – 800 iterations) from
    fresh Dirichlet starts.  Whenever a better sequence is found it
    replaces the current global best.

3️⃣  **Long refinement** – a lengthy exponentiated sub‑gradient descent
    (full‑budget) from the best sequence followed by a smooth‑max
    Adam polishing step.

All vectors are kept on the probability simplex, the objective
`2·n·max(conv(a,a))/sum(a)^2` is evaluated by the supplied
`evaluate_sequence` (the implementation here uses the same exact
formula).  The algorithm adapts the problem size (`N`) to the time
budget: a smaller `N` gives more iterations, a larger `N` gives a
finer discretisation.

The returned list contains non‑negative floats and respects the
required format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (all normalised to sum 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “power‑ramp” (optional flat plateau)."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_shape(n: int, decay: float) -> np.ndarray:
    """Monotone exponential decay from the left edge."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-decay * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (fixed‑iteration variant)
# ----------------------------------------------------------------------
def _exp_subgradient_descent_fixed(a0: np.ndarray,
                                   max_iters: int) -> tuple[np.ndarray, float]:
    """Short exponentiated sub‑gradient run (no time check)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    eta0 = 0.8
    for it in range(1, max_iters + 1):
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 2000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Full exponentiated sub‑gradient descent (time‑bounded)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Long run version that stops at a given wall‑clock deadline."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        eta0 = 0.8
        eta = eta0 / (1.0 + it / 2000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05
    T0 = 0.02
    T_end = 1e-4
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # temperature schedule (linear anneal)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of the smooth‑max term
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate N gives many iterations while still
    # being fine enough to represent useful shapes.
    # --------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool
    # --------------------------------------------------------------
    TOP_K = 12
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # deterministic families
    _add(_make_uniform(N))

    for alpha in np.linspace(0.05, 1.5, 30):
        _add(_make_beta_shape(N, alpha))

    alphas = [0.2, 0.5, 0.8, 1.2]
    betas = [0.2, 0.5, 0.8, 1.2]
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    for p in np.linspace(0.2, 3.0, 12):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add(_make_shape(N, p, plateau))

    # Gaussian windows (different sigmas)
    for sigma_frac in [0.05, 0.10, 0.20, 0.30]:
        sigma = sigma_frac * N
        _add(_make_gaussian_shape(N, sigma))

    # Exponential decays (different rates)
    for decay in [0.0005, 0.001, 0.002, 0.004]:
        _add(_make_exp_shape(N, decay))

    # Random Dirichlet draws (both dense and sparse)
    rng = np.random.default_rng(seed + 12345)
    for conc in [0.2, 0.4, 0.8, 1.5, 5.0]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s > 0:
                a /= s
                _add(a)

    # Incorporate a possible global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _add(starter_up)
    except Exception:
        pass

    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Random‑restart short sub‑gradient phase
    # --------------------------------------------------------------
    max_restart_iters = 800   # length of each quick descent
    restart_deadline = start_time + (deadline - start_time) * 0.35
    while time.time() < restart_deadline:
        # draw a fresh random start
        conc = 10 ** np.random.uniform(-2, 0)   # concentration in [0.01,1]
        a = rng.gamma(conc, 1.0, size=N)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        seq, sc = _exp_subgradient_descent_fixed(a, max_restart_iters)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 3) Long exponentiated sub‑gradient descent from the best so far
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        seq, sc = _exp_subgradient_descent(best_seq, deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Long ExpGD improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 4) Smooth‑max Adam polishing for any remaining time
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq, sc = _smooth_gradient_descent(best_seq, deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq.tolist()]
