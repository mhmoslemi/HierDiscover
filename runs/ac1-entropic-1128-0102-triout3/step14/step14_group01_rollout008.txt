def _hill_climb(a0, best_seq, best_score, deadline):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    cur_score = 2.0 * n * _conv_self(a).max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
    rng = np.random.default_rng()
    # Temperature schedule linear decay from T0 to near 0
    T0 = 0.02
    total = max(0.0, deadline - time.time())
    decay = (1e-6 / T0) ** (1.0 / total) if total>0 else 1.0
    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
        frac = rng.random() * 0.005 + 1e-6 # up to .5%
        eps = a[src] * frac
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if delta < -1e-12:
            accept = True
        else:
            prob = math.exp(-delta / max(T,1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill climb new best {best_score:.6f}")
    return best_seq, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  

The optimiser works in three stages:

1.  **Random Dirichlet initialisation** – a quick Monte‑Carlo sweep
    over many concentration parameters to find a good starting point.
2.  **Smooth‑max + variance Adam optimisation** – minimise a
    differentiable surrogate

        J(a) =  (1/τ)·log Σ exp(τ·(a*a))
               + λ·∑(a*a−μ)² ,

    where μ = 1/(2·n−1) is the average of the convolution.
    τ starts small (smooth) and is increased, while λ grows
    from 0 to a few units, encouraging the convolution to become
    flat.  The gradient of both terms is computed by FFT‑based
    convolutions; Adam updates are projected onto the probability
    simplex after each step.
3.  **Directed hill‑climb / simulated annealing** – tiny mass transfers
    from indices with a large positive sub‑gradient to those with a
    large negative one, with a simple temperature schedule to escape
    shallow local minima.

All stages respect the given time budget (default 1000 s).  The final
sequence is normalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT (full length).
    Returns an array of length len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.ifft(X * Y).real[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.ifft(Af * Af).real[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score used internally: 2·n·max(conv(seq,seq)) (seq already sums to 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (non‑smooth) max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _grad_smoothmax(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of (1/τ)·log Σ exp(τ·conv).
    """
    n = a.shape[0]
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))      # numerically stable
    w = w_raw / w_raw.sum()
    # grad = 2·(w * a_rev)  (restricted to length n)
    grad = 2.0 * _conv_fft(w, a[::-1])[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the variance term  Σ (conv‑μ)²
# ----------------------------------------------------------------------
def _grad_variance(a: np.ndarray, conv: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of Σ (conv‑μ)² .  (μ is constant – the mean of the convolution.)
    """
    n = a.shape[0]
    w = conv - mu
    grad = 4.0 * _conv_fft(w, a[::-1])[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet sampling (wide concentration range)
# ----------------------------------------------------------------------
def _dirichlet_trials(a0: np.ndarray,
                      best_seq: np.ndarray,
                      best_score: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # log‑uniform concentration in [0.05, 10]
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = math.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=a0.shape[0])
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (the best sequence found).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 proved effective in the reference run.
    # ------------------------------------------------------------------
    n = 4096
    mu = 1.0 / (2 * n - 1)                # average of the full convolution

    # ------------------------------------------------------------------
    # Initial candidate – uniform distribution.
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n)
    best_seq = a.copy()
    best_score = _objective_score(a)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optionally import a global starter (height_sequence_1) and normalise it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size))
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq, a = sc, starter.copy(), starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – cheap Dirichlet sampling (≈30 s)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 30.0)
    if time.time() < phase1_deadline:
        best_seq, best_score = _dirichlet_trials(a, best_seq, best_score, phase1_deadline)
        a = best_seq.copy()

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimiser on smooth‑max + variance.
    # Reserve ~50 s for a final hill‑climb, the rest for Adam.
    # ------------------------------------------------------------------
    hill_time = 50.0
    phase2_deadline = deadline - hill_time
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.05                     # base step size
        it = 0
        while time.time() < phase2_deadline:
            it += 1
            conv = _conv_self(a)      # full self‑convolution
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,}  score={cur_score:.6f}")

            # ----- schedule for temperature τ and variance weight λ -----
            progress = (time.time() - start_time) / (phase2_deadline - start_time)
            # τ grows from ~1 to a few hundred
            tau = 1.0 + 300.0 * (progress ** 1.5)
            # λ grows from 0 to about 3.0
            lam = 3.0 * progress

            # ----- gradients -----
            grad_s = _grad_smoothmax(a, conv, tau)
            grad_v = _grad_variance(a, conv, mu)
            grad = grad_s + lam * grad_v

            # ----- Adam update & simplex projection -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)

            a = a_new

    # ------------------------------------------------------------------
    # Phase 3 – directed hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        # start from the best point discovered so far
        a_hc = best_seq.copy()
        cur_score = 2.0 * n * _conv_self(a_hc).max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a_hc.copy()

        # temperature schedule (linear decay)
        T0 = 0.02
        total_hc = max(0.0, deadline - time.time())
        decay = (1e-6 / T0) ** (1.0 / total_hc) if total_hc > 0 else 1.0
        rng = np.random.default_rng()

        while time.time() < deadline:
            conv = _conv_self(a_hc)
            grad = _subgradient_maxconv(a_hc, conv)

            # source = index with largest positive sub‑gradient that actually holds mass
            src_cand = np.where(a_hc > 1e-12)[0]
            if src_cand.size == 0:
                break
            src = src_cand[np.argmax(grad[src_cand])]
            dst = int(np.argmin(grad))

            # occasional random move to avoid stagnation
            if src == dst or rng.random() < 0.05:
                src = rng.integers(n)
                dst = rng.integers(n)
                if src == dst or a_hc[src] < 1e-12:
                    continue

            # transfer a tiny fraction (up to ~0.5 %)
            frac = rng.random() * 0.004 + 1e-6
            eps_move = a_hc[src] * frac
            if eps_move <= 0.0:
                continue

            a_prop = a_hc.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = 2.0 * n * _conv_self(a_prop).max()
            delta = new_score - cur_score
            T = T0 * (decay ** (time.time() - start_time))

            accept = False
            if delta < -1e-12:
                accept = True
            else:
                prob = math.exp(-delta / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a_hc = a_prop
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_hc.copy()
                    print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
