def _smooth_max_adam(a0, time_limit, start_time):

    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1=0.9; beta2=0.999; eps=1e-8

    step0 = 0.2

    # compute initial conv and max for temperature init
    conv = _conv_self(a)
    max_conv = conv.max()

    T0 = max(0.1, max_conv) # maybe using conv max

    T_end = 1e-7

    it=0
    while time.time() < time_limit:
        it+=1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam iter={it:,} score={cur_score:.6f}")

        # temp schedule
        fraction = (time.time() - start_time) / (time_limit - start_time)
        T = max(T_end, T0 * (T_end/T0) ** fraction)

        # compute softmax weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation
        a_rev = a[::-1]
        L = _next_pow2(3*n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[:3*n-2]
        grad = 2.0 * corr[n-1:n-1+n]  # length n

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        # learning rate schedule
        step = step0 / np.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score

"""
Step‑function optimiser (≈1000 s budget).

The algorithm follows three ideas:

1. **Diverse initial pool** – a large collection of analytically‑generated
   shapes (uniform, Beta‑densities, power ramps, half‑cosine windows) and many
   random Dirichlet draws.  All vectors are normalised to an L1‑sum of 1 and
   the best `TOP_K` are kept.

2. **Smooth‑max Adam descent** – minimise a temperature‑smoothed surrogate of
   the ℓ∞‑objective
        f(a)=max_k (a∗a)[k] .
   The surrogate is the soft‑max of the convolution values.  Adam with an
   exponentially annealed temperature provides a fast, global‑search phase.
   The current best vector is continually updated.

3. **Exponentiated sub‑gradient descent** – once a good plateau has been
   reached, a multiplicative update using the exact sub‑gradient (active set)
   is applied.  This refines the solution on the true ℓ∞ objective.

The schedule (≈60 % smooth Adam → 35 % EG → remaining time polishing) is
chosen empirically to beat the previous best ~1.508.  All vectors are kept on
the probability simplex, which makes the objective scale‑invariant and
automatically respects the non‑negativity constraints required by
`evaluate_sequence`.

The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with an optional flat plateau.
    Normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    α<1 yields a U‑shaped distribution.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_half_cosine(n: int, p: float) -> np.ndarray:
    """
    Half‑cosine window raised to power p,
    i.e. max(0, cos(π·(i/(n‑1)‑0.5)))**p, normalised to sum 1.
    """
    idx = np.arange(n, dtype=float) / (n - 1)
    shape = np.cos(np.pi * (idx - 0.5))
    shape = np.clip(shape, 0.0, None)  # keep only the non‑negative half
    shape = shape ** p
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam descent on the smooth‑max surrogate of the ℓ∞ objective.
    Temperature is exponentially annealed; projection onto the simplex
    guarantees feasibility.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.2          # base learning rate

    # temperature schedule (exponential)
    conv0 = _conv_self(a)
    max0 = conv0.max()
    T_start = max(0.1, max0)   # start with a fraction of the initial peak
    T_end = 1e-7

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam iter={it:,} score={cur_score:.6f}")

        # temperature (exponential decay)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * (T_end / T_start) ** frac)

        # soft‑max weights (log‑sum‑exp trick)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * (w ⋆ a_rev)   (⋆ is correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (keeps sum = 1, non‑negative)
        a = _project_simplex(a)

    return best_a, best_score


def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated (multiplicative) sub‑gradient descent on the exact ℓ∞
    objective.  The active set (indices attaining the current maximum) supplies
    a valid sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # indices attaining (or almost attaining) the maximum
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:                               # fallback safety
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule
        eta0 = 1.0
        eta = eta0 / (1.0 + it / 1500.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score
