def _gradient(a, T):
    conv = _conv_self(a)
    max_conv = conv.max()
    if T > 1e-8:
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()
    else:
        # subgradient: uniform over max indices
        eps = 1e-12
        mask = conv >= max_conv - eps
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(float) / cnt
    # compute correlation gradient 2 * (w * a_rev) conv
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)

    corr = np.fft.irfft(W * AR, n=L)
    grad = 2.0 * corr[len(a)-1:len(a)-1+len(a)]

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 ,
\]

where `*` denotes the discrete convolution.  
The objective is homogeneous, therefore we may restrict the search to the
probability simplex  

\[
\mathcal S=\Bigl\{a\in\mathbb R_{\ge0}^n\;\Big|\;\sum_i a_i = 1\Bigr\}.
\]

On the simplex the objective simplifies to  

\[
F(a)=2n\;\max\bigl(a*a\bigr).
\]

The programme below follows three‑stage optimisation:

1. **Diverse initial pool** – a small set of promising candidates formed from
   analytic shapes (uniform, β‑densities, power‑ramps) and from Dirichlet
   draws.  The best *K* vectors are kept.

2. **Projected Adam on a smooth approximation of the max** –  
   we minimise  

   \[
   \tilde F_T(a)=2n\;T\log\!\Bigl(\sum_k\exp\bigl((a*a)_k/T\bigr)\Bigr)
   \]

   with an exponentially decaying temperature `T`.  The gradient of the
   smoothed max is a convolution of the weight vector  
   \(w_k\propto\exp((a*a)_k/T)\) with the reversed sequence.
   Adam’s moment estimates are used and after each step the vector is
   projected back onto the simplex.

   When the temperature becomes tiny (≈ 1e‑9) the weight vector collapses
   to a uniform distribution over the (almost) maximal indices, i.e. we
   switch to a *true* sub‑gradient of the original max‑objective.

3. **Very short simulated‑annealing polish** – after the Adam phase a
   few thousand random mass‑moves are tried; moves that improve the
   real objective (or are accepted by the Metropolis rule) are kept.

All phases respect the 1000 s budget.  The implementation works for any
vector length; a size of **8192** gives a good trade‑off between speed
and resolution.  If more time is available the size is increased
automatically.

The code is self‑contained, uses only NumPy and standard library
functions, and returns a plain list of non‑negative floats.

```python
"""
Propose a step‐function (non‑negative heights) that minimises  

        2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three phases:

1. **Initial pool** – create a handful of analytically‑defined shapes
   (uniform, symmetric β‑densities, power‑ramps) and many random
   Dirichlet vectors.  Keep the K best candidates w.r.t. the true
   objective.

2. **Projected Adam on a smooth‑max surrogate** – minimise
   T·log∑exp(conv/T) with a temperature that decays exponentially from
   0.5 to 1e‑9.  The gradient is obtained by a single FFT‑based
   correlation; after each Adam step the iterate is projected onto the
   probability simplex.  When the temperature is below a tiny threshold
   the weight vector collapses to the (sub‑)gradient of the exact max.

3. **Short simulated‑annealing polish** – random mass swaps are performed
   for the remaining time; moves are accepted if they improve the real
   objective or with the Metropolis probability.

The whole procedure runs inside the given time budget (≤ 1000 s) and
returns a list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of a with itself, using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a vector that already sums to 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad / trim to exactly n elements
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, T: float) -> tuple[np.ndarray, float]:
    """
    Return (grad, max_conv) where grad is a sub‑gradient of
    the smoothed max with temperature T.
    """
    n = a.shape[0]

    # convolution a*a
    conv = _conv_self(a)
    max_conv = conv.max()

    # weight vector w
    if T > 1e-9:
        # smooth: soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()
    else:
        # exact sub‑gradient: uniform over (almost) maximal indices
        eps = 1e-12
        mask = conv >= max_conv - eps
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(float) / cnt

    # correlation w * a_rev gives gradient = 2 * (w ⋆ a_rev)
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    AR = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * AR, n=L)
    grad = 2.0 * corr[n - 1 : n - 1 + n]

    return grad, max_conv


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   T_start: float = 0.5,
                   T_end: float = 1e-9,
                   base_lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max objective.
    Returns the best (vector, score) seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    # evaluate the *exact* objective for bookkeeping
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of smoothed objective
        # temperature decays exponentially with the fraction of elapsed time
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        grad, max_conv = _smooth_gradient(a, T)

        # keep the exact objective if it improves
        cur_score = float(2.0 * n * max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{elapsed:.2f}s] Adam iter={it:,} T={T:.2e} score={cur_score:.6f}")

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (decay with sqrt(iter))
        lr = base_lr / math.sqrt(it)

        # additive update + simplex projection
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """Mass‑swap SA; moves are accepted if they improve the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    # annealing schedule
    T_start = 0.2
    T_end = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        # choose a source index proportionally to its mass
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # amount to move
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose vector length based on the remaining budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    elif remaining > 120:
        N = 16384
    else:
        N = 8192

    # ------------------------------------------------------------------
    # 1) Build a small diverse pool of candidates
    # ------------------------------------------------------------------
    TOP_K = 5
    pool = []                    # list of (score, vector)

    # uniform baseline
    uniform = _make_uniform(N)
    _push_top = lambda p, s, v, k: (_push_top_impl(p, s, v, k))
    def _push_top_impl(p, s, v, k):
        if len(p) < k:
            p.append((s, v.copy()))
            p.sort(key=lambda x: x[0])
        elif s < p[-1][0]:
            p[-1] = (s, v.copy())
            p.sort(key=lambda x: x[0])

    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 7):
        for plat in np.linspace(0.0, 0.5, 5):
            a = _make_power_ramp(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes
    for alpha in np.arange(0.05, 1.0, 0.09):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws with various concentrations
    concs = [0.01, 0.03, 0.07, 0.15, 0.30, 0.60]
    for conc in concs:
        for _ in range(10):
            if time.time() >= deadline:
                break
            draws = rng.gamma(conc, 1.0, size=N)
            if draws.sum() == 0:
                continue
            a = draws / draws.sum()
            _push_top(pool, _objective(a), a, TOP_K)

    # try to incorporate the global best from previous runs if present
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # simple linear interpolation / repetition
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement for each top candidate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15:
        per_candidate = remaining / (len(pool) if len(pool) > 0 else 1)
    else:
        per_candidate = remaining * 0.8

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per_candidate
        if sub_deadline >= deadline:
            break
        refined_vec, refined_score = _adam_optimize(vec, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_vec.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Very short SA polish (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, start_time, rng)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
```
