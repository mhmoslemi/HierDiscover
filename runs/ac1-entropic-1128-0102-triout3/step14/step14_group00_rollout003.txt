        it += 1
        # compute a via softmax
        y_shift = y - np.max(y)   # for numerical stability
        a = np.exp(y_shift)
        a_sum = a.sum()

"""
Optimiser for the self‑convolution bound.

The algorithm works on the probability simplex via a soft‑max
parameterisation.  Adam is used on the log‑odds (the “logits”) with
periodic random perturbations (re‑initialisation of the logits) to
escape possible plateaux.  The objective is the scale‑invariant quantity

    2·n·max(conv(a,a)) / (sum(a)^2) ,

which, after fixing sum(a)=1, reduces to 2·n·max(conv(a,a)).  The code
starts from a good analytic seed (previous best if available, otherwise a
random Beta‑distribution shape) and then refines it until the time budget
expires.

The routine returns a plain list of non‑negative floats that sum to 1,
ready to be passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact (un‑clamped) objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The input is assumed to be a non‑negative vector.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised Beta(a,b) density on {0,…,n‑1}, normalised to sum = 1.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _initial_sequence(seed: int) -> np.ndarray:
    """
    Return a good starting probability vector.
    Preference is given to the global variable ``height_sequence_1`` if it
    exists; otherwise a small pool of Beta‑shapes is sampled and the best
    one (according to the exact objective) is returned.
    """
    rng = np.random.default_rng(seed)

    # 1) global starter, if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0.0:
                seq = seq / seq.sum()
                return seq
    except Exception:
        pass

    # 2) build a tiny pool of analytic candidates
    n = 4096  # a reasonable default size
    pool = []

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # random Beta shapes
    for _ in range(40):
        a = 10 ** rng.uniform(-2.0, 0.7)   # ~0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        pool.append(_make_beta_shape(n, a, b))

    # pick the best by exact objective
    best = min(pool, key=_objective)
    return best.copy()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a list of floats whose sum equals 1.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    a0 = _initial_sequence(seed)
    n = a0.shape[0]

    # log‑odds (logits) for the soft‑max parametrisation
    y = np.log(a0 + 1e-14)          # avoid log(0)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.30                     # base learning rate

    # best solution tracking
    best_y = y.copy()
    best_score = _objective(a0)

    it = 0
    iters_since_improve = 0
    next_print = start_time + 5.0   # periodic status output

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- soft‑max to obtain the probability vector -----
        y_shift = y - np.max(y)                # numerical stability
        a = np.exp(y_shift)
        a = a / a.sum()

        # ----- self‑convolution and active set -----
        conv = _conv_self(a)
        max_val = conv.max()
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of max(conv) w.r.t. a -----
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # ----- gradient w.r.t. logits (soft‑max chain rule) -----
        g_dot_a = np.dot(grad, a)
        grad_y = a * (grad - g_dot_a)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        # (optional) keep logits bounded to avoid overflow
        # y = np.clip(y, -30.0, 30.0)

        # ----- evaluate current objective -----
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_y = y.copy()
            iters_since_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  new best={cur_score:.6f}")
        else:
            iters_since_improve += 1

        # ----- periodic status line -----
        if time.time() >= next_print:
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  best={best_score:.6f}")
            next_print = time.time() + 5.0

        # ----- random perturbation if we have stalled -----
        if iters_since_improve > 200_000:
            # Re‑initialise logits with a small Gaussian kick around the best point
            y = best_y + rng.normal(scale=0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            iters_since_improve = 0
            # (no immediate print – the next loop will output the status)

    # ------------------------------------------------------------------
    # finalise best solution
    # ------------------------------------------------------------------
    y_shift = best_y - np.max(best_y)
    a_best = np.exp(y_shift)
    a_best = a_best / a_best.sum()
    a_best = np.maximum(a_best, 0.0)
    total = a_best.sum()
    if total > 0.0:
        a_best /= total
    else:
        a_best = np.full_like(a_best, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final best score ≈ {best_score:.6f}")
    return [float(v) for v in a_best]
