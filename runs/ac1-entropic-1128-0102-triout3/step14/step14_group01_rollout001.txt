"""
Step‑function optimiser – improved version.

The algorithm searches for a non‑negative sequence a of length *n*
(minimum n=4096, we use n=8192) with Σa = 1 that minimises

        2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence`` after normalising
the sum to one.  The search consists of four stages:

1.  **Extensive parametric sweep** – families that are known to give good
    autocorrelation behaviour (geometric, power, beta, stretched‑exp,
    logistic) plus a few new ones (cosine, sine, mirrored exponential,
    plateau+tails).  The sweep is timed – it stops when the allocated budget
    for this phase expires.

2.  **Massive random Dirichlet draws** – many random points on the simplex
    with a wide range of concentration parameters.

3.  **Smooth‑max Adam optimisation** – we minimise a smooth approximation
    of the max‑convolution using Adam with an exponentially increasing
    temperature τ.  The learning rate is small and the temperature is
    increased every iteration (τ←τ·1.001) so the optimiser follows the
    true objective very closely.  The optimiser runs until the time budget
    for this phase is exhausted.

4.  **Adaptive hill‑climb / simulated annealing** – we repeatedly move a
    tiny amount of mass from the index with the largest positive sub‑gradient
    to the one with the most negative sub‑gradient.  The step size decays
    and a temperature schedule (Metropolis acceptance) allows occasional
    uphill moves.

All stages keep the current best sequence and score; the final normalised
sequence is returned as a plain Python list.  The code respects the 1000 s
budget, prints progress information and works with the *evaluate_sequence*
function that will be present in the execution environment.
"""

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, projection, sub‑gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two 1‑D arrays using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L_fft)
    conv = irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """
    Score used internally (identical to evaluate_sequence after normalising
    Σa = 1): 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_families(   # noqa: C901
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Exhaustive sweep over a collection of parametric families.
    Each candidate is normalised to sum‑one and its score is computed.
    The best candidate is kept.
    """
    idx = np.arange(n, dtype=np.float64)

    # ---- uniform (baseline) ------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ---------------------------------------------------------
    eps_grid = np.logspace(-5, -1, 200)          # 200 values – cheap

            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0.0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}"
                )

    # ---- power --------------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0.0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}"
            )

    # ---- beta‑type ---------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 61)
    betas = np.linspace(0.05, 4.0, 61)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0.0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}"
                )

    # ---- stretched exponential -----------------------------------------------
    scales = np.linspace(0.25, 1.0, 11)   # fraction of n
    p_vals = np.linspace(0.2, 5.0, 41)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0.0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Stretched‑exp p={p:.2f} scale={scale:.2f} score={sc:.6f}"
                )

    # ---- logistic ------------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 13)   # position of midpoint (as fraction of n)
    s_factors = np.linspace(0.01, 0.2, 11)   # slope (as fraction of n)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}"
                )

    # ---- cosine (single‑sided) -----------------------------------------------
    # a_i ∝ (1 + cos(π i/(n-1))) / 2    – decreasing cosine wave
    cos_seq = 0.5 * (1.0 + np.cos(np.pi * idx / (n - 1)))
    cos_seq /= cos_seq.sum()
    sc = _objective_score(cos_seq)
    if sc < best_score:
        best_score, best_seq = sc, cos_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Cosine score={sc:.6f}")

    # ---- sine (half‑period) -------------------------------------------------
    sin_seq = np.sin(np.pi * (idx + 0.5) / n)  # strictly positive
    sin_seq = np.clip(sin_seq, 1e-12, None)
    sin_seq /= sin_seq.sum()
    sc = _objective_score(sin_seq)
    if sc < best_score:
        best_score, best_seq = sc, sin_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Sine score={sc:.6f}")

    # ---- mirrored exponential mixture ----------------------------------------
    # a_i ∝ λ·exp(-α·i) + (1−λ)·exp(-α·(n−1−i))
    alphas_exp = np.logspace(-4, -0.2, 15)          # decay rates
    lambdas = np.linspace(0.0, 1.0, 11)
    for a_exp in alphas_exp:
        if time.time() >= deadline:
            break
        left = np.exp(-a_exp * idx)
        right = left[::-1]                         # symmetric tail
        for lam in lambdas:
            if time.time() >= deadline:
                break
            seq = lam * left + (1.0 - lam) * right
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] MirrExp α={a_exp:.3e} λ={lam:.2f} score={sc:.6f}"
                )

    # ---- plateau with geometric tails ---------------------------------------
    # parameters: plateau width fraction w∈[0.2,0.8], tail ratio r∈[0.01,0.3]
    widths = np.linspace(0.2, 0.8, 7)
    ratios = np.logspace(-2.5, -0.2, 12)   # geometric ratio in tails
    for w in widths:
        if time.time() >= deadline:
            break
        t_len = int((1.0 - w) * n / 2)        # length of each tail
        plateau_len = n - 2 * t_len
        for r in ratios:
            if time.time() >= deadline:
                break
            # left tail: r^{t_len-1},…,r,1
            left_tail = r ** np.arange(t_len)[::-1]
            right_tail = left_tail[::-1]
            plateau = np.ones(plateau_len)
            seq = np.concatenate([left_tail, plateau, right_tail])
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] PlatTail w={w:.2f} r={r:.3f} score={sc:.6f}"
                )

    return best_seq, best_score

def _random_dirichlet_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Draw a large number of random points on the simplex."""
    rng = np.random.default_rng()
    # The number of trials is adapted to the remaining time budget.
    # Roughly 0.5 µs per trial → 250 ms per 500 k trials is safe.
    max_trials = 5000
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # draw a concentration from a log‑uniform interval
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = math.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth‑max approximation of max(conv(a,a)) with Adam.
    The temperature τ starts modest (5) and is multiplied by 1.001 each
    iteration, giving a very smooth annealing into the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02

    tau = 5.0
    tau_mul = 1.001

    it = 0
    while time.time() < deadline:
        it += 1

        # current (exact) score – also the checkpoint for best
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth‑max weights (clip to avoid overflow)
        max_c = conv.max()
        diff = tau * (conv - max_c)
        diff = np.clip(diff, -700, 700)                # stabilise exp()
        w_raw = np.exp(diff)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum

        # gradient of the smooth‑max term
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)                   # length = 3·n‑2
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        tau = min(tau * tau_mul, 1e9)   # avoid overflow

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated annealing (adaptive step)
# ----------------------------------------------------------------------
def _final_hill_climb(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Adaptive hill‑climb that moves a *fraction* of the source mass.
    Early in the phase the fraction is large (up to 5 % of source);
    it decays exponentially.  Temperature follows a simple exponential
    decay so that uphill moves become unlikely towards the end.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    # compute current score once
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # total time left for hill‑climb
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0.0:
        return best_seq, best_score

    # schedule parameters
    T0 = 0.02 * cur_score                      # temperature proportional to score
    decay = (1e-8 / T0) ** (1.0 / total_hc_time)  # T → ~0 at the end
    step_start = 0.05                         # 5 % of source mass at start
    step_end = 0.001                          # 0.1 % at the very end
    step_decay = (step_end / step_start) ** (1.0 / total_hc_time)

    cur_step = step_start

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source index: positive gradient and non‑zero mass
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # transfer a fraction of the source mass
        eps = a[src] * cur_step
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        delta = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        if delta < -1e-12:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}"
                )

        # decay step size and temperature continuously
        cur_step = max(step_end, cur_step * step_decay)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine follows a four‑stage pipeline
    (parametric sweep → random Dirichlet draws → Adam smoothing → hill‑climb)
    and respects the given time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives more flexibility, still fast enough)
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with the uniform distribution (or a provided starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(
        f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}"
    )

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}"
                    )
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (reserve ~200 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0   # keep ~250 s for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline, L_fft
        )
        print(
            f"[{time.time() - start_time:.2f}s] After sweep best score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (reserve ~150 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )
        print(
            f"[{time.time() - start_time:.2f}s] After Dirichlet best score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation of smooth‑max (reserve ~480 s)
    # ------------------------------------------------------------------
    hill_time = 130.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, smooth_deadline, best_seq, best_score, L_fft
        )
        print(
            f"[{time.time() - start_time:.2f}s] After Adam best score≈{best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb with simulated‑annealing flavour
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score, L_fft
        )
        print(
            f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    # sanity check with the provided evaluator (optional, does not affect result)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] evaluate_sequence reports {val:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
