import time, math, numpy as np

# define utilities:
def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _project_simplex(v):
    # Euclidean projection onto probability simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho]/(rho+1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _eval_fast(a, L_self):
    # a must sum to 1
    conv = _conv_self_fft(a, L_self)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _make_monotonic_shape(n, r, p):
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

def _make_U_shape(n, r, p=0.0):
    # symmetric U-shaped: r**i + r**(n-1-i) with optional power factor
    i = np.arange(n, dtype=np.float64)
    shape = np.power(r, i) + np.power(r, n-1-i)
    if p != 0.0:
        # optional extra power weighting by distance from centre
        dist = np.abs(i - (n-1)/2) + 1.0
        shape *= np.power(dist, p)
    shape /= shape.sum()
    return shape

def _mix_with_uniform(shape, lam):
    n = shape.shape[0]
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

    Hybrid optimisation for the step‑function objective.
    1. Coarse sweep over two parametric families (geometric‑decreasing
       and symmetric “U‑shaped”) mixed with the uniform distribution.
    2. Deterministic‑annealing Adam optimisation of the exact objective
       using a soft‑max approximation of the max‑convolution.
    3. Tiny stochastic hill‑climb that moves a small amount of mass
       between random positions.
    The routine respects the time budget and returns the best vector
    (list of floats) found.
    """

        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        a_candidate = a - step * m_hat
        # maintain the simplex
        a_candidate = _project_simplex(a_candidate)

        # optional symmetrisation – the optimal solution is usually symmetric
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a_candidate = _project_simplex(a_candidate)

            best_seq = a_candidate.copy()
            print(f"[{time.time()-start:.2f}s] Adam it={it:,} τ={tau:.2f} score={cur_val:.6f}")

        # always move forward (even if not improved) – helps escaping plateaus
        a = a_candidate

        # --- schedule: increase τ slowly, decay step size via its own schedule ---
        if it % 2000 == 0:

        # occasional random jitter to avoid being stuck
        if it % 5000 == 0:
            eps_vec = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb (tiny mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_best = best_score
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move a small proportion of the source mass
            delta = a[src] * rng.random() * 1e-3   # up to 0.1 % of source
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            # re‑project onto simplex (the operation already preserves sum)
            a_prop = _project_simplex(a_prop)
            sc = _eval_fast(a_prop, L_self)
            if sc < cur_best:
                cur_best = sc
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[{time.time()-start:.2f}s] final hill‑climb score={sc:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)

    print(f"[{time.time()-start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

Hybrid optimiser for the step‑function objective.

1️⃣  Coarse grid search over two simple families:
   – a monotone geometric × power shape,
   – a symmetric “U‑shaped’’ geometric sum.
   Both families are optionally mixed with the uniform distribution.

2️⃣  Smooth Adam optimisation.
   The max‑convolution is approximated by a soft‑max:
        w_k ∝ exp( τ·(a*a)_k ).
   The gradient of the smooth objective is
        ∇ = 2·(w ⋆ a)   (cross‑correlation),
   which is computed by a single FFT‑based convolution.
   An Adam update (with projection onto the probability simplex) is
   performed while τ is slowly increased, turning the smooth proxy
   into the true max.

3️⃣  Tiny stochastic hill‑climb that transfers a very small amount
   of mass between random coordinates – useful for escaping any
   remaining plateaus.

The routine respects a 1000 s wall‑clock budget and returns the best
vector (as a plain list of floats) it has found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_self)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _make_monotonic_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape (decreasing)."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)          # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_U_shape(n: int, r: float, p: float = 0.0) -> np.ndarray:
    """
    Symmetric U‑shaped distribution:
        a_i ∝ r**i + r**(n‑1‑i)  (optional distance‑power factor p).
    """
    i = np.arange(n, dtype=np.float64)
    shape = np.power(r, i) + np.power(r, n - 1 - i)
    if p != 0.0:
        # extra weighting by distance from the centre
        dist = np.abs(i - (n - 1) / 2) + 1.0
        shape *= np.power(dist, p)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """lam·uniform + (1‑lam)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm consists of a coarse sweep, an Adam‑based smooth
    optimisation and a final tiny hill‑climb.  It respects the given
    time budget and returns the best vector found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2      # safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    L_self = _next_pow2(2 * n - 1)        # for a*a
    L_cross = _next_pow2(3 * n - 2)       # for w ⋆ a (linear)

    # --------------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global starter (if provided via the global name `height_sequence_1`)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                cur = _eval_fast(starter, L_self)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = start + 150.0
    r_grid = np.logspace(np.log10(0.85), np.log10(0.9999), num=30)
    p_grid = np.linspace(-2.0, 2.0, 11)   # only for the monotone family
    lam_grid = np.linspace(0.0, 0.5, 6)   # mixing with uniform

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        # monotone family
        for p in p_grid:
            shape = _make_monotonic_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_self)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start:.2f}s] sweep mono r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

        # U‑shaped family (no extra power parameter for speed)
        for lam in lam_grid:
            shape = _make_U_shape(n, r)
            cand = _mix_with_uniform(shape, lam)
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start:.2f}s] sweep U r={r:.5f} λ={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth Adam optimisation (soft‑max of the max‑convolution)
    # --------------------------------------------------------------
    # reserve most of the remaining budget for this phase
    grad_deadline = deadline - 200.0
    a = best_seq.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.15          # base learning rate
    tau = 3.0             # temperature of the soft‑max (smoothness)
    it = 0

    while time.time() < grad_deadline:
        it += 1

        # ----- self‑convolution (needed for the objective and for w) -----
        conv = _conv_self_fft(a, L_self)          # length 2·n‑1
        max_conv = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        # shift by max_conv for numerical stability
        w_raw = np.exp(tau * (conv - max_conv))
        w = w_raw / w_raw.sum()                   # sum(w) = 1

        # ----- gradient: 2·(w ⋆ a) (cross‑correlation) -----
        a_rev = a[::-1]
        # linear convolution of w with a_rev (length 3·n‑2)
        fft_w = np.fft.rfft(w, n=L_cross)
        fft_a_rev = np.fft.rfft(a_rev, n=L_cross)
        cross_full = np.fft.irfft(fft_w * np.conj(fft_a_rev), n=L_cross)
        cross_full = cross_full[: (len(w) + len(a_rev) - 1)]   # keep linear part
        grad = 2.0 * cross_full[(n - 1) : (n - 1 + n)]        # slice n entries

        # ----- Adam update (projected) -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # projection onto the simplex
        a_candidate = _project_simplex(a_candidate)

        # enforce (optional) symmetry – the optimal solution is typically symmetric
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a_candidate = _project_simplex(a_candidate)

        # evaluate the true objective
        cur_score = _eval_fast(a_candidate, L_self)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"[{time.time() - start:.2f}s] Adam it={it:,} τ={tau:.2f} score={cur_score:.6f}")

        # always move forward (helps to leave plateaus)
        a = a_candidate

        # ----- schedule: make τ larger (sharper) over time -----
        if it % 2000 == 0:
            tau = min(tau * 1.07, 120.0)   # slowly turn the smooth proxy into the true max

        # occasional random jitter to dodge tiny plateaus
        if it % 5000 == 0:
            eps_vec = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣  Final tiny hill‑climb (mass transfer between random indices)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_best = best_score
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move up to 0.1 % of the source mass
            delta = a[src] * rng.random() * 1e-3
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            a_prop = _project_simplex(a_prop)
            sc = _eval_fast(a_prop, L_self)
            if sc < cur_best:
                cur_best = sc
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[{time.time() - start:.2f}s] final hill‑climb score={sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
