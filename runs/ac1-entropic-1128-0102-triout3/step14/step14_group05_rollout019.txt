"""
propose_candidate – search for a step‑function (non‑negative sequence)
that minimises 2·n·max(conv(a,a)) / (sum(a)**2).

The routine combines several ideas:
  * a cheap parametric sweep (geometric‑power families);
  * a temperature‑annealed L‑BFGS optimisation of a smooth‑max surrogate;
  * a dedicated sub‑gradient descent on the *true* max‑convolution
    (Adam optimiser in the soft‑max logit space);
  * a final stochastic hill‑climb that moves tiny masses between
    coordinates.

All steps respect a hard 1000 s wall‑clock budget and keep the best
sequence found.
"""

import time
import numpy as np

# optional – may be missing in a minimal environment
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper primitives
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT.
    Returns the (2·len(a)‑1)‑long result."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                    # very unlikely
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


def _objective(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Smooth‑max (log‑sum‑exp) surrogate of max(conv) together with its gradient
    w.r.t. the log‑probability parameters `z`.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # length for a*a
        self.L_corr = L_corr          # length for w * a_rev

    def __call__(self, z: np.ndarray):
        """Returns (surrogate value, gradient w.r.t. z)."""
        a = _softmax(z)                     # a ≥0, Σa = 1
        # self‑convolution
        conv = _conv_self_fft(a, self.L_conv)     # length 2n‑1
        max_c = conv.max()
        # smooth‑max via log‑sum‑exp
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights (length 2n‑1)
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient w.r.t. a (2 * w * a_rev convolution)
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]   # length n

        # chain rule through the soft‑max (softmax Jacobian)
        dot = np.dot(grad_a, a)                     # scalar
        grad_z = a * (grad_a - dot)                 # shape (n,)
        return surrogate, grad_z


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def _grad_at_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Gradient of max(conv(a,a)) w.r.t. a (sub‑gradient when several maxima).
    Implements the analytic formula g_i = 2·∑_k w_k·a_{k‑i}, where w_k is a
    uniform weight over the maximal indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    cnt = max_idx.size
    for k in max_idx:
        if k <= n - 1:
            # contribution from i = 0 .. k
            grad[: k + 1] += a[k :: -1]          # a[k], a[k-1], …, a[0]
        else:
            start = k - (n - 1)                  # first i with a_{k-i} defined
            # contribution from i = start .. n‑1
            grad[start:] += a[n - 1 : start - 1 : -1]
    grad *= 2.0 / cnt
    return grad


def _refine_max_gradient(a0: np.ndarray, n: int, L_fft: int,
                         deadline: float, start_time: float):
    """
    Adam‑based optimisation of the exact max‑convolution.
    Operates in the log‑probability space (softmax ensures non‑negativity,
    unit sum).  Returns the best vector found and its objective value.
    """
    eps = 1e-12
    z = np.log(np.maximum(a0, eps))

    best_a = a0.copy()
    best_score = _objective(best_a, n, L_fft)

    # Adam hyper‑parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.2
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline - 0.5:
        a = _softmax(z)
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[{time.time() - start_time:.2f}s] max‑grad iter={t} "
                      f"best_score={best_score:.6f}")

        # indices attaining the maximum (allow a tiny tolerance)
        tol = max_val * 1e-12 + 1e-15
        max_idx = np.where(conv >= max_val - tol)[0]

        # sub‑gradient w.r.t. a and then w.r.t. z
        grad_a = _grad_at_max(a, max_idx)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update on the log‑probabilities
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional tiny random jitter to escape plateaus
        if t % 2000 == 0:
            z += rng.normal(scale=1e-4, size=n)
            m.fill(0.0)
            v.fill(0.0)

        # simple learning‑rate decay
        if t % 5000 == 0:
            lr *= 0.8

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap hill‑climb with the exact objective
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """Random local mass transfers that are kept only if they improve the true score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(best_a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.5 % of the source mass
        eps = best_a[src] * rng.random() * 5e-3
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _objective(cand, n, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine respects a 1000 s wall‑clock budget, records the best
    sequence found, and finally returns it as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15      # tiny safety margin

    # problem size – 4096 works well on a single core
    n = 4096
    L_fft = _next_pow2(2 * n - 1)               # for exact convolutions
    L_corr = _next_pow2(3 * n - 2)              # for smooth‑max gradients

    # --------------------------------------------------------------
    # 0️⃣ uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # optional global starter (if the notebook defines `height_sequence_1`)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _objective(starter, n, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # --------------------------------------------------------------
    # 1️⃣ fast parametric sweep (geometric‑power families)
    # --------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index
    r_grid = np.logspace(np.log10(0.90), np.log10(0.99999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 200:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _objective(cand, n, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                          f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ smooth‑max L‑BFGS optimisation (annealed β)
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 250:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0]
        # more iterations for the later (sharper) temperatures
        maxfun_per_beta = {0.5: 200, 1.0: 200, 2.0: 200, 5.0: 400,
                           10.0: 400, 20.0: 600, 50.0: 800,
                           100.0: 1200, 200.0: 1500, 500.0: 2000,
                           1000.0: 2500}

        for beta in beta_schedule:
            if time.time() >= deadline - 250:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))
            maxfun = maxfun_per_beta.get(beta, 500)
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            sc = _objective(cand, n, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → "
                      f"improved score={sc:.6f}")
            # warm‑start the next temperature
            z0 = z_opt

    # --------------------------------------------------------------
    # 3️⃣ refinement on the true max‑convolution (sub‑gradient + Adam)
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        best_seq, best_score = _refine_max_gradient(
            best_seq, n, L_fft, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] post‑refine best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ final cheap hill‑climb polishing (exact objective)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # post‑process: ensure non‑negative, normalise to unit sum and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
