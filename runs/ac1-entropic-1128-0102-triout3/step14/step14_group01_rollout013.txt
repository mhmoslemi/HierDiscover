        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # For j in lo..hi: f[k - j].
        # Equivalent: fill A_ub[r, lo:hi+1] = f[k - np.arange(lo,hi+1)]
        # indices: j = lo..hi, k - j runs from k-lo down to k-hi.

        # Let's compute array idx = np.arange(lo, hi+1); A_ub[r, idx] = f[k - idx]
        idx = np.arange(lo, hi+1)
        A_ub[r, idx] = f[k - idx]
    b_ub = np.full(rows, M)

"""
Step‑function optimizer – fine‑grained exponentiated‑gradient descent.

The evaluator:
    score = 2·n·max(conv(a,a))  (a is normalised to sum‑1)
Thus we have to minimise the peak of the self‑convolution on the
probability simplex.

Strategy
---------
1.  Initialise with a large set of random Dirichlet draws (wide range
    of concentration parameters) and keep the best.
2.  Refine the best candidate with a mirror‑descent / exponentiated‑
    gradient scheme that moves mass from indices with a positive
    sub‑gradient to those with a negative one.  The update is
      a_i ← a_i·exp(−η·∇_i)      (projected onto the simplex by renormalising)
    η is decreased as 1/√t.  The sub‑gradient of  max(conv) is computed
    by averaging over all active (tight) convolution shifts.
3.  After the gradient phase a quick “mix‑with‑uniform’’ sweep
    (convex combination of the best sequence and the uniform one)
    is performed; a tiny amount of mass can sometimes smooth the peak
    further.
4.  The whole routine is time‑aware (budget ≈ 1000 s) and prints
    progress information.  The final sequence is returned as a plain
    Python list with non‑negative entries and sum ≈ 1.

The implementation uses only numpy and scipy (for a tiny linear‑program
fallback).  No external files, network calls or hidden state are used.
"""

import time
import numpy as np

# optional: scipy is only needed for a cheap LP fallback (not essential)
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance – treat anything within 1e‑12 of the max as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]            # contribution a_{k-i}
    if active.size:
        grad = 2.0 * grad / active.size    # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)),  a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet draws (wide concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # a few thousand draws are cheap; we keep going until time runs out
    while time.time() < deadline:
        # concentration in [0.02, 20] drawn log‑uniformly
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Exponentiated‑gradient (mirror descent) on the simplex
# ----------------------------------------------------------------------
def _expgrad_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of
    max(conv(a,a)).  η (step) decays like 1/√t.
    """
    a = a0.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        # current objective
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # step size η – 1/√t scaling, capped for stability
        eta = 0.7 / np.sqrt(it)

        # multiplicative update, clipped exponent to avoid overflow
        exp_factor = np.exp(-eta * grad)
        # Clip very large/small factors to keep numbers in a reasonable range
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)

        a_new = a * exp_factor
        # simplex projection – here simply renormalise (positivity is preserved)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # fall back to uniform if something went wrong
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        # accept if we made progress; otherwise keep the old a
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            # occasional simulated‑annealing style acceptance to escape plateaus
            # temperature decays linearly with remaining time
            T0 = 1e-3
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < accept_prob:
                a = a_new
        # loop continues
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final fine‑grained mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform.
    The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)
    # start with λ=0 (uniform) and λ=1 (current a); then refine around best λ.
    lam_grid = np.linspace(0.0, 1.0, 41)   # coarse sweep
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # Refine in a narrow window around the best λ (if any improvement was seen)
    # Find the λ that gave the best score
    # (a simple approach – re‑run with finer grid around the best λ)
    # If best_seq changed, recompute λ estimate
    # For simplicity we just stop here – the time budget is usually exhausted.
    return best_seq, best_score

    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()

            if s > 0:

                sc = _objective_score(starter, n)
