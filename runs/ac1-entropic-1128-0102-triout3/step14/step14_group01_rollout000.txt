"""
Step‑function optimiser – “max‑convolution” challenge.

Main ideas
----------
1.  Initialise with the best known construction (`height_sequence_1`) if it
    exists, otherwise with the uniform distribution.
2.  Perform a short parametric sweep (geometric, power, beta, stretched‑exp,
    logistic, Hann‑window).  The best candidate from this stage seeds the
    optimiser.
3.  Optimise the exact objective `max(conv(a,a))` via a projected Adam
    method that uses the *true* sub‑gradient (no smooth approximation).
    The update is performed on the probability simplex.
4.  A final simulated‑annealing hill‑climb moves a small amount of mass from
    indices with the largest positive sub‑gradient to those with the most
    negative, accepting worsening moves with a slowly decaying temperature.
5.  The whole pipeline respects the supplied time budget (default 1000 s)
    and returns a list of non‑negative floats whose sum is 1.0.

All heavy work is done with FFT‑based convolutions (O(n log n)) and an
O(n) Euclidean simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance for “active” shifts
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) for sequences with sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_families(n: int, best_seq: np.ndarray,

    idx = np.arange(n, dtype=np.float64)

    # uniform (baseline)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] uniform  score={sc:.6f}")

    for r in np.linspace(0.2, 0.9999, 80):
        if time.time() >= deadline:

        if sc < best_score:

    # -------- beta‑type ----------

    betas = np.linspace(0.1, 4.0, 31)
    log_i = np.log(idx + 1.0)

    for a in alphas:

        for b in betas:

            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] beta a={a:.2f} b={b:.2f}  score={sc:.6f}")

    # -------- stretched exponential ----------
    for scale in np.linspace(0.3, 1.0, 9):

            seq = np.exp(-np.clip(factor, 0, 700))

            sc = _objective_score(seq)
            if sc < best_score:

            sc = _objective_score(seq)
            if sc < best_score:

# 2️⃣  Adam optimisation using the *exact* sub‑gradient of max‑conv
# ----------------------------------------------------------------------
def _adam_subgrad_optimize(a0: np.ndarray,
                           start_time: float,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution objective.
    The objective on the simplex is simply `max(conv(a,a))`.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02                     # modest base learning rate
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑subgrad it={it:,}  score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # Adam update (move opposite to the gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # optional enforcement of symmetry – often beneficial
        a_new = (a_new + a_new[::-1]) * 0.5

        a = a_new

    return best_seq, best_score

# 3️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Tiny mass transfers guided by the sub‑gradient; accepts worsening moves."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule – decays to ~0 at the deadline
    total = max(0.0, deadline - time.time())
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total) if total > 0 else 0.0

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source: index with largest positive sub‑gradient that still has mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination: index with smallest sub‑gradient
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer a *tiny* but occasionally larger fraction
        frac = rng.random() * 0.008 + 1e-6   # up to ≈0.8 % of source mass
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)      # keep on simplex
        new_score = 2.0 * n * _conv_self(a_prop, L_fft).max()
        Δ = new_score - cur_score

        # temperature at this moment
        T = T0 * (decay ** (time.time() - start_time))

        accept = Δ < -1e-14 or rng.random() < np.exp(-Δ / max(T, 1e-12))
        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb best={best_score:.6f}")

    return best_seq, best_score

    # Baseline initialisation
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform  score={best_score:.6f}")

    if starter is not None:
        sc = _objective_score(starter)
        if sc < best_score:
            best_score, best_seq = sc, starter.copy()
            print(f"[{time.time() - start_time:.2f}s] starter provided  score={sc:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (≈15 % of total time)
    # --------------------------------------------------------------
    phase_deadline = start_time + 0.15 * budget_s
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation on the exact max‑convolution (≈55 % time)
    # --------------------------------------------------------------
    phase_deadline = start_time + 0.70 * budget_s   # leave ~30 % for final hill‑climb
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] start Adam‑subgrad optimisation")
        best_seq, best_score = _adam_subgrad_optimize(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] after Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb / simulated annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] start final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
