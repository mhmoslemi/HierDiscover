"""
Step‑function optimiser (improved)

- Builds a richer initial pool: uniform, exponential decay (forward & reverse),
  power‑law, symmetric / asymmetric Beta, random Dirichlet draws and the optional
  global starter if present.
- Keeps a larger top‑k pool (12 candidates) and runs a quick simulated‑annealing
  warm‑start.
- Performs exponentiated sub‑gradient descent on the exact ℓ∞ objective from
  each top candidate (time‑sliced).  The step‑size is adaptively reduced and
  the active set is weighted uniformly.
- Finishes with a smooth‑max Adam optimiser that uses a temperature schedule
  and soft‑max weighting, allowing a finer polishing of the best solution.
- All heavy work uses FFT‑based convolution for O(n log n) per iteration.
- The routine respects the 1000 s budget and returns a non‑negative list that
  can be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution / objective utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Shape generators
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential_shape(n: int, r: float) -> np.ndarray:
    """Forward exponential decay a_i ∝ r**i, normalised to sum 1 (0<r<1)."""
    if not (0.0 < r < 1.0):
        raise ValueError("r must be in (0,1)")
    powers = np.power(r, np.arange(n, dtype=float))
    shape = powers / powers.sum()
    return shape


def _make_powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = np.power(idx, -p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a size‑k sorted pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active indices (within floating tolerance)
        tol = max(1e-15, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # uniform weight on active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # decreasing step size
        eta = eta0 / (1.0 + it / 2500.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        # avoid all‑zero vector
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 2000 == 0:
                print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam descent on the smooth‑max surrogate of the ℓ∞ objective.
    Uses a soft‑max weighting of all convolution entries.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.08                # base learning rate
    T0 = 0.02
    T_end = 1e-6                # final temperature (much lower)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # temperature schedule (exponential annealing)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, (elapsed) / total)
        T = max(T_end, T0 * (T_end / T0) ** frac)

        # soft‑max weights (log‑sum‑exp trick for numerical stability)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w * a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # --------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool
    # --------------------------------------------------------------
    TOP_K = 12
    top = []      # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        # ensure non‑negative and normalised
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec /= s
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform baseline
    _add(np.full(N, 1.0 / N))

    # exponential decays (forward & reverse)
    for r in np.linspace(0.5, 0.999, 12):
        _add(_make_exponential_shape(N, r))          # left‑heavy
        _add(_make_exponential_shape(N, r)[::-1])   # right‑heavy

    # power‑law shapes
    for p in np.linspace(0.4, 3.0, 10):
        _add(_make_powerlaw_shape(N, p))

    # symmetric Beta shapes (log‑space for α)
    for alpha in np.logspace(-2, 1, 12):
        _add(_make_beta_shape(N, float(alpha)))

    # asymmetric Beta shapes (α small, β larger) – richer grid
    alphas = np.logspace(-2, 0, 8)       # 0.01 … 1
    betas = np.logspace(0, 1, 8)        # 1 … 10
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, float(a), float(b)))

    # random Dirichlet vectors (various concentration levels)
    rng = np.random.default_rng(seed + 9999)
    for _ in range(80):
        if time.time() >= deadline:
            break
        # concentration drawn uniformly from [0.1, 2.0]
        conc = rng.uniform(0.1, 2.0)
        vec = rng.gamma(conc, 1.0, size=N)
        if vec.sum() > 0:
            vec /= vec.sum()
            _add(vec)

    # optional global starter (if defined in the notebook)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _add(starter_up)
    except Exception:
        pass

    # best of the initial pool
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated‑annealing warm‑start (≈10 % of the total budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.10
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    rng = np.random.default_rng(seed + 12345)
    max_move_factor = 0.35
    T_start = 0.5
    T_end = 1e-9
    it = 0

    while time.time() < phase1_deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * (T_end / T_start) ** frac

        src = rng.choice(N, p=cur_seq)
        if cur_seq[src] < 1e-15:
            continue
        dst = rng.integers(N)
        if src == dst:
            continue

        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue

        a_prop = cur_seq.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        dif = new_score - cur_score

        if dif < 0.0 or rng.random() < math.exp(-dif / T):
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

        # occasional restart from a different top‑candidate
        if it % 200000 == 0 and len(top) > 1:
            idx = rng.integers(len(top))
            cur_seq = top[idx][1].copy()
            cur_score = _objective(cur_seq)

        # occasional Gaussian jitter
        if it % 80000 == 0:
            sigma = 0.008
            noise = rng.normal(0.0, sigma, size=N)
            a_noisy = cur_seq + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            s = a_noisy.sum()
            if s > 0:
                a_noisy /= s
                cur_seq = a_noisy
                cur_score = _objective(cur_seq)

    # --------------------------------------------------------------
    # 3) Exponentiated sub‑gradient descent from each top candidate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        # allocate 60 % of the remaining time to sub‑gradient runs,
        # split equally among the current top‑K candidates.
        sub_total = time.time() + remaining * 0.60
        per_candidate = (sub_total - time.time()) / len(top)

        # first, improve the SA result
        seq, sc = _exp_subgradient_descent(best_global_seq, sub_total, start_time)
        if sc < best_global_score:
            best_global_seq, best_global_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD (SA start) improved best={sc:.6f}")

        # then run from each pooled vector (time‑boxed)
        for idx, (_, vec) in enumerate(top):
            if time.time() >= sub_total:
                break
            end_time = time.time() + per_candidate
            seq, sc = _exp_subgradient_descent(vec, end_time, start_time)
            if sc < best_global_score:
                best_global_seq, best_global_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 4) Smooth‑max (Adam) polishing with the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_seq, best_global_score = refined_seq, refined_score
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best={refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion
    # --------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
