def _adam_subgradient_descent(a0, deadline, start_time, best_score_initial, step0=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(a0)
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute subgradient at shift k
        g = _subgrad_at_shift(a, k)
        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] adam it={t:,} score={cur_score:.6f}")
        else:

            a = np.maximum(a + eps_move, 0.0)

"""
Search for a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases

1️⃣  **Quick random parametric sweep** – generate vectors of the form
    a_i ∝ r**i·(i+1)**p and mix them with the uniform distribution.
    The best candidate from this cheap exploration is taken as a warm‑start.

2️⃣  **Projected Adam on the true sub‑gradient** – starting from the warm‑start
    we run a few Adam‑style updates on the sub‑gradient of the maximal
    convolution entry.  The update is projected onto the probability simplex
    after each step; occasional uphill moves and tiny random jitters are added
    to escape plateaus.

3️⃣  **Final tiny hill‑climb** – a simple random mass‑transfer move between
    two coordinates is performed while time remains to polish the solution.

All stages respect the 1000 s budget.  The routine returns the best sequence
found as a plain list of floats (already normalised to sum = 1)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power) and mixture with uniform
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Returns a normalised vector of length n with components
    proportional to r**i * (i+1)**p (i = 0…n‑1).
    Computation is performed in log‑space for stability.
    """
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # log‑space to avoid overflow/underflow
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)                     # shift for stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns lam * uniform + (1‑lam) * shape (both already sum to 1).
    The result automatically sums to 1.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()                               # re‑normalise for safety
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient for a given convolution shift
# ----------------------------------------------------------------------


def _subgrad_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the k‑th entry of the self‑convolution a*a.
    The gradient is exactly the derivative of Σ_i a_i a_{k-i};
    diagonal terms receive a factor 2 (handled automatically).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] += a[j]
    g[j] += a[i]                     # diagonal entries get added twice → 2·a_i
    return g


# ----------------------------------------------------------------------
# Projected Adam on the sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float,
                              step0: float = 0.25,
                              beta1: float = 0.9,
                              beta2: float = 0.999,
                              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that follows the sub‑gradient of the currently
    maximal convolution entry.
    Returns the best vector encountered and its score.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    m = np.zeros(n)
    v = np.zeros(n)
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # --- locate the maximal convolution entry --------------------
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # --- sub‑gradient for this shift ----------------------------
        g = _subgrad_at_shift(a, k)

        # --- Adam moment updates ------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # --- step --------------------------------------------------
        step = step0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        # --- evaluate ------------------------------------------------
        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")
        else:
            # occasional probabilistic uphill move
            prob = np.exp(-(cur_score - best_score) * 9.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # --- small jitter every few thousand iterations -------------
        if it % 3000 == 0:
            eps_move = rng.random(n) * a * 1e-4
            a = np.maximum(a + eps_move, 0.0)
            a = _project_simplex(a)

    return best_a, best_score
