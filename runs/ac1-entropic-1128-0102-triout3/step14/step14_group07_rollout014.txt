def _make_beta_asym(n, alpha, beta):
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1-eps)
    # beta density: x^{alpha-1}(1-x)^{beta-1}
    shape = np.exp((alpha-1)*np.log(x) + (beta-1)*np.log(1-x))
    total = shape.sum()
    if total>0:
        shape /= total
    else:
        shape = np.full(n, 1.0/n)
    return shape

"""
Implementation of a generic search for a low‑score step function.
The algorithm proceeds in three stages:
  1. Diverse initial pool (analytic families, Dirichlet, asymmetric Beta,
     previous best).  The pool is kept small (TOP_K) but very varied.
  2. Projected Adam optimisation that directly uses the exact sub‑gradient
     of the max‑convolution (no smooth‑max surrogate).  The best few
     candidates receive the bulk of the remaining time.
  3. A short simulated‑annealing mass‑swap polish on the final best
     sequence.  If time permits the final sequence is up‑sampled to a
     larger resolution and a last Adam/SA pass is performed.
All iterates stay on the probability simplex (non‑negative, sum = 1)
via Euclidean projection.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution a*a (real FFT, length 2·len(a)-1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the standard simplex (L2‑projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_decay(n: int, p: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i) (lam ≥ 0)."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sym_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric ramp used in the reference implementation."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = shape / shape.sum() if shape.sum() > 0 else np.full(n, 1.0 / n)
    return shape


def _make_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted pool of size ≤ k (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution for a given lag k
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Return ∇_a (max_k conv)[k] = ∇_a conv[k] for a given index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k <= n - 1:
        # indices 0 … k : g[i] = 2*a[k-i]
        g[: k + 1] = 2.0 * a[k :: -1]  # reversed slice
    else:
        start = k - (n - 1)           # first index that participates
        # slice a[start:] reversed and placed at g[start:]
        g[start:] = 2.0 * a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Projected Adam optimiser using the exact sub‑gradient of max‑conv
# ----------------------------------------------------------------------
def _adam_exact_max(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the true max‑convolution objective.
    Returns the best (seq,score) seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters (standard)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.25                     # initial learning rate (aggressive)
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # find the set of tying indices (within tiny tolerance)
        tol = 1e-12
        max_idxs = np.where(conv >= max_val - tol)[0]
        # choose one randomly if more than one
        if max_idxs.size > 1:
            k = int(rng.choice(max_idxs))
        else:
            k = int(max_idxs[0])

        grad = _grad_max_conv(a, k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        it_f = float(it)
        m_hat = m / (1.0 - beta1 ** it_f)
        v_hat = v / (1.0 - beta2 ** it_f)
        step = step0 / math.sqrt(it_f)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """Very simple SA that repeatedly transfers a small amount of mass."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.3
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # pick a source with probability proportional to its weight
        src = rng.choice(n, p=a)
        if a[src] < 1e-20:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsample a vector (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, N_target: int) -> np.ndarray:
    """Interpolate linearly to a new length."""
    n = seq.shape[0]
    if n == N_target:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, N_target, endpoint=False)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 123456)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a modest size for the exploratory phase (fast FFTs)
    # ------------------------------------------------------------------
    N = 16384                     # initial resolution
    N_large = 65536               # final resolution if time permits

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool
    # ------------------------------------------------------------------
    TOP_K = 10
    pool = []                     # list of (score, vector)

    # Uniform baseline
    _push_top(pool, _objective(_make_uniform(N)), _make_uniform(N), TOP_K)

    # Symmetric power‑ramp (plateau)
    for p in np.linspace(0.2, 2.0, 7):
        for plat in np.linspace(0.0, 0.5, 5):
            vec = _make_sym_ramp(N, p, plat)
            _push_top(pool, _objective(vec), vec, TOP_K)

    # Monotone power‑decay
    for p in [0.3, 0.6, 1.0, 1.5, 2.5]:
        vec = _make_power_decay(N, p)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Exponential decay
    for lam in [0.001, 0.003, 0.01, 0.03, 0.1]:
        vec = _make_exponential_decay(N, lam)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Asymmetric Beta shapes (random parameters)
    for _ in range(30):
        alpha = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        beta = 10 ** rng.uniform(-3, 1)
        vec = _make_beta_asym(N, alpha, beta)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Random Dirichlet draws with a variety of concentration parameters
    concs = [0.02, 0.05, 0.1, 0.3, 0.6, 1.0]
    for conc in concs:
        for _ in range(6):
            if time.time() >= deadline:
                break
            raw = rng.gamma(conc, 1.0, size=N)
            total = raw.sum()
            if total == 0.0:
                continue
            vec = raw / total
            _push_top(pool, _objective(vec), vec, TOP_K)

    # Incorporate the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                starter_up = _upsample(starter, N)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0.0:
                starter_up /= s
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        # silence any access errors – the variable may be undefined
        pass

    # Current best from the pool
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine the top candidates with Adam (exact max‑conv gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # give the majority of the remaining budget to the top 4 candidates
    top_to_refine = min(4, len(pool))
    per_candidate = max(5.0, remaining * 0.7 / top_to_refine)

    for idx in range(top_to_refine):
        vec = pool[idx][1]
        sub_deadline = time.time() + per_candidate
        if sub_deadline > deadline:
            sub_deadline = deadline
        refined, refined_score = _adam_exact_max(vec, sub_deadline, start_time, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to a larger resolution and polish if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq.shape[0] != N_large:
        # interpolate to the large grid
        best_seq = _upsample(best_seq, N_large)
        best_seq = _project_simplex(best_seq)   # ensure exact simplex
        # a short Adam run on the large grid
        sub_deadline = time.time() + remaining * 0.5
        refined_large, refined_large_score = _adam_exact_max(best_seq, sub_deadline, start_time, rng)
        if refined_large_score < best_score:
            best_score = refined_large_score
            best_seq = refined_large.copy()
            print(f"[{time.time() - start_time:.2f}s] Upsampled Adam improved best to {refined_large_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline, start_time, rng,
                                                    max_move_frac=0.025)

    # ------------------------------------------------------------------
    # 5) Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(best_seq.shape[0], 1.0 / best_seq.shape[0])

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]

"""
Implementation of a low‑score step‑function search.

The algorithm works in three stages:

1. **Initial pool** – generate a varied collection of analytic shapes
   (uniform, power‑decay, exponential, symmetric ramps,
   asymmetric Beta, random Dirichlet) and keep the TOP_K best
   candidates (according to the exact objective).

2. **Exact‑gradient Adam** – for the top few candidates run a projected
   Adam optimiser that uses the *real* sub‑gradient of the max‑convolution
   (no smooth‑max surrogate).  The best sequence found is retained.

3. **Polish** – if time allows the best sequence is up‑sampled to a larger
   resolution and a final Adam pass + a short simulated‑annealing mass‑swap
   phase are performed.  The result is projected onto the probability
   simplex and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (zero‑padded)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a probability vector."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape factories
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} , normalized."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) , normalized."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sym_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau (used in the reference)."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a size‑k sorted pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution for a given lag k
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """∇_a conv[k] for a fixed index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k <= n - 1:
        # indices 0 … k receive 2·a[k‑i]
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        start = k - (n - 1)
        g[start:] = 2.0 * a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Projected Adam optimiser using the exact max‑convolution gradient
# ----------------------------------------------------------------------
def _adam_exact_max(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam on the true max‑conv objective. Returns the best (seq,score)."""
    a = a0.copy()
    n = a.shape[0]

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.25                     # aggressive initial step
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # indices achieving the maximum (within a tiny tolerance)
        tol = 1e-12
        idxs = np.where(conv >= max_val - tol)[0]
        k = int(rng.choice(idxs)) if idxs.size > 1 else int(idxs[0])

        grad = _grad_max_conv(a, k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        it_f = float(it)
        m_hat = m / (1.0 - beta1 ** it_f)
        v_hat = v / (1.0 - beta2 ** it_f)
        step = step0 / math.sqrt(it_f)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.025) -> tuple[np.ndarray, float]:
    """Simple SA that transfers a small fraction of mass between two indices."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.3
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-20:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling via linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, N_target: int) -> np.ndarray:
    """Resize a vector to length N_target with linear interpolation."""
    n = seq.shape[0]
    if n == N_target:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, N_target, endpoint=False)
    return np.interp(x_new, x_old, seq)
