def _pgd_optimize(a, start_time, deadline, best_seq, best_score):
    n = len(a)
    L_fft_self = _next_pow2(2 * n - 1)
    L_fft_grad = _next_pow2(3 * n)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv
        conv = _conv_self(a, L_fft_self)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD iter={iteration} score={best_score:.6f}")

        # build w
        tol = maxc * 1e-12
        active_idx = np.where(conv >= maxc - tol)[0]
        w = np.zeros_like(conv)
        if active_idx.size == 0:
            # should not happen
            break
        w[active_idx] = 1.0 / active_idx.size

        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)  # length 3n-2, using fft with proper L
        grad = 2.0 * grad_full[n-1:n-1+n]   # slice

            # ensure non-negative before projection - but projection will enforce anyway
            a_new = _project_simplex(a_new)  # projection also ensures nonnegatives
            # compute new conv and score

            idx_src = np.random.randint(n)

"""
Step‑function optimiser for the max‑convolution challenge.

Approach
--------
* Initialise with the best known sequence (``height_sequence_1``) or the
  uniform distribution.
* Perform a *random‑restart projected sub‑gradient descent*:
  – draw a Dirichlet‑type candidate (different concentration parameters);
  – run a short PGD to drive the ℓ∞‑norm of the self‑convolution down;
  – keep the best candidate seen.
* After a large exploration phase, start a long *refinement* PGD from the
  best sequence discovered.
* The PGD uses the exact sub‑gradient of ``max(conv(a,a))`` (obtained from
  the active shifts) and a back‑tracking line‑search; every iterate is
  projected onto the probability simplex (non‑negative, sum = 1).  All
  convolutions are performed with FFTs (O(n log n)).
* The routine respects the 1000 s time‑budget and returns the best
  sequence found.  All numbers are non‑negative; the final list is
  normalised to sum = 1, exactly as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT‑based convolutions, simplex projection, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    if L is None:
        L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on  f(a)=max(conv(a,a))  (sum(a)=1)
# ----------------------------------------------------------------------
def _pgd_optimize(
    a: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on the ℓ∞‑norm of the
    self‑convolution (sum(a)=1).  ``a`` is assumed to be feasible.
    Returns the best sequence seen and its score.
    """
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)          # length for w * a_rev convolution
    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        it += 1

        # ---- evaluate current point ---------------------------------
        conv = _conv_self(a, L_self)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} "
                  f"score={best_score:.6f}")

        # ---- build the sub‑gradient of max(conv) --------------------
        # active shifts where the max is attained
        tol = maxc * 1e-12
        active = np.where(conv >= maxc - tol)[0]
        if active.size == 0:
            break    # should not happen

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size           # equal weight on the active set

        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev, L_grad)  # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice to length n

        # ---- descent direction & line search ------------------------
        # we move opposite to the sub‑gradient
        direction = -grad
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            # flat region – stop
            break

        # heuristic initial step (scaled to the magnitude of the objective)
        step = 0.1 * maxc / (dir_norm + 1e-15)

        # back‑tracking line search (projected onto the simplex)
        for _ in range(12):
            a_trial = a + step * direction
            a_trial = _project_simplex(a_trial)   # guarantees non‑neg, sum=1

            conv_trial = _conv_self(a_trial, L_self)
            new_score = 2.0 * n * conv_trial.max()
            if new_score <= cur_score + 1e-15:
                break
            step *= 0.5
        else:
            # no improvement – try a tiny random jitter to escape
            jitter = np.random.default_rng().normal(scale=1e-5, size=n)
            a_trial = a + jitter
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = _project_simplex(a_trial)
            conv_trial = _conv_self(a_trial, L_self)
            new_score = 2.0 * n * conv_trial.max()
            step = 0.0

        a = a_trial
        # continue – the loop will recompute the objective on the next
        # iteration and update best if it improved.

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine performs a random‑restart projected
    sub‑gradient descent, spending roughly 60 % of the budget on
    exploration (many short runs) and the remaining 40 % on a long
    refinement from the best candidate found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (the same size used in the reference run)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # initialise with the global best if it is available, otherwise uniform
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0)
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
            else:
                starter = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    best_seq = starter.copy()
    conv0 = _conv_self(best_seq)
    best_score = 2.0 * n * conv0.max()
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Exploration phase: many random Dirichlet starts, each run a short PGD
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.60 * (deadline - start_time)
    rng = np.random.default_rng(seed + 12345)

    while time.time() < explore_deadline:
        # draw a concentration parameter from a log‑uniform range
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a = a / a.sum()

        # short PGD (≈1500 iterations or time‑limited)
        a, best_score = _pgd_optimize(
            a, start_time, explore_deadline, best_seq, best_score, max_iters=1800
        )
        # keep the best sequence seen so far
        if best_score < 2.0 * n * _conv_self(a).max():
            # should not happen, but keep consistency
            best_seq = a.copy()
        else:
            # we already updated best_seq inside _pgd_optimize when it found
            # a better point, so just copy it back for the next restart
            best_seq = best_seq.copy()

    print(f"[{time.time() - start_time:.2f}s] Exploration finished – "
          f"best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement: long PGD from the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final refinement")
        best_seq, best_score = _pgd_optimize(
            best_seq, start_time, deadline, best_seq, best_score, max_iters=None
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – returning sequence")
    return [float(x) for x in best_seq]
