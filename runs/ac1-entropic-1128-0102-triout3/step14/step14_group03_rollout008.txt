"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of four stages:

1️⃣  Baseline / optional global starter.
2️⃣  A coarse parametric sweep over a geometric × power family and mixing
    with the uniform distribution.
3️⃣  Mirror‑descent optimisation of the *smooth* max‑convolution.
    The smooth max is realised by a soft‑max (log‑sum‑exp) with a
    gradually increasing temperature; the gradient is obtained by a
    single extra FFT, so each iteration costs only two FFTs.
4️⃣  A final cheap random mass‑transfer hill‑climb to polish the result.

All stages respect a 1000 s budget, regularly check the time limit and
keep the best sequence seen so far.  The returned list is normalised to
sum to 1 (the objective is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution of x and y using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape families (geometric × power) and mixing
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Return a normalised vector ∝ r**i * (i+1)**p."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)            # prevent overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()                      # final normalisation
    return mixed


# ----------------------------------------------------------------------
# Mirror‑descent on the smooth max‑convolution
# ----------------------------------------------------------------------


def _smooth_max_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(alpha·b_k) / Σ exp(alpha·b) in a
    numerically stable way.  Returns a 1‑D array of the same length as b.
    """
    b_max = b.max()
    exp_shift = np.exp(alpha * (b - b_max))
    w = exp_shift / exp_shift.sum()
    return w


def _mirror_descent_step(a: np.ndarray, L_self: int, L_grad: int,
                         alpha: float, eta: float) -> np.ndarray:
    """
    Perform one mirror‑descent iteration:
        a ← proj_simplex( a * exp(‑eta·∇f_smooth(a)) )
    where ∇f_smooth is the gradient of the smooth max‑convolution.
    """
    # self‑convolution (needed for the smooth weights)
    b = _conv_self_fft(a, L_self)

    # soft‑max weights for the smooth max
    w = _smooth_max_weights(b, alpha)

    # gradient of the smooth max (2·(w * a) truncated to length n)
    grad_full = 2.0 * _conv_fft(w, a, L_grad)
    grad = grad_full[: a.shape[0]]

    # mirror‑descent update (KL‑projection is just a renormalisation)
    factor = np.exp(-eta * grad)                # all entries > 0
    a_new = a * factor
    a_new = np.maximum(a_new, 1e-20)           # avoid exact zeros
    a_new /= a_new.sum()
    return a_new


def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation until ``deadline``.
    Returns the best sequence seen and its (exact) objective value.
    """
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score

    rng = np.random.default_rng()
    # schedule for the smoothness parameter α
    alpha_start = 800.0
    alpha_end = 20000.0
    eta0 = 3.0                     # initial step‑size (will decay as 1/√t)

    it = 0
    while time.time() < deadline:
        it += 1
        # linear schedule for α (slowly tighten the approximation of max)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 20000.0)

        eta = eta0 / np.sqrt(it)

        a = _mirror_descent_step(a, L_self, L_grad, alpha, eta)

        # exact objective (no smoothing) for bookkeeping
        b = _conv_self_fft(a, L_self)
        cur_sc = 2.0 * n * b.max()           # sum(a)=1 by construction

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] mirror it={it:,} "
                  f"α={alpha:.1f} η={eta:.4e} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the objective.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderately large vector gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 8192
    # If a global starter is present and already has a suitable size we
    # adopt its length (this can give us a head start on the search).
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=np.float64)
        if starter.ndim == 1 and starter.size >= 1024:
            n = int(starter.size)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (and possibly the global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # use a global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=np.float64)
        if starter.ndim == 1 and starter.size > 0:
            # adjust size if necessary
            if starter.size != n:
                if starter.size < n:
                    pad = np.full(n - starter.size, 0.0, dtype=np.float64)
                    starter = np.concatenate([starter, pad])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = evaluate_sequence(starter.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep (geometric × power + uniform mix)
    # ------------------------------------------------------------------
    sweep_deadline = start + 80.0
    r_grid = np.logspace(np.log10(0.93), np.log10(0.9999), 30)
    p_grid = np.linspace(-2.0, 2.0, 41)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} "
                          f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb over the three parametric variables
    # ------------------------------------------------------------------
    param_deadline = start + 300.0
    cur_r = 0.97
    cur_p = 0.0
    cur_lam = 0.3
    sigma_r = 0.015
    sigma_p = 0.20
    sigma_lam = 0.12
    cur_seq = best_seq.copy()
    cur_score = best_score

    while time.time() < param_deadline:
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape(n, r_new, p_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = evaluate_sequence(cand_new.tolist())

        if sc_new < cur_score:
            cur_r, cur_p, cur_lam = r_new, p_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start:.2f}s] param‑hill r={r_new:.6f} "
                      f"p={p_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter to escape plateaus
        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq,
                                 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = evaluate_sequence(cur_seq.tolist())

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑descent on the exact max‑convolution (smooth approximation)
    # ------------------------------------------------------------------
    mirror_deadline = deadline - 80.0
    if time.time() < mirror_deadline:
        a_opt, sc_opt = _run_mirror_descent(best_seq, n,
                                            mirror_deadline, start,
                                            best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] mirror‑descent final score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        L_self = _next_pow2(2 * n - 1)
        best_sc = evaluate_sequence(a.tolist())
        best_a = a.copy()
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = a[src] * rng.random() * 2e-3  # move ≤0.2 % of source mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            sc = evaluate_sequence(a_prop.tolist())
            if sc < best_sc:
                a = a_prop
                best_sc = sc
                best_a = a_prop.copy()
                print(f"[{time.time() - start:.2f}s] final hill‑climb score={best_sc:.6f}")

        best_seq = best_a
        best_score = best_sc

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
