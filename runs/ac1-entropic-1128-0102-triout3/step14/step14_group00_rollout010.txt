"""
Propose a low‑score non‑negative step function.

This implementation follows a *smooth‑max Adam* optimisation on the
probability simplex.  The objective 2·n·max(conv(a,a)) (with ∑a=1)
is differentiable if the max is replaced by a smooth approximation
w_k ∝ exp(-β·(max‑conv_k)).  By keeping the variables in log‑space
(u_i = log a_i) we obtain unrestricted updates; Adam is used for
adaptive step‑sizes.  The algorithm starts from the best known
construction (`height_sequence_1` if available) or from a random
Dirichlet vector, and runs until the supplied time budget (≈1000 s)
expires.  The best sequence encountered is returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _resize_to_n(seq: np.ndarray, n: int) -> np.ndarray:
    """Linearly interpolate a 1‑D array to length n and normalise."""
    old_n = seq.shape[0]
    if old_n == n:
        out = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        out = np.interp(x_new, x_old, seq)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s > 0.0:
        out /= s
    else:
        out = np.full(n, 1.0 / n)
    return out


def _objective_fast(a: np.ndarray, n: int, L_full: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) given a (already normalised to sum‑1).
    Uses a single FFT of length L_full (≥2·n‑1).
    """
    Af = np.fft.rfft(a, n=L_full)
    conv_full = np.fft.irfft(Af * Af, n=L_full)
    conv = conv_full[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of evaluate_sequence().
    Returns a list of non‑negative floats (sum is normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a length (power‑of‑two) – a trade‑off between quality
    #    and speed.  2¹⁵ = 32768 works well within the time budget.
    # ------------------------------------------------------------------
    n = 32768                          # length of the sequence
    L_full = _next_pow2(3 * n - 2)     # FFT length used for gradient step
    L_conv = _next_pow2(2 * n - 1)     # length for the pure self‑convolution

    # ------------------------------------------------------------------
    # 2) Initialise a probability vector.
    #    Prefer the global best if it exists, otherwise use a random Dirichlet.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _resize_to_n(starter, n)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.dirichlet(np.ones(n))

    # Ensure strict positivity (helps the log‑transform).
    eps_floor = 1e-14
    a0 = np.maximum(a0, eps_floor)
    a0 /= a0.sum()

    # Work in log‑space (unconstrained) – soft‑max will enforce positivity
    # and normalisation at each iteration.
    u = np.log(a0)

    # ------------------------------------------------------------------
    # 3) Adam optimiser hyper‑parameters.
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999      # momentum terms
    eps = 1e-8                     # Adam's epsilon
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # learning‑rate schedule (initial value, then decays like 1/√t)
    lr0 = 0.5

    # smoothing factor for the soft‑max approximation of the max.
    # We start moderate and increase slowly.
    beta_smooth = 1e5
    beta_growth = 1.05   # multiplicative increase every 5000 iterations

    # ------------------------------------------------------------------
    # 4) Main optimisation loop.
    # ------------------------------------------------------------------
    best_u = u.copy()
    best_score = math.inf
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- soft‑max to obtain the current probability vector -----
        u_shift = u - np.max(u)                 # for numerical stability
        a = np.exp(u_shift)
        a = np.maximum(a, eps_floor)           # tiny floor to avoid exact zeroes
        a_sum = a.sum()
        a = a / a_sum

        # ----- self‑convolution (auto‑correlation) via FFT -----
        Af = np.fft.rfft(a, n=L_conv)
        conv_full = np.fft.irfft(Af * Af, n=L_conv)
        conv = conv_full[: 2 * n - 1]           # linear convolution, length 2n‑1
        max_conv = conv.max()

        # ----- smooth approximation of the max (exponential weighting) -----
        # weight w_k ∝ exp(-β·(max‑conv_k)), larger β focuses more on the peak.
        w = np.exp(-beta_smooth * (max_conv - conv))
        w_sum = w.sum()
        if w_sum == 0.0:                       # fallback (should be rare)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max w.r.t. a (convolution) -----
        # we need g = 2 * Σ_k w_k ∇ conv_k , and ∇ conv_k[i] = 2·a_{k-i}
        # which reduces to a single convolution of a with the reversed weight.
        w_rev = w[::-1]
        Bf = np.fft.rfft(w_rev, n=L_full)
        # note: use the same FFT length for the cross‑convolution
        # (L_full ≥ 3n‑2 ensures the linear convolution is captured).
        Af_full = np.fft.rfft(a, n=L_full)
        conv_aw = np.fft.irfft(Af_full * Bf, n=L_full)

        # extract the required slice and reverse it (see analysis)
        slice_grad = conv_aw[n - 1: 2 * n - 1]          # length n
        grad_a = 2.0 * slice_grad[::-1]                # final gradient w.r.t. a

        # ----- map gradient to log‑space (soft‑max Jacobian) -----
        avg = np.dot(a, grad_a)        # scalar  ⟨a,∇J⟩
        g_u = a * (grad_a - avg)       # ∇J in log‑space (unconstrained)

        # ----- Adam update on the log‑variables -----
        m = beta1 * m + (1.0 - beta1) * g_u
        v = beta2 * v + (1.0 - beta2) * (g_u * g_u)
        lr = lr0 / (math.sqrt(it) + 1.0)   # simple decay

        u -= lr * m / (np.sqrt(v) + eps)

        # ----- occasionally evaluate the true objective (cheap) -----
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_u = u.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  score={cur_score:.6f}")

        # ----- increase the smoothing parameter slowly (more focus on the peak) -----
        if it % 5000 == 0:
            beta_smooth = min(5e6, beta_smooth * beta_growth)

        # ----- stop early if very little time remains -----
        if time.time() + 0.5 >= deadline:
            break

    # ------------------------------------------------------------------
    # 5) Recover the best probability vector and normalise.
    # ------------------------------------------------------------------
    u_best = best_u
    a_best = np.exp(u_best - np.max(u_best))
    a_best = np.maximum(a_best, eps_floor)
    a_best /= a_best.sum()

    # ------------------------------------------------------------------
    # 6) Final sanity check with the provided evaluate_sequence().
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(a_best.tolist())
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in a_best]
