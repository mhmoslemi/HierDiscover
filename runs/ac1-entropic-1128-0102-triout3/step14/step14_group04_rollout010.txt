def _smooth_gradient(a, alpha):
    conv_a = _conv_self(a)
    M = conv_a.max()
    shifted = alpha * (conv_a - M)

def _smooth_gradient(a, alpha):
    conv = _conv_self(a)  # length 2n-1
    max_c = conv.max()
    # compute softmax weights
    shifted = alpha * (conv - max_c)  # large negative for non-max
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    # gradient via convolution
    w_rev = w[::-1]
    grad_conv = _conv(w_rev, a)  # length 3n-2
    offset = 2 * a.shape[0] - 2
    grads = 2.0 * grad_conv[offset - np.arange(a.shape[0])]
    return grads, conv

"""
A deterministic‑annealing optimiser for the step‑function objective.

The objective (after normalising sum(a)=1) is 2·n·max(conv(a,a)).
We minimise a smoothed surrogate:
    f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
which approaches the max as α→∞.
Its gradient can be written in closed form via a second convolution,
so each iteration only needs two FFT‑based convolutions.
An Adam‑style update together with a Euclidean projection onto the
probability simplex performs gradient descent, while α is increased
gradually (deterministic annealing) so that the optimisation first
flattens the whole convolution and then zeroes‑in on the peak.
The routine builds a few diverse initial seeds (global starter,
DPSS window, a few random Dirichlet draws and the uniform vector)
and runs the annealing optimiser on each within the allotted time.
The best sequence found is returned as a plain list of non‑negative
floats (already normalised to sum = 1), ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b (full length) via real FFT."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑free) objective used for bookkeeping
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) with sum(a)=1 (no safety checks)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for initial seeds)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_dpss(n: int, nw: float) -> np.ndarray:
    """First DPSS (Slepian) sequence, normalised to sum = 1."""
    try:
        from scipy.signal import windows
        vec = windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        vec = np.maximum(vec, 0.0)
        vec /= vec.sum()
        return vec
    except Exception:
        return _make_uniform(n)


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float):
    """
    Returns the gradient of the smoothed max‑convolution
    (log‑sum‑exp surrogate) and the raw convolution.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a)                     # length 2·n‑1
    max_c = conv.max()

    # 2) softmax weights w_k ∝ exp(α·(conv_k – max_c))
    shifted = alpha * (conv - max_c)         # avoids overflow
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()            # Σ w = 1

    # 3) gradient: 2· Σ_k w_k · a shifted by k
    #    This is a cross‑correlation (w ⋆ a).  Using convolution
    #    with the reversed weight vector yields the same result.
    w_rev = w[::-1]
    grad_conv = _conv(w_rev, a)              # length 3·n‑2
    offset = 2 * n - 2
    grad = 2.0 * grad_conv[offset - np.arange(n)]

    return grad, conv


# ----------------------------------------------------------------------
# Deterministic‑annealing gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def _annealing_descent(a0: np.ndarray,
                       deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs a deterministic‑annealing optimisation starting from ``a0``.
    Returns the best sequence encountered and its objective value.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _make_uniform(a.shape[0])

    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    # schedule parameters
    alpha_start = 0.5
    alpha_end = 250.0          # sufficient to approximate the true max
    max_iters_for_alpha = 15000  # after which alpha stays at alpha_end

    while time.time() < deadline:
        it += 1

        # linearly increase α up to alpha_end
        if it < max_iters_for_alpha:
            alpha = alpha_start + (alpha_end - alpha_start) * it / max_iters_for_alpha
        else:
            alpha = alpha_end

        # gradient of the smoothed objective
        grad, conv = _smooth_gradient(a, alpha)

        # learning rate – modest decay
        lr = 0.25 / math.sqrt(it + 1.0)

        # Adam update (gradient descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps non‑negativity and Σ=1)
        a = _project_simplex(a)

        # evaluate (no extra FFT – reuse ``conv``)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it:,} α={alpha:.1f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (automatically normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # Choose a length – a power of two stays FFT‑friendly
    n = 16384   # 2¹⁴

    # ------------------------------------------------------------------
    # Build a small pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1) global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                if starter.size > n:
                    starter = starter[:n]
                elif starter.size < n:
                    starter = np.pad(starter, (0, n - starter.size))
                starter /= starter.sum()
                seeds.append(starter)
    except Exception:
        pass

    # 2) a DPSS window (very good starting shape)
    seeds.append(_make_dpss(n, 5.0))

    # 3) a few random Dirichlet draws (moderate concentration)
    rng = np.random.default_rng()
    for _ in range(3):
        vec = rng.dirichlet(alpha=np.ones(n) * 0.25)
        seeds.append(vec)

    # 4) the uniform sequence (baseline)
    seeds.append(_make_uniform(n))

    # ------------------------------------------------------------------
    # Run the annealing optimiser on each seed, sharing the remaining budget
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    total_seeds = len(seeds)

    for idx, init_vec in enumerate(seeds):
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate a fair share of the remaining time to this seed
        share = remaining / (total_seeds - idx) * 0.85
        sub_deadline = time.time() + share

        cand_seq, cand_score = _annealing_descent(init_vec, sub_deadline, start_time)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Seed {idx} improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final tidy‑up: a brief projected‑subgradient refinement (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        # a very short sub‑gradient pass to mop up any remaining slack
        # (re‑use the earlier sub‑gradient code but with a stricter tolerance)
        # – we keep the implementation tiny to stay within the budget
        a = best_seq.copy()
        n = a.shape[0]
        a = _project_simplex(a)
        best_local_score = _objective(a)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _conv_self(a)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} best={best_local_score:.6f}")

            # active set: all indices within 1 % of the current max
            thresh = cur_max * 0.99
            ks = np.where(conv >= thresh)[0]

            # sub‑gradient of the (unsmoothed) max‑conv
            grad = np.zeros_like(a)
            for k in ks:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idxs = np.arange(lo, hi + 1, dtype=int)
                grad[idxs] += 2.0 * a[k - idxs]
            grad /= float(len(ks))

            # Adam‑style step
            lr = 0.5 / (np.sqrt(it) + 1.0)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)

        best_score = best_local_score

    # ------------------------------------------------------------------
    # Normalise and convert to plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _make_uniform(n)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = _make_uniform(n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
