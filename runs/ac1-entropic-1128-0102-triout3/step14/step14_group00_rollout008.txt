def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 32768

    # build pool
    pool = []

    uniform = np.full(n, 1.0/n)
    pool.append((_objective(uniform), uniform))

    sqrt_shape = _make_sqrt_uniform_shape(n)
    pool.append((_objective(sqrt_shape), sqrt_shape))

    # random beta shapes (a,b log-uniform)
    for _ in range(150):
        a = 10 ** rng.uniform(-2.0, 0.7)
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v))

    # random mixtures of two shapes
    for _ in range(50):
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1-w) * _make_beta_shape(n, a2, b2)
        v = np.maximum(v, 0)
        v /= v.sum()
        pool.append((_objective(v), v))

    # keep best few
    pool.sort(key=lambda x: x[0])
    pool = pool[:5]   # top 5
    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # allocate time across EG runs for each candidate
    # compute remaining time
    remaining = deadline - time.time()
    # allocate maybe 0.5 for EG runs, 0.5 for anneal.

    # Determine per-candidate EG budget

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Choose a power‑of‑two length `n` (2¹⁵) – a good trade‑off between
   resolution and speed for the FFT based convolution.

2. Build a small pool of promising analytic shapes:
   * uniform,
   * the *square‑root‑of‑uniform* shape a_i = C(2i,i)/4^i  (the
     coefficients of (1‑z)^‑½, i.e. a_i ≈ 1/√i),
   * many beta‑distribution shapes,
   * mixtures of two beta‑shapes.
   Keep the best few (5).

3. Refine each candidate with a dedicated run of
   exponentiated‑gradient descent on the simplex.
   The update is multiplicative,
   the sub‑gradient is built from **all** convolution entries that are
   within a small relative tolerance of the current maximum
   (instead of only the exact maximisers).  The step size η follows a
   1/√t schedule, with a slightly larger initial η.

4. The best vector after the EG phase is polished by a short
   simulated‑annealing phase that swaps tiny packets of mass between
   random positions, accepting moves with a Metropolis rule.

5. Finally the vector is clipped to non‑negative values,
   normalised to sum = 1 and returned as a plain Python list.

All steps respect the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution using a real FFT (fast)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective: 2·n·max(conv) / (sum(a)^2)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return the objective value without the security checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sqrt_uniform_shape(n: int) -> np.ndarray:
    """
    Coefficients a_i = C(2i,i) / 4^i (the expansion of (1‑z)^‑½),
    truncated to length n and normalised to sum = 1.
    These coefficients satisfy (Σ a_i z^i)^2 = Σ_{k=0}^{2n‑2} z^k
    for the first n indices, which gives a very flat self‑convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(0, n - 1):
        # recurrence: a_{i+1} = a_i * ((2i+2)(2i+1)) / ((i+1)^2 * 4)
        num = (2 * i + 2) * (2 * i + 1)
        den = (i + 1) * (i + 1) * 4.0
        a[i + 1] = a[i] * num / den
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    eta0: float = 1.0,
    rel_tol: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (EG) descent on the probability simplex.
    Returns the best vector seen and its objective value.
    `rel_tol` controls the size of the active set:
      all convolution entries >= (1‑rel_tol)·max are used.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set – a small relative band around the maximum
        cutoff = max_val * (1.0 - rel_tol)
        ks = np.where(conv >= cutoff)[0]

        # sub‑gradient: average contribution of the active indices
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # avoid exact zeros, then renormalise
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(
    seq: np.ndarray, deadline: float, start_time: float
) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from 0.02 to 1e-6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)  # bias towards larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5  # up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step function within the given time budget.
    Returns a list of non‑negative floats that sum to 1 (the scale is
    irrelevant for the evaluation function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # 1) Choose the FFT‑friendly length
    # --------------------------------------------------------------
    n = 32768  # 2¹⁵ – gives a good resolution while staying fast

    # --------------------------------------------------------------
    # 2) Build an initial pool of candidate shapes
    # --------------------------------------------------------------
    pool: list[tuple[float, np.ndarray]] = []

    # uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # sqrt‑uniform analytic shape (central‑binomial coefficients)
    sqrt_unif = _make_sqrt_uniform_shape(n)
    pool.append((_objective(sqrt_unif), sqrt_unif.copy()))

    # many random beta shapes
    for _ in range(150):
        a = 10 ** rng.uniform(-2.0, 0.7)  # roughly 0.01 … 5
        b = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # random convex combinations of two beta shapes
    for _ in range(50):
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v1 = _make_beta_shape(n, a1, b1)
        v2 = _make_beta_shape(n, a2, b2)
        v = w * v1 + (1.0 - w) * v2
        v = np.maximum(v, 0.0)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:5]  # keep the top 5
    best_score, best_seq = pool[0]

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Run EG on each of the top candidates (time‑share)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        eg_total = remaining * 0.6
        anneal_total = remaining - eg_total
        per_cand = eg_total / len(pool)

        for idx, (sc, vec) in enumerate(pool):
            cand_deadline = time.time() + per_cand
            cand_seq, cand_sc = _exp_gradient_descent(
                vec, cand_deadline, start_time, eta0=1.0, rel_tol=1e-4
            )
            if cand_sc < best_score:
                best_score, best_seq = cand_sc, cand_seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] New global best from EG (idx {idx}) = {best_score:.6f}"
                )
    else:
        anneal_total = remaining

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = time.time() + anneal_total
        best_seq, best_score = _anneal(best_seq, anneal_deadline, start_time)
        print(
            f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 5) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    totals = best_seq.sum()
    if totals > 0.0:
        best_seq /= totals
    else:
        best_seq = np.full(n, 1.0 / n)

    print(
        f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}"
    )
    return [float(v) for v in best_seq]
