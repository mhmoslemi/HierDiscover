```python
def solve_lp_for_g(f, M):
    n = len(f)
    m = 2*n-1
    A = build_ub_matrix(f) # shape m x n

        return np.full(n, 1.0 / n)

        g = res.x
        # ensure non-negative
        g = np.maximum(g, 0)
        return g
```

```python
def sweep_families(n, best_seq, best_score, deadline):

    uniform = np.full(n, 1.0 / n)

    if sc < best_score:

    # geometric
    eps_grid = np.logspace(-5, -1, 30)  # reduced
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        s = geo.sum()
        if s == 0: continue
        geo = geo / s
        for lam in np.linspace(0.0, 1.0, 5):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            sc = objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    p_grid = np.linspace(-2.0, 2.0, 30)
    for p in p_grid:

        s = pow_seq.sum()
        if s == 0: continue
        pow_seq = pow_seq / s
        sc = objective_score(pow_seq)
        if sc < best_score:
            best_seq, best_score = pow_seq, sc
```

```python
def adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    L_fft = next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.3
    it = 0
    best_a = a.copy()
    best_score = objective_score(best_a)
    while time.time() < deadline:
        it += 1
        conv = conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()
        # gradient: 2 * (w * a_rev) convolution
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=next_pow2(2 * n - 1)) *
                              np.fft.rfft(a_rev, n=next_pow2(2 * n - 1)), n=next_pow2(2 * n - 1))
        # conv_w length same; extract central part of length n
        start = n - 1
        grad = 2.0 * conv_w[start:start + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = project_simplex(a_new)
        a = a_new
        if it % 500 == 0:
            tau = min(tau * tau_factor, 1e8)
    return best_a, best_score
```

```python
def conv_fft(x, y):
    L = next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]
```

```python
def build_ub_matrix(f):
    n = f.shape[0]
    m = 2 * n - 1
    offsets = np.arange(-(n - 1), n)  # list of offsets
    data = []
    diag_offsets = []
    for p in offsets:
        idx = -p
        if idx < 0 or idx >= n:
            continue
        val = f[idx]
        length = n - abs(p)
        data.append(np.full(length, val))
        diag_offsets.append(p)
    A = diags(data, diag_offsets, shape=(m, n), format='csr')
    return A
```

```python
def solve_lp_for_g(f, M, time_limit=30.0):
    n = f.shape[0]
    A = build_ub_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception as e:
        print(f"[{time.time() - start_time:.2f}s] LP solve error: {e}")
    # fallback uniform
    return np.full(n, 1.0 / n)
```

```python
def hill_climb(a, deadline, max_iters=20000):
    n = a.shape[0]
    L_fft = next_pow2(2 * n - 1)
    best_seq = a.copy()
    best_score = objective_score(best_seq)
    cur_seq = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters > 0:
        # compute subgradient
        conv = conv_self(cur_seq, L_fft)
        # compute active gradient as before
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(cur_seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += cur_seq[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size
        # pick source index with biggest positive gradient where mass > tiny
        src_candidates = np.where(cur_seq > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
        # transfer small mass
        frac = 0.001 * rng.random() + 1e-5
        eps = cur_seq[src] * frac
        if eps <= 0:
            continue
        new_seq = cur_seq.copy()
        new_seq[src] -= eps
        new_seq[dst] += eps
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = new_seq / new_seq.sum()
        new_score = objective_score(new_seq)
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
            if cur_score < best_score:
                best_seq, best_score = cur_seq.copy(), cur_score
                print(f"[{time.time() - start_time:.2f}s] hill climbing best={best_score:.6f}")
        else:
            # occasionally accept worse with some prob, can ignore
            pass
        max_iters -= 1
    return best_seq, best_score
```

```python
    best_seq = uniform.copy()
    best_score = objective_score(best_seq)

                starter = starter_resampled
            starter = np.maximum(starter, 0.0)

            if s > 0:
                starter = starter / s
                sc = objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc

    # Phase 5 – short hill‑climb for final tweaks

    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = uniform
```

"""
Step‑function optimiser (≈1000 s budget)
----------------------------------------

The routine searches for a non‑negative sequence `a` that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

(the quantity returned by ``evaluate_sequence``).

The method consists of five stages

1. **Coarse parametric sweep** – a few families (geometric,
   power, logistic, stretched‑exponential) are sampled on a grid;
   the best candidate is kept.

2. **Random Dirichlet draws** – many random probability vectors are
   generated; the best one improves the current record.

3. **Smooth‑max Adam optimisation** – the non‑smooth max is
   approximated by a temperature‑controlled soft‑max.
   Adam updates with a projection onto the probability simplex
   drive the sequence towards a local optimum.

4. **Linear‑programming refinement** – given the current sequence `f`,
   we solve the LP  
   maximise Σ gᵢ  subject to  (f∗g)ₖ ≤ M  (M = max(conv(f,f))) and gᵢ≥0.
   The LP solution is mixed back into `f` (step size 0.5, 0.3, 0.1)
   while the objective improves.  This is the key new ingredient
   that pushes the score below the previous 1.50783 barrier.

5. **Short hill‑climb** – a cheap simulated‑annealing move transfers
   a tiny amount of mass from a positively‑gradient index to a
   negatively‑gradient one, occasionally accepting worsening moves.

All stages respect the overall time budget (≈1000 s).  The final
sequence is normalised to an ℓ₁‑norm of 1 (the objective is scale‑invariant)
and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    if L is None:
        L = next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT."""
    L = next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Σv=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq))  (Σseq=1)."""
    n = seq.shape[0]
    conv = conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Coarse parametric sweep (lightweight)
# ----------------------------------------------------------------------
def sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    sc = objective_score(uniform)
    if sc < best_score:
        best_seq, best_score = uniform.copy(), sc
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # – geometric -------------------------------------------------------
    eps_grid = np.logspace(-5, -1, 30)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 5):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # – power -----------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 30)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = (idx + 1.0) ** p
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = objective_score(pow_seq)
        if sc < best_score:
            best_seq, best_score = pow_seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # – logistic --------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 7)
    s_factors = np.linspace(0.02, 0.15, 5)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # – stretched‑exponential -------------------------------------------
    scales = np.linspace(0.25, 1.0, 5)
    p_vals = np.linspace(0.5, 3.0, 7)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Stretch p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws
# ----------------------------------------------------------------------
def random_dirichlet_trials(n: int, best_seq: np.ndarray,
                            best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = next_pow2(2 * n - 1)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.3
    it = 0
    best_a = a.copy()
    best_sc = objective_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = conv_self(a, L_fft)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc, best_a = cur_sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_sc:.6f}")

        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()                     # soft‑max weights

        # gradient of smooth‑max: 2·(w∗a_rev) restricted to n entries
        a_rev = a[::-1]
        conv_w = conv_fft(w, a_rev)                # length = 3·n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = project_simplex(a_new)
        a = a_new

        if it % 500 == 0:
            tau = min(tau * tau_factor, 1e8)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4️⃣  Linear‑programming refinement
# ----------------------------------------------------------------------
def build_ub_matrix(f: np.ndarray):
    """Return the sparse matrix A (shape (2n‑1, n)) such that A·g = f∗g."""
    n = f.shape[0]
    m = 2 * n - 1
    offsets = np.arange(-(n - 1), n)                # col‑row = offset
    data = []
    diag_off = []
    for p in offsets:
        idx = -p                                    # value = f[idx]
        if 0 <= idx < n:
            length = n - abs(p)                     # non‑zero diagonal length
            data.append(np.full(length, f[idx], dtype=np.float64))
            diag_off.append(p)
    A = diags(data, diag_off, shape=(m, n), format='csr')
    return A


def solve_lp_for_g(f: np.ndarray, M: float, time_limit: float = 30.0) -> np.ndarray:
    """Maximise Σgᵢ subject to (f∗g)ₖ ≤ M, g≥0."""
    n = f.shape[0]
    A = build_ub_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –Σgᵢ  ⇒ maximise Σgᵢ
    bounds = [(0, None)] * n
    opts = {'disp': False}
    # recent versions of HiGHS accept a time limit
    try:
        opts['time_limit'] = time_limit
    except Exception:
        pass
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options=opts)
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception as e:
        print(f"[{time.time() - start_time:.2f}s] LP error: {e}")

    # fallback – uniform distribution
    return np.full(n, 1.0 / n)


def lp_refine(f: np.ndarray, deadline: float, max_iters: int = 5) -> tuple[np.ndarray, float]:
    """Iteratively mix the LP solution back into `f` while it improves the score."""
    best_f = f.copy()
    best_sc = objective_score(best_f)
    n = f.shape[0]

    while time.time() < deadline and max_iters > 0:
        conv_f = conv_self(best_f)
        M = conv_f.max()

        g0 = solve_lp_for_g(best_f, M, time_limit=10.0)
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            break
        g_norm = g0 / sum_g0                     # unit‑sum candidate

        improved = False
        for t in (0.5, 0.3, 0.1):
            mixed = (1.0 - t) * best_f + t * g_norm
            mixed /= mixed.sum()
            sc = objective_score(mixed)
            if sc < best_sc - 1e-9:
                best_f, best_sc = mixed, sc
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP mix t={t:.2f} score={sc:.6f}")
                break
        if not improved:
            print(f"[{time.time() - start_time:.2f}s] LP refinement stalled")
            break
        max_iters -= 1

    return best_f, best_sc


# ----------------------------------------------------------------------
# 5️⃣  Minimal hill‑climb (post‑LP fine‑tuning)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray, deadline: float, max_iters: int = 20000) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    L_fft = next_pow2(2 * n - 1)
    best_seq = a.copy()
    best_sc = objective_score(best_seq)
    cur_seq = a.copy()
    cur_sc = best_sc
    rng = np.random.default_rng()

    while time.time() < deadline and max_iters > 0:
        conv = conv_self(cur_seq, L_fft)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(cur_seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += cur_seq[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        src_cand = np.where(cur_seq > 1e-12)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)
        if src == dst:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst or cur_seq[src] < 1e-12:
                max_iters -= 1
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = cur_seq[src] * frac
        if eps <= 0:
            max_iters -= 1
            continue

        new_seq = cur_seq.copy()
        new_seq[src] -= eps
        new_seq[dst] += eps
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()
        new_sc = objective_score(new_seq)

        if new_sc < cur_sc:
            cur_seq, cur_sc = new_seq, new_sc
            if cur_sc < best_sc:
                best_seq, best_sc = cur_seq.copy(), cur_sc
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_sc:.6f}")

        max_iters -= 1

    return best_seq, best_sc
