"""
Improved search for a low‑score non‑negative step function.

The algorithm works on a fixed length (n = 4096) and uses a smooth
approximation of the “max” in the objective.  The search proceeds as
follows:

1. Initialise the sequence from the global starter (if present) or from
   a uniform distribution.
2. Optimise the soft‑max parametrisation a_i = exp(w_i)/∑exp(w) with an
   Adam optimiser.  The true objective 2·n·max(conv) is replaced by a
   smooth maximum using a temperature β that grows during the run.
   Gradients are obtained analytically:
        ∂f/∂a_j = 4·n·∑_k p_k·a_{k‑j}
   where p_k are the normalised soft‑max weights of the convolution.
3. Periodically evaluate the exact (non‑smooth) objective and keep the
   best vector found.
4. With any remaining budget run a short simulated‑annealing polish
   (mass moves between two positions) to escape the last local basin.

The routine respects the 1000 s time budget and returns a list of
non‑negative floats whose sum is normalised to one (the scaling is
irrelevant for the evaluation function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based convolution, exact objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simulated‑annealing polish (same as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Simple anneal that moves a random packet of mass between two indices."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 600_000  # a bit larger than the reference

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from 0.02 down to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # source biased by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

    beta_inc = (beta_target - beta) / 60000   # linear increase

    it = 0
    lr0 = 0.12                               # initial learning rate (Adam uses decay)

    while time.time() < deadline:
        it += 1

        # ------- soft‑max to obtain a on the simplex -------
        max_w = w.max()
        a = np.exp(w - max_w)
        a = a / a.sum()

        # ------- convolution and smooth‑max weights -------
        conv = _conv_self(a)                # length 2·n‑1
        conv_max = conv.max()
        # use a numerically stable soft‑max over the convolution values
        exp_arg = beta * (conv - conv_max)
        p = np.exp(exp_arg)
        p_sum = p.sum()
        if p_sum == 0.0:        # should not happen, but guard against underflow
            p = np.full_like(p, 1.0 / p.size)
        else:
            p = p / p_sum

        # ------- gradient wrt a (smooth objective) -------
        # grad_a = 4·n·corr(p, a)  ; correlation can be computed via np.correlate
        grad_a = 4.0 * n * np.correlate(p, a, mode='valid')

        # ------- gradient wrt logits w (soft‑max chain rule) -------
        scalar = np.dot(a, grad_a)
        grad_w = a * (grad_a - scalar)

        # ------- Adam update (with a simple 1/√t decay) -------
        lr = lr0 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- increase smoothing temperature -------
        if beta < beta_target:
            beta += beta_inc

        # ------- occasional exact evaluation (every few hundred iters) -------
        if it % 300 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} β={beta:.1f} score={cur_score:.6f}")

        # safety break if we have spent most of the budget
        if it >= 200_000:   # generous upper bound; typically we stop earlier
            break

    # --------------------------------------------------------------
    # 3) Final annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _anneal(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] after anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Normalise and convert to plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished, final exact score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
