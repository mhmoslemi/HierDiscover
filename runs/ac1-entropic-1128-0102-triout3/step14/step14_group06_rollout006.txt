"""
Step‑function optimiser – 2·n·max(conv(a,a))/sum(a)².

The search works in three stages:

1. **Diverse initial pool** – a handful of analytically‑defined shapes
   (uniform, symmetric/asymmetric Beta densities, power‑ramps,
   exponential‐decay, two‑block step shapes) together with a few random
   Dirichlet vectors.  The best‑k candidates (by the exact objective) are
   kept.

2. **Simulated‑annealing warm‑start** – a fast Metropolis walk that moves
   mass between coordinates (both single‑move and swaps).  It runs on a
   modest fraction of the time budget and supplies a good starting point.

3. **Refinement** – first an exponentiated (multiplicative) sub‑gradient
   descent on the true ℓ∞ objective, followed by a smooth‑max Adam
   descent.  Because the problem size (N) is kept moderate (≈1.6·10⁴) the
   FFT‑based convolutions are cheap, allowing many thousands of iterations
   and a tighter final bound.

The code adapts the problem size to the remaining time, prints progress
messages and finally returns a plain Python list of non‑negative floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/(sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “power‑ramp” (optional flat plateau) – normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1} – normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) – normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_decay_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential‑decay shape: a_i ∝ e^{-λ·i}+e^{-λ·(n‑1‑i)}."""
    i = np.arange(n, dtype=float)
    shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_two_block_shape(n: int, edge_len: int, edge_mass: float) -> np.ndarray:
    """
    Symmetric two‑block shape:
        * first `edge_len` entries on each side have the same height,
        * the centre has a constant (different) height.
    `edge_mass` is the total probability mass allocated to both edge blocks.
    """
    edge_len = max(0, min(n // 2 - 1, edge_len))
    edge_mass = max(0.0, min(1.0, edge_mass))
    mid_mass = 1.0 - edge_mass
    # number of positions holding the middle height
    mid_len = n - 2 * edge_len
    if edge_len == 0:
        edge_h = 0.0
    else:
        edge_h = edge_mass / (2 * edge_len)
    mid_h = mid_mass / mid_len if mid_len > 0 else 0.0
    shape = np.full(n, mid_h, dtype=float)
    if edge_len > 0:
        shape[:edge_len] = edge_h
        shape[-edge_len:] = edge_h
    # normalise (numerical safety)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Optimisation primitives

def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on the exact
    ℓ∞ objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # active set (indices attaining the current maximum, tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * correlation(w, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # decreasing step‑size
        eta0 = 0.5
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        # project onto simplex
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam descent on a smooth‑max surrogate of the max‑convolution.
    The temperature is linearly annealed to a tiny value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.07          # base learning rate
    T0 = 0.02             # initial temperature
    T_end = 1e-5          # final temperature

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # anneal temperature
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * correlation(w, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (keep N moderate so we can afford many iterations)
    # ------------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 800:
        N = 16384            # still large enough for a fine discretisation
    elif remaining_total > 400:
        N = 8192
    else:
        N = 4096

    print(f"[{time.time() - start_time:.2f}s] Problem size N = {N}")

    # ------------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    TOP_K = 7
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform
    _add(np.full(N, 1.0 / N))

    # symmetric Beta shapes
    for alpha in np.linspace(0.1, 1.5, 20):
        _add(_make_beta_shape(N, alpha))

    # asymmetric Beta shapes
    for a in [0.3, 0.5, 0.8, 1.2]:
        for b in [0.3, 0.5, 0.8, 1.2]:
            _add(_make_beta_shape_asym(N, a, b))

    # power‑ramp shapes (including a few steeper ramps)
    for p in np.linspace(0.2, 4.0, 12):
        for plateau in np.linspace(0.0, 0.5, 4):
            _add(_make_shape(N, p, plateau))

    # exponential‑decay shapes
    for lam in [0.0005, 0.001, 0.003, 0.01, 0.03]:
        _add(_make_exp_decay_shape(N, lam))

    # two‑block step shapes (edge length / edge mass grid)
    edge_lens = [max(1, int(N * frac)) for frac in np.linspace(0.01, 0.25, 6)]
    edge_masses = np.linspace(0.05, 0.45, 8)
    for L in edge_lens:
        for em in edge_masses:
            _add(_make_two_block_shape(N, L, em))

    # random Dirichlet vectors (sparse)
    rng = np.random.default_rng(seed + 9999)
    for _ in range(60):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.5, 1.0, size=N)   # 0.5 → promotes sparsity
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
            _add(a)

    # incorporate a global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _add(starter_up)
    except Exception:
        pass

    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing warm‑start (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.15
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    rng = np.random.default_rng(seed + 12345)
    max_move_factor = 0.25
    T_start = 0.6
    T_end = 1e-9
    it = 0

    while time.time() < phase1_deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * (T_end / T_start) ** frac

        # decide which proposal to use
        if rng.random() < 0.7:          # mass‑move proposal
            src = rng.choice(N, p=cur_seq)
            if cur_seq[src] < 1e-15:
                continue
            dst = rng.integers(N)
            if src == dst:
                continue
            delta = cur_seq[src] * max_move_factor * rng.random()
            if delta <= 0.0:
                continue
            a_prop = cur_seq.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
        else:                           # swap proposal (exchange two values)
            i, j = rng.choice(N, size=2, replace=False)
            a_prop = cur_seq.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        new_score = _objective(a_prop)
        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / T):
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

        # occasional restarts from the pool
        if it % 200000 == 0 and len(top) > 1:
            choice = rng.integers(len(top))
            cur_seq = top[choice][1].copy()
            cur_score = _objective(cur_seq)

        # occasional Gaussian jitter to keep exploration alive
        if it % 80000 == 0:
            sigma = 0.02
            noisy = cur_seq + rng.normal(0.0, sigma, size=N)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                noisy /= s
                cur_seq = noisy
                cur_score = _objective(cur_seq)

    # ------------------------------------------------------------------
    # 3) Exponentiated sub‑gradient descent (≈70 % of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_end = time.time() + remaining * 0.70

        # start from the best SA result
        seq, sc = _exp_subgradient_descent(best_global_seq, sub_end, start_time)
        if sc < best_global_score:
            best_global_seq, best_global_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD (SA start) improved best={sc:.6f}")

        # additionally try the other top‑k candidates while time permits
        idx = 0
        while time.time() < sub_end and idx < len(top):
            vec = top[idx][1]
            seq, sc = _exp_subgradient_descent(vec, sub_end, start_time)
            if sc < best_global_score:
                best_global_seq, best_global_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved best={sc:.6f}")
            idx += 1

    # ------------------------------------------------------------------
    # 4) Smooth‑max (Adam) polishing for the leftover time (≈15 %)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_seq, best_global_score = refined_seq, refined_score
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
