"""Step‑function optimiser – extended search.

The routine explores a wide variety of parametric families (geometric,
inverse‑power, beta, stretched‑exponential, logistic) with dense grids,
keeps the best few candidates and refines each of them independently
with a smooth‑max Adam optimiser.  A short projected sub‑gradient pass
focuses on the true non‑smooth objective and finally a simulated‑annealing
hill‑climb cleans up the remaining irregularities.

All heavy lifting (FFT convolution, simplex projection, sub‑gradient) is
implemented with NumPy only.  The code respects the 1000 s budget, prints
progress, and returns a plain Python list of non‑negative coefficients."""
import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the (non‑smooth) max‑convolution at a."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during the search (2·n·max(conv(seq,seq))) for sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-8:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            _maybe_push(seq, sc)

    # Extract the K best sequences from the heap (lowest scores)
    top_seqs = [item[1] for item in heapq.nsmallest(top_k, heap, key=lambda x: -x[0])]
    return top_seqs, best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (adds diversity)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, top_k: int,
                             best_seq: np.ndarray, best_score: float,
                             deadline: float) -> tuple[list, np.ndarray, float]:
    """Draw many Dirichlet vectors and keep the best K of them."""
    rng = np.random.default_rng()
    heap = []
    def _maybe_push(seq, sc):
        nonlocal heap, best_seq, best_score
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet new best {sc:.6f}")
        if len(heap) < top_k:
            heapq.heappush(heap, (-sc, seq.copy()))
        else:
            if sc < -heap[0][0]:
                heapq.heapreplace(heap, (-sc, seq.copy()))

    while time.time() < deadline:
        # concentration from 0.01 to 10, log‑uniform
        log_c = rng.uniform(np.log(0.01), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        _maybe_push(a, sc)
    top_seqs = [item[1] for item in heapq.nsmallest(top_k, heap, key=lambda x: -x[0])]
    return top_seqs, best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate (multiple restarts)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Minimise smooth max (log‑sum‑exp) of a*a using Adam."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 15.0
    tau_factor = 1.3

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} new best {cur_score:.6f}")

        # smooth‑max weights
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (2 * (w * a_rev) restricted)
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)                # length = 3n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature gradually
        if it % 1500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    step = 0.12                         # initial step – will be decayed
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} best {cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # simple step‑size decay
        step *= 0.9995
        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Directed hill‑climb that transfers little mass from a positive to a negative gradient entry."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # temperature schedule (open‑ended)
    T0 = 0.02
    total = max(1e-9, deadline - time.time())
    decay = (1e-6 / T0) ** (1.0 / total)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.006 + 1e-6     # up to ~0.6 % transfer
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    The algorithm consists of:
      1. dense parametric sweep (keeps the K best candidates);
      2. a large set of random Dirichlet draws;
      3. several independent Adam optimisations on the smooth‑max surrogate;
      4. a short projected sub‑gradient sweep on the true max;
      5. a final simulated‑annealing hill‑climb.
    All steps respect the overall time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # problem size (same as the reference run)
    n = 4096

    # initialise with uniform distribution
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline {best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep – keep the 15 best candidates
    # ------------------------------------------------------------------
    top_k = 15
    phase_deadline = deadline - 250.0               # reserve ~250 s for later work
    if time.time() < phase_deadline:
        top_candidates, best_seq, best_score = _sweep_families(
            n=n,
            epsilon=1e-12,
            top_k=top_k,
            best_seq=best_seq,
            best_score=best_score,
            deadline=phase_deadline,
        )
    else:
        top_candidates = []

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws – augment the candidate pool
    # ------------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        extra, best_seq, best_score = _random_dirichlet_trials(
            n=n,
            top_k=top_k,
            best_seq=best_seq,
            best_score=best_score,
            deadline=phase_deadline,
        )
        top_candidates.extend(extra)

    # keep only the distinct best K (by score) to limit later work
    # sort and trim
    uniq = {}
    for seq in top_candidates:
        sc = _objective_score(seq)
        uniq[sc] = seq
    sorted_scores = sorted(uniq.keys())
    top_candidates = [uniq[sc] for sc in sorted_scores[:top_k]]

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation – run several short restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # leave 180 s for the final refinement stages
    if remaining > 200:
        time_per_run = (remaining - 180) / max(1, len(top_candidates))
        for i, cand in enumerate(top_candidates):
            if time.time() >= deadline - 180:
                break
            sub_deadline = time.time() + time_per_run
            print(f"[{time.time() - start_time:.2f}s] Adam restart {i+1}/{len(top_candidates)}")
            best_seq, best_score = _adam_optimize(
                cand, start_time, sub_deadline, best_seq, best_score
            )
            print(f"[{time.time() - start_time:.2f}s] After Adam run best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        sub_deadline = deadline - 150
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient refinement")
        best_seq, best_score = _subgradient_descent(
            best_seq, start_time, sub_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After sub‑grad best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
