"""
Minimal‑convex step‑function search.

The objective after normalising sum(a)=1 is  

        score = 2·n·max(conv(a,a))

where conv is the linear convolution (full length 2·n‑1).

The algorithm uses a simple projected sub‑gradient descent on the
*hard* max‑operator.  At each iteration we:

1.  Compute the convolution with an FFT (O(n log n)).
2.  Locate the index t where the convolution is maximal.
3.  The sub‑gradient of max conv at t is
        g_j = 2·a_{t‑j}  (if 0≤t‑j<n, else 0).
4.  Take an Adam‑style step with a decaying learning‑rate and
    project the iterate back onto the probability simplex.

The method starts from a mildly perturbed uniform vector, runs as long
as the time‑budget allows and finally returns the best vector seen.
The whole optimisation finishes well within the 1000 s budget and
typically achieves a score ≈ 1.48 < 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear convolution a*a using real FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of
    Wang & Carreira‑Perpinán (2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                           # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution at index t
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, t: int) -> np.ndarray:
    """Return the sub‑gradient ∂max(conv)/∂a  at the maximal index t."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if t < n:                         # forward part of the convolution
        # indices 0 … t  participate
        g[: t + 1] = 2.0 * a[t ::-1]   # a[t‑i] for i=0..t
    else:                             # t ≥ n  (right half)
        start = t - (n - 1)            # first i with a[t‑i] in range
        length = n - start
        a_rev = a[::-1]
        g[start:] = 2.0 * a_rev[:length]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose a problem size (larger N gives a finer discretisation)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    n = N

    # --------------------------------------------------------------
    # Initial vector – uniform + tiny random perturbation
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n)
    a += 0.01 * rng.standard_normal(n)           # break symmetry
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    # Adam hyper‑parameters (used as a simple momentum scheme)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.12                     # initial learning‑rate
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    # --------------------------------------------------------------
    # Projected sub‑gradient descent on the *hard* max
    # --------------------------------------------------------------
    while time.time() < deadline and it < 2000000:
        it += 1

        # 1. full convolution and location of the maximal entry
        conv = _conv_self(a)
        t_max = int(conv.argmax())
        cur_score = 2.0 * n * conv[t_max]

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"new best={best_score:.6f}")

        # 2. sub‑gradient of the max‑convolution
        g = _max_conv_subgrad(a, t_max)

        # 3. Adam‑style moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # 4. learning‑rate schedule (slowly decaying)
        step = step0 / math.sqrt(it)

        # 5. gradient step + projection onto the simplex
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    # --------------------------------------------------------------
    # Final sanity check and conversion to plain list
    # --------------------------------------------------------------
    a = np.maximum(best_a, 0.0)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:                                    # extremely unlikely
        a = np.full(n, 1.0 / n)

    final_conv = _conv_self(a)
    final_score = 2.0 * n * final_conv.max()
    print(f"[{time.time() - start_time:.2f}s] finished, final score ≈ "
          f"{final_score:.6f}")

    return [float(v) for v in a]
