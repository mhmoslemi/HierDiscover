def _final_hill_climb_enhanced(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    if cur_score < best_score: best_seq,best_score = a.copy(), cur_score
    # schedule: start with max_transfer = 0.1 (10% of source mass) gradually decay
    max_frac_start = 0.1
    min_frac = 0.001
    frac = max_frac_start
    dec_period = (deadline - time.time()) / (np.log(max_frac_start / min_frac))
    # We'll just linear decay based on time.
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linearly reduce frac
        frac = max_frac_start * (1 - elapsed / (deadline - start_time))
        if frac < min_frac:
            frac = min_frac
        # compute subgradient for direction
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
        eps = a[src] * (rng.random() * frac + 1e-6)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue
        a_prop = _project_simplex(a_prop)  # ensure sum=1, non-neg
        new_score = _objective_score(a_prop)
        delta = new_score - cur_score
        # temperature schedule
        T = max(0.02 * (1 - elapsed/(deadline-start_time)), 1e-6)
        accept = delta < 0 or rng.random() < np.exp(-delta / T)
        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill climb improved score {best_score:.6f}")
    return best_seq, best_score

"""
Propose a low‑autocorrelation step function.

The search works in three phases (all run within a single 1000‑second budget)

1️⃣  **Seeding.**  If a previous construction ``height_sequence_1`` is
    available it is normalised and used as a starting point;
    otherwise a uniform distribution of length 4096 is taken.
    A large number of random Dirichlet draws (different concentration
    parameters) are evaluated with a fast FFT‑based objective
    ``2·n·max(conv(a,a))`` (the sum is kept = 1, therefore this equals the
    value returned by ``evaluate_sequence``).

2️⃣  **Projected sub‑gradient descent.**  The true objective
    ``max(conv(a,a))`` is non‑smooth; its sub‑gradient at a point is
    obtained by ``_subgradient_maxconv`` (average of the gradients of all
    active convolution indices).  We repeatedly move
    ``a ← a – η·grad`` with an adaptive step size η, clip to non‑negativity
    and project back onto the probability simplex (Euclidean projection).
    The step size is increased when a move improves the score and reduced
    otherwise – a simple line‑search‑free scheme that works well in practice.
    Random “jitter’’ (a tiny Dirichlet mixture) is added every few hundred
    iterations to escape plateaus.

3️⃣  **Aggressive hill‑climb / simulated‑annealing.**  From the best point
    found by the descent we perform a directed mass‑transfer hill‑climb:
    the index with the largest *positive* sub‑gradient loses a small
    fraction of its mass, which is added to the index with the most
    *negative* sub‑gradient.  The transfer size decays linearly with time.
    Moves that worsen the score are sometimes accepted according to a
    simple temperature schedule (Metropolis criterion).  This phase can
    still improve the solution after the gradient descent has converged.

All heavy work uses FFT‑based convolutions (``_conv_self``) and the
projection onto the simplex ``_project_simplex`` – both O(n log n) with
n≈4096.  The routine respects the time budget, prints progress information
and finally returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used internally: 2·n·max(self‑conv(seq,seq))   (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet seeding
# ----------------------------------------------------------------------
def _random_dirichlet_trials(
    n: int, best_seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # many cheap draws – the loop stops when the deadline is reached
    while time.time() < deadline:
        # log‑uniform concentration between 0.05 and 10
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _gradient_descent_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the non‑smooth objective.
    Uses an adaptive step size (increase on improvement, decrease on
    rejection) and occasional random jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    step = 0.12            # initial learning rate
    min_step = 0.001
    max_step = 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # take a step opposite to the sub‑gradient
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        cur_score = _objective_score(a)
        new_score = _objective_score(a_new)

        # keep the globally best sequence
        if new_score < best_score:
            best_score, best_seq = new_score, a_new.copy()
            print(
                f"[{time.time() - start_time:.2f}s] GD it={it:,} score={new_score:.6f} step={step:.4f}"
            )

        # accept the move if it improves, otherwise shrink the step
        if new_score < cur_score:
            a = a_new
            step = min(step * 1.07, max_step)   # gently increase
        else:
            step = max(step * 0.73, min_step)   # shrink when not improving

        # occasional random jitter to shake out of plateaus
        if it % 1000 == 0:
            if rng.random() < 0.25:
                jitter = rng.gamma(0.2, 1.0, size=n)
                jitter /= jitter.sum()
                a = 0.9 * a + 0.1 * jitter
                a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Aggressive hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb_enhanced(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that moves a (time‑dependent) fraction of mass
    from the index with the largest positive sub‑gradient to that with the
    most negative one.  A simple linear cooling schedule provides occasional
    acceptance of worsening moves.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # parameters for the mass‑transfer fraction
    max_frac = 0.10          # start with up‑to‑10 % of the source mass
    min_frac = 0.001         # final minimum
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    # linear decay of the allowed fraction
    def current_frac():
        elapsed = time.time() - start_time
        frac = max_frac * (1.0 - elapsed / (deadline - start_time))
        return max(frac, min_frac)

    while time.time() < deadline:
        # compute the sub‑gradient of the (non‑smooth) max‑convolution
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source = index with largest *positive* gradient that actually holds mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]

        # destination = index with most negative gradient
        dst = int(np.argmin(grad))

        if src == dst:
            # fallback: pick two random distinct indices
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer a tiny amount of mass
        frac = current_frac()
        eps = a[src] * (rng.random() * frac + 1e-6)   # ensure a non‑zero move
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # re‑project to keep the simplex constraint (numerical safety)
        a_prop = _project_simplex(a_prop)

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score

        # temperature that decays to almost zero at the deadline
        T0 = 0.02
        decay = (1e-6 / T0) ** (1.0 / total_hc_time)
        T = T0 * (decay ** (time.time() - start_time))

        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Hill‑climb improved score={best_score:.6f}"
                )
    return best_seq, best_score
