"""Search for a step function with low evaluation score.

The algorithm proceeds in three stages:
1.  On a moderate length (8192) the coefficients are modelled as an
    exponentiated cosine series a[i] = exp( Σ_k c_k cos(2π k i / n) )
    and the parameters c_k are optimised with Adam on the simplex
    using the original sub‑gradient of the max‑convolution objective.
2.  The resulting shape is interpolated to a larger length (32768)
    and refined further with the fast exponentiated‑gradient descent
    of the original implementation.
3.  A short simulated‑annealing polishing phase is applied.
All phases respect the supplied time budget.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the probability simplex."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set: values within relative tolerance
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)

        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Light simulated‑annealing polish."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

def _cosine_basis(n: int, K: int) -> np.ndarray:
    """Return a (K, n) array of cosine basis vectors."""
    i = np.arange(n, dtype=np.float64)
    B = np.empty((K, n), dtype=np.float64)
    for k in range(K):
        B[k] = np.cos(2.0 * np.pi * k * i / n)
    return B

def _adam_cosine_optimize(n: int,
                          K: int,
                          deadline: float,
                          start_time: float,
                          eta: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Optimise a non‑negative vector of length ``n`` that is constrained to
    lie on the simplex.  The vector is parametrised as

        a_i = exp( Σ_{k=0}^{K-1} c_k cos(2π k i / n) )  ,   Σ a_i = 1.

    Adam is used on the low‑dimensional coefficient vector ``c``.
    The gradient of the original max‑convolution objective
    (sub‑gradient based on the active set) is back‑propagated to ``c``.
    """
    rng = np.random.default_rng()
    # initialise coefficients close to the uniform distribution
    c = rng.normal(scale=0.05, size=K)

    B = _cosine_basis(n, K)            # shape (K, n)
    m = np.zeros(K)
    v = np.zeros(K)
    it = 0
    best_score = np.inf
    best_a = None

    while time.time() < deadline:
        it += 1

        # construct a from the current coefficients
        log_a = np.dot(c, B)            # shape (n,)
        # numerical stability: shift before exponentiation
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  score={cur_score:.6f}")

        # compute sub‑gradient w.r.t. a
        conv = _conv_self(a)
        max_val = conv.max()
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        grad_a = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= float(len(ks))

        # back‑propagation to the coefficients
        mu = np.dot(B, a)                     # (K,)
        s = np.dot(grad_a * a, np.ones(n))    # scalar = Σ_i grad_a_i a_i
        tmp = np.dot(grad_a * a, B.T)         # (K,)
        grad_c = tmp - mu * s

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_c
        v = beta2 * v + (1.0 - beta2) * (grad_c ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        c -= eta * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score

def _upscale(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a larger size."""
    if a_small.size == n_large:
        return a_small.copy()
    x_small = np.linspace(0.0, 1.0, a_small.size)

    total = a_large.sum()
    if total > 0.0:
        a_large /= total
    else:
        a_large = np.full(n_large, 1.0 / n_large)
    return a_large

"""
Search for a step function with a low evaluation score.

The algorithm works in four phases:

1. **Cosine‑Adam optimisation (n=8192).**  
   The coefficients are modelled as  
   a[i] = exp( Σ_{k=0}^{K‑1} c_k·cos(2π k i / n) ) ,  Σ a_i = 1.  
   The low‑dimensional parameter vector `c` is optimised with Adam.
   The gradient of the original max‑convolution objective (sub‑gradient
   based on the active set) is back‑propagated to `c`.

2. **Upscaling.**  
   The resulting shape is interpolated to a larger resolution
   (n=32768).  If a global starter `height_sequence_1` is available it is
   blended in (5 % weight).

3. **Exponentiated‑gradient refinement** on the large vector, exactly as
   in the reference implementation.

4. **Simulated‑annealing polish** for the remaining time budget.

All phases respect the supplied time budget, print progress information
and finally return a list of non‑negative floats that sum to 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (phase 3)
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Multiplicative (EG) descent on the probability simplex."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (phase 4)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly move tiny packets of mass between positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

# Cosine basis utilities (phase 1)

def _cosine_basis(n: int, K: int) -> np.ndarray:
    """Return a (K, n) array of cosine basis vectors."""
    i = np.arange(n, dtype=np.float64)
    B = np.empty((K, n), dtype=np.float64)
    for k in range(K):
        B[k] = np.cos(2.0 * np.pi * k * i / n)
    return B


def _adam_cosine_optimize(n: int,
                          K: int,
                          deadline: float,
                          start_time: float,
                          eta: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:

    Optimise a vector of length ``n`` that lies on the simplex.
    The vector is parametrised as

        a_i = exp( Σ_{k=0}^{K‑1} c_k·cos(2π k i / n) ),   Σ a_i = 1.

    from the active set) is back‑propagated to ``c``.
    """

    B = _cosine_basis(n, K)                    # (K, n)
    m = np.zeros(K)
    v = np.zeros(K)
    it = 0
    best_score = np.inf
    best_a = None

    while time.time() < deadline:
        it += 1

        # construct a from the current coefficients
        log_a = np.dot(c, B)                    # (n,)
        log_a -= np.max(log_a)                  # for numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  score={cur_score:.6f}")
