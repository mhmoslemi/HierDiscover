def _make_extended_shape(n, r, p, q):
    i = np.arange(n, dtype=np.float64)
    log_r = np.log(r) if r>0 else -np.inf # but r>0
    log_shape = i * log_r + p * np.log(i + 1.0) + q * np.log(n - i)
    # shift max for numeric stability
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in four phases:

1️⃣  **Coarse parametric sweep** – a three‑parameter family
    a_i ∝ r**i · (i+1)^p · (n‑i)^q  (geometric·power·reverse‑power) is
    mixed with the uniform distribution.  The best parameters are kept.

2️⃣  **Fine stochastic hill‑climb** on the four parameters (r, p, q, λ)
    with exponentially decaying perturbations (simulated‑annealing flavour).

3️⃣  **Projected sub‑gradient descent** on the exact objective.
    At each iteration the shift attaining the maximal convolution entry
    supplies the sub‑gradient; a diminishing step size and occasional
    uphill moves avoid premature stagnation.

4️⃣  **Tiny random mass‑transfer hill‑climb** for a last polish.

All phases respect the 1000 s global budget.  The final vector is projected
onto the probability simplex (sum = 1, non‑negative) and returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power × reverse‑power) and mixture
# ----------------------------------------------------------------------


def _make_extended_shape(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Returns a normalised vector of length n with components proportional to

        r**i * (i+1)**p * (n-i)**q    (i = 0 … n‑1).

    Computation is performed in log‑space for stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space: avoid overflow/underflow
    log_r = np.log(r) if r > 0 else -np.inf
    log_shape = i * log_r + p * np.log(i + 1.0) + q * np.log(n - i)
    # shift to keep exponentials moderate
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        # fallback to uniform – should rarely happen
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns lam·uniform + (1‑lam)·shape (both already sum to 1).
    The result automatically sums to 1.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()        # guard against tiny numerical drift
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true objective.
    The sub‑gradient corresponds to a single maximal shift of the convolution.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.18                     # slightly larger initial step
    it = 0

    while time.time() < deadline:
        it += 1

        # locate the (first) maximal convolution entry with an FFT
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # --- sub‑gradient for shift k (O(n)) ---------------------------------
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # for i≠j also add the symmetric contribution a_i to ∂/∂a_j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # --- step with diminishing size ------------------------------------
        step = step0 / (it ** 0.55)          # a bit slower decay for stability
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional probabilistic uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 10.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations to escape plateaux
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine consists of a coarse parametric sweep, a fine
    stochastic hill‑climb on the shape parameters, a projected sub‑gradient
    refinement, and a final tiny random mass‑transfer hill‑climb.
    It respects the 1000 s time budget and returns the best sequence as a list
    of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off between quality
    # and runtime.  8192 proved to be a sweet spot on the reference hardware.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a globally supplied starter exists, use it (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep over (r, p, q, λ)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 180.0   # ~3 min for grid search
    r_grid = np.logspace(np.log10(0.93), np.log10(0.9999), num=12)
    p_grid = np.linspace(-5.0, 5.0, 11)
    q_grid = np.linspace(-5.0, 5.0, 11)
    lam_grid = np.linspace(0.0, 0.5, 6)   # mixing >0.5 rarely helps

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            for q in q_grid:
                shape = _make_extended_shape(n, r, p, q)
                for lam in lam_grid:
                    cand = _mix_with_uniform(shape, lam)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        cur_params = (r, p, q, lam)
                        print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} "
                              f"p={p:.2f} q={q:.2f} λ={lam:.3f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fine stochastic hill‑climb on the four parameters (r, p, q, λ)
    # ------------------------------------------------------------------
    try:
        cur_r, cur_p, cur_q, cur_lam = cur_params
    except NameError:
        cur_r, cur_p, cur_q, cur_lam = 0.97, 0.0, 0.0, 0.3

    param_deadline = deadline - 250.0   # reserve ~250 s for sub‑gradient
    sigma_r = 0.02
    sigma_p = 0.5
    sigma_q = 0.5
    sigma_lam = 0.1
    cur_seq = best_seq.copy()
    cur_score = best_score
    it_param = 0

    while time.time() < param_deadline:
        it_param += 1

        # Propose perturbations (log‑normal for r, additive for others)
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        # Clamp to admissible ranges
        r_new = float(np.clip(r_new, 0.85, 0.999999))
        p_new = float(p_new)
        q_new = float(q_new)
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_extended_shape(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        # Accept only improving moves (occasionally accept a worsening one)
        if sc_new < cur_score:
            cur_seq = cand_new
            cur_score = sc_new
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
        else:
            if rng.random() < 0.001:      # tiny chance to jump out of a plateau
                cur_seq = cand_new
                cur_score = sc_new
                cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new

        if sc_new < best_score:
            best_score = sc_new
            best_seq = cand_new.copy()
            print(f"[{time.time() - start_time:.2f}s] param‑hill it={it_param} "
                  f"r={r_new:.6f} p={p_new:.3f} q={q_new:.3f} λ={lam_new:.3f} "
                  f"score={sc_new:.6f}")

        # Diminish perturbation amplitudes
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 55.0:        # leave a few seconds for final tweak
        sub_deadline = deadline - 55.0
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        L_fft = _next_pow2(2 * n - 1)
        best_a = a.copy()
        best_sc = _eval_fast(a, L_fft)

        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # transfer up to 0.2 % of the source mass
            eps = a[src] * rng.random() * 2e-3
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            a_prop = _project_simplex(a_prop)
            new_sc = _eval_fast(a_prop, L_fft)
            if new_sc < best_sc:
                a = a_prop
                best_sc = new_sc
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={best_sc:.6f}")

        best_seq = best_a
        best_score = best_sc

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
