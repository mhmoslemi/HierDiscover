"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Coarse LP optimisation (tiny dimension)**
   • Work with a modest size `n_small` (≈1024) to keep linear
     programming cheap.
   • Initialise the vector either uniformly or from a very sparse
     Dirichlet draw.
   • For a few iterations:
       – Compute `M = max(conv(f,f))`.
       – Build the convolution matrix `A` (size (2n‑1)×n) such that
         `A @ g = conv(f,g)`.
       – Solve the LP  
           maximise   1ᵀ g
           subject to A·g ≤ M·1 ,   g ≥ 0 .
       – Normalise `g` to sum = 1 and mix with the current `f`:
         `f ← (1‑t)·f + t·g`  (t≈0.2).
       – Keep the best small‑scale vector found.

2. **Upsample**
   • Interpolate the best small vector to a large size `N`
     (16 k – 65 k depending on the remaining time) and renormalise.

3. **Smooth‑max Adam refinement**
   • Optimise the true objective `2·N·max(conv(a,a))` using a
     temperature‑annealed soft‑max (the same routine that the previous
     submission used) for the bulk of the remaining budget.

4. **Sub‑gradient descent**
   • Treat `max(conv(a,a))` as a non‑smooth function.
   • At each step compute the shift `k` attaining the maximum,
     obtain the sub‑gradient `g_j = 2·a_{k‑j}` (when the index is in range),
     take a diminishing step and project back onto the probability simplex.
   • Keep the best vector seen.

5. **Simulated‑annealing polish**
   • Short mass‑swap SA biased toward lowering the objective.

All phases respect the global 1000 s budget and the final sequence is
scaled to sum = 1 (scaling does not affect the evaluated score).  The
procedure is deterministic for a given seed but explores several
initialisations.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear convolution a*a (real FFT, length 2·len(a)-1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Objective for a probability vector (sum==1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate of the true objective
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    T_start = 0.30
    T_end = 1e-6
    step0 = 0.15

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # temperature schedule (exponential)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max = 2·(w ⊛ a_rev)   (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 10**6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        idx = np.arange(n)
        src = k - idx
        mask = (src >= 0) & (src < n)
        grad[mask] = 2.0 * a[src[mask]]

        # diminishing step; 0.2 is a safe start
        step = 0.20 / (1.0 + math.sqrt(it))
        a -= step * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose final discretisation size based on total budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()

    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # Phase 1 : tiny‑scale LP refinement (n_small ≈ 1024)
    # --------------------------------------------------------------
    n_small = 1024 if (deadline - time.time()) > 200 else 512
    best_small = None
    best_small_score = float('inf')

    # two different starts – uniform and a very sparse Dirichlet
    init_vectors = []
    init_vectors.append(np.full(n_small, 1.0 / n_small))
    # sparse Dirichlet (concentration << 1 gives spiky vectors)
    a_dir = rng.gamma(0.02, 1.0, size=n_small)
    if a_dir.sum() == 0:
        a_dir = np.full(n_small, 1.0 / n_small)
    else:
        a_dir /= a_dir.sum()
    init_vectors.append(a_dir)

    for start_idx, f in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        cur_best_score = _objective(f)
        # LP iteration loop
        for it in range(30):
            if time.time() >= deadline:
                break
            conv_f = _conv_self(f)
            M = conv_f.max()

            # Build convolution matrix A  (A @ g = conv(f,g))
            n = n_small
            A = np.zeros((2 * n - 1, n), dtype=float)
            for j in range(n):
                A[j:j + n, j] = f

            # Linear programme: maximise sum(g)  <=> minimise -sum(g)
            c = -np.ones(n)
            b = M * np.ones(2 * n - 1)

            res = linprog(c,
                          A_ub=A,
                          b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})
            if not res.success:
                # if the LP fails, quit this start‑vector
                break

            g0 = res.x
            s = g0.sum()
            if s <= 0:
                g = np.full(n, 1.0 / n)
            else:
                g = g0 / s

            # convex combination with a modest step size
            t = 0.2
            f = (1 - t) * f + t * g
            f = _project_simplex(f)

            cur_score = _objective(f)
            if cur_score < cur_best_score:
                cur_best_score = cur_score

        print(f"[{time.time() - start_time:.2f}s] Small‑scale start {start_idx} best score {cur_best_score:.6f}")

        if cur_best_score < best_small_score:
            best_small_score = cur_best_score
            best_small = f.copy()

    # If, for any reason, the LP phase produced nothing useful, fall back
    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # --------------------------------------------------------------
    # Phase 2 : up‑sample to the target size N
    # --------------------------------------------------------------
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_large = np.linspace(0, 1, N, endpoint=False)
    a = np.interp(x_large, x_small, best_small)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full(N, 1.0 / N)

    # --------------------------------------------------------------
    # Phase 3 : Adam smooth‑max optimisation (majority of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20:
        sub_deadline = time.time() + remaining * 0.70

    # Phase 4 : sub‑gradient descent (helps to push the max down a bit)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10:

        a, _ = _subgradient_descent(a, sub_deadline, start_time)

    # Phase 5 : short simulated‑annealing polish (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5:

    # Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    a = np.maximum(a, 0.0)
    s = a.sum()

        a /= s
    else:
        a = np.full(N, 1.0 / N)

    final_score = _objective(a)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in a]
