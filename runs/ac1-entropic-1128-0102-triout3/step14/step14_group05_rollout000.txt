def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.2

    np.random.seed(seed)
    n = 8192  # maybe adjust based on time, but choose 8192.
    L_fft = _next_pow2(2*n - 1)

    # baseline uniform
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # incorporate global starter if exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            scorer = _eval_fast(starter, L_fft)
            if scorer < best_score:
                best_seq = starter.copy()
                best_score = scorer
                print(f"[{time.time() - start:.2f}s] global starter score={scorer:.6f}")
    except Exception:
        pass

    # Stage 1: parametric sweep
    # Use geometric families (similar to before) with more refined grid
    idx = np.arange(1, n+1, dtype=float) # 1-index
    r_grid = np.logspace(np.log10(0.85), np.log10(0.9999), num=40)
    p_grid = np.linspace(-3.5, 2.5, num=53) # more values
    for r in r_grid:
        if time.time() > deadline - 200:
            break
        geo = np.power(r, np.arange(n, dtype=float))
        for p in p_grid:
            if time.time() > deadline - 200:
                break
            shape = geo * np.power(idx, p)
            total = shape.sum()
            if total == 0:
                continue
            shape /= total
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_seq = cand.copy()
                    best_score = sc
                    print(f"[{time.time() - start:.2f}s] sweep r={r:.5f} p={p:.3f} λ={lam} score={sc:.6f}")

    # Stage 2: L-BFGS on surrogate with extensive iteration
    if _SCIPY_AVAILABLE and time.time() < deadline - 120:
        # start from log of best_seq
        eps = 1e-12

The algorithm works in four phases:

0️⃣  Initialise with a uniform distribution and, if available, a previously

    annealed from 0.5 up to 1000 and the number of function evaluations is

3️⃣  Perform a long exponentiated‑gradient descent on the exact objective.
    The sub‑gradient of ``max(conv(a,a))`` is ``2·a[t_max‑i]``; a multiplicative
    update in the simplex (a ← a·exp(‑η·g), renormalise) naturally respects
    non‑negativity and the sum‑to‑one constraint.  The step size η decays as
    1/√iteration and the vector is symmetrised after each update.

4️⃣  A cheap random hill‑climb (tiny mass transfers) finishes the optimisation.

All phases respect a 1000 s wall‑clock budget; the best sequence encountered
is returned as a plain ``list[float]`` (non‑negative, normalised to unit sum).

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT.  Returns length 2·len(a)‑1."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Surrogate (smooth‑max) used by L‑BFGS
# ----------------------------------------------------------------------
class _SurrogateMax:
    """
    Callable implementing the log‑sum‑exp surrogate of ``max(conv)`` together
    with its gradient w.r.t. the log‑probability parameters ``z``.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv      # for a*a
        self.L_corr = L_corr      # for w * a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = _conv_self_fft(a, self.L_conv)   # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, length 2n‑1
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient w.r.t a: g_i = 2·∑_k w_k·a_{k‑i}
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]   # length n

        # chain rule through soft‑max (∂a_i/∂z_j = a_i·(δ_ij – a_j))
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        return surrogate, grad_z

# ----------------------------------------------------------------------
def _final_hill_climb(a: np.ndarray, deadline: float,

    objective.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:

        eps = best_a[src] * rng.random() * 1e-2

        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
