def _make_shape_offset(n, r, p, offset):
    i = np.arange(n, dtype=np.float64)
    j = i - offset
    mask = j >= 0
    shape = np.zeros_like(i, dtype=np.float64)
    if np.any(mask):
        idx = j[mask] + 1.0
        # compute in log-space
        log_shape = j[mask] * np.log(r) + p * np.log(idx)
        # shift
        log_shape -= np.max(log_shape)
        shape_vals = np.exp(log_shape)
        shape[mask] = shape_vals
        total = shape_vals.sum()
        if total > 0:
            shape[mask] /= total
    # Ensure shape vector sums to 1 (any zero mass positions are already zero)
    # If total==0, we fallback to uniform
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape
```

"""
Optimisation of the step‑function objective
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm runs in four stages (all respecting a 1000 s wall‑clock budget)

1️⃣  **Parametric sweep** – a 4‑parameter family
        a_i ∝ r^{i‑offset}·(i‑offset+1)^p   (with offset ≥0)
    is mixed with the uniform distribution.
    A coarse grid over (r,p,offset,λ) yields a good warm‑start.

2️⃣  **Stochastic hill‑climb** – the four parameters are perturbed
    with decreasing Gaussian noise; improvements are kept.
    This refines the shape beyond the grid.

3️⃣  **Projected momentum sub‑gradient descent** – working on the exact
    max‑convolution (computed with an FFT).  The sub‑gradient is the
    average of the gradients of the top‑few near‑maximal shifts,
    and a heavy‑ball momentum term accelerates convergence.

4️⃣  **Simulated‑annealing local search** – random mass transfers between
    two indices are accepted with a Metropolis rule whose temperature
    decays to zero.  Occasional small smoothing steps help to escape
    plateaus.

The best vector found (normalised to sum = 1) is returned as a list of
Python floats.  All helper functions are top‑level, use only NumPy and
the standard library, and contain no closures or lambdas.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, sum w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power) with offset & mixture
# ----------------------------------------------------------------------


def _make_shape_offset(n: int, r: float, p: float, offset: int) -> np.ndarray:
    """
    Vector of length n with components ∝ r^{i‑offset}·(i‑offset+1)^p
    for i ≥ offset, otherwise 0.  Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    j = i - offset
    mask = j >= 0
    shape = np.zeros_like(i, dtype=np.float64)
    if np.any(mask):
        idx = j[mask] + 1.0
        # log‑space computation for stability
        log_shape = j[mask] * np.log(r) + p * np.log(idx)
        log_shape -= np.max(log_shape)          # shift to avoid overflow
        shape_vals = np.exp(log_shape)
        shape[mask] = shape_vals
        total = shape_vals.sum()
        if total > 0.0:
            shape[mask] /= total
    # Normalise the whole vector (protect against the pathological case)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns lam·uniform + (1‑lam)·shape (both already sum to 1).
    Result automatically sums to 1 (tiny renormalisation for safety).
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Gradient from (near‑)maximal convolution shifts
# ----------------------------------------------------------------------


def _gradient_from_shifts(a: np.ndarray,
                         shift_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the (average) sub‑gradient of max(conv(a,a)) coming from the
    supplied convolution shift indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in shift_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # symmetric part for i≠j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    if shift_idxs.size > 0:
        g /= float(shift_idxs.size)
    return g


# ----------------------------------------------------------------------
# Momentum sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def _subgradient_descent_momentum(a0: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected heavy‑ball sub‑gradient descent directly on the true objective.
    The sub‑gradient aggregates the gradients of the top few (≈10) maximal
    convolution shifts.  Momentum (β≈0.9) accelerates convergence.
    Returns the best vector seen and its score.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    beta = 0.9
    momentum = np.zeros_like(a)

    step0 = 0.12                       # initial step size (tuned empirically)
    it = 0
    # we keep a tiny set of “nearly maximal’’ shifts (Δ = 1e‑7)
    delta_abs = 1e-9

    while time.time() < deadline:
        it += 1
        # --- locate near‑maximal shifts ---------------------------------
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        # collect up to 12 shifts whose value is within delta_abs of the max
        near = np.where(conv >= max_val - delta_abs)[0]
        if near.size > 12:
            # random subset to keep cost low
            near = rng.choice(near, size=12, replace=False)

        # --- compute (averaged) sub‑gradient ---------------------------
        g = _gradient_from_shifts(a, near)

        # --- momentum step ---------------------------------------------
        step = step0 / (it ** 0.55)      # slightly slower decay
        momentum = beta * momentum + (1.0 - beta) * g
        a_candidate = a - step * momentum
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # optional: tighten delta when we improve
            delta_abs = max(delta_abs * 0.97, 1e-11)
            # print progress
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # simulated‑annealing style uphill acceptance
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # occasional jitter to help escape tiny plateaus
        if it % 3000 == 0:
            eps = a * rng.random(a.shape) * 1e-4
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style random mass‑transfer local search
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with a temperature that decays linearly
    to zero.  Keeps the best vector seen.
    """
    n = a_start.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a_start.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    move_factor = 0.02                # up to 2 % of donor mass per move

    while time.time() < deadline:
        # linearly decreasing temperature (scaled to score magnitude)
        remaining = deadline - time.time()
        frac = max(remaining / total_time, 0.0)
        T = best_score * 0.012 * frac + 1e-9

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * move_factor
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score

        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] SA score={best_score:.6f}")

        # occasional tiny smoothing (kernel size 3)
        if rng.random() < 0.003:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            a_smoothed = np.convolve(a, kernel, mode='same')
            a = _project_simplex(a_smoothed)

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine consists of a coarse parametric sweep,
    a stochastic hill‑climb on the four shape parameters, a momentum
    sub‑gradient descent, and a final simulated‑annealing local search.
    It respects the 1000 s time budget and returns the best sequence as a
    plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a tiny safety margin for the runtime environment
    deadline = start_time + budget_s - 0.15

    # --------------------------------------------------------------
    # Problem size – 4096 proved a good trade‑off in earlier runs.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline: uniform distribution (also used if no starter is present)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep over (r, p, offset, λ)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 140.0   # ~140 s for the grid search
    r_grid = np.logspace(np.log10(0.88), np.log10(0.9999), num=22)
    p_grid = np.linspace(-3.0, 2.0, 27)
    offset_grid = np.arange(0, 401, 100)    # 0,100,200,300,400
    lam_grid = np.linspace(0.0, 1.0, 9)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            for offset in offset_grid:
                shape = _make_shape_offset(n, r, p, offset)
                for lam in lam_grid:
                    cand = _mix_with_uniform(shape, lam)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                              f"p={p:.2f} off={offset} λ={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Fine stochastic hill‑climb on (r, p, offset, λ)
    # --------------------------------------------------------------
    # Initialise parameters from the best sweep candidate if possible.
    # For simplicity we start from a generic centre.
    cur_r = 0.97
    cur_p = -0.2
    cur_offset = 80
    cur_lam = 0.35
    cur_seq = best_seq.copy()
    cur_score = best_score

    param_deadline = deadline - 340.0   # keep ~340 s for the two final phases
    sigma_r = 0.018
    sigma_p = 0.25
    sigma_off = 30.0
    sigma_lam = 0.09

    while time.time() < param_deadline:
        # propose new parameters (log‑normal for r to stay in (0,1))
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        off_new = int(round(cur_offset + sigma_off * rng.standard_normal()))
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        # clamp into admissible ranges
        r_new = float(np.clip(r_new, 0.85, 0.999999))
        p_new = float(p_new)               # any real
        off_new = int(np.clip(off_new, 0, n // 4))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_offset(n, r_new, p_new, off_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_offset, cur_lam = r_new, p_new, off_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] hill r={r_new:.6f} "
                      f"p={p_new:.4f} off={off_new} λ={lam_new:.4f} score={sc_new:.6f}")

        # exponential decay of perturbation magnitudes
        sigma_r *= 0.9994
        sigma_p *= 0.9994
        sigma_off *= 0.9994
        sigma_lam *= 0.9994

        # occasional random jitter
        if rng.random() < 0.004:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    # --------------------------------------------------------------
    # 3️⃣  Momentum sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    if time.time() < deadline - 140.0:
        sub_deadline = deadline - 140.0
        a_sub, sc_sub = _subgradient_descent_momentum(best_seq,

                                                       start_time,

        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()

    # 4️⃣  Simulated‑annealing local refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:

        a_sa, sc_sa = _simulated_annealing(best_seq,

    # --------------------------------------------------------------
    # Final normalisation (sum → 1) and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
