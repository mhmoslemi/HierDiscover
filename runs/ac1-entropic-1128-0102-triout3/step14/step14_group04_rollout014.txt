"""
Propose a low‑score step‑function.

The score evaluated by the competition is

    2 * n * max(conv(a, a)) / (sum(a)**2)

which for a probability vector (sum(a)=1) reduces to 2·n·max(conv(a,a)).
The code below searches many diverse families of windows, random sparse
vectors (including asymmetric “mirror‑free” draws) and refines the most
promising candidates with a simple projected sub‑gradient method that
focuses on the *single* worst convolution entry.  A final short
simulated‑annealing phase moves tiny packets of mass between indices.
All operations respect the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalized vector (sum == 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (sum(seq)=1)"""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                      # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps with a flat plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0

        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian window, normalised to sum = 1."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_exponential(n: int, decay: float) -> np.ndarray:
    """Monotonically decreasing exponential, normalised to sum = 1."""
    i = np.arange(n)
    shape = np.exp(-decay * i)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_asymmetric_random(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Dirichlet draw with concentration `conc` and then enforce
    a_i * a_{n-1-i} = 0 for every mirrored pair (break symmetry).
    """
    vec = rng.dirichlet(np.full(n, conc))
    half = n // 2
    for i in range(half):
        j = n - 1 - i
        if vec[i] >= vec[j]:
            vec[j] = 0.0
        else:
            vec[i] = 0.0
    if n % 2 == 1:
        vec[half] = 0.0
    total = vec.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    vec /= total
    return vec


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (focus on the single worst entry)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    tol: float = 0.01,
    top_k: int = 1,
) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    Only the `top_k` largest convolution entries are used to build
    the gradient (default = 1 → the absolute worst entry).
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam‑style momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.6

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)  # denominator = 1 (sum==1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        # indices of the largest entries
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            ks = np.where(conv >= cur_max * (1.0 - tol))[0]

        # gradient of the (average) active convolutions
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # each pair (i, k-i) contributes twice except when i == k-i,
            # which is already handled by the factor 2 below
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Random mass moves with a simple Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = max(1e-5, 0.2 * (0.01 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)      # bias to larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 40 %) of src's mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1) Length – the merit factor is essentially independent of n,
    #    but n≈4096 gives a good trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2) Build a diverse pool of candidates
    # ------------------------------------------------------------------
    pool = []                # (score, vector)
    pool_cap = 25            # keep a modest elite set

    def _maybe_add(vec: np.ndarray):
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # classic symmetric windows
    _maybe_add(_make_uniform(n))
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))
    _maybe_add(_make_cosine_tip(n))

    # Kaiser family – explore a wider beta range
    for beta in np.linspace(0.0, 15.0, 31):
        _maybe_add(_make_kaiser(n, beta))

    # Power‑ramp family – finer grid
    for p in np.linspace(0.1, 5.0, 25):
        for alpha in np.linspace(0.0, 0.5, 11):
            _maybe_add(_make_power_shape(n, p, alpha))

    # Beta‑distribution family – finer grid over a,b
    a_grid = np.linspace(0.2, 10.0, 21)
    b_grid = np.linspace(0.2, 10.0, 21)
    for a_par in a_grid:
        for b_par in b_grid:
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    # Gaussian windows – sweep sigma as a fraction of n
    for frac in np.linspace(0.05, 0.5, 10):
        sigma = frac * n
        _maybe_add(_make_gaussian(n, sigma))

    # Simple exponential decays
    for decay in np.linspace(0.001, 0.02, 10):
        _maybe_add(_make_exponential(n, decay))

    # Random Dirichlet draws with several concentration values
    for conc in [0.01, 0.05, 0.1, 0.3, 0.7]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.dirichlet(np.full(n, conc))
            _maybe_add(vec)

    # Asymmetric (mirror‑free) random draws – break the a_i–a_{n‑1‑i} symmetry
    for conc in [0.05, 0.2, 0.5]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = _make_asymmetric_random(n, conc, rng)
            _maybe_add(vec)

    # ------------------------------------------------------------------
    # 3) Keep the current global starter if it is present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                starter = starter / total
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                if sc < pool[-1][0]:
                    _maybe_add(starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4) Pick the best from the pool and run focused refinements
    # ------------------------------------------------------------------
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # Allocate time for several short sub‑gradient runs starting from the
    # elite members of the pool.
    remaining = deadline - time.time()
    if remaining > 60.0:
        # run refinement on the top 5 pool members, each gets a slice of time
        elite = [vec for (_, vec) in pool[:5]]
        per_candidate = remaining / len(elite)
        for i, vec in enumerate(elite, 1):
            sub_deadline = time.time() + per_candidate * 0.85
            refined_seq, refined_score = _subgradient_descent(
                vec, sub_deadline, start_time, tol=0.005, top_k=1
            )
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Refined (elite {i}) new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
