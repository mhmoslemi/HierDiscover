function _subgradient_descent(a_initial, deadline, start_time, rng):

    a = a_initial.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_obj = _objective(a)

    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)  # compute conv full
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] SD iter={it:,} best={cur_obj:.6f}")

        # compute gradient of max
        # Use tolerance fraction to include near-max points
        # We'll include conv within alpha*(max_val) maybe relative tolerance 1e-12 absolute to avoid numeric issues.
        tol = 1e-12
        ks = np.where(conv >= max_val - tol)[0]  # indices of max

        # compute gradient g
        g = np.zeros_like(a)
        for k in ks:
            imin = max(0, k - n + 1)
            imax = min(n - 1, k)
            idx = np.arange(imin, imax + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # step schedule
        step = 0.6 / np.sqrt(it)   # step0 maybe 0.6
        # exponentiated gradient update:
        a = a * np.exp(-step * g)
        # re-normalize
        total = a.sum()
        if total == 0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # occasional random swap
        if it % 5000 == 0:
            # small random perturbation
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * 0.01 * rng.random()
                a[src] -= delta
                a[dst] += delta
                # ensure nonnegative
                a = np.maximum(a, 0)
                a /= a.sum()

    return best_a, best_obj

"""
Hybrid search for a low‑score step‑function.

1. Initialise a pool of diverse probability vectors (uniform, power‑ramps,
   symmetric Beta densities, Dirichlet draws and the previous best if it
   exists).  Keep the best K candidates.

2. Pick the current best candidate and run a long *sub‑gradient descent*
   on the true objective  f(a)=2·n·max(conv(a,a)).
   The gradient of the max‑convolution is obtained analytically;
   we use an exponentiated‑gradient (mirror‑descent) update and
   Euclidean projection onto the simplex.  The step size decays as
   η / sqrt(iter) with a large initial η so that sizeable moves are made
   early on.

3. When time is low a short simulated‑annealing polish (mass swaps) is
   performed.

All work respects the 1000 s budget, uses only NumPy and stays on the
probability simplex.  The returned list sums to one and contains only
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Real convolution a*a using FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau, symmetrised."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 200.0,
                     max_move_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on  f(a)=2·n·max(conv(a,a)).
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG iter={it:,} "
                  f"best={cur_score:.6f}")

        # -------- gradient of the max‑convolution (sub‑gradient) ----------
        # all indices attaining the maximum (numerical tolerance)
        ks = np.where(conv >= max_val - 1e-14)[0]

        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # -------- exponentiated‑gradient update -------------------------
        step = step0 / math.sqrt(it)
        # avoid overflow/under‑flow
        expo = np.exp(-step * g)
        a = a * expo
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # -------- occasional random mass move (escape) -----------------
        if it % 5000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,

    # symmetric Beta shapes (including U‑shaped)

    for conc in concs:

                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)

    # start from the previous best if available
    try:
        starter = globals().get('height_sequence_1')
        if starter is not None:

                starter_up = np.maximum(starter_up, 0.0)

    # 2) Sub‑gradient descent on the best candidate (majority of time)
    # --------------------------------------------------------------

        refined_seq, refined_score = _subgrad_descent(best_seq, sub_deadline,
                                                     start_time, rng,

    # --------------------------------------------------------------

    else:

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")
