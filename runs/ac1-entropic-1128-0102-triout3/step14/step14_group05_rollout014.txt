"""
Enhanced step‑function optimiser.

The routine proceeds through several cheap exploratory phases
(uniform baseline, simple parametric families, random Dirichlet
samples) before a long annealed L‑BFGS optimisation of a
log‑soft‑max parametrisation.  The L‑BFGS schedule uses a much
larger number of function evaluations than the reference
implementation, allowing the smooth‑max surrogate to settle
closer to the true objective.  After the smooth optimisation a
few random restarts are performed, followed by a final hill‑climb
that moves a larger fraction of mass between coordinates.
All phases respect the 1000 s wall‑clock budget; the best vector
found is returned as a plain list of floats (normalised to a unit
sum, non‑negative).
"""

import time
import numpy as np

# optional – may be missing in a minimal environment
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# FFT utilities (fast exact objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length = 2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact evaluation of the objective
        2·n·max(conv(a,a)) / (sum(a)²)
    with no security checks (assumes a ≥ 0, sum(a) > 0).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max utilities (log‑probability parametrisation)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:          # fallback – should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable returning (value, gradient) of a smooth approximation
    of max(conv(a,a)).  The optimisation variable is `z`,
    the log‑probability vector (a = softmax(z)).
    """

    def __init__(self, n: int, beta: float,
                 L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # for a*a
        self.L_corr = L_corr          # for w*a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥0, Σa = 1
        # ----- self‑convolution -----------------------------------------
        conv = _conv_self_fft(a, self.L_conv)   # length 2n‑1
        max_c = conv.max()
        # smooth‑max via log‑sum‑exp
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, length 2n‑1
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # ----- gradient w.r.t. a (2·∑_k w_k·a_{k-i}) --------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]          # length n

        # ----- chain rule through soft‑max -------------------------------
        # ∂a_i/∂z_j = a_i·(δ_{ij} – a_j)
        dot = np.dot(grad_a, a)               # scalar
        grad_z = a * (grad_a - dot)           # shape (n,)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # standard algorithm (see Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.maximum(v, 0.0) / np.maximum(v.sum(), 1e-12)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    # due to rounding errors, renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum


# ----------------------------------------------------------------------
# Final hill‑climb (larger moves than the reference implementation)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """
    Random mass‑transfer hill‑climb that accepts strictly
    improving moves.  Transfer size may be up to 5 % of the source
    mass, which allows the algorithm to escape shallow local minima.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 5 % of source mass
        eps = best_a[src] * rng.random() * 5e-2
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The procedure combines cheap sweeps, random Dirichlet samples,
    an annealed L‑BFGS optimisation of a smooth‑max surrogate, a
    few random restarts and a final hill‑climb.  The best vector
    discovered (normalised to unit sum) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # problem dimensions
    # ------------------------------------------------------------------
    n = 4096                                    # size used by the reference code
    L_fft = _next_pow2(2 * n - 1)               # length for exact evaluations
    L_corr = _next_pow2(3 * n - 2)              # length for gradient convolutions

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional starter from the global variable `height_sequence_1`
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric‑power families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 150:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling (exploratory)
    # ------------------------------------------------------------------
    dirichlet_alphas = [0.2, 0.5, 1.0, 5.0]
    samples_per_alpha = 300   # total ~1200 evaluations, cheap on the time budget
    rng = np.random.default_rng(seed + 9999)

    for alpha in dirichlet_alphas:
        for _ in range(samples_per_alpha):
            if time.time() >= deadline - 100:
                break
            x = rng.gamma(alpha, size=n)
            if x.sum() == 0.0:
                continue
            cand = x / x.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet α={alpha:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max L‑BFGS optimisation (extended schedule)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 250:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # a more aggressive temperature schedule
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

        # number of surrogate evaluations per temperature
        maxfun_per_beta = 2000

        for beta in beta_schedule:
            if time.time() >= deadline - 250:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun_per_beta,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # warm‑start for the next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 4️⃣  Random restarts (short L‑BFGS runs from perturbed starts)
    # ------------------------------------------------------------------
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline - 150:
            break
        rng = np.random.default_rng(seed + r + 12345)
        noisy = best_seq + rng.normal(scale=0.02, size=n)   # a bit noisier than before
        noisy = np.maximum(noisy, 0.0)
        noisy_sum = noisy.sum()
        if noisy_sum == 0.0:
            continue
        noisy /= noisy_sum
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 10.0
        obj = _SurrogateMax(n=n, beta=beta,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=_next_pow2(3 * n - 2))
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=2000,
                disp=False,
                factr=1e7,
            )
        except Exception:
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb polishing (exact objective, larger moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: guarantee feasibility and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
