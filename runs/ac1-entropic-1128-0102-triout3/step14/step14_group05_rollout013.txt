def proj_simplex(v):
    # Ensure nonneg sum to 1.
    # Algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns a real array of length 2*len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    # Numerical noise may produce tiny negative values – clamp to 0.
    return np.maximum(conv, 0.0)


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    `a` is assumed non‑negative.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
        A @ g == conv(f, g)        (linear convolution, f fixed)
    and therefore the linear constraints A @ g ≤ M implement
        (f * g)_k ≤ M    for all k.
    Only non‑negative offsets are needed because conv(f,g)[k] = Σ f_{k‑j} g_j,
    and f_{k‑j} is defined only when k‑j ≥ 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    offsets = []
    for d in range(n):                     # offset = row - col = d ≥ 0
        length = n - d                     # number of entries on this diagonal
        diag = np.full(length, f[d], dtype=np.float64)
        data.append(diag)
        offsets.append(d)
    A = sp.diags(data, offsets, shape=(rows, n), format="csr")
    return A


def _solve_lp_max_sum_g(f: np.ndarray, M: float):
    """
    Solve the linear program:
        maximise   Σ g_i
        subject to   g ≥ 0,
                     conv(f, g) ≤ M   (componentwise)
    Returns the optimal vector g (not normalised).
    """
    if not _SCIPY_AVAILABLE:
        raise RuntimeError("SciPy is required for the LP step.")
    n = f.shape[0]
    A = _build_conv_ub_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # linprog minimises → maximise sum(g)
    bounds = [(0, None)] * n
    # HiGHS (the default) handles sparse A_ub efficiently.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method="highs", options={"presolve": True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine combines a linear‑program steering step with a projected
    sub‑gradient polishing phase and respects the overall time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (the original challenge used n = 4096)
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # for exact convolution evaluations

    # --------------------------------------------------------------
    # 0️⃣  Initialise – use the global starter if it exists,
    #                  otherwise a uniform distribution.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0.0:
            raise ValueError
        if starter.shape[0] != n:
            # pad/truncate to the required length
            if starter.shape[0] < n:
                starter = np.concatenate([starter,
                                          np.zeros(n - starter.shape[0])])
            else:
                starter = starter[:n]
        a = starter / starter.sum()
    except Exception:  # pragma: no cover
        a = np.full(n, 1.0 / n)

    best_seq = a.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  LP‑driven peak reduction loop
    # --------------------------------------------------------------
    t_vals = [0.05, 0.10, 0.20, 0.35, 0.50]   # mixing parameters to try
    lp_iter = 0
    while time.time() < deadline - 5.0 and _SCIPY_AVAILABLE:
        # current peak M
        conv_a = _conv_self_fft(a, L_fft)
        M = conv_a.max()
        # solve LP for a direction g
        try:
            g = _solve_lp_max_sum_g(a, M)
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP failed ({e}); breaking LP loop.")
            break
        if g.sum() == 0.0:
            # no slack – cannot improve further with this step
            break
        g_norm = g / g.sum()               # keep unit sum for mixing
        improved = False
        # try several mixing strengths and keep the best improvement
        for t in t_vals:
            cand = (1.0 - t) * a + t * g_norm
            # re‑normalise (tiny numerical drift)
            cand = cand / cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score - 1e-9:
                best_score = sc
                best_seq = cand.copy()
                a = cand.copy()
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter}, t={t:.2f} → score {sc:.6f}")
                break          # accept first improving t (larger t tends to be better)
        lp_iter += 1
        if not improved:
            # no improvement for any t – stop the LP phase
            break

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient polishing (exact objective)
    # --------------------------------------------------------------
    step = 0.5           # initial step length
    it = 0
    while time.time() < deadline - 0.2:
        conv_a = _conv_self_fft(a, L_fft)
        max_val = conv_a.max()
        k = int(conv_a.argmax())          # index where the peak occurs

        # sub‑gradient of the max‑convolution at a:
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        # take a step opposite to the sub‑gradient and project onto simplex
        a_new = a - step * g
        a_new = _project_simplex(a_new)

        sc = _eval_fast(a_new, L_fft)
        if sc < best_score - 1e-9:
            best_score = sc
            best_seq = a_new.copy()
            a = a_new
            step = min(step * 1.2, 1.0)   # increase step when we make progress
            print(f"[{time.time() - start_time:.2f}s] subgrad iter {it} → score {sc:.6f}")
        else:
            # no improvement – shrink step
            step = max(step * 0.6, 1e-6)

        it += 1

    # --------------------------------------------------------------
    # Post‑process: ensure non‑negative, unit sum and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
