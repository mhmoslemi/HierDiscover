"""
Propose a non‑negative step function (vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm focuses on a long projected sub‑gradient descent on the true
objective (the maximal entry of the self‑convolution).  The bulk of the
1000 s budget is spent on this descent; a tiny final random‑mass‑transfer
hill‑climb polishes the result.  If a global starter vector
`height_sequence_1` is available it is used as the initial point, otherwise
the uniform distribution is taken.  The code is completely self‑contained
and only relies on NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent (long run)
# ----------------------------------------------------------------------


def _subgradient_descent_long(init_a: np.ndarray,
                              n: int,
                              L_fft: int,
                              deadline: float,
                              start_time: float,
                              best_initial_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    A simple clipping‑and‑renormalising projection is used (much cheaper than
    sorting‑based Euclidean projection) because the sum is kept constant.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_score = best_initial_score
    it = 0
    step0 = 0.45          # a generous initial step size
    rng = np.random.default_rng()
    # exponent < 0.5 yields a slower step decay → more exploration
    decay_exp = 0.45

    while time.time() < deadline:
        it += 1

        # --- locate maximal convolution entry (via FFT) -----------------
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))

        # --- sub‑gradient for shift k (O(n)) ---------------------------
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # --- step ------------------------------------------------------
        step = step0 / (it ** decay_exp)
        a_candidate = a - step * g

        # cheap projection: clip negatives and renormalise to sum 1
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0.0:
            a_candidate = a_candidate / s
        else:
            a_candidate = np.full_like(a_candidate, 1.0 / n)

        cur_score = _eval_fast(a_candidate, L_fft)

        # --- accept / keep‑the‑best ------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # --- small jitter every few thousand iterations ---------------
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny random mass‑transfer hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of mass from one index to another.
    Occasionally applies a small smoothing move.
    """
    rng = np.random.default_rng()
    best_a = a.copy()
    best_sc = best_score

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

        # occasional gentle smoothing (averaging neighbours)
        if rng.random() < 0.008:
            a_smooth = (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0
            a_smooth = a_smooth / a_smooth.sum()
            sc_smooth = _eval_fast(a_smooth, L_fft)
            if sc_smooth < best_sc:
                best_sc = sc_smooth
                best_a = a_smooth.copy()
                a = a_smooth
                print(f"[{time.time() - start_time:.2f}s] smoothing step score={sc_smooth:.6f}")
