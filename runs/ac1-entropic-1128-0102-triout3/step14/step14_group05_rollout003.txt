The function `propose_candidate` builds a step‑function that minimises  

\[
\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

The search consists of four independent phases that together respect the
1000 s wall‑clock limit.

1. **Uniform baseline** – the trivial solution (score ≈ 2) is stored.
2. **Deterministic sweep of Beta‑shapes** –  
   \(a_i \propto (i+1)^p\,(n-i)^q\) with a small grid on `p` and `q`.  
   The best shape (mixed with the uniform vector) is kept.
3. **Sub‑gradient descent on the exact `max`** –  
   *parameterisation*:   \(a_i = \mathrm{softmax}(z)_i\) (guaranteed
   non‑negativity and unit sum).  
   In each iteration the exact self‑convolution is computed by FFT, the
   active shifts (those attaining the maximum) are identified and an
   averaged sub‑gradient of the max‑operator is built analytically
   (no extra FFTs).  The update is performed in the `z`‑space with a
   simple decreasing step‑size, and the best vector encountered is
   recorded.
4. **Local hill‑climb polishing** – random mass transfers (up to 1 % of the
   source entry) are accepted only if they improve the *exact* objective.
   This phase finishes the search.

All helper routines are defined at top level, no closures or I/O are used,
and the algorithm automatically stops before the 1000 s deadline, returning
the best sequence it has found.

```python
"""Optimise a non‑negative step function for the given
evaluation function.

The search proceeds through:
  1️⃣ uniform baseline,
  2️⃣ a deterministic sweep of Beta‑shaped families,
  3️⃣ fast sub‑gradient descent on the exact max‑convolution
     (soft‑max parametrisation, analytic sub‑gradient, FFT for the
     convolution only),
  4️⃣ a final random hill‑climb polishing phase.

The implementation respects a 1000 s wall‑clock budget and returns the
best sequence (as a plain list of floats)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)
    with a already non‑negative and (approximately) normalised.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                      # should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Hill‑climb (exact objective) – cheap local mass transfers
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """
    Very local random mass transfers accepted only if they improve the
    exact objective. Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    # maximal transfer size (relative to the source entry)
    max_fraction = 0.01          # up to 1 % moves

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_a[src] * rng.random() * max_fraction
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine comprises a uniform baseline, a deterministic sweep of
    Beta‑shaped families, a fast sub‑gradient descent on the *exact*
    max‑convolution, and a final hill‑climb polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for final housekeeping
    deadline = start_time + budget_s - 0.15

    # --------------------------------------------------------------
    # problem size – a power‑of‑two is convenient for the FFTs
    # --------------------------------------------------------------
    n = 8192                     # double the previous size – still cheap
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣ uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ deterministic sweep of Beta‑shaped families
    #    a_i ∝ (i+1)^p * (n-i)^q
    # --------------------------------------------------------------
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    rev = n - np.arange(0, n, dtype=np.float64)       # n … 1

    p_grid = np.linspace(-2.5, 2.5, 21)
    q_grid = np.linspace(-2.5, 2.5, 21)

    for p in p_grid:
        if time.time() >= deadline - 100:
            break
        ip = np.power(i, p, where=(i > 0))
        for q in q_grid:
            if time.time() >= deadline - 100:
                break
            shape = ip * np.power(rev, q, where=(rev > 0))
            # numerically safe normalisation
            s = shape.sum()
            if s == 0.0 or np.isnan(s) or np.isinf(s):
                continue
            shape /= s
            # mix with the uniform vector – λ ∈ [0,1]
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ sub‑gradient descent on the *exact* max‑convolution
    # --------------------------------------------------------------
    # initialise the log‑probability (soft‑max) vector
    eps = 1e-12
    a = best_seq.copy()
    z = np.log(np.maximum(a, eps))

    # simple decreasing step‑size schedule
    eta0 = 0.5
    it = 0
    max_iters = 50000               # upper bound – real stop is time‑driven

    while time.time() < deadline - 30 and it < max_iters:
        # (a) compute the full linear convolution a*a
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()

        # (b) active indices (within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # corresponding shift values (range: –(n‑1)…+(n‑1))
        shifts = active - (n - 1)
        n_active = len(shifts)

        # (c) analytic sub‑gradient of the max‑operator w.r.t. a
        grad_a = np.zeros_like(a)
        for dt in shifts:
            if dt == 0:
                grad_a += 2.0 * a
            else:
                sft = abs(dt)
                # a_{i+dt} contribution
                grad_a[: n - sft] += a[sft:]
                # a_{i‑dt} contribution
                grad_a[sft:] += a[: n - sft]
        grad_a /= n_active                     # average over the active set

        # (d) chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # (e) step‑size and update
        eta = eta0 / np.sqrt(it + 1)
        z -= eta * grad_z
        a = _softmax(z)                        # stays on the simplex

        # (f) evaluate and keep the best vector found so far
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it} improved score={sc:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 3️⃣ final hill‑climb (exact objective)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # post‑process: guarantee non‑negativity and unit sum
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # return plain Python list, as required by the evaluation harness
    return [float(v) for v in best_seq]
```
