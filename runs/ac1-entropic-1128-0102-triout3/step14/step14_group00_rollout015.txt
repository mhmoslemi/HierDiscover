def _exp_gradient_descent(a0, deadline, start_time, eta0=0.7, tol_factor=1e-8):
    # returns best vector and score
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    eps_floor = 1e-14
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # tolerance
        tol = max_val * tol_factor
        ks = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2.0 * a[k-idx]
        grad /= float(len(ks))
        # step size schedule
        eta = eta0 / math.sqrt(it)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # enforce floor and renorm
        a = np.maximum(a, eps_floor)
        sum_a = a.sum()
        if sum_a > 0:
            a /= sum_a
        else:
            a = np.full(n, 1.0/n)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,} score={cur_score:.6f}")
        # optional random perturb
        if it % 200000 == 0:
            # add small random noise to escape
            noise = (np.random.rand(n) - 0.5) * 1e-3  # nice magnitude
            a = a + noise
            a = np.maximum(a, eps_floor)
            a /= a.sum()
    return best, best_score
```

"""
A fast, multi‑stage optimizer for the self‑convolution L∞ problem.

The algorithm
---------------
1. **Candidate generation** – build a diverse pool of analytic shapes:
   uniform, a dense beta‑grid, power‑law (U‑shaped) families,
   a sinusoid, the arcsine (Beta(0.5,0.5)) and a handful of random
   Dirichlet draws.  Every vector is normalised to L₁=1.

2. **Screening** – evaluate the exact objective for all candidates
   (using an FFT based linear self‑convolution) and keep the best few.

3. **Refinement** – each of the best candidates is polished with an
   exponentiated‑gradient (multiplicative) descent on the probability
   simplex.  The gradient is a sub‑gradient of the max‑convolution;
   a relatively large initial step (η₀≈1.0) and a looser tolerance for
   the active set give a smoother search.  The optimiser runs only
   for a time slice that is shared between the candidates.

4. **Annealing polish** – a light simulated‑annealing stage moves tiny
   packets of mass between random positions, accepting moves with a
   standard Metropolis rule.  This helps to escape the last few flat
   regions of the objective landscape.

All steps respect the overall time budget (default 1000 s).  The final
sequence consists of non‑negative floats, sums to 1 and is returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions for FFT based linear self‑convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    Continuous pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n        # mid‑point rule on (0,1)
    # Guard against a,b very close to 0 (which would produce infinities)
    shape = np.where(xs > 0,
                     np.where(xs < 1,
                              (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0)),
                              0.0),
                     0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_powerlaw_shape(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law (U‑shaped) profile:
        w[i] ∝ (x_i)^{-p} + (1‑x_i)^{-p},
    with x_i = (i+0.5)/n.
    """
    xs = (np.arange(n) + 0.5) / n
    eps = 1e-12
    shape = (xs + eps) ** (-p) + (1.0 - xs + eps) ** (-p)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Multiplicative (exponentiated‑gradient) descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 1.0,
                          tol_factor: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Multiplicative (EG) descent on the probability simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()

        # active set: all indices within a small relative tolerance
        tol = max_val * tol_factor
        ks = np.where(conv >= max_val - tol)[0]

        # sub‑gradient (average over the active set)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # tiny floor + renormalisation (probability simplex)
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

        # occasional random jitter to help escape plateaus
        if it % 200_000 == 0:
            jitter = (rng.random(n) - 0.5) * 1e-3
            a = a + jitter
            a = np.maximum(a, eps_floor)
            a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 2_000_000  # generous but still cheap per step

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling from T_start down to T_end
        T_start, T_end = 0.02, 1e-6
        T = T_start * (T_end / T_start) ** (it / max_steps)

        src = rng.choice(n, p=a)               # bias source towards larger mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine builds a large pool of analytic shapes, keeps the best few,
    refines each of them with multiplicative gradient descent and finishes
    with a simulated‑annealing polish.  The returned list contains non‑negative
    floats and respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length for fast FFTs
    # --------------------------------------------------------------
    n = 16384   # 2¹⁴ – reasonable compromise between resolution and speed

    # --------------------------------------------------------------
    # 2) Build a diversified pool of candidate shapes
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # Beta‑grid
    beta_vals = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    for a in beta_vals:
        for b in beta_vals:
            v = _make_beta_shape(n, a, b)
            pool.append((_objective(v), v.copy()))

    # Power‑law (U‑shaped) family
    power_vals = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0, 4.0]
    for p in power_vals:
        v = _make_powerlaw_shape(n, p)
        pool.append((_objective(v), v.copy()))

    # Sinusoid (low in the centre, high on the sides)
    xs = (np.arange(n) + 0.5) / n
    sinus = np.sin(np.pi * xs)          # zero at ends, one in the middle
    sinus = sinus / sinus.sum()
    pool.append((_objective(sinus), sinus.copy()))

    # Arcsine distribution (Beta(0.5,0.5))
    arcsine = _make_beta_shape(n, 0.5, 0.5)
    pool.append((_objective(arcsine), arcsine.copy()))

    # Random Dirichlet draws (sparse) – useful for “Sidon‑set‑like” patterns
    conc_vals = [0.2, 0.5, 0.8, 1.5]
    for conc in conc_vals:
        for _ in range(15):
            v = rng.dirichlet(np.full(n, conc))
            pool.append((_objective(v), v.copy()))

    # Random mixtures of two beta shapes – enrich the search space
    for _ in range(40):
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1.0 - w) * _make_beta_shape(n, a2, b2)
        v = v / v.sum()
        pool.append((_objective(v), v.copy()))

    # Optional global starter (if the environment provides it)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                pool.append((sc, starter.copy()))
    except Exception:
        pass

    # Keep only the best few candidates for the refinement stage
    pool.sort(key=lambda x: x[0])
    top_k = 8
    pool = pool[:top_k]

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Refine each top candidate with exponentiated‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:   # reserve a little time for annealing later
        # allocate approx. 70 % of the remaining budget to EG, split equally
        anneal_time = min(30.0, remaining * 0.2)
        eg_time = (remaining - anneal_time) / len(pool)

        for idx, (sc, vec) in enumerate(pool):
            eg_deadline = time.time() + eg_time
            refined_vec, refined_score = _exp_gradient_descent(
                vec,
                eg_deadline,
                start_time,
                eta0=1.0,
                tol_factor=1e-8
            )
            print(f"[{time.time() - start_time:.2f}s] Refined candidate {idx+1}/{len(pool)} score={refined_score:.6f}")
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_vec.copy()

        # --------------------------------------------------------------
        # 4) Light simulated‑annealing polish (use whatever time is left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            best_seq, best_score = _anneal(best_seq, deadline, start_time)
            print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Normalisation & conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
