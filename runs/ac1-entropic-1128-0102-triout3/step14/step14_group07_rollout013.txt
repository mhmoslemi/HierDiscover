def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - .5

    # Choose N
    if deadline - start > 800: N=65536
    elif > 500: N=32768
    else: N=16384

    # Helper functions: conv_self etc (FFT)

    # Candidate generation
    candidates = []

    # uniform
    a = np.full(N, 1.0/N)
    candidates.append(a)

    # symmetric ramp shapes (use original _make_shape) but also asymmetric
    for p in np.linspace(0.2, 2.5, 7):
        for plateau in np.linspace(0, 0.5, 5):
            candidates.append(_make_shape(N, p, plateau))
            # also asymmetric: decreasing ramp only
            ramp = _make_asymmetric_ramp(N, p, left=False) # maybe decreasing
            candidates.append(ramp)

    # Beta with low alpha
    for alpha in np.concatenate((np.arange(0.02, 0.24, 0.04), np.arange(0.5, 1.2, 0.2))):
        candidates.append(_make_beta_shape(N, alpha))

    # Gaussian
    for mu_frac in [0.25, 0.5, 0.75]:
        sigma_frac = 0.1
        candidates.append(_make_gaussian(N, mu_frac*N, sigma_frac*N))

    # Bimodal
    for centers in [(0.2, 0.8), (0.3,0.7), (0.4,0.6)]:
        candidates.append(_make_bimodal(N, centers[0]*N, centers[1]*N, sigma_frac*N))

    # Random Dirichlet with multiple concentration
    for conc in [0.001,0.005,0.01,0.02,0.05,0.1,0.2]:
        for _ in range(5):
            a = rng.gamma(conc, 1.0, size=N)
            a_sum = a.sum()
            if a_sum>0:
                a = a / a_sum
                candidates.append(a)
                # also randomized permutation
                perm = rng.permutation(N)
                candidates.append(a[perm])

    # Compute scores for each candidate, keep top K
    K = 15
    pool = []
    for a in candidates:
        score = _objective(a)
        _push_top(pool, score, a, K)
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f'Initial best {best_score:.6f}')

    # Phase: Adam refinement
    # allocate time per candidate
    remaining = deadline - time.time()
    if remaining > 30:
        per = remaining / len(pool)
    else:
        per = remaining * 0.9

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per
        if sub_deadline >= deadline: break
        refined, refined_score = _adam_smooth_max_custom(vec, sub_deadline, start)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f'Candidate {idx} improved best to {best_score:.6f}')

    # final SA polish
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline, start, rng, max_move_frac=0.05)

    # final normalize
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total>0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0/N)

    final_obj = _objective(best_seq)
    print(f'Final score {final_obj:.6f}')
    return best_seq.tolist()
```

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The optimisation proceeds in four stages:

1. **Diverse initial pool** – generate analytic families (uniform,
   power‑ramps, asymmetric ramps, Beta‑densities, Gaussians,
   bimodal Gaussians) and many random Dirichlet vectors (several
   concentration parameters).  For each Dirichlet draw also try a
   random permutation to break symmetry.  All candidates are projected
   onto the probability simplex and the best K are kept.

2. **Projected Adam on a smooth‑max surrogate** – from each retained
   candidate run Adam on the temperature‑smoothed objective
   `f_T(a)=T·log Σ exp(conv/T)`.  The temperature starts at 1.0 and
   decays exponentially to 1e‑6; the learning‑rate decays as 1/√t.
   After each step the vector is projected back onto the simplex.
   The best vector seen for the candidate is retained.

3. **Projected sub‑gradient descent on the exact max‑conv** – a second
   refinement phase uses the true objective (max of the convolution).
   The sub‑gradient of the active convolution entry is
   `g_i = 2·a_{t*‑i}` (zero when the index is out of range).  Steps are
   taken with a 1/√t schedule and the simplex projection is reapplied.

4. **Fine‑grained simulated‑annealing polish** – a short SA run with
   mass‑transfer moves (move a small δ from a source index to a target
   index) finalises the solution.

All phases respect the overall time budget (default 1000 s).  The
returned list is a plain Python list of non‑negative floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear convolution a*a (real FFT, length 2·len(a)-1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Objective for a probability vector (sum==1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with a flat centre."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_asymmetric_power_ramp(n: int, p: float, left: bool = True) -> np.ndarray:
    """Monotone power‑ramp: large values on the left (left=True) or right."""
    idx = np.arange(n, dtype=float) / (n - 1)  # 0 … 1
    if left:
        shape = (1.0 - idx) ** p
    else:
        shape = idx ** p
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    """Discrete Gaussian on {0,…,n‑1}."""
    x = np.arange(n, dtype=float)
    shape = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_bimodal(n: int, mu1: float, mu2: float, sigma: float) -> np.ndarray:
    """Sum of two Gaussians."""
    x = np.arange(n, dtype=float)
    shape = np.exp(-0.5 * ((x - mu1) / sigma) ** 2) + np.exp(
        -0.5 * ((x - mu2) / sigma) ** 2
    )
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    step0: float = 0.2,
    T_start: float = 1.0,
    T_end: float = 1e-6,
    max_iter: int = 10**7,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # forward pass
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}"
            )

        # temperature schedule (exponential)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (correlation with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]  # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # back to simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑conv objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    step0: float = 0.3,
    max_iter: int = 10**6,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        t_star = int(conv.argmax())

        # compute sub‑gradient of the active convolution term
        grad = np.zeros_like(a)
        idx = np.arange(a.shape[0])
        j = t_star - idx
        mask = (j >= 0) & (j < a.shape[0])
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(it)
        a -= step * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} score={cur_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
    max_move_frac: float = 0.03,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # select source proportional to its weight (larger weights move more often)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}"
                )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a low‑score step‑function under the given time budget."""
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a discretisation size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool
    # ------------------------------------------------------------------
    TOP_K = 20
    pool = []  # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # symmetric power‑ramps (plateau + exponent grid)
    for p in np.linspace(0.2, 2.5, 7):
        for plat in np.linspace(0.0, 0.5, 5):
            a = _make_shape(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # asymmetric power‑ramps (only left‑hand side)
    for p in np.linspace(0.2, 2.5, 5):
        a = _make_asymmetric_power_ramp(N, p, left=True)
        _push_top(pool, _objective(a), a, TOP_K)

    # Beta densities (including very spiky U‑shapes)
    for alpha in np.concatenate((np.arange(0.02, 0.22, 0.04), np.arange(0.5, 1.5, 0.2))):
        a = _make_beta_shape(N, float(alpha))
        _push_top(pool, _objective(a), a, TOP_K)

    # Gaussians centred at a few positions
    for mu_frac in [0.2, 0.35, 0.5, 0.65, 0.8]:
        for sigma_frac in [0.07, 0.12]:
            mu = mu_frac * (N - 1)
            sigma = sigma_frac * N
            a = _make_gaussian(N, mu, sigma)
            _push_top(pool, _objective(a), a, TOP_K)

    # Bimodal Gaussians
    for mu1_frac, mu2_frac in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7)]:
        for sigma_frac in [0.08, 0.12]:
            mu1 = mu1_frac * (N - 1)
            mu2 = mu2_frac * (N - 1)
            sigma = sigma_frac * N
            a = _make_bimodal(N, mu1, mu2, sigma)
            _push_top(pool, _objective(a), a, TOP_K)

    # Random Dirichlet vectors with many concentration levels
    concs = [0.001, 0.003, 0.008, 0.02, 0.05, 0.12, 0.25]
    for conc in concs:
        for _ in range(10):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)
            # also try a random permutation (break symmetry)
            perm = rng.permutation(N)
            a_perm = a[perm]
            _push_top(pool, _objective(a_perm), a_perm, TOP_K)

    # Incorporate the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
                starter_up = starter_up.astype(float)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑style smooth‑max refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        per_candidate = remaining / min(len(pool), TOP_K)
    else:
        per_candidate = remaining * 0.9

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per_candidate
        if sub_deadline >= deadline:
            break
        refined, refined_score = _adam_opt(vec, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {refined_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3) Sub‑gradient descent (exact max) on the current best
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sub_deadline = time.time() + remaining * 0.6
        best_seq, best_score = _subgradient_descent(
            best_seq, sub_deadline, start_time
        )
        print(
            f"[{time.time() - start_time:.2f}s] After sub‑gradient best = {best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4) Optional short SA polish (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, start_time, rng, max_move_frac=0.02
        )

    # ------------------------------------------------------------------
    # 5) Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
