    for rank, (score0, vec0) in enumerate(pool):

def _make_exponential(n, r):
    # a_i proportional to r**i
    # compute geometric series sum
    # if r close 1, use alternative
    if abs(r - 1.0) < 1e-12:
        shape = np.full(n, 1.0 / n)

        # geometric series sum = (1 - r**n) / (1 - r)
        geom_sum = (1 - r**n) / (1 - r)
        shape = (r ** np.arange(n)) / geom_sum
    return shape

def _smooth_gradient_descent(a0, deadline, start_time,
                             max_iters=2000, 
                             beta0=0.5, beta_mult=1.03,
                             lr0=0.2,
                             beta_max=200.0):
    a = a0.copy()
    a = _project_simplex(a)  # ensure normalized
    best_a = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    # precompute FFT length for gradient conv_rev
    L = _next_pow2(3*n)   # for conv_rev length

    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = beta0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv
        conv = _conv_self(a)  # length 2n-1
        cur_max = float(conv.max())
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it}  score={cur_score:.6f}")

        # softmax weights
        # use shift for numerical stability
        max_conv = conv.max()
        # compute exponentials
        exp_weights = np.exp(beta * (conv - max_conv))
        sum_exp = exp_weights.sum()
        p = exp_weights / sum_exp   # shape length 2n-1

        # gradient via convolution of rev_a and p
        rev_a = a[::-1]
        # FFTs
        Af = np.fft.rfft(rev_a, n=L)
        Pf = np.fft.rfft(p, n=L)
        grad_conv = np.fft.irfft(Af * Pf, n=L)[: 3*n - 1]  # length 3n-1
        # extract central n slice
        grad = 2.0 * grad_conv[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = lr0 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

        # increase beta periodically
        if it % 20 == 0:
            beta = min(beta * beta_mult, beta_max)

    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

The algorithm:
1. Choose a power‑of‑two length (4096) – cheap FFTs and enough resolution.
2. Assemble a diverse pool of candidate shapes:
   * uniform, exponential, Gaussian, power‑ramp, beta, cosine tip,
   * a few random Dirichlet draws, and the optional global starter
     (height_sequence_1) if it is present.
3. Keep the best few candidates (by the true objective 2·n·max(conv)/sum²).
4. Refine each kept candidate with a **smooth‑max gradient descent**:
   – Replace the nondifferentiable max by a softmax (log‑sum‑exp) with a
     temperature β that is slowly increased.
   – The gradient of the smoothed objective is obtained analytically
     via a single convolution of the reversed vector with the softmax
     weights (all done with FFTs).
   – Adam‐style momentum and projection onto the probability simplex keep
     the iterate feasible.
5. Finish with a light simulated‑annealing “mass‑swap” polish (as in the
   reference code) to escape tiny local minima.
6. Return the best sequence found, normalised to sum = 1.

The whole search respects the 1000 s budget; intermediate progress and
objective values are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective, used for selection.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1."""
    if abs(r - 1.0) < 1e-12:
        return _make_uniform(n)
    # geometric series sum = (1 - r**n) / (1 - r)
    denom = (1.0 - r**n) / (1.0 - r)
    a = (r ** np.arange(n)) / denom
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = _make_uniform(n)
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps with a flat plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # midpoint rule on [0,1]
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    """Cosine tip: a_i ∝ 1+cos(π i/(n‑1)), vanishes at the ends."""
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (with Adam) on the simplex
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             max_iters: int = 3000,
                             beta0: float = 0.5,
                             beta_multiplier: float = 1.04,
                             beta_max: float = 200.0,
                             lr0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the *smooth* version of the objective:
        fβ(a) = 2·n·(log‑sum‑exp(β·conv(a))/β).
    The smoother (small β) yields a well‑behaved gradient; β is raised
    gradually so that the optimisation focuses on the true max.
    """
    a = a0.copy()
    a = _project_simplex(a)          # enforce Σa = 1, a ≥ 0
    best_a = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    # FFT length for the gradient convolution (rev_a * p)
    L_grad = _next_pow2(3 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta = beta0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # current full convolution (linear)
        conv = _conv_self(a)                     # length 2n‑1
        cur_max = float(conv.max())
        cur_score = float(2.0 * n * cur_max)     # sum(a)=1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD it={it:,}  obj={cur_score:.6f}")

        # ---- smooth max (log‑sum‑exp) ----
        # stable soft‑max weights
        max_c = conv.max()
        exp_weights = np.exp(beta * (conv - max_c))
        sum_exp = exp_weights.sum()
        p = exp_weights / sum_exp                 # shape (2n‑1,)

        # ---- gradient of the smooth max ----
        # grad = 2 * (rev_a * p)_{central slice}
        rev_a = a[::-1]
        Af = np.fft.rfft(rev_a, n=L_grad)
        Pf = np.fft.rfft(p, n=L_grad)
        conv_rev = np.fft.irfft(Af * Pf, n=L_grad)[: 3 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]     # length n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

        # slowly raise the smoothing parameter
        if it % 20 == 0:
            beta = min(beta * beta_multiplier, beta_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly move tiny packets of mass, Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 600_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = max(1e-4, 0.2 * (0.03 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)          # prefer moving mass from larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (Σa = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1) Length – power of two for cheap FFTs (moderate size)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 2) Build a diverse initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # deterministic families
    candidates.append(_make_uniform(n))
    candidates.append(_make_cosine_tip(n))

    # exponential decays
    for r in np.linspace(0.94, 0.999, 5):
        candidates.append(_make_exponential(n, r))

    # Gaussian‑like shapes (centered)
    for sigma in np.linspace(0.07, 0.25, 5):
        candidates.append(_make_gaussian(n, sigma))

    # power‑ramp family
    for p in np.linspace(0.2, 2.0, 5):
        for alpha in np.linspace(0.0, 0.5, 3):
            candidates.append(_make_power_shape(n, p, alpha))

    # beta‑distribution family
    for a_par in [0.5, 1.0, 2.0, 4.0]:
        for b_par in [0.5, 1.0, 2.0, 4.0]:
            candidates.append(_make_beta_shape(n, a_par, b_par))

    # random Dirichlet draws (moderate concentration)
    rng = np.random.default_rng()
    for _ in range(120):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.1, 0.8)   # varying concentration
        candidates.append(rng.dirichlet(np.full(n, conc)))

    # optional global starter (if the host provides it)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                candidates.append(starter)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Evaluate and keep the most promising few
    # --------------------------------------------------------------
    pool = []                 # list of (score, vector)
    pool_cap = 10

    def _maybe_add(vec):
        sc = _objective(vec)
        if np.isfinite(sc):
            if len(pool) < pool_cap:
                pool.append((sc, vec.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, vec.copy())
                pool.sort(key=lambda x: x[0])

    for v in candidates:
        _maybe_add(v)

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial pool best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Refine the top candidates with smooth‑max gradient descent
    # --------------------------------------------------------------
    for idx, (init_sc, init_vec) in enumerate(pool):
        if time.time() >= deadline - 5.0:
            break
        print(f"[{time.time() - start_time:.2f}s] Gradient‑descent start on candidate {idx+1}")
        refined_vec, refined_score = _smooth_gradient_descent(
            init_vec,
            deadline,
            start_time,
            max_iters=2500,
            beta0=0.5,
            beta_multiplier=1.06,
            beta_max=150.0,
            lr0=0.3,
        )
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_vec.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after GD = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final annealing polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing polish")
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalise & return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
