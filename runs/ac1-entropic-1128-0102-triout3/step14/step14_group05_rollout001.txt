def _simulated_annealing_hill_climb(a, L_fft, deadline, start_time):
    rng = np.random.default_rng()
    cur = a.copy()
    cur_score = _objective(cur, L_fft)
    best = cur.copy()
    best_score = cur_score
    T0 = 0.1
    T_end = 1e-6
    total_iters = 200000
    iter = 0
    while time.time() < deadline and iter < total_iters:
        # pick src, dst
        src = rng.integers(n)
        if cur[src] <= 0:
            continue
        dst = rng.integers(n)
        if src == dst: continue
        # transfer proportion p between 0 and 0.1
        p = rng.random() * 0.1
        eps = cur[src] * p
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_score = _objective(cand, L_fft)
        delta = cand_score - cur_score
        # temperature schedule linear
        T = T0 * ((T_end/T0) ** (iter/total_iters))
        accept = False
        if delta < 0:
            accept = True
        else:
            if np.exp(-delta / T) > rng.random():
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter {iter} new best {best_score:.6f}")
        iter += 1
    return best, best_score

def _project_onto_simplex(v):
    # from https://gist.github.com/jaorenliu/e2749286b820b6ce5ea2
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _subgradient_descent(a, L_fft, deadline, start_time):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a, L_fft)
    eta = 0.5
    t = 0
    while time.time() < deadline:
        # evaluate conv
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        idx_max = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
        # compute subgradient
        grad = np.zeros_like(a)
        for k in idx_max:
            start_i = max(0, k - (n-1))
            end_i = min(k, n-1)
            i_range = np.arange(start_i, end_i+1)
            grad[i_range] += 2 * a[k - i_range]
        grad /= len(idx_max) if idx_max.size>0 else 1

        # step size schedule: eta_t = eta0 / sqrt(t+1)
        step = eta / np.sqrt(t+1)
        a_new = a - step * grad
        # projection onto simplex
        a_new = _project_onto_simplex(a_new)

        # evaluate objective
        score = _objective(a_new, L_fft)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] iter {t}, new best {best_score:.6f}")

        a = a_new
        t += 1
        if t >= 200000: # safety
            break
    return best_a, best_score

def _project_onto_simplex(v):
    # Based on Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <= 0 -> return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _simulated_annealing(a, L_fft, deadline, start_time):
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur, L_fft)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # schedule
    max_iters = 2000000  # maybe large but will break by deadline
    # parameters
    init_temp = 0.01
    final_temp = 1e-6
    it = 0
    while time.time() < deadline and it < max_iters:
        src = rng.integers(n)
        if cur[src] <= 0:
            it += 1
            continue
        dst = rng.integers(n)
        if src == dst:
            it += 1
            continue
        # choose transfer fraction up to say 0.2
        frac = rng.random() * 0.2
        eps = cur[src] * frac
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        # do not need projection because sum remains 1 and positivity holds
        cand_score = _objective(cand, L_fft)
        delta = cand_score - cur_score
        # temperature schedule
        progress = it / max_iters
        T = init_temp * (final_temp / init_temp) ** progress
        accept = False
        if delta < 0:
            accept = True
        else:
            if np.exp(-delta / (T + 1e-12)) > rng.random():
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter {it}, new best {best_score:.6f}")
        it += 1
    return best, best_score
```

    2·n·max(conv(a,a)) / (sum(a)**2)
over the probability simplex (non‑negative, unit‑sum).  The objective is
convex; its sub‑gradient at a point a is obtained from the index(s) k at
which the self‑convolution attains its maximum:
    ∂max(conv)/∂a_i = 2·a_{k‑i}  (if 0≤k‑i<n, else 0).
A projected sub‑gradient method (Euclidean projection onto the simplex)
runs for many thousands of cheap FFT‑based iterations, far more than the
previous L‑BFGS schedule.  After this global optimisation a short
simulated‑annealing hill‑climb with larger mass transfers refines the
solution.  The whole routine respects a 1000 s wall‑clock budget and
returns the best sequence found as a plain list[float].
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector a is assumed non‑negative; the function works for any sum
    (the sum is squared in the denominator).
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    max_iters: int = 200_000,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Runs a projected sub‑gradient method on the convex function
    max(conv(a,a)).  Returns the best vector seen and its objective.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(best_a, L_fft)

    # pre‑allocate gradient array
    grad = np.empty_like(a)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- compute self‑convolution and locate maximal index(es) -----
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        idx_max = np.where(np.isclose(conv, max_val, atol=1e-12))[0]

        # ----- sub‑gradient of max(conv) -----
        grad.fill(0.0)
        for k in idx_max:
            # valid i satisfy 0 ≤ k‑i < n  ⇔  max(0,k‑n+1) ≤ i ≤ min(k,n‑1)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]
        grad /= float(len(idx_max))

        # ----- step size (standard diminishing schedule) -----
        step = step0 / np.sqrt(t + 1.0)

        # ----- Euclidean projection onto the simplex -----
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # ----- evaluate exact objective -----
        score = _objective(a_candidate, L_fft)

        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad iter {t:5d} "
                f"new best {best_score:.6f}"
            )

        a = a_candidate

        # occasional random jitter to avoid possible stalls
        if (t + 1) % 5_000 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (exact objective)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers a random fraction of mass from one index to
    another.  The temperature drops exponentially; only moves that improve
    the exact objective are always accepted, worsening moves are accepted
    with probability exp(-Δ/Temp).
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = _objective(cur, L_fft)

    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    # temperature schedule (fast exponential cooling)
    T0 = 0.05
    Tf = 1e-6

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # pick source (must have some mass) and destination
        src = rng.integers(n)
        if cur[src] <= 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction (up to 20 %) of the source mass
        frac = rng.random() * 0.20
        eps = cur[src] * frac

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps  # sum stays 1, non‑negativity preserved

        cand_score = _objective(cand, L_fft)
        delta = cand_score - cur_score

        # linearly map iteration to progress in [0,1] for temperature
        progress = it / max_iters
        T = T0 * (Tf / T0) ** progress

        accept = False
        if delta < 0:
            accept = True
        else:
            if np.exp(-delta / (T + 1e-12)) > rng.random():
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA iter {it:7d} "
                    f"new best {best_score:.6f}"
                )

    return best, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine comprises:
      1. a baseline (uniform or global starter)
      2. a quick geometric / power‑law sweep
      3. many‑thousand‑iteration projected sub‑gradient descent
      4. a final simulated‑annealing hill‑climb
    The best sequence (normalised to unit sum) is returned as a list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between expressiveness
    # and speed on a single core
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – use global starter if present, otherwise uniform
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size == n and starter.sum() > 0:
            a0 = np.maximum(starter, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    best_seq = a0.copy()
    best_score = _objective(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric‑power families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)  # 1‑based
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=25)
    p_grid = np.linspace(-3.0, 2.0, 31)

    for r in r_grid:
        if time.time() >= deadline - 150:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _objective(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                        f"p={p:5.2f} λ={lam:.2f} → {best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (many cheap iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200:
        a_sgd, score_sgd = _subgradient_descent(
            best_seq, L_fft, deadline, start_time, max_iters=200_000, step0=0.6
        )
        if score_sgd < best_score:
            best_score = score_sgd
            best_seq = a_sgd.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad final {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing hill‑climb (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        a_sa, score_sa = _simulated_annealing(
            best_seq, L_fft, deadline, start_time, max_iters=2_000_000
        )
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA final {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final clean‑up – ensure feasibility and normalise
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
