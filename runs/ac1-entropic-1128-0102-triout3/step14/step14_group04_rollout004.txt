def lp_refine(a, deadline, start_time, max_iters=50):
    n=len(a)
    active_tol=0.005  # 0.5% maybe.
    step=0.3 (initial)
    while time.time()<deadline and max_iters>0:
        conv = _conv_self(a)
        M = conv.max()
        # active indices where conv >= M * (1 - active_tol)
        active_idx = np.where(conv >= M * (1 - active_tol))[0]
        if len(active_idx) == 0:
            break
        # Build A_ub rows
        rows = []
        # We'll build as list of arrays.
        for k in active_idx:
            row = np.zeros(n+1)
            # fill b coefficients
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # i from lo..hi: coefficient for b_{k - i} = a[i]
            # Equivalent: for j in range(lo, hi+1): b_idx = k - j
            # row[b_idx] = a[j]
            # We can vectorize:
            # j = np.arange(lo, hi+1)
            # b_idx = k - j
            # row[b_idx] = a[j]
            row[k - np.arange(lo, hi+1)] = a[lo:hi+1]
            row[-1] = -1.0  # coefficient for t
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(len(active_idx))
        A_eq = np.zeros((1, n+1))
        A_eq[0,:n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0.0, None)] * n + [(0.0, None)]  # t >= 0
        # Solve LP
        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]), A_ub=A_ub, b_ub=b_ub,
                       A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                       method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print("LP failed, break.")
            break
        b = res.x[:n]
        # mixing step
        a = (1 - step) * a + step * b
        # reproject onto simplex (should sum already 1 but numeric). Ensure non-negative.
        a = np.maximum(a, 0.0)

        max_iters -= 1
        # optionally compute new score and print.
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] LP iter score={score:.6f}")
    return a, _objective(a)

"""
Hybrid optimiser for the max‑convolution objective

The algorithm proceeds in three stages:

1. **Diverse initial pool** – a handful of analytic windows (uniform,
   triangular, Gaussian, Kaiser, Tukey, cosine tip, etc.) together with
   a few random Dirichlet draws are normalised to sum = 1 and the best
   one (lowest 2·n·max(conv)/sum²) is kept.

2. **Active‑set linear‑programming refinement** – given the current
   sequence *a*, we solve a small LP:
        minimise   t
        s.t.       (a * b)_k ≤ t   for all “active’’ indices k
                   Σ b_i = 1 ,  b_i ≥ 0
   The constraints are only kept for convolution entries that are within
   a small tolerance of the current maximum (default 2 %).  The LP
   therefore has at most a few hundred constraints even for n≈4000,
   which makes it fast.  The resulting *b* is mixed into *a* with a
   decreasing step size; the process repeats while time permits,
   monotonically decreasing the objective.

3. **Polish** – a short projected sub‑gradient descent (with Adam‑style
   momentum) followed by a light simulated‑annealing mass‑swap phase
   are applied to squeeze out the last bits of improvement.

All vectors are kept on the probability simplex (non‑negative,
sum = 1).  The routine respects the 1000 s budget and returns a plain
Python list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of windows (all normalised to Σ=1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_triangular(n: int) -> np.ndarray:
    """Symmetric triangular (Bartlett) window."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - center) / (center + 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float = 0.2) -> np.ndarray:
    """Gaussian window centred in the middle, sigma relative to n."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    sigma_abs = sigma * n
    shape = np.exp(-0.5 * ((i - center) / sigma_abs) ** 2)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0

        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_tukey(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey window – cosine taper of proportion alpha."""
    if alpha <= 0:
        return _make_uniform(n)
    if alpha >= 1:
        # equivalent to Hann
        i = np.arange(n)
        shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        shape /= shape.sum()
        return shape
    i = np.arange(n)
    period = n - 1
    taper_len = int(alpha * period / 2)
    shape = np.ones(n)
    # left taper
    left = i[:taper_len]
    shape[:taper_len] = 0.5 * (1 + np.cos(np.pi * (-1 + 2 * left / (alpha * period))))
    # right taper
    right = i[-taper_len:]
    shape[-taper_len:] = 0.5 * (1 + np.cos(np.pi * (1 - 2 * (right - (n - taper_len)) / (alpha * period))))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x≥0, Σx=1} (Wang & Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with Adam‑style momentum."""
    a = a0.copy()
    n = a.shape[0]
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)  # denominator = 1 (Σa=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Random mass‑swap moves with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = max(1e-4, 0.1 * (0.05 / 0.1) ** (it / max_steps))

        src = rng.choice(n, p=a)          # bias toward larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP that minimises max_k (a*b)_k over the active
    set of convolution entries.  Mix the solution back into ``a``.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # attempt to import linprog once
    try:
        from scipy.optimize import linprog
    except Exception as exc:
        print("Scipy linprog not available – skipping LP refinement.")
        return best_a, best_score

    step = 0.3
    tol = 0.02          # active‑set tolerance
    max_iters = 30

    while time.time() < deadline and max_iters > 0:
        conv = _conv_self(a)
        M = conv.max()
        # active indices (within tol of the current maximum)
        active = np.where(conv >= M * (1.0 - tol))[0]
        if active.size == 0:
            break

        # Build the inequality matrix (one row per active index)
        rows = []
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1, dtype=int)               # i‑indices
            row = np.zeros(n + 1, dtype=float)                 # b[0…n‑1] and t
            row[k - j] = a[j]                                   # a_i for b_{k-i}
            row[-1] = -1.0                                      # -t
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(A_ub.shape[0])

        # Equality: Σb = 1
        A_eq = np.zeros((1, n + 1), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=float)

        bounds = [(0.0, None)] * n + [(0.0, None)]  # b ≥0, t ≥0

        remaining = max(0.0, deadline - time.time())
        try:
            res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                          A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'time_limit': remaining})
        except Exception as e:
            print("LP solve failed:", e)
            break

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        b_seq = res.x[:n]

        # Mix the new vector into the current one
        a = (1.0 - step) * a + step * b_seq
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter improvement score={cur_score:.6f}")

        # tighten parameters
        step = max(0.01, step * 0.85)
        tol = max(0.005, tol * 0.95)
        max_iters -= 1

    return best_a, best_score
