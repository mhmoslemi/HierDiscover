def _smooth_descent(a0, time_limit, start_time, beta0=10., beta_factor=1.05, beta_max=2000., step0=0.5):
    a = a0.copy()
    a = _project_simplex(a) # ensure sum=1
    best_a = a.copy()
    best_score = _objective(a)   # exact objective

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    beta = beta0

    while time.time() < time_limit:
        it += 1
        # compute conv and smooth gradient
        conv = _conv_self(a)       # length 2n-1
        max_conv = conv.max()
        # compute softmax weights
        # use shift for numerical stability
        shift = max_conv
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        p = exp_beta / sum_exp

        # compute gradient
        grad = _grad_from_weights(a, p)  # returns vector size n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size decays
        step = step0 / (np.sqrt(it) + 1)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

"""
A smooth‑max optimiser for the step‑function objective.

The method builds a diverse pool of analytic windows and random
Dirichlet draws, then refines the best few candidates by minimising
a log‑sum‑exp approximation of the max‑self‑convolution using an
Adam optimiser on the probability simplex.  Finally a short
mass‑swap simulated‑annealing polish is applied.  All work fits
inside the 1000 s budget and returns a plain Python list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum‑free, sum(a)=1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection  (Wang & Carreira‑Perpiñán, 2013)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # numerical guard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic window families (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    w = (0.42
         - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 1.0 + np.cos(np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two symmetric power ramps with a flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule on [0,1]
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) w.r.t. the probability vector
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute  ∂/∂a   (∑_k p_k a * a_shifted)   = 2·∑_i a_i p_{i+j}.
    The formula reduces to a convolution of `a` with the reversed weight
    vector `p`.  All work is done with real FFTs.
    """
    n = a.shape[0]
    # reverse p for the convolution formula derived in the analysis
    p_rev = p[::-1]
    L = _next_pow2(n + p_rev.shape[0] - 1)      # = next_pow2(3·n‑2)
    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p_rev, n=L)
    conv_rev = np.fft.irfft(Af * Pf, n=L)      # length L
    # we only need the first 3n‑2 entries
    conv_rev = conv_rev[: (3 * n - 2) ]
    offset = p.shape[0] - 1                    # = 2·n‑2
    idx = offset - np.arange(n)                # descending indices
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    beta0: float = 10.0,
                    beta_factor: float = 1.03,
                    beta_max: float = 2000.0,
                    step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))

    The iteration stops when the time budget expires.
    Returns the best (exact) objective seen during the run.
    """
    a = _project_simplex(a0.copy())
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    beta = beta0

    while time.time() < deadline:
        it += 1

        # ----- convolution and smooth max -----
        conv = _conv_self(a)                     # length 2·n‑1
        max_conv = conv.max()
        # log‑sum‑exp (numerically stable)
        shift = max_conv
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        p = exp_beta / sum_exp                   # softmax weights, sum(p)=1

        # ----- gradient of the smooth max -----
        grad = _grad_from_weights(a, p)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- stay on the simplex -----
        a = _project_simplex(a)

        # ----- occasional exact evaluation -----
        if it % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"smooth it={it:,} β={beta:.1f} best={cur_score:.6f}")

        # ----- raise the smoothness parameter slowly -----
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑swap annealing, identical to the previous script."""
    a = seq.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]

    it = 0
    max_it = 600_000
    while it < max_it and time.time() < deadline:
        it += 1
        # exponential cooling
        T = max(1e-5, 0.1 * (0.02 / 0.1) ** (it / max_it))

        src = rng.choice(n, p=a)          # favour larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.4   # up to 40 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate pool construction
# ----------------------------------------------------------------------
def _maybe_add(pool: list, vec: np.ndarray, cap: int) -> None:
    """Insert a vector into the pool if it improves the current worst entry."""
    score = _objective(vec)
    if len(pool) < cap:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # truncate or pad to match the current length
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                _maybe_add(pool, starter, pool_cap)
    except Exception:
        pass

    # ----- report initial pool best -----
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Smooth‑max refinement of the top candidates
    # ------------------------------------------------------------------
    # take the three best starters for a more thorough exploration
    top_candidates = [vec for _, vec in sorted(pool, key=lambda x: x[0])[:3]]

    for idx, cand in enumerate(top_candidates, 1):
        if time.time() >= deadline:
            break
        remains = deadline - time.time()
        # allocate at most 200 s per candidate, but never exceed the remaining budget
        target = min(200.0, remains * 0.6)
        sub_deadline = time.time() + target
        print(f"[{time.time() - start_time:.2f}s] Refining candidate {idx} for up to {target:.1f}s")
        refined_seq, refined_score = _smooth_descent(
            cand, sub_deadline, start_time,
            beta0=12.0, beta_factor=1.04, beta_max=2500.0, step0=0.6)

        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing polish")
        annealed_seq, annealed_score = _anneal(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score, best_seq = annealed_score, annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
