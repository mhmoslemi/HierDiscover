def _exponentiated_gradient_descent(a0, deadline, start_time):
    n = a0.shape[0]
    a = a0.copy()
    y = np.log(a + 1e-30)  # log weights
    best_a = a.copy()
    # compute initial objective
    conv = _conv_self(a)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv

    it = 0
    # pre-allocate arrays for speed? Ok.
    # schedule parameters
    eta0 = 0.5
    # tolerance
    # Possibly implement early stop if improvement tiny.

    while time.time() < deadline:
        it += 1
        # compute convolution and gradient via combined method
        # compute FFT length L = next_pow2(3*n-2)
        L = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=L)
        # auto-convolution
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,} best_score={cur_score:.6f}")

        # compute mask of active constraints
        tol = 1e-12 * max(1.0, max_conv)
        w = (conv >= max_conv - tol).astype(np.float64)  # shape length 2n-1

        # compute w_rev FFT
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, n=L)
        # gradient
        grad_full = np.fft.irfft(Af * W_rev, n=L)
        # gradient indices: 2n-2 - np.arange(n)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]  # shape (n,)

        # step size
        eta = eta0 / np.sqrt(it)
        # Exponentiated update
        y -= eta * grad
        # Stability: subtract max
        y_shift = y - np.max(y)
        a = np.exp(y_shift)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Choose a discretisation size N (larger N gives a better bound but
   costs more per iteration).  The size is selected from the remaining
   time budget.

2. Build a tiny pool of diverse starting vectors (uniform, a few Beta
   shapes, a handful of Dirichlet draws, and the previous best if it
   exists).  The best one (according to the exact objective) is kept.

3. Run an **exponentiated‑gradient** (mirror‑descent with KL‑divergence)
   on the exact ∞‑norm objective.  The sub‑gradient of
   `max(conv(a,a))` is obtained analytically:
        g_k = 2 * Σ_i w_i * a_{i‑k},
   where w is the binary mask of indices attaining the current maximum.
   The convolution needed for the gradient is performed by FFT, so each
   iteration costs only a few FFTs.

   The update is
        y ← y – η_t · g        (y are log‑weights)
        a ← softmax(y)       (projection onto the probability simplex)
   with a step‑size η_t = η₀/√t.  The best sequence seen so far is
   remembered.

4. If time permits, a short **simulated‑annealing** (mass‑swap) polish
   is applied to the result of step 3.

All operations stay on the probability simplex, therefore the external
`evaluate_sequence` function will see a sum of 1 and the returned score
is exactly `2·n·max(conv)`.  The routine respects the 1000 s budget and
always returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective for a probability vector (sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Light‑weight SA that swaps a small amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a, best_score = a.copy(), cur_score

    T_start, T_end = 0.2, 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        # biased source selection (more mass → higher chance)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,}  new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with KL‑divergence (exponentiated gradient) on
    f(a)=max(conv(a,a)).  Returns the best vector seen and its score.
    """
    n = a0.shape[0]
    a = a0.copy()
    # initialise log‑weights safely
    y = np.log(a + 1e-30)

    # pre‑compute index mapping for the gradient (2n‑2‑k)
    idx = (2 * n - 2) - np.arange(n)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5                     # initial step size
    while time.time() < deadline:
        it += 1
        # --- full FFT size sufficient for both conv and gradient ---
        L = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=L)

        # auto‑convolution (full)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]                 # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  best_score={cur_score:.6f}")

        # active‑set mask (tolerance for numerical ties)
        tol = 1e-12 * max(1.0, max_conv)
        w = (conv >= max_conv - tol).astype(np.float64)   # shape (2n‑1,)

        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        w_rev = w[::-1]                                   # reverse
        W_rev = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(Af * W_rev, n=L)         # length L, contains conv(a,w_rev)
        grad = 2.0 * grad_full[idx]                       # pick indices (2n‑2‑k)

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # exponentiated update
        y -= eta * grad
        # avoid overflow/underflow: shift by max
        y_shift = y - np.max(y)
        a = np.exp(y_shift)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed + 9999)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a discretisation size (larger N → better theoretical bound)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 2) Build a tiny pool of seeds and pick the best one
    # --------------------------------------------------------------
    pool = []

    # uniform
    pool.append(np.full(N, 1.0 / N))

    # a few symmetric Beta shapes
    for alpha in (0.05, 0.10, 0.20, 0.35, 0.50):
        pool.append(_make_beta_shape(N, alpha))

    # random Dirichlet draws with various concentration parameters
    for conc in (0.02, 0.05, 0.10, 0.30, 0.60):
        for _ in range(4):
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s > 0.0:
                a /= s
                pool.append(a)

    # incorporate the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # linear interpolation to required size
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0.0:
                starter_up /= starter_up.sum()
                pool.append(starter_up)
    except Exception:
        pass

    # evaluate pool and keep the best seed
    best_seq = None
    best_score = float('inf')
    for cand in pool:
        sc = _objective(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Exponentiated‑gradient descent from the best seed
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _exp_gradient_descent(best_seq, deadline, start_time)

    # --------------------------------------------------------------
    # 4) Optional SA polish (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.02)

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
