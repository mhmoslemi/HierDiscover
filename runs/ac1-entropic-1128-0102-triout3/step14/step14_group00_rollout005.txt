Projected onto simplex function:

def project_simplex(v):
    """Projection onto probability simplex."""
    # Implementation using sorted descending approach.
    # Ensure v is np.ndarray.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _softmax_grad(a, lam):
    conv = _conv_self(a)
    # compute softmax weights
    # lam could be large, so compute scaled = lam * conv to avoid overflow later
    scaled = lam * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()
    # compute gradient
    a_rev = a[::-1]
    n = a.shape[0]
    L = _next_pow2(3*n - 2)
    # compute convolution of p with a_rev
    Cp = np.fft.rfft(p, n=L)
    Ca_rev = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(Cp * Ca_rev, n=L)[: 3*n - 2]
    grad = 2.0 * conv_full[n-1 : 2*n-1]   # length n
    return grad

    a = a0.copy()

    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        t += 1
        # schedule lambda linearly increase to final
        lam = lam0 + (lam_final - lam0) * (t / 10000)  # ramp up first 10000 steps
        if lam > lam_final:
            lam = lam_final
        grad = _softmax_grad(a, lam)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce non-negativity and projection onto simplex
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / a.shape[0])
        # optionally symmetrize
        # a = (a + a[::-1]) / 2
        # a = a / a.sum()
        cur_score = _objective(a)
        if cur_score < best_score:

            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} lam={lam:.2f} score={cur_score:.6f}")

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096   # moderate size

    # Build pool
    pool = []

    # Uniform
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))
    # Beta shapes
    for _ in range(30):
        a = 10**rng.uniform(-2, 0.6)
        b = 10**rng.uniform(-2, 0.6)
        shape = _make_beta_shape(n, a, b)
        pool.append((_objective(shape), shape.copy()))
    # Random sparse
    for k in [64, 96, 128, 160]:
        for _ in range(5):
            cand = _make_sparse_candidate(n, k, rng)
            pool.append((_objective(cand), cand.copy()))
    # Global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                # If length differs, resample
                if starter.size != n:
                    # linear interpolation
                    xs_old = np.arange(starter.size) / (starter.size - 1)
                    xs_new = np.arange(n) / (n - 1)
                    starter = np.interp(xs_new, xs_old, starter)
                starter = starter / starter.sum()
                pool.append((_objective(starter), starter.copy()))
    except Exception:
        pass

    # Keep best few
    pool.sort(key=lambda x: x[0])
    pool = pool[:5]

    best_score, best_seq = pool[0]

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # Run Adam descent on each candidate (or best) within time budget
    # Allocate roughly 80% of remaining time for Adam, 20% for anneal
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + 0.8 * remaining
        # Use best candidate from pool as starting point for longer run
        best_seq, best_score = _adam_descent(best_seq, adam_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After Adam best score = {best_score:.6f}")

    # Annealing polish
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # Normalise and return list
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]

def _softmax_grad(a, lam):
    n = a.shape[0]
    conv = _conv_self(a)
    scaled = lam * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()

    grad = 2.0 * conv_full[n-1 : n-1 + n]  # gradient of max_conv(approx) w.r.t a
    grad = grad * (2.0 * n)  # scale to gradient of O
    return grad

def _project_simplex(v):
    """Projects a vector onto the probability simplex (non-negative entries summing to 1)."""
    if np.all(v <= 0):
        # return uniform
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find the rho where u_i > cssv_i / i
    rho = np.where(u > cssv / ind)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_descent(a0, deadline, start_time,
                  lam0=0.5, lam_final=200.0, eta=0.1,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  lam_ramp_iters=2000):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    # Optionally enforce symmetry each iteration
    while time.time() < deadline:
        t += 1
        # schedule lambda
        if t <= lam_ramp_iters:
            lam = lam0 + (lam_final - lam0) * (t / lam_ramp_iters)
        else:
            lam = lam_final

        grad = _softmax_grad(a, lam)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # enforce non-negative and simplex projection
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # symmetrize if needed
        a = (a + a[::-1]) / 2.0
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} lam={lam:.1f} score={cur_score:.6f}")

    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

The algorithm builds a modest pool of analytic and sparse candidates,
keeps the best few, and then refines the current best vector with a
projected Adam optimiser that minimises a smooth approximation of the
objective 2·n·max(conv(a,a)).  The approximation is a soft‑max (log‑sum‑exp)
over the convolution values; the temperature is slowly increased so the
optimiser eventually targets the true maximum.  After the continuous
optimisation a short simulated‑annealing polish is applied.

All updates are projected onto the probability simplex (non‑negative,
unit‑sum) and the vector is symmetrised at every step – the optimal
shape is believed to be symmetric.  The routine respects the supplied
time budget (default 1000 s) and returns a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) (no safety checks, assumes sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple analytic families
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta pdf on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sparse_candidate(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random set of k positions (k « n)."""
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=1}."""
    if np.all(v <= 0):
        # uniform fallback
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of 2·n·softmax_λ(conv(a,a)) w.r.t. a.
    lam is the temperature (larger → tighter approximation of max).
    """
    n = a.shape[0]

    # convolution a*a
    conv = _conv_self(a)                     # length 2·n‑1
    scaled = lam * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()        # soft‑max weights, sum to 1

    # gradient of each conv_k w.r.t. a is 2·a shifted
    a_rev = a[::-1]

    L = _next_pow2(3 * n - 2)                # length needed for the cross‑correlation
    fp = np.fft.rfft(p, n=L)
    fa_rev = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(fp * fa_rev, n=L)[: 3 * n - 2]

    # extract the n relevant entries (centered)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # ∂max_conv/∂a (smooth)
    grad *= 2.0 * n                             # scale to ∂O/∂a where O = 2·n·max_conv
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (with symmetrisation)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  lam0: float = 0.5,
                  lam_final: float = 200.0,
                  eta: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  lam_ramp_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the simplex, minimising the true objective.
    The temperature `lam` is increased linearly during the first
    `lam_ramp_iters` iterations, after which it stays at `lam_final`.
    The iterate is forced to be symmetric (a[i]=a[-1-i]) at every step.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    while time.time() < deadline:
        t += 1

        # temperature schedule
        if t <= lam_ramp_iters:
            lam = lam0 + (lam_final - lam0) * (t / lam_ramp_iters)
        else:
            lam = lam_final

        grad = _softmax_grad(a, lam)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep on the simplex and enforce symmetry
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        a = (a + a[::-1]) / 2.0          # symmetrisation
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} lam={lam:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the previous code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random mass swaps with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from 0.02 down to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # more likely to move mass from larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5   # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score
