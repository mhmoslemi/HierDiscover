def _targeted_hill_climb(a, deadline, start_time, L_fft):
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute conv
        conv = _conv_self_fft(best, L_fft)
        s_star = int(np.argmax(conv))
        # compute contributions
        # source indices in range to have both a_i and a_{s-i} defined
        start_i = max(0, s_star - (len(best)-1))
        end_i = min(s_star, len(best)-1)
        # contributions
        # create array of zeros length n
        contrib = np.zeros_like(best)
        # compute contributions for i: a[i] * a[s-i]
        # using slicing for speed:
        i_range = np.arange(start_i, end_i+1)
        contrib[i_range] = best[i_range] * best[s_star - i_range]

        # find src idx with largest contrib (if zero then break)
        src = int(np.argmax(contrib))
        if best[src] < 1e-14:
            # choose random src with positive mass
            pos = np.where(best > 1e-14)[0]
            if pos.size == 0:
                break
            src = int(rng.choice(pos))
        # choose dst idx with smallest contrib (pref zero)
        # could pick a random index with low contribution
        # compute sorted indices for contributions (maybe precomputed)
        # choose dst among bottom 10%.
        low_mask = (best > 1e-14)  # ensure not negative? Actually we can transfer to any index
        # we can pick any index, but avoid src
        all_idx = np.arange(len(best))
        dst = int(rng.choice(all_idx[all_idx != src]))
        # transfer epsilon
        eps = best[src] * rng.random() * 1e-2  # up to 1% maybe more
        if eps <= 0:
            continue
        cand = best.copy()
        cand[src] -= eps
        cand[dst] += eps
        # maybe enforce min 0
        cand = np.maximum(cand, 0.0)
        # renorm to sum 1
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        sc = _eval_fast(cand, L_fft)
        if sc < best_score - 1e-12:
            best_score = sc
            best = cand
            print(f"[{time.time() - start_time:.2f}s] targeted climb score={best_score:.6f}")
    return best, best_score

"""
Propose a step‑function (non‑negative sequence) that minimizes

      2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine mixes a few fast parametric families (Beta‑shapes,
geometric‑power, sparse spikes) with a soft‑max surrogate L‑BFGS
optimiser and a *targeted* hill‑climb that moves mass away from the
indices that contribute most to the current maximal convolution entry.
All work is done with FFT‑based convolutions for speed and respects a
≈ 1000 s wall‑clock budget.
"""

import time
import numpy as np

# optional SciPy – if unavailable we fall back to gradient‑free search
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)
    with a already non‑negative and (approximately) normalised.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Surrogate used for the L‑BFGS phase (smooth max of the convolution)
# ----------------------------------------------------------------------
class _SurrogateMax:
    """Smooth‑max (log‑sum‑exp) of the self‑convolution together with its gradient."""

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv
        self.L_corr = L_corr

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        conv = _conv_self_fft(a, self.L_conv)   # length 2n‑1
        max_c = conv.max()

        # smooth‑max (log‑sum‑exp)
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient w.r.t. a (2·(w * a_rev) shifted)
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]      # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        return surrogate, grad_z


# ----------------------------------------------------------------------
# Targeted hill‑climb (uses exact objective, moves mass from the
# “worst’’ contributing index to a low‑contributing one)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray, deadline: float,
                         start_time: float, L_fft: int):
    """
    Repeatedly move a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to an
    index that contributes least.  Accept only improving moves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        # current convolution and the shift attaining the max
        conv = _conv_self_fft(best_a, L_fft)
        s_star = int(np.argmax(conv))

        # contributions a_i * a_{s*-i}
        low = max(0, s_star - (n - 1))
        high = min(s_star, n - 1)
        idx_range = np.arange(low, high + 1, dtype=int)
        contrib = np.zeros_like(best_a)
        contrib[idx_range] = best_a[idx_range] * best_a[s_star - idx_range]

        # source = index with largest contribution (if all zero pick a random positive entry)
        src = int(np.argmax(contrib))
        if best_a[src] < 1e-14:
            pos = np.where(best_a > 1e-14)[0]
            if pos.size == 0:
                break
            src = int(rng.choice(pos))

        # destination = index with smallest contribution (prefer zero)
        # we restrict dst not to be src
        mask = np.arange(n) != src
        # take bottom 10 % of contributions (including zeros)
        sorted_idx = np.argsort(contrib[mask])
        n_candidates = max(1, int(0.1 * n))
        dst_candidates = np.where(mask)[0][sorted_idx[:n_candidates]]
        dst = int(rng.choice(dst_candidates))

        # move a tiny fraction of mass
        eps = best_a[src] * rng.random() * 5e-3   # up to 0.5 % of source mass
        if eps <= 0.0:
            continue
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build a Beta‑shape (arcsine‑like) distribution
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta‑like shape on {0,…,n‑1} (or None if degenerate)."""
    # 1‑based indices to avoid zero‑power issues
    i = np.arange(1, n + 1, dtype=np.float64)
    # a_i ∝ i^{α‑1} (n+1‑i)^{β‑1}
    with np.errstate(divide='ignore', over='ignore', invalid='ignore'):
        shape = np.power(i, alpha - 1.0) * np.power(n + 1 - i, beta - 1.0)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0.0:
        return None
    return shape / s


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine combines:
      * a few cheap parametric families (geometric‑power, Beta shapes,
        sparse random spikes),
      * an annealed L‑BFGS optimisation of a smooth‑max surrogate,
      * several random restarts,
      * a fast targeted hill‑climb that uses the exact objective.
    The best sequence (normalised to unit sum) is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15        # a tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096                               # 2^12 – good compromise between speed and expressiveness
    L_fft = _next_pow2(2 * n - 1)          # length for exact convolutions
    L_corr = _next_pow2(3 * n - 2)         # length for gradient‑related convolutions

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Small parametric sweeps
    # ------------------------------------------------------------------
    # geometric‑power families (already used in the reference implementation)
    idx = np.arange(1, n + 1, dtype=np.float64)
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=20)
    p_grid = np.linspace(-3.0, 2.0, 31)

    for r in r_grid:
        if time.time() >= deadline - 150:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            # mix a little with uniform – λ∈{0,0.25,0.5,0.75}
            for lam in (0.0, 0.25, 0.5, 0.75):
                cand = lam * best_seq + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] geo r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # Beta‑like shapes (heavy at the edges when α,β<1)
    alpha_grid = np.linspace(0.2, 1.8, 17)   # 0.2 … 1.8
    beta_grid = np.linspace(0.2, 1.8, 17)
    for a in alpha_grid:
        if time.time() >= deadline - 130:
            break
        for b in beta_grid:
            if time.time() >= deadline - 130:
                break
            shape = _beta_shape(a, b, n)
            if shape is None:
                continue
            sc = _eval_fast(shape, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"[{time.time() - start_time:.2f}s] beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # Sparse random spikes – try to approach the “Sidon‑set’’ intuition
    rng = np.random.default_rng(seed + 12345)
    for k in (60, 70, 80, 90, 100):
        if time.time() >= deadline - 110:
            break
        for _ in range(12):          # a few attempts per k
            idxs = rng.choice(n, size=k, replace=False)
            cand = np.zeros(n)
            cand[idxs] = 1.0 / k
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sparse k={k} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (temperature annealing)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 80:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
        for beta in beta_schedule:
            if time.time() >= deadline - 80:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))
            # allow a generous number of evaluations
            maxfun = 1200
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    factr=1e7,
                    disp=False,
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # warm‑start next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Random restarts (short L‑BFGS runs)
    # ------------------------------------------------------------------
    for r in range(4):
        if time.time() >= deadline - 55:
            break
        rng = np.random.default_rng(seed + r + 9999)
        noisy = best_seq + rng.normal(scale=0.02, size=n)
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 10.0
        obj = _SurrogateMax(n=n, beta=beta,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=_next_pow2(3 * n - 2))
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=400,
                disp=False,
                factr=1e7,
            )
        except Exception:  # pragma: no cover
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/4 improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Targeted hill‑climb (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Return a clean, normalised vector (non‑negative, sum = 1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
