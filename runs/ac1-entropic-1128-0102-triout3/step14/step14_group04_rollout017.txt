def _subgradient_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _objective(a)
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print("[{:.2f}s] SubGrad it={}, score={:.6f}".format(time.time() - start_time, it, cur_score))
        # active indices:
        top_k = 3
        ks = np.argpartition(conv, -top_k)[-top_k:]
        # compute gradient:
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # a[k-idx] vector:
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(top_k)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # projection:
        a = _project_simplex(a)
    return best_a, best_score

def _anneal_incremental(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    # ensure a sum =1
    a = a / a.sum()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # determine total steps based on time left
    # We will estimate steps per second by measuring loops; however we can just iterate until deadline.
    steps = 0
    max_steps = 5_000_000  # upper bound
    while time.time() < deadline and steps < max_steps:
        steps += 1
        # linear temperature schedule
        T = max(1e-5, 1.0 * (1 - steps / max_steps))
        # pick src weighted by a
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue
        if a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 0.5  # up to 50%
        # Update conv incrementally
        # use original a for slices.
        # linear part
        conv[src:src+n] -= 2.0 * eps * a
        conv[dst:dst+n] += 2.0 * eps * a
        # quadratic
        idx2src = 2*src
        idx2dst = 2*dst
        idxsd = src + dst
        if idx2src < conv.size:
            conv[idx2src] += eps * eps
        if idx2dst < conv.size:
            conv[idx2dst] += eps * eps
        if idxsd < conv.size:
            conv[idxsd] -= 2.0 * eps * eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score
        accept = False
        if delta < 0 or math.exp(-delta / T) > rng.random():
            accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA step={steps} best={best_score:.6f}")
        else:
            # revert changes
            conv[src:src+n] += 2.0 * eps * a
            conv[dst:dst+n] -= 2.0 * eps * a
            if idx2src < conv.size:
                conv[idx2src] -= eps * eps
            if idx2dst < conv.size:
                conv[idx2dst] -= eps * eps
            if idxsd < conv.size:
                conv[idxsd] += 2.0 * eps * eps
            # a unchanged
    return best_a, best_score

"""
Adaptive two‑phase optimizer for the step‑function objective.

Phase 1 – focused sub‑gradient descent on the few largest convolution
entries (top‑k).  An Adam‑style momentum with simplex projection is used
to quickly drive the maximum of a*a down.

Phase 2 – massive light‑annealing (mass‑swap) that updates the
self‑convolution incrementally (O(n) per move).  A linear cooling schedule
allows the search to escape local minima and to polish the best candidate.

The whole search respects the 1000 s budget, works with any length n
(power of two for cheap FFT), and returns a plain Python list that can be
fed to ``evaluate_sequence``.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes (used to build a diverse start pool)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps with a flat plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # midpoint rule on [0,1]
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    shape /= shape.sum()
    return shape


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 1.0 + np.cos(np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_exp_shape(n: int, rate: float) -> np.ndarray:
    """Simple exponential decay, normalised to sum = 1."""
    i = np.arange(n)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Focused sub‑gradient descent (top‑k active set) with Adam momentum
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          top_k: int = 3,
                          step0: float = 0.5,
                          tol: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by minimising max(conv(a,a)).
    Only the `top_k` largest convolution entries are used to build the
    sub‑gradient; Adam momentum and simple step‑size decay are employed.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep a on the simplex
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)=1 → denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,}  score={cur_score:.6f}")

        # indices of the `top_k` largest entries
        ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the selected max‑entries (average)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]     # same formula as in the reference code
        grad /= float(top_k)

        # Adam update (gradient descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Massive light simulated‑annealing (mass‑swap moves) – incremental update
# ----------------------------------------------------------------------
def _anneal_incremental(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of mass between positions.
    The convolution is updated incrementally (O(n) per move) and a
    simple linear cooling schedule is used.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep exact convolution for fast max‑query
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    max_steps = 4_000_000          # generous upper bound; loop stops at deadline
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1

        # linear cooling: T goes from 1.0 down to ~1e‑4
        T = max(1e-4, 1.0 * (1.0 - step / max_steps))

        # source chosen proportionally to its mass (large entries move more often)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-18:
            continue

        # move up to 40 % of the source mass (adaptable)
        eps = a[src] * rng.random() * 0.4

        # --- incremental convolution update ---------------------------------
        # linear part
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        # quadratic correction
        i2s = 2 * src
        i2d = 2 * dst
        ids = src + dst
        if i2s < conv.size:
            conv[i2s] += eps * eps
        if i2d < conv.size:
            conv[i2d] += eps * eps
        if ids < conv.size:
            conv[ids] -= 2.0 * eps * eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score

        accept = False
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal step={step:,}  best={best_score:.6f}")
        else:
            # revert the convolution (a stays unchanged)
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            if i2s < conv.size:
                conv[i2s] -= eps * eps
            if i2d < conv.size:
                conv[i2d] -= eps * eps
            if ids < conv.size:
                conv[ids] += 2.0 * eps * eps

        # occasional full recomputation to wipe out accumulated rounding errors
        if step % 200_000 == 0:
            conv = _conv_self(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a length – a power of two makes FFT cheap
    # --------------------------------------------------------------
    n = 4096                     # 2¹ⁿⁿ – fine trade‑off between resolution and speed

    # --------------------------------------------------------------
    # 2) Build a diverse pool of promising candidates
    # --------------------------------------------------------------
    pool = []                     # (score, vector)
    pool_cap = 20

    def _maybe_add(vec):
        vec = np.asarray(vec, dtype=float)
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s <= 0.0:
            return
        vec /= s
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # classic windows and simple analytical families
    _maybe_add(_make_uniform(n))
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))
    _maybe_add(_make_cosine_tip(n))

    for p in [0.5, 1.0, 2.0]:
        for alpha in [0.0, 0.1, 0.2]:
            _maybe_add(_make_power_shape(n, p, alpha))

    for a_par in [0.5, 2.0, 5.0]:
        for b_par in [0.5, 2.0, 5.0]:
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    for rate in [0.0005, 0.001, 0.002]:
        _maybe_add(_make_exp_shape(n, rate))

    # random Dirichlet draws (moderate concentration)
    rng = np.random.default_rng()
    for _ in range(300):
        if time.time() >= deadline:
            break
        vec = rng.dirichlet(alpha=np.ones(n) * 0.5)
        _maybe_add(vec)

    # --------------------------------------------------------------
    # 3) If a globally supplied starter exists, try it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                _maybe_add(starter)
    except Exception:
        pass

    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Local refinement – focused sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:                     # need enough time for a meaningful run
        sub_deadline = time.time() + 0.75 * remaining
        refined_seq, refined_score = _subgradient_optimize(
            best_seq, sub_deadline, start_time, top_k=3, step0=0.6)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (mass‑swap moves, many iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal_incremental(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score, best_seq = annealed_score, annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved score = {annealed_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    return [float(v) for v in best_seq]
