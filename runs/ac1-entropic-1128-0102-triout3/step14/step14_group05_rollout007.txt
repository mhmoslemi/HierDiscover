"""
step_search_v2.py – A refined optimisation pipeline for the step‑function
challenge.

Key ideas
---------
* A fast parametric sweep (geometric‑power families) with a *coarse* grid
  followed by a time‑limited random sampling for finer exploration.
* A temperature‑annealed L‑BFGS optimiser on the smooth (log‑sum‑exp)
  surrogate of the max‑convolution – we raise the maximum number of
  function evaluations per temperature and use a richer β‑schedule.
* A *guided* hill‑climb that always moves mass away from the indices that
  contribute most to the current maximal convolution (the exact sub‑gradient
  of the true objective).  The step size is adaptive and we also fall back
  to small random transfers when no improvement is found.
* Simple projection onto the probability simplex keeps the iterate
  non‑negative and normalised without expensive LPs.

The routine respects the 1000 s wall‑clock budget, logs progress, and
returns the best sequence (a list of non‑negative floats) found.
"""

import time
import numpy as np

# optional – may be missing in a minimal environment
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Primitive utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (smooth max) used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max surrogate of max(conv) together
    with its gradient w.r.t. the log‑probability parameters `z`.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv
        self.L_corr = L_corr

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                      # a ≥ 0, Σa = 1
        # --- self‑convolution -------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)          # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, length 2n‑1
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # --- gradient w.r.t. a ------------------------------------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]      # length n

        # --- chain rule through the soft‑max transformation -------------------
        dot = np.dot(grad_a, a)               # scalar
        grad_z = a * (grad_a - dot)           # shape (n,)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Guided hill‑climb (exact objective & sub‑gradient)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray, deadline: float,
                       start_time: float, L_fft: int):
    """
    Local optimisation that repeatedly moves mass away from the positions
    that most contribute to the current maximal self‑convolution.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    cur_a = best_a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    # adaptive step size; start relatively aggressive
    eps_factor = 0.02

    while time.time() < deadline:
        # compute exact convolution and locate the maximal index
        conv = _conv_self_fft(cur_a, L_fft)
        i_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        # g_k = 2 * a[i_max - k]  for k s.t. 0 ≤ i_max‑k < n
        grad = np.zeros_like(cur_a)
        lo = max(0, i_max - (n - 1))
        hi = min(i_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * cur_a[i_max - idx]

        # source = index with largest gradient (most harmful)
        src = int(np.argmax(grad))
        # destination = index with smallest current mass
        dst = int(np.argmin(cur_a))

        # propose a transfer; amount limited by source mass and a modest factor
        max_transfer = cur_a[src] * eps_factor
        if max_transfer < 1e-12:
            # source became too small – fall back to a tiny random move
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst:
                continue
            max_transfer = cur_a[src] * eps_factor

        eps = max_transfer * rng.random()
        cand = cur_a.copy()
        cand[src] -= eps
        cand[dst] += eps

        # maintain feasibility (non‑negativity + sum = 1)
        cand = _project_to_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)

        if cand_score < cur_score:
            # accept as new current iterate
            cur_a = cand
            cur_score = cand_score
            # update global best if necessary
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] guided hill‑climb "
                      f"new best score={best_score:.6f}")
        else:
            # no improvement → shrink the step size a little
            eps_factor *= 0.99
            if eps_factor < 1e-4:
                eps_factor = 1e-4

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.001:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src != dst and cur_a[src] > 0:
                eps = cur_a[src] * 0.005 * rng.random()
                cand = cur_a.copy()
                cand[src] -= eps
                cand[dst] += eps
                cand = _project_to_simplex(cand)
                cand_score = _eval_fast(cand, L_fft)
                if cand_score < cur_score:
                    cur_a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] random jump "
                              f"new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine consists of:
      0️⃣  uniform baseline,
      1️⃣  fast parametric sweep (coarse + random sampling),
      2️⃣  annealed L‑BFGS on a smooth‑max surrogate (more iterations than
          the previous implementation),
      3️⃣  a guided hill‑climb that follows the exact sub‑gradient,
      4️⃣  a final safety projection & return.

    The best sequence (normalised to unit sum) is returned as a plain
    ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.15

    # problem size – 4096 works well on a single CPU core
    n = 4096
    L_fft = _next_pow2(2 * n - 1)        # for exact evaluations
    L_corr = _next_pow2(3 * n - 2)       # for surrogate gradient

    # ------------------------------------------------------------------
    # 0️⃣  Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (coarse grid)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)      # 1‑based index
    # coarse geometric ratio grid (log‑space)
    r_grid = np.logspace(np.log10(0.93), np.log10(0.999999), num=30)
    p_grid = np.linspace(-4.0, 3.0, 35)              # exponent grid
    lam_grid = np.linspace(0.0, 1.0, 5)              # mixture with uniform

    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        # pre‑compute geometric term once
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 200:
                break
            # base shape: r^i * i^p
            shape = geo * np.power(idx, p)
            # avoid NaNs when p is negative and idx==0 (not possible here)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            # also try the symmetric version (mirror + original)
            shape_sym = (shape + shape[::-1]) * 0.5
            for base in (shape, shape_sym):
                for lam in lam_grid:
                    cand = lam * uniform + (1.0 - lam) * base
                    cand /= cand.sum()
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep r={r:.6f} "
                              f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣‑b  Random sampling of parameters (time‑bounded)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    while time.time() < deadline - 150:
        r = rng.uniform(0.93, 0.999999)
        p = rng.uniform(-5.0, 4.0)
        lam = rng.uniform(0.0, 1.0)
        geo = np.power(r, np.arange(n, dtype=np.float64))
        shape = geo * np.power(idx, p)
        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s
        # random choice between plain and symmetric
        if rng.random() < 0.5:
            shape = (shape + shape[::-1]) * 0.5
        cand = lam * uniform + (1.0 - lam) * shape
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] random param r={r:.6f} "
                  f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (richer schedule)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 120:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # a slightly finer temperature schedule
        beta_schedule = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0,
                         20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 120:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))
            # we allow a larger budget per temperature – still cheap enough
            maxfun = 1500
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,          # loose tolerance – we rely on the schedule
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → "
                      f"improved score={real_score:.6f}")

            # warm‑start next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Random restarts of the L‑BFGS stage (short bursts)
    # ------------------------------------------------------------------
    restarts = 4
    for r in range(restarts):
        if time.time() >= deadline - 80:
            break
        rng = np.random.default_rng(seed + r + 77777)
        noisy = best_seq + rng.normal(scale=0.02, size=n)   # a bit more jitter
        noisy = np.maximum(noisy, 0.0)
        noisy = _project_to_simplex(noisy)
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 5.0
        obj = _SurrogateMax(n=n, beta=beta,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=_next_pow2(3 * n - 2))
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=500,
                disp=False,
                factr=1e7,
            )
        except Exception:  # pragma: no cover
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} "
                  f"improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Guided hill‑climb (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _guided_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final guided hill‑climb "
                  f"score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: ensure non‑negative, normalise to unit sum and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")

    # Convert to plain Python list as required by the external harness
    return [float(v) for v in best_seq]
