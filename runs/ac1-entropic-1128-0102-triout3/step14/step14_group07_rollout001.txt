"""
Search for a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages under the 1000 s budget:

1. **Coarse sampling** – generate a small, diverse pool of candidates
   (uniform, Hann window, flat top, Beta‑shapes and random Dirichlet vectors)
   on a modest size `N_small` (≈ 8 k).  The `TOP_K` best are kept.

2. **Exact sub‑gradient descent** – for each pooled candidate we run a
   projected sub‑gradient method on the *true* objective
   (the max‑convolution, not a smooth approximation).  The gradient at the
   current worst shift `k*` is simply `g_i = 2·a_{k*‑i}` (zero‑padded).
   A decreasing step size (`η∝1/√t`) and Euclidean projection onto the
   probability simplex keep the iterates feasible.  Many cheap iterations
   are possible because `N_small` is small.

3. **Simulated‑annealing polish** – a short SA that swaps mass between
   two coordinates (mass‑move proposals) refines the best sequence.
   Finally the best small‑scale vector is linearly up‑sampled to a large
   size (`N_big` ≈ 64 k), symmetrised and re‑normalised.

All steps respect the time budget and only use NumPy (no external
solvers).  The returned list sums to 1, contains only non‑negative floats,
and can be fed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Objective for a probability vector (sum == 1): 2·n·max(conv)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted list of the k best (lowest) items."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Analytic shapes used for the initial pool
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution
# ----------------------------------------------------------------------
def _grad_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_k conv[k] at the current worst index `k`.
    The formula is g_i = 2·a_{k‑i} (zero‑padded outside [0,n‑1]).
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _sgd_subgradient(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     max_iter: int = 10**9) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv) with projection onto the simplex.
    Returns the best sequence seen before `deadline`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        k = int(conv.argmax())
        grad = _grad_at_max(a, k)

        # step size: η_t = η0 / sqrt(t)
        eta0 = 0.6
        step = eta0 / math.sqrt(it)

        a -= step * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SGD iter={it:,} best={cur_score:.6f}")

        # occasional random jitter to escape flat plateaux
        if it % 2000 == 0:
            noise = rng.normal(scale=1e-4, size=a.shape)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Light‑weight simulated annealing (mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # pick source proportional to its weight (biased towards large masses)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

def _upsample(a_small: np.ndarray, target_n: int) -> np.ndarray:
    n = a_small.shape[0]

    s = a_up.sum()
    if s > 0:
        a_up /= s
    else:
        a_up = np.full(target_n, 1.0 / target_n)
    return a_up

    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve a short slice for the final SA polish
    time_for_sa = min(30.0, max(5.0, remaining * 0.1))
    time_for_sgd = max(0.0, remaining - time_for_sa)
    per_candidate = time_for_sgd / max(1, len(pool))

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per_candidate
        if sub_deadline >= deadline:
            break
        refined_seq, refined_score = _sgd_subgradient(vec, sub_deadline,
                                                     start_time, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) Optional short SA polish (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.02)

    # --------------------------------------------------------------
    # 4) Upsample to the final length, symmetrise and re‑normalise
    # --------------------------------------------------------------
    final_seq = _upsample(best_seq, N_big)

    # Enforce (optional) symmetry – this often reduces the peak of the
    # autocorrelation without hurting the sum.
    final_seq = (final_seq + final_seq[::-1]) / 2.0
    final_seq = np.maximum(final_seq, 0.0)

    total = final_seq.sum()
    if total > 0:
        final_seq /= total
    else:
        final_seq = np.full(N_big, 1.0 / N_big)

    final_score = _objective(final_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in final_seq]
