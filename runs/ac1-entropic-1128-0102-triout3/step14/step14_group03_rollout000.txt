"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses a smooth‑approximation of the max‑convolution and
projected Adam optimisation on the probability simplex.

Key ideas
---------
* The objective is homogeneous, so we keep the vector normalised (sum = 1)
  and optimise only its shape.
* For a given vector a we compute the full self‑convolution via a single
  FFT (O(n log n)).
* The non‑smooth max is replaced by a temperature‑controlled smooth‑max
  (log‑sum‑exp).  Its gradient w.r.t. a can be written as a correlation
  between the soft‑max weights and a, which is also evaluated by FFT.
* Adam (first‑/second‑moment adaptation) together with Euclidean projection
  onto the probability simplex provides a fast, stable descent.
* The temperature is exponentially cooled, turning the smoothed problem
  gradually into the original one.
* A final short “mass‑transfer” hill‑climb refines the best vector found.

The routine respects the 1000 s budget, prints progress and returns the
best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the full convolution (length 2·len(a)‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed to be non‑negative, but the routine works for any real a.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Returns (smooth_max, weights) where
        smooth_max = tau * logsumexp(conv / tau)
    and weights = softmax(conv / tau) (they sum to 1).
    """
    max_c = conv.max()
    # shift for numerical stability
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = max_c + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth_max, weights


def _correlate_valid(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Computes the valid part of the correlation (w ⋆ a)[k] = Σ_j w_{k+j} * a_j
    for k = 0 … len(a)‑1.
    Both w and a are real, L is the FFT length (≥ len(w)+len(a)-1).
    """
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    start = a.shape[0] - 1
    return corr_full[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  Uses a smoothed max‑convolution, projected Adam optimisation,
    and a final tiny hill‑climb.  Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Problem size – use the same length as the supplied starter if
    # available, otherwise fall back to n = 4096.
    # --------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.ndim == 1 and seq0.size > 0:
            n = seq0.size
        else:
            n = 4096
    except Exception:
        n = 4096

    # FFT lengths
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n)              # for correlation w ⋆ a

    # --------------------------------------------------------------
    # Initialise a (sum = 1, non‑negative)
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=float)
        else:
            a0 = a0 / a0.sum()
    except Exception:
        a0 = np.full(n, 1.0 / n, dtype=float)

    # Baseline score
    best_seq = a0.copy()
    best_score = _eval_fast(best_seq, L_conv)
    print(f"[{time.time() - start_time:.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser on the smoothed objective
    # --------------------------------------------------------------
    a = a0.copy()
    m = np.zeros_like(a)                 # first moment
    v = np.zeros_like(a)                 # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                           # base learning rate (empirically good)
    beta1_pow = beta1
    beta2_pow = beta2

    # temperature schedule
    tau_start = 0.30
    tau_end = 5e-5
    tau = tau_start
    tau_decay = 0.9996                   # multiplicative decay per iteration

    it = 0
    no_improve_iters = 0                # for occasional jitter
    while time.time() < deadline:
        it += 1
        # ---- update temperature ----
        tau = max(tau_end, tau * tau_decay)

        # ---- FFT of current a ----
        a_fft = np.fft.rfft(a, n=L_corr)

        # ---- self‑convolution (exact) ----
        conv = np.fft.irfft(a_fft * a_fft, n=L_corr)[: 2 * n - 1]

        # ---- smooth max and soft‑max weights ----
        _, w = _smooth_max_and_weights(conv, tau)   # w sums to 1, length 2n‑1

        # ---- gradient of smooth max w.r.t. a ----
        grad = 2.0 * _correlate_valid(w, a, L_corr)

        # ---- Adam moments ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # ---- step ----
        step = lr0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # ---- projection onto the simplex (keeps sum = 1) ----
        a_candidate = _project_simplex(a_candidate)

        # ---- evaluate the true objective (not the smoothed one) ----
        cur_score = _eval_fast(a, L_conv)
        cand_score = _eval_fast(a_candidate, L_conv)

        # ---- accept / reject (simulated‑annealing flavour) ----
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            no_improve_iters = 0
        else:
            # probability of uphill move decreases with iteration count
            prob = math.exp(-(cand_score - cur_score) * 8.0 / (math.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate
                cur_score = cand_score
                no_improve_iters = 0
            else:
                no_improve_iters += 1

        # ---- keep global best ----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it:,} "
                  f"temp={tau:.2e} score={cur_score:.6f}")

        # ---- occasional jitter to escape flat plateaus ----
        if no_improve_iters >= 4000:
            eps_jitter = 1e-4 * rng.random(n) * a
            a = _project_simplex(np.maximum(a + eps_jitter, 0.0))
            no_improve_iters = 0

        # ---- safety break if we are very close to deadline ----
        if time.time() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # Final tiny random mass‑transfer hill‑climb (pure exploitation)
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move a tiny fraction of mass from src to dst
        eps_move = a[src] * rng.random() * 2e-3      # ≤0.2 % of src
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move
        new_score = _eval_fast(a_prop, L_conv)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=float)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
