    a = a0.copy()
    n = a.shape[0]
    v = np.log(a + 1e-12)  # logits for softmax
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 1
    # initial tau: maybe set to a small fraction of max conv.
    while time.time() < deadline:
        # compute a = softmax(v)
        max_v = np.max(v)
        exp_v = np.exp(v - max_v)  # for numeric stability
        a = exp_v / np.sum(exp_v)
        # conv
        conv = _conv_self(a)
        max_conv = conv.max()
        # compute smoothing temperature tau; we can reduce gradually
        tau = max_conv * 0.1  # set to 10% of current max
        # weight w

        w = w / np.sum(w)
        # compute gradient wrt a

        L = _next_pow2(len(a) + len(w_rev) - 1)

"""
Low‑peak step‑function search

This implementation builds a diverse pool of analytic window shapes,
creates many random Dirichlet starts and then refines each candidate with
a smooth approximation of the L∞‑norm of the self‑convolution.
The smooth loss is

    τ·log( Σ_k exp(conv_k/τ) )

with a temperature τ that is tied to the current peak.
Its gradient can be expressed as a single convolution, so each
iteration costs only two FFTs.  Adam on the soft‑max logits together
with projection onto the simplex yields fast descent.  A short
simulated‑annealing phase finishes the search.

The whole routine respects the 1000 s budget and returns a plain
Python list of non‑negative floats (sum≈1).  The algorithm is
fully self‑contained and does not rely on external files or network
access.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns an array of length 2·n‑1 (full convolution).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  (sum is assumed >0)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Wang–Carreira‑Perpiñán)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # numerical safety net
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (a few useful ones)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n)


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to a simple cosine if scipy is missing."""
    try:
        from scipy.signal import windows
        w = windows.kaiser(n, beta, sym=False)
    except Exception:
        # elementary approximation based on Bessel I0
        from scipy.special import i0   # scipy may still be present for i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        w = i0(beta * np.sqrt(1.0 - (i / alpha) ** 2))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_power_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Two symmetric power ramps with an optional flat centre.
    p > 0 controls curvature, plateau_frac∈[0,0.5] the relative size of the flat part.
    The result is normalised to sum = 1.
    """
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution (a,b>0)."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth optimisation of max(conv)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray, deadline: float, start_time: float):
    """
    Refine a probability vector using the smooth log‑sum‑exp surrogate
    of the peak of its self‑convolution.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise logits for a soft‑max representation (ensures positivity & sum=1)
    v = np.log(a + 1e-12)

    # Adam state
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    best_a = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        it += 1

        # ----- soft‑max to obtain a probability vector -----
        maxv = np.max(v)
        expv = np.exp(v - maxv)
        a = expv / expv.sum()

        # ----- convolution and smooth weights -----
        conv = _conv_self(a)
        max_c = conv.max()
        # temperature set to a modest fraction of the current peak
        tau = max_c * 0.2 + 1e-16

        # stable exponentiation
        max_conv = np.max(conv)
        w = np.exp((conv - max_conv) / tau)
        w /= w.sum()                     # now Σ w_k = 1

        # ----- gradient of the smooth loss w.r.t. a -----
        # grad_a = 2 * Σ_k w_k * (shifted copy of a)
        w_rev = w[::-1]                  # reversed weighting
        L = _next_pow2(len(a) + len(w_rev) - 1)  # length for FFT
        Af = np.fft.rfft(a, n=L)
        Wf = np.fft.rfft(w_rev, n=L)
        conv_aw = np.fft.irfft(Af * Wf, n=L)

        rev_idx = len(w) - 1            # = 2·n‑2
        # grad_i = 2 * conv_aw[rev_idx - i]  for i=0..n‑1
        grad_a = 2.0 * conv_aw[rev_idx - np.arange(n)]

        # ----- back‑propagation through the soft‑max -----
        mean_grad = np.dot(grad_a, a)
        grad_v = a * (grad_a - mean_grad)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = 0.4 / (np.sqrt(it) + 1.0)   # heuristic step‑size schedule
        v -= step * m_hat / (np.sqrt(v2_hat) + eps)

        # ----- track best -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑opt it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float):
    """
    Randomly move small packets of mass between positions.
    Simple Metropolis acceptance with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = max(1e-5, 0.1 * (0.01 / 0.1) ** (it / max_steps))

        src = rng.choice(n, p=a)          # favour larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 40 %) of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,

    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list (sum normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # tiny safety margin

    # --------------------------------------------------------------
    # 1) choose a length (power of two makes FFT cheap)
    # --------------------------------------------------------------
    n = 8192          # 2¹³ – enough resolution while staying fast

    # --------------------------------------------------------------
    # 2) build an initial pool of diverse candidates
    # --------------------------------------------------------------
    pool = []                     # list of (score, vector)
    pool_cap = 12

    def _maybe_add(vec):
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # deterministic analytic starters
    _maybe_add(_make_uniform(n))
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    for beta in (0.0, 2.0, 5.0, 8.0):
        _maybe_add(_make_kaiser(n, beta))
    _maybe_add(_make_cosine_tip(n))

    # power‑ramp family
    for p in (0.5, 1.0, 1.5, 2.0):
        for alp in (0.0, 0.1, 0.3, 0.5):
            _maybe_add(_make_power_shape(n, p, alp))

    # beta‑distribution family (grid)
    for a_par in (0.5, 1.0, 2.0, 4.0):
        for b_par in (0.5, 1.0, 2.0, 4.0):
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    # random Dirichlet draws (a variety of concentrations)
    rng = np.random.default_rng()
    for conc in (0.1, 0.3, 0.6, 1.0, 2.0):
        for _ in range(8):
            if time.time() >= deadline:
                break
            vec = rng.dirichlet(np.full(n, conc))
            _maybe_add(vec)

    # ------------------------------------------------------------------
    # 3) incorporate a global starter if present (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
                _maybe_add(starter)
    except Exception:
        pass

    # keep the best element of the pool as the current champion
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] initial pool best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) local refinement – smooth optimisation on each pool member
    # ------------------------------------------------------------------
    # give each pool entry a modest time slice (≈5 s) while we still have budget
    for _, vec in pool:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate at most 5 s per start, but not more than half of the remaining budget
        per_deadline = time.time() + min(5.0, remaining * 0.5)
        seq, sc = _smooth_optimize(vec, per_deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] after smooth‑opt new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 5) final annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, sc = _anneal(best_seq, deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improved best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 6) final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = _make_uniform(n)

    print(f"[{time.time() - start_time:.2f}s] finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
