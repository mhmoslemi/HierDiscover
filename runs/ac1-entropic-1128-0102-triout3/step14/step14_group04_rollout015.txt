"""
Gradient‑softmax optimisation of a step function.

The evaluation function is

    2·n·max(conv(a,a)) / (sum(a)**2)

with a non‑negative vector a.  Since the objective is scale‑invariant,
we normalise all candidates to sum = 1 and minimise

    2·n·max(conv(a,a))               (conv = linear self‑convolution)

A smooth approximation of the max is the soft‑max
    φβ(a) = (1/β)·log Σ exp(β·conv(a)[k]).
Its gradient can be written in closed form as a convolution,
which allows us to use an Adam optimiser together with a projection
onto the probability simplex.  The temperature β is increased
gradually, turning the smooth objective into the true max.
The routine runs for the whole 1000 s budget, starts from the
previous best construction (``height_sequence_1``), and returns the
best sequence found.

The algorithm is completely self‑contained and uses only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT of length L_fft."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    # keep only the useful part (length 2·n‑1)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (the vector is normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # --------------------------------------------------------------
    # 1) Problem size – a power of two gives cheap FFTs
    # --------------------------------------------------------------
    n = 16384                # 2¹⁴ – a good compromise between resolution and speed

    # --------------------------------------------------------------
    # 2) Initialise from the best known construction (if available)
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the competition framework.
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        # Upsample / downsample linearly to length n, preserving total mass.
        x_old = np.linspace(0.0, 1.0, init.size)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, init)
    except Exception:
        # fallback: uniform distribution
        a = np.full(n, 1.0 / n)

    # safeguard – non‑negativity and normalisation
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_a = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start_time:.2f}s] Start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Pre‑computed FFT length (used for both self‑convolution and
    #    the gradient convolution a * p)
    # --------------------------------------------------------------
    L_fft = _next_pow2(3 * n)      # >= n + (2n‑1) – 1

    # Adam parameters for the smooth optimisation
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    # temperature schedule (soft‑max concentration)
    def temperature(it):
        # start at β=5, grow roughly linearly, cap at 2000
        return min(2000.0, 5.0 + 0.1 * it)

    # --------------------------------------------------------------
    # 4) Main optimisation loop – smooth‑max → hard max
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- self‑convolution (objective) ---------------------------------
        Af = np.fft.rfft(a, n=L_fft)                 # FFT of current a
        conv_full = np.fft.irfft(Af * Af, n=L_fft)   # a*a of length 2n‑1 + padding
        conv = conv_full[: 2 * n - 1]                # keep only the useful part
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Iter {it:,} → new best {best_score:.6f}")

        # ---- soft‑max over the convolution values -------------------------
        beta = temperature(it)
        # stable soft‑max
        tmp = beta * conv
        mtmp = tmp.max()
        exp_tmp = np.exp(tmp - mtmp)
        p = exp_tmp / exp_tmp.sum()                 # length 2n‑1, non‑negative, sums to 1

        # ---- gradient of the smooth objective ----------------------------
        # gradient = 2 * (a ⊛ p)  (⊛ = linear convolution)
        Pf = np.fft.rfft(p, n=L_fft)
        grad_full = np.fft.irfft(Af * Pf, n=L_fft)   # length ≥ 3n‑2
        grad = 2.0 * grad_full[:n]                  # first n entries are the needed ones

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – a simple schedule that shrinks with √it
        step = 0.2 / (math.sqrt(it) + 1.0)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection back onto the simplex -----------------------------
        a = _project_simplex(a)

        # occasional safeguard: re‑normalise (numerical drift)
        if abs(a.sum() - 1.0) > 1e-12:
            a /= a.sum()

        # every few thousand iterations report progress (cheap)
        if it % 5000 == 0:
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] it={it:,}  cur={cur_score:.6f}  best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light polishing with the original sub‑gradient method
    #    (few iterations only – it is cheap and often gives a tiny extra gain)
    # --------------------------------------------------------------
    try:
        # reuse the fast sub‑gradient code from the reference implementation
        # (re‑implemented locally to avoid closures)
        def _subgrad_refine(a0, sub_deadline):
            a = a0.copy()
            a = _project_simplex(a)
            best = a.copy()
            best_val = _objective(a)

            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8

            it = 0
            while time.time() < sub_deadline:
                it += 1
                conv = _conv_self(a, _next_pow2(2 * n - 1))
                cur_max = conv.max()
                cur_val = 2.0 * n * cur_max
                if cur_val < best_val:
                    best_val = cur_val
                    best = a.copy()
                # active set (2 % tolerance)
                thresh = cur_max * 0.98
                ks = np.where(conv >= thresh)[0]

                grad = np.zeros_like(a)
                for k in ks:
                    lo = max(0, k - (n - 1))
                    hi = min(n - 1, k)
                    idx = np.arange(lo, hi + 1, dtype=int)
                    grad[idx] += 2.0 * a[k - idx]
                grad /= float(len(ks))

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                step = 0.5 / (np.sqrt(it) + 1.0)
                a -= step * m_hat / (np.sqrt(v_hat) + eps)
                a = _project_simplex(a)
            return best, best_val

        remaining = deadline - time.time()
        if remaining > 30.0:
            sub_dead = time.time() + 0.7 * remaining
            refined, refined_score = _subgrad_refine(best_a, sub_dead)
            if refined_score < best_score:
                best_score, best_a = refined_score, refined.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved to {best_score:.6f}")
    except Exception:
        # any problem in the polishing stage is non‑fatal
        pass

    # --------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
