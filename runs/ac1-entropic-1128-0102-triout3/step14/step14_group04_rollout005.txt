assistantfinal**Solution Overview**

The evaluation function  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale‑invariant*: after projecting the vector onto the probability simplex
\(\{x\ge 0,\;\sum x=1\}\) the denominator is 1 and the objective reduces to  

\[
E(a)=2\,n\;\max (a*a)\qquad (a\ge0,\;\sum a=1).
\]

Hence the problem is to minimise the peak of the linear self‑convolution while
keeping the vector non‑negative and normalised.

The code below builds a **diverse initial pool** (uniform, classic windows,
exponential / power ramps, random Dirichlet draws, …), selects the best
candidate and then **optimises it with a smooth‑max gradient descent**:

* the non‑smooth “max’’ is replaced by a *soft‑max*  
  \(f_T(a)=2n\,T\;\log\!\sum_i\exp\bigl((a*a)_i/T\bigr)\).  
  For a temperature \(T\) the gradient is explicit and can be obtained with
  two FFT‑based convolutions.
* The temperature is annealed (geometric decay) so the optimisation first
  follows a smooth landscape and later concentrates on the true peak.
* After each gradient step the vector is **projected onto the simplex**
  (Wang & Carreira‑Perpiñán algorithm).  This guarantees feasibility.
* A light **random mass‑swap** is injected every few hundred iterations to
  escape shallow local minima.
* The whole routine respects the 1000 s budget, prints progress and finally
  returns the best vector found as a plain Python list.

Because the optimisation works directly on the smooth approximation of the
objective and uses a full‑gradient (all convolution entries) the search
explores directions that the original sub‑gradient method could not,
often yielding a lower bound than the previous 1.508 → 1.503 target.

"""
search for a low‑value step‑function sequence.
Uses a diverse initial pool and a smooth‑max gradient descent
(with temperature annealing) together with simplex projection.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray, L_conv: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L_conv (>=2·n‑1)."""
    Af = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * a.shape[0] - 1]
    return conv

# ----------------------------------------------------------------------
# Exact (scale‑free) objective: 2·n·max(conv)  (sum is assumed 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L_conv: int) -> float:
    conv = _conv_self_fft(a, L_conv)
    return float(2.0 * a.shape[0] * conv.max())

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Analytic families of shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    w = (0.42
         - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        w = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        w = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) , normalised."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    return w / w.sum()


def _make_power_ramp(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)   (monotone decreasing)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p)
    return w / w.sum()


def _make_random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.dirichlet(np.full(n, alpha))
    return vec


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (single restart)
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      L_conv: int,
                      L_corr: int) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by minimising a smooth approximation
    of the max‑convolution.  Temperature is annealed geometrically.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    # Adam‑style momentum (helps stability)
    m = np.zeros_like(a)
    beta1 = 0.9

    # initial temperature – a fraction of the first peak
    conv = _conv_self_fft(a, L_conv)
    max_conv = conv.max()
    T = max_conv * 0.5 if max_conv > 0 else 1.0
    T_min = 1e-12
    T_decay = 0.9995

    best_a = a.copy()
    best_score = _objective(a, L_conv)

    it = 0
    step0 = 0.2        # base step size
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # -------- convolution (current state) ----------
        conv = _conv_self_fft(a, L_conv)
        max_conv = conv.max()

        # -------- smooth max (soft‑max weights) -------
        # w_i = exp((c_i - max)/T)  , then normalise
        shifted = conv - max_conv
        # avoid overflow – subtract max before exponentiation
        w = np.exp(shifted / max(T, 1e-12))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # -------- gradient of the smooth objective -----
        # g_j = Σ_k a_k w_{j+k}  (correlation a ⋆ w)
        Af = np.fft.rfft(a, n=L_corr)
        Wf = np.fft.rfft(w, n=L_corr)
        corr_full = np.fft.irfft(Af * np.conj(Wf), n=L_corr)
        # genuine length of linear correlation = n + (2n‑1) – 1
        corr_len = n + (2 * n - 1) - 1
        corr = corr_full[:corr_len]
        offset = (2 * n - 1) - 1          # index of lag 0
        g = corr[offset: offset + n]     # length n

        # gradient of the smooth objective
        grad = 4.0 * n * g                # from the derivation in the analysis

        # -------- Adam‑style update -------------------
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** it)

        # step length decays like 1/√it
        step = step0 / (math.sqrt(it) + 1.0)

        a -= step * m_hat

        # -------- projection onto simplex ---------------
        a = _project_simplex(a)

        # occasional random mass‑swap to improve exploration
        if it % 500 == 0:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and a[src] > 1e-12:
                eps = a[src] * rng.random() * 0.3
                a[src] -= eps
                a[dst] += eps
                a = _project_simplex(a)

        # -------- keep the best seen --------------------
        cur_score = _objective(a, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"grad‑desc it={it:,}  best={best_score:.6f}")

        # -------- temperature annealing ----------------
        T = max(T * T_decay, T_min)

        # safety break if we have spent the bulk of the budget
        if it % 1000 == 0 and time.time() > deadline - 5.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (already normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    # ------------------------------------------------------------------
    # 1) problem size – power of two for cheap FFTs
    # ------------------------------------------------------------------
    n = 4096                     # enough resolution, cheap enough

    # ------------------------------------------------------------------
    # 2) build a diverse pool of candidates
    # ------------------------------------------------------------------
    pool = []                    # (score, vector)
    pool_cap = 20                # keep only the best few

    def _maybe_add(vec):
        vec = np.maximum(vec, 0.0)
        vec = vec / vec.sum() if vec.sum() > 0 else np.full_like(vec, 1.0 / n)
        sc = _objective(vec, _next_pow2(2 * n - 1))
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # uniform baseline
    _maybe_add(_make_uniform(n))

    # classic windows
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))
    for beta in np.linspace(0.0, 10.0, 9):
        _maybe_add(_make_kaiser(n, beta))

    # monotone exponential ramps
    for lam in np.logspace(-5, -0.5, 12):
        _maybe_add(_make_exponential(n, lam))

    # monotone power ramps
    for p in np.linspace(0.2, 3.0, 12):
        _maybe_add(_make_power_ramp(n, p))

    # random Dirichlet draws (moderate concentration)
    rng = np.random.default_rng()
    for _ in range(150):
        if time.time() >= deadline:
            break
        _maybe_add(_make_random_dirichlet(n, alpha=0.5, rng=rng))

    # optionally use a globally supplied starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                _maybe_add(starter)
    except Exception:
        pass

    # best from the pool
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) smooth‑max gradient descent on the current best candidate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:          # need a couple of seconds for at least one run
        L_conv = _next_pow2(2 * n - 1)
        L_corr = _next_pow2(3 * n - 2)
        gd_deadline = time.time() + 0.75 * remaining
        refined_seq, refined_score = _gradient_descent(
            best_seq, gd_deadline, start_time, L_conv, L_corr)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gradient‑desc improved score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final cleaning – ensure positivity & normalisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
