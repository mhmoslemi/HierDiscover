All phases adapt the problem size to the time that is left; the algorithm
prefers a larger N when the budget permits, otherwise it works with a
smaller N (still ≥ 16384) to keep the number of FFTs cheap during the
exploration phase.  The code never uses closures, lambda‑expressions or
file‑IO and prints a short log of the current best objective.

The returned list is a normalised non‑negative step function that can be
directly fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float, plateau: float) -> np.ndarray:
    """
    Symmetric power‑ramp (x⁽ᵖ⁾ on the sides, optional flat centre).
    Parameters are chosen so that the shape sums to 1.
    """
    plateau = max(0.0, min(0.5, plateau))
    plateau_len = int(round(plateau * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n, dtype=float)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration (α)."""
    a = rng.gamma(concentration, 1.0, size=n)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Top‑K pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """
    Keep only the best `k` entries (lowest score).  The pool is sorted.
    """
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (exact ℓ∞ objective)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             log_prefix: str = "") -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 1.0                 # larger than the old 0.8 – more aggressive
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set – all lags attaining (within tolerance) the maximum
        tol = max(1e-14, max_conv * 1e-14)
        active = np.where(conv >= max_conv - tol)[0]

        # Sub‑gradient (correlation with the indicator of the active set)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Step‑size schedule (slow decay – many iterations)
        eta = eta0 / (1.0 + it / 2000.0)

        # Multiplicative update, clipped for stability
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # Renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s]{log_prefix} ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – fine polishing
# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.07               # a bit larger than the old 0.05
    T0 = 0.03
    T_end = 5e-5

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam iter={it:,} score={cur_score:.6f}")

        # temperature schedule (linear anneal)
        frac = it / (it + 2000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of the soft‑max surrogate
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = 2.0 * np.fft.irfft(np.fft.rfft(w, n=L) *
                                 np.fft.rfft(a_rev, n=L), n=L)[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing warm‑start (fast cheap moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_vec: np.ndarray,
                         deadline: float,
                         start_time: float,
                         log_prefix: str = "") -> tuple[np.ndarray, float]:
    cur_seq = start_vec.copy()
    cur_score = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    n = cur_seq.shape[0]
    rng = np.random.default_rng(int(time.time()) ^ 0x5A5A5A)

    max_move_factor = 0.25
    T_start = 0.6
    T_end = 1e-9
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * (T_end / T_start) ** frac

        # pick a source proportional to its current mass
        src = rng.choice(n, p=cur_seq)
        if cur_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue
        prop = cur_seq.copy()
        prop[src] -= delta
        prop[dst] += delta

        new_score = _objective(prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_seq = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s]{log_prefix} SA iter={it:,} new best={new_score:.6f}")

        # occasional restart from a random pool entry (handled outside)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size – larger N gives finer resolution but costs more.
    # --------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 131072                     # try to use the full power of the budget
    elif remaining > 400:
        N = 65536
    else:
        N = 32768

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool on a *small* grid (fast evaluation)
    # --------------------------------------------------------------
    SMALL_N = max(16384, N // 8)        # cheap evaluations
    TOP_K = 12
    top = []                           # (score, vector) – best first

    def _add_small(vec: np.ndarray) -> None:
        # always project onto the simplex before scoring
        vec = _project_simplex(vec)
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform
    _add_small(np.full(SMALL_N, 1.0 / SMALL_N))

    # symmetric Beta – dense log‑grid
    for alpha in np.geomspace(0.03, 2.0, 30):
        _add_small(_make_beta_symmetric(SMALL_N, float(alpha)))

    # asymmetric Beta – explore a wide rectangle
    alphas = np.geomspace(0.03, 2.0, 12)
    betas = np.geomspace(0.03, 2.0, 12)
    for a in alphas:
        for b in betas:
            _add_small(_make_beta_asym(SMALL_N, float(a), float(b)))

    # power‑ramps
    for p in np.linspace(0.1, 4.0, 12):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add_small(_make_power_ramp(SMALL_N, float(p), float(plateau)))

    # random Dirichlet vectors – various concentrations
    rng = np.random.default_rng(seed + 7777)
    for conc in np.geomspace(0.05, 5.0, 15):
        for _ in range(25):
            vec = _make_random_dirichlet(SMALL_N, float(conc), rng)
            _add_small(vec)

                x_new = np.linspace(0, 1, SMALL_N, endpoint=False)
                starter_small = np.interp(x_new, x_old, starter)
            else:
                starter_small = starter.copy()
            starter_small = np.maximum(starter_small, 0.0)
            if starter_small.sum() > 0:
                starter_small /= starter_small.sum()
                _add_small(starter_small)
    except Exception:
        pass

    best_score, best_vec_small = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Upsample the best small vectors to the full size N
    # --------------------------------------------------------------
    def _upsample(vec_small: np.ndarray) -> np.ndarray:
        if N == vec_small.size:
            return vec_small.copy()
        x_old = np.linspace(0, 1, vec_small.size, endpoint=False)
        x_new = np.linspace(0, 1, N, endpoint=False)
        up = np.interp(x_new, x_old, vec_small)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0:
            up /= s
        else:
            up = np.full(N, 1.0 / N)
        return up

    # Upsample all pool entries
    top_full = []
    for score, vec_small in top:
        vec_full = _upsample(vec_small)
        _push_top(top_full, _objective(vec_full), vec_full, TOP_K)

    best_score, best_vec = top_full[0][0], top_full[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Upsampled best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing warm‑start (≈25 % of the remaining budget)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + (deadline - time.time()) * 0.25
    cur_vec, cur_score = _simulated_annealing(best_vec, phase1_deadline, start_time,

    best_vec, best_score = cur_vec.copy(), cur_score

    # occasional random restarts from the pool while SA is active
    rng = np.random.default_rng(seed + 999)

    # --------------------------------------------------------------

        # start from the SA result
        vec, sc = _exp_subgradient_descent(best_vec, sub_deadline, start_time,

        if sc < best_score:
            best_vec, best_score = vec, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD from SA improved best={sc:.6f}")

        # also try the other pool members while time permits
        for idx, (_, pool_vec) in enumerate(top_full):
            if time.time() >= sub_deadline:
                break
            vec, sc = _exp_subgradient_descent(pool_vec, sub_deadline, start_time,
                                               log_prefix=f" (ExpGD‑pool{idx}) ")

                best_vec, best_score = vec, sc
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 5) Final smooth‑max Adam polishing (remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        vec, sc = _smooth_adam_descent(best_vec, deadline, start_time)
        if sc < best_score:
            best_vec, best_score = vec, sc
            print(f"[{time.time() - start_time:.2f}s] Adam polishing improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 6) Final clean‑up and conversion to plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
