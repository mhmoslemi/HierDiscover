"""
Propose a low‑score step function.

The routine builds a large, diverse pool of analytic shapes (exponential,
power‑law, double‑power, beta, windows, random Dirichlet draws) for a
moderate length n (4096).  All candidates are evaluated with the exact
objective 2·n·max(conv(a,a))/sum(a)² (scale‑invariant) using an FFT‑based
self‑convolution.

The best pool members are refined independently by a short Adam‑style
projected gradient descent on a smoothed max‑convolution (log‑sum‑exp)
and by a light simulated‑annealing “mass‑swap’’ polish.  The whole
search runs within the 1000 s budget and returns the best sequence found
as a plain list of non‑negative floats (already normalised to sum = 1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  No safety checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-14:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = _make_uniform(n)
    return a


def _make_power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = _make_uniform(n)
    return a


def _make_double_power(n: int, p_left: float, p_right: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p_left) * (n-i)^(-p_right).
    Asymmetric two‑sided power law.
    """
    i = np.arange(n, dtype=float)
    left = (i + 1.0) ** (-p_left)
    right = (n - i) ** (-p_right)
    a = left * right
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = _make_uniform(n)
    return a


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1}."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_dpss(n: int, nw: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        vec = windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        vec = np.maximum(vec, 0.0)
        vec /= vec.sum()
        return vec
    except Exception:
        return _make_uniform(n)


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_objective(a: np.ndarray, alpha: float) -> float:
    """
    Returns 2·n·S where S = (1/α)·log∑_k exp(α·conv_k).  a is assumed to sum to 1.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    # use log‑sum‑exp in a numerically stable way
    max_c = conv.max()
    s = np.exp(alpha * (conv - max_c)).sum()
    smooth_max = max_c + np.log(s) / alpha
    return 2.0 * n * smooth_max


def _smooth_grad(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t a (assuming Σa=1, no projection).
    """
    n = a.shape[0]
    conv = _conv_self(a)

    # softmax weights for each conv entry
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()          # shape (2n‑1,)

    # Build the matrix Q where Q_{i,j} = w_{i+j}.
    # Gradient = 2 * Q·a  (see analysis in the docstring above).
    # Efficiently compute Q·a via convolution with an offset.
    # Place w into a longer array shifted by (n‑1) so that
    #   (a * w_shifted)[n‑1 + i] = Σ_j w_{i+j} a_j.
    L = _next_pow2(3 * n)          # enough for a (len n) + w (len 2n‑1)
    a_pad = np.zeros(L)
    w_pad = np.zeros(L)
    a_pad[:n] = a
    start = n - 1
    w_pad[start:start + 2 * n - 1] = w
    Af = np.fft.rfft(a_pad)
    Wf = np.fft.rfft(w_pad)
    conv_full = np.fft.irfft(Af * Wf, n=L)
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Short Adam‑style projected gradient descent on the smoothed objective
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    alpha: float = 15.0,
                    step0: float = 0.5,
                    max_iter: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Starting from a0 (already on the simplex) perform Adam‑style
    updates on the smoothed objective.  The routine stops at `deadline`
    or after `max_iter` iterations, whichever comes first.
    Returns the best vector seen and its *exact* objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        grad = _smooth_grad(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto simplex (maintains sum = 1)
        a = _project_simplex(a)

        # Record improvement
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s]   smooth‑descent it={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly move a small packet of mass between two positions,
    accepting moves with a Metropolis rule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Linear cooling schedule (very fast)
        T = max(1e-4, 0.1 * (0.01 / 0.1) ** (it / max_steps))

        src = rng.choice(n, p=a)          # more mass → higher chance to be source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # Transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s]   anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a length – power of two for cheap FFTs
    # --------------------------------------------------------------
    n = 4096                     # comfortable to evaluate many candidates

    # --------------------------------------------------------------
    # 2) Populate a diverse pool of candidate vectors
    # --------------------------------------------------------------
    pool_cap = 30                # keep only the best few
    pool = []                    # list of (score, vector)

    def _maybe_add(vec):
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # analytic families -------------------------------------------------
    _maybe_add(_make_uniform(n))
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))

    for beta in np.linspace(0.0, 10.0, 9):
        _maybe_add(_make_kaiser(n, beta))

    _maybe_add(_make_cosine_tip(n))

    # exponential decay -------------------------------------------------
    for lam in np.linspace(0.0, 4.0, 81):
        _maybe_add(_make_exponential(n, lam))

    # pure power law ----------------------------------------------------
    for p in np.linspace(0.0, 4.0, 81):
        _maybe_add(_make_power_decay(n, p))

    # double‑power (asymmetric) ----------------------------------------
    p_grid = np.linspace(0.0, 2.0, 21)
    for pL in p_grid:
        for pR in p_grid:
            _maybe_add(_make_double_power(n, pL, pR))

    # beta distribution (skewed) ----------------------------------------
    a_vals = np.logspace(-1.5, 1.5, 25)   # ~0.03 … 30
    b_vals = np.logspace(-1.5, 1.5, 25)
    for a_par in a_vals:
        for b_par in b_vals:
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    # DPSS (Slepian) windows --------------------------------------------
    for nw in [2.0, 3.0, 4.0, 5.0]:
        _maybe_add(_make_dpss(n, nw))

    # Random Dirichlet draws ---------------------------------------------
    rng = np.random.default_rng()
    for _ in range(2000):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.05, 3.0)                     # concentration parameter
        vec = rng.dirichlet(alpha=np.full(n, conc))
        _maybe_add(vec)

    # --------------------------------------------------------------
    # 3) Report best of the initial pool
    # --------------------------------------------------------------
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Multi‑start smoothing‑descent on the top candidates
    # --------------------------------------------------------------
    # Take the top few (up to 8) distinct vectors for local refinement.
    top_candidates = [vec for _, vec in pool[:8]]

    for idx, init_vec in enumerate(top_candidates):
        remaining = deadline - time.time()
        if remaining < 5.0:          # not enough time for another run
            break
        run_deadline = time.time() + min(10.0, remaining * 0.3)   # short bursts
        refined_vec, refined_score = _smooth_descent(
            init_vec, run_deadline, start_time, alpha=15.0,
            step0=0.6, max_iter=8000)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_vec.copy()
            print(f"[{time.time() - start_time:.2f}s]  Refined candidate #{idx+1} improved to {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final annealing polish on the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        ann_seq, ann_score = _anneal(best_seq, anneal_deadline, start_time)
        if ann_score < best_score:
            best_score, best_seq = ann_score, ann_seq.copy()
            print(f"[{time.time() - start_time:.2f}s]  Anneal improvement to {ann_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = _make_uniform(n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
