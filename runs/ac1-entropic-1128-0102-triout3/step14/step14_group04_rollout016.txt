"""
Improved search for a low‑score step‑function.

The algorithm builds a small, diverse pool of candidates (uniform,
classical windows, a global starter if available and many random Dirichlet
samples).  The best pool member is refined by a projected sub‑gradient
descent that uses a tighter active‑set tolerance (0.5 %) and a larger
Adam‑style step‑size schedule.  After the gradient phase a focused
simulated‑annealing polish moves tiny packets of mass between indices.
All objective calculations during the search use a fast FFT‑based
convolution (identical to the one in the reference code) whereas the
final score is verified with the official ``evaluate_sequence`` routine.
The routine respects the 1000 s budget and always returns a valid
non‑negative list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (exact up to floating point)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast objective used during optimisation (identical to the reference)
# ----------------------------------------------------------------------
def _fast_objective(v: np.ndarray) -> float:
    """2·n·max(conv(v,v)) / (sum(v)**2)  (no safety checks)."""
    n = v.shape[0]
    conv = _conv_self(v)
    max_conv = conv.max()
    s = v.sum()
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Exact objective (the one the grader will call)
# ----------------------------------------------------------------------
def _true_objective(v: np.ndarray) -> float:
    """Call the provided ``evaluate_sequence`` on a plain list."""
    return evaluate_sequence([float(x) for x in v])


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Wang & Carreira‑Perpiñán – O(n) simplex projection."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes (a subset – enough for a diverse pool)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    a = 1.0 + np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps with a flat centre – normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau = int(round(alpha * n))
    if plateau % 2 == 1 and plateau > 0:
        plateau -= 1
    side = (n - plateau) // 2
    if side <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side + 1, dtype=float) / side
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Adam momentum, tighter tolerance)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.005,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """Refine `a0` (already on the simplex) by projected sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    a = _project_simplex(a)

    best = a.copy()
    best_score = _fast_objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    step0 = 0.7

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.1f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        # active set – either top‑k or within the tolerance band
        if top_k is not None and top_k < conv.size:
            active = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max‑conv (average over active indices)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(active))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # back onto the simplex
        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Light Simulated‑Annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny packets of mass; simple exponential cooling."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _fast_objective(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (fast exponential decay)
        T = max(1e-4, 0.2 * (0.05 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)          # larger entries are more likely sources
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a small fraction of the source mass (up to 30 %)
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _fast_objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.1f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine builds a small pool of candidates,
    refines the best one with sub‑gradient descent (tight tolerance) and
    finishes with a short simulated‑annealing polish, all respecting the
    supplied time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # Choose a length – a power of two keeps FFT cheap
    # --------------------------------------------------------------
    n = 8192                                 # 2¹³ – enough resolution, still fast

    # --------------------------------------------------------------
    # Build a first pool of diverse candidates
    # --------------------------------------------------------------
    pool = []                                 # list of (score, vector)
    pool_cap = 15

    def _add(vec):
        sc = _fast_objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # uniform baseline
    _add(_make_uniform(n))

    # classic windows
    _add(_make_hann(n))
    _add(_make_cosine_tip(n))

    # power‑ramp family (explore a few settings)
    for p in (0.3, 0.6, 1.0, 1.5, 2.0):
        for a in (0.0, 0.2, 0.4):
            _add(_make_power_shape(n, p, a))

    # beta‑distribution family
    for a_par in (0.5, 2.0, 5.0):
        for b_par in (0.5, 2.0, 5.0):
            _add(_make_beta_shape(n, a_par, b_par))

    # try to utilise the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            # put it onto the same length (pad / truncate) – this keeps the search simple
            if starter.size < n:
                pad = np.zeros(n - starter.size)
                starter = np.concatenate([starter, pad])
            elif starter.size > n:
                starter = starter[:n]
            starter = _project_simplex(starter)
            _add(starter)
    except Exception:
        pass

    # a few random Dirichlet draws (moderately concentrated)
    rng = np.random.default_rng()
    for _ in range(250):
        if time.time() >= deadline:
            break
        vec = rng.dirichlet(alpha=np.ones(n) * 0.3)
        _add(vec)

    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient refinement (the heavy part)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:                     # leave some time for the annealer
        sub_deadline = time.time() + 0.80 * remaining
        refined, refined_score = _subgradient_descent(
            best_seq,
            sub_deadline,
            start,
            tol=0.005,               # tighter active‑set tolerance
            top_k=None)              # use all near‑max entries
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[{time.time() - start:.2f}s] After sub‑grad best = {refined_score:.6f}")

    # --------------------------------------------------------------
    # Final annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _anneal(best_seq, deadline, start)
        if final_score < best_score:
            best_score, best_seq = final_score, final_seq.copy()
            print(f"[{time.time() - start:.2f}s] After anneal best = {final_score:.6f}")

    # --------------------------------------------------------------
    # Verify with the true objective (just for information)
    # --------------------------------------------------------------
    true_score = _true_objective(best_seq)
    print(f"[{time.time() - start:.2f}s] True score (evaluate_sequence) = {true_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and output a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
