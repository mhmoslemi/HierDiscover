        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1, dtype=int)

"""Efficient mixed LP + projected sub‑gradient optimisation for the
step‑function bound.

The search works in two stages.

* Stage 1 – a low‑dimensional linear programme.
  For a modest length n₀ (≈ 1 024) we repeatedly
  – compute M = max( (f∗f) )
  – solve the LP   maximise Σgⱼ   s.t. f∗g ≤ M, g≥0
    (the constraints are linear because the convolution with f is a Toeplitz
    matrix).  The LP pushes mass onto indices where f is small.
  – mix the current f with the LP‑solution g and project back onto the
    probability simplex.
  After a few dozen iterations the vector f is already far from the
    uniform distribution and has a much smaller ∞‑norm of the
    self‑convolution.

* Stage 2 – up‑sample to a large size (≈ 16 384) and polish
  with the projected sub‑gradient method used in the reference code.
  A decreasing tolerance on the active set together with a modest
  top‑k selection makes the descent fast and accurate.
  Finally a short simulated‑annealing phase swaps tiny packets of mass
  between positions.

All operations respect the global 1000 s time budget.
The returned list is normalised to sum = 1 (the evaluator rescales it
automatically)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (linear self‑convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑free, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)   (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (kept for diversity of seeds)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with optional top‑k active set)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    Active set: either all indices with conv ≥ (1‑tol)·max or the `top_k`
    largest entries.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam‑style momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)      # sum(a)=1 → denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # Determine active indices
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        # Gradient of the (averaged) maximal convolution entry
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of mass between positions,
    accepting according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 1_000_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Simple exponential cooling schedule
        T = max(5e-5, 0.2 * (0.05 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)          # source tends to be a heavy index
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step (small n)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float) -> np.ndarray:
    """Given f and a bound M, solve  maximise Σg  subject to f∗g ≤ M, g≥0."""
    try:
        from scipy.optimize import linprog
    except Exception:
        # If SciPy is unavailable, just return the input unchanged.
        return f

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz matrix representing the linear convolution f∗g.
    # Row k contains the coefficients f_{k‑j} for indices j that fit.
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        A[k, idx] = f[k - idx]

    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g  → minimise − Σ g
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'disp': False})
    if not res.success:
        # fallback – keep the current vector
        return f

    g0 = res.x
    if g0.sum() <= 0.0:
        return f
    g = g0 / g0.sum()
    return g

# Multi‑stage optimiser
# ----------------------------------------------------------------------

      A decreasing tolerance on the active set together with a modest

    • Finally perform a short simulated‑annealing phase that swaps tiny
      packets of mass between positions.

All operations respect the 1000 s global budget.  The returned list is
normalised to sum = 1 (the evaluator rescales it automatically)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – linear self‑convolution (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑free, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)   (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (used as seeds for the small‑n pool)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (optional top‑k active set)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    Active set = either all conv entries ≥ (1−tol)·max, or the `top_k`
    largest entries.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam‑style momentum parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # ---- active set -------------------------------------------------
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        # ---- gradient of the averaged active entry ----------------------
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection -------------------------------------------------
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of mass between positions,
    accepting according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 1_000_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = max(5e-5, 0.2 * (0.05 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)          # source favours large entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step (small n)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float) -> np.ndarray:
    """
    Given a probability vector f and a bound M, solve
        maximise Σ g_j   subject to  f∗g ≤ M ,  g ≥ 0 .
    The convolution f∗g is linear in g, therefore the constraints form a
    standard linear programme.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        # SciPy unavailable – fall back to the current vector.
        return f

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz matrix representing the linear convolution f∗g.
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        A[k, idx] = f[k - idx]

    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g → minimise –Σ g
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'disp': False})
    if not res.success:
        return f

    g0 = res.x
    if g0.sum() <= 0.0:
        return f
    g = g0 / g0.sum()
    return g

# ----------------------------------------------------------------------

    """

    deadline = start_time + budget_s - 0.5   # safety margin

    # Stage 1 – low‑dimensional LP optimisation (n₀ ≈ 1024)
    # ------------------------------------------------------------------

    pool = [_make_hann(n_small), _make_hamming(n_small),
            _make_blackman(n_small), _make_cosine_tip(n_small)]
    rng = np.random.default_rng()
    for _ in range(20):
        pool.append(rng.dirichlet(alpha=np.ones(n_small) * 0.4))

    best_small_score = np.inf
    best_small_vec = None
    for vec in pool:
        vec = _project_simplex(vec)
        sc = _objective(vec)
        if sc < best_small_score:
            best_small_score, best_small_vec = sc, vec.copy()
    f = best_small_vec
    print(f"[{time.time() - start_time:.2f}s] Small‑size start score = {best_small_score:.6f}")

    # iterative LP improvement – generous share of the time budget
    lp_deadline = deadline - 30.0          # reserve ~30 s for the fine phase
    lp_iter = 0
    best_lp_score = best_small_score
    best_lp_vec = f.copy()

        g = _lp_improve(f, M)
        t = 0.2 / (1 + lp_iter)            # decreasing mixing factor

        f = _project_simplex(f)

        cur_score = _objective(f)
        if cur_score < best_lp_score:
            best_lp_score, best_lp_vec = cur_score, f.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={lp_iter:3d} score={cur_score:.6f}")

        lp_iter += 1

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to a large length and polish
    # ------------------------------------------------------------------
    N = 16384                                 # final resolution (2¹⁴)
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, N)
    a = np.interp(x_big, x_small, best_lp_vec)
    a = np.maximum(a, 0.0)

                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                if sc < _objective(a):
                    a = starter.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase A – sub‑gradient descent with a decreasing tolerance schedule
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + 0.70 * remaining

        # (fraction of remaining, tolerance, top_k) schedule
        schedule = [

            (0.30, 0.02, 200),
            (0.40, 0.005, 400),
        ]

        cur_vec = a.copy()
        cur_best_score = _objective(cur_vec)
        cur_best_vec = cur_vec.copy()

        for frac, tol, topk in schedule:
            phase_deadline = time.time() + (sub_deadline - time.time()) * frac
            cur_vec, cur_score = _subgradient_descent(cur_vec,
                                                      phase_deadline,
                                                      start_time,
                                                      tol=tol,
                                                      top_k=topk)
            if cur_score < cur_best_score:
                cur_best_score, cur_best_vec = cur_score, cur_vec.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad (tol={tol}) best={cur_score:.6f}")

        a = cur_best_vec
        best_score = cur_best_score
    else:
        best_score = _objective(a)

    # ------------------------------------------------------------------
    # Phase B – final annealing polish (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_vec, anneal_score = _anneal(a, deadline, start_time)
        if anneal_score < best_score:

    # Normalisation & conversion to plain Python list
    # ------------------------------------------------------------------

    total = a.sum()
    if total > 0.0:
        a /= total
    else:

    return [float(v) for v in a]
