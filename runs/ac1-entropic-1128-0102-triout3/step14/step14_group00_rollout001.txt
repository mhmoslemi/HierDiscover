def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # reverse a
    rev_a = a[::-1]
    L = _next_pow2(3*n - 2)
    Af = np.fft.rfft(rev_a, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(Af * Wf, n=L)[:3*n-2]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad
```

def _project_simplex(v):
    # from https://arxiv.org/pdf/1101.6081.pdf
    # ensure nonnegative and sum to 1
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score non‑negative step function.

Algorithm
---------

1. **Initial pool** – generate a few analytic candidates (uniform,
   random beta shapes, Dirichlet draws, an up‑sampled global starter if
   available). Keep the best one (lowest objective).

2. **Smooth max‑gradient descent** – minimise the scale‑invariant objective
   `2·n·max(conv(a,a))` by gradient descent on the probability simplex.
   The non‑smooth max is approximated by a soft‑max with a temperature
   that slowly decays.  Adam‐type adaptive steps are used and after each
   update the vector is projected back onto the simplex.  A symmetry
   trick (`a ← (a+rev(a))/2`) keeps the distribution mirror‑symmetric,
   which is natural for the convolution problem.

3. **Stagnation handling** – if no improvement is seen for many
   iterations a small random perturbation is added to escape local minima.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain list of floats.  All values are
non‑negative and sum to 1, which satisfies the security checks of
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length L_self)."""
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * a.shape[0] - 1]
    return conv


def _grad_from_weights(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max weighted max‑convolution.
    Returns 2 * sum_k w_k * a_{k-i} for each i.
    Implemented as convolution of reversed a with w.
    """
    rev_a = a[::-1]                                   # shape (n,)
    Af = np.fft.rfft(rev_a, n=L_grad)
    Wf = np.fft.rfft(w, n=L_grad)
    conv_rev = np.fft.irfft(Af * Wf, n=L_grad)[: 3 * a.shape[0] - 2]
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L_self: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (already normalised, sum=1)."""
    n = a.shape[0]
    conv = _conv_self(a, L_self)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta‑distribution on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape[:] = 1.0 / n
    return shape


def _upsample(src: np.ndarray, n: int) -> np.ndarray:
    """Linear up‑sampling of a 1‑D probability vector to length n."""
    src = np.asarray(src, dtype=float)
    src = np.maximum(src, 0.0)
    if src.sum() == 0.0:
        src[:] = 1.0 / src.size
    else:
        src /= src.sum()
    m = src.size
    src_x = np.linspace(0.0, 1.0, m)
    tgt_x = (np.arange(n) + 0.5) / n
    up = np.interp(tgt_x, src_x, src)
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum > 0.0:
        up /= up_sum
    else:
        up[:] = 1.0 / n
    return up


# ----------------------------------------------------------------------
# Core optimiser (soft‑max + Adam on the simplex)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        L_self: int,
                        L_grad: int,
                        enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam‑type optimisation of the max‑convolution objective.
    Returns best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Learning‑rate schedule (initially large, decays like 1/√t)
    lr0 = 10.0

    # Temperature schedule factors
    temp_factor_start = 0.2   # start with 20 % of current max value
    temp_decay_rate = 0.9995  # per iteration decay

    best_a = a.copy()
    best_score = _objective(a, L_self)

    it = 0
    no_improve = 0
    # thresholds for stagnation handling
    STAGNATE_ITERS = 20000
    MAX_NOIMP = 50000

    while time.time() < deadline:
        it += 1

        # --- current convolution and soft‑max weights ---
        conv = _conv_self(a, L_self)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        # keep the best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  score={cur_score:.6f}")
        else:
            no_improve += 1

        # temperature (decays multiplicatively)
        temp = max(1e-12, max_conv * temp_factor_start)
        temp_factor_start *= temp_decay_rate

        # stable soft‑max
        y = conv / temp
        y_max = np.max(y)
        w = np.exp(y - y_max)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed max ---
        grad = _grad_from_weights(a, w, L_grad)

        # --- Adam update ---
        lr = lr0 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce non‑negativity & simplex
        a = _project_simplex(a)

        # optional symmetry enforcement (helps a lot for convolution problems)
        if enforce_sym:
            a = 0.5 * (a + a[::-1])
            a = _project_simplex(a)

        # stagnation handling – inject a tiny random jitter if we are stuck
        if no_improve >= STAGNATE_ITERS:
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.02, size=n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            if enforce_sym:
                a = 0.5 * (a + a[::-1])
                a = _project_simplex(a)
            no_improve = 0

        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score

        v /= v.sum()
        pool.append((_objective(v, L_self), v.copy()))

    # 4) Random Dirichlet draws (highly sparse or smooth)
    for _ in range(10):
        alpha = rng.uniform(0.1, 2.0, size=n)
        v = rng.gamma(shape=alpha, scale=1.0)
        v_sum = v.sum()
        if v_sum > 0.0:
            v /= v_sum
        else:
            v[:] = 1.0 / n
        pool.append((_objective(v, L_self), v.copy()))

    # 5) Upsampled global starter (if the variable exists)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                up = _upsample(starter, n)
                pool.append((_objective(up, L_self), up.copy()))
                print(f"[{time.time() - start_time:.2f}s] Added up‑sampled global starter.")
    except Exception:
        pass

    # Keep the best few candidates (here just the top one)
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smooth‑max Adam descent on the current best vector
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:  # ensure we have enough time for a meaningful run
        eg_deadline = time.time() + 0.85 * remaining
        best_seq, best_score = _smooth_max_descent(best_seq,
                                                   eg_deadline,
                                                   start_time,
                                                   L_self,
                                                   L_grad,
                                                   enforce_sym=True)
        print(f"[{time.time() - start_time:.2f}s] After optimisation best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
