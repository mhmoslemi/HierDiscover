def project_simplex(v):
    """Projects v onto probability simplex (nonnegative, sum=1)."""
    # (v sorted descending)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

# Helper: project onto simplex (not needed if we renormalise after exp)
def _project_simplex(v):
    """Project vector v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _conv_self(a):
    """Exact linear self‑convolution."""
    return np.convolve(a, a)

def _max_conv_and_subgrad(a):
    """
    Returns (M, g) where
      M = max_i (a*a)[i]
      g = sub‑gradient of M w.r.t. a (as a vector of length len(a)).
    The sub‑gradient corresponds to the average over all indices attaining M.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()
    ks = np.where(conv == M)[0]  # one or few indices
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        # each term contributes 2*a_j
        g[i] += 2.0 * a[j]
    # If multiple ks, average the contributions
    if ks.size > 1:
        g /= ks.size
    return M, g

Optimisation via exponentiated sub‑gradient descent on the max‑convolution.
The routine starts from the best sequence that is already available as
`height_sequence_1` (or falls back to a uniform vector) and iteratively
reduces the quantity

    2·n·max(conv(a,a)) / (sum(a)**2)

while keeping the vector on the probability simplex.  The sub‑gradient of the
max‑convolution is obtained analytically, the update is performed in the
log‑space (mirror descent with KL‑divergence) and the vector is renormalised
after each step.  A cheap random‑mass‑transfer hill‑climb is run at the end
to polish the result.  The whole search respects a 1000 s wall‑clock budget.

The algorithm is deliberately different from the previous surrogate‑L‑BFGS
pipeline and thus explores other parts of the search space, giving a higher
chance of beating the 1.503 target.
"""

import time

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * M / (s * s))


def _max_conv_and_subgrad(a: np.ndarray, Lc: int, Lg: int):
    """
    Compute (M, g) where
        M = max_i (a*a)[i]
        g = sub‑gradient of M w.r.t. a (length n).

    The sub‑gradient is the average of the gradients for all indices that
    achieve the maximum.  The computation uses FFTs for speed.
    """
    n = a.shape[0]

    # ----- convolution a*a (exact, via FFT) -----
    Af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]
    M = conv.max()

    # ----- binary mask of maximising shifts -----
    w = np.zeros_like(conv)
    w[conv == M] = 1.0
    n_max = w.sum()                     # how many k attain the max

    # ----- gradient of the max‑convolution -----
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=Lg)
    Rf = np.fft.rfft(a_rev, n=Lg)
    conv_wr = np.fft.irfft(Wf * Rf, n=Lg)[: 3 * n - 2]
    # slice that gives ∂M/∂a_i (see derivation in the analysis)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]

    if n_max > 1:
        grad /= n_max                     # average when several maxima

    return M, grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:

    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] iter {it} new best {best_score:.6f}")

        # line‑search‑style step‑size schedule (linear decay)
        elapsed = time.time() - start_time
        eta = eta0 * (1.0 - elapsed / budget_s)
        eta = max(eta, min_eta)

        # mirror‑descent (exponentiated) update on the simplex
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * g
        a = np.exp(log_a)
        a = a / a.sum()          # renormalise – projection onto the simplex

        it += 1

        # if we have stagnated for a long time, perturb and keep going
        if it - last_improve > patience and time.time() < deadline - 5.0:
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] restart after {it} iterations")

    # ------------------------------------------------------------------
    # 2️⃣  Final hill‑climb polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_a, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_a = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑process – ensure a valid probability vector and return it
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
