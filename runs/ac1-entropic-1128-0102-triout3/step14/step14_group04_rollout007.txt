def _smooth_gradient_descent(a0, deadline, start_time, alpha=80.0, step0=0.5):
    a = a0.copy()
    n = a.shape[0]
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _objective(a)
    # Precompute FFT length
    L = _next_pow2(3 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        # Forward FFT of a
        Af = np.fft.rfft(a, n=L)

        # Self-convolution to compute conv (for evaluation)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # Compute smoothed weights w
        diff = conv - max_conv  # max diff <=0
        # Exponential trick for stability
        w_raw = np.exp(alpha * diff)
        Z = w_raw.sum()
        w = w_raw / Z

        # Compute FFT of w_rev
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, n=L)

        # Cross-correlation (a * w_rev)
        corr_full = np.fft.irfft(Af * W, n=L)

        # Gradient = 2 * sum_j a_j w_{i+j} = 2 * corr slice reversed
        slice_ = corr_full[n - 1 : n - 1 + n]
        grad = 2.0 * slice_[::-1]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto simplex
        a = _project_simplex(a)

        # Evaluate current true objective (if time to report)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent it={it} best={cur_score:.6f}")

    return best_a, best_score

"""
Search for a low‑score non‑negative step function.

The algorithm follows three stages:

1. Build a diverse pool of candidate shapes (windows, beta / power ramps,
   arcsine, DPSS, random Dirichlet draws, …) and keep the best few.
2. Refine the best candidates by a smooth gradient descent.  The max‑convolution
   is approximated with a log‑sum‑exp (parameter ``alpha``) which yields a
   differentiable surrogate.  Gradients are obtained efficiently with FFT‑based
   convolutions and the iterates are projected onto the probability simplex.
3. A light simulated‑annealing “mass‑swap’’ polish finishes the search.

All work respects the prescribed time budget (default 1000 s).  The returned
list is normalised to sum = 1, i.e. the objective that the judge evaluates is
``2·n·max(conv)``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  The vector is assumed non‑negative."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two symmetric power ramps with a flat centre."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) on (0,1)."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_dpss(n: int, nw: float) -> np.ndarray:
    try:
        from scipy.signal import windows
        vec = windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        vec = np.maximum(vec, 0.0)
        vec /= vec.sum()
        return vec
    except Exception:
        return _make_uniform(n)


def _make_arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the classic arcsine Law (U‑shaped)."""
    xs = (np.arange(n) + 0.5) / n
    shape = 1.0 / np.sqrt(xs * (1.0 - xs))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed gradient descent on the log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             alpha: float = 80.0,
                             step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected (Adam‑style) descent on
        L(a) = max(conv) + log‑sum‑exp(alpha·(conv‑max(conv))) / alpha
    The true objective is 2·n·max(conv) (the denominator is 1 after simplex
    projection).  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective(a)

    # pre‑compute FFT length for all convolutions
    L = _next_pow2(3 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1

        # Forward FFT of the current iterate
        Af = np.fft.rfft(a, n=L)

        # Self‑convolution (needed for the weight construction)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # Log‑sum‑exp weights (stabilised by subtracting max)
        diff = conv - max_conv          # ≤ 0
        w_raw = np.exp(alpha * diff)    # safe, max entry = 1
        Z = w_raw.sum()
        w = w_raw / Z

        # Cross‑correlation of a with w (gradient of the smoothed max)
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, n=L)
        corr_full = np.fft.irfft(Af * W, n=L)

        # Extract the relevant n entries and reverse to obtain grad_i
        slice_ = corr_full[n - 1 : n - 1 + n]
        grad = 2.0 * slice_[::-1]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex (keeps sum = 1)
        a = _project_simplex(a)

        # Periodically evaluate the true objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent it={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Randomly move a small packet of mass between two indices,
    accepting according to a simple exponential cooling schedule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Linear cooling (temperature never below 1e-5)
        T = max(1e-5, 0.2 * (0.01 / 0.2) ** (it / max_steps))

        src = rng.choice(n, p=a)          # source favours large entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # Move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (probability‑simplex normalised).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1) Choose a length – a power of two gives cheap FFTs
    # --------------------------------------------------------------
    n = 16384  # 2¹⁴ – good trade‑off between flexibility and speed

    # --------------------------------------------------------------
    # 2) Build a diverse initial pool
    # --------------------------------------------------------------
    pool = []                     # list of (score, vector)
    pool_cap = 20

    def _maybe_add(vec: np.ndarray):
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # classic deterministic families
    _maybe_add(_make_uniform(n))
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))
    for beta in np.linspace(0.0, 12.0, 7):
        _maybe_add(_make_kaiser(n, beta))
    _maybe_add(_make_cosine_tip(n))
    _maybe_add(_make_arcsine(n))

    # power‑ramp family
    for p in np.linspace(0.2, 3.0, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            _maybe_add(_make_power_shape(n, p, alpha))

    # beta‑distribution grid (including U‑shaped region)
    for a_par in np.linspace(0.05, 8.0, 9):
        for b_par in np.linspace(0.05, 8.0, 9):
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    # DPSS (Slepian) sequences
    for nw in [2.0, 3.0, 4.0, 5.0]:
        _maybe_add(_make_dpss(n, nw))

    # --------------------------------------------------------------
    # 3) Randomised candidates (beta, power ramps, Dirichlet)
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # random beta shapes, favour extreme (U‑shaped) parameters
    for _ in range(120):
        if time.time() >= deadline:
            break
        a_par = 10 ** rng.uniform(-2.0, 0.9)   # roughly 0.01 … 8
        b_par = 10 ** rng.uniform(-2.0, 0.9)
        _maybe_add(_make_beta_shape(n, a_par, b_par))

    # random power‑ramp shapes
    for _ in range(30):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.2, 4.0)
        alpha = rng.uniform(0.0, 0.5)
        _maybe_add(_make_power_shape(n, p, alpha))

    # random Dirichlet draws (low concentration → spiky, high → flat)
    for _ in range(150):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.05, 0.7)
        vec = rng.dirichlet(alpha=np.full(n, conc))
        _maybe_add(vec)

    # --------------------------------------------------------------
    # 4) Global starter, if the platform supplies one
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                _maybe_add(starter)
    except Exception:
        pass

    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Smooth gradient descent from the (up to) three best pool members
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        top_candidates = pool[:3]                     # best three
        time_per = remaining * 0.70 / len(top_candidates)
        for sc, vec in top_candidates:
            sub_deadline = time.time() + time_per
            refined_seq, refined_score = _smooth_gradient_descent(
                vec, sub_deadline, start_time, alpha=80.0, step0=0.5)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Refined candidate score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 6) Light annealing polish (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved best to {anneal_score:.6f}")

    # --------------------------------------------------------------
    # 7) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
