def _weighted_gradient(a, w):
    # a: ndarray shape (n)
    # w: ndarray shape (2n-1)
    n = a.shape[0]
    # reverse w
    w_rev = w[::-1]
    # convolution w_rev * a
    conv_full = np.convolve(w_rev, a)   # length n + (2n-1) - 1 = 3n-2
    m = 2*n - 1
    # slice
    # g[j] = conv_full[m-1 - j] for j=0..n-1
    start_idx = m-1
    g = conv_full[start_idx : start_idx - n : -1]  # decrement
    # Equivalent: g = conv_full[start_idx - np.arange(n)]
    # Return g
    return g
