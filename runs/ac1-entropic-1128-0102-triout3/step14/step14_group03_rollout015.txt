"""
Propose a non‑negative step function that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three parts:

1️⃣  A quick parametric sweep over the family a[i] ∝ r**i·(i+1)**p mixed with the
    uniform distribution.  The best shape from this stage is kept as a warm‑start.

2️⃣  A smooth‑max Adam optimiser.  The quantity
        J_ε(a) = (1/ε)·log Σ_i exp(ε·(a*a)_i)
    is a differentiable surrogate of the true max.  Its gradient can be written as
        ∇J_ε = 2·conv(w, a_rev)[(n‑1):(n‑1)+n] ,
    where w is the soft‑max over the self‑convolution entries.  We perform an
    Adam update and project onto the probability simplex after each step.

3️⃣  A cheap final hill‑climb that moves a tiny amount of mass between random
    indices; this cleans up any remaining plateau.

All stages respect the 1000 s wall‑time limit and the final vector is returned as
a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power) and mixture with uniform
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Returns a normalised vector of length n with components proportional
    to r**i·(i+1)**p (i = 0…n‑1).  Computation is performed in log‑space.
    """
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)            # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns lam·uniform + (1‑lam)·shape (both already sum to 1).
    The result automatically sums to 1.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()                      # guard against round‑off
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max gradient (Adam) utilities
# ----------------------------------------------------------------------


def _smooth_weights_and_gradient(
    a: np.ndarray,
    eps: float,
    L_self: int,
    L_grad: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, gradient) where
        conv = a*a (full convolution),
        gradient = ∇J_ε(a)  (smooth‑max surrogate).
    J_ε = (1/ε)·log Σ_i exp(ε·conv_i).
    """
    n = a.shape[0]

    # ----- self‑convolution (needed for the surrogate value and the weights) -----
    conv = _conv_self_fft(a, L_self)                      # length 2n‑1

    # ----- soft‑max weights -----
    shift = conv.max()                                     # avoid overflow
    exp_term = np.exp(eps * (conv - shift))
    w = exp_term / exp_term.sum()                         # Σ w = 1, w ≥ 0

    # ----- gradient: 2·conv(w, a_rev) on the centre slice -----
    a_rev = a[::-1]
    # convolution w * a_rev (full length = (2n‑1)+n‑1 = 3n‑2)
    conv_wr_full = np.fft.irfft(
        np.fft.rfft(w, n=L_grad) * np.fft.rfft(a_rev, n=L_grad), n=L_grad
    )[: len(w) + len(a_rev) - 1]
    # centre slice yields a vector of length n
    grad = 2.0 * conv_wr_full[(n - 1) : (n - 1) + n]

    return conv, grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  Returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a power of two for FFT efficiency.
    # Larger n gives a finer discretisation; we use 8192 (≈8 k).
    # --------------------------------------------------------------
    n = 8192
    L_self = _next_pow2(2 * n - 1)          # length for self‑convolution FFT
    L_grad = _next_pow2(3 * n - 2)          # length for the smooth‑max gradient

    # --------------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a globally supplied starter if it exists (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                # adapt length to n (crop / pad with zeros)
                if starter.size > n:
                    starter = starter[:n]
                elif starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_self)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Coarse parametric sweep (geometric × power + uniform mix)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 120.0   # ~2 min for the sweep
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9995), num=20)
    p_grid = np.linspace(-2.5, 1.5, 21)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_self)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f}"
                        f" λ={lam:.2f} score={sc:.6f}"
                    )

    # --------------------------------------------------------------
    # 2️⃣ Smooth‑max Adam optimisation
    # --------------------------------------------------------------
    # Hyper‑parameters
    eps0 = 5.0            # start with a moderate smoothing
    eps_target = 30.0     # final (tighter) smoothing
    eps_steps = 2000      # number of iterations over which eps grows
    lr0 = 0.06
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    # reserve ~180 s for the final tiny hill‑climb
    grad_deadline = deadline - 180.0

    while time.time() < grad_deadline:
        it += 1

        # linearly increase smoothing factor
        eps = eps0 + (eps_target - eps0) * min(it / eps_steps, 1.0)

        # compute surrogate gradient
        _, grad = _smooth_weights_and_gradient(a, eps, L_self, L_grad)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size (decays very slowly – we rely on Adam’s adaptive scaling)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} eps={eps:.2f}"
                f" score={cur_score:.6f}"
            )
        else:
            # occasional uphill acceptance to escape shallow plateaus
            if rng.random() < 1e-4:
                a = a_candidate

        # small random jitter every few thousand iterations
        if it % 2500 == 0:
            jitter = 1e-4 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣ Final tiny hill‑climb (mass transfer between random indices)
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_a = a.copy()
    best_sc = best_score

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.5 % of the source mass
        eps_move = a[src] * rng.random() * 5e-3
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move
        new_sc = _eval_fast(a_prop, L_self)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_sc:.6f}")
    return [float(v) for v in best_a]
