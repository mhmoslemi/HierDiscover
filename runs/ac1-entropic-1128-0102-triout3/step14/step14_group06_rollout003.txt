"""
Step‑function optimiser (≈1000 s budget)

Strategy
---------
1.  Choose a problem size N (powers of two up to 65536) that fits the remaining time.
2.  Build a compact “top‑k” pool of diverse candidates:
    * uniform, symmetric and asymmetric Beta densities (including very small α),
    * symmetric power‑ramps with optional plateau,
    * explicit U‑shapes generated as (x·(1‑x))⁻ᵖ,
    * sparse Dirichlet draws.
3.  Keep the best TOP_K vectors (simplex‑normalised) and start from the overall best.
4.  Major optimisation – exponentiated sub‑gradient descent with a larger,
    slowly‑decaying step‑size (mirror‑descent on the simplex).
5.  Local deterministic refinement – repeatedly move a small amount of mass
    from the index with the largest sub‑gradient to the index with the smallest
    one (pairwise exchange).  After each move the vector is re‑projected onto
    the probability simplex.
6.  Final polishing – Adam on a smooth‑max surrogate (the same code as the
    reference implementation, but only if time remains).
7.  Return the best sequence (normalised to sum 1).  All entries are non‑negative,
    the evaluator will clip them automatically if needed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp”.  The shape is normalised to sum 1.
    `p` controls the steepness of the ramps, `alpha` the relative length
    of an optional flat plateau (0…0.5).
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    For α<1 the shape is U‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    log_shape = np.clip(log_shape, -700, 700)          # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    log_shape = np.clip(log_shape, -700, 700)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Small pool helper
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score, low is better)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with exponential updates on the exact ℓ∞ objective.
    Uses a larger initial step‑size (η₀=2) and a very slow decay → more aggressive moves.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (all indices attaining the current maximum, up to a tiny tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # uniform weight on the active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # sub‑gradient of max‑convolution w.r.t. a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule – very gentle decay
        eta0 = 2.0
        eta = eta0 / (1.0 + it / 2000.0)

        # multiplicative update (mirror‑descent on the simplex)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – same code as reference but with
# a slightly higher initial learning rate.
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on the smooth‑max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.08          # a bit larger than the reference
    T0 = 0.03
    T_end = 5e-5

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # temperature schedule (linear freeze)
        frac = it / (it + 1200.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w * a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise exchange local refinement
# ----------------------------------------------------------------------
def _pairwise_exchange(a0: np.ndarray,
                       deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from the index with largest sub‑gradient to the smallest."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # compute active‑set gradient (same as in ExpGD)
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        # maximum transferable mass from the “worst’’ index
        delta_max = a[i_max]
        if delta_max <= 0:
            continue

        improved = False
        for factor in (0.5, 0.2, 0.1, 0.05, 0.01):
            delta = delta_max * factor
            if delta <= 0:
                continue
            a_prop = a.copy()
            a_prop[i_max] -= delta
            a_prop[i_min] += delta
            val = _objective(a_prop)
            if val < best_val:
                best_val = val
                best_a = a_prop.copy()
                a = a_prop
                improved = True
                if it % 2000 == 0:
                    print(f"[{time.time() - start_time:.2f}s] Pairwise iter={it:,} best={val:.6f}")
                break
        if not improved and it % 4000 == 0:
            # occasional random tiny perturbation to escape plateaus
            src = np.random.choice(n, p=a)
            dst = np.random.randint(n)
            if src != dst:
                eps = a[src] * 0.015
                prop = a.copy()
                prop[src] = max(prop[src] - eps, 0.0)
                prop[dst] += eps
                prop /= prop.sum()
                val = _objective(prop)
                if val < best_val:
                    best_val = val
                    best_a = prop.copy()
                    a = prop

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under a given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # --------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 800:
        N = 65536
    elif remaining_total > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool (TOP_K best vectors)
    # --------------------------------------------------------------
    TOP_K = 12
    top: list[tuple[float, np.ndarray]] = []

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform baseline
    _add(np.full(N, 1.0 / N))

    # symmetric Beta shapes – include very small alphas
    alphas = np.concatenate([np.linspace(0.02, 0.2, 9), np.linspace(0.25, 1.0, 8), np.linspace(1.1, 1.5, 4)])
    for alpha in alphas:
        _add(_make_beta_shape(N, float(alpha)))

    # asymmetric Beta shapes (small grid)
    alphas_asym = [0.1, 0.2, 0.3, 0.5]
    betas_asym = [0.1, 0.2, 0.3, 0.5]
    for a in alphas_asym:
        for b in betas_asym:
            _add(_make_beta_shape_asym(N, float(a), float(b)))

    # power‑ramp shapes (larger range of p)
    for p in [0.2, 0.5, 1.0, 2.0, 4.0, 6.0]:
        for plateau in [0.0, 0.1, 0.25, 0.4]:
            _add(_make_shape(N, p, plateau))

    # explicit U‑shapes via (x·(1‑x))⁻ᵖ
    xs = (np.arange(N, dtype=float) + 0.5) / N
    for p in [0.5, 1.0, 1.5, 2.0]:
        shape = (xs * (1.0 - xs)) ** (-p)
        shape[np.isinf(shape)] = 0.0
        shape = np.maximum(shape, 0.0)
        tot = shape.sum()
        if tot > 0:
            shape /= tot
        else:
            shape = np.full(N, 1.0 / N)
        _add(shape)

    # random sparse Dirichlet vectors
    rng = np.random.default_rng(seed + 9999)
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.2, 1.0, size=N)   # shape < 1 → sparsity
        s = a.sum()
        if s > 0:
            a /= s
            _add(a)

    # incorporate a global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _add(starter_up)
    except Exception:
        pass

    # Best candidate from the pool
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Major optimisation – exponentiated sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate ~60 % of the remaining budget to mirror‑descent
        sub_deadline = time.time() + remaining * 0.60

        seq, sc = _exp_subgradient_descent(best_seq, sub_deadline, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD (start) improved best={sc:.6f}")

        # also try the other top‑k seeds while time permits
        for idx, (_, vec) in enumerate(top[1:], start=1):
            if time.time() >= sub_deadline:
                break
            seq2, sc2 = _exp_subgradient_descent(vec, sub_deadline, start_time)
            if sc2 < best_score:
                best_seq, best_score = seq2, sc2
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved best={sc2:.6f}")

    # --------------------------------------------------------------
    # 3) Local deterministic refinement – pairwise exchange
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        pair_deadline = time.time() + remaining * 0.55
        seq, sc = _pairwise_exchange(best_seq, pair_deadline, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] Pairwise exchange improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 4) Final polishing – smooth‑max Adam descent (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        seq, sc = _smooth_gradient_descent(best_seq, deadline, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best={sc:.6f}")

    # --------------------------------------------------------------
    # 5) Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
