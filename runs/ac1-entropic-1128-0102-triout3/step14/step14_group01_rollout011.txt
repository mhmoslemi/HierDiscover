        # source candidates with mass >0
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0: break
        src_idx = src_candidates[np.argmax(grad[src_candidates])]
        dst_idx = np.argmin(grad)
        if src_idx == dst_idx:
            # random move
            src_idx = rng.integers(n)
            dst_idx = rng.integers(n)
            if src_idx == dst_idx: continue
        # attempt improvement via line search fractions
        fractions = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
        # compute a_src
        a_src_val = a[src_idx]
        best_delta = 0.0

        # Precompute d using slice operations only once

"""
Step‑function optimiser for the “max‑convolution” challenge.

The core idea is a *gradient‑guided line‑search hill‑climb* that
moves a tunable amount of mass from a source index (with the most
positive sub‑gradient) to a destination index (with the most negative
sub‑gradient).  The exact effect of a move on the self‑convolution is
computed analytically, so the new maximum can be evaluated in O(n)
instead of recomputing a full FFT each time.  A tiny simulated‑annealing
schedule allows occasional uphill moves and random Dirichlet restarts
help escape local minima.

The optimiser works in three stages:
1.  Initialise from the global `height_sequence_1` if it exists,
    otherwise use the uniform distribution.
2.  (Optional) a short burst of random Dirichlet draws to possibly find a
    better starting point.
3.  Gradient‑guided hill‑climb with line‑search until the time budget
    expires.  The best sequence found is returned as a plain Python list.

All numbers are non‑negative and the sequence is kept normalised to a
sum of 1, which makes the evaluation function invariant to scaling.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (O(n log n))."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """
    Equivalent to `evaluate_sequence` for a normalised sequence (sum=1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).  If several shifts reach the maximum,
    the contributions are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# --------------------------------------------------------------
# Random Dirichlet initialiser (optional, fast)
# --------------------------------------------------------------
def _random_dirichlet_best(n: int, best_seq: np.ndarray,
                           best_score: float,
                           deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Generate many Dirichlet samples until the deadline and keep the best."""
    while time.time() < deadline:
        # concentration drawn log‑uniformly
        log_c = rng.uniform(np.log(0.05), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Gradient‑guided line‑search hill‑climb with simulated annealing
# --------------------------------------------------------------
def _hillclimb_line_search(a0: np.ndarray,
                           start_time: float,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from the index with the largest positive
    sub‑gradient to the index with the most negative one.
    The exact effect on the self‑convolution is computed analytically,
    allowing a cheap line‑search over several step sizes.
    A tiny temperature schedule permits occasional uphill moves,
    and very rare random Dirichlet restarts help escape deep local minima.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L_fft)
    cur_score = 2.0 * n * conv.max()

    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()
        print(f"[{time.time() - start_time:.2f}s] Initial hill‑climb score={cur_score:.6f}")

    rng = np.random.default_rng()
    total_time = deadline - start_time
    # Initial temperature (≈1 % of the current score)
    T0 = max(cur_score * 0.01, 1e-12)

    # fractions of source mass to try (large → small)
    fractions = [0.20, 0.10, 0.05, 0.02, 0.01, 0.005, 0.001]

    # reusable buffer for the shifted difference vector
    d = np.empty(2 * n - 1, dtype=np.float64)

    while time.time() < deadline:
        # ---- sub‑gradient & choice of source / destination -----------------
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # avoid the degenerate case; fall back to a random pair
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        # ---- build the shifted‑difference vector d = shift(dst) – shift(src) ---
        d.fill(0.0)
        d[dst:dst + n] += a
        d[src:src + n] -= a

        a_src = a[src]
        best_delta = 0.0
        best_new_score = cur_score

        # ---- line‑search over several fractions of the source mass ----------
        for f in fractions:
            delta = a_src * f
            if delta <= 1e-18:
                continue

            # tentative new convolution (copy + linear part)
            new_conv = conv + 2.0 * delta * d
            # adjust the three indices where the quadratic terms appear
            new_conv[2 * dst] += delta * delta          # (dst,dst) term
            new_conv[2 * src] -= delta * delta          # (src,src) term
            new_conv[src + dst] -= 2.0 * delta * delta  # (src,dst) & (dst,src)

            new_score = 2.0 * n * new_conv.max()
            if new_score < best_new_score:
                best_new_score = new_score
                best_delta = delta

        # ---- apply the best improving move -------------------------------
        if best_delta > 0.0:
            # update the sequence
            a[src] -= best_delta
            a[dst] += best_delta

            # update the convolution in‑place
            conv += 2.0 * best_delta * d
            conv[2 * dst] += best_delta * best_delta
            conv[2 * src] -= best_delta * best_delta
            conv[src + dst] -= 2.0 * best_delta * best_delta

            cur_score = best_new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Improved score={best_score:.6f}")
            continue

        # ---- no improving step found – try a tiny uphill move ------------
        # (simulated annealing flavour)
        delta = a_src * (rng.random() * 0.001 + 1e-6)
        if delta <= 0:
            continue

        new_conv = conv + 2.0 * delta * d
        new_conv[2 * dst] += delta * delta
        new_conv[2 * src] -= delta * delta
        new_conv[src + dst] -= 2.0 * delta * delta
        new_score = 2.0 * n * new_conv.max()
        Δ = new_score - cur_score

        # linear temperature decay
        remaining = max(deadline - time.time(), 0.0)
        T = T0 * (remaining / total_time)

        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv += 2.0 * delta * d
            conv[2 * dst] += delta * delta
            conv[2 * src] -= delta * delta
            conv[src + dst] -= 2.0 * delta * delta
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA accepted new best={best_score:.6f}")

        # ---- occasional random restart (very low probability) -----------
        if rng.random() < 0.0005:
            # generate a fresh Dirichlet candidate and switch if it is better
            log_c = rng.uniform(np.log(0.05), np.log(10.0))
            c = np.exp(log_c)
            cand = rng.gamma(shape=c, scale=1.0, size=n)
            if cand.sum() == 0:
                continue
            cand /= cand.sum()
            cand_score = _objective_score(cand)
            if cand_score < best_score:
                a = cand
                conv = _conv_self(a, L_fft)
                cur_score = cand_score
                best_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Random restart improved score={best_score:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine uses a fast gradient‑guided hill‑climb with line‑search and a
    tiny simulated‑annealing schedule.  It respects the provided time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ----------------------------------------------------------
    # Problem size – keep the same length as the reference run
    # ----------------------------------------------------------
    n = 4096

    # ----------------------------------------------------------
    # Initialise (prefer the globally supplied construction if any)
    # ----------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ----------------------------------------------------------
    # Optional quick Dirichlet sweep (gives a few diverse starts)
    # ----------------------------------------------------------
    dirichlet_deadline = start_time + min(30.0, (deadline - start_time) * 0.05)
    if time.time() < dirichlet_deadline:
        rng = np.random.default_rng()
        best_seq, best_score = _random_dirichlet_best(
            n, best_seq, best_score, dirichlet_deadline, rng
        )

    # ----------------------------------------------------------
    # Gradient‑guided hill‑climb with line‑search – main workhorse
    # ----------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hillclimb_line_search(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished best≈{best_score:.6f}")

    # ----------------------------------------------------------
    # Normalise again (numerical drift) and return as plain list
    # ----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
