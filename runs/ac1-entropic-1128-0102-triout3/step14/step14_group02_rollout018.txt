    idx = np.arange(1, n+1)
    # shape: (i)^{a-1} * (n+1-i)^{b-1}
    log_seq = (a-1) * np.log(idx) + (b-1) * np.log(n+1 - idx)

Search for a non‑negative step function (probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search proceeds in three stages (all respecting the 1000 s budget):

1️⃣  **Massive random sampling** of simple parametric families
    (Beta‑shapes, double‑exponential, power laws and Dirichlet draws).
    The best sample becomes the starting point for the next stage.

2️⃣  **Projected gradient descent** on a smooth approximation of the max.
    The gradient of the soft‑max of the convolution is computed via FFT,
    a direction step is taken, and the iterate is projected back onto the
    probability simplex.  An adaptive back‑tracking line‑search and a
    slowly increasing temperature τ turn the smooth objective into the true
    max while keeping the iteration stable.

3️⃣  **Fine‑grained hill‑climbing** that repeatedly transfers a small
    fraction of mass from the index with the largest sub‑gradient to the
    index with the smallest, always keeping improvements (or performing a
    random perturbation after many failures).

The routine returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – FFT convolution and simplex projection
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, out_len: int) -> np.ndarray:
    """Full linear convolution via FFT, result length = out_len."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:out_len]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (length = 2·len(a) – 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective and gradient helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))  (a is already normalised, sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _smooth_max_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of 2·n·softmax_tau(conv(a,a)).
    softmax_tau(x) = log(sum exp(tau·x)) / tau.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (more uniform for small tau, concentrate on the peak for large tau)
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        w_sum = 1.0
    w /= w_sum

    # grad = 2 * (w * a_rev)  (convolution of w with reversed a)
    a_rev = a[::-1]
    grad_full = _conv_fft(w, a_rev, out_len=3 * n - 2)   # length = 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]               # centre part corresponds to a‑indices
    return grad


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max (not smoothed) at a.
    When several shifts attain the max the contributions are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Random families – a fast way to generate diverse candidates
# ----------------------------------------------------------------------
def _beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    a = 10 ** rng.uniform(-2, 1)   # α ∈ [0.01,10]
    b = 10 ** rng.uniform(-2, 1)   # β ∈ [0.01,10]
    idx = np.arange(1, n + 1, dtype=np.float64)
    log_seq = (a - 1.0) * np.log(idx) + (b - 1.0) * np.log(n + 1 - idx)
    seq = np.exp(log_seq - np.max(log_seq))
    seq /= seq.sum()
    # optional uniform mixing
    lam = rng.random()
    seq = lam * (np.full(n, 1.0 / n)) + (1.0 - lam) * seq
    seq /= seq.sum()
    return seq


def _double_exp_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    lam1 = rng.uniform(0.0, 5.0)
    lam2 = rng.uniform(0.0, 5.0)
    idx = np.arange(n, dtype=np.float64)
    left = np.exp(-lam1 * idx)
    right = np.exp(-lam2 * (n - 1 - idx))
    seq = left + right
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    # mix with uniform as above
    lam = rng.random()
    seq = lam * (np.full(n, 1.0 / n)) + (1.0 - lam) * seq
    seq /= seq.sum()
    return seq


def _power_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(-5.0, 5.0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(idx, p)
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq.sum()
    lam = rng.random()
    seq = lam * (np.full(n, 1.0 / n)) + (1.0 - lam) * seq
    seq /= seq.sum()
    return seq


def _dirichlet_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    # concentration sampled on a log‑scale
    logc = rng.uniform(-3.0, 1.0)   # c ∈ [0.001, 10]
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Stage 2 – projected gradient descent on the smooth max
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                deadline: float,
                                best_seq: np.ndarray,
                                best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    tau = 0.5                 # start with a very smooth approximation
    tau_factor = 1.2
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} τ={tau:.3f} score={cur_score:.6f}")

        # gradient of the smooth‑max objective
        grad = _smooth_max_gradient(a, tau)

        # try a few step sizes with back‑tracking line search
        success = False
        for eta in (0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01):
            a_new = a - eta * grad
            a_new = _project_simplex(np.maximum(a_new, 0.0))
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score + 1e-12 < cur_score:          # improvement
                a = a_new
                success = True
                break
        if not success:
            # if none of the tried step sizes improve, shrink τ (more smoothing)
            tau = max(tau * 0.7, 0.01)
        else:
            # gradual increase of τ to tighten the approximation
            if it % 200 == 0:
                tau = min(tau * tau_factor, 1e5)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – fine hill‑climb (transfer small mass from “high‑grad” to “low‑grad”)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # allow a fairly large transfer – up to 5 % of src mass
        frac = rng.random() * 0.045 + 0.001   # [0.001, 0.046]
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)   # guarantees feasibility

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score - 1e-12:
            a = a_prop
            best_seq, best_score = a.copy(), cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill climb new best={cur_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 4000:
                # escape a possible plateau by a random jitter
                jitter = rng.normal(scale=0.001, size=n)
                a = _project_simplex(np.maximum(a + jitter, 0.0))
                no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats (a probability vector) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.20      # small safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length makes the search fast yet
    # expressive enough to beat the previous 1.507831 bound.
    # --------------------------------------------------------------
    n = 2048
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Random parametric sweep
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    families = [_beta_candidate, _double_exp_candidate,
                _power_candidate, _dirichlet_candidate]
    sample_budget = int(2.5e4)   # total number of random candidates
    for _ in range(sample_budget):
        if time.time() >= deadline:
            break
        # pick a random family
        cand_func = rng.choice(families)
        a = cand_func(n, rng)
        sc = _objective(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Random sample improved score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected gradient descent (smooth‑max)
    # --------------------------------------------------------------
    gd_deadline = deadline - 250.0
    if time.time() < gd_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected gradient descent")
        best_seq, best_score = _projected_gradient_descent(
            best_seq, gd_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After GD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Fine hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
