"""
Step‑function optimiser (coarse‑to‑fine).

* Stage 1 – small problem (N≈8 k).  A large pool of analytic shapes
  (Beta, power‑ramps, random Dirichlet with many concentrations) is built
  and the best candidate is refined by many iterations of exponentiated
  sub‑gradient descent on the exact ℓ∞ objective.

* Stage 2 – up‑sample the best small‑scale solution to the full size
  (N≈65 k) and continue optimisation:
    – exponentiated sub‑gradient descent (active‑set) for the bulk of the
      remaining budget,
    – finally a smooth‑max Adam polishing phase (soft‑max surrogate of the
      max‑convolution).

All updates keep the vector non‑negative and renormalise it to a probability
vector, which makes the objective 2·n·max(conv)/sum² scale‑invariant.
The algorithm is deliberately time‑aware and adapts the length of each
phase to the remaining budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(seq,seq))/(sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators (all return a probability vector, sum≈1)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Symmetric power‑ramp a_i ∝ (i/side_len)^p on the two sides,
    optionally with a flat plateau of relative length `plateau_frac`.
    """
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:                # safety padding
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector (α = concentration)."""
    if concentration <= 0:
        concentration = 1e-3
    a = rng.gamma(concentration, 1.0, size=n)
    tot = a.sum()
    if tot > 0:
        a /= tot
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (exact ℓ∞ objective)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             time_end: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) update on the exact max–convolution
    objective.  a is kept on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < time_end:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices attaining the max up to a tiny tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # weight vector w (uniform on the active set)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)  via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule (decreasing)
        eta0 = 0.9
        eta = eta0 / (1.0 + it / 1800.0)

        # multiplicative update, safely clamped
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # simplex projection (renormalise)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – surrogate for the ℓ∞ objective
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     time_end: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv(a,a)).
    Temperature is annealed to turn the surrogate into the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.04
    T0 = 0.02
    T_end = 1e-4

    it = 0
    while time.time() < time_end:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule (linear anneal)
        frac = min(1.0, (time.time() - start_time) / (time_end - start_time))
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * (w * a_rev) via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score

# Upsampling helper
# ----------------------------------------------------------------------
def _upsample_vector(v_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from len(v_small) to n_big, followed by simplex normalisation."""
    n_small = v_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    v_big = np.interp(x_big, x_small, v_small)
    v_big = np.maximum(v_big, 0.0)
    tot = v_big.sum()
    if tot > 0:
        v_big /= tot
    else:
        v_big = np.full(n_big, 1.0 / n_big)
    return v_big

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Hierarchical optimisation for the step‑function bound.
    Returns a list of non‑negative heights (probability vector).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # --------------------------------------------------------------
    # Stage 1 – coarse optimisation on a modest size
    # --------------------------------------------------------------
    N_small = 8192  # fast enough for many iterations
    rng = np.random.default_rng(seed + 12345)

    # Build a diverse initial pool (score, vector)
    TOP_K = 15
    pool = []   # list of (score, vector)

    def _push(score, vec):
        """Insert (score,vec) into the top‑K pool (ascending scores)."""
        if len(pool) < TOP_K:
            pool.append((score, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif score < pool[-1][0]:
            pool[-1] = (score, vec.copy())
            pool.sort(key=lambda x: x[0])

    # Uniform baseline
    _push(_objective(np.full(N_small, 1.0 / N_small)),
          np.full(N_small, 1.0 / N_small))

    # Symmetric Beta family (dense sweep)
    for alpha in np.linspace(0.2, 2.0, 30):
        shape = _make_beta_shape(N_small, alpha)
        _push(_objective(shape), shape)

    # Asymmetric Beta sweep (coarse grid)
    for a in [0.3, 0.6, 1.0, 1.5, 2.0]:
        for b in [0.3, 0.6, 1.0, 1.5, 2.0]:
            shape = _make_beta_shape_asym(N_small, a, b)
            _push(_objective(shape), shape)

    # Power‑ramp shapes
    for p in np.linspace(0.1, 4.0, 12):
        for plateau in np.linspace(0.0, 0.5, 5):
            shape = _make_power_ramp(N_small, p, plateau)
            _push(_objective(shape), shape)

    # Random Dirichlet vectors with many concentration parameters
    concentrations = [0.05, 0.1, 0.2, 0.4, 0.8, 1.5, 3.0]
    for conc in concentrations:
        for _ in range(20):
            vec = _make_random_dirichlet(N_small, conc, rng)
            _push(_objective(vec), vec)

    # Optional warm‑start from a previously found global sequence
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N_small:
                if N_small % starter.size == 0:
                    factor = N_small // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N_small, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push(_objective(starter_up), starter_up)
    except Exception:
        pass

    # Best candidate after the pool construction
    best_small_seq = pool[0][1].copy()
    best_small_score = pool[0][0]
    print(f"[{time.time() - start_time:.2f}s] Stage 1 pool built, best score = {best_small_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 refinement – exponentiated sub‑gradient descent
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.25 * (deadline - start_time)  # roughly 25 % of total time
    best_small_seq, best_small_score = _exp_subgradient_descent(best_small_seq,
                                                               phase1_deadline,
                                                               start_time)
    print(f"[{time.time() - start_time:.2f}s] Stage 1 refined, score = {best_small_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – upscale to the full resolution
    # --------------------------------------------------------------
    N_big = 65536
    best_big_seq = _upsample_vector(best_small_seq, N_big)
    cur_score = _objective(best_big_seq)
    print(f"[{time.time() - start_time:.2f}s] Upsampled to N={N_big}, start score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 main optimisation – exponentiated sub‑gradient
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate 70 % of the remainder to sub‑gradient, rest to smooth‑max
    sub_end = time.time() + 0.70 * remaining
    best_big_seq, best_big_score = _exp_subgradient_descent(best_big_seq,
                                                            sub_end,
                                                            start_time)
    print(f"[{time.time() - start_time:.2f}s] Stage 2 ExpGD finished, best score = {best_big_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 polishing – smooth‑max Adam descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_big_seq, best_big_score = _smooth_max_adam(best_big_seq,
                                                       deadline,
                                                       start_time)
        print(f"[{time.time() - start_time:.2f}s] Stage 2 Smooth‑GD polished, final score = {best_big_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation & return
    # --------------------------------------------------------------
    best_big_seq = np.maximum(best_big_seq, 0.0)
    tot = best_big_seq.sum()
    if tot > 0:
        best_big_seq /= tot
    else:
        best_big_seq = np.full(N_big, 1.0 / N_big)

    final_score = _objective(best_big_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final evaluation ≈ {final_score:.6f}")

    return [float(v) for v in best_big_seq]
