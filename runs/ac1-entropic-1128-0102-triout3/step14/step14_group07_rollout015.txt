assistantfinal**Solution Overview**

The aim is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 ,
\]

where `a*a` is the linear convolution of the vector with itself.  
Because the score is *scale‑invariant* we may restrict ourselves to vectors on the
probability simplex (non‑negative and summing to 1).  
For such vectors  

\[
\text{score}(a)=2\,n\,\max (a*a).
\]

A very good solution is obtained by searching the simplex with a combination of

* **very diverse analytic families** (Beta‑distributions,
  asymmetric power‑ramps, block‑wise constant shapes, random Dirichlet vectors);
* **projected Adam optimisation** on a smooth approximation of the
  `max`‑operator (soft‑max temperature annealing);
* **short stochastic polishing** (mass swaps) to escape any remaining local
  minima.

The code below follows this scheme, makes the search space considerably larger
than the reference implementation and spends the remaining time on a final
refinement of the best candidate.

```python
"""
Propose a non‑negative step‑function (list of floats) that minimises

                2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Broad sampling** –  many analytic families (Beta, asymmetric power‑ramps,
   block‑wise constant shapes) and random Dirichlet draws are generated.
   The best K candidates (lowest score) are kept.

2. **Projected Adam optimisation** – each retained candidate is refined by
   minimising a smooth approximation of the `max`‑operator
   (soft‑max with exponentially decaying temperature).  The iterates are
   projected onto the probability simplex after each step.

3. **Simulated‑annealing polish** – a short mass‑swap annealing run starting
   from the best refined sequence gives a final nudge.

All phases respect the 1000 s budget; the routine always returns the best
sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (zero‑padded to next power of two)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Score for a vector that already sums to 1 (no extra scaling needed)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())   # sum(seq)==1


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the pool of size at most k (kept sorted)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n                # mid‑point rule
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_shape(n: int, p_left: float, p_right: float,
                     plateau_frac: float = 0.0) -> np.ndarray:
    """
    Asymmetric power‑ramp.
    Left side (indices 0 … left_len‑1) : (i/left_len)^p_left
    Right side (indices n‑right_len … n‑1) : ((right_len‑i)/right_len)^p_right
    Optional flat plateau in the centre.
    """
    plateau_len = int(round(plateau_frac * n))
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        # left ramp
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        left = idx ** p_left
        # right ramp (mirror of left but with its own exponent)
        right = idx ** p_right
        # assemble
        shape = np.concatenate([left,
                                np.full(plateau_len, 1.0),
                                right[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_block_shape(n: int, rng: np.random.Generator,
                      min_blocks: int = 5, max_blocks: int = 200) -> np.ndarray:
    """
    Random piece‑wise constant shape.
    A random number B of blocks is chosen, each block receives a random total
    mass; the mass is spread uniformly over the block.
    """
    B = rng.integers(min_blocks, max_blocks + 1)
    # random block lengths (Dirichlet → integer partition)
    lengths_frac = rng.random(B)
    lengths_frac /= lengths_frac.sum()
    lengths = np.round(lengths_frac * n).astype(int)

    # fix rounding errors
    diff = n - lengths.sum()
    if diff > 0:
        lengths[:diff] += 1
    elif diff < 0:
        lengths[: -diff] -= 1

    # random block masses (Dirichlet)
    masses = rng.random(B)
    masses /= masses.sum()                     # total mass = 1
    # per‑element height in each block
    heights = masses / lengths
    blocks = [np.full(lengths[i], heights[i], dtype=float) for i in range(B)]
    seq = np.concatenate(blocks) if blocks else np.full(n, 1.0 / n)
    if seq.size > n:
        seq = seq[:n]
    elif seq.size < n:
        seq = np.concatenate([seq, np.zeros(n - seq.size)])
    return seq


def _make_random_dirichlet(n: int, rng: np.random.Generator,
                          concentration: float) -> np.ndarray:
    """Draw a random point on the simplex with Dirichlet(concentration,…,conc)."""
    a = rng.gamma(concentration, 1.0, size=n)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv(a,a)).
    Temperature decays exponentially from T_start to T_end.
    Returns the best (seq,score) seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    T_start, T_end = 0.30, 1e-6
    step0 = 0.15                         # initial learning rate

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # forward pass
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (exponential)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # smooth‑max weights (stable soft‑max)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max = 2 * (w  *  a_rev)   (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex (keeps sum==1)
        a = _project_simplex(a)

    return best_a, best_score


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """Very light mass‑swap annealing starting from *a0*."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a, best_score = a.copy(), cur_score

    n = a.shape[0]
    T_start, T_end = 0.2, 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # choose a source index proportional to its mass (more massive ⇒ more likely)
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose discretisation size (larger N ⇒ finer search)
    # --------------------------------------------------------------
    if deadline - start_time > 600:
        N = 65536
    elif deadline - start_time > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial diverse pool of candidates
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []   # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # --------- Analytic families ------------------------------------
    # symmetric Beta
    for alpha in np.logspace(-2, 0.8, 20):            # 0.01 … 6.3
        shape = _make_beta_shape(N, float(alpha), float(alpha))
        _push_top(pool, _objective(shape), shape, TOP_K)

    # asymmetric Beta (random α,β)
    for _ in range(120):
        if time.time() >= deadline:
            break
        a = 10 ** rng.uniform(-2, 0.8)               # log‑uniform 0.01 … 6.3
        b = 10 ** rng.uniform(-2, 0.8)
        shape = _make_beta_shape(N, float(a), float(b))
        _push_top(pool, _objective(shape), shape, TOP_K)

    # asymmetric power‑ramps
    exponents = np.linspace(-0.5, 3.0, 9)             # -0.5 … 3
    for pl in np.linspace(0.0, 0.5, 4):               # plateau fraction
        for pL in exponents:
            for pR in exponents:
                if time.time() >= deadline:
                    break
                shape = _make_power_shape(N, float(pL), float(pR), float(pl))
                _push_top(pool, _objective(shape), shape, TOP_K)

    # block‑wise constant shapes
    for _ in range(80):
        if time.time() >= deadline:
            break
        shape = _make_block_shape(N, rng,
                                 min_blocks=5, max_blocks=150)
        _push_top(pool, _objective(shape), shape, TOP_K)

    # random Dirichlet draws with many concentration values
    concentrations = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in concentrations:
        for _ in range(30):
            if time.time() >= deadline:
                break
            shape = _make_random_dirichlet(N, rng, conc)
            _push_top(pool, _objective(shape), shape, TOP_K)

    # incorporate a global starter if it exists in the namespace
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # resize / interpolate to required size N
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
                _push_top(pool, _objective(starter), starter, TOP_K)
    except Exception:
        pass

    # current best
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Adam‑style refinement of the retained candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    per_candidate = max(5.0, remaining / (len(pool) * 2.0))

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per_candidate
        if sub_deadline >= deadline:
            break
        refined_seq, refined_score = _adam_smooth_max(vec, sub_deadline,
                                                     start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) One longer refinement of the current best (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        final_seq, final_score = _adam_smooth_max(best_seq, deadline,
                                                  start_time)
        if final_score < best_score:
            best_score, best_seq = final_score, final_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final Adam polish improved best to {final_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light simulated‑annealing polish (mass swaps)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.025)

    # --------------------------------------------------------------
    # 5) Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
```
