"""
Propose a step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search works in five stages, each respecting the remaining time budget:

1️⃣  Wide parametric sweeps (geometric, power, symmetric/asymmetric Beta,
    plus a Gaussian family) over a large dimension n (≈ 8000).
2️⃣  Random Dirichlet initialisers with a broad range of concentration.
3️⃣  Random spectral initialisers – take a constant‑modulus FFT with random
    phases, clip to non‑negative, and project onto the simplex.
4️⃣  Smooth‑max continuation using Adam (τ grows geometrically, turning the
    smooth max into the true max).  The gradient of the smooth max is obtained
    via an FFT‑based convolution, and the parameters are projected onto the
    probability simplex after each update.
5️⃣  A final directed hill‑climb that moves a tiny amount of mass from the
    index with the largest positive sub‑gradient to the index with the most
    negative one, accepting only improving moves.

The algorithm keeps the best vector found (always normalised to sum = 1) and
returns it as a plain Python list.  All operations are vectorised and use
FFT‑based convolutions, making the search feasible within the 1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Assuming sum(seq)=1, return 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# 1️⃣  Parametric families
# --------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    global start_time
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 250)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    global start_time
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 141):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    global start_time
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    alphas = np.linspace(0.05, 4.0, 71)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in alphas:            # symmetric case (b == a) was used before,
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    global start_time
    """Explore the full (α,β) grid (α≠β)."""
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    alphas = np.linspace(0.1, 5.0, 19)   # 19×19 = 361 combos
    betas = alphas

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_gaussian(n, best_seq, best_score, deadline):
    global start_time
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma_grid = np.logspace(np.log10(n / 200), np.log10(n / 2), 40)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        gauss = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        s = gauss.sum()
        if s == 0:
            continue
        seq = gauss / s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gauss σ={sigma:.1f} score={sc:.6f}")
    return best_seq, best_score

# 2️⃣  Random Dirichlet initialisers
# --------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    global start_time
    rng = np.random.default_rng()
    trials = 300
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.8)      # concentrations ~0.01 … 6
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣  Spectral random initialisers
# --------------------------------------------------------------
def _spectral_random_trials(n, best_seq, best_score, deadline):
    global start_time
    rng = np.random.default_rng()
    trials = 120
    half = n // 2
    for t in range(trials):
        if time.time() >= deadline:
            break
        # random phases for frequencies 1 … half-1
        phases = rng.uniform(0, 2 * np.pi, half - 1)
        spec = np.zeros(n, dtype=np.complex128)
        # DC and Nyquist components set to 0 (purely AC)
        spec[0] = 0.0
        spec[1:half] = np.exp(1j * phases)
        spec[half] = 0.0
        spec[half + 1:] = np.conj(spec[1:half][::-1])
        a = np.fft.ifft(spec).real
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Spectral #{t} score={sc:.6f}")
    return best_seq, best_score

# 4️⃣  Smooth‑max continuation (Adam optimiser)
# --------------------------------------------------------------
def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):
    global start_time
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 0.2               # start very smooth
    tau_factor = 1.8
    tau_update_every = 2500
    it = 0

    # pre‑allocated length for the gradient convolution
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # current convolution and score
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth soft‑max over convolution entries
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w * rev(a))  (length 3n‑2, then extract middle n entries)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)           # length = 3n‑2 (full)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature occasionally
        if it % tau_update_every == 0:
            tau = min(tau * tau_factor, 1e6)

    return best_seq, best_score

# 5️⃣  Directed hill‑climb
# --------------------------------------------------------------
def _hill_climb(start_a, deadline):
    global start_time
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        frac = rng.random() * 0.006 + 1e-6   # up to 0.6 % of source mass
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill climb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 8000:
                # small random jitter to escape plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.006 + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine follows a staged strategy (parametric sweeps,
    random draws, spectral seeds, Adam optimisation, hill‑climb) and
    respects the given time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.12

    # --------------------------------------------------------------
    # Problem size – use a larger n if time permits
    # --------------------------------------------------------------
    # If we have ample time choose n=8192, otherwise fall back to 4096.
    if budget_s >= 400:
        n = 8192
    else:
        n = 4096

    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline (n={n}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power, Beta, Gaussian)
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_gaussian(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Spectral random initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 190.0
    if time.time() < phase_deadline:
        best_seq, best_score = _spectral_random_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 4️⃣  Smooth‑max continuation with Adam
    # --------------------------------------------------------------
    smooth_deadline = deadline - 80.0
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max_descent(best_seq, smooth_deadline,
                                                        best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
