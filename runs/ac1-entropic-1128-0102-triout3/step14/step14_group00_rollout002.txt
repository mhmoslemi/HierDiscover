    n = len(a)
    A = []
    b_ub = []

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    
    start = time.time()
    deadline = start + budget_s - 1.0
    
    # Choose n
    n = 4096  # maybe adapt based on time
    # Build initial candidate
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim == 1 and init.size > 0:
            # adjust length to n: if length differs, resample
            if init.size != n:
                # interpolate to n points
                x_old = np.linspace(0, 1, init.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                init = np.interp(x_new, x_old, init)
            init = np.maximum(init, 0.0)
            if init.sum() > 0:
                init = init / init.sum()
        else:
            init = None
    else:
        init = None
    
    if init is None:
        # random beta shape
        a = 10**(rng.uniform(-2, 0.5))
        b = 10**(rng.uniform(-2, 0.5))
        init = _make_beta_shape(n, a, b)
    
    cur_seq = init.copy()
    cur_score = _objective(cur_seq)
    print(f"[{time.time()-start:.2f}s] initial score = {cur_score:.6f}")
    
    # LP improvement loop
    max_lp_iters = 20
    lambda_mix = 0.2  # base mixing factor
    while time.time() < deadline and max_lp_iters > 0:
        conv = _conv_self(cur_seq)
        M = conv.max()
        # Build LP matrix
        # For each k, row contains a_i where indices valid.
        # We can construct dense A.
        # We'll produce as list of rows for SciPy.
        rows = []
        b_ub = []
        for k in range(2*n-1):
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            # b appears at positions idx = lo..hi
            row = np.zeros(n, dtype=float)
            idx = np.arange(lo, hi+1)
            row[idx] = cur_seq[k-idx]
            rows.append(row)
            b_ub.append(M)
        A_ub = np.stack(rows, axis=0)
        # Solve LP: max sum(b) => min -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # Solve
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                                     method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time()-start:.2f}s] LP failed {res.message}")
            break
        b_vec = res.x
        # Normalize b to sum 1
        if b_vec.sum() > 0:
            b_vec = b_vec / b_vec.sum()
        else:
            b_vec = np.full(n, 1.0/n)

"""
Propose a low‑score non‑negative step function.

The algorithm mixes three ideas that complement each other:

1. **Analytic seeding** – a small pool of beta‑distribution shapes (including
   the previously known best `height_sequence_1`) is evaluated and the best
   one is taken as a start.

2. **Active‑set LP improvement** – from the current vector `a` we compute its
   self‑convolution `c = a*a`.  The maximal entries of `c` are the tight
   constraints of the optimisation problem
        minimise   max_k (a*b)[k]
        subject to b ≥ 0,  Σb = 1 .
   Using only the *active* indices (the K largest values of `c`) we build a
   tiny linear program (≈ K constraints, K≈20) that maximises Σb under the
   constraint that the convolution `a*b` does not exceed the current maximal
   value.  The solution `b` is then mixed with `a` (line‑search over a few
   mixing factors).  This step typically produces a noticeable drop in the
   objective.

3. **Fine‑grained mirror descent** – after the LP phase we run an
   exponentiated‑gradient (KL‑mirror) descent on the simplex, which is a
   sub‑gradient method for the L∞‑norm of the self‑convolution.
   The step‑size decays as η₀/√t (η₀≈0.8).  The descent runs for as much
   time as remains after the LP phase.

4. **Optional annealing & up‑sampling** – a short simulated‑annealing
   polish is performed if time permits, and if a large time budget (> 200 s)
   remains we double the resolution (interpolate to 2 · n) and finish with a
   final mirror‑descent pass.

All numbers are kept non‑negative, the vector is normalised to sum = 1
(so the objective reduces to `2·n·max(conv)`), and the routine respects the
overall 1000 s time budget.
"""

import time
import math
import numpy as np

# Optional SciPy – used for the tiny linear programs.
try:
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (fast for powers of two)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) (sum(seq) assumed = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)   # sum(seq)==1 ⇒ denominator = 1


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised Beta( a , b ) on {0,…,n‑1}, normalised to sum = 1.
    The pdf ∝ x^(a‑1) (1‑x)^(b‑1).
    """
    xs = (np.arange(n, dtype=float) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:                       # degenerate parameters – fall back to uniform
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the probability simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent (KL) for minimising max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set – indices whose value is within a tiny absolute tolerance
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        # sub‑gradient: average of gradients of the active entries
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update, enforce a tiny floor and re‑normalise
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, eps_floor)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score

# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule: T∈[0.02, 1e‑6]
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # pick source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5   # move up to 50 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Solve a tiny LP based on the currently active convolution constraints
# ----------------------------------------------------------------------
def _solve_lp_active(a: np.ndarray,
                     M: float,
                     ks: np.ndarray) -> np.ndarray:
    """
    Solve  max Σb_i   s.t.   (a * b)[k] ≤ M  for all k ∈ ks,
    b_i ≥ 0, Σb_i = 1.
    The constraints are linear in b because a is fixed.
    """
    if linprog is None:                     # pragma: no cover
        # Fallback – return the current vector (no improvement possible)
        return a.copy()

    n = a.shape[0]
    rows = np.zeros((len(ks), n), dtype=float)

    for row_i, k in enumerate(ks):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        rows[row_i, idx] = a[k - idx]       # (a * b)[k] = Σ a_{k-i} b_i

    # Linear programme: minimise – Σ b_i  (i.e. maximise Σ b_i)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    b_ub = np.full(len(ks), M, dtype=float)

    res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # If the LP fails for any reason, fall back to the current vector.
        return a.copy()

    b = res.x
    s = b.sum()
    if s > 0.0:
        b /= s
    else:
        b = np.full(n, 1.0 / n)

    return b

                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                pool.append((_objective(starter), starter.copy()))
    except Exception:
        pass

    # 2) Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # 3) Random beta‑distribution shapes
    for _ in range(30):
        a_par = 10 ** rng.uniform(-2.0, 0.7)
        b_par = 10 ** rng.uniform(-2.0, 0.7)
        v = _make_beta_shape(n, a_par, b_par)
        pool.append((_objective(v), v.copy()))

    # 4) Random mixtures of two beta shapes
    for _ in range(20):
        a1 = 10 ** rng.uniform(-2.0, 0.7)
        b1 = 10 ** rng.uniform(-2.0, 0.7)
        a2 = 10 ** rng.uniform(-2.0, 0.7)
        b2 = 10 ** rng.uniform(-2.0, 0.7)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1.0 - w) * _make_beta_shape(n, a2, b2)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:pool_size]

    best_seq = pool[0][1].copy()
    best_score = pool[0][0]

    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Active‑set LP improvement phase
    # --------------------------------------------------------------
    max_lp_iters = 15
    K_active = 20                     # number of active constraints used in LP

    while time.time() < deadline and max_lp_iters > 0:
        conv = _conv_self(best_seq)
        M = conv.max()

        # Pick the K indices with largest convolution values (tight constraints)
        if K_active >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(-conv, K_active - 1)[:K_active]

        # Solve the tiny linear program
        b_vec = _solve_lp_active(best_seq, M, ks)

        # Line‑search over a few mixing coefficients
        improved = False
        for t in (0.05, 0.10, 0.20, 0.30, 0.50):
            cand = (1.0 - t) * best_seq + t * b_vec          # Σ stays =1
            cand_score = _objective(cand)
            if cand_score < best_score - 1e-12:
                best_seq = cand
                best_score = cand_score
                improved = True
                print(f"[{time.time() - start:.2f}s] LP mix t={t:.2f} → score={cand_score:.6f}")
                break

        if not improved:
            # No improvement – stop the LP phase early
            break

        max_lp_iters -= 1

    # --------------------------------------------------------------
    # 2) Exponentiated‑gradient descent (mirror descent)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        eg_deadline = time.time() + 0.70 * remaining
        best_seq, best_score = _exp_gradient_descent(best_seq,
                                                     eg_deadline,
                                                     start,
                                                     eta0=0.8)
        print(f"[{time.time() - start:.2f}s] After EG best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Optional up‑sampling + final refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and n < 8192:
        # Double the resolution by linear interpolation
        n2 = n * 2
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
        best_seq = np.interp(x_new, x_old, best_seq)
        best_seq /= best_seq.sum()
        n = n2
        print(f"[{time.time() - start:.2f}s] Upsampled to n={n}")

        # One more short EG pass on the finer grid
        remaining = deadline - time.time()
        if remaining > 10.0:
            eg_deadline = time.time() + 0.70 * remaining
            best_seq, best_score = _exp_gradient_descent(best_seq,
                                                         eg_deadline,
                                                         start,
                                                         eta0=0.8)
            print(f"[{time.time() - start:.2f}s] After up‑sampled EG score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start)
        print(f"[{time.time() - start:.2f}s] After anneal best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
