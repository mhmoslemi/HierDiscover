def _hill_climb(a, L_fft, deadline, start_time):
    n = len(a)
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # choose source index weighted by a (higher chance from high value)
        src = rng.choice(np.arange(n), p=best / best.sum())
        # choose dest uniform from all other indices maybe
        dst = rng.integers(n)
        if src == dst:
            continue
        # choose transfer fraction random between 0 and max_transfer where max_transfer maybe 0.1 of src or 0.1 of total
        max_transfer = best[src] * 0.2  # allow up to 20% of source mass
        eps = rng.random() * max_transfer
        cand = best.copy()
        cand[src] -= eps
        cand[dst] += eps
        # evaluate
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[{time.time() - start_time:.2f}s] hillclimb improved {best_score:.6f}")
        # else maybe accept with prob
    return best, best_score

"""
step_search_v2.py – A faster, more aggressive optimiser for the step‑function
challenge.

Main ideas
----------
1. **Broad random exploration** of the low‑dimensional family
   a_i ∝ r^i·(i+1)^p mixed with a uniform baseline.
2. **Local refinement** of the best parameters with a small‑step Gaussian
   walk (a cheap hill‑climb in the (r,p,λ) space).
3. **Annealed smooth‑max L‑BFGS** (as in the previous version) but with a
   much larger budget of function evaluations and a larger final β.
4. **Targeted hill‑climb** on the exact objective: move mass from indices
   that contribute most to the currently maximal convolution entry to indices
   that contribute the least.  The move size is adaptive (up to 5 % of the
   source mass) and the algorithm accepts only improving moves.
5. **Time‑aware phases** – each part stops before the 1 000 s deadline, so the
   routine never exceeds the allowed wall‑clock budget.

The routine returns a list of non‑negative numbers normalised to unit sum.
"""

import time
import numpy as np

# optional – may be missing in a minimal environment
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Primitive linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT. Returns length 2·len(a)‑1."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  a must be non‑negative."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Surrogate (smooth‑max) objective used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max surrogate of max(conv) together
    with its gradient w.r.t. the soft‑max parameters `z`.
    The surrogate becomes exact as β → ∞.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv      # length for a*a
        self.L_corr = L_corr      # length for w * a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                       # a ≥ 0, Σa = 1
        # --- self‑convolution -------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)           # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()          # smooth‑max weights
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # --- gradient w.r.t. a -------------------------------------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]        # length n

        # --- chain rule through soft‑max ----------------------------------------
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)                    # shape (n,)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Targeted hill‑climb on the exact objective
# ----------------------------------------------------------------------


def _targeted_hill_climb(a: np.ndarray, L_fft: int,
                         deadline: float, start_time: float):
    """
    Move mass from indices that contribute heavily to the current maximal
    convolution entry to indices that contribute little.  Accept only
    strictly improving moves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current convolution and the location of the maximum
        conv = _conv_self_fft(best_a, L_fft)
        max_val = conv.max()
        k = int(conv.argmax())

        # contribution of each index to conv[k]
        idx = np.arange(n)
        j1 = k - idx
        mask1 = (j1 >= 0) & (j1 < n)
        contrib = np.zeros(n)
        contrib[mask1] = best_a[idx[mask1]] * best_a[j1[mask1]]

        j2 = idx - k
        mask2 = (j2 >= 0) & (j2 < n)
        contrib[mask2] += best_a[idx[mask2]] * best_a[j2[mask2]]

        # if everything is zero we are stuck
        if contrib.sum() == 0.0:
            break

        # source – sample preferentially from large contributors
        src = rng.choice(n, p=contrib / contrib.sum())

        # destination – prefer indices with small contribution
        inv = contrib.max() - contrib + 1e-12
        dst = rng.choice(n, p=inv / inv.sum())
        if src == dst:
            continue

        # transfer up to 5 % of the source mass (larger steps help escape)
        eps = best_a[src] * rng.random() * 0.05
        if eps <= 0.0:
            continue

        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps

        # project back onto the simplex (non‑negativity + unit sum)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build a candidate from low‑dimensional parameters
# ----------------------------------------------------------------------


def _make_candidate(r: float, p: float, lam: float, n: int,
                    uniform: np.ndarray) -> np.ndarray | None:
    """
    Returns a normalised candidate vector:
        cand = lam·uniform + (1‑lam)·shape,
    where shape_i = r**i * (i+1)**p   (i = 0 … n‑1).
    If the shape collapses to all zeros (numerical under‑flow) the function
    returns None.
    """
    i = np.arange(n, dtype=np.float64)
    # compute in log‑space to improve stability for extreme p
    with np.errstate(over='ignore', under='ignore', invalid='ignore'):
        log_shape = i * np.log(r) + p * np.log(i + 1.0)
        shape = np.exp(log_shape)
    if not np.isfinite(shape).all():
        return None
    total = shape.sum()
    if total == 0.0:
        return None
    shape /= total
    cand = lam * uniform + (1.0 - lam) * shape
    cand_sum = cand.sum()
    if cand_sum <= 0.0:
        return None
    cand /= cand_sum
    return cand

    # ------------------------------------------------------------------

    sigma_r = 0.005
    sigma_p = 0.2

    for _ in range(15000):

        lam = np.clip(cur_lam + np.random.normal(0.0, sigma_lam), 0.0, 1.0)
        cand = _make_candidate(r, p, lam, n, uniform)
        if cand is None:
            continue
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            cur_r, cur_p, cur_lam = r, p, lam
            print(f"[{time.time() - start_time:.2f}s] refined r={r:.5f} p={p:.2f} λ={lam:.2f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Annealed smooth‑max L‑BFGS optimisation
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 150:
        # initialise soft‑max parameters from the best discrete candidate
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # temperature schedule – grow β gradually
        beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 150:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=L_corr)
            # more function evaluations for the tighter (large‑β) stages
            maxfun = 2000 if beta >= 100.0 else 800
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score, best_seq = real_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.0f} → improved score={real_score:.6f}")

            # warm‑start the next β‑stage
            z0 = z_opt

    # ------------------------------------------------------------------
    # 4️⃣  Targeted hill‑climb on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        a_hc, sc_hc = _targeted_hill_climb(best_seq, L_fft,
                                            deadline, start_time)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  One final short L‑BFGS pass (very high β) for polishing
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 5:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))
        obj = _SurrogateMax(n=n, beta=2000.0,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=L_corr)
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=500,
                disp=False,
                factr=1e7,
            )
            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score, best_seq = real_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] final polish β=2000 → {real_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Post‑process: enforce non‑negativity, normalise to unit sum and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
