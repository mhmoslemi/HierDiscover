"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm is a multi‑phase search inside the 1000 s budget:

1.  Initialise from the global ``height_sequence_1`` (if present) or from the
    uniform distribution.
2.  Broad parametric sweeps (geometric, power, symmetric / asymmetric Beta
    and exponential families).  A few thousand simple candidates are tested.
3.  Random Dirichlet draws with a wide range of concentration parameters.
4.  Smooth‑max continuation using Adam (a differentiable approximation of the
    ``max`` is increased gradually, providing a fast first‑order optimiser).
5.  Aggressive hill‑climbing: repeatedly move a *large* fraction of mass from
    the index with the highest sub‑gradient to the index with the lowest,
    performing a tiny line‑search on the transferred amount.  Occasionally
    random perturbations are added to escape plateaus.

All intermediate best scores are printed.  The final vector is projected onto
the probability simplex (non‑negative, sum = 1) and returned as a plain Python
list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real input)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers (sum is assumed to be 1, i.e. on the simplex)
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _sweep_symmetric_beta(n, best_seq, best_score, deadline):

    log_rev = np.log(np.maximum(n - idx, 1))

    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):

    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Left‑aligned and centred exponential decays."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0

    # left‑aligned
    for lam in np.logspace(-3, 0, 150):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * idx)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp left λ={lam:.3e} score={sc:.6f}")

        # centred version (symmetric)
        seqc = np.exp(-lam * np.abs(idx - centre))
        seqc /= seqc.sum()
        sc = _objective_score(seqc)
        if sc < best_score:
            best_score, best_seq = sc, seqc.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp cent λ={lam:.3e} score={sc:.6f}")

        # mix with uniform (helps fine‑tuning)
        uniform = np.full(n, 1.0 / n)
        for lam_mix in np.linspace(0.0, 1.0, 5):
            mixed = lam_mix * uniform + (1.0 - lam_mix) * seqc
            mixed /= mixed.sum()
            sc = _objective_score(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
                print(f"[{time.time() - start_time:.2f}s] Exp mix λ={lam:.3e} mix={lam_mix:.2f} score={sc:.6f}")

    return best_seq, best_score


# --------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# --------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 250
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.5, 0.8)      # concentration ≈ 0.003–6.3
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# 3️⃣  Smooth‑max continuation (Adam optimiser)
# --------------------------------------------------------------
def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):
    """Adam on the smooth approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 5.0               # initial temperature (smoothness)
    tau_factor = 2.0
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth soft‑max over the convolution entries
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smooth max part
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)   # stay on the simplex

        a = a_new

        # raise temperature every few thousand iterations
        if it % 4000 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# --------------------------------------------------------------
# 4️⃣  Aggressive hill‑climb (large transfers + tiny line‑search)
# --------------------------------------------------------------
def _aggressive_hill_climb(a_start, deadline):
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    # fractions of the source mass we are willing to try moving
    eps_fractions = np.array([0.001, 0.005, 0.01, 0.05, 0.2, 0.5, 1.0])

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_cand = np.where(a > 1e-15)[0]
        if src_cand.size == 0:
            break

        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        # avoid degenerate src==dst
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        max_move = a[src]
        if max_move <= 0:
            break

        # line‑search over a handful of fractions
        best_local_score = best_score
        best_eps = None
        for f in eps_fractions:
            eps = max_move * f
            if eps > a[src]:
                eps = a[src]
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            cur_score = _objective_score(a_prop)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_eps = eps

        if best_eps is not None and best_local_score < best_score - 1e-12:
            # Accept the best move found
            a[src] -= best_eps
            a[dst] += best_eps
            best_score = best_local_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] AggHill score={best_score:.6f}")
        else:
            no_improve += 1
            # occasional random shake‑up
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.02 + 1e-5)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

# Main interface
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ---------------------------------------------------------
    # Determine problem size (use the length of the global starter if available)
    # ---------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.shape[0]
        else:
            raise ValueError
    except Exception:
        n = 4096

    uniform = np.full(n, 1.0 / n)

    # ---------------------------------------------------------
    # Initialise best sequence (global starter if valid)
    # ---------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.ndim == 1 and seq0.size > 0:
            seq0 = np.maximum(seq0, 0.0)
            if seq0.sum() > 0:
                seq0 = seq0 / seq0.sum()
                sc0 = _objective_score(seq0)
                if sc0 < best_score:
                    best_seq, best_score = seq0.copy(), sc0
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc0:.6f}")
    except Exception:
        pass

    # ---------------------------------------------------------
    # 1️⃣ Parametric sweeps
    # ---------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)

    # ---------------------------------------------------------
    # 2️⃣ Random Dirichlet draws
    # ---------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # ---------------------------------------------------------
    # 3️⃣ Smooth‑max Adam optimisation
    # ---------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max_descent(best_seq, phase_deadline,
                                                        best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam score={best_score:.6f}")

    # ---------------------------------------------------------
    # 4️⃣ Aggressive hill‑climb (large mass transfers)
    # ---------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive hill‑climb")
        best_seq, best_score = _aggressive_hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ---------------------------------------------------------
    # Normalise and return
    # ---------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
