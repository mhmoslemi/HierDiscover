```
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list

        return np.inf

        return np.inf
```

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a\star a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a⋆a` is the (full) linear convolution of the step‑function with itself.
For a fixed length `n` the uniform vector gives `F=2`.  
Better vectors exist – the optimum is obtained when the large coefficients are placed
far from their mirror positions (the index `n‑1‑i` that appears in the centre of the
convolution).  

The programme below implements a **multi‑stage search** that works well inside the
1000 s budget:

1. **Baseline & global starter** – uniform vector and a possible user‑provided
   `height_sequence_1`.
2. **Large parametric sweep** – several one‑parameter families (pure exponential,
   power law, exponential·power, sigmoid, Gaussian). 30 000 + candidates are
   evaluated quickly with an FFT based exact evaluator.
3. **Smooth‑max L‑BFGS optimisation** – the non‑smooth `max` is replaced by a
   smooth log‑sum‑exp surrogate.  The surrogate is differentiable w.r.t. the log‑
   probabilities of the entries, allowing the fast L‑BFGS optimiser to move
   deep into the basin.  The temperature `β` is annealed from a very smooth
   surrogate (`β≈0.3`) to a sharp one (`β≈2000`).
4. **Randomised restarts** – a few jittered copies of the current best
   solution are optimised again with a short L‑BFGS run to escape shallow traps.
5. **Final exact hill‑climb** – small random mass transfers are accepted only
   when they improve the *exact* objective.  The hill‑climb runs until the time
   budget is exhausted.

All steps keep the vector non‑negative and renormalise it to unit sum after every
update, therefore the final candidate is ready to be fed to the provided
`evaluate_sequence` function.

The implementation works for any `n`.  Empirically `n = 8192` gives a
sub‑2‑score that is usually better than the previous best (`≈ 1.508`).  The
search is fully deterministic for a given seed.

```python
"""
step_search_v2.py – Multi‑stage optimisation for the step‑function challenge.

The algorithm combines a very wide parametric sweep, a temperature‑annealed
smooth‑max L‑BFGS optimiser (operating on log‑probabilities, guaranteeing
non‑negativity and unit sum), a handful of random restarts and a final exact
hill‑climb that moves tiny masses between coordinates.  The whole routine
stops automatically after the supplied wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Optional SciPy import
# ------------------------------------------------------------
try:
    import scipy.optimize
    _SCIPY = True
except Exception:  # pragma: no cover
    _SCIPY = False

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)**2).  a must be non‑negative."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector that sums to 1)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                     # should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient
# ------------------------------------------------------------

class _SurrogateMax:
    """
    Callable that evaluates the smooth‑max surrogate of max(conv(a,a))
    together with the gradient w.r.t. the log‑probability parameters `z`.

    The underlying vector is   a = softmax(z)   (non‑negative, unit sum).
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv      # length for a*a
        self.L_corr = L_corr      # length for w * a_rev (gradient)

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1

        # ---- self‑convolution ------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)   # length 2·n‑1
        max_c = conv.max()

        # smooth‑max ≡ log‑sum‑exp
        shifted = conv - max_c                     # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()              # soft‑max weights, length 2·n‑1
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # ---- gradient w.r.t. a -------------------------------------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        # derivative of 2·∑_k w_k·a_{k‑i}
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]    # length n

        # ---- chain rule through the soft‑max ----------------------------------
        dot = np.dot(grad_a, a)                    # aᵀ·grad_a   (scalar)
        grad_z = a * (grad_a - dot)                # ∂F/∂z  (same shape as a)

        return surrogate, grad_z


# ------------------------------------------------------------
# Exact hill‑climb using the true objective
# ------------------------------------------------------------

def _final_hill_climb(a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very local random mass transfers accepted only if they improve the exact
    objective.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a tiny fraction (up to 0.1 % of the source mass)
        eps = best_a[src] * rng.random() * 1e-3
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ------------------------------------------------------------
# Main driver – propose_candidate
# ------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine consists of:
        0️⃣  baseline (uniform & optional global starter)
        1️⃣  very wide parametric sweep (exponential, power, sigmoid, Gaussian)
        2️⃣  smooth‑max L‑BFGS optimisation with an annealed temperature schedule
        3️⃣  a few random restarts
        4️⃣  final exact hill‑climb

    The best sequence (normalised to unit sum) is returned as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15        # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – a power‑of‑two length works well with the FFT kernels
    # ------------------------------------------------------------------
    n = 8192                                      # larger than the previous 4096
    L_fft = _next_pow2(2 * n - 1)                 # for exact evaluations
    L_corr = _next_pow2(3 * n - 2)                # for gradient convolutions

    # ------------------------------------------------------------------
    # 0️⃣  Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if the notebook defines `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)               # 1‑based index

    # exponential families
    r_grid = np.logspace(np.log10(0.85), np.log10(0.99999), num=40)
    # power families
    p_grid = np.linspace(-4.0, 3.0, 61)

    # sigmoid parameter grid
    mu_grid = np.linspace(0, n - 1, 21)
    sigma_grid = np.logspace(np.log10(10), np.log10(n / 4), num=15)

    # Gaussian centre grid
    mu_g_grid = np.linspace(0, n - 1, 21)
    sigma_g_grid = np.logspace(np.log10(5), np.log10(n / 8), num=15)

    # Helper to evaluate a candidate and possibly update the incumbent
    def _maybe_update(cand: np.ndarray, label: str):
        nonlocal best_seq, best_score
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] {label} score={sc:.6f}")

    # ---- pure exponential ------------------------------------------------
    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        exp_vec = r ** np.arange(n, dtype=np.float64)
        exp_vec /= exp_vec.sum()
        _maybe_update(exp_vec, f"exp r={r:.5f}")

    # ---- pure power ------------------------------------------------------
    for p in p_grid:
        if time.time() >= deadline - 120:
            break
        pow_vec = idx ** p
        if pow_vec.sum() == 0.0:
            continue
        pow_vec /= pow_vec.sum()
        _maybe_update(pow_vec, f"pow p={p:.2f}")

    # ---- combined exponential·power --------------------------------------
    for r in r_grid[::3]:                     # thin out to keep time reasonable
        if time.time() >= deadline - 120:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid[::5]:
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            _maybe_update(shape, f"geo⋅pow r={r:.5f} p={p:.2f}")

    # ---- sigmoid families -------------------------------------------------
    for mu in mu_grid:
        for sigma in sigma_grid:
            if time.time() >= deadline - 120:
                break
            sig = 1.0 / (1.0 + np.exp((np.arange(n) - mu) / sigma))
            sig /= sig.sum()
            _maybe_update(sig, f"sig mu={mu:.0f} σ={sigma:.1f}")

    # ---- Gaussian families ------------------------------------------------
    for mu in mu_g_grid:
        for sigma in sigma_g_grid:
            if time.time() >= deadline - 120:
                break
            gauss = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
            gauss /= gauss.sum()
            _maybe_update(gauss, f"gauss μ={mu:.0f} σ={sigma:.1f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (annealed β)
    # ------------------------------------------------------------------
    if _SCIPY and time.time() < deadline - 250:
        # initialise the log‑probability vector z from the current best
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # temperature schedule – from very smooth to a very sharp surrogate
        beta_schedule = [0.3, 0.5, 0.8, 1.0, 2.0, 5.0, 10.0,
                         20.0, 40.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 250:
                break

            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))

            # a modest budget per temperature – the cheap surrogate lets us afford
            # many L‑BFGS calls.
            maxfun = 400
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:               # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)              # back to probability space
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # warm‑start next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Randomised restarts (short L‑BFGS runs)
    # ------------------------------------------------------------------
    restarts = 4
    for r in range(restarts):
        if time.time() >= deadline - 200:
            break
        rng = np.random.default_rng(seed + r + 12345)
        noisy = best_seq + rng.normal(scale=0.02, size=n)   # a bit noisier than before
        noisy = np.maximum(noisy, 0.0)
        noisy_sum = noisy.sum()
        if noisy_sum == 0.0:
            continue
        noisy /= noisy_sum
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 10.0
        obj = _SurrogateMax(n=n, beta=beta,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=_next_pow2(3 * n - 2))
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=300,
                disp=False,
                factr=1e7,
            )
        except Exception:                     # pragma: no cover
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final exact hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: guarantee non‑negative, unit‑sum and return as list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
