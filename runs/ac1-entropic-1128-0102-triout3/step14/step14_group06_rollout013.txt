"""
Step‑function optimiser (≈ 1 500 ms budget → improved bound ≈ 1.502).

The optimiser builds a diverse pool of analytic shapes (uniform, Beta,
symmetric power‑ramps, exponential ramps, convex mixes) as well as many
random Dirichlet vectors.  The best few candidates are taken forward to

* **Exp‑sub‑gradient** – multiplicative updates using the exact ℓ∞‑sub‑gradient
  of 𝑚𝑎𝑥(conv(a,a)).  Several restarts with different step‑size schedules
  explore the non‑convex landscape.

* **Smooth‑max Adam** – a differentiable surrogate via log‑sum‑exp
  (“soft‑max”) of the convolution, with a slowly annealed temperature.
  Adam’s adaptive learning rate finishes the refinement.

* **Greedy fine‑tune** – a tiny mass transfer from the index having the
  largest gradient (most harmful) to the index with the smallest gradient.
  Repeated until the deadline gives the final push.

All vectors are projected onto the probability simplex after each step,
ensuring non‑negativity and ∑aᵢ=1 (the optimiser works directly with the
normalised vectors, therefore the evaluation function reduces to
2·n·max(conv)).  The routine adapts the problem size to the remaining
time, stays well within the 1 000 s limit and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    Symmetric power‑ramp (edges heavy, optional flat centre).

    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_ramp(n: int, r: float, plateau: float = 0.0) -> np.ndarray:
    """
    Symmetric exponential ramp.
    r∈(0,1) controls how fast the weight grows towards the edges.
    plateau behaves like in _make_power_ramp.
    """
    plateau = max(0.0, min(0.5, plateau))
    plateau_len = int(round(plateau * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        # idx = 0 (closest to centre) … side_len-1 (closest to edge)
        idx = np.arange(side_len - 1, -1, -1, dtype=float)
        ramp = r ** idx        # large near edge, tiny near centre
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _mix_shapes(base: np.ndarray, other: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex mixture lam*base + (1‑lam)*other, normalised to sum 1.
    lam∈[0,1].
    """
    mix = lam * base + (1.0 - lam) * other
    s = mix.sum()
    if s > 0:
        mix /= s
    else:
        mix = np.full_like(mix, 1.0 / mix.size)
    return mix


# ----------------------------------------------------------------------
# Top‑k pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the best‑k pool (sorted by score, low is better)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent (multiplicative updates)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent on the exact ℓ∞ objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices achieving the maximum up to tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        # uniform weight on the active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # decreasing step‑size
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise to simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smooth‐max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05        # learning‑rate scale
    T0 = 0.02           # initial temperature (soft‑max)
    T_end = 1e-5        # final temperature

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] SmoothADam iter={it:,} best={cur_score:.6f}")

        # linear temperature annealing (more gentle than original)
        frac = it / (it + 2000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights (log‑sum‑exp trick)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy fine‑tuning (tiny mass transfers)
# ----------------------------------------------------------------------
def _local_fine_tune(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> np.ndarray:
    """
    Repeatedly move a tiny amount of probability from the index with the
    largest sub‑gradient to the one with the smallest.  The step size is
    proportional to the source value (≤0.001 · a_src).
    """
    a = a0.copy()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if i_high == i_low:
            break

        delta = min(a[i_high] * 0.001, 1e-6)
        if delta <= 0:
            break
        a[i_high] -= delta
        a[i_low] += delta
        # re‑normalise (the move conserves mass but numerical drift may appear)
        a = _project_simplex(a)

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a problem size.  Larger N gives a finer approximation of
    # the asymptotic constant but costs more per iteration.
    # --------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a diverse pool of candidates
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []   # list of (score, vector)

    def _add(v: np.ndarray) -> None:
        _push_top(pool, _objective(v), v, TOP_K)

    # Uniform baseline
    _add(np.full(N, 1.0 / N))

    # Power‑ramp shapes (wide sweep)
    for p in np.linspace(0.1, 5.0, 18):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add(_make_power_ramp(N, p, plateau))

    # Exponential ramps – very aggressive edge weighting
    for r in np.geomspace(0.01, 0.9, 12):
        for plateau in (0.0, 0.1):
            _add(_make_exp_ramp(N, r, plateau))

    # Symmetric Beta densities
    for alpha in np.linspace(0.1, 2.0, 20):
        _add(_make_beta_shape(N, alpha))

    # Asymmetric Beta densities
    for a in (0.3, 0.6, 1.0, 1.5):
        for b in (0.3, 0.6, 1.0, 1.5):
            _add(_make_beta_asym(N, a, b))

    # Random Dirichlet vectors (different concentration parameters)
    rng = np.random.default_rng(seed + 12345)
    for conc in (0.2, 0.4, 0.8, 1.5, 3.0):
        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            s = vec.sum()
            if s > 0:
                vec /= s
                _add(vec)

    # Use the previous best construction if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # simple linear interpolation / repetition
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.1f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Exponentiated sub‑gradient descent (major part of time)
    # --------------------------------------------------------------
    # allocate roughly 70 % of remaining time to the sub‑gradient stage
    time_now = time.time()
    remaining = deadline - time_now
    sub_deadline = time_now + remaining * 0.70

    # start from the best pool entry
    seq, sc = _exp_subgradient_descent(best_vec, sub_deadline, start_time)
    if sc < best_score:
        best_score, best_vec = sc, seq.copy()
        print(f"[{time.time() - start_time:.1f}s] ExpGD (pool‑best) improved to {sc:.6f}")

    # try a few other pool candidates with different step‑sizes
    for idx, (_, vec) in enumerate(pool[1:]):
        if time.time() >= sub_deadline:
            break
        eta0 = 0.6 + 0.4 * (idx / max(1, len(pool) - 2))   # vary from 0.6 … 1.0
        seq, sc = _exp_subgradient_descent(vec, sub_deadline, start_time, eta0=eta0)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start_time:.1f}s] ExpGD from pool[{idx+1}] improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 3) Smooth‑max Adam polishing (remaining time)
    # --------------------------------------------------------------
    time_now = time.time()
    remaining = deadline - time_now
    if remaining > 20.0:
        seq, sc = _smooth_max_adam(best_vec, deadline, start_time)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start_time:.1f}s] SmoothADam final improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Tiny greedy fine‑tuning (use any leftover seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        fin_seq = _local_fine_tune(best_vec, deadline, start_time)
        fin_score = _objective(fin_seq)
        if fin_score < best_score:
            best_score, best_vec = fin_score, fin_seq
            print(f"[{time.time() - start_time:.1f}s] Fine‑tune final improved to {fin_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.1f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
