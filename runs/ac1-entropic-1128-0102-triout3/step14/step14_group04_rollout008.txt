"""
Propose a low‑score non‑negative step function.

The algorithm works in two stages:

1. **Coarse optimisation on a small problem (n≈256).**
   - Build a diverse pool of candidate windows (uniform, sine‑power,
     exponential, beta, Dirichlet draws, etc.).
   - Pick the best candidate and improve it with a *alternating linear‑
     programming* scheme derived from the classic f–g iteration:
       * given f, solve   max ∑g  s.t.   f∗g ≤ M   (M = max(f∗f)),
       * mix the solution g back into f (small step size) and repeat.
   - The LP is tiny (≤ 511×256) and solved with SciPy's ``highs`` solver.

2. **Fine‑grained polishing on the full size (n≈8192).**
   - Upsample the small‑scale solution to a larger length,
     re‑normalise it to a probability vector.
   - Run a long simulated‑annealing / mass‑swap heuristic that moves
     tiny packets of probability mass between indices while accepting
     uphill moves according to an exponential cooling schedule.
   - Keep the best vector seen during annealing.

Both stages respect the simplex (non‑negative, sum = 1) and the final
vector is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective (no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (sum(seq)=1 ⇒ denominator =1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    # seq is assumed to be normalised to sum==1
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π i/(n‑1))**p (Hann is p=2)."""
    i = np.arange(n)
    shape = np.sin(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0) ** p
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = _make_uniform(n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam≥0)."""
    i = np.arange(n)
    shape = np.exp(-lam * i)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = _make_uniform(n)
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = _make_uniform(n)
    return shape


def _make_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """Two symmetric power ramps with optional flat plateau (α∈[0,0.5])."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Build the convolution matrix A where (A·g)_k = Σ_i f_i g_{k-i}
# (i.e. the linear constraints f∗g ≤ M)
# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]
    return A

# Alternating LP‑based optimisation (small n)
# ----------------------------------------------------------------------
def _alternating_optimize(f0: np.ndarray,
                          iterations: int,
                          t: float,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform the classic f–g alternating scheme on a small problem.
    f0 must already be normalised to sum==1.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = _objective(f)

    for it in range(iterations):
        if time.time() >= deadline:
            break

        # current bound M = max(f∗f)
        M = _conv_self(f).max()

        # construct LP: max Σg  s.t.  A·g ≤ M, g ≥ 0
        A = _build_convolution_matrix(f)
        b = np.full(A.shape[0], M, dtype=float)
        c = -np.ones_like(f)  # linprog minimises, so we use -Σg

        # solve LP with HiGHS (fast for small dense problems)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * f.shape[0],
                      method='highs', options={'presolve': True})

        if not res.success:
            # fallback: keep current f
            break

        g = res.x
        if g.sum() == 0.0:
            break

        # Rescale g to have the same total mass as f (which is 1)
        g = g / g.sum()

        # Mix (small step size to retain stability)
        f = (1.0 - t) * f + t * g
        f = _project_simplex(f)  # enforce exact sum==1

        cur_score = _objective(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:.2f}s] Alternating it={it+1} score={cur_score:.6f}")

    return best_f, best_score

# Light simulated‑annealing polish (large n)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of probability mass between positions.
    Accept uphill moves according to an exponential cooling schedule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 2_000_000  # generous upper bound; actual will stop earlier

    # Cooling: start high (≈1.0), decay to ≈1e‑4
    T0 = 1.0
    Tf = 1e-4

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = T0 * ((Tf / T0) ** (it / max_steps))

        # pick source proportional to its mass (big entries move more)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer a modest fraction (≤30 %) of source mass
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsample a probability vector to a larger length (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vec = np.interp(x_new, x_old, vec)
    new_vec = np.maximum(new_vec, 0.0)
    total = new_vec.sum()
    if total > 0.0:
        new_vec /= total
    else:
        new_vec = np.full(new_n, 1.0 / new_n)
    return new_vec

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ----- 1. Small‑scale work (n≈256) ---------------------------------
    n_small = 256
    pool = []                     # list of (score, vector)
    pool_cap = 12

    def _maybe_add(vec):
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            vec = _make_uniform(vec.size)
        vec /= vec.sum()
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # a few analytic families
    _maybe_add(_make_uniform(n_small))
    _maybe_add(_make_cosine_tip(n_small))
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        _maybe_add(_make_sine_power(n_small, p))
    for lam in [0.001, 0.005, 0.01, 0.02, 0.05]:
        _maybe_add(_make_exponential(n_small, lam))
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        for alpha in np.linspace(0.0, 0.5, 5):
            _maybe_add(_make_power_ramp(n_small, p, alpha))
    for a_par in [0.5, 1.0, 2.0, 4.0]:
        for b_par in [0.5, 1.0, 2.0, 4.0]:
            _maybe_add(_make_beta_shape(n_small, a_par, b_par))

    # random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(400):
        if time.time() >= deadline:
            break
        vec = rng.dirichlet(alpha=np.ones(n_small) * 0.5)
        _maybe_add(vec)

    # try to incorporate a global starter if provided
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter_small = _upsample(starter, n_small)
        _maybe_add(starter_small)
    except Exception:
        pass

    best_score_small, best_vec_small = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Small‑scale pool best score = {best_score_small:.6f}")

    # ----- 2. Alternating LP optimisation (tiny “f–g” loop) ----------
    remaining = deadline - time.time()
    if remaining > 30.0:
        alt_deadline = time.time() + 0.6 * remaining
        best_vec_small, best_score_small = _alternating_optimize(
            best_vec_small,
            iterations=30,
            t=0.15,
            deadline=alt_deadline,
            start_time=start_time)
        print(f"[{time.time() - start_time:.2f}s] After alternation score = {best_score_small:.6f}")

    # ----- 3. Upsample to a larger problem (n≈8192) -----------------
    n_large = 8192
    best_vec_large = _upsample(best_vec_small, n_large)

    # ----- 4. Light sub‑gradient refinement (optional, fast) ----------
    if time.time() < deadline - 20.0:
        # a single short sub‑gradient run (full active set)
        sub_deadline = time.time() + 5.0
        best_vec_large, best_score_large = _alternating_optimize(
            best_vec_large,
            iterations=5,
            t=0.05,
            deadline=sub_deadline,
            start_time=start_time)
        print(f"[{time.time() - start_time:.2f}s] Sub‑grad brief score = {best_score_large:.6f}")

    # ----- 5. Simulated‑annealing polish (main work) -----------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_vec_large, deadline, start_time)
        if anneal_score < best_score_small:
            best_vec_large, best_score_small = anneal_seq, anneal_score
            print(f"[{time.time() - start_time:.2f}s] Anneal final improvement = {anneal_score:.6f}")

    # ----- 6. Normalise & return ---------------------------------------
    best_vec_large = np.maximum(best_vec_large, 0.0)
    total = best_vec_large.sum()
    if total > 0.0:
        best_vec_large /= total
    else:
        best_vec_large = np.full(n_large, 1.0 / n_large)

    final_score = _objective(best_vec_large)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec_large]
