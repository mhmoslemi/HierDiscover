def propose_candidate(seed=0, budget_s=1000, **kwargs):

    start = time.time()

    rng = np.random.default_rng(seed)

    # Choose n.

    n = 8192

    # generate initial candidates
    candidates = []

    def add_candidate(vec):
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total < 1e-12:
            return
        vec = vec / total
        candidates.append(vec)

    # uniform
    add_candidate(np.full(n, 1.0 / n))

def _anneal(seq, deadline, start_time):
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400000
    for it in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        # Simulated annealing temperature schedule
        T = max(1e-6, 0.5 * (0.01 / 0.5) ** (it / max_steps))  # maybe adjust
        src = rng.choice(n, p=a)  # biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 0.5  # up to 50% of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()

"""
Gradient‑descent + annealing optimizer for the step‑function L∞‑objective.

The algorithm:
1. Build a diverse pool of candidate probability vectors (uniform, windows,
   beta / power shapes, random Dirichlet draws, optional global starter).
2. Pick the best initial candidate w.r.t. the exact objective.
3. Refine it with a smooth‑max surrogate:
   –   L(a) = (1/β) log Σ exp(β·conv(a,a))  (β grows from 5 to 80).
   –   Gradient of L is a simple cross‑correlation; we use Adam‑style
      updates and project back onto the probability simplex.
4. Polish the result with a light simulated‑annealing mass‑swap phase.
5. Return the final vector (non‑negative, sum normalised to 1).

The whole search respects a 1000 s budget; intermediate progress is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _valid_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation corr[k] = Σ_j w[k+j]·a[j] for k=0…len(a)-1.
    Returns the “valid” part of length len(a).
    """
    Nw = w.shape[0]
    Na = a.shape[0]
    L = _next_pow2(Nw + Na - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    start = Na - 1
    end = Nw
    return corr_full[start:end]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Adam)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by minimising a smooth approximation of
    max(conv).  Uses Adam, a linearly increasing softmax temperature β,
    and projects onto the simplex after each step.
    Returns the best vector seen and its exact objective.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # β schedule
    beta_start = 5.0
    beta_end = 80.0
    beta_ramp_iters = 3000  # after this β stays at beta_end

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        maxc = conv.max()

        # smooth weights (log‑sum‑exp)
        beta = beta_start + (beta_end - beta_start) * min(it / beta_ramp_iters, 1.0)
        exp_arg = beta * (conv - maxc)          # max term becomes 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smooth surrogate
        grad = 2.0 * _valid_corr(w, a)  # shape (n,)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (decreasing like 1/√t)
        step = 0.3 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

        # occasional exact evaluation
        if it % 20 == 0:
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth descent it={it:,}  best={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Small random mass‑swap moves accepted with a Metropolis rule.
    Works on the exact objective; useful for a final polishing stage.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = max(1e-6, 0.5 * (0.01 / 0.5) ** (it / max_steps))

        src = rng.choice(n, p=a)            # more mass moves faster
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Candidate generators (simple analytic families)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_half_uniform(n: int) -> np.ndarray:
    """Mass uniformly distributed on the first half, zero elsewhere."""
    half = np.zeros(n, dtype=float)
    half[: n // 2] = 2.0 / n   # sum = 1
    return half


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    tip = 1.0 + np.cos(np.pi * i / (n - 1))
    tip = tip / tip.sum()
    return tip


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _make_beta_shape(n: int, a_par: float, b_par: float) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a_par - 1.0)) * ((1.0 - xs) ** (b_par - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

    """Exact objective 2·n·max(conv(seq,seq)) (assumes sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


def _valid_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation corr[k] = Σ_j w[k+j]·a[j] for k=0…len(a)-1.
    Returns the “valid” part of length len(a).
    """
    Nw = w.shape[0]
    Na = a.shape[0]
    L = _next_pow2(Nw + Na - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    start = Na - 1
    end = Nw
    return corr_full[start:end]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Adam)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by minimising a smooth approximation of
    max(conv).  Uses Adam, a linearly increasing temperature β, and
    projects onto the simplex after each step.
    Returns the best vector seen and its exact objective.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_start = 5.0
    beta_end = 80.0
    beta_ramp_iters = 3000          # after this β stays at beta_end

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        maxc = conv.max()

        # smooth weights (log‑sum‑exp)
        beta = beta_start + (beta_end - beta_start) * min(it / beta_ramp_iters, 1.0)
        exp_arg = beta * (conv - maxc)          # max term becomes zero
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of the smooth surrogate
        grad = 2.0 * _valid_corr(w, a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size (decreases like 1/√t)
        step = 0.3 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto the simplex
        a = _project_simplex(a)

        # occasional exact evaluation
        if it % 20 == 0:
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth descent it={it:,}  best={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑swap moves accepted with a Metropolis rule.
    Works on the exact objective; useful for a final polishing stage.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = max(1e-6, 0.5 * (0.01 / 0.5) ** (it / max_steps))

        src = rng.choice(n, p=a)          # larger entries move more often
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple candidate generator family
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_cosine_tip(n: int) -> np.ndarray:
    i = np.arange(n)
    tip = 1.0 + np.cos(np.pi * i / (n - 1))
    return tip / tip.sum()


def _make_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _make_beta_shape(n: int, a_par: float, b_par: float) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a_par - 1.0)) * ((1.0 - xs) ** (b_par - 1.0))
    total = shape.sum()
    return shape / total if total > 0.0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (the vector is normalised to sum ≈ 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Choose a length – a power of two makes the FFT cheap
    # --------------------------------------------------------------
    n = 8192   # 2¹³  (fast FFT, enough degrees of freedom)

    # --------------------------------------------------------------
    # 1) Build a diverse pool of seed vectors
    # --------------------------------------------------------------
    candidates = []

    def _add(vec: np.ndarray) -> None:
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total < 1e-12:
            return
        candidates.append(vec / total)

    # analytic families
    _add(_make_uniform(n))
    _add(_make_cosine_tip(n))
    _add(_make_hann(n))

    # a handful of beta‑distribution shapes
    for a_par in (0.3, 0.7, 1.5, 3.0, 6.0):
        for b_par in (0.3, 0.7, 1.5, 3.0, 6.0):
            _add(_make_beta_shape(n, a_par, b_par))

    # optional global starter (if the notebook defines height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size < n:
                tmp = np.zeros(n, dtype=float)
                tmp[: starter.size] = starter
                starter = tmp
            else:
                starter = starter[:n]
            _add(starter)
    except Exception:
        pass

    # random Dirichlet draws (stop when time runs out or we have enough)
    while time.time() < deadline and len(candidates) < 200:
        alpha = rng.uniform(0.05, 4.0)               # concentration parameter
        vec = rng.dirichlet(np.full(n, alpha))
        _add(vec)

    # --------------------------------------------------------------
    # 2) Pick the best seed according to the exact objective
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for seq in candidates:
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Init best score = {sc:.6f}")

    # --------------------------------------------------------------
    # 3) Refine with smooth‑max gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate the majority of the remaining budget to descent
        desc_deadline = time.time() + 0.65 * remaining
        refined_seq, refined_score = _smooth_descent(best_seq, desc_deadline, start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score, best_seq = annealed_score, annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved to {annealed_score:.6f}")

    # --------------------------------------------------------------
    # 5) Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
