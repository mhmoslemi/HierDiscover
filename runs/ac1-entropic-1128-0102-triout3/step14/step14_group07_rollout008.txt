"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Coarse global search (N≈8 k)** – many random Dirichlet vectors
   (a wide range of concentration parameters) and a grid of analytic
   β‑ and power‑ramps are generated.  The best K≈30 candidates are kept
   according to the scale‑invariant metric  n·max(conv).

2. **Fine refinement (N=65 536)** – the current best coarse candidate is
   up‑sampled to the full length and optimised with an Adam‑type descent
   on the smooth‑max surrogate of the objective.  The optimiser runs as
   long as the global budget allows; a few random perturbations are
   injected when time permits to avoid premature convergence.

3. **Short simulated‑annealing polish** – a final SA run (mass swaps of
   two indices) nudges the solution out of any remaining local minima.

All steps respect the 1000 s budget, use only NumPy and the provided
``evaluate_sequence`` function, and finally return a plain Python list
ready for evaluation.  The implementation is fully vectorised and
operates on the probability simplex (the objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _metric(seq: np.ndarray) -> float:
    """Scale‑invariant metric: n * max(conv(seq,seq))."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(n * conv.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {a≥0, Σa=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Analytic shape generators (always return vectors that sum to 1)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_power_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Symmetric power‑ramp: a^p on each side with an optional flat plateau.
    The vector is normalised to sum 1.
    """
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Simple heap‑like container for the K best candidates
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted list of the k lowest‑score vectors."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Adam on  f_T(a)=T·log∑exp(conv/T)  with exponential temperature cooling."""
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    T_start = 0.30
    T_end = 1e-6
    step0 = 0.15

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature (exponential decay)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # soft‑max weights (stable)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on simplex
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated annealing (mass swaps) – cheap final polish
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = float(2.0 * a.shape[0] * _conv_self(a).max())
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # sample source index proportionally to its weight
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = float(2.0 * n * _conv_self(a_prop).max())
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Upsampling by linear interpolation
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, target_n: int) -> np.ndarray:
    src_n = vec.shape[0]
    if src_n == target_n:
        return vec.copy()
    x_src = np.linspace(0.0, 1.0, src_n, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, target_n, endpoint=False)
    return np.interp(x_tgt, x_src, vec)

    if best_seq.sum() == 0.0:
        best_seq = np.full(N_fine, 1.0 / N_fine)
    else:
        best_seq /= best_seq.sum()

    # allocate the majority of the remaining time to Adam optimisation
    remaining = deadline - time.time()
    adam_deadline = time.time() + max(0.0, remaining * 0.85)

    # main Adam run
    best_seq, best_score = _adam_smooth_max(best_seq, adam_deadline, start_time)

    # occasional random perturbations if time still left (to escape plateaus)
    while time.time() < deadline - 5.0:
        # small random jitter
        jitter = rng.normal(scale=0.02, size=best_seq.shape)
        pert = best_seq + jitter
        pert = np.maximum(pert, 0.0)
        if pert.sum() == 0.0:
            continue
        pert /= pert.sum()
        # run a short Adam on the perturbed start
        sub_deadline = time.time() + 5.0
        pert, score = _adam_smooth_max(pert, sub_deadline, start_time)
        if score < best_score:
            best_score = score
            best_seq = pert.copy()
            print(f"[{time.time() - start_time:.2f}s] Perturb‑Adam improved best to {score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final simulated‑annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.03)

    # ------------------------------------------------------------------
    # 4) Normalise and return the plain‑Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N_fine, 1.0 / N_fine)

    final_score = float(2.0 * best_seq.shape[0] * _conv_self(best_seq).max())
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
