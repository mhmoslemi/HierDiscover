def downsample(seq, target_n):
    N = len(seq)
    if N == target_n:
        return np.array(seq)
    if N < target_n:
        # pad zeros
        out = np.zeros(target_n)
        out[:N] = seq
        return out
    # else N>target_n
    # use reshape if divisible
    # compute bin edges as integer
    out = np.empty(target_n, dtype=float)
    for i in range(target_n):
        start = int(i * N / target_n)
        end = int((i+1) * N / target_n)
        if end > start:
            out[i] = seq[start:end].mean()
        else:
            out[i] = seq[start] # case such that end==start
    return out

"""
Propose a non‑negative step‑function (a probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of two phases:

1️⃣  Frank‑Wolfe style optimisation.
    For the current distribution f we compute M = max(conv(f,f)).
    The linear oracle solves a linear programme

        maximise   Σ b_i
        subject to  (f * b)_k ≤ M   for all shifts k,
                    b_i ≥ 0.

    The solution b tends to concentrate mass where f is small,
    yielding a new extreme point g = b / Σb.  A convex combination
    f ← (1‑t) f + t g (with the usual FW step size t = 2/(it+2))
    strictly reduces the objective while staying on the simplex.

2️⃣  A very cheap hill‑climb refinement.
    Using the sub‑gradient of max(conv(f,f)) we repeatedly transfer a tiny
    amount of mass from the index with the most positive sub‑gradient to the
    one with the most negative, accepting only improving moves.

Both phases respect the 1000 s budget; intermediate best scores are printed.
The final vector is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution / objective helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a·a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Small utility – down‑sample / pad a vector to a target length
# ----------------------------------------------------------------------
def _reshape_to_length(vec: np.ndarray, target: int) -> np.ndarray:
    """Reshape `vec` to length `target` by averaging bins (or padding)."""
    src = vec.astype(float)
    N = src.size
    if N == target:
        return src.copy()
    if N < target:
        out = np.zeros(target, dtype=float)
        out[:N] = src
        return out
    # N > target – block‑average
    out = np.empty(target, dtype=float)
    for i in range(target):
        lo = int(i * N / target)
        hi = int((i + 1) * N / target)
        if hi > lo:
            out[i] = src[lo:hi].mean()
        else:               # should not happen, but guard
            out[i] = src[lo]
    return out


# ----------------------------------------------------------------------
# Build the convolution matrix A such that A @ b = conv(f, b) (full length)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A with rows = 2*len(f)-1 and columns = len(f)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
# Frank‑Wolfe optimisation (LP oracle)
# ----------------------------------------------------------------------
def _frank_wolfe_optimize(init_f: np.ndarray, deadline: float):
    """
    Perform Frank‑Wolfe iterations.
    Returns the best vector seen and its objective value.
    """
    f = init_f.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_score = _objective_score(f)
    it = 0

    while time.time() < deadline:
        conv_ff = _conv_self(f)
        M = conv_ff.max()

        # Build linear constraints A @ b ≤ M
        A = _build_conv_matrix(f)

        # Linear programme: maximise Σ b_i  <=> minimise -Σ b_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        # No improvement possible if sum_b ≤ 1 (the current point is already optimal)
        if sum_b <= 1.0 + 1e-12:
            break

        # New extreme point on the simplex
        g = b / sum_b

        # Standard FW step size
        t = 2.0 / (it + 2.0)
        if t > 1.0:
            t = 1.0

        # Convex combination, stays on the simplex
        f = (1.0 - t) * f + t * g
        f /= f.sum()                     # numerical safety

        it += 1
        cur_score = _objective_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(
                f"[{time.time() - start_time:.2f}s] FW it={it:4d} t={t:.3f} score={cur_score:.6f}"
            )
        # early stop if improvement becomes tiny
        if it >= 200:   # generous safety guard
            # we have already spent enough iterations for this phase
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (mass transfer) refinement
# ----------------------------------------------------------------------
def _hill_climb_refine(start_f: np.ndarray, deadline: float):
    """
    Very cheap hill‑climb that moves a tiny amount of mass from the
    most “harmful’’ coordinate to the most “helpful’’ one.
    """
    f = start_f.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_score = _objective_score(f)
    rng = np.random.default_rng()
    stagnant = 0

    while time.time() < deadline:
        conv = _conv_self(f)
        grad = _subgradient_maxconv(f, conv)

        src_candidates = np.where(f > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # fall back to a random move
            src, dst = rng.choice(n, size=2, replace=False)

        # move a tiny fraction of the source mass
        frac = rng.random() * 0.005 + 1e-6   # ≤0.5 % of the source
        eps = f[src] * frac
        if eps <= 0.0:
            continue

        f_prop = f.copy()
        f_prop[src] -= eps
        f_prop[dst] += eps
        if f_prop[src] < -1e-15:
            continue

        f_prop = np.maximum(f_prop, 0.0)
        f_prop /= f_prop.sum()

        cur_score = _objective_score(f_prop)
        if cur_score < best_score - 1e-12:
            f = f_prop
            best_f = f.copy()
            best_score = cur_score
            stagnant = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={cur_score:.6f}")
        else:
            stagnant += 1
            if stagnant > 5000:
                # occasional random jitter to escape plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = f[i1] * (rng.random() * 0.005 + 1e-6)
                f[i1] -= eps2
                f[i2] += eps2
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stagnant = 0

    return best_f, best_score
