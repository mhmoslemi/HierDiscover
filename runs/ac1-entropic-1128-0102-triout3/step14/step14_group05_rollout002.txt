def _project_simplex(v):
    """Projects an arbitrary vector onto the probability simplex."""
    # Reference: Wang & Carreira-Perpiñán 2013.
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending sort
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a * b using FFT of length L.
    Returns the first len(a)+len(b)‑1 entries.
    """
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[: a.shape[0] + b.shape[0] - 1]


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return _conv_fft(a, a, L)


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact evaluation of the objective
        2·n·max(conv(a,a)) / (sum(a)**2)
    for a non‑negative vector a (sum may be ≈1).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    shift = np.max(z)
    e = np.exp(z - shift)
    tot = e.sum()
    if tot == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / tot


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max (log‑sum‑exp) surrogate of
    max(conv(a,a)) together with its gradient w.r.t. the soft‑max
    parameters `z`.  The vector `z` lives in ℝⁿ and is mapped to a
    probability vector a = softmax(z).
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # size for a*a
        self.L_corr = L_corr          # size for w * a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        # ----- self‑convolution -------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)       # length 2n‑1
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) and its weight vector
        shifted = conv - max_c
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()               # soft‑max over shifts
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # ----- gradient w.r.t. a ------------------------------------------------
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        # slice (n‑1 … 0) → length n
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]

        # ----- chain rule through the soft‑max ----------------------------------
        dot = np.dot(grad_a, a)                # scalar
        grad_z = a * (grad_a - dot)            # ∂f/∂z, same shape as z
        return surrogate, grad_z


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) minimising

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The function respects the given wall‑clock budget and returns the best
    sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # problem size – 2048 provides a good trade‑off between speed and expressive power
    # --------------------------------------------------------------
    n = 2048
    L_fft = _next_pow2(2 * n - 1)          # for exact evaluations
    L_corr = _next_pow2(3 * n - 2)         # for surrogate gradient

    # --------------------------------------------------------------
    # 0️⃣  uniform baseline (and optional global starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # try to use a globally supplied starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            # if length differs, linearly resample to our n
            if starter.size != n:
                x_old = np.linspace(0.0, 1.0, starter.size)
                x_new = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / s
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # --------------------------------------------------------------
    # 1️⃣  Large parametric sweep (geometric‑power families & mixtures)
    # --------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=float)            # 1‑based index
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-4.0, 3.0, 61)
    lam_grid = np.linspace(0.0, 1.0, 6)

    for r in r_grid:
        if time.time() >= deadline - 300:
            break
        geo = np.power(r, np.arange(n, dtype=float))
        for p in p_grid:
            if time.time() >= deadline - 300:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # a few random mixtures of two geometric series
    for _ in range(30):
        if time.time() >= deadline - 300:
            break
        r1 = 0.90 + np.random.rand() * 0.099
        r2 = 0.90 + np.random.rand() * 0.099
        w1 = np.random.rand()
        w2 = 1.0 - w1
        geo1 = np.power(r1, np.arange(n))
        geo2 = np.power(r2, np.arange(n))
        shape = w1 * geo1 + w2 * geo2
        shape /= shape.sum()
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = shape.copy()
            print(f"[{time.time() - start_time:.2f}s] random mixture score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Annealed L‑BFGS on the smooth‑max surrogate
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 200:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # temperature schedule – increasing β makes the surrogate sharper
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 200:
                break
            obj = _SurrogateMax(
                n=n,
                beta=beta,
                L_conv=_next_pow2(2 * n - 1),
                L_corr=_next_pow2(3 * n - 2),
            )
            # a generous budget for each temperature
            maxfun = 2000
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → real score={real_score:.6f}")

            # warm‑start next temperature with the current solution
            z0 = z_opt

    # --------------------------------------------------------------
    # 3️⃣  A few random L‑BFGS restarts (stronger perturbations)
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 150:
        restarts = 5
        for r in range(restarts):
            if time.time() >= deadline - 150:
                break
            rng = np.random.default_rng(seed + r + 12345)
            noise = rng.normal(scale=0.02, size=n)
            noisy = best_seq + noise
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            z0 = np.log(np.maximum(noisy, 1e-12))
            beta = 20.0
            obj = _SurrogateMax(
                n=n,
                beta=beta,
                L_conv=_next_pow2(2 * n - 1),
                L_corr=_next_pow2(3 * n - 2),
            )
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=1500,
                    disp=False,
                    factr=1e7,
                )
            except Exception:
                continue
            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} score={real_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    def _subgradient_descent(a_start):
        a = a_start.copy()
        best_local = a.copy()
        best_local_score = _eval_fast(a, L_fft)
        itr = 0
        # continue until only a few seconds are left for the final hill‑climb
        while time.time() < deadline - 30:
            # self‑convolution and maximum value
            conv = _conv_self_fft(a, L_fft)               # length 2n‑1
            M = conv.max()
            # active shifts (allow a tiny tolerance)
            eps_active = max(M * 1e-12, 1e-14)
            w = (conv >= M - eps_active).astype(float)    # indicator vector
            num_active = w.sum()
            if num_active == 0:
                break

            # convolution w * a_rev to obtain Σ_k a_{k‑i}
            a_rev = a[::-1]
            conv_wa = _conv_fft(w, a_rev, L_corr)         # length 3n‑2
            sum_shift = conv_wa[n - 1 : n - 1 + n]         # length n
            avg_shift = sum_shift / num_active

            # sub‑gradient of the exact objective (see analysis)
            grad = 4.0 * n * (avg_shift - M)

            # diminishing step size
            step = 0.5 / np.sqrt(itr + 1)
            a_candidate = a - step * grad
            a = _project_simplex(a_candidate)

            sc = _eval_fast(a, L_fft)
            if sc < best_local_score:
                best_local_score = sc
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] subgrad iter {itr} → score={sc:.6f}")

            itr += 1
        return best_local, best_local_score

    if time.time() < deadline - 30:
        a_sgd, sc_sgd = _subgradient_descent(best_seq)
        if sc_sgd < best_score:
            best_score = sc_sgd
            best_seq = a_sgd.copy()
            print(f"[{time.time() - start_time:.2f}s] after sub‑gradient best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final local hill‑climb (larger transfers)
    # --------------------------------------------------------------
    def _final_hill_climb(a, time_limit):
        n_local = a.shape[0]
        best = a.copy()
        best_sc = _eval_fast(best, L_fft)
        rng = np.random.default_rng()
        while time.time() < time_limit:
            i = rng.integers(n_local)
            if best[i] < 1e-15:
                continue
            j = rng.integers(n_local)
            if i == j:
                continue
            # move up to 5 % of the source mass
            eps = best[i] * rng.random() * 0.05
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            # projected back onto simplex (preserves sum)
            cand = _project_simplex(cand)
            sc = _eval_fast(cand, L_fft)
            if sc < best_sc:
                best_sc = sc
                best = cand
                print(f"[{time.time() - start_time:.2f}s] hill‑climb improved score={best_sc:.6f}")
        return best, best_sc

    if time.time() < deadline - 5:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑process – ensure non‑negativity and unit sum
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
