"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

The optimiser is a three‑stage heuristic:

1. **Global exploration** – random Dirichlet draws and a fast
   differential‑evolution (DE) search over a low‑dimensional spline
   (12 knots) that parameterises the whole step‑function.
2. **Smooth‑max gradient descent** – Adam optimisation of the
   log‑sum‑exp surrogate of the max‑convolution (temperature τ is
   increased gradually).  The iterate is kept on the probability simplex.
3. **Fine‑grained hill‑climb** – simulated‑annealing style moves that
   transfer a tiny fraction of mass from the index with the largest
   positive sub‑gradient to the one with the most negative,
   with a temperature schedule that slowly freezes.

All stages respect the 1000 s wall‑time budget, keep the sequence
non‑negative and normalise it to unit sum (the objective is homogeneous
of degree 0, so any scaling is irrelevant).  The code uses FFT‑based
convolutions (O(n log n)) and a Euclidean simplex projection (O(n)).
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Self‑convolution a * a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score used in the optimiser: 2·n·max(conv(seq,seq)) (∑seq=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max‑convolution (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Global exploration (Dirichlet + DE over spline knots)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # draw a concentration from a log‑uniform range
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _de_spline_opt(n: int, best_seq: np.ndarray,
                   best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Differential‑evolution over a 12‑knot piecewise‑linear spline."""
    K = 12                     # number of knots – a sweet spot between flexibility and speed
    positions = np.linspace(0, n - 1, K)

    # objective for DE – higher‑level shape → full sequence → normalise → score
    def de_obj(x):
        if np.any(x < 0):
            return np.inf
        # linear interpolation
        seq = np.interp(np.arange(n), positions, x)
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0:
            return np.inf
        seq /= total
        return _objective_score(seq)

    # bounds: knots are non‑negative, we cap them at a modest value (10) – scaling is irrelevant
    bounds = [(0.0, 10.0) for _ in range(K)]

    # time‑controlled callback
    def de_callback(xk, convergence):
        if time.time() >= deadline:
            return True
        return False

    # run DE – we keep the iteration numbers modest; the callback will stop earlier if needed
    result = differential_evolution(
        de_obj,
        bounds,
        maxiter=200,
        polish=False,
        updating='deferred',
        seed=seed,
        callback=de_callback,
        disp=False,
    )
    # construct the sequence from the best DE candidate
    best_de = result.x
    seq = np.interp(np.arange(n), positions, best_de)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] DE spline score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02
    tau = 8.0
    tau_factor = 1.15
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth max (log‑sum‑exp)
        mval = conv.max()
        w_raw = np.exp(tau * (conv - mval))
        w = w_raw / w_raw.sum()

        # gradient of smooth max
        a_rev = a[::-1]
        grad = 2.0 * _conv_fft(w, a_rev)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # temperature schedule for the surrogate
        if it % 500 == 0:
            tau = min(tau * tau_factor, 1e7)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final fine‑grained hill‑climb / simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """Directed hill‑climb with a simple annealing schedule."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    # annealing parameters
    total_hc = max(0.0, deadline - time.time())
    if total_hc <= 0:
        return best_seq, best_score
    T_start = 0.02          # relative to score magnitude
    T_end = 1e-6
    frac_start = 0.04       # up to 4 % of source mass initially
    frac_end = 1e-5

    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear decay of temperature and transfer fraction
        T = T_start * (remaining / total_hc) + T_end * (1 - remaining / total_hc)
        frac = frac_start * (remaining / total_hc) + frac_end * (1 - remaining / total_hc)

        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source – highest positive sub‑gradient among indices that actually own mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – most negative sub‑gradient
        dst = int(np.argmin(grad))

        if src == dst:
            # fallback random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        eps = a[src] * (rng.random() * frac + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        delta = new_score - cur_score

        if delta < -1e-12:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T, 1e-12))

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine comprises:
    1) a quick random Dirichlet sweep,
    2) a DE optimisation over a 12‑knot spline,
    3) Adam optimisation of a smooth‑max surrogate,
    4) a final hill‑climb with simulated‑annealing flavour.
    All work respects the given time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # tiny safety margin

    # problem size
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---- baseline (uniform) ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ---- optional global starter (height_sequence_1) ----
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Random Dirichlet draws (wide concentration range)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0  # leave room for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣ Differential Evolution on a low‑dimensional spline
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _de_spline_opt(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣ Adam optimisation of the smooth‑max surrogate
    # ------------------------------------------------------------------
    # Reserve ~130 s for the final hill‑climb
    hill_time = 130.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
