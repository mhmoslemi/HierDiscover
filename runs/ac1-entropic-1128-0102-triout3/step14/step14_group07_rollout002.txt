def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random
    start = time.time()
    N = 1024
    best_seq = np.full(N, 1.0/N)
    best_score = evaluate_sequence(best_seq.tolist())
    # random hill climbing for some time
    while time.time() - start < budget_s - 0.1:
        # generate random Dirichlet
        vec = np.random.default_rng(seed).gamma(0.5, 1.0, size=N)
        vec /= vec.sum()
        sc = evaluate_sequence(vec.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vec
    return best_seq.tolist()
