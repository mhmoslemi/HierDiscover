def adam_descent(a0, deadline, start_time):
    n = a0.shape[0]
    L = _next_pow2(2*n-1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # gradient
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
        # g already twice as needed (since they added symmetric). So correct.

        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # step size schedule: alpha decaying
        alpha0 = 0.3
        alpha = alpha0 / (it**0.4)  # slower decay
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        # small negative values may appear; project onto simplex
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()

"""
Propose a non‑negative step‑function (vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The optimiser works in four phases within the 1000 s budget:

1️⃣  Coarse search over two simple families of shapes:
    – monotone (a_i ∝ r**i·(i+1)**p)
    – edge‑concentrated symmetric (a_i ∝ r**min(i,n‑1‑i)·(min(i,…)+1)**p)
    Each shape is mixed with the uniform distribution (parameter λ).
2️⃣  Fine hill‑climb on the three continuous parameters (r, p, λ) of the
    best family found in (1).
3️⃣  A long projected‑Adam descent on the true objective.
    The gradient is the sub‑gradient of the currently maximal convolution
    entry; after each Adam step the vector is projected onto the probability
    simplex.
4️⃣  Final deterministic pairwise mass‑transfer guided by the gradient:
    mass is moved from indices with large gradient (increase the max‑conv)
    to those with small gradient, always accepting improvements.

All phases respect the overall time budget; intermediate scores and timestamps
are printed for monitoring.  The returned list sums to 1 (and therefore the
evaluation function sees sum≈1).  The routine also uses a globally supplied
starter (height_sequence_1) when available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _make_shape_monotonic(n: int, r: float, p: float) -> np.ndarray:
    """a_i ∝ r**i * (i+1)**p, i = 0…n‑1."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)               # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_shape_symmetric(n: int, r: float, p: float) -> np.ndarray:
    """a_i ∝ r**min(i,n‑1‑i) * (min(i,…)+1)**p, symmetric w.r.t. edges."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)               # distance to nearest edge
    log_shape = dist * np.log(r) + p * np.log(dist + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    # re‑normalise to wipe numerical drift
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Gradient / sub‑gradient helpers
# ----------------------------------------------------------------------


def _max_shift_and_gradient(a: np.ndarray) -> tuple[int, np.ndarray]:
    """
    Locate the shift k attaining max(conv(a,a)) and return the sub‑gradient
    g_i = ∂/∂a_i max_conv = 2·a_{k‑i} (zero outside the support).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    k = int(np.argmax(conv))

    # compute gradient efficiently (the same logic as in the reference code)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_vals = k - i_vals
    g[i_vals] += a[j_vals]
    mask = i_vals != j_vals
    if mask.any():
        g[j_vals[mask]] += a[i_vals[mask]]
    # the loop above already accounts for the factor 2
    return k, g


# ----------------------------------------------------------------------
# Projected Adam descent on the true objective
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the max‑convolution objective.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    # base learning rate – tuned experimentally
    lr0 = 0.30

    while time.time() < deadline:
        it += 1
        k, g = _max_shift_and_gradient(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # slowly decaying effective step size
        lr = lr0 / (it ** 0.4)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")
        else:
            # occasional uphill move to avoid stagnation
            prob = np.exp(-(cur_score - best_score) * 5.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

    return best_a, best_score


# ----------------------------------------------------------------------
# A tiny gradient‑guided pairwise mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _pairwise_local_search(a: np.ndarray,
                           deadline: float,
                           start_time: float,
                           best_score_initial: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from a high‑gradient index to a
    low‑gradient one, accepting only improving moves.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    cur = a.copy()
    best_a = cur.copy()
    best_score = best_score_initial

    while time.time() < deadline:
        # gradient of the currently active shift
        _, g = _max_shift_and_gradient(cur)

        src = int(np.argmax(g))        # location that hurts the objective most
        dst = int(np.argmin(g))        # location that helps most

        if cur[src] < 1e-12:
            # fallback to a random source when the chosen one is empty
            src = rng.integers(n)
            if cur[src] < 1e-12:
                continue

        # transfer up to 0.2 % of the source mass
        eps = cur[src] * rng.random() * 2e-3
        prop = cur.copy()
        prop[src] -= eps
        prop[dst] += eps
        prop = _project_simplex(prop)

        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = prop.copy()
            cur = prop
            print(f"[{time.time() - start_time:.2f}s] pairwise move score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    evaluation function.  The strategy combines a coarse parametric
    sweep, a hill‑climb on the parameters, a long Adam descent
    on the exact objective and a final gradient‑guided local search.
    The routine respects the total time budget and returns the best
    vector found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin before the 1000 s hard limit
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Problem size – 4096 provides a good trade‑off between quality
    # and speed (FFT of length 8192 fits comfortably in memory).
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _eval_fast(starter, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep over (r, p, λ) for both families
    # --------------------------------------------------------------
    sweep_deadline = start_time + 130.0      # leave enough time for later phases
    r_grid = np.logspace(np.log10(0.85), np.log10(0.9995), num=24)
    p_grid = np.linspace(-3.0, 2.0, 31)
    lam_grid = np.linspace(0.0, 0.6, 13)

    for family in ("mono", "sym"):
        if time.time() >= sweep_deadline:
            break
        for r in r_grid:
            for p in p_grid:
                # generate the basic shape
                if family == "mono":
                    base = _make_shape_monotonic(n, r, p)
                else:
                    base = _make_shape_symmetric(n, r, p)

                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep {family} r={r:.4f} p={p:.3f} λ={lam:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the three parameters (r, p, λ)
    # --------------------------------------------------------------
    # Initialise from the best shape family discovered in the sweep.
    # If the best vector came from the monotone family we approximate its
    # parameters; otherwise we start from a safe centre.
    cur_r = 0.97
    cur_p = 0.0
    cur_lam = 0.30
    shape_family = "mono"          # fall‑back – can be changed later if desired

    param_deadline = deadline - 300.0   # reserve ~300 s for Adam & final search
    sigma_r = 0.018
    sigma_p = 0.25
    sigma_lam = 0.15

    cur_seq = best_seq.copy()
    cur_score = best_score

    while time.time() < param_deadline:
        # propose new parameters (log‑normal for r to enforce (0,1))
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))
        # p is unrestricted

        if shape_family == "mono":
            shape_new = _make_shape_monotonic(n, r_new, p_new)
        else:
            shape_new = _make_shape_symmetric(n, r_new, p_new)

        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_lam = r_new, p_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param‑hill {shape_family} r={r_new:.6f} p={p_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        # slowly shrink the perturbation scales
        sigma_r *= 0.9996
        sigma_p *= 0.9996
        sigma_lam *= 0.9996

        # occasional random jitter
        if rng.random() < 0.006:
            eps = 1e-4 * rng.standard_normal(n) * cur_seq
            cur_seq = np.maximum(cur_seq + eps, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    # --------------------------------------------------------------
    # 3️⃣  Projected Adam descent on the exact objective
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:   # keep ~2 min for the final refinement
        adam_deadline = deadline - 120.0
        a_adam, sc_adam = _adam_descent(best_seq, adam_deadline,
                                        start_time, best_score, rng)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam final score={sc_adam:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Gradient‑guided pairwise local search (few minutes)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        a_final, sc_final = _pairwise_local_search(best_seq,
                                                   final_deadline,
                                                   start_time,
                                                   best_score,
                                                   rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Pairwise final score={sc_final:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
