"""
Step‑function optimiser – L2‑flattening + sub‑gradient + Adam polishing.

The algorithm works in three phases:

1. **Diverse seeding** – a small pool of analytic shapes (beta, power‑ramps,
   uniform, random Dirichlet) and the previous best solution (if any).  The
   best few candidates (by the exact objective) are kept.

2. **L2‑uniform descent** – we minimise the L2 distance between the
   self‑convolution and the (theoretical) uniform distribution.
   The loss  ‖a∗a – 1/(2n−1)‖₂² is smooth; its gradient is obtained by a single
   FFT‑based correlation.  An Adam optimiser together with a Euclidean
   projection onto the probability simplex drives the shape toward a flat
   sum distribution, which dramatically lowers the ℓ∞‑objective.

3. **Fine‑tuning** – a short exponentiated‑gradient descent on the exact
   ℓ∞‑objective followed by a smooth‑max (soft‑max) Adam descent.

Finally the vector is up‑sampled to the required length (65536), re‑normalised
and returned as a plain Python list.  The whole search respects the given
time budget (default 1000 s) and prints occasional status updates.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (same as the reference implementation)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool (keeps the lowest scores)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool, sorted by increasing score."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# L2‑uniform descent (Adam on ‖a∗a – 1/(2n−1)‖₂²)
# ----------------------------------------------------------------------
def _l2_uniform_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Flatten the self‑convolution toward a uniform distribution."""
    a = a0.copy()
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution and deviation from uniform target
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv          # sum(a)=1 ⇒ denominator =1
        diff = conv - avg                     # length 2n‑1

        # gradient of L2 loss: 4·(diff ∗ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        D = np.fft.rfft(diff, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(D * A_rev, n=L)[: 3 * n - 2]
        grad = 4.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_simplex(a)

        if it % 300 == 0:
            # occasional score check
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] L2 iter={it:,} best={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (exact ℓ∞ objective)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative update based on the active set of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices achieving (almost) the max)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2·(w ∗ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # decreasing step‑size
        eta0 = 0.8
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_sc = _objective(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – polishing stage
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on the smooth‑max surrogate of the ℓ∞‑objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05
    T0 = 0.02
    T_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_sc:.6f}")

        # temperature schedule (linear annealing)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w ∗ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Resize vec to length new_n by linear interpolation and renormalise."""
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    The routine builds a small, diverse pool of candidate shapes,
    launches a heavy L2‑flattening descent (Adam) on the best one,
    refines with exponentiated‑gradient and smooth‑max phases,
    and finally upsamples to the canonical length of 65536.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a working size (smaller sizes allow more iterations)
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 700:
        work_N = 65536
    elif remaining > 400:
        work_N = 32768
    else:
        work_N = 16384

    # ------------------------------------------------------------------
    # 1) Build a modest pool of diverse seeds
    # ------------------------------------------------------------------
    TOP_K = 5
    top = []  # (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform
    _add(np.full(work_N, 1.0 / work_N))

    # symmetric Beta shapes
    for alpha in np.linspace(0.2, 1.2, 20):
        _add(_make_beta_shape(work_N, alpha))

    # asymmetric Beta shapes
    for a in [0.3, 0.6, 0.9, 1.2]:
        for b in [0.3, 0.6, 0.9, 1.2]:
            _add(_make_beta_shape_asym(work_N, a, b))

    # power‑ramp shapes
    for p in np.linspace(0.2, 3.0, 8):
        for plateau in np.linspace(0.0, 0.4, 3):
            _add(_make_shape(work_N, p, plateau))

    # random Dirichlet draws with a few concentration values
    rng = np.random.default_rng(seed + 123456)
    for concentration in [0.2, 0.5, 0.8, 1.2]:
        for _ in range(30):
            if time.time() >= deadline:
                break
            draws = rng.gamma(concentration, 1.0, size=work_N)
            if draws.sum() == 0:
                continue
            draws /= draws.sum()
            _add(draws)

    # incorporate the historic best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # fit it to the current work size
            if prev.size != work_N:
                if work_N % prev.size == 0:
                    factor = work_N // prev.size
                    prev_up = np.repeat(prev, factor)
                else:
                    prev_up = _upsample(prev, work_N)
            else:
                prev_up = prev.copy()
            prev_up = np.maximum(prev_up, 0.0)
            if prev_up.sum() > 0:
                prev_up /= prev_up.sum()
                _add(prev_up)
    except Exception:
        pass

    # Keep the best from the pool as the starting point
    best_seq = top[0][1].copy()
    best_score = top[0][0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) L2‑uniform descent (major part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                         # ensure we have SOME time left afterwards
        l2_end = time.time() + remaining * 0.70
        seq, sc = _l2_uniform_descent(best_seq, l2_end, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] L2 descent improved best = {sc:.6f}")

        # try a couple of other top‑k seeds if time permits
        for _, vec in top[1:]:
            if time.time() >= l2_end:
                break
            seq, sc = _l2_uniform_descent(vec, l2_end, start_time)
            if sc < best_score:
                best_seq, best_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] L2 from pool improved best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Exponentiated sub‑gradient descent (fine‑tune exact ℓ∞)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        exp_end = time.time() + remaining * 0.5
        seq, sc = _exp_subgradient_descent(best_seq, exp_end, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD improved best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Smooth‑max Adam polishing (use whatever tiny budget is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        seq, sc = _smooth_gradient_descent(best_seq, deadline, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Upsample to the required final length (65536)
    # ------------------------------------------------------------------
    final_N = 65536
    if best_seq.shape[0] != final_N:
        best_seq = _upsample(best_seq, final_N)

        # a very short final polishing (≈5 s) if we still have time
        if time.time() < deadline - 5.0:
            seq, sc = _smooth_gradient_descent(best_seq, deadline, start_time)
            if sc < best_score:
                best_seq, best_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] Final smooth polishing improved best = {sc:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_N, 1.0 / final_N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
