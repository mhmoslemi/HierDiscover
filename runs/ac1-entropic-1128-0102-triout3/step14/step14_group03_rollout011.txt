"""
Propose a step‑function (non‑negative vector) minimizing

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of several independent phases, all respecting a 1000 s
budget.

1️⃣  **Rich parametric sweep** – we explore three families of shapes
    (geometric × power, symmetric power, exponential from the ends) and
    mix each with the uniform distribution.  The best candidate is kept.

2️⃣  **Random Dirichlet restarts** – a large number of random probability
    vectors (different concentration parameters) are sampled and evaluated.

3️⃣  **Smoothed gradient descent** – we minimise a smooth approximation of
    the max‑convolution obtained via a log‑sum‑exp.  The temperature
    β is increased gradually, providing a deterministic analogue of simulated
    annealing.  The iterate is projected onto the probability simplex at
    each step.

4️⃣  **Exact sub‑gradient descent** – starting from the best vector found so
    far we run a classic projected sub‑gradient method on the true
    L∞‑objective (the maximal entry of the self‑convolution).

5️⃣  **Tiny random hill‑climb** – final mass‑transfer moves between random
    indices polish the solution.

All intermediate results are printed with timestamps; the best sequence
found is returned as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Shape families

def _make_geo_power_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Shape ∝ r**i * (i+1)**p   (i = 0…n‑1)
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)            # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sym_power_shape(n: int, p: float, q: float) -> np.ndarray:
    """
    Symmetric shape ∝ (i+1)**p * (n-i)**q  (i = 0…n‑1)
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_end_exp_shape(n: int, r: float) -> np.ndarray:
    """
    Exponential decay from both ends:
    shape_i ∝ r**min(i, n‑1‑i)
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    log_shape = d * np.log(r)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return λ·uniform + (1‑λ)·shape (both already sum to 1).
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()          # guard against tiny rounding drift
    return mixed

def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth proxy of the L∞‑objective:
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))

    β is increased gradually (annealing).  After each step the iterate is
    projected onto the simplex.
    """
    n = a0.shape[0]
    L_self = _next_pow2(2 * n - 1)           # for a*a
    L_grad = _next_pow2(3 * n - 2)           # for a*w
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    beta = 0.5
    beta_max = 5000.0
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1

        # ----- compute convolution a*a -----
        conv = _conv_self_fft(a, L_self)            # length 2n‑1

        # ----- softmax weights over the convolution -----
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)    # stabilised
        Z = exp_scaled.sum()
        w = exp_scaled / Z                         # length 2n‑1

        # ----- gradient = 2·(a * w) (central n entries) -----
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w, n=L_grad)
        conv_full = np.fft.irfft(Af * Wf, n=L_grad)  # length n+2n‑2 = 3n‑2
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]    # central block

        # ----- step & projection -----
        step = step0 / np.sqrt(it)                  # simple diminishing step
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,} β={beta:.1f} score={cur_score:.6f}")
        else:
            # occasional uphill move (temperature = 1/β)
            prob = np.exp(-(cur_score - best_score) * (1.0 / (beta + 1.0)))
            if rng.random() < prob:
                a = a_candidate

        # increase β slowly
        beta = min(beta * 1.001, beta_max)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true L∞‑objective.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # construct sub‑gradient for shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # step size – slower decay than 1/√it
        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter occasionally
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # --- symmetric power --------------------------------------------------
    p_grid2 = np.linspace(-2.0, 2.0, 21)
    q_grid2 = np.linspace(-2.0, 2.0, 21)
    for p in p_grid2:
        for q in q_grid2:
            shape = _make_sym_power_shape(n, p, q)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep sym p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    # --- exponential from ends --------------------------------------------
    r_grid2 = np.logspace(np.log10(0.90), np.log10(0.9999), num=20)
    for r in r_grid2:
        shape = _make_end_exp_shape(n, r)
        for lam in lam_grid:
            cand = _mix_with_uniform(shape, lam)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep end‑exp r={r:.4f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet restarts (diverse concentration parameters)
    # ------------------------------------------------------------------
    dirichlet_deadline = start_time + 250.0
    alphas = [0.1, 0.3, 0.6, 1.0, 2.0, 5.0]
    while time.time() < dirichlet_deadline:
        alpha = float(rng.choice(alphas))
        # Gamma sampling -> Dirichlet
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        cand = g / g.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] dirichlet α={alpha:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed gradient descent (log‑sum‑exp proxy)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        smooth_deadline = deadline - 200.0
        a_smooth, sc_smooth = _smooth_gradient_descent(best_seq, smooth_deadline,
                                                       start_time, best_score)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad final score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        sub_deadline = deadline - 80.0
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final random mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        best_a = a.copy()
        best_sc = _eval_fast(a, L_fft)
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move a tiny fraction of the source mass
            eps = a[src] * rng.random() * 2e-3
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_sc = _eval_fast(a_prop, L_fft)
            if new_sc < best_sc:
                a = a_prop
                best_sc = new_sc
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={best_sc:.6f}")

        best_seq = best_a
        best_score = best_sc

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
