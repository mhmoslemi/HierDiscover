"""
Propose a non‑negative step function (probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  Generate a diverse pool of seed vectors by sampling parametric families
    (geometric, power, Beta‑like, semicircle, logistic) and many random
    Dirichlet draws.  All vectors are projected onto the probability simplex
    and the best few (lowest scores) are kept.

2️⃣  From each seed we launch a fast sub‑gradient descent on the
    non‑smooth objective.  The sub‑gradient of `max(conv(a,a))` is obtained
    analytically; a step is taken opposite to this gradient, the result is
    projected back onto the simplex (Euclidean projection).  A diminishing
    step size (η₀/√t) is used and the best improvement over the whole run is
    retained.

3️⃣  A short directed hill‑climb finishes the optimisation by repeatedly
    moving a tiny amount of mass from the coordinate with the largest positive
    sub‑gradient to the one with the most negative sub‑gradient.

All stages respect the 1000 s budget; intermediate best scores are printed.
The returned value is a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np
import heapq

# --------------------------------------------------------------
# FFT utilities (full convolution)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                       # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers
# --------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# --------------------------------------------------------------
# Simple deterministic families (geometric, power, beta, etc.)
# --------------------------------------------------------------
def _geometric(n, r):
    idx = np.arange(n, dtype=np.float64)
    seq = r ** idx
    return seq / seq.sum()


def _power(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** p
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


def _beta(n, alpha, beta):
    """Discrete Beta‑like shape: (i+1)^{α-1} (n‑i)^{β-1}."""
    idx = np.arange(n, dtype=np.float64)
    # work in log‑space to avoid overflow
    log_seq = (alpha - 1.0) * np.log(idx + 1.0) + (beta - 1.0) * np.log(np.maximum(n - idx, 1.0))
    seq = np.exp(log_seq - np.max(log_seq))
    s = seq.sum()
    if s == 0 or not np.isfinite(s):
        return None
    return seq / s


def _semicircle(n):
    """a_i ∝ sqrt(1 – (2i/(n‑1) – 1)^2)  (a discrete semi‑circle)."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0
    vals = np.sqrt(np.maximum(0.0, 1.0 - t * t))
    s = vals.sum()
    if s == 0:
        return None
    return vals / s


def _logistic(n, k, mu):
    """a_i ∝ 1 / (1 + exp(k·(i‑μ)))  (logistic shape)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / (1.0 + np.exp(k * (i - mu)))
    s = vals.sum()
    if s == 0:
        return None
    return vals / s


# --------------------------------------------------------------
# Sub‑gradient descent (Euclidean projection)
# --------------------------------------------------------------
def _sg_descent(a0: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float):
    """Simple sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    n = a.shape[0]
    step0 = 0.2            # initial step magnitude
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution term
        grad = _subgradient_maxconv(a, conv)

        # make the gradient lie in the tangent space of the simplex
        grad -= grad.mean()

        # diminishing step size
        step = step0 / np.sqrt(it)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    return best_seq, best_score


# --------------------------------------------------------------
# Small hill‑climb (transfer tiny mass from src→dst)
# --------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # ensure we move something
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # transfer a tiny fraction of mass
        frac = rng.random() * 0.005 + 1e-6   # up to 0.5 % of source
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill climb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                # occasional random shake‑up
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.005 + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation metric.
    The method creates many random / parametric seeds, runs a fast
    sub‑gradient descent from each and finishes with a short hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ----------------------------------------------------------
    # Problem size – keep the same as in the reference solution
    # ----------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ----------------------------------------------------------
    # 1️⃣  Build a pool of promising seeds
    # ----------------------------------------------------------
    seed_pool = []                # min‑heap of (–score, seq) keeping the K best
    K = 15                        # number of seeds we will descend from

    def _maybe_push(seq, score):
        """Maintain a heap of the K lowest scores seen so far."""
        if len(seed_pool) < K:
            heapq.heappush(seed_pool, (-score, seq.copy()))
        else:
            # heap root holds the *worst* (largest score) because of minus sign
            if -seed_pool[0][0] > score:
                heapq.heapreplace(seed_pool, (-score, seq.copy()))

    # ----- deterministic families -------------------------------------------------
    # geometric families
    for r in [0.85, 0.90, 0.95, 0.97, 0.99]:
        seq = _geometric(n, r)
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geometric r={r:.2f} new best {sc:.6f}")
        _maybe_push(seq, sc)

    # power families
    for p in np.linspace(-1.5, 1.5, 13):
        seq = _power(n, p)
        if seq is None:
            continue
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} new best {sc:.6f}")
        _maybe_push(seq, sc)

    # symmetric beta families (α=β)
    for a in [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        seq = _beta(n, a, a)
        if seq is None:
            continue
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta a={a:.1f} new best {sc:.6f}")
        _maybe_push(seq, sc)

    # a few asymmetric beta examples
    for a, b in [(0.3, 5.0), (5.0, 0.3), (0.5, 8.0), (8.0, 0.5)]:
        seq = _beta(n, a, b)
        if seq is None:
            continue
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.1f} β={b:.1f} new best {sc:.6f}")
        _maybe_push(seq, sc)

    # semicircle
    seq = _semicircle(n)
    if seq is not None:
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Semi‑circle new best {sc:.6f}")
        _maybe_push(seq, sc)

    # logistic shapes
    for k in [0.02, 0.05, 0.1]:
        for mu in [n * 0.3, n * 0.5, n * 0.7]:
            seq = _logistic(n, k, mu)
            if seq is None:
                continue
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic k={k:.3f} μ={mu:.0f} new best {sc:.6f}")
            _maybe_push(seq, sc)

    # ----- random Dirichlet draws -------------------------------------------------
    rng = np.random.default_rng()
    dirichlet_trials = 800
    for _ in range(dirichlet_trials):
        if time.time() >= deadline - 30.0:   # reserve time for descent + hill‑climb
            break
        # draw concentration on a log‑scale (range ≈ 0.01 … 20)
        logc = rng.uniform(-2.0, 1.3)        # 10^(-2) … 10^1.3 ≈ 0.01 … 20
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} new best {sc:.6f}")
        _maybe_push(a, sc)

    # ----------------------------------------------------------
    # 2️⃣  Sub‑gradient descent from the K best seeds
    # ----------------------------------------------------------
    # Give each descent a few seconds; the loop will recycle while time permits.
    descent_time_per_seed = 5.0
    seed_list = [seq for _, seq in sorted(seed_pool, key=lambda x: -x[0])]  # lowest scores first
    seed_idx = 0
    while time.time() < deadline - 30.0:         # keep about 30 s for final hill‑climb
        a0 = seed_list[seed_idx % len(seed_list)].copy()
        # add a tiny random jitter to avoid getting stuck on identical points
        jitter = rng.normal(scale=1e-6, size=n)
        a0 = np.maximum(a0 + jitter, 0.0)
        a0 = _project_simplex(a0)

        sub_deadline = min(deadline - 30.0, time.time() + descent_time_per_seed)
        best_seq, best_score = _sg_descent(a0, sub_deadline, best_seq, best_score)

        seed_idx += 1
        if time.time() >= deadline - 30.0:
            break

    # ----------------------------------------------------------
    # 3️⃣  Final short hill‑climb
    # ----------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ----------------------------------------------------------
    # Normalise and return as a plain Python list
    # ----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
