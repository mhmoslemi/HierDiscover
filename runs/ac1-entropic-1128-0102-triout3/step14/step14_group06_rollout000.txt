The program builds a low‑score step‑function in two stages.

1. **Low‑dimensional LP‑based optimisation**  
   *Work on a tiny problem ( `N≈512` ) where a linear programme can be solved fast.*  
   – start from several diverse shapes (U‑shaped beta, power‑ramps, random Dirichlet).  
   – for the current shape `f` compute `M = max( f * f )`.  
   – solve  

     ```
     maximise   Σ g_j
     subject to  (f * g)[k] ≤ M   for all k
                 g_j ≥ 0
     ```

     which is a linear programme because `f` is fixed.  
   – obtain the optimal `g`, normalise it to sum = 1 and move a small step towards it  

     `f ← (1‑t)·f + t·g`   ( `t` slowly decreases ).  

   – repeat until the allotted time for the low‑dimensional phase runs out.  
   The LP step injects mass into positions where `f` is small, thereby flattening the
   self‑convolution and reducing the objective.

2. **Upscaling & high‑dimensional polishing**  
   The best low‑dimensional shape is interpolated to a high‑resolution vector
   (`N≈65536`).  Afterwards a few thousand iterations of *smooth‑max Adam*
   descent are performed – the same surrogate used in the reference code but
   now with far more iterations because each step is cheap on the large vector.

Both phases respect the required time‑budget (`budget_s`).  
All vectors are kept on the probability simplex (non‑negative, sum = 1) which

```python
"""
Step‑function optimiser (LP + smooth‑max Adam).

Stage 1 – work on a small problem (N≈512).  For the current shape f
          compute the maximal self‑convolution M.
          Solve a linear programme that maximises Σ g_j under the
          linear constraints   (f * g)[k] ≤ M   (all k)   and   g ≥ 0.
          Move a small step towards the normalised optimiser g.
          Re‑iterate – this quickly reduces the ℓ∞‑convolution bound.

Stage 2 – interpolate the best small shape to a large N (≈65536) and
          polish it with many iterations of a smooth‑max Adam descent,
          i.e. minimise a temperature‑smoothed version of
          2·n·max(conv(a,a))/sum(a)².

The whole routine respects the supplied time budget (default 1000 s)
and always returns a non‑negative list that can be fed to evaluate_sequence().
"""

import time
import math
import numpy as np
from typing import Tuple, List

# ----------------------------------------------------------------------
# Utility functions (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for the initial pool)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape[:] = 1.0 / n
    return shape


def _make_power_ramp(n: int, p: float, plateau: float) -> np.ndarray:
    """
    Symmetric power‑ramp with an optional flat plateau.
    The vector is normalised to sum = 1.
    """
    plateau_len = int(round(plateau * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape[:] = 1.0 / n
    return shape


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw with concentration `conc` (< 1 favours sparsity)."""
    a = rng.gamma(conc, 1.0, size=n)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement (fixed‑f, optimise g)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> Tuple[np.ndarray, bool]:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to (f * g)[k] ≤ M   (∀k)
                   g ≥ 0
    Returns (g, success).  `g` is *not* normalised.
    """
    n = f.shape[0]
    K = 2 * n - 1

    # Build the Toeplitz‑like matrix A where A[k, j] = f_{k-j} (zero outside).
    # For each column j we fill rows j … j+n-1 with the vector f.
    A = np.zeros((K, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # Upper bounds for every convolution entry are M.
    b_ub = np.full(K, M, dtype=float)

    # Bounds for variables: g_j ≥ 0, no explicit upper bound.
    bounds = [(0.0, None) for _ in range(n)]

    # Solve with SciPy's HiGHS backend (fast for dense problems of this size).
    try:
        from scipy.optimize import linprog
    except Exception:          # pragma: no cover – fallback if SciPy missing
        return np.zeros_like(f), False

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        return np.zeros_like(f), False
    g = res.x
    return g, True


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (high‑dimensional polishing)
# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 init_lr: float = 0.05,
                 T_start: float = 0.2,
                 T_end: float = 1e-5) -> Tuple[np.ndarray, float]:
    """
    Adam descent on the soft‑max surrogate of the max‑convolution.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_c = conv.max()
        # current exact objective
        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} best={cur_score:.6f}")

        # temperature schedule (geometric)
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = max(T_end, T_start * (T_end / T_start) ** frac)

        # soft‑max weights for the convolution entries
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * corr(a_rev, w)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = init_lr / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a low‑score step function under the given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1) Choose problem sizes
    # ------------------------------------------------------------------
    N_large = 65536
    # Small problem on which the LP step is cheap.
    N_small = 512

    # ------------------------------------------------------------------
    # 2) Build an initial pool (small dimension) and keep the best.
    # ------------------------------------------------------------------
    TOP_K = 5
    pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def _push(score: float, vec: np.ndarray) -> None:
        if len(pool) < TOP_K:
            pool.append((score, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif score < pool[-1][0]:
            pool[-1] = (score, vec.copy())
            pool.sort(key=lambda x: x[0])

    # uniform baseline
    _push(_objective(np.full(N_small, 1.0 / N_small)),
          np.full(N_small, 1.0 / N_small))

    # a few symmetric beta shapes (U‑shaped and bell‑shaped)
    for a in (0.25, 0.35, 0.5, 0.7, 1.0, 1.3):
        shape = _make_beta_shape(N_small, a)
        _push(_objective(shape), shape)

    # power‑ramps with various exponents and plateau sizes
    for p in (0.2, 0.5, 1.0, 2.0):
        for plat in (0.0, 0.1, 0.3):
            shape = _make_power_ramp(N_small, p, plat)
            _push(_objective(shape), shape)

    # a handful of random Dirichlet draws (concentration <1 favours sparsity)
    for conc in (0.3, 0.5, 0.8):
        for _ in range(5):
            shape = _random_dirichlet(N_small, conc, rng)
            _push(_objective(shape), shape)

    # current best (the global variable `height_sequence_1` if available)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # down‑sample / up‑sample to N_small
            if starter.size != N_small:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N_small, endpoint=False)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                _push(_objective(starter), starter)
    except Exception:
        pass

    # best vector from the pool becomes the starting point for the LP loop
    best_small = pool[0][1].copy()
    best_score_small = pool[0][0]
    print(f"[{time.time() - start:.2f}s] Initial best (small) ≈ {best_score_small:.6f}")

    # ------------------------------------------------------------------
    # 3) LP‑based improvement loop (majority of the budget)
    # ------------------------------------------------------------------
    # Allocate roughly 70 % of the total time to the LP loop.
    lp_deadline = start + (deadline - start) * 0.70
    iteration = 0
    t_step = 0.2          # initial convex‑combination weight
    while time.time() < lp_deadline:
        iteration += 1
        conv = _conv_self(best_small)
        M = conv.max()
        g, ok = _lp_step(best_small, M)
        if not ok:
            # If the LP failed (e.g. numerical trouble) fall back to a tiny perturbation.
            delta = rng.normal(0.0, 0.02, size=N_small)
            g = best_small + delta
            g = np.maximum(g, 0.0)
            g = _project_simplex(g)

        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g                # enforce Σ g = 1

        # Adaptive step size: decay slowly with the number of iterations.
        t = t_step / (1.0 + iteration / 500.0)

        candidate = (1.0 - t) * best_small + t * g_norm
        cand_score = _objective(candidate)

        # Accept only if we see improvement – otherwise keep the old vector.
        if cand_score < best_score_small:
            best_small = candidate
            best_score_small = cand_score
            print(f"[{time.time() - start:.2f}s] LP iter={iteration:,} score={cand_score:.6f}")

        # occasional re‑initialisation from the pool to escape possible stalls
        if iteration % 200 == 0 and len(pool) > 1:
            idx = rng.integers(len(pool))
            best_small = pool[idx][1].copy()
            best_score_small = pool[idx][0]

    print(f"[{time.time() - start:.2f}s] End of LP phase, best small ≈ {best_score_small:.6f}")

    # ------------------------------------------------------------------
    # 4) Upscale to the large dimension required by the evaluator.
    # ------------------------------------------------------------------
    # Linear interpolation preserves the shape while keeping non‑negativity.
    x_small = (np.arange(N_small, dtype=float) + 0.5) / N_small
    x_large = (np.arange(N_large, dtype=float) + 0.5) / N_large
    large_vec = np.interp(x_large, x_small, best_small)
    large_vec = np.maximum(large_vec, 0.0)
    large_vec /= large_vec.sum()          # back to the simplex

    # ------------------------------------------------------------------
    # 5) Final Adam polishing (uses remaining budget, if any)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:                   # need at least a few seconds for a useful run
        final_vec, final_score = _smooth_adam(large_vec,
                                              deadline,
                                              start,
                                              init_lr=0.08,
                                              T_start=0.15,
                                              T_end=1e-5)
        # Keep the better of the two (rarely the Adam step worsens the score)
        if final_score < _objective(large_vec):
            large_vec = final_vec
            print(f"[{time.time() - start:.2f}s] Adam polishing improved to {final_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final sanity checks and return
    # ------------------------------------------------------------------
    large_vec = np.maximum(large_vec, 0.0)
    total = large_vec.sum()
    if total > 0:
        large_vec /= total
    else:
        large_vec = np.full(N_large, 1.0 / N_large)

    final_obj = _objective(large_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_obj:.6f}")

    # Convert to plain Python list as required by the evaluator.
    return [float(v) for v in large_vec]
```
