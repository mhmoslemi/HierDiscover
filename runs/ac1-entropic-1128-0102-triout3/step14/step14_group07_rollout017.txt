"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The strategy is a lightweight three‑phase search that can be run in a
single CPU thread inside the 1000 s budget.

1. **Diverse initial pool** – we create many analytic shapes
   (inverse power‑law, symmetric β‑densities, sparse binary masks)
   together with a few random Dirichlet draws.  Each candidate is
   projected to the probability simplex (non‑negative, sum = 1) and the
   `TOP_K` best (lowest) scores are retained.

2. **Sub‑gradient (mirror‑descent) polish** – starting from the best
   candidate we minimise the true objective `max(conv)` directly.
   At every iteration we locate the index `k*` where the convolution
   attains its maximum and move the vector opposite to the sub‑gradient
   `∂max/∂a_i = a_{k*-i}`.  After the additive update we renormalise
   onto the simplex.  A decreasing step size (`η / √t`) together with a
   simple run‑time allocation yields a fast descent without the need for
   soft‑max smoothing or Adam.

3. **Optional short simulated‑annealing finish** – if a few seconds
   remain we perform mass‑transfer swaps between two indices to escape
   possible local minima.

All heavy work (convolution, gradients) uses an FFT‑based linear
convolution routine.  The final sequence is returned as a plain Python
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2·len(a)-1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a sequence that already sums to 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex by simple clipping + renormalisation."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Analytic families of candidate shapes
# ----------------------------------------------------------------------
def _make_inverse_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric a_i ∝ (i+1)^{-p} + (n-i)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p) + (n - idx + 1) ** (-p)
    total = shape.sum()
    shape /= total
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sparse_mask(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Random binary mask of density `frac`, entries equal 1/m where m is the mask size."""
    m = max(1, int(round(frac * n)))
    idx = rng.choice(n, size=m, replace=False)
    shape = np.zeros(n, dtype=float)
    shape[idx] = 1.0 / m
    return shape


# ----------------------------------------------------------------------
# Pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted pool of at most k items with the smallest scores."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         eta0: float = 0.6,
                         max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) on the simplex by a simple additive
    sub‑gradient step: a ← a - η·g,   g_i = a_{k*‑i}.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_k = int(conv.argmax())
        max_c = conv[max_k]

        # compute sub‑gradient of the max‑convolution entry
        g = np.zeros_like(a)
        if max_k < n:
            # overlapping indices 0 … max_k
            g[: max_k + 1] = a[max_k :: -1]
        else:
            # overlapping indices max_k-(n-1) … n-1
            start = max_k - (n - 1)
            g[start:] = a[n - 1 : start - 1 : -1]

        # decreasing step size
        step = eta0 / math.sqrt(it)

        a = a - step * g
        a = _simplex_normalise(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # keep a small safety margin

    # --------------------------------------------------------------
    # Choose discretisation size (larger N → finer search, but also slower)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []       # (score, vector)

    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

        a = _make_inverse_power_shape(N, p)

    for alpha in np.arange(0.05, 1.0, 0.05):
        a = _make_beta_shape(N, alpha)

        a = _make_sparse_mask(N, frac, rng)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws (different concentration parameters)
    concs = [0.001, 0.005, 0.02, 0.06, 0.15, 0.4]
    for conc in concs:
        for _ in range(6):
            if time.time() >= deadline:
                break
            sample = rng.gamma(conc, 1.0, size=N)
            if sample.sum() == 0:
                continue
            a = sample / sample.sum()
            _push_top(pool, _objective(a), a, TOP_K)

    for base_score, base_vec in list(pool):   # shallow copy of current pool
        for t in [0.05, 0.10, 0.20]:
            mixed = (1 - t) * base_vec + t * uniform
            mixed = _simplex_normalise(mixed)
            _push_top(pool, _objective(mixed), mixed, TOP_K)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
