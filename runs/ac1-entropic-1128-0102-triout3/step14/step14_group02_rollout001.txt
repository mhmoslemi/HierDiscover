"""
Multi‑resolution optimisation for the step‑function attack.

The search proceeds in four phases:

1️⃣  Coarse exploration (n≈256).  A huge number of random Dirichlet
    draws (different concentration parameters) and simple parametric
    families (geometric, power, Beta) are screened – this is cheap
    because the FFT size is tiny.

2️⃣  Upsample the best coarse vector to a medium size (n≈1024) and
    run a smooth‑max Adam optimiser.  The smooth‑max
    `log‑sum‑exp` surrogate turns the non‑smooth “max‑conv” into a
    differentiable objective; the temperature τ is increased
    geometrically so the surrogate tightens gradually.

3️⃣  Upsample again to the target resolution (n≈4096) and continue the
    Adam optimisation (more iterations, smaller step‑size).

4️⃣  A final projected‑gradient hill‑climb moves a tiny amount of mass
    from the index with the largest positive sub‑gradient to the index
    with the most negative one; occasional random perturbations escape
    plateaux.

All updates stay on the probability simplex (sum = 1, aᵢ ≥ 0) by means
of the O(n) Euclidean projection of Duchi et al.  The routine respects
the 1000 s time budget, prints intermediate scores and finally returns
a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# FFT utilities (full linear convolution)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via FFT (real vectors)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al.
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers (sum is forced to 1 → score = 2·n·max(conv))
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# --------------------------------------------------------------
# Parametric families (used on the coarse grid)
# --------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for eps in np.logspace(-6, -1, 200):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mixing with uniform sometimes helps
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 201):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    """Full (α,β) grid – relatively cheap at small n."""
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    alphas = np.linspace(0.1, 5.0, 15)   # 15×15 = 225 combos (fast)
    betas = alphas

    for a in alphas:
        if time.time() >= deadline:
            break
        part_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = part_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=3000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [0.01, 5]
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# Smooth‑max Adam optimiser (scale‑invariant)
# --------------------------------------------------------------
def _adam_smoothmax(a0, deadline, best_seq, best_score,
                    init_step=0.04, init_tau=5.0, tau_factor=1.5,
                    step_factor=0.95):
    """Projected Adam on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = init_tau
    step0 = init_step
    it = 0

    # pre‑allocate large enough FFT length for gradient convolution
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of smooth max = 2 * (w * rev(a)) (restricted to n)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)               # length = 3n‑2 (full)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # simple learning‑rate decay to stabilise later
        step0 *= step_factor

        a = a_new

        # increase temperature occasionally – makes surrogate tighter
        if it % 800 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# --------------------------------------------------------------
# Directed hill‑climb (projected sub‑gradient moves)
# --------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    stagnation = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # if the obvious move is useless, pick a random swap
        if src == dst:
            i1, i2 = rng.choice(n, size=2, replace=False)
            src, dst = i1, i2
            if a[src] < 1e-12:
                continue

        # transfer a modest amount – up to 1 % of the source mass
        eps = a[src] * (rng.random() * 0.01 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = _objective_score(a_prop)
        if cur_score < best_score - 1e-9:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")
        else:
            stagnation += 1
            if stagnation > 4000:
                # random jitter to escape a plateau
                i, j = rng.choice(n, size=2, replace=False)
                eps_j = a[i] * (rng.random() * 0.01 + 1e-6)
                a[i] -= eps_j
                a[j] += eps_j
                a = _project_simplex(np.maximum(a, 0.0))
                stagnation = 0

    return best_a, best_score


# --------------------------------------------------------------
# Upsampling utilities (linear interpolation + renorm)
# --------------------------------------------------------------
def _upsample(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(v) to new_n and renormalise."""
    if v.shape[0] == new_n:
        return v.copy()
    old_n = v.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.maximum(v_new, 0.0)
    total = v_new.sum()
    if total > 0:
        v_new /= total
    else:   # fallback – uniform
        v_new = np.full(new_n, 1.0 / new_n)
    return v_new

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (sum ≈ 1) that minimises
    ``evaluate_sequence``.  The procedure follows a four‑phase,
    multi‑resolution strategy described in the module docstring.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search (n≈256)
    # ------------------------------------------------------------------
    n0 = 256
    uniform0 = np.full(n0, 1.0 / n0)
    best_seq = uniform0.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Coarse uniform baseline score={best_score:.6f}")

    phase1_deadline = deadline - 500.0   # reserve the rest for refinement
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_geometric(
            n0, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_power(
            n0, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(
            n0, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n0, best_seq, best_score, phase1_deadline, trials=4000)

    # ------------------------------------------------------------------
    # 2️⃣  Medium resolution (n≈1024) + Adam warm‑up
    # ------------------------------------------------------------------
    n1 = 1024
    best_seq = _upsample(best_seq, n1)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Upsampled to {n1}, score={best_score:.6f}")

    medium_deadline = deadline - 250.0
    if time.time() < medium_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam at {n1}")
        best_seq, best_score = _adam_smoothmax(
            best_seq, medium_deadline, best_seq, best_score,
            init_step=0.04, init_tau=5.0)

    # ------------------------------------------------------------------
    # 3️⃣  Target resolution (n≈4096) + Adam fine‑tuning
    # ------------------------------------------------------------------
    n_target = 4096
    best_seq = _upsample(best_seq, n_target)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Upsampled to target {n_target}, score={best_score:.6f}")

    fine_deadline = deadline - 80.0
    if time.time() < fine_deadline:
        print(f"[{time.time() - start_time:.2f}s] Adam fine‑tuning")
        best_seq, best_score = _adam_smoothmax(
            best_seq, fine_deadline, best_seq, best_score,
            init_step=0.02, init_tau=20.0)

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Normalise one last time and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
