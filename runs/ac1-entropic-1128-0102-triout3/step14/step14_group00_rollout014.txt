def _generate_candidate_shapes(n, rng, num_random=200):
    candidates = []
    # symmetric beta shapes
    for _ in range(num_random):
        a = 10**rng.uniform(-2, 1)  # 0.01 to 10

    for _ in range(num_random//2):
        a1 = 10**rng.uniform(-2, 1)
        b1 = 10**rng.uniform(-2, 1)
        a2 = 10**rng.uniform(-2, 1)
        b2 = 10**rng.uniform(-2, 1)

        shape = w * _make_beta_shape(n, a1, b1) + (1 - w) * _make_beta_shape(n, a2, b2)

        candidates.append(shape)
    # uniform
    candidates.append(np.full(n, 1.0/n))

def _pgd_optimize(a0, deadline, start_time, step0=0.05):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active indices within tolerance
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2 * a[k - idx]
        grad /= float(len(active))

        # step size schedule: decaying
        step = step0 / np.sqrt(it)

        # additive update
        a_new = a - step * grad

        # project onto simplex
        a_new = _project_simplex(a_new)

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD iter={it} score={cur_score:.6f}")
        a = a_new
    return best_a, best_score

Now for _project_simplex implementation:

```
def _project_simplex(v):
    # ensures sum(v) = 1 and v >= 0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _guided_hill_climb(a0, deadline, start_time, max_trials=5):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    attempts = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # find active set
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi+1)
            g[idx] += 2 * a[k - idx]
        g /= float(len(active))

        # choose src with largest g where a>1e-12
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        g_src = g[src_candidates]
        src = src_candidates[np.argmax(g_src)]

        # choose dst with smallest g
        dst = np.argmin(g)

        if src == dst:
            # random perturbation
            dst = np.random.randint(n)
            if dst == src:
                continue

        eps = a[src] * 0.3  # move 30% of src mass
        # ensure we don't remove all mass
        eps = min(eps, a[src] * 0.9)

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # renorm not needed (sum unchanged)
        cur_score = _objective(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_score = cur_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] HillClimb move src={src} dst={dst} eps={eps:.3e} score={cur_score:.6f}")
        else:
            # occasionally accept worsening move for escapement?
            attempts += 1
            if attempts % 20 == 0:
                # random tweak
                src = np.random.choice(np.where(a > 1e-12)[0])
                dst = np.random.randint(n)
                eps = a[src] * 0.1
                a_prop = a.copy()
                a_prop[src] -= eps
                a_prop[dst] += eps
                a = a_prop
    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

The search proceeds in three stages:

1. **Broad initialization** – a large pool of analytic candidates
   (Beta‑distributions, simple power‑law curves and a few random mixes)
   is generated and the best few are kept.

2. **Projected gradient descent on the ℓ∞‑norm of the auto‑convolution**.
   For a fixed ℓ¹‑normalisation the objective equals
   ``2·n·max(conv(a,a))``.  At every iteration we compute the active
   set of indices attaining the maximum, form the sub‑gradient of the
   max‑functional and take an additive step followed by projection onto
   the probability simplex.  Step size decays as η₀/√t.

3. **Guided hill‑climbing** – using the same active set we move a small
   packet of mass from the index with the largest sub‑gradient (the
   “worst” contributor to the peak) to the index with the smallest
   sub‑gradient.  Each move is accepted only if it reduces the objective.
   The process is repeated until the time budget is exhausted.

All stages are time‑boxed; the best vector encountered is finally
renormalised (sum = 1) and returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used for optimisation (scale‑invariant)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised Beta( a , b ) on {0,…,n‑1}, normalised to sum = 1.
    The PDF is proportional to x^{a‑1}(1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns w such that w ≥ 0 and w.sum() = 1.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the max‑convolution functional.
    At each iteration the sub‑gradient (averaged over the active set)
    is subtracted, then the result is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set (indices that are (almost) maximal)
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of the max‑functional
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(active))

        # decaying additive step
        step = step0 / math.sqrt(it)

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        a = a_new

    return best_a, best_score


def _guided_hill_climb(a0: np.ndarray,
                       deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer to reduce the maximal convolution entry.
    Uses the active‑set sub‑gradient to pick a source (large gradient)
    and a destination (small gradient) and moves a fraction of the
    source mass.  Accepts only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    failed_moves = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient (averaged over active set)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(active))

        # source = index with large gradient and non‑trivial mass
        cand_src = np.where(a > 1e-12)[0]
        if cand_src.size == 0:
            break
        src = cand_src[np.argmax(grad[cand_src])]

        # destination = index with minimal gradient
        dst = int(np.argmin(grad))

        if src == dst:
            # tiny random shuffle if the algorithm stalls
            src = int(np.random.choice(cand_src))
            dst = int(np.random.randint(n))
            if src == dst:
                continue

        eps = min(a[src] * 0.35, a[src] * 0.9)   # move up to 35 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps   # sum is preserved, non‑negativity holds

        cur_score = _objective(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_score = cur_score
            best_a = a_prop.copy()
            failed_moves = 0
            print(f"[{time.time() - start_time:.2f}s] HillClimb src={src} dst={dst} eps={eps:.3e} score={cur_score:.6f}")
        else:
            failed_moves += 1
            # after many unsuccessful attempts try a tiny random perturbation
            if failed_moves > 30:
                src = int(np.random.choice(cand_src))
                dst = int(np.random.randint(n))
                eps = a[src] * 0.05
                a_prop = a.copy()
                a_prop[src] -= eps
                a_prop[dst] += eps
                a = a_prop
                failed_moves = 0

    return best_a, best_score


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   beta_smooth: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of the max‑functional.
    The smooth max is  (1/β)·log∑exp(β·conv).  The gradient can be
    evaluated by a single convolution of the weight vector with a reversed
    copy of a, which is performed with an FFT for speed.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        maxc = conv.max()
        # smooth weights (log‑sum‑exp trick)
        w_exp = np.exp(beta_smooth * (conv - maxc))
        w = w_exp / w_exp.sum()                     # shape (2n‑1,)

        # gradient = 2 * (reversed a) * w  (convolution, central slice)
        a_rev = a[::-1]
        L = _next_pow2(len(w) + n - 1)
        Af = np.fft.rfft(a_rev, n=L)
        Wf = np.fft.rfft(w, n=L)
        prod = np.fft.irfft(Af * Wf, n=L)[:len(w) + n - 1]
        grad = 2.0 * prod[(n - 1):(n - 1) + n]

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = 0.08 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t} score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function under the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # 1) Choose a length that balances resolution and speed.
    # --------------------------------------------------------------
    n = 8192                               # 2¹¹³ – fast FFT, enough freedom

    # --------------------------------------------------------------
    # 2) Build a generous pool of analytic candidates.
    # --------------------------------------------------------------
    raw_pool = _generate_initial_pool(n, rng)

    # keep only the best 12 candidates for the intensive local search
    raw_pool.sort(key=lambda x: x[0])
    pool = raw_pool[:12]

    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Do a short PGD run on each of the top few starters.
    # --------------------------------------------------------------
    for idx, (_, vec) in enumerate(pool):
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # allocate a modest slice of the remaining budget to this starter
        alloc = min(remaining * 0.35, 25.0)   # at most 25 s per starter
        sub_deadline = time.time() + alloc
        seq, sc = _pgd_optimize(vec, sub_deadline, start_time)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"[{time.time() - start_time:.2f}s] New global best by PGD on starter {idx} = {sc:.6f}")

    print(f"[{time.time() - start_time:.2f}s] After PGD phase best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Guided hill‑climbing to fine‑tune the best vector.
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + (deadline - time.time()) * 0.45
        best_seq, best_score = _guided_hill_climb(best_seq, hill_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Short Adam‑based polish (if time still permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _adam_optimize(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After Adam polish best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation & conversion to list of Python floats.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
