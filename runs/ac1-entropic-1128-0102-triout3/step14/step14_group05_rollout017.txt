"""
Optimisation of a non‑negative step function for the min‑max convolution
objective.

The algorithm proceeds in three stages:

1. **Coarse parametric sweep** – evaluates a large family of
   geometric‑power shapes (a_i ∝ r**i·(i+1)**p) for many choices of
   r and p, keeping the best sequence found.

2. **Annealed smooth‑max optimisation** – a surrogate objective based on
   a log‑sum‑exp approximation of ``max(conv(a,a))`` is minimised
   with L‑BFGS.  The temperature (``beta``) is raised from a smooth
   regime up to a very sharp one (β≈1000) while allowing many
   function evaluations, which yields a far more accurate solution
   than the original 250‑call budget.

3. **Targeted hill‑climb** – repeatedly identifies the shift that
   attains the current maximum of ``conv(a,a)`` and moves a small
   fraction of mass from indices with a large contribution to the
   maximum toward indices that contribute little.  This greedy step
   exploits the exact objective and typically improves the solution
   beyond the surrogate phase.

All steps respect the global wall‑clock budget (default 1000 s).  The
best sequence discovered is normalised to unit sum and returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# optional SciPy – may be missing in a minimal environment
try:
    import scipy.optimize
    import scipy.sparse
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    ``a`` must be non‑negative; the routine does not perform any clipping.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                     # should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max surrogate of max(conv) together
    with its gradient w.r.t. the log‑probability parameters ``z``.
    ``z`` lives in ℝⁿ; ``a = softmax(z)`` is a probability vector.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # size used for a*a
        self.L_corr = L_corr          # size used for w * a_rev

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a ≥0, Σa = 1
        # self‑convolution of a
        conv = _conv_self_fft(a, self.L_conv)   # length 2n‑1
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) and its normalised weights
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, length 2n‑1
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient wrt a (see derivation in the original code)
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]   # length n

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)        # scalar
        grad_z = a * (grad_a - dot)    # (n,)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Targeted hill‑climb (exact objective)
# ----------------------------------------------------------------------


def _targeted_hill_climb(a: np.ndarray, deadline: float,
                         start_time: float, L_fft: int):
    """
    Greedy local optimiser that repeatedly reduces the contribution of the
    shift currently attaining the maximum of ``conv(a,a)``.
    At each iteration a small amount of mass is moved from a source index
    with large gradient (large partner weight) to a destination index with a
    small gradient.  Only moves that strictly improve the exact objective are
    accepted.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()
    # parameters controlling step size
    move_fraction = 0.02   # move up to 2 % of source mass per step
    low_percent = 0.10     # percentile used to pick a low‑gradient destination

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_idx = int(conv.argmax())
        # gradient of the maximal entry w.r.t. a_i is 2 * a_{max_idx-i} (if inside)
        grad = np.zeros(n)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx = np.arange(i_start, i_end + 1)
        partner = max_idx - idx
        grad[idx] = 2.0 * best_a[partner]

        # source – an index that carries mass and has a relatively large gradient
        mass_mask = best_a > 1e-15
        if not np.any(mass_mask):
            break
        src_candidates = np.where(mass_mask)[0]
        # prefer large product of mass and gradient
        src_score = grad[src_candidates] * best_a[src_candidates]
        src = src_candidates[np.argmax(src_score)]

        # destination – an index with a small gradient (ideally zero)
        # pick from the lower ``low_percent`` of gradient values
        sorted_by_grad = np.argsort(grad)
        low_cut = max(1, int(low_percent * n))
        low_set = sorted_by_grad[:low_cut]
        # make sure we don't pick the source again
        low_set = low_set[low_set != src]
        if low_set.size == 0:
            # fall back to a completely random different index
            dst = rng.integers(n)
            if dst == src:
                continue
        else:
            dst = rng.choice(low_set)

        eps = best_a[src] * move_fraction
        if eps <= 0.0:
            continue
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        # numerical safety
        if cand[src] < 0:
            continue
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] targeted hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine consists of:
      1. a fast parametric sweep over geometric‑power families,
      2. an annealed L‑BFGS optimisation of a smooth‑max surrogate
         (significantly more iterations than the reference code),
      3. a greedy targeted hill‑climb that uses the exact objective.
    The best sequence (normalised to unit sum) is returned as a list of floats.
    """

    n = 4096

    L_corr = _next_pow2(3 * n - 2)      # FFT length for surrogate gradient

    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()

                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc

            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover

    # enlarge grids compared with the reference implementation

                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (elongated β schedule)
    # ------------------------------------------------------------------

"""
Optimisation of a non‑negative step function for the min‑max convolution
objective.

The algorithm proceeds in four stages:

1. **Coarse parametric sweep** – evaluates many geometric‑power shapes
   (a_i ∝ r**i·(i+1)**p) on a fine grid of (r,p) values and keeps the
   best sequence found.

2. **Annealed smooth‑max optimisation** – a surrogate objective based on a
   log‑sum‑exp (soft‑max) approximation of ``max(conv(a,a))`` is minimised
   with L‑BFGS.  The temperature β is raised from a smooth regime up to a
   very sharp one (β≈1000) while allowing many function evaluations,
   giving a much more accurate solution than the reference 250‑call budget.

3. **Random restarts** – short L‑BFGS runs from perturbed versions of the
   current best sequence help escape shallow local minima.

4. **Targeted hill‑climb** – repeatedly identifies the shift that attains the
   current maximum of ``conv(a,a)`` and moves a small fraction of mass from
   indices with a large contribution to that shift toward indices that
   contribute little.  Only moves that strictly improve the exact objective
   are accepted.  This greedy step typically pushes the solution below the
   previous best (≈ 1.508).

All steps respect the global wall‑clock budget (default 1000 s).  The
best sequence discovered is normalised to unit sum and returned as a plain
``list[float]``.
"""

import time
import math
import numpy as np

# optional SciPy – may be missing in a minimal environment
try:
    import scipy.optimize
    import scipy.sparse
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max surrogate of max(conv) together
    with its gradient w.r.t. the log‑probability parameters ``z``.
    ``z`` lives in ℝⁿ; ``a = softmax(z)`` is a probability vector.
    """

    def __init__(self, n: int, beta: float, L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv
        self.L_corr = L_corr

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                         # a ≥0, Σa = 1
        conv = _conv_self_fft(a, self.L_conv)    # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                   # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()            # soft‑max weights

        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # gradient w.r.t. a
        a_rev = a[::-1]
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]      # length n

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Targeted hill‑climb (exact objective)
# ----------------------------------------------------------------------


def _targeted_hill_climb(a: np.ndarray, deadline: float,
                         start_time: float, L_fft: int):
    """
    Greedy optimiser that repeatedly reduces the contribution of the shift
    currently attaining the maximum of ``conv(a,a)``.  At each iteration a
    small amount of mass is moved from a source index with a large gradient
    to a destination index with a small gradient.  Only improvements are
    accepted.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()
    move_fraction = 0.02      # move up to 2 % of source mass each step
    low_percent = 0.10        # percentile for low‑gradient destinations

    while time.time() < deadline:
        conv = _conv_self_fft(best_a, L_fft)
        max_idx = int(conv.argmax())
        # gradient of the maximal entry w.r.t. a_i is 2·a_{max_idx−i}
        grad = np.zeros(n)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = max_idx - idx
        grad[idx] = 2.0 * best_a[partner]

        # source: an index that holds mass and has a relatively large gradient
        mass_mask = best_a > 1e-15
        if not np.any(mass_mask):
            break
        src_candidates = np.where(mass_mask)[0]
        src_score = grad[src_candidates] * best_a[src_candidates]
        src = src_candidates[np.argmax(src_score)]

        # destination: an index with a small gradient (preferably zero)
        sorted_by_grad = np.argsort(grad)
        low_cut = max(1, int(low_percent * n))
        low_set = sorted_by_grad[:low_cut]
        low_set = low_set[low_set != src]                     # avoid self‑move
        if low_set.size == 0:
            dst = rng.integers(n)
            if dst == src:
                continue
        else:
            dst = rng.choice(low_set)

        eps = best_a[src] * move_fraction
        if eps <= 0.0:
            continue
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        if cand[src] < 0.0:
            continue

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] targeted hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine consists of:
      1. a fine parametric sweep over geometric‑power families,
      2. an annealed L‑BFGS optimisation of a smooth‑max surrogate,
      3. several short random restarts,
      4. a targeted hill‑climb that uses the exact objective.

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # problem size – 4096 works well on a single core
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if ``height_sequence_1`` exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size:

            if sc < best_score:

    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fine parametric sweep (geometric‑power families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=60)
    p_grid = np.linspace(-5.0, 3.0, num=81)
