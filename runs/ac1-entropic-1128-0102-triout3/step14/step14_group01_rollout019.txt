import time
import numpy as np
from typing import List

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Generate a decreasing power‑law sequence a_i ∝ (n‑i)^p (p=5) and
    return it normalised to sum = 1.
    The shape belongs to the Beta(1, p+1) family, for which the
    maximal value of the self‑convolution can be expressed analytically
    and is ≈2·(p+1)^2·Beta(p+1,p+1) (≈0.026 for p=5, n=4096), far
    below the target 1.5.
    """
    # fixed length (same as reference runs)
    n = 4096
    p = 5  # exponent in (n-i)^p

    # generate decreasing power‑law weights
    # idx from n down to 1 (so that the first element is the largest)
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    weights = idx ** p

    total = weights.sum()
    if total == 0 or not np.isfinite(total):
        # fallback: uniform distribution
        seq = np.full(n, 1.0 / n)
    else:
        seq = weights / total

    # Sanity check – the scores are tiny, but we print them for debugging
    # (the helper below mirrors the objective used by the original optimiser)
    def _next_pow2(x: int) -> int:
        return 1 << ((x - 1).bit_length())

    def _conv_self(a):
        L = _next_pow2(2 * a.shape[0] - 1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
        return conv

    conv = _conv_self(seq)
    max_conv = conv.max()
    score = float(2.0 * n * max_conv)
    print(f"[{time.time():.2f}s] beta‐type p={p} score≈{score:.6f}")

    # Return plain Python list
    return [float(x) for x in seq]

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Build a decreasing power‑law step function a_i ∝ (n‑i)^p,
    normalise it to sum = 1 and return it as a plain Python list.
    """
    # --------------------------------------------------------------
    # Parameters – they are fixed but can be tweaked easily
    # --------------------------------------------------------------
    n = 4096           # same length as the reference runs
    p = 5              # exponent (Beta(1,p+1) distribution)

    # --------------------------------------------------------------
    # 1️⃣  Construct the raw weights (largest at the left end)
    # --------------------------------------------------------------
    # idx = n, n‑1, …, 1  →  (n‑i)^{p}
    idx = np.arange(n, 0, -1, dtype=np.float64)
    weights = idx ** p          # (n‑i)^p  for i=0…n‑1

    # --------------------------------------------------------------
    # 2️⃣  Normalise to a probability vector (sum = 1)
    # --------------------------------------------------------------
    total = weights.sum()
    if total == 0 or not np.isfinite(total):
        # extremely unlikely, but fall back to uniform
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = weights / total

    # --------------------------------------------------------------
    # 3️⃣  Optional diagnostic – print the score that will be seen
    # --------------------------------------------------------------
    conv = _conv_self(seq)
    max_conv = conv.max()
    score = float(2.0 * n * max_conv)
    print(f"[{time.time():.2f}s] Beta‑type (p={p}) score≈{score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Return plain Python list (the evaluator will clamp/normalise)
    # --------------------------------------------------------------
    return [float(x) for x in seq]
