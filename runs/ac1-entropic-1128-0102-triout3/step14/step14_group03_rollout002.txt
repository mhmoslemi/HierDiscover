"""
A hybrid search for a low‑peak self‑convolution step function.

The algorithm works in three phases (all respecting a 1000 s budget).

1️⃣  **Warm start** – random exploration of a simple geometric × power
    family (a_i ∝ r^i·(i+1)^p) possibly mixed with the uniform vector.
    The best shape found is taken as the initial point.

2️⃣  **Multi‑peak sub‑gradient descent** – at each iteration the
    convolution is computed, all shifts whose value is within a small
    relative tolerance of the maximum are collected and the gradients
    for those shifts are averaged.  This avoids getting trapped on a
    single active constraint.  An adaptive step size together with a
    Euclidean projection onto the probability simplex is used.

3️⃣  **Simulated‑annealing mass‑transfer** – the current best vector is
    perturbed by moving a random fraction of mass from one index to
    another.  Moves are accepted according to a temperature schedule
    (Metropolis rule).  The temperature and the maximal transferred
    fraction decay exponentially, allowing large exploratory jumps at
    the beginning and fine‑grained polishing at the end.

The code is completely self‑contained, uses only NumPy and the
already‑imported ``evaluate_sequence`` for final verification, and
returns the best vector (as a plain list of Python floats) found before
the time limit expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using a pre‑allocated FFT length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective (the sum is assumed ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  ``a`` is a probability vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple parametric family (geometric × power) and mixing with uniform
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Return a normalised vector proportional to r**i·(i+1)**p (i = 0…n‑1)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)               # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Multi‑peak sub‑gradient descent (stage 2)
# ----------------------------------------------------------------------


def _multi_subgradient_descent(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that averages the gradients of all
    convolution shifts whose value lies within ``rel_tol`` of the maximum.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.20                     # initial step size (empirically good)
    rel_tol = 5e-4                  # consider shifts with conv >= (1‑rel_tol)*max
    it = 0

    while time.time() < deadline:
        it += 1

        # convolution and set of “active” shifts
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # --- gradient that is the average over the active shifts -------------
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals

            g[i_vals] += a[j_vals]            # contribution a_j → ∂/∂a_i
            mask = i_vals != j_vals
            if mask.any():
                g[j_vals[mask]] += a[i_vals[mask]]

        g = g / max(1, len(active))            # average

        # --- step with a diminishing schedule --------------------------------
        step = step0 / (it ** 0.55)            # slower decay than 1/√it
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        # --- accept improvement or a probabilistic uphill move ----------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")
        else:
            # Metropolis‑style acceptance with a very mild temperature
            prob = np.exp(-(cur_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # occasional jitter to keep the iterates from stagnating
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (stage 3)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Moves are accepted with the Metropolis rule; temperature and the
    maximal transferred fraction decay exponentially.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    cur_score = best_score_initial
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    # annealing schedule parameters
    T_start = 2.5e-2
    T_end = 1e-6
    frac_start = 2.0e-1
    frac_end = 1.0e-3
    total = max(deadline - time.time(), 1e-6)

    while time.time() < deadline:
        elapsed = time.time() - (deadline - total)
        frac = frac_start * (frac_end / frac_start) ** (elapsed / total)
        T = T_start * (T_end / T_start) ** (elapsed / total)

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * rng.random() * frac
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _eval_fast(a_prop, L_fft)

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve score={new_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_prop
                cur_score = new_score

        # tiny jitter every few thousand iterations (helps break ties)
        if rng.random() < 0.001:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA final score={sc_sa:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (sum → 1) and return a plain list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
