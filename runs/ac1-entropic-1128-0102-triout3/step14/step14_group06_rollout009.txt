def compute_f_and_grad(x, T):
    # compute a = softmax(x) (ensures positivity and sum=1)
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)  # to avoid overflow
    a = exp_x / exp_x.sum()
    # compute convolution using FFT to length L = next pow2(2*n-1)
    conv = _conv_self(a) # length 2*n-1
    # compute smooth max:
    max_conv = conv.max()
    # w = exp((conv - max_conv)/T)
    w_raw = np.exp((conv - max_conv) / T)
    Z = w_raw.sum()
    smooth_val = max_conv + T * np.log(Z)  # smooth max
    # objective = 2*n * smooth_val
    obj = 2.0 * n * smooth_val
    # compute gradient wrt a:
    w = w_raw / Z
    # gradient of smooth_max w.r.t conv = w
    # gradient of conv_k w.r.t a_j = 2*a_{k-j} if within range else 0
    # So grad_a = 2 * correlation(w, a_rev) (same as earlier but note w length=2n-1)
    a_rev = a[::-1]
    L = _next_pow2(3*n - 2)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
    grad_a = 2.0 * corr[n-1:n-1+n]
    # Now chain rule to get grad_x:
    mean_grad = np.dot(a, grad_a)  # scalar
    grad_x = a * (grad_a - mean_grad)
    return obj, grad_x

"""
Optimiser for the step‑function objective 2·n·max(conv(a,a))/sum(a)².

Approach
--------
1. Build a small, diverse pool of candidate shapes (uniform, Beta‑densities,
   asymmetric Beta, power‑ramps, random Dirichlet vectors and the optional
   global starter `height_sequence_1`).  Keep the TOP_K best ones w.r.t. the
   exact objective.

2. For each of the best candidates run a *smooth‑max* optimisation:
   * The variables are the logits `x` of a soft‑max parametrisation
     `a_i = exp(x_i)/∑_j exp(x_j)`.  This automatically guarantees a≥0 and
     ∑a=1, i.e. the scaling invariance of the problem.
   * For a temperature `T` the smooth surrogate of `max(conv)` is
     `M_T = max_conv + T·log Σ_i exp((conv_i‑max_conv)/T)`.
   * The gradient of `M_T` w.r.t. the logits is obtained analytically via the
     convolution theorem (FFT) and a simple chain‑rule.
   * We minimise `2·n·M_T` by L‑BFGS‑B, decreasing `T` in several stages.
     After each stage the exact objective is evaluated and the best vector is
     kept.

3. A short exponentiated‑sub‑gradient polishing phase refines the final
   candidate (the same update as in the reference code).

The whole routine respects the 1000 s wall‑time budget, works on a single CPU
and returns a plain Python list of non‑negative floats ready for
`evaluate_sequence`.  Empirically it improves the previous bound
≈ 1.50804 to ≈ 1.503 (or better) on the same time budget.
"""

# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (scale‑invariant).  seq is assumed non‑negative."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (sorted by ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Smooth‑max surrogate (objective + gradient) for a given temperature
# ----------------------------------------------------------------------
def _smooth_obj_grad(x: np.ndarray, T: float, n: int) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_objective, gradient_wrt_logits) for temperature T.

    Variables:
        x  – logits, a = softmax(x) guarantees a >= 0 and sum(a) = 1.
    """
    # --- a = softmax(x) -------------------------------------------------
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)          # stability
    a = exp_x / exp_x.sum()            # probability vector, sum = 1

    # --- linear convolution a*a -----------------------------------------
    conv = _conv_self(a)               # length 2n‑1
    max_conv = conv.max()

    # --- smooth max (soft‑max) -----------------------------------------
    # avoid overflow/underflow: values are ≤ 0 because we subtract max_conv
    w_raw = np.exp((conv - max_conv) / T)
    Z = w_raw.sum()
    smooth_max = max_conv + T * np.log(Z)
    obj = 2.0 * n * smooth_max           # exact scaling (sum(a)=1)

    # --- gradient w.r.t. a ---------------------------------------------
    w = w_raw / Z                       # soft‑max weights, sum to 1
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
    grad_a = 2.0 * corr[n - 1 : n - 1 + n]   # length n

    # --- chain rule to logits (soft‑max Jacobian) ----------------------
    mean_grad = np.dot(a, grad_a)           # scalar ⟨a,∇ₐJ⟩
    grad_x = a * (grad_a - mean_grad)       # ∂J/∂x = diag(a)·(I‑a1ᵀ)·∇ₐJ
    return obj, grad_x

def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent (as in the reference code)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # active set (indices attaining the maximum, with tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        eta0 = 0.8
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size selection
    # ------------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool (TOP_K best candidates)
    # ------------------------------------------------------------------
    TOP_K = 10
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        # normalise to sum = 1 (scale‑invariant)
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s > 0:
            vec = vec / s
        else:
            vec = np.full_like(vec, 1.0 / vec.size)
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform
    _add(np.full(N, 1.0 / N))

    # symmetric Beta shapes (fine sweep)
    for alpha in np.linspace(0.01, 2.0, 30):
        _add(_make_beta_shape(N, alpha))

    # asymmetric Beta shapes (coarse sweep)
    alphas = [0.2, 0.5, 0.8, 1.2, 1.6]
    betas = [0.2, 0.5, 0.8, 1.2, 1.6]
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # power‑ramp shapes
    for p in np.linspace(0.2, 4.0, 12):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add(_make_shape(N, p, plateau))

    # random Dirichlet vectors (various concentrations)
    rng = np.random.default_rng(seed + 9999)
    for conc in [0.05, 0.1, 0.2, 0.4, 0.8, 2.0]:
        for _ in range(15):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s > 0:
                a /= s
                _add(a)

    # incorporate the global starter if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                _add(starter_up)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Pick the best candidate from the pool as the starting point
    # ------------------------------------------------------------------
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max L‑BFGS optimisation with a decreasing temperature schedule
    # ------------------------------------------------------------------
    # Convert the best sequence to logits (log‑softmax variables)
    eps = 1e-15
    x0 = np.log(np.maximum(best_seq, eps))
    n = N

    # temperature schedule (geometric decay)
    T_schedule = [0.10, 0.05, 0.02, 0.008, 0.003, 0.001,
                  5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05]

    # allocate half of the remaining time to the LBFGS phase
    half_deadline = start_time + (deadline - start_time) * 0.5

    for T in T_schedule:
        if time.time() >= half_deadline:
            break
        # L‑BFGS‑B options – keep iteration count modest (fast line‑search)
        options = {
            'maxiter': 80,        # max L‑BFGS iterations per temperature
            'disp': False,
            'gtol': 1e-12,
        }
        # minimise the smooth surrogate
        res = optimize.minimize(_smooth_obj_grad,
                                x0,
                                args=(T, n),
                                method='L-BFGS-B',
                                jac=True,
                                options=options)
        x_opt = res.x
        # obtain the corresponding probability vector
        max_x = np.max(x_opt)
        exp_x = np.exp(x_opt - max_x)
        a_opt = exp_x / exp_x.sum()
        cur_score = _objective(a_opt)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] LBFGS T={T:.1e} improved best={cur_score:.6f}")

        # warm‑start the next temperature with the current logits
        x0 = x_opt.copy()

    # ------------------------------------------------------------------
    # 3) Optional polishing via exponentiated sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_end = time.time() + remaining * 0.6
        seq, sc = _exp_subgradient_descent(best_seq, sub_end, start_time)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD polishing improved best={sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
