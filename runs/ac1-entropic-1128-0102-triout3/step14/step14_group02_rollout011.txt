"""
A fast multi‑stage heuristic for minimizing

    2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm works on the probability simplex (sum = 1) because the
objective is scale‑invariant.  It builds a small library of simple
shapes (uniform, geometric, power‑law, Beta‑like) and then explores
random convex combinations of those shapes.  The best candidate is
refined by a simulated‑annealing hill‑climb that repeatedly moves a
tiny fraction of mass from the index with the largest sub‑gradient of
the max‑convolution to the index with the smallest one.  The temperature
decays linearly, allowing occasional uphill moves early on and a fine
deterministic descent later.  All work is done with FFT‑based
convolutions, so a full iteration costs only a few milliseconds even for
n≈4000.  The routine respects the 1000 s budget and returns a Python
list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a single FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (the internal version works on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # a tolerance to decide which lags are active
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# ----------------------------------------------------------------------
def _build_basis(n: int):
    """Return a list of normalized vectors that span a rich set of shapes."""
    basis = []

    # 1) uniform

    idx = np.arange(n, dtype=float)

        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        basis.append(geo)

    # 3) power‑law families (a_i ∝ (i+1)^p)
    p_vals = [-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 2.0]

        seq_sum = seq.sum()

    betas = [0.2, 0.5, 1.0, 2.0, 5.0]

def _greedy_hill_climb(a: np.ndarray,
                       cur_score: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    A simple projected gradient / simulated‑annealing climb.
    Moves a fraction of mass from the index with the largest sub‑gradient
    to the index with the smallest.  Early on uphill moves are accepted
    with a temperature that decays linearly.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_sc = cur_score

    # linear temperature schedule
    T0 = cur_score * 0.05          # 5 % of the current score
    T_end = cur_score * 1e-5

    while time.time() < deadline:
        # compute gradient of the current max‑convolution
        conv = _conv_self(best_a)
        grad = _subgradient_maxconv(best_a, conv)

        # candidate source / destination
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # step size (fraction of the source mass) – shrink as temperature falls
        remaining_frac = (deadline - time.time()) / (deadline - (deadline - 1000))
        frac = 0.08 * remaining_frac  # start near 8 % and decay
        frac = max(frac, 1e-4)        # never smaller than a tiny value
        delta = best_a[src] * frac
        if delta <= 0.0:
            continue

        # propose new vector (sum unchanged, only two entries differ)
        a_prop = best_a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # protect against negative rounding errors
        if a_prop[src] < -1e-12:
            continue
        a_prop = np.maximum(a_prop, 0.0)
        a_prop /= a_prop.sum()          # renormalise (tiny correction)

        new_score = _objective_score(a_prop)

        # current temperature
        T = T0 * (deadline - time.time()) / (deadline - (deadline - 1000)) + T_end

        # accept improvement or a temperature‑guided uphill move
        if new_score < best_sc or rng.random() < math.exp(-(new_score - best_sc) / max(T, 1e-12)):
            best_a = a_prop
            best_sc = new_score

    return best_a, best_sc

    # give the hill‑climb a few seconds to initialise the temperature
    init_sc = best_score
    best_seq, best_score = _greedy_hill_climb(best_seq, init_sc, climb_deadline, rng)

    # ------------------------------------------------------------------
    # Final normalisation and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
