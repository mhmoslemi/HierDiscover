def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    4. An enhanced hill‑climb with a larger transfer step.

    The best sequence (normalized to sum = 1) is returned as a plain list.

"""
Enhanced search for a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm follows four stages while respecting a 1000 s budget:

1.  A broader parametric sweep (geometric, power, symmetric/asymmetric Beta,
    symmetric exponential, and a two‑block piecewise‑constant family).
2.  Random Dirichlet draws with many concentration values.
3.  Multiplicative‑weight descent that repeatedly shrinks coordinates with
    large sub‑gradient (a projected exponential‑gradient method on the simplex).
4.  An improved hill‑climb that transfers a larger fraction of mass and
    performs occasional random perturbations.

All convolutions use an FFT‑based routine; the simplex projection uses the
O(n) algorithm of Duchi et al.  The best sequence found is normalised to sum = 1
and returned as a plain Python list ready for ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):

                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")

def _multiplicative_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5                      # initial step size
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] MultiDescent it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)        # sub‑gradient (all ≥0)
        grad_centered = grad - grad.mean()           # make it zero‑sum
        eta = eta0 / (1.0 + it * 1e-4)                # slow decay

        factor = np.exp(-eta * grad_centered)
        factor = np.clip(factor, 1e-12, 1e12)        # safety against overflow
        a = a * factor
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score

def _hill_climb_enhanced(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # dynamic transfer fraction (up to 5 % of the source mass)
        max_frac = 0.05
        frac = rng.random() * max_frac + 1e-5
        eps = a[src] * frac
        if eps <= 0:

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-14:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
