"""
Propose a low‑score non‑negative step function.

The algorithm consists of three stages:

1. **Candidate pool** – generate a few analytic shapes (uniform, beta,
   mixtures, a previous best if available) and keep the best one.

2. **Continuation gradient descent** – minimise a smooth approximation of
   the objective `2·n·max(conv(f,f))/sum(f)^2`.  The smooth max is a
   temperature‑controlled soft‑max (log‑sum‑exp) of the self‑convolution.
   The gradient of the smooth max is obtained analytically and evaluated
   by a single FFT‑based convolution.  Adam with projection onto the
   probability simplex drives the update; the temperature is slowly
   increased so that the optimisation first sees a smoothed landscape
   and finally concentrates on the true maximum.

3. **Short annealing polish** – a lightweight simulated‑annealing step
   swaps tiny packets of mass between random positions to escape any
   remaining local minimum.

The whole routine respects a 1000 s time budget and returns a plain
list of Python floats (non‑negative, summing to one).  The implementation
uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_i w_i·(a*a)_i  with respect to a.
    It equals 2·∑_i w_i·shift_i(a).  Computed by one FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse the weight array
    w_rev = w[::-1]
    L = _next_pow2(3 * n - 2)          # length needed for a * w_rev
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    full = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]   # linear convolution
    # g_k = 2 * full[2n-2 - k]   for k = 0..n-1
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the competition:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule on (0,1)
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# continuation gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of the max‑convolution.
    Returns the best vector seen and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1

        # 1) self‑convolution
        conv = _conv_self(a)
        max_conv = conv.max()

        # 2) soft‑max weights (temperature schedule)
        # early iterations: low temperature → smooth landscape
        # later iterations: high temperature → close to true max
        progress = (time.time() - start_time) / (deadline - start_time)
        # α grows from 2e3 to 1e5 roughly
        alpha = 2e3 * (1.0 + 49.0 * progress)          # 2e3 … 1e5
        # stable soft‑max
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # 3) gradient of the smooth max
        grad = _weighted_grad(a, w)

        # 4) Adam update (minimise → step opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decay learning rate roughly as 1/√t, capped to keep steps moderate
        base_lr = 0.18
        lr = base_lr / (math.sqrt(it) + 1.0)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # 5) occasional random jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = a + jitter
        # 6) project back onto the simplex
        a = _project_simplex(a)

        # 7) true objective (expensive but cheap compared to FFTs)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score

# light simulated‑annealing polish (same as previous code, slightly tweaked)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny simulated‑annealing polish."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 400_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling from 2e‑2 to 1e‑6
        T = 2e-2 * (1e-6 / 2e-2) ** (it / max_steps)

        src = rng.choice(n, p=a)          # bias towards mass for faster changes
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * rng.random() * 0.45   # move up to 45 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

        best_seq, best_score = _softmax_gradient_descent(best_seq,
                                                         gd_deadline,
                                                         start_time)
        print(f"[{time.time() - start_time:.2f}s] After GD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light simulated‑annealing polish (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalise and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
