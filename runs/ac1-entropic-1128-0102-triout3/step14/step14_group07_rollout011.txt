def _subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # find positions within tolerance
        # use tolerance relative: e.g., 1e-12 absolute may be too strict.
        # We'll select all indices where max_val - conv <= 1e-12 (small).
        # Determine indices where conv close to max.
        mask = conv >= max_val - 1e-12
        # create weight vector w of same length as conv, uniform on mask.
        w = np.zeros_like(conv)
        count = mask.sum()
        if count == 0:
            # Should not happen
            w[np.argmax(conv)] = 1.0
        else:
            w[mask] = 1.0 / count

        # compute gradient:
        a_rev = a[::-1]
        L = _next_pow2(3 * a.shape[0] - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * a.shape[0] - 2]
        grad = 2.0 * corr[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # step size schedule:
        eta0 = 0.2
        eta = eta0 / math.sqrt(it)
        # gradient descent
        a -= eta * grad
        # projection onto simplex
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} score={cur_score:.6f}")
        # optional early exit if gradient norm small?
    return best_a, best_score

def _targeted_simulated_annealing(a0, deadline, start_time, rng, max_move_frac=0.03):
   a = a0.copy()
   cur_score = _objective(a)
   best_a = a.copy()
   best_score = cur_score
   n = a.shape[0]
   T_start = 0.1
   T_end = 1e-8
   # precompute conv each step.
   while time.time() < deadline:
       elapsed = time.time() - start_time
       frac = elapsed / (deadline - start_time)
       T = max(T_end, T_start * ((T_end/T_start) ** frac))
       conv = _conv_self(a)
       max_val = conv.max()
       # compute contributions for each index for the max shift(s)
       kmax = np.argmax(conv)
       # compute contributions array c_i = a_i * a_{kmax-i} if in range else 0
       c = np.zeros(n)
       start_i = max(0, kmax - (n-1))
       end_i = min(n-1, kmax)
       # vectorized:
       i_arr = np.arange(start_i, end_i+1)
       j_arr = kmax - i_arr
       c[i_arr] = a[i_arr] * a[j_arr]
       # sample src index weighted by contributions
       total_c = c.sum()
       if total_c > 0:
           src = rng.choice(n, p=c/total_c)
       else:
           src = rng.choice(n, p=a)  # fallback
       # avoid src being zero
       if a[src] < 1e-16:
           continue
       # select dst random (different)
       dst = rng.integers(n)
       if dst == src:
           continue
       # propose mass shift
       delta = a[src] * max_move_frac * rng.random()
       if delta <= 0.0:
           continue
       a_prop = a.copy()
       a_prop[src] -= delta
       a_prop[dst] += delta
       new_score = _objective(a_prop)
       if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
           a = a_prop
           cur_score = new_score
           if new_score < best_score:
               best_score = new_score
               best_a = a.copy()
               print(f"[{time.time() - start_time:.2f}s] Target SA new best={new_score:.6f}")
   return best_a, best_score

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The search runs three stages:

1. **Diverse seeding** – a richer set of analytic shapes (power‑ramps,
   symmetric Beta densities, exponential‑edge profiles) and a few
   random Dirichlet draws are generated.  The best K candidates are kept.

2. **Targeted optimisation** – the current best candidate is refined by a
   pure sub‑gradient descent on the exact max‑convolution (no smoothing),
   projected onto the probability simplex.  This drives the maximal
   convolution down directly.  A short Adam run on the top few candidates
   adds extra diversity.

3. **Focused simulated‑annealing** – mass is transferred from indices that
   contribute most to the current worst‑case convolution shift to random
   destinations, with a temperature schedule.  This fine‑tunes the last
   few digits of the bound.

All phases respect a 1000 s budget and return a plain list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·len(a)−1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Objective for a probability vector (sum ≈ 1). Returns 2·n·max(conv)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted pool of the k best (lowest) candidates."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_decay_shape(n: int, decay: float) -> np.ndarray:
    """Symmetric exponential edge profile a[i] ∝ exp(-decay·i)+exp(-decay·(n-1-i))."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-decay * idx) + np.exp(-decay * (n - 1 - idx))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Adam optimiser on a temperature‑softened max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    T_start, T_end = 0.30, 1e-6
    step0 = 0.15

    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    n = a.shape[0]

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # indices attaining (or nearly attaining) the maximum
        mask = conv >= max_val - 1e-12
        w = np.zeros_like(conv)
        cnt = mask.sum()
        if cnt == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[mask] = 1.0 / cnt

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # simple diminishing step size
        eta0 = 0.2
        eta = eta0 / math.sqrt(it)
        a -= eta * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


def _targeted_simulated_annealing(a0: np.ndarray,
                                  deadline: float,
                                  start_time: float,
                                  rng: np.random.Generator,
                                  max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """SA that preferentially moves mass from indices that dominate the worst shift."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    T_start, T_end = 0.1, 1e-8
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # find the shift with the current maximum
        conv = _conv_self(a)
        kmax = int(conv.argmax())

        # contributions of each index to that shift
        contrib = np.zeros(n)
        i0 = max(0, kmax - (n - 1))
        i1 = min(n - 1, kmax)
        i = np.arange(i0, i1 + 1)
        j = kmax - i
        contrib[i] = a[i] * a[j]

        # choose a source index weighted by its contribution
        total_c = contrib.sum()
        if total_c > 0:
            src = rng.choice(n, p=contrib / total_c)
        else:
            src = rng.choice(n, p=a)  # fallback to probability‑weighted

        if a[src] < 1e-16:
            continue

        # choose a destination uniformly (different from src)
        dst = rng.integers(n)
        if dst == src:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Target SA new best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 9999)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose a discretisation size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a diverse pool of candidates
    # --------------------------------------------------------------
    TOP_K = 12
    pool = []      # (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps (broader grid)
    p_grid = np.concatenate([np.linspace(0.1, 0.5, 5),
                             np.linspace(0.6, 3.0, 8)])
    plate_grid = np.linspace(0.0, 0.6, 9)
    for p in p_grid:
        for plat in plate_grid:
            a = _make_shape(N, p, plat)
            _push_top(pool, _objective(a), a, TOP_K)

    # symmetric Beta shapes (more alphas)
    alphas = np.concatenate([np.arange(0.01, 0.11, 0.01),
                             np.arange(0.15, 0.95, 0.1)])
    for alpha in alphas:
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _objective(a), a, TOP_K)

    # exponential edge profiles
    decays = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]
    for d in decays:
        a = _make_exp_decay_shape(N, d)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 0.8]
    for conc in concs:
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)

    # incorporate a previously found global starter, if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Refine the current best candidate with sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        sub_deadline = time.time() + 0.55 * remaining
        best_seq, best_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light‑weight Adam runs on a few top candidates (optional)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20:
        per_candidate = remaining / min(len(pool), 3)
        for idx, (_, vec) in enumerate(pool[:3]):
            sub_dl = time.time() + per_candidate
            if sub_dl >= deadline:
                break
            refined_seq, refined_score = _adam_smooth_max(vec, sub_dl, start_time)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam candidate {idx} improved best to {refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Targeted simulated‑annealing polish
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_score = _targeted_simulated_annealing(
            best_seq, deadline, start_time, rng, max_move_frac=0.02)

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
