def project_simplex(v):
    """Project v onto the probability simplex {x ≥0, sum=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # ensures sum(v) = 1, v >= 0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros? return uniform distribution
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    return np.fft.irfft(Af * Bf, n=L)[:len(a) + len(b) - 1]

def _project_simplex(v):
    """Project onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero or negative, return uniform distribution
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent search for a non‑negative step function that minimizes
    ``2·n·max(conv(a,a))/(sum a)^2``.  The algorithm works on the probability
    simplex (sum = 1) and uses a smoothed (log‑sum‑exp) approximation of the
    max‑convolution to obtain a dense gradient.  A simple projected‑gradient
    scheme with a multiplicative increase of the smoothing parameter drives the
    sequence toward a flat self‑convolution, which empirically beats the
    previous exponentiated‑gradient + annealing baseline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Choose a power‑of‑two size that balances resolution and speed.
    n = 8192  # 2¹³

    # Initial candidate: Dirichlet (α=1) → uniform random point on the simplex.
    a = rng.random(n)
    a = a / a.sum()

    # Hyper‑parameters
    step0 = 0.4                # initial additive step size
    beta = 5.0                 # start smoothing (log‑sum‑exp temperature)
    beta_max = 200.0
    beta_factor = 1.02
    it = 0
    best_a = a.copy()
    # Current actual score (using exact max)
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    # Main projected‑gradient descent loop.
    while time.time() < deadline:
        it += 1

        # FFT of current vector – reused for all three convolutions.
        L = _next_pow2(3 * n)   # enough for a * w and a * w_rev
        Af = np.fft.rfft(a, n=L)

        # 1) exact self‑convolution for the true objective.
        conv_self = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        cur_max = conv_self.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  new best={cur_score:.6f}")

        # 2) softmax weights that concentrate on large entries of conv_self.
        scaled = beta * conv_self
        # stable exponentiation
        shift = scaled.max()
        w = np.exp(scaled - shift)
        w /= w.sum()                     # shape (2n‑1)

        # 3) gradient of the smoothed objective.
        w_rev = w[::-1]

        # FFT of the weight vectors (same length L).
        Wf = np.fft.rfft(w, n=L)
        Wrevf = np.fft.rfft(w_rev, n=L)

        # a * w  and  a * w_rev  (linear convolutions).
        conv_aw = np.fft.irfft(Af * Wf, n=L)[: n + len(w) - 1]      # length 3n‑2
        conv_aw_rev = np.fft.irfft(Af * Wrevf, n=L)[: n + len(w) - 1]

        # term2 = Σ_j a_j w[i‑j]   → first n entries of conv_aw
        term2 = conv_aw[:n]

        # term1 = Σ_j a_j w[i + j]   → indices (len(w)‑1‑i) of conv_aw_rev
        idx = (len(w) - 1) - np.arange(n)   # integer array length n
        term1 = conv_aw_rev[idx]

        grad = 2.0 * (term1 + term2)          # ∇_a  ( log‑sum‑exp(conv) )

        # 4) additive step (gradient descent) with diminishing step size.
        step = step0 / math.sqrt(it)
        a_new = a - step * grad

        # 5) projection back onto the simplex.
        a = _project_simplex(a_new)

        # 6) slowly tighten the smoothing.
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    # --------------------------------------------------------------
    # 7) final polishing stage – short simulated annealing.
    # --------------------------------------------------------------
    # If we still have a few seconds left, run a very light anneal.
    if time.time() + 5.0 < deadline:
        a = best_a.copy()
        cur_score = best_score
        # Simple temperature schedule (linear cooling).
        T0 = 0.05
        Tend = 1e-6
        max_steps = 200_000
        for step_no in range(1, max_steps + 1):
            if time.time() >= deadline:
                break
            T = T0 * (Tend / T0) ** (step_no / max_steps)
            # Choose a source proportional to current mass.
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
            eps = a[src] * rng.random() * 0.3   # move up to 30% of src mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            # quick objective (full conv) – inexpensive compared to many steps.
            conv_test = _conv_self(a_prop)
            new_score = 2.0 * n * conv_test.max()
            if new_score < cur_score or math.exp(-(new_score - cur_score) / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={step_no:,}  best={new_score:.6f}")

    # --------------------------------------------------------------
    # 8) return the best sequence found (as plain Python floats).
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]

Propose a low‑score non‑negative step function.

The algorithm works on the probability simplex (∑a_i = 1) and uses a
smooth (log‑sum‑exp) approximation of the max‑convolution to obtain a dense
gradient.  A simple projected‑gradient descent with a gradually increasing
smoothing parameter drives the self‑convolution of the vector towards a flat
profile, which empirically beats the previous exponentiated‑gradient + anneal
baseline.  A light simulated‑annealing polish is applied at the end if time
allows.

The returned list contains non‑negative floats; its scale is irrelevant
(because the evaluation function is scale‑invariant) but we keep the sum

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """
    Project onto the probability simplex.
    Here we simply clip negatives to a tiny floor and renormalise.
    """
    eps = 1e-15
    v = np.maximum(v, eps)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v

    beta_max = 200.0

    it = 0

    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    L = _next_pow2(3 * n)        # enough for a * w and a * w_rev (length 3 n‑2)

    # ------------------------------------------------------------------

    while time.time() < deadline:

        Af = np.fft.rfft(a, n=L)

        conv_self = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        cur_max = conv_self.max()

            a_prop[src] -= eps

            if new_score < cur_score or math.exp(-(new_score - cur_score) / T) > rng.random():
                a = a_prop

    # 6) Normalise the best vector and return it as a plain Python list.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
