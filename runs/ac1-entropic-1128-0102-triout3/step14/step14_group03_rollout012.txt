    Af = np.fft.rfft(a, n=L_conv)

"""
Hybrid optimisation for the 2·n·max(conv(a,a)) / (sum(a)**2) problem.

1️⃣  A cheap parametric sweep explores two families:
    – geometric × power (r**i·(i+1)**p) mixed with the uniform vector
    – centred Gaussian shapes (σ‑controlled) mixed with uniform.
   The best sequence from this stage seeds the next phase.

2️⃣  A smoothing‑gradient descent minimises a smooth approximation of the
    max‑convolution:
        Sα(a) = (1/α)·log∑_k exp(α·(a*a)_k) .
    Its gradient is
        ∇Sα = 2·corr(a , w)   with   w_k ∝ exp(α·(a*a)_k) .
    The algorithm performs projected (onto the probability simplex) gradient
    steps with a decreasing step length, slowly raising α to focus on the true
    max.  The best iterate (according to the exact objective) is kept.

3️⃣  A final tiny random “mass‑transfer” hill‑climb cleans up any remaining
    spikes.

All phases respect the total 1000 s time budget.  The returned list
contains non‑negative numbers that sum to 1, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Exact linear self‑convolution a*a (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _correlate_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation of a (len n) with w (len m) → length n+m‑1.
    Returns the sub‑vector that corresponds to
        g_i = 2·∑_j a_j·w_{i+j}   (i = 0…n‑1)
    """
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(Af * np.conj(Wf), n=L)
    # keep the linear part only
    corr = corr_full[: a.shape[0] + w.shape[0] - 1]
    offset = w.shape[0] - 1
    grad = 2.0 * corr[offset : offset + a.shape[0]]
    return grad


def _eval_fast(a: np.ndarray, L_conv: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self_fft(a, L_conv)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)   # denominator = 1 because sum(a)=1


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------


def _make_geo_pow(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape: proportional to r**i·(i+1)**p."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)                     # prevent overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered discrete Gaussian (truncated) normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _mix_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """lam·uniform + (1‑lam)·shape (both already sum = 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()       # safeguard against rounding errors
    return mixed


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------


def _smooth_max_gradient_descent(a0: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 best_score: float,
                                 best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on Sα(a) = (1/α)·log∑exp(α·(a*a)_k).
    α is increased slowly; the exact objective is monitored and the best
    iterate is kept.
    """
    n = a0.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)          # sufficient for correlation
    a = a0.copy()
    best_a = best_seq.copy()
    cur_score = best_score

    rng = np.random.default_rng()
    step_scale = 0.5                       # controls Euclidean step size
    alpha_start = 5.0
    alpha_end = 80.0
    ramp_iters = 20000                     # iterations over which α grows

    it = 0
    while time.time() < deadline:
        it += 1

        # --- convolution (for b and for soft‑max) -----------------------
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]

        # --- soft‑max weights -----------------------------------------
        alpha = (alpha_start + (alpha_end - alpha_start) *
                 min(1.0, it / ramp_iters))
        maxc = conv.max()
        # stabilise exp
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()          # length 2·n‑1, sums to 1

        # --- gradient (correlation a⟂w) --------------------------------
        g = _correlate_fft(a, w, L_corr)

        # normalise step length
        norm_g = np.linalg.norm(g) + 1e-30
        step = step_scale / norm_g

        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        # exact objective of the candidate (fast because we already have FFT)
        conv_cand = _conv_self_fft(a_candidate, L_conv)
        cand_score = float(2.0 * n * conv_cand.max())     # sum≈1

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            cur_score = cand_score
            a = a_candidate
            print(
                f"[{time.time() - start_time:.2f}s] grad it={it:,} "
                f"α={alpha:.1f} score={cand_score:.6f}"
            )
            continue

        # occasional uphill move (simulated annealing flavour)
        if cand_score < cur_score or rng.random() < np.exp(
            -(cand_score - cur_score) * 5.0 / np.sqrt(it + 1.0)
        ):
            a = a_candidate
            cur_score = cand_score

        # small random jitter every few thousand iterations
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray,
                      deadline: float,
                      start_time: float,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Randomly moves a tiny fraction of mass between two positions."""
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3       # ≤0.2 % transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv = _conv_self_fft(a_prop, L_fft)
        cand_score = float(2.0 * n * conv.max())
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_prop.copy()
            a = a_prop
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={cand_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).

    The routine performs:
      1. a fast parametric sweep (geometric‑power & Gaussian families),
      2. a smoothed‑max projected gradient descent,
      3. a tiny random hill‑climb.
    It respects the 1000 s budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_conv = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initial candidate: uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_conv)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _eval_fast(starter, L_conv)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                    print(f"[{time.time() - start:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = start + 130.0          # 130 s for sweeping
    # (r,p,λ) family
    r_grid = np.logspace(np.log10(0.92), np.log10(0.9995), num=25)
    p_grid = np.linspace(-2.5, 1.5, 31)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_geo_pow(n, r, p)
            for lam in lam_grid:
                cand = _mix_uniform(shape, lam)
                sc = _eval_fast(cand, L_conv)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start:.2f}s] sweep geo r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # (Gaussian, λ) family
    sigma_grid = np.linspace(n / 40.0, n / 3.0, 15)  # from ~0.025·n to ~0.33·n
    for sigma in sigma_grid:
        if time.time() >= sweep_deadline:
            break
        shape = _make_gaussian(n, sigma)
        for lam in lam_grid:
            cand = _mix_uniform(shape, lam)
            sc = _eval_fast(cand, L_conv)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start:.2f}s] sweep gauss σ={sigma:.1f} λ={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed‑max projected gradient descent
    # --------------------------------------------------------------
    grad_deadline = deadline - 150.0          # reserve ~150 s for final hill‑climb
    if time.time() < grad_deadline:
        best_seq, best_score = _smooth_max_gradient_descent(
            best_seq, grad_deadline, start, best_score, best_seq
        )
        print(f"[{time.time() - start:.2f}s] after gradient descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final tiny hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _final_hill_climb(best_seq, deadline, start, best_score)

    # --------------------------------------------------------------
    # Normalise finally (numerical safety) and return as list of floats.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
