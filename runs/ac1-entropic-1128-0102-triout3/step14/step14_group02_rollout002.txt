def simulated_annealing(initial_seq, time_budget, start_time):
    a = initial_seq.copy()
    n = len(a)
    cur_score = _objective_score(a)
    best_seq = a.copy()
    best_score = cur_score
    # schedule: initial temperature based on observed score range
    T0 = 1e-2  # tuned
    T_min = 1e-6
    # exponential decay: determine rate such that after certain number of steps we reach T_min
    # We'll just update T = max(T_min, T0 * 0.9999**step)
    step = 0
    while time.time() < time_budget:
        step += 1
        # Choose move
        src = rng.integers(n)
        while a[src] < 1e-12:
            src = rng.integers(n)
        dst = rng.integers(n)
        # ensure different
        while dst == src:
            dst = rng.integers(n)
        # transfer amount
        frac = rng.random() * 0.02 + 1e-5   # up to 2%
        eps = a[src] * frac
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # keep nonnegative and sum preserved
        # ensure positivity (should hold)
        # compute score
        new_score = _objective_score(a_new)
        delta = new_score - cur_score
        # temperature schedule
        T = max(T_min, T0 * (0.9995 ** step))

def _simulated_annealing(initial_seq, deadline, start_time):
    rng = np.random.default_rng()
    a = initial_seq.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    best_seq = a.copy()
    best_score = cur_score

    step = 0
    T0 = 0.01
    T_min = 1e-8
    decay = 0.9995

    while time.time() < deadline:
        step += 1
        # choose move type
        if rng.random() < 0.9:
            # simple transfer
            src = rng.integers(n)
            while a[src] < 1e-14:
                src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

            a_new[src] -= eps
            a_new[dst] += eps

"""
Propose a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search is based on a multi‑restart simulated‑annealing (SA) scheme
augmented with a few cheap random Dirichlet seeds and a final simplex
projection.  The algorithm works on the full length (default n=4096) and
uses FFT for fast convolution.  A SA run consists of tiny mass‑transfer
moves that are sometimes accepted uphill (temperature‑controlled).  The
temperature decays exponentially; occasional random perturbations keep
the walk from stalling.  All time is split between quick random starts
and repeated SA runs until the 1000 s budget is exhausted.  The best
vector found (normalised to a probability simplex) is returned as a
plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------------
# Linear‑algebra helpers
# ------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------------
# Objective helper (scale‑invariant – sum is assumed to be 1)
# ------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------------
# Simulated‑annealing optimiser (single run)
# ------------------------------------------------------------------
def _simulated_annealing(initial_seq: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform SA on a probability vector (sum=1, non‑negative).
    Returns the best vector seen and its objective value.
    """
    rng = np.random.default_rng()
    a = initial_seq.copy()
    n = a.shape[0]

    cur_score = _objective_score(a)
    best_seq = a.copy()
    best_score = cur_score

    step = 0
    # temperature schedule
    T0 = 5e-3          # start temperature (tuned experimentally)
    T_min = 1e-9
    decay = 0.9995     # exponential decay factor

    while time.time() < deadline:
        step += 1

        # ----- propose a move -------------------------------------------------
        if rng.random() < 0.85:                     # simple mass transfer
            src = rng.integers(n)
            while a[src] < 1e-14:
                src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

            # early in the run we allow a slightly larger fraction
            max_frac = 0.02 if step % 2000 == 0 else 0.005
            frac = rng.random() * max_frac + 1e-6
            eps = a[src] * frac
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
        else:                                       # small random perturbation
            a_new = a.copy()
            idxs = rng.choice(n, size=3, replace=False)
            a_new[idxs] += rng.normal(scale=0.003, size=3)
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)

        # ----- evaluate the move ---------------------------------------------
        # (mass‑transfer preserves the ℓ1‑norm; perturbation already projected)
        new_score = _objective_score(a_new)
        delta = new_score - cur_score

        # ----- Metropolis acceptance -----------------------------------------
        T = max(T_min, T0 * (decay ** step))
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] SA new best score={best_score:.6f}")

        # ----- occasional large perturbation to escape plateaus ---------------
        if step % 20000 == 0:
            # re‑normalise to the simplex (numerical drift may appear)
            a = _project_simplex(np.maximum(a, 0.0))
            cur_score = _objective_score(a)

    return best_seq, best_score

# Main entry point
# ------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The implementation uses many short random starts
    followed by simulated‑annealing with restarts; all work respects the
    supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Quick random Dirichlet initialisers (consume a few seconds)
    # ------------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)          # concentration between ~0.01 and 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet seed score={sc:.6f}")

    # ------------------------------------------------------------------
    # Main SA loop with restarts
    # ------------------------------------------------------------------
    # Each SA run gets roughly 35‑45 s; the loop continues until time out.
    while time.time() < deadline:
        time_left = deadline - time.time()
        run_time = min(42.0, time_left)          # a few seconds for the final fragment are useless
        if run_time < 5.0:
            break

        # Decide the starting point for this SA run
        if rng.random() < 0.3:
            # completely fresh random start
            conc = 10 ** rng.uniform(-2.5, 0.5)
            a0 = rng.gamma(shape=conc, scale=1.0, size=n)
            a0 /= a0.sum()
        else:
            # perturb the current global best a little
            a0 = best_seq.copy()
            idxs = rng.choice(n, size=12, replace=False)
            a0[idxs] += rng.normal(scale=0.008, size=12)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

        # Run a single SA instance
        seq, sc = _simulated_annealing(a0, time.time() + run_time, start_time)

        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after SA run score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
