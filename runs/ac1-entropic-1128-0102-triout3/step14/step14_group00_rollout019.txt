"""
Propose a low‑score non‑negative step function.

The algorithm works in three stages:

1.  Build a modest pool of analytic candidates (beta‑distributions,
    random convex mixtures, uniform).  Keep the best few.

2.  Refine each of those candidates with Adam (adaptive‑momentum) on
    the probability simplex.  The gradient is the sub‑gradient of the
    maximal entry of the self‑convolution, obtained from the active
    set of indices.  A small learning‑rate schedule and bias‑correction
    are used.  The best sequence seen so far is remembered.

3.  Return the best sequence, normalised to sum = 1 (the objective is
    scale‑invariant).  The whole routine respects the supplied time
    budget (default 1000 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   budget_s: float,
                   start_time: float,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam (adaptive‑momentum) descent on the probability simplex.
    Returns the best vector seen and its objective value.
    """
    n = a0.shape[0]
    # initialise logits (avoid log(0))
    eps_floor = 1e-14
    a = np.maximum(a0, eps_floor)
    w = np.log(a)

    # Adam moments
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    deadline = start_time + budget_s

    while time.time() < deadline:
        t += 1

        # softmax → simplex
        a = np.exp(w)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # convolution and active set
        conv = _conv_self(a)
        max_val = conv.max()
        # tolerance relative to the maximum
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of the max‑convolution w.r.t. a
        grad_a = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            # derivative = 2 * a_{k-i}
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= float(len(ks))

        # transform gradient to logits (mirror descent)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        w = w - step

        # evaluate
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a length that gives a good trade‑off (FFT speed vs resolution)
    # --------------------------------------------------------------
    n = 4096      # 2¹¹ – modest size, fast FFTs, still enough resolution

    # --------------------------------------------------------------
    # 2) Build a small pool of promising analytic candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # Random beta‑distribution shapes (parameters drawn log‑uniformly,
    # allowing very sharp U‑shapes)
    for _ in range(200):
        a = 10 ** rng.uniform(-6.0, 1.0)   # roughly 1e‑6 … 10
        b = 10 ** rng.uniform(-6.0, 1.0)
        v = _make_beta_shape(n, a, b)
        pool.append((_objective(v), v.copy()))

    # Random convex combinations of two beta shapes
    for _ in range(100):
        a1 = 10 ** rng.uniform(-6.0, 1.0)
        b1 = 10 ** rng.uniform(-6.0, 1.0)
        a2 = 10 ** rng.uniform(-6.0, 1.0)
        b2 = 10 ** rng.uniform(-6.0, 1.0)
        w = rng.random()
        v = w * _make_beta_shape(n, a1, b1) + (1.0 - w) * _make_beta_shape(n, a2, b2)
        v /= v.sum() if v.sum() > 0 else 1.0
        pool.append((_objective(v), v.copy()))

    # Keep the best few candidates for refinement
    pool.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in pool[:5]]
    best_score, best_seq = pool[0]

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Refine each candidate with Adam (budget‑aware)
    # --------------------------------------------------------------
    for idx, cand in enumerate(top_candidates):
        remaining = deadline - time.time()
        if remaining < 10.0:        # not enough time left
            break
        # allocate at most half of the remaining time to a single run
        run_budget = min(remaining * 0.5, 300.0)
        seq, score = _adam_optimize(cand, run_budget, start_time)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after Adam on cand {idx} = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
