def _adam_subgradient(a0, deadline, start_time, max_iter=int(1e7)):
    a = a0.copy()
    n = a.shape[0]

    beta1=0.9
    beta2=0.999
    eps=1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)  # as before computes 2*n*max

    it=0
    # maybe fixed learning rate 0.1 decaying
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        max_idx = conv.argmax()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam subgrad iter {it} new best {cur_score:.6f}")

        # gradient for argmax: for i in [left,right], g_i = 2 * a[max_idx - i]
        g = np.zeros_like(a)
        left = max(0, max_idx - (n-1))
        right = min(n-1, max_idx)
        # indices i = left..right
        i_idx = np.arange(left, right+1)
        j_idx = max_idx - i_idx
        g[i_idx] = 2.0 * a[j_idx]

        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.1 / np.sqrt(it)  # maybe adapt
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _directed_sa(a0, deadline, start_time, rng, max_move_frac=0.03):

    cur_max = conv.max()
    cur_idx = conv.argmax()
    cur_score = 2.0 * len(a) * cur_max
    best_a = a.copy()
    best_score = cur_score

    n = len(a)

    while time.time() < deadline:
        # compute gradient direction using current max idx
        k = cur_idx
        # compute weight w_i = a[k-i] for i in range where k-i within [0,n-1]; else 0
        w = np.zeros(n)
        left = max(0, k - (n-1))
        right = min(n-1, k)
        i_idx = np.arange(left, right+1)
        j_idx = k - i_idx
        w[i_idx] = a[j_idx]

        # pick source probabilistically according to w * a maybe? Actually choose source with high w (large contribution)
        # we'll sample source from indices with probability proportional to a[i] * w[i] (the contribution to max conv)
        also compute contrib = a * w
        total_contrib = contrib.sum()
        if total_contrib == 0:
            break
        contrib_prob = contrib / total_contrib
        src = rng.choice(n, p=contrib_prob)
        # find dest with low w (maybe zero)
        # we can sample dest uniformly among indices where w is low (e.g., bottom 20% percentile)
        low_indices = np.where(w < np.percentile(w, 20))[0]
        if low_indices.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_indices)

        if src == dst:
            continue

        # compute delta = min(max_move_frac * a[src], maybe also limited)
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 1e-12:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # compute new conv and score
        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            cur_max = new_max
            cur_idx = conv_prop.argmax()
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Directed SA new best {new_score:.6f}")
        else:
            # accept with probability exp(-(new_score-cur_score)/T)
            # Temperature schedule similar to SA_t = T_start * ((T_end/T_start)**frac)
            # compute frac = elapsed / total
            elapsed = time.time() - start_time
            total = deadline - start_time

                cur_score = new_score
                cur_max = new_max
                cur_idx = conv_prop.argmax()

"""
Propose a step‑function (non‑negative heights) that minimizes  

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Initial pool** – a handful of diverse probability vectors
   (uniform, a lifted previous best, a few random Dirichlet draws).

2. **Projected sub‑gradient Adam** – the true (non‑smoothed) max‑convolution
   gradient is a simple “mirrored” vector:
   ∂max/∂a_i = 2·a_{k*‑i} where k* is the current arg‑max index.
   We run an Adam optimiser on this sub‑gradient, projecting the iterate
   back onto the simplex after each step.

3. **Directed simulated‑annealing polish** – using the current worst‑contributing
   indices (large a_i·a_{k*‑i}) we move a small fraction of mass to the
   least‑contributing positions.  The move is accepted either if it lowers
   the objective or with a temperature‑controlled Metropolis probability.

All phases share the global time budget (default 1000 s).  The final
vector is normalised to sum 1 and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective = 2·n·max(conv(a,a)) (the sum of a is assumed to be 1)."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain the k best (lowest) items in *pool*."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Adam optimisation using the *exact* sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Adam on the sub‑gradient ∂max/∂a_i = 2·a_{k*‑i}."""
    a = a0.copy()
    n = a.shape[0]

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # a modest initial learning‑rate; it decays with √t
    lr0 = 0.2

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        max_idx = np.argmax(conv)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # ---- sub‑gradient for the current arg‑max ----
        g = np.zeros_like(a)
        left = max(0, max_idx - (n - 1))
        right = min(n - 1, max_idx)
        i_idx = np.arange(left, right + 1, dtype=int)
        j_idx = max_idx - i_idx
        g[i_idx] = 2.0 * a[j_idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)                     # decaying step
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Directed simulated‑annealing (mass moves biased by contribution)
# ----------------------------------------------------------------------
def _directed_sa(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 rng: np.random.Generator,
                 max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    """Move mass away from indices that heavily contribute to the peak."""
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_self(a)
    cur_max = conv.max()
    cur_idx = np.argmax(conv)
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (same as the previous implementation)
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # contribution of each index to the current max (a_i * a_{k*-i})
        left = max(0, cur_idx - (n - 1))
        right = min(n - 1, cur_idx)
        contrib = np.zeros(n)
        i_idx = np.arange(left, right + 1, dtype=int)
        j_idx = cur_idx - i_idx
        contrib[i_idx] = a[i_idx] * a[j_idx]

        # source: biased to large contributions
        if contrib.sum() == 0.0:
            break
        src = rng.choice(n, p=contrib / contrib.sum())

        # destination: look for indices with *tiny* contributions
        low_thr = np.percentile(contrib, 20)
        low_candidates = np.flatnonzero(contrib <= low_thr)
        if low_candidates.size == 0:
            dst = rng.integers(n)
        else:
            dst = rng.choice(low_candidates)

        if src == dst:
            continue

        # amount to move (a small fraction of the source weight)
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 1e-16:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # evaluate the new candidate
        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            cur_max = new_max
            cur_idx = np.argmax(conv_prop)
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a list of non‑negative floats (probability vector).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a discretisation size (larger gives more freedom, but costs time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build an initial pool of candidates
    # ------------------------------------------------------------------
    TOP_K = 5
    pool = []   # holds (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # previous best, if the global variable exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != N:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, prev.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                prev /= s
                _push_top(pool, _objective(prev), prev, TOP_K)
    except Exception:
        pass

    # a few random Dirichlet draws with different concentration parameters
    concs = [0.05, 0.2, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(4):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            total = vec.sum()
            if total > 0:
                vec /= total
                _push_top(pool, _objective(vec), vec, TOP_K)

    # current best
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation with the exact sub‑gradient (one run per top candidate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_candidate = remaining / max(len(pool), 1)

    for idx, (_, vec) in enumerate(pool):
        sub_deadline = time.time() + per_candidate
        if sub_deadline >= deadline:
            break
        opt_vec, opt_score = _adam_subgrad(vec, sub_deadline, start_time)
        if opt_score < best_score:
            best_score, best_vec = opt_score, opt_vec.copy()
            print(f"[{time.time() - start_time:.2f}s] Candidate {idx} improved best to {opt_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Directed simulated‑annealing polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_vec, best_score = _directed_sa(best_vec, deadline, start_time, rng)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
