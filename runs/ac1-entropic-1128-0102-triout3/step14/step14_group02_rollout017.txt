"""
Propose a non‑negative step function (probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is scale‑invariant, therefore we keep the vector on the
simplex (sum = 1).  The search consists of three main phases:

1️⃣  **Fast parametric / random families** – geometric, two‑sided
    mixtures, power laws and random Beta‑shapes.  Hundreds of
    candidates are generated and evaluated with a cheap FFT‑based
    objective.

2️⃣  **Large‑scale random Dirichlet sampling** – draw thousands of
    Dirichlet vectors with a very wide log‑scale concentration
    parameter (down to 10⁻⁴) to explore extremely sparse and very
    dense regimes.

3️⃣  **Iterated sub‑gradient descent + final hill‑climb** – from the
    best candidates found so far we run a simple projected sub‑gradient
    method (step size ∝ 1/√t) that directly uses the sub‑gradient of the
    max‑convolution.  After the time budget is nearly exhausted a
    directed hill‑climb with line‑search (moving a tunable fraction of
    mass from the most “harmful’’ index to the most “helpful’’ one) gives
    the final polish.

All stages respect the global 1000 s budget.  The routine prints
intermediate scores so that progress can be followed.  The returned list
contains plain Python floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective assuming sum(seq) == 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# ----------------------------------------------------------------------
def _random_parametric_sweep(n, best_seq, best_score, deadline):
    """
    Generate a few thousand candidate vectors from simple analytic families
    (geometric, two‑sided mixtures, power, random Beta) and keep the best.
    """
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)

    # pre‑compute logarithms for Beta‑type families
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    # ----- geometric (single side) ------------------------------------
    for _ in range(200):
        if time.time() >= deadline:

        r = rng.uniform(0.90, 0.9999)          # decay factor
        geo = np.power(r, idx)
        seq = geo / geo.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo r={r:.5f} score={sc:.6f}")

    # ----- two‑sided geometric mixture --------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        r_left = rng.uniform(0.90, 0.9999)
        r_right = rng.uniform(0.90, 0.9999)
        lam = rng.random()
        left = np.power(r_left, idx)
        right = np.power(r_right, idx[::-1])
        seq = lam * left + (1.0 - lam) * right
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] MixGeo λ={lam:.3f} rl={r_left:.5f} rr={r_right:.5f} score={sc:.6f}")

    # ----- power law --------------------------------------------------
    for p in np.linspace(-5.0, 5.0, 301):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ----- random Beta‑type families (α,β) ---------------------------
    for _ in range(500):
        if time.time() >= deadline:
            break
        a = 10 ** rng.uniform(-2, 1)          # α ∈ [0.01,10]
        b = 10 ** rng.uniform(-2, 1)          # β ∈ [0.01,10]
        log_seq = (a - 1.0) * log_idx + (b - 1.0) * log_rev
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandBeta α={a:.3f} β={b:.3f} score={sc:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def _massive_dirichlet_trials(n, best_seq, best_score, deadline):
    """
    Draw thousands of Dirichlet vectors with concentration ranging from
    10⁻⁴ up to 10² (log‑uniform) and keep the best.
    """
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration on a log‑scale
        logc = rng.uniform(-4.0, 2.0)          # c ∈ [10⁻⁴, 10²]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.5f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(a0, inner_deadline, best_seq, best_score):
    """
    Simple projected sub‑gradient method.
    Uses step size η_t = η₀ / sqrt(t) and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5
    while time.time() < inner_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        step = eta0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = _project_simplex(np.maximum(a_new, 0.0))
        a = a_new
    return best_seq, best_score, a


def _multi_subgradient_stage(n, best_seq, best_score, deadline):
    """
    Run many short sub‑gradient runs from random starts and from the
    current best vector.  Each run gets a few seconds of CPU time.
    """
    # keep a reference to the globally best solution
    while time.time() < deadline:
        # pick a start: 50 % current best, 50 % fresh Dirichlet
        if np.random.rand() < 0.5:
            start = best_seq.copy()
        else:
            logc = np.random.uniform(-4.0, 2.0)
            c = 10 ** logc
            start = np.random.gamma(shape=c, scale=1.0, size=n)
            start /= start.sum()
        # give each run at most 6 seconds (adjustable)
        inner_deadline = min(deadline, time.time() + 6.0)
        best_seq, best_score, _ = _subgradient_descent(start, inner_deadline,
                                                       best_seq, best_score)
        # quick escape if we are about to hit the final deadline
        if time.time() + 10.0 >= deadline:
            break
    return best_seq, best_score

def _hill_climb_line_search(start_a, deadline, best_seq, best_score):
    """
    Move a small but tunable fraction of mass from the index with the
    largest positive sub‑gradient to the index with the most negative.
    A simple line‑search over a few fractions is performed.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    fail_counter = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src = np.argmax(grad)          # most “harmful’’ entry
        dst = np.argmin(grad)          # most “helpful’’ entry

        if a[src] <= 1e-14:
            # fallback: pick a random source that actually has mass
            nonzero = np.where(a > 1e-14)[0]
            if nonzero.size == 0:

            src = rng.choice(nonzero)
            dst = rng.integers(n)
            if src == dst:

        improved = False
        for f in fracs:

            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            cur_score = _objective_score(a_prop)
            if cur_score < best_score:
                a = a_prop
                best_a = a.copy()
                best_score = cur_score
                improved = True
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")
                break

        if not improved:
            fail_counter += 1
            if fail_counter > 2000:
                # occasional random jitter to escape flat regions

                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))

    return best_a, best_score
