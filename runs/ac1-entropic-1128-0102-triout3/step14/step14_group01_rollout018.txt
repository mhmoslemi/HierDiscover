"""
Propose a step‑function (non‑negative list) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is built around fast FFT‑based convolutions and works in three
phases:

1️⃣  Random Dirichlet draws (wide range of concentration parameters) give a
    diverse pool of feasible start points.
2️⃣  For the best few candidates we run a fast *sub‑gradient descent* on the
    true max‑convolution, followed by a few hundred Adam steps on a
    *smooth‑max* surrogate (soft‑max over the convolution entries) with a
    slowly rising temperature τ.  All updates are projected onto the
    probability simplex (sum = 1, a_i ≥ 0).
3️⃣  A final simulated‑annealing hill‑climb moves a tiny fraction of mass from
    the index with the largest positive sub‑gradient to the one with the most
    negative, accepting improving moves and occasionally accepting worsening
    ones.  This refines the best solution found.

The routine respects a 1000 s budget, prints progress, and returns the best
sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT of length L (full length)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_{k} (a*a)[k] at a.
    If several shifts attain the maximum their contributions are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, L_self: int) -> float:
    """2·n·max(self‑convolution) for a (assumed to sum to 1)."""
    n = a.shape[0]
    conv = _conv_self(a, L_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, L_self: int,
                         max_iter: int, best_score: float,
                         best_seq: np.ndarray,
                         start_time: float, deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    eta = 0.2                     # base step size
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:5d} score={cur_score:.6f}")
        grad = _subgradient_maxconv(a, conv)
        step = eta / np.sqrt(it)
        a = _project_simplex(a - step * grad)
        # periodic bookkeeping
        if it % 200 == 0 and time.time() - start_time > 1.0:
            # cheap progress indicator
            pass
    return best_seq, best_score


def _adam_smooth(a0: np.ndarray, L_self: int, L_w: int,
                 max_iter: int, best_score: float,
                 best_seq: np.ndarray,
                 start_time: float, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smooth‑max surrogate of the convolution."""
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04                     # learning rate
    tau = 0.5                     # initial temperature
    tau_factor = 1.01
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # self‑convolution
        conv = _conv_self(a, L_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:5d} score={cur_score:.6f}")
        # soft‑max over the convolution entry
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum != 0.0 else np.eye(1, len(conv), conv.argmax())[0]
        # gradient of smooth‑max: 2·conv(w, a_rev)
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev, L_w)          # length = 3·n‑2
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # slowly raise temperature – the surrogate becomes the true max
        if it % 500 == 0:
            tau = min(tau * tau_factor, 200.0)

    return best_seq, best_score


def _final_hill_climb(a0: np.ndarray, L_self: int,
                      start_time: float, deadline: float,
                      best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Directed hill‑climb with a simple annealing schedule."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = 2.0 * n * _conv_self(a, L_self).max()
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0.0:
        return best_seq, best_score

    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_self)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer up to 2 % of the source mass
        eps_frac = rng.random() * 0.019 + 1e-6
        eps = a[src] * eps_frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, L_self).max()
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

                sc = _objective_score(starter, L_self)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

            pool.append((sc, a.copy()))
        else:
            worst_idx = np.argmax([p[0] for p in pool])
            if sc < pool[worst_idx][0]:
                pool[worst_idx] = (sc, a.copy())

    # --------------------------------------------------------------
    # 2️⃣  Intensive refinement of the best pool members
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])          # best first
    for idx, (init_sc, a0) in enumerate(pool):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] Refining candidate {idx+1}/{len(pool)} (init score={init_sc:.6f})")
        # a) Sub‑gradient descent (fast, many iterations)
        best_seq, best_score = _subgradient_descent(
            a0, L_self,
            max_iter=3000,
            best_score=best_score,
            best_seq=best_seq,
            start_time=start_time,
            deadline=deadline,
        )
        # b) Adam on smooth‑max (fine‑tune)
        best_seq, best_score = _adam_smooth(
            best_seq, L_self, L_w,
            max_iter=4000,
            best_score=best_score,
            best_seq=best_seq,
            start_time=start_time,
            deadline=deadline,
        )

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _final_hill_climb(
            best_seq, L_self,
            start_time,
            deadline,
            best_seq,
            best_score,
        )

    # --------------------------------------------------------------
    # Normalise to sum = 1 and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
