"""
Improved search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of several time‑controlled phases:

1️⃣  **Extended parametric sweep** – dense grids for geometric, power‑law and
    full (α,β)‑Beta families (α,β ∈ [0.01, 30] with log‑spacing).  This finds
    good analytic shapes.

2️⃣  **Large‑scale Dirichlet sampling** – 5 000 draws with concentration
    parameters spanning 10⁻³ … 10³.  Many random points increase the chance
    of hitting a good region of the simplex.

3️⃣  **Multi‑restart smooth‑max Adam** – a differentiable surrogate for the
    max‑convolution (soft‑max with temperature τ).  Several short Adam runs
    start from the best candidates found so far; the temperature is
    increased gradually so the surrogate converges to the true objective.

4️⃣  **Simulated annealing** – a cheap global optimiser that repeatedly
    transfers a random mass chunk between two coordinates, accepting
    worsening moves with probability exp(−Δ/τ).  The temperature decays
    exponentially.

5️⃣  **Aggressive hill‑climb** – a deterministic refinement that moves a
    tiny amount of mass from the index with the largest sub‑gradient to the
    index with the smallest, keeping only improving steps.

All phases respect a 1000 s budget (with a small safety margin).  The final
sequence is projected onto the probability simplex, clipped to non‑negative,
and returned as a plain list of floats ready for ``evaluate_sequence``.
"""
import time
import numpy as np

# --------------------------------------------------------------
# FFT‑based convolution utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution (FFT) – used for the smooth‑max gradient."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – assumes sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):

def _sweep_beta(n, best_seq, best_score, deadline):
    """Full (α,β) sweep on a wide log‑spaced grid."""
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))

    # combine a log‑grid covering [0.01, 30] and a few linear points for safety
    alphas = np.unique(
        np.concatenate((
            np.logspace(-2, -0.5, 30),
            np.linspace(0.1, 5.0, 40),
            np.logspace(0.5, 1.5, 30)
        ))
    )
    betas = alphas

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * log_rev
            # avoid under/overflow
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):

            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}")
    return best_seq, best_score

def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):

    a = a0.copy()

    v = np.zeros_like(a)

        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # gradient via convolution with reversed a
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # raise temperature every few thousand iterations
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


def _multi_restart_adam(n, best_seq, best_score, deadline, max_restarts=4):
    """
    Perform several short Adam runs starting from the current best.
    Each run gets a slice of the remaining budget.
    """
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq, best_score

    # split the remaining time evenly among the restarts
    per_run = remaining / max_restarts
    for r in range(max_restarts):
        run_deadline = time.time() + per_run
        # a small random perturbation before each restart helps escaping minima
        rng = np.random.default_rng()
        perturb = rng.uniform(-1e-3, 1e-3, size=n)
        a0 = best_seq + perturb
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)
        best_seq, best_score = _adam_smooth_max_descent(a0, run_deadline, best_seq, best_score)

        if time.time() >= deadline:
            break
    return best_seq, best_score


# --------------------------------------------------------------
# 4️⃣  Simulated annealing (global) 
# --------------------------------------------------------------
def _simulated_annealing(start_a, deadline, best_seq, best_score):
    """
    Simple SA that transfers a random mass chunk between two coordinates.
    Temperature decays geometrically.
    """
    a = start_a.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    rng = np.random.default_rng()
    T = 0.05                # initial temperature
    decay = 0.99997         # per‑iteration decay
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # pick source (must have some mass) and destination
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to 1 % of source mass)
        frac = rng.random() * 0.01 + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        new_score = _objective_score(a_prop)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={iteration:,} score={cur_score:.6f}")

        T *= decay
        # keep temperature from vanishing completely
        if T < 1e-7:
            T = 1e-7

    return best_seq, best_score


# --------------------------------------------------------------
# 5️⃣  Aggressive hill‑climb (deterministic)
# --------------------------------------------------------------
def _hill_climb(start_a, deadline):
    """
    Greedy mass‑transfer hill‑climb using the true sub‑gradient.
    Stops when no improvement is seen for a long while.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # random perturbation if gradient is degenerate
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move a tiny fraction (up to 0.3 %) – larger than the previous 0.05 %
        frac = rng.random() * 0.003 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # stay feasible
        if a_prop[src] < -1e-15:
            continue

        new_score = _objective_score(a_prop)

        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 6000:
                # occasional random shake‑up
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.005 + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns a plain list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.30  # a tiny safety margin

    # ----------------------------------------------------------
    # Choose problem size
    # ----------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ----------------------------------------------------------
    # Optional: use a globally supplied starter if available
    # ----------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ----------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # ----------------------------------------------------------
    sweep_deadline = deadline - 400.0  # keep time for later phases
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, sweep_deadline)

    # ----------------------------------------------------------
    # 2️⃣  Massive Dirichlet sampling
    # ----------------------------------------------------------
    dirichlet_deadline = deadline - 250.0
    if time.time() < dirichlet_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, dirichlet_deadline)

    # ----------------------------------------------------------
    # 3️⃣  Multi‑restart Adam smooth‑max descent
    # ----------------------------------------------------------
    adam_deadline = deadline - 120.0
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting multi‑restart Adam")
        best_seq, best_score = _multi_restart_adam(
            n, best_seq, best_score, adam_deadline, max_restarts=5)
        print(f"[{time.time() - start_time:.2f}s] After Adam best_score={best_score:.6f}")

    # ----------------------------------------------------------
    # 4️⃣  Simulated annealing
    # ----------------------------------------------------------
    sa_deadline = deadline - 60.0
    if time.time() < sa_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, sa_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After SA best_score={best_score:.6f}")

    # ----------------------------------------------------------
    # 5️⃣  Final aggressive hill‑climb
    # ----------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ----------------------------------------------------------
    # Normalise and return a plain Python list
    # ----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
