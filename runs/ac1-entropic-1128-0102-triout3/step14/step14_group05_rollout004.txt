"""
step_search_v2.py – A more aggressive optimisation pipeline.

Main ideas
----------
* Work with a larger problem size (n = 8192) – the continuous limit of the
  objective is independent of n, but a finer discretisation lets us
  approximate the optimal density more accurately.
* A fast *beta‑family* sweep (a·x^p·(1‑x)^q) discovers good non‑uniform
  baselines that were missed by the earlier geometric sweep.
* An annealed smooth‑max surrogate (log‑sum‑exp) is minimised with L‑BFGS.
  The optimiser is given many more function evaluations (≫ 250 in the
  previous code) so it can converge much closer to a local optimum.
* After the smooth phase a large‑step random hill‑climb (up to 5 % of the
  source mass) refines the exact objective.  Occasionally a whole‑swap is
  tried – this helps escape shallow plateaus that tiny moves cannot leave.
* The whole routine respects the 1000 s wall‑clock budget, records the best
  sequence found and finally returns it as a plain ``list[float]``.
"""

import time
import numpy as np

# ---- optional SciPy ---------------------------------------------------------
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a must be non‑negative; the routine normalises the sum internally.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:                         # should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Smooth‑max surrogate used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max (log‑sum‑exp) surrogate of
    max(conv) together with its gradient w.r.t. the log‑probability
    vector `z`.
    """

    def __init__(self, n: int, beta: float, L_conv: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv

    def __call__(self, z: np.ndarray):
        """
        Returns (value, gradient) where `value` is the smooth surrogate
        and `gradient` = ∂value/∂z (same shape as `z`).
        """
        a = _softmax(z)                     # a ≥ 0, Σa = 1
        # ----- self‑convolution -------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)     # length 2n‑1
        max_c = conv.max()
        # ----- smooth‑max (log‑sum‑exp) -----------------------------------------
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # ----- gradient w.r.t. a -------------------------------------------------
        # w_i = exp(beta*(conv_i-max_c)) / Σ
        w = exp_vals / exp_vals.sum()
        # convolution w * a_rev  (a_rev = a[::-1])
        a_rev = a[::-1]
        # use FFT for speed
        L_corr = _next_pow2(3 * self.n - 2)
        prod = np.fft.rfft(w, n=L_corr) * np.fft.rfft(a_rev, n=L_corr)
        conv_wa = np.fft.irfft(prod, n=L_corr)[: 3 * self.n - 2]
        # gradient w.r.t. a_i = 2 * (conv_wa[n‑1‑i])
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]   # length n

        # ----- back‑prop through the soft‑max ------------------------------------
        dot = np.dot(grad_a, a)                     # scalar
        grad_z = a * (grad_a - dot)                 # shape (n,)
        return surrogate, grad_z


# ----------------------------------------------------------------------
# Aggressive hill‑climb on the exact objective
# ----------------------------------------------------------------------


def _final_hill_climb_aggressive(a: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 L_fft: int):
    """
    Random mass transfers (up to 5 % of the source mass) accepted only if they
    improve the exact objective.  Occasionally a full swap of two entries is
    attempted.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- tiny mass move -------------------------------------------------
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move a random fraction (up to 5 %) of the source mass
        eps = best_a[src] * rng.random() * 0.05
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

        # ----- occasional full‑swap -----------------------------------------
        if rng.random() < 0.01:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                cand2 = best_a.copy()
                cand2[i], cand2[j] = cand2[j], cand2[i]
                sc2 = _eval_fast(cand2, L_fft)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = cand2
                    print(f"[{time.time() - start_time:.2f}s] swap‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine consists of
        1. a uniform baseline,
        2. a dense sweep over Beta‑family shapes,
        3. an annealed L‑BFGS optimisation of a smooth‑max surrogate,
        4. a short random‑restart phase,
        5. an aggressive hill‑climb on the exact objective.

    The best sequence (unit‑summed) is returned as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 0.2 s safety margin
    deadline = start_time + budget_s - 0.20

    # problem size – 8192 works well on a single CPU core and gives a finer grid
    n = 8192
    L_fft = _next_pow2(2 * n - 1)      # for exact evaluations
    L_sur = _next_pow2(2 * n - 1)      # for the surrogate (same length)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – up‑sample if needed
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size:
            # up‑sample / down‑sample to length n by linear interpolation
            if starter.size != n:
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep – Beta‑family (i+1)^p·(n‑i)^q
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    p_grid = np.linspace(-0.9, 2.0, 31)   # exponent on the left side
    q_grid = np.linspace(-0.9, 2.0, 31)   # exponent on the right side
    lam_grid = (0.0, 0.25, 0.5, 0.75, 1.0)

    for p in p_grid:
        if time.time() >= deadline - 150:
            break
        for q in q_grid:
            if time.time() >= deadline - 150:
                break
            # shape[i] = (i+1)^p * (n-i)^q   – symmetric for p==q
            shape = np.power(idx + 1.0, p) * np.power(n - idx, q)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] β‑sweep p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (annealed β)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 250:
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # temperature schedule – start very smooth, finish very sharp
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         50.0, 100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 250:
                break
            obj = _SurrogateMax(n=n, beta=beta, L_conv=L_sur)

            # allocate a generous budget of function evaluations
            maxfun = 2500
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # warm‑start the next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Random restarts – short L‑BFGS runs from noisy starts
    # ------------------------------------------------------------------
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline - 200:
            break
        rng = np.random.default_rng(seed + r + 12345)
        noisy = best_seq + rng.normal(scale=0.01, size=n)
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 5.0
        obj = _SurrogateMax(n=n, beta=beta, L_conv=L_sur)
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=500,
                disp=False,
                factr=1e7,
            )
        except Exception:  # pragma: no cover
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final aggressive hill‑climb on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb_aggressive(best_seq,
                                                   deadline,
                                                   start_time,
                                                   L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process – enforce non‑negativity & unit sum, return list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
