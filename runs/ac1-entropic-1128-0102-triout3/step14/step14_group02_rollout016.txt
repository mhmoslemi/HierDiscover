"""
Propose a non‑negative step‑function (probability vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in four time‑bounded stages:

1️⃣  Broad parametric sweeps (geometric, power, symmetric / asymmetric Beta and
    a new “edge‑dropoff’’ family).  All candidates are normalised to sum=1
    and scored with the cheap objective 2·n·max(conv).

2️⃣  Random initialisers – Dirichlet draws with a very wide range of
    concentration and sparse “m‑point’’ vectors (uniform weight on a random
    subset of indices).  A few hundred trials are performed.

3️⃣  Projected sub‑gradient descent on the exact max‑convolution objective.
    The sub‑gradient is computed analytically; a diminishing step‑size and
    Euclidean projection onto the simplex are used.  Several thousand
    iterations are performed, improving the best candidate from the previous
    phases.

4️⃣  A final directed hill‑climb that moves a tiny amount of mass from the
    index with the largest positive sub‑gradient to the one with the most
    negative sub‑gradient, accepting only improving moves.  This refines the
    solution to the limit of the allotted time.

All linear convolutions are performed with FFT for speed, and the simplex
projection uses the O(n) algorithm of Duchi et al.  The routine respects the
1000 s budget; intermediate best scores are printed for monitoring.

The returned list of floats is ready for ``evaluate_sequence``.
"""
import time
import numpy as np

# --------------------------------------------------------------
# Linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a (a * a)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    return np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w

# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    return float(2.0 * n * _conv_self(seq).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (average over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# 1️⃣  Parametric families (geometric, power, beta, drop‑off)
# --------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 101):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 61)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * np.log(idx + 1.0)
        log_rev = (a - 1.0) * np.log(np.maximum(n - idx, 1))
        log_seq = log_a + log_rev
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    betas = alphas
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_dropoff(n, best_seq, best_score, deadline):
    """
    Family a_i ∝ (i+1)^{p} * (n-i)^{q}.
    By choosing negative exponents we obtain large weights near the ends
    and small weights in the middle.
    """
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-12
    p_vals = np.linspace(-3.0, 1.0, 13)   # -3 … 1
    q_vals = np.linspace(-3.0, 1.0, 13)
    for p in p_vals:
        if time.time() >= deadline:
            break
        log_i = p * np.log(idx + 1.0 + eps)
        for q in q_vals:
            if time.time() >= deadline:
                break
            log_seq = log_i + q * np.log(n - idx + eps)
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Dropoff p={p:.2f} q={q:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 200
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-4.0, 1.0)   # widen range: c from 1e‑4 to 10
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.5f} score={sc:.6f}")
    return best_seq, best_score


def _random_sparse_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 150
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # pick support size on a log‑scale between 5 and n//2
        logm = rng.uniform(np.log(5), np.log(max(10, n // 2)))
        m = int(np.exp(logm))
        if m < 2:
            m = 2
        idx = rng.choice(n, size=m, replace=False)
        seq = np.zeros(n)
        seq[idx] = 1.0
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse m={m} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣  Projected sub‑gradient descent on the exact max‑conv objective
# --------------------------------------------------------------
def _subgradient_descent(start_a, deadline, best_seq, best_score):
    a = start_a.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.2                     # initial step size (tuned empirically)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)       # direction that increases max
        # move opposite to the gradient
        step = step0 / np.sqrt(it)                # classic diminishing step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

    return best_seq, best_score

# 4️⃣  Directed hill‑climb (tiny mass transfers)
# --------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_cand = np.where(a > 1e-14)[0]
        if src_cand.size == 0:
            break

        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        frac = rng.random() * 0.008 + 1e-6          # up to 0.8 % of source mass
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill climb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # occasional random nudge to escape plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * 0.008 + 1e-6)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm runs four stages (parametric sweep, random initialisers,

    allotted 1000 s budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:

            s = starter.sum()
            if s > 0:

                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (leave ~300 s for later stages)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(
            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_dropoff(
            n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet / sparse initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:

            n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_sparse_trials(
            n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 80.0
    if time.time() < subgrad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(
            best_seq, subgrad_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
