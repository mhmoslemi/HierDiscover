def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    beta1,beta2 = 0.9,0.999
    eps_adam=1e-8
    step0=0.02
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    tau = 0.5
    tau_max = 200.0
    tau_factor = 1.15
    it = 0
    while time.time() < deadline:
        it+=1
        conv = _conv_self(a)
        cur_score = float(2.0*n*conv.max())
        if cur_score < best_score:
            best_seq,best_score = a.copy(),cur_score
            print(f"[{time.time()-start_time:.2f}s] Adam iter {it} score {cur_score:.6f} tau {tau:.3f}")
        # smooth weights
        max_c = conv.max()
        w = np.exp(tau*(conv - max_c))
        w_sum = w.sum()
        if w_sum ==0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)]=1.0
            w_sum=1.0
        w/=w_sum
        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)  # length = 3n-2
        grad = 2.0 * grad_full[n-1:n-1+n]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = step0 * m_hat/(np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new
        # increase tau
        if it % 2000 == 0:
            tau = min(tau * tau_factor, tau_max)
    return best_seq, best_score

def _gradient_descent_maxconv(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it+=1
        conv = _conv_self(a)
        cur_score = float(2.0*n*conv.max())
        if cur_score < best_score:
            best_seq,best_score = a.copy(),cur_score
            print(f"[{time.time()-start_time:.2f}s] Subgrad descent iter {it} new best {cur_score:.6f}")
        # compute subgradient of max
        grad = _subgradient_maxconv(a, conv)
        if np.all(grad == 0):
            break
        # descent direction is -grad
        direction = -grad
        # compute feasible step size
        # for indices where direction < 0 (i.e., decreasing a), max step = a_i / (-direction_i)
        # for direction > 0, no upper bound (increase unlimited but limited by sum after projection)
        max_step = np.inf
        dec_indices = np.where(direction < -1e-12)[0]
        if dec_indices.size > 0:
            max_step = min(max_step, np.min(a[dec_indices] / (-direction[dec_indices])))
        # try step fractions
        step_factor = 0.5
        step = max_step * step_factor if max_step != np.inf else step_factor
        # propagate a step
        a_candidate = a + step * direction
        # ensure nonneg
        a_candidate = np.maximum(a_candidate, 0.0)
        # project to simplex (to maintain sum)
        a_candidate = _project_simplex(a_candidate)
        # compute objective
        conv_c = _conv_self(a_candidate)
        cand_score = float(2.0*n*conv_c.max())
        if cand_score < cur_score:
            a = a_candidate
        else:
            # line search reduce step
            improved = False
            for factor in [0.25, 0.125, 0.0625]:
                step = max_step * factor if max_step != np.inf else factor

def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    cur_score = float(2.0*n*_conv_self(a).max())
    t_start = time.time()
    total_time = deadline - t_start
    # schedule temperature linearly decreasing from T_start to T_end
    T_start = 1.0
    T_end = 0.001
    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = T_start * (1 - elapsed/total_time) + T_end * (elapsed/total_time)
        # propose random transfer
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = a[i] * (np.random.rand() * 0.02 + 1e-4)  # up to 2% of source
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        # ensure nonnegative, project onto simplex
        a_prop = _project_simplex(np.maximum(a_prop,0.0))
        score_prop = float(2.0*n*_conv_self(a_prop).max())
        if score_prop < cur_score:
            a = a_prop
            cur_score = score_prop
            if score_prop < best_score:
                best_seq, best_score = a.copy(), score_prop
                print(f"[{time.time()-start_time:.2f}s] Anneal improved to {score_prop:.6f}")
        else:
            # accept with probability exp(-(score_prop - cur_score)/T)
            if np.random.rand() < np.exp(-(score_prop - cur_score) / max(T, 1e-12)):
                a = a_prop
                cur_score = score_prop
    return best_seq, best_score

"""
Improved search for a non‑negative step function (probability vector) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)^2)

The algorithm proceeds in five time‑bounded stages:

1.  Very broad parametric sweeps – geometric, power and a *full* Beta
    family (α,β drawn from a dense log‑grid, allowing extreme U‑shapes).
2.  Random Dirichlet draws – concentration parameter spans 10⁻⁴ … 5,
    giving both very spiky and very flat candidates.
3.  Smooth‑max continuation with Adam – the smoothness (τ) starts tiny
    (≈0.5) and is increased gradually, which gives a smooth landscape for
    the optimiser and helps to escape local minima.
4.  Projected sub‑gradient descent on the true max‑convolution (no
    smoothing).  The sub‑gradient of the max‑entry is computed analytically,
    a step is taken and the result is projected back onto the probability
    simplex.
5.  Simulated‑annealing fine‑tuning – random mass transfers are accepted
    with a temperature schedule, allowing occasional uphill moves and a
    thorough exploration of the remaining time budget.

All stages respect the global 1000 s deadline, keep the vector on the
simplex (sum = 1, non‑negative) and constantly update the best‑so‑far
solution, printing progress to the console.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (v_i ≥ 0, Σ v_i = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming Σ seq = 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# --------------------------------------------------------------
# 1️⃣  Parametric families
# --------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(
                    f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}"
                )
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta_extended(n, best_seq, best_score, deadline):
    """Full (α,β) sweep on a log‑grid, allowing extreme U‑shapes."""
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    # α,β from 0.01 to 10 (log‑spaced)
    alphas = np.logspace(np.log10(0.01), np.log10(10.0), 25)
    betas = alphas

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * log_rev
            # stabilise the exponentials
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(
                    f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}"
                )
    return best_seq, best_score


# --------------------------------------------------------------
# 2️⃣  Random Dirichlet draws
# --------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 400  # more draws than the previous version
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-4.0, 0.7)      # concentration ∈ [10⁻⁴, ~5]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# 3️⃣  Smooth‑max continuation (Adam)
# --------------------------------------------------------------
def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):
    """Adam on a smooth approximation of max(conv).  τ starts tiny and grows."""
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.02

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 0.5               # start with a very smooth max
    tau_max = 200.0
    tau_factor = 1.15

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_seq, best_score = a.copy(), cur_score
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2f} score={cur_score:.6f}"
            )

        # smooth‑max weights
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smooth max (2 * conv(w, a_rev))
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)           # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature occasionally
        if it % 2000 == 0:
            tau = min(tau * tau_factor, tau_max)

    return best_seq, best_score


# --------------------------------------------------------------
# 4️⃣  Projected sub‑gradient descent on the *true* max‑conv
# --------------------------------------------------------------
def _proj_subgradient_descent(a0, deadline, best_seq, best_score):
    """Take steps opposite to the sub‑gradient of max(conv) and project."""
    a = a0.copy()
    n = a.shape[0]
    step = 0.5            # initial step size
    min_step = 1e-6

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        grad = _subgradient_maxconv(a, conv)      # gradient of the max‑entry
        if np.allclose(grad, 0):
            break

        improved = False
        # try a few geometrically decreasing step sizes
        for attempt in range(5):
            step_try = step * (0.5 ** attempt)
            if step_try < min_step:
                break
            a_candidate = a - step_try * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            cand_score = float(2.0 * n * _conv_self(a_candidate).max())
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                step = step_try * 1.2                 # gently increase for next iteration
                if cand_score < best_score:
                    best_seq, best_score = a.copy(), cand_score
                    print(
                        f"[{time.time() - start_time:.2f}s] Sub‑grad descent improved to {cand_score:.6f}"
                    )
                break

        if not improved:
            step *= 0.5
            if step < min_step:
                # occasional random shake to get out of a plateau
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                if a[i] > 1e-12:
                    delta = a[i] * (np.random.rand() * 0.01 + 1e-4)
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(np.maximum(a, 0.0))
                step = 0.5

    return best_seq, best_score


# --------------------------------------------------------------
# 5️⃣  Simulated‑annealing fine‑tuning
# --------------------------------------------------------------
def _simulated_annealing(a0, deadline, best_seq, best_score):
    """Random mass transfers, Metropolis acceptance with a linear cooling schedule."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = float(2.0 * n * _conv_self(a).max())

    t_start = time.time()
    total = deadline - t_start
    T_start = 1.0
    T_end = 1e-3

    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = T_start * (1 - elapsed / total) + T_end * (elapsed / total)

        # pick a source with non‑negligible mass
        src = np.random.randint(n)
        if a[src] <= 1e-12:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # move a modest fraction of the source mass
        delta = a[src] * (np.random.rand() * 0.02 + 1e-4)
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cand_score = float(2.0 * n * _conv_self(a_prop).max())
        if cand_score < cur_score:
            a, cur_score = a_prop, cand_score
            if cand_score < best_score:
                best_seq, best_score = a.copy(), cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] Annealing improved to {cand_score:.6f}"
                )
        else:
            # uphill move with Metropolis probability
            if np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_prop, cand_score

    return best_seq, best_score

# Main entry point required by the outer harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects a ~1000 s budget and returns a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (keep the same as the reference run)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Expanded parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 500.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(
            n, best_seq, best_score, phase_deadline
        )
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(
            n, best_seq, best_score, phase_deadline
        )
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta_extended(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more draws)
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max continuation with Adam (τ starts tiny)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max_descent(
            best_seq, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent on the exact max‑conv
    # --------------------------------------------------------------
    phase_deadline = deadline - 100.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected sub‑gradient descent")
        best_seq, best_score = _proj_subgradient_descent(
            best_seq, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After sub‑gradient descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
