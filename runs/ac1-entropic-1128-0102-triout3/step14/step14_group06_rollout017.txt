"""
Step‑function optimiser – evolutionary search + refinement.

The algorithm works in three stages:

1. **Coarse population search (small n).**
   A diverse initial population (uniform, beta‑densities,
   power‑ramps, exponentials, random Dirichlet vectors) is created.
   A simple (μ+λ) evolutionary loop with tournament selection,
   crossover (averaging) and Gaussian mutation explores the space.
   Candidates are always projected onto the probability simplex
   (non‑negative, sum = 1).  The best few individuals are kept.

2. **Upscaling.**
   The current best low‑dimensional vector is interpolated to a
   larger size (up to 65536 items) – the objective does not depend
   on the absolute scale, only on the shape, so the up‑sampled
   vector is a valid starting point for a fine‑grained optimisation.

3. **Fine refinement.**
   The up‑scaled candidate is fed to the same two deterministic
   refinements used in the reference code:
   – exponentiated sub‑gradient descent on the exact ℓ∞
     objective (active‑set),
   – smooth‑max Adam descent (soft‑max surrogate).

The whole procedure adapts its problem size to the remaining time
budget (1000 s) and prints progress information.  The returned list
contains non‑negative floats; `evaluate_sequence` will compute the
final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution / objective utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # guard against division by zero
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _normalise(v: np.ndarray) -> np.ndarray:
    """Make v non‑negative and sum to 1 (probability vector)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.shape[0])
    return v


# ----------------------------------------------------------------------
# Analytic shape generators (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with optional plateau.
    p  – exponent of the ramp,
    alpha – fraction of the total length kept as a flat plateau (0…0.5).
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    shape = _normalise(shape)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape = _normalise(shape)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape = _normalise(shape)
    return shape


def _make_exponential(n: int, r: float) -> np.ndarray:
    """Decreasing exponential a_i ∝ r**i, r∈(0,1)."""
    x = np.arange(n, dtype=float)
    shape = r ** x
    return _normalise(shape)


def _make_sine(n: int) -> np.ndarray:
    """Positive half‑sine on [0,1] (non‑negative)."""
    x = np.arange(n, dtype=float) / n
    shape = np.sin(np.pi * x)
    return _normalise(shape)

# Deterministic refinements (identical to the reference implementation)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on the smooth‑max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05
    T0 = 0.02
    T_end = 1e-4
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # temperature schedule (linear freeze)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w * a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best_a, best_score


def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on the exact max‑convolution objective.
    The active set defines a sub‑gradient; multiplicative updates keep the vector
    non‑negative, a final projection restores the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # indices attaining the current maximum (with tiny tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * correlation( w , a_rev )
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule
        eta0 = 0.8
        eta = eta0 / (1.0 + it / 2000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score

# Evolutionary search on a modest dimension
# ----------------------------------------------------------------------
def _initial_population(n: int, rng: np.random.Generator) -> list[tuple[float, np.ndarray]]:
    """Create a diverse initial pool (score, vector)."""
    pool = []

    def _add(vec: np.ndarray) -> None:
        vec = _normalise(vec)
        pool.append((_objective(vec), vec))

    _add(np.full(n, 1.0 / n))  # uniform

    # symmetric Beta shapes
    for a in np.linspace(0.6, 2.0, 7):
        _add(_make_beta_shape(n, a))

    # asymmetric Beta shapes
    for a in [0.5, 1.0, 1.5]:
        for b in [0.5, 1.0, 1.5]:
            _add(_make_beta_shape_asym(n, a, b))

    # power‑ramps
    for p in [0.5, 1.0, 2.0]:
        _add(_make_shape(n, p, 0.0))

    # exponential ramps
    for r in [0.9995, 0.997, 0.99]:
        _add(_make_exponential(n, r))

    # half‑sine
    _add(_make_sine(n))

    # random Dirichlet vectors (sparse)
    for _ in range(30):
        a = rng.gamma(0.5, 1.0, size=n)
        _add(a)

    # use the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # simple linear interpolation to match size
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                starter = np.interp(x_new, x_old, starter)
            starter = _normalise(starter)
            _add(starter)
    except Exception:
        pass

    pool.sort(key=lambda x: x[0])
    return pool[:20]                # keep the best 20


def _evolution(pop: list[tuple[float, np.ndarray]],
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple (μ+λ) evolutionary loop.
    Returns the best vector found and its score.
    """
    rng = np.random.default_rng()
    best_vec, best_score = pop[0][1], pop[0][0]

    n = best_vec.shape[0]
    sigma_initial = 0.02
    it = 0
    while time.time() < deadline:
        it += 1

        # tournament selection for two parents
        def _select() -> np.ndarray:
            cand = rng.choice(len(pop), size=3, replace=False)
            best = min(cand, key=lambda i: pop[i][0])
            return pop[best][1]

        p1, p2 = _select(), _select()
        child = (p1 + p2) * 0.5

        # annealing Gaussian mutation
        sigma = sigma_initial * (1.0 - (time.time() - start_time) / (deadline - start_time))
        child += rng.normal(0.0, sigma, size=n)
        child = _normalise(child)

        score = _objective(child)

        # keep the best POP_SIZE individuals
        if score < pop[-1][0]:
            pop[-1] = (score, child)
            pop.sort(key=lambda x: x[0])
            if score < best_score:
                best_score, best_vec = score, child
                print(f"[{time.time() - start_time:.2f}s] Evolution iter={it:,} new best={score:.6f}")

    return best_vec, best_score

# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # small safety margin

    # ------------------------------------------------------------------
    # Choose an intermediate problem size for the evolutionary phase.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N_small = 16384
    elif remaining > 400:
        N_small = 8192
    else:
        N_small = 4096

    rng = np.random.default_rng(seed + 9876)

    # ------------------------------------------------------------------
    # 1) Initialise a diverse population
    # ------------------------------------------------------------------
    population = _initial_population(N_small, rng)

    best_vec, best_score = population[0][1], population[0][0]
    print(f"[{time.time() - start:.2f}s] Initial best score (n={N_small}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Evolutionary optimisation on the small problem
    # ------------------------------------------------------------------
    evo_deadline = start + (deadline - start) * 0.40   # ~40 % of total budget
    best_vec, best_score = _evolution(population, evo_deadline, start)
    print(f"[{time.time() - start:.2f}s] After evolution best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upscale to a larger size (as large as remaining time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        if remaining > 250:
            N_big = 65536
        elif remaining > 120:
            N_big = 32768
        else:
            N_big = 16384

        # linear interpolation to the larger grid
        x_big = np.linspace(0, 1, N_big, endpoint=False)
        x_small = np.linspace(0, 1, best_vec.shape[0], endpoint=False)
        best_big = np.interp(x_big, x_small, best_vec)
        best_big = _normalise(best_big)
        best_score_big = _objective(best_big)
        print(f"[{time.time() - start:.2f}s] Upscaled to n={N_big} score={best_score_big:.6f}")

        # ------------------------------------------------------------------
        # 4) Deterministic refinement on the large problem
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 20:
            sub_end = time.time() + remaining * 0.70
            # a short run of the exponentiated sub‑gradient method
            best_big, refined_score = _exp_subgradient_descent(best_big, sub_end, start)
            if refined_score < best_score_big:
                best_score_big = refined_score
                print(f"[{time.time() - start:.2f}s] ExpGD refined best={refined_score:.6f}")

            # final Adam smoothing if time permits
            remaining = deadline - time.time()
            if remaining > 5:
                best_big, refined_score = _smooth_gradient_descent(best_big,
                                                                   deadline,
                                                                   start)
                if refined_score < best_score_big:
                    best_score_big = refined_score
                    print(f"[{time.time() - start:.2f}s] SmoothGD refined best={refined_score:.6f}")

        final_vec = best_big
    else:
        final_vec = best_vec

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    final_vec = _normalise(final_vec)
    final_score = _objective(final_vec)

    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in final_vec]
