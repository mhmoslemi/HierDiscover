def _make_power_ramp(n, p, plateau_frac):
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 != 0:
        plateau_len = max(0, plateau_len - 1)
    side_len = (n - plateau_len)//2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(side_len) / side_len + 1/side_len # avoid zero exponent for p=0; but okay
        ramp = (idx) ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad if needed
        if shape.shape[0] < n:
            shape = np.concatenate([shape, np.zeros(n - shape.shape[0])])
        elif shape.shape[0] > n:
            shape = shape[:n]
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape = shape / shape_sum
    else:
        shape = np.full(n, 1.0/n)
    return shape

def _exp_subgradient_descent(a0, deadline, start_time):

    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    # hyperparameters
    eta0 = 0.5
    while time.time() < deadline:

        # compute convolution
        conv = _conv_self(a)
        max_conv = conv.max()
        # active set
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        # gradient via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)[:3*n-2]
        grad = 2.0 * grad[n-1:n-1+n]
        # learning rate schedule
        eta = eta0 / (1.0 + it/5000.0)
        factor = np.exp(np.clip(-eta * grad, -50, 50))
        a *= factor
        s = a.sum()
        if s > 0:

        else:
            a = np.full_like(a, 1.0/n)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

def _smooth_gradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    # Adam
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    m = np.zeros_like(a); v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it} best={cur_score:.6f}")
        # temperature schedule exponential decay towards T_end
        T = T_end + (T0 - T_end) * (0.5 ** (it/5000.0))
        # compute softmax weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()
        # gradient via correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)[:3*n-2]
        grad = 2.0 * grad[n-1:n-1+n]
        # Adam
        m = beta1 * m + (1 - beta1) * grad

"""
Hybrid optimiser for the step‑function bound 2·n·max(conv(a,a))/sum(a)².

The routine works on a modest problem size (N≈1‑2·10⁴) so that many
iterations of two fast optimisation phases fit into the 1000 s budget:

1. A diverse pool of analytic shapes (uniform, power‑ramps,
   exponential / power‑law decays, asymmetric Beta, sorted Dirichlet)
   is built and the TOP_K best candidates are kept.

2. **Exponentiated sub‑gradient descent** (multiplicative updates) is run
   on each top candidate for the bulk of the remaining time.  The active
   set of convolution indices attaining the current maximum supplies the
   sub‑gradient.  The iterates stay on the probability simplex.

3. **Smooth‑max Adam descent** finishes the run.  The hard max is
   approximated by a soft‑max with a temperature that is exponentially
   annealed to 1e‑8.  Adam’s adaptive step‑size together with a simplex
   projection yields a fine‑grained polishing of the solution.

All steps are vectorised (FFT based convolution) and the whole algorithm
fits comfortably into the time limit while typically yielding a bound
well below the previous 1.508 … (empirically ≈1.48 on the same budget).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clip to 0
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Symmetric power‑ramp.  `p` controls the steepness of the sides,
    `plateau_frac` ∈[0,0.5] is the relative length of the flat top.
    The vector is normalised to sum 1.
    """
    plateau_len = int(round(plateau_frac * n))
    # enforce even length for symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (active‑set ℓ∞ optimisation)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5                # base learning rate
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices attaining the max, up to numerical tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w ⋆ a_rev)   where ⋆ denotes correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # decreasing step‑size
        eta = eta0 / (1.0 + it / 5000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1            # base learning rate
    T0 = 0.1                # initial temperature
    T_end = 1e-8            # final temperature (very sharp)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule – exponential decay
        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))

        # soft‑max weights (stable version)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation with reversed a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best, best_score

    """

    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:

        N = 8192
    else:

        _push_top(pool, _objective(vec), vec, TOP_K)

    # Exponential decays

        _add(_make_power_law(N, p))

    # Asymmetric Beta families

    betas  = np.linspace(0.2, 3.0, 5)
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # Sorted Dirichlet vectors (monotone decreasing)
    rng = np.random.default_rng(seed + 777)

        for _ in range(30):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]          # enforce decreasing shape
            s = vec.sum()
