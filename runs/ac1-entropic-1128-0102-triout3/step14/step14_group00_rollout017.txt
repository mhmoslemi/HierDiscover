"""
Optimising non‑negative step functions for the L∞ self‑convolution bound.

The pipeline consists of:

1.  A cheap high‑level search over analytic β‑distribution shapes on a
    modest size (n_small = 2¹² = 4096).  The best shape is kept.

2.  An LP‑based “gap‑filling’’ step (only if SciPy is available).
    For the current sequence `f` we compute its worst convolution value
    `M = max(f*f)`.  By solving a linear programme that maximises the
    L₁‑mass of a helper vector `g` under the constraints
    `f * g ≤ M` (only the active constraints are kept), we obtain a
    vector that spreads the mass where it hurts the least.
    A short line‑search mixes `f` and the scaled `g`, strictly decreasing
    the objective.  This loop is repeated a few times.

3.  The resulting shape is up‑sampled to a larger resolution
    (n_big = 2¹⁵ = 32768) by linear interpolation and re‑normalised.

4.  A long exponentiated‑gradient descent on the simplex (multiplicative
    updates) refines the sequence using the exact self‑convolution
    gradient (FFT‑based).  The step‑size decays as η = η₀/√t.

5.  Finally a light simulated‑annealing polish moves tiny packets of
    mass between positions and accepts worsening moves with a cooling
    schedule.

All phases respect the global time‑budget (default 1000 s) and the
returned list is a non‑negative probability vector (∑aᵢ = 1).  The
evaluation function `evaluate_sequence` will treat this vector as a
valid step‑function and compute the required bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities for linear convolution (real FFT, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (both real) using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑free) objective used during the search.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  No safety checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Analytic families – β‑distribution on a discrete grid
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta pdf on {0,…,n‑1} and normalised to sum = 1.
    The continuous pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Multiplicative (EG) descent on the probability simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # use only the true maximiser (single index) – faster and still
        # provides a descent direction
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = a[k - idx]
        # multiplicative update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # enforce floor then renormalise
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling from 0.02 to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement (only if SciPy is present)
# ----------------------------------------------------------------------
def _lp_improve(seq: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑based mixing described in the
    reference.  Each iteration solves a small LP (active constraints only),
    rescales the solution to keep the L₁‑mass fixed and mixes it into `seq`.
    Returns the improved sequence and its new objective value.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        # SciPy not available – skip this phase
        return seq, _objective(seq)

    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _objective(best_seq)

    # number of LP rounds – stop early if no improvement
    for lp_round in range(15):
        if time.time() >= deadline:
            break

        conv = _conv_self(best_seq)
        M = conv.max()
        # active set (indices whose value is within 1e‑6 of the maximum)
        tol = M * 1e-6
        ks = np.where(conv >= M - tol)[0]
        if ks.size == 0:
            break
        # keep at most 200 active indices (most critical)
        if ks.size > 200:
            ks = ks[np.argsort(-(conv[ks]))[:200]]
        rows, cols, data = [], [], []
        for row_id, k in enumerate(ks):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            col_range = np.arange(lo, hi + 1, dtype=int)
            coeffs = best_seq[k - col_range]   # f_{k‑j}
            rows.extend([row_id] * len(col_range))
            cols.extend(col_range.tolist())
            data.extend(coeffs.tolist())
        A = csr_matrix((data, (rows, cols)), shape=(len(ks), n))
        b = np.full(len(ks), M)
        c = -np.ones(n)                       # maximise sum(g)
        bounds = [(0, None)] * n

        # solve the LP (HiGHS is the default modern solver)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP round {lp_round} failed")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break
        # rescale to keep the same L₁‑mass as the current sequence
        S = best_seq.sum()
        g = g0 * (S / sum_g0)

        # pre‑compute the three convolutions needed for the line‑search
        conv_ff = conv                              # f*f
        conv_fg = _conv_cross(best_seq, g)          # f*g
        conv_gg = _conv_self(g)                     # g*g

        best_t, best_M = 0.0, M
        # coarse scan of mixing coefficients
        for t in np.linspace(0.0, 1.0, 21):
            coeff = 1.0 - t
            mixed = (coeff * coeff) * conv_ff + 2.0 * t * coeff * conv_fg + (t * t) * conv_gg
            m = mixed.max()
            if m < best_M:
                best_M, best_t = m, t

        if best_t == 0.0:          # no improvement possible
            break

        # mix and renormalise (renorm is actually unnecessary because both have the same sum)
        best_seq = (1.0 - best_t) * best_seq + best_t * g
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:.2f}s] LP round={lp_round}  score={best_score:.6f}  t={best_t:.3f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine builds a modest pool of analytic β‑shapes,
    refines the best one with a linear‑programming mixing phase,
    upsamples to a higher resolution and finishes with exponentiated‑gradient
    descent and a short annealing polish.  The returned list contains
    non‑negative floats (∑aᵢ = 1) and respects the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) problem size decisions
    # ------------------------------------------------------------------
    n_small = 4096            # 2¹¹ – used for the cheap LP phase
    n_big   = 32768           # 2¹⁵ – final resolution

    # ------------------------------------------------------------------
    # 1) Build a small pool of analytic candidates on the small grid
    # ------------------------------------------------------------------
    pool = []                     # list of (score, vector)
    # uniform baseline
    uniform = np.full(n_small, 1.0 / n_small)
    pool.append((_objective(uniform), uniform.copy()))

    # random beta shapes
    for _ in range(250):
        a = 10 ** rng.uniform(-2.5, 0.8)   # roughly 0.003 … 6.3
        b = 10 ** rng.uniform(-2.5, 0.8)
        v = _make_beta_shape(n_small, a, b)
        pool.append((_objective(v), v.copy()))

    # convex combinations of two beta shapes
    for _ in range(100):
        a1 = 10 ** rng.uniform(-2.5, 0.8)
        b1 = 10 ** rng.uniform(-2.5, 0.8)
        a2 = 10 ** rng.uniform(-2.5, 0.8)
        b2 = 10 ** rng.uniform(-2.5, 0.8)
        w = rng.random()
        v = w * _make_beta_shape(n_small, a1, b1) + (1.0 - w) * _make_beta_shape(n_small, a2, b2)
        v /= v.sum()
        pool.append((_objective(v), v.copy()))

    # optional global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # resample to n_small if needed
            if starter.size != n_small:
                xs_old = np.arange(starter.size) / starter.size
                xs_new = np.arange(n_small) / n_small
                starter = np.interp(xs_new, xs_old, starter)
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            pool.append((_objective(starter), starter.copy()))
    except Exception:
        pass

    # keep only the best few candidates
    pool.sort(key=lambda x: x[0])
    pool = pool[:5]

    best_seq, best_score = pool[0][1], pool[0][0]
    print(f"[{time.time() - start_time:.2f}s] Initial small‑grid best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based mixing on the small grid (if SciPy is present)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _lp_improve(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After LP phase score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to the large grid
    # ------------------------------------------------------------------
    if time.time() < deadline and n_big != n_small:
        xs_small = np.arange(n_small) / n_small
        xs_big   = np.arange(n_big)   / n_big
        upsampled = np.interp(xs_big, xs_small, best_seq)
        upsampled = np.maximum(upsampled, 0.0)
        upsampled /= upsampled.sum()
        best_seq = upsampled
        print(f"[{time.time() - start_time:.2f}s] Upsampled to n={n_big}")

    # ------------------------------------------------------------------
    # 4) Exponentiated‑gradient descent on the final resolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 20.0:
            eg_deadline = time.time() + 0.70 * remaining
            best_seq, best_score = _exp_gradient_descent(best_seq,
                                                         eg_deadline,
                                                         start_time,
                                                         eta0=0.7)
            print(f"[{time.time() - start_time:.2f}s] After EG best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
