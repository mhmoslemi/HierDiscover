            if rng.random() < math.exp(-(new_score - cur_score) / T):

"""
Two‑stage optimiser for the step‑function bound:
    2·n·max(conv(a,a)) / (sum(a)²)

Stage 1 (fast) works on a down‑sampled problem (≈8192 points) and
performs a long simulated‑annealing walk together with a very diverse
initial pool (uniform, Beta‑densities, power‑ramps, monotone Dirichlet
samples).  The best small‑N shape is then up‑sampled to the target size
(16 k–65 k).  Stage 2 refines the large‑N candidate with an
exponentiated sub‑gradient descent on the exact ℓ∞‑objective followed by
a smooth‑max Adam optimiser and a tiny greedy “max‑reduction” sweep.

All vectors are kept on the probability simplex (scaling‑invariant), the
objective is calculated with an FFT‑based convolution, and the routine
terminates well before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the (probability) simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (lower score = better)."""
    if len(pool) < k:
        pool.append((float(score), vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (float(score), vec.copy())
        pool.sort(key=lambda x: x[0])

def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with optional flat plateau.
    The shape is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    For α<1 the shape is U‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates on the exact max‑convolution objective.
    Active‑set (indices attaining the current maximum) provides the sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices attaining the max, with tiny tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)   (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size schedule (decreasing)
        eta0 = 0.9
        eta = eta0 / (1.0 + it / 1500.0)

        # multiplicative update, safely clipped
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score

def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05
    T0 = 0.02
    T_end = 1e-5

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule (linear cooling on a log‑scale)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights (stable log‑sum‑exp)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation with reversed a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score

def _local_refine(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Very cheap greedy sweep moving mass from high‑impact to low‑impact indices."""
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        k = conv.argmax()

        # gradient of the selected maximal entry
        w = np.zeros_like(conv)
        w[k] = 1.0
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        i = np.argmax(grad)   # most responsible index
        j = np.argmin(grad)   # least responsible index
        if i == j or a[i] <= 0:
            break

        # transfer a tiny amount of mass from i to j
        eps = min(a[i] * 0.03, 1e-6 + 0.02 * (a[i] + a[j]))
        if eps <= 0:
            break
        a[i] -= eps
        a[j] += eps
        # ensure non‑negativity (tiny rounding errors)
        if a[i] < 0:
            a[j] += a[i]
            a[i] = 0.0
    return a

def propose_candidate(seed: int = 0,

    Search for a low‑score step‑function under the supplied time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size selection (large N for final result)
    # ------------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 800:
        target_N = 65536
    elif remaining_total > 400:
        target_N = 32768
    else:
        target_N = 16384

    # Small N for fast exploration
    small_N = min(8192, target_N)
    up_factor = max(1, target_N // small_N)
    large_N = small_N * up_factor

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool (size small_N)
    # ------------------------------------------------------------------
    TOP_K = 7
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform baseline
    _add(np.full(small_N, 1.0 / small_N))

    # symmetric Beta shapes (fine grid)
    for alpha in np.linspace(0.1, 1.2, 30):
        _add(_make_beta_shape(small_N, alpha))

    # asymmetric Beta shapes (finer grid)
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(small_N, a, b))

    # power‑ramp shapes
    for p in np.linspace(0.2, 5.0, 15):
        for plateau in np.linspace(0.0, 0.5, 5):
            _add(_make_shape(small_N, p, plateau))

    # monotone decreasing shapes (sorted gammas)
    rng = np.random.default_rng(seed + 123)
    for _ in range(30):
        v = rng.gamma(0.5, 1.0, size=small_N)
        v = np.sort(v)[::-1]       # decreasing
        v /= v.sum()
        _add(v)

    # random Dirichlet vectors with various concentrations
    for conc in [0.05, 0.1, 0.2, 0.4, 0.8, 1.5]:
        for _ in range(15):
            v = rng.gamma(conc, 1.0, size=small_N)
            s = v.sum()
            if s > 0:
                v /= s
                _add(v)

    # incorporate a global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != small_N:
                if small_N % starter.size == 0:
                    factor = small_N // starter.size
                    starter_small = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, small_N, endpoint=False)
                    starter_small = np.interp(x_new, x_old, starter)
            else:
                starter_small = starter.copy()
            starter_small = np.maximum(starter_small, 0.0)
            total = starter_small.sum()
            if total > 0:
                starter_small /= total
                _add(starter_small)
    except Exception:
        pass

    top.sort(key=lambda x: x[0])
    best_score, best_vec = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best (N={small_N}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing warm‑start (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * (deadline - start_time)
    cur_seq = best_vec.copy()
    cur_score = best_score
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    rng = np.random.default_rng(seed + 9999)
    max_move_factor = 0.30
    T_start = 1.0
    T_end = 1e-7
    it = 0

    while time.time() < phase1_deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        # geometric cooling
        T = T_start * (T_end / T_start) ** frac

        # pick source proportional to current weight
        src = rng.choice(small_N, p=cur_seq)
        if cur_seq[src] <= 1e-15:
            continue
        dst = rng.integers(small_N)
        if src == dst:
            continue

        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue
        a_prop = cur_seq.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_seq = cur_seq.copy()
                best_global_score = new_score
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

        # occasional restart from a random top‑candidate
        if it % 200000 == 0 and len(top) > 1:
            cur_seq = top[rng.integers(len(top))][1].copy()
            cur_score = _objective(cur_seq)

        # occasional small Gaussian perturbation
        if it % 50000 == 0:
            sigma = 0.01
            noisy = cur_seq + rng.normal(0.0, sigma, size=small_N)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                noisy /= s
                cur_seq = noisy
                cur_score = _objective(cur_seq)
                if cur_score < best_global_score:
                    best_global_seq = cur_seq.copy()
                    best_global_score = cur_score
                    print(f"[{time.time() - start_time:.2f}s] SA noise improved best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to the final size
    # ------------------------------------------------------------------
    if up_factor > 1:
        large_seq = np.repeat(best_global_seq, up_factor)
    else:
        # linear interpolation if sizes are not commensurate
        x_old = np.linspace(0, 1, small_N, endpoint=False)
        x_new = np.linspace(0, 1, large_N, endpoint=False)
        large_seq = np.interp(x_new, x_old, best_global_seq)

    large_seq = np.maximum(large_seq, 0.0)
    s = large_seq.sum()
    if s > 0:
        large_seq /= s
    else:
        large_seq = np.full(large_N, 1.0 / large_N)

    up_score = _objective(large_seq)
    print(f"[{time.time() - start_time:.2f}s] Upsampled to N={large_N}, score={up_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Refine at full size with exponentiated sub‑gradient & Adam
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate the majority to exp GD, keep a little time for final smooth GD
        sub_end = time.time() + 0.70 * remaining
        seq, sc = _exp_subgradient_descent(large_seq, sub_end, start_time)
        if sc < best_global_score:
            best_global_seq, best_global_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD (from upsample) improved to {sc:.6f}")

        # try a few other top candidates as well
        idx = 0
        while time.time() < sub_end and idx < len(top):
            vec = top[idx][1]
            if up_factor > 1:
                vec_up = np.repeat(vec, up_factor)
            else:
                x_old = np.linspace(0, 1, small_N, endpoint=False)
                x_new = np.linspace(0, 1, large_N, endpoint=False)
                vec_up = np.interp(x_new, x_old, vec)
            vec_up = np.maximum(vec_up, 0.0)
            vec_up /= vec_up.sum()
            seq2, sc2 = _exp_subgradient_descent(vec_up, sub_end, start_time)
            if sc2 < best_global_score:
                best_global_seq, best_global_score = seq2, sc2
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved to {sc2:.6f}")
            idx += 1

    # ------------------------------------------------------------------
    # 5) Final smooth‑max Adam polishing (any leftover time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_seq, best_global_score = refined_seq, refined_score
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Tiny greedy local refinement (few seconds, if possible)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        refined = _local_refine(best_global_seq, deadline)
        refined_score = _objective(refined)
        if refined_score < best_global_score:
            best_global_seq, best_global_score = refined, refined_score
            print(f"[{time.time() - start_time:.2f}s] Local refine improved to {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full_like(best_global_seq, 1.0 / best_global_seq.shape[0])

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
