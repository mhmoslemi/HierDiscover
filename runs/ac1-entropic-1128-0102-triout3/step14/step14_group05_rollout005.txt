def _project_simplex(v):
    # ensures v >=0 sum=1
    # sort descending, find rho, theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
step_search_v2.py – Gradient‑descent (sub‑gradient) optimisation on the exact
objective.

The algorithm works on a probability vector a (non‑negative, sum = 1) and
optimises the exact evaluation function

        J(a) = 2·n·max(conv(a,a)) / (sum(a)²)

by repeatedly
    • computing the linear self‑convolution via FFT,
    • locating the index k* where the convolution attains its maximum,
    • forming a sub‑gradient of J at a (the gradient of the max‑convolution
      together with the derivative of the normalising denominator),
    • taking an additive step *a ← a – η·∇J*,
    • projecting back onto the simplex by clipping negatives and renormalising.

A tiny exponential‑mirror step (a ← a·exp(‑η·∇J)) is also provided and used
as a fallback when the additive step would drive many coordinates to zero.
Multiple random restarts, a simple hill‑climb polishing phase and a final
fallback to the previously best known construction are included.

All work is done inside the supplied 1000 s wall‑clock budget and the best
sequence encountered is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility primitives (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple simplex projection (clip‑&‑renorm) – used after additive steps
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Clips negatives to 0 and rescales to have sum = 1."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total > 0.0:
        v /= total
    else:
        # fallback to uniform if everything vanished
        n = v.shape[0]
        v[:] = 1.0 / n
    return v


# ----------------------------------------------------------------------
# Sub‑gradient of the exact objective w.r.t. a (the probability vector)
# ----------------------------------------------------------------------


def _subgradient_J(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns the sub‑gradient ∂J/∂a for the current a and its self‑convolution.
    """
    n = a.shape[0]
    # locate a (possibly leftmost) maximal index; exploit symmetry
    k = int(conv.argmax())
    if k > n - 1:                     # use symmetry conv[k] = conv[2n-2-k]
        k = 2 * n - 2 - k

    # build the gradient of max(conv) w.r.t. a (g_i = 2·a_{k-i} when defined)
    g = np.zeros_like(a)
    shift = n - 1 - k                  # shift of the reversed array
    a_rev = a[::-1]

    if shift >= 0:
        # indices 0 … n‑shift‑1 get contributions
        g[: n - shift] = 2.0 * a_rev[shift:]
    else:
        # shift < 0  →  start at -shift
        start = -shift
        g[start:] = 2.0 * a_rev[: n + shift]

    # constants for the full objective
    sum_a = a.sum()
    max_conv = conv.max()
    C = 2.0 * n / (sum_a * sum_a)

    # ∂J/∂a = C·g – 2·C·max_conv/ sum_a · 1
    const = 2.0 * C * max_conv / sum_a
    grad = C * g - const
    return grad


# ----------------------------------------------------------------------
# Core optimisation loop (sub‑gradient descent with simplex projection)
# ----------------------------------------------------------------------


def _optimize_subgrad(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Runs a simple sub‑gradient descent from a0 until time runs out or a
    fixed number of iterations is reached. Returns the best vector seen
    and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)

    it = 0
    while time.time() < deadline and it < max_iters:
        # --- convolution -------------------------------------------------
        conv = _conv_self_fft(a, L_fft)

        # --- gradient ----------------------------------------------------
        grad = _subgradient_J(a, conv)

        # --- step size ---------------------------------------------------
        # Choose η so that the L1‑change is ≈ 0.01 (adjustable)
        norm_grad = np.linalg.norm(grad, 1)  # L1 norm is cheap
        if norm_grad == 0.0:
            step = 0.0
        else:
            step = 0.01 / norm_grad
        # upper‑bound step to keep updates stable
        step = min(step, 0.5)

        # --- additive update + projection --------------------------------
        a_new = a - step * grad
        a = _project_onto_simplex(a_new)

        # --- occasional exponential‑mirror fallback (helps avoid stagnation)
        if it % 500 == 0:
            # compute an exponential‑mirror update and keep the better one
            eps = 1e-12
            a_exp = a * np.exp(-step * grad)
            a_exp = _project_onto_simplex(a_exp)
            sc_cur = _eval_fast(a, L_fft)
            sc_exp = _eval_fast(a_exp, L_fft)
            if sc_exp < sc_cur:
                a = a_exp
                best_score = sc_exp
                best_a = a.copy()
            else:
                if sc_cur < best_score:
                    best_score = sc_cur
                    best_a = a.copy()
        else:
            # regular monitoring
            if it % 1000 == 0:
                sc = _eval_fast(a, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] iter {it} "
                        f"sub‑grad score={sc:.6f}"
                    )

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap hill‑climb (tiny random mass transfers)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray, deadline: float, start_time: float, L_fft: int):
    """
    Very local random mass transfers accepted only if they improve the exact
    objective. Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.2 % of the source mass
        eps = best_a[src] * rng.random() * 2e-3
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_onto_simplex(cand)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
    return best_a, best_score

# Main entry point – propose_candidate
# ----------------------------------------------------------------------
