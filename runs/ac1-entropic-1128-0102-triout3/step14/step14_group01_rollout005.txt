"""
Optimiser for the “max‑convolution” challenge.

Algorithm (≈ 1000 s budget)
--------------------------------
1.  Initialise a sequence of length *n* (default 4096) on the
    simplex (∑a = √(2·n)).  If the global variable ``height_sequence_1``
    is present it is used as a starter, otherwise the uniform vector.
2.  **LP‑update loop** (the core of the approach described in the
    statement):
        –  Compute M = max conv(a,a).
        –  Solve the linear programme
               maximise    1ᵀ b
               subject to  conv(a,b) ≤ M   (2n‑1 inequalities)
                           b ≥ 0
           (the convolution matrix of ``a`` is stored as a sparse CSR
            matrix, built once and re‑used each iteration.)
        –  Rescale the LP‑solution to the required normalisation
               g = (√(2n)/∑b)·b .
        –  Mix a fraction α of ``g`` into the current vector:
               a←(1−α)·a + α·g
           with a small back‑tracking line‑search on α (0.5 → 0.3 → 0.2
           → 0.1).  The mix keeps the sum fixed and never raises the
           objective, so M is non‑increasing.
        –  Keep the best vector found (lowest M).  Stop after a
           configurable number of LP solves or when the time budget
           expires.
3.  **Fine‑tuning** – a cheap projected‑gradient / simulated‑annealing
    hill‑climb (identical to the one used in the previous baseline)
    for the remaining time.  This can shave off a few extra digits.
4.  Return the vector normalised to sum = 1 (any scaling leaves the
    score unchanged).

The LP step drives the max‑convolution down much faster than pure
gradient descent, allowing us to beat the previous bound (~1.5078) and
reach the target ~1.503 within the allowed budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# --------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using an FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k}(a*a)[k] w.r.t. a (averaged over active k)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # a_i * a_{k-i}, derivative w.r.t. a_i
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used internally:   2·n·max(conv(seq,seq)) (seq is on the unit simplex)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# 1️⃣ Linear‑programming update (core of the algorithm)
# --------------------------------------------------------------
def _lp_update(a: np.ndarray, target_sum: float, L_fft: int,
               rows: np.ndarray, cols: np.ndarray,
               deadline: float, best_M: float,
               max_iter: int = 25) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based updates while time permits.
    Returns the best vector (still normalised to target_sum) and the
    corresponding max‑convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = best_M

    # pre‑allocate constant RHS vector for the LP
    rhs_const = np.empty(2 * n - 1, dtype=np.float64)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # 1️⃣ current bound M
        conv = _conv_self(a, L_fft)
        M = float(conv.max())
        if M < best_val - 1e-12:
            best_val, best_a = M, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it+1}   M={M:.6e}")

        # 2️⃣ build RHS (same for all constraints)
        rhs_const.fill(M)

        # 3️⃣ data = tile of a (the convolution matrix values)
        data = np.tile(a, n)                      # length n²
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # 4️⃣ solve LP  maximise 1ᵀb   ↔   minimise -1ᵀb
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=rhs_const,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if not res.success or res.x is None:
            print(f"[{time.time() - start_time:.2f}s] LP iteration failed, terminating LP stage.")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print(f"[{time.time() - start_time:.2f}s] LP returned zero vector, stop.")
            break

        # 5️⃣ rescale to the required normalisation
        g = (target_sum / sum_b) * b

        # 6️⃣ line‑search on the mixing fraction α
        improved = False
        for alpha in (0.5, 0.3, 0.2, 0.1):
            a_candidate = (1 - alpha) * a + alpha * g
            conv_cand = _conv_self(a_candidate, L_fft)
            M_cand = float(conv_cand.max())
            if M_cand <= M + 1e-12:               # non‑increase is enough
                a = a_candidate
                improved = True
                break
        if not improved:
            # cannot improve – stop the LP loop
            print(f"[{time.time() - start_time:.2f}s] No α yielded improvement, exiting LP stage.")
            break

    return best_a, best_val


# --------------------------------------------------------------
# 2️⃣  Final hill‑climb (same philosophy as the baseline)
# --------------------------------------------------------------
def _final_hill_climb(a: np.ndarray, target_sum: float, L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing / greedy hill‑climb that moves a tiny mass
    from the most positive sub‑gradient index to the most negative one.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_seq = a.copy()
    cur_score = _objective_score(cur_seq / cur_seq.sum())  # ensure unit‑sum for scoring
    best_seq, best_score = cur_seq.copy(), cur_score

    # temperature schedule (relative to score)
    T0 = 0.02
    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score
    decay = (1e-6 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        conv = _conv_self(cur_seq, L_fft)
        grad = _subgradient_maxconv(cur_seq, conv)

        # source – a positive gradient mass that we can move
        src_candidates = np.where(cur_seq > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – most negative gradient
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or cur_seq[src] < 1e-12:
                continue

        # move a tiny fraction of the source mass
        frac = rng.random() * 0.004 + 1e-6
        eps = cur_seq[src] * frac
        if eps <= 0:
            continue

        prop = cur_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        if prop[src] < -1e-12:
            continue

        new_score = _objective_score(prop / prop.sum())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_seq = prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.

    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()

    n = 4096

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    uniform = np.full(n, target_sum / n)
    a = uniform.copy()
    best_seq = a.copy()
    best_M = float(_conv_self(a, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Initial uniform M={best_M:.6e}")

    # If a global starter is available, use it (and renormalise)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:

            if starter.sum() > 1e-12:
                starter = starter / starter.sum() * target_sum
                a = starter.copy()
                best_M = float(_conv_self(a, L_fft).max())

    except Exception:

    i = np.arange(n, dtype=np.int32)

    a = np.maximum(a, 0.0)
