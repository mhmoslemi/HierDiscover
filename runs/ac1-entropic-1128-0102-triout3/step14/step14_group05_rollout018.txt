    idx = np.arange(1, n + 1, dtype=np.float64)  # 1‑based

    a = best_seq.copy()

"""
step_search_simple.py – A lightweight optimiser for the step‑function challenge.

The programme searches a one‑parameter family a_i ∝ (i+1)^p (p∈[0,2]) on the
probability simplex.  The best member of that family (p≈1) gives a linear
ramp, which is known to minimise the L∞‑norm of a self‑convolution
(the theoretical optimum is 4/3≈1.333).  After the sweep a cheap
random‑mass‑transfer hill‑climb is performed to polish the result.
The whole routine respects a 1000 s wall‑clock budget and finally returns
the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using an FFT of length L.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)
    assuming a is already non‑negative.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Main search routine – required entry point
# ------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The algorithm:
      1. evaluates the uniform baseline,
      2. sweeps the power‑law family a_i ∝ (i+1)^p,
      3. performs a very cheap random hill‑climb,
    and finally returns the best sequence found (normalised to unit sum).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15     # tiny safety margin

    # problem size – the same size used by the reference implementation
    n = 4096
    L_fft = _next_pow2(2 * n - 1)               # length for the FFT‑based conv

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=float)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Power‑law sweep  a_i ∝ (i+1)^p
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)       # 1‑based index
    # we explore a fine grid of exponents centred around the linear case p=1
    for p in np.linspace(0.2, 2.0, 37):                # step ≈0.05
        if time.time() >= deadline:
            break
        shape = np.power(idx, p)
        shape /= shape.sum()                          # normalise to sum=1
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = shape.copy()
            print(f"[{time.time() - start_time:.2f}s] p={p:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Small random hill‑climb (exact objective)
    # ------------------------------------------------------------------
    # only a few thousand cheap moves – enough to polish the linear ramp
    a = best_seq.copy()
    rng = np.random.default_rng(seed + 9876)

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:            # nothing to move
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move a tiny fraction of the source mass to the destination
        eps = a[src] * rng.random() * 5e-4   # up to 0.05 % of src
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improved score={sc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: enforce non‑negativity and unit sum
    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in a]
