"""
Simple monotone power‑ramp optimiser.

The optimal continuous solution of
    minimise  2·n·max_{t∈[0,2]} (f∗f)(t)   subject to f≥0 , ∫_0^1 f = 1
is an increasing density f(x) = (p+1)·x^p (p>0).  After discretisation
with n points this gives the vector  

    a_i ∝ ((i+0.5)/n)^p ,   i=0,…,n‑1 .

For any p>0 the maximum of the self‑convolution scales like 1/n, while the
edge term a_{n‑1}² scales like 1/n², so for a reasonably large n the
objective ≈2·(p+1)²·B(p+1,p+1) can be made arbitrarily small.  The
implementation below tries a short grid of exponents, keeps the best,
optionally refines it with a few steps of the projected sub‑gradient
descent from the previous solution, and finally returns the normalised
sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum‑free, scale invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  –  fast version."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Monotone power‑ramp: a_i ∝ ((i+0.5)/n)^p , Σa_i = 1
# ----------------------------------------------------------------------
def _make_monotone_power(n: int, p: float) -> np.ndarray:
    """Increasing power‑ramp (no symmetry) normalised to sum = 1."""
    idx = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.power(idx, p, out=idx)  # reuse idx as buffer
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (light Adam momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    All indices with convolution within `tol`·max are treated as active
    (or the `top_k` largest if given).  Adam‑style momentum is used.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 → denominator =1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        # active set
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        # sub‑gradient of the (smoothed) max‑conv
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the scoring harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Build a monotone power‑ramp sequence, optionally improve it with a few
    projected sub‑gradient steps, and return the best vector found within
    the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # spare margin

    # ------------------------------------------------------------------
    # 1) Choose a length – large enough that edge effects are negligible
    # ------------------------------------------------------------------
    n = 16384                     # power of two → fast FFT

    # ------------------------------------------------------------------
    # 2) Scan a small grid of exponents p>0
    # ------------------------------------------------------------------
    exponents = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 6.0, 8.0, 10.0]
    best_seq = None
    best_score = float("inf")

    for p in exponents:
        if time.time() >= deadline:
            break
        vec = _make_monotone_power(n, p)
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(f"[{time.time() - start_time:.2f}s] p={p:>4}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Tiny local refinement (only if we still have time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        sub_deadline = deadline - 1.0
        refined_seq, refined_score = _subgradient_descent(
            best_seq, sub_deadline, start_time,
            tol=0.01, top_k=None
        )
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Guard against pathological empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 5) Normalise (just in case) and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
