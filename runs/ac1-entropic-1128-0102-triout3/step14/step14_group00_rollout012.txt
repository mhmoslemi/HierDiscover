def _project_to_simplex(v):
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score non‑negative step function.

The strategy consists of three stages:

1️⃣  Build a modest library of analytic shapes (Beta, sinusoidal, Gaussian,
    uniform) and keep the few best candidates (objective = 2·n·max(conv)/∑²).

2️⃣  Refine each candidate with a *smooth‑max* gradient descent on the simplex.
    We minimise a smooth approximation of   max_k (a*a)[k]   using
    Adam‐style additive updates and a projection onto the probability simplex.
    The smoothness parameter β is increased from 0.5 to 200 during the run,
    turning the smooth loss into the true max‑loss.  After each update the
    vector is symmetrised (a ← (a+rev(a))/2) – the optimal shape is empirically
    symmetric – and re‑projected onto the simplex.

3️⃣  Finish with a short simulated‑annealing polish (tiny mass swaps) to
    escape any remaining sub‑optimal plateaus.

All heavy work uses real FFTs, so a length of a few thousand points is fast
enough to perform many‑hundreds of thousands of iterations inside the 1000 s
budget.  The routine returns a plain list of Python floats that respects the
required non‑negativity and normalisation constraints.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective: 2·n·max(conv)/∑²   (no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised symmetric beta pdf (∝ x^{a‑1}(1‑x)^{b‑1}) on n points."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_shape(n: int, p: float) -> np.ndarray:
    """Shape proportional to sin(π·x)^p, x∈(0,1)."""
    xs = (np.arange(n) + 0.5) / n
    shape = np.sin(np.pi * xs) ** p
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, σ = sigma_frac·n."""
    sigma = max(sigma_frac * n, 1e-6)
    xs = np.arange(n)
    centre = (n - 1) / 2.0
    shape = np.exp(-((xs - centre) ** 2) / (2.0 * sigma ** 2))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("vector required")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max helpers (soft approximation of max(conv))
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(β·(conv_k−max)).  Sum w_k = 1."""
    maxc = conv.max()
    # subtract maxc for numerical stability
    exps = np.exp(beta * (conv - maxc))
    w = exps / exps.sum()
    return w


def _gradient_smoothmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max loss L = (1/β)·log∑exp(β·conv_k)
    w.r.t. a (the sequence).  Implements
        ∂L/∂a_i = 2 * Σ_j a_j·w_{i+j}
    where w are the smooth‑max weights.
    """
    n = a.shape[0]
    w = _smooth_weights(conv, beta)               # length 2·n‑1
    Lw = len(w)

    # correlation a ⋆ w (no reversal) using FFT:
    L2 = _next_pow2(n + Lw - 1)
    A = np.fft.rfft(a, n=L2)
    W = np.fft.rfft(w, n=L2)
    corr_full = np.fft.irfft(np.conj(A) * W, n=L2)[: n + Lw - 1]

    start = Lw - 1                     # index of shift 0
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max gradient descent with Adam, simplex projection and symmetry
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style additive updates on the smooth‑max surrogate.
    Returns the best sequence seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.1

    # smooth‑max schedule
    beta_start, beta_end = 0.5, 200.0
    beta_ramp_iters = 100_000          # after this β stays at beta_end

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and smooth‑max weight
        conv = _conv_self(a)
        # linearly raise β during the first ramp phase
        if it <= beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        else:
            beta = beta_end

        grad = _gradient_smoothmax(a, conv, beta)

        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        lr = lr0 / math.sqrt(it)        # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection & symmetry (the shape is known to be symmetric at optimum)
        a = _project_to_simplex(a)
        a = (a + a[::-1]) * 0.5

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} score={cur_score:.6f}")

    return best_a, best_score

    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from 0.02 to 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # bias towards larger sites
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5   # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial pool creation (analytic families + random mixtures)
# ----------------------------------------------------------------------
def _initial_pool(n: int, rng: np.random.Generator, pool_sz: int = 10):
    """Return a list of (score, vector) tuples, sorted by score ascending."""
    candidates = []

    # 1. uniform baseline
    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    # 2. random beta shapes
    for _ in range(30):
        a = 10 ** rng.uniform(-2.0, 1.0)   # ~0.01 … 10
        b = 10 ** rng.uniform(-2.0, 1.0)
        candidates.append(_make_beta_shape(n, a, b))

    # 3. sinusoidal ^p shapes
    for _ in range(20):
        p = 10 ** rng.uniform(-2.0, 2.0)   # 0.01 … 100
        candidates.append(_make_sine_shape(n, p))

    # 4. discrete Gaussians
    for _ in range(20):
        sigma_frac = 10 ** rng.uniform(-2.0, 0.0)   # 0.01 … 1
        candidates.append(_make_gaussian_shape(n, sigma_frac))

    # 5. random convex mixtures of two previously generated shapes
    for _ in range(30):
        s1 = candidates[rng.integers(len(candidates))]
        s2 = candidates[rng.integers(len(candidates))]
        w = rng.random()
        mix = w * s1 + (1.0 - w) * s2
        mix /= mix.sum()
        candidates.append(mix)

    # evaluate and keep the best `pool_sz` candidates
    scored = [(_objective(v), v) for v in candidates]
    scored.sort(key=lambda x: x[0])
    return scored[:pool_sz]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    Returns a plain Python list of non‑negative floats (the sequence).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a length that balances resolution and speed.
    # ------------------------------------------------------------------
    n = 2048                      # power of two → FFT friendly

    # ------------------------------------------------------------------
    # 1) Build a small pool of promising analytic candidates.
    # ------------------------------------------------------------------
    pool = _initial_pool(n, rng, pool_sz=8)
    best_score, best_seq = pool[0]

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine each candidate with smooth‑max gradient descent.
    #    We allocate roughly 70 % of the remaining budget to this phase.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + 0.70 * remaining
        for idx, (_, vec) in enumerate(pool):
            if time.time() >= gd_deadline:
                break
            print(f"[{time.time() - start_time:.2f}s] GD start on pool member {idx}")
            refined, score = _smooth_gradient_descent(vec, gd_deadline, start_time)
            if score < best_score:
                best_score, best_seq = score, refined
                print(f"[{time.time() - start_time:.2f}s] New pool best = {score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final annealing polish (if any time left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing polish")
        best_seq, best_score = _anneal(best_seq, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Normalise and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
