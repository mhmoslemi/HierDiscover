"""
step_search_v2.py – Gradient‑descent + large‑step hill‑climb.

The driver starts from the best known construction (the global variable
`height_sequence_1` if present) and performs three phases:

1.  A quick parametric sweep over geometric‑power families to obtain a
    decent seed (far better than the uniform baseline).

2.  Projected gradient descent on the *exact* objective
    2·n·max(conv(a,a))/sum(a)².  The sub‑gradient is known in closed form
    (the max of the convolution is attained at a single shift k*,
    ∂max/∂a_i = 2·a_{k*‑i} when the index is valid, else 0).  Each step
    is projected onto the probability simplex (non‑negative, unit sum).

3.  A stochastic “large‑transfer” hill‑climb (random mass moves up to
    several percent of the source) together with a tiny amount of
    simulated‑annealing.  The loop runs until the 1000 s wall‑clock
    budget is exhausted.

All heavy work (convolution, evaluation) is performed with FFT‑based
routines, so thousands of objective evaluations comfortably fit into the
budget.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas – plain functions only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    prod = Af * Af
    conv = np.fft.irfft(prod, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  Faster than
    evaluate_sequence because it works directly on a NumPy array."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _conv_max_index(a: np.ndarray, L_fft: int):
    """Return (max_value, index_of_max) of the self‑convolution."""
    conv = _conv_self_fft(a, L_fft)
    idx = int(np.argmax(conv))
    return conv[idx], idx


def _subgradient(a: np.ndarray, L_fft: int):
    """
    Compute the sub‑gradient of the exact objective w.r.t. a.
    Returns grad (same shape as a) and also the current max‑conv value
    and the index where it is attained.
    """
    n = a.shape[0]
    S = a.sum()
    max_val, k = _conv_max_index(a, L_fft)  # k ∈ [0, 2n‑2]

    # constant term that appears for every component
    const = (4.0 * n * max_val) / (S ** 3)

    # term that depends on a_{k‑i}
    # Build a shifted view: b[i] = a[k‑i] when 0≤k‑i<n else 0
    idx = np.arange(n)
    j = k - idx
    valid = (j >= 0) & (j < n)
    shifted = np.zeros_like(a)
    shifted[valid] = a[j[valid]]

    grad = (4.0 * n / (S ** 2)) * shifted - const
    return grad, max_val, k


def _project_to_simplex(v: np.ndarray):
    """
    Project a non‑negative vector onto the simplex Σx_i = 1.
    Simple clipping + renormalisation (sufficient because we never
    introduce negative values after the clipping step).
    """
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0:
        # fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine performs a fast parametric sweep, followed by
    projected gradient descent and a final stochastic hill‑climb.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the evaluator
    deadline = start_time + budget_s - 0.10

    # ------------------------------------------------------------------
    # Determine problem size and initialise a seed sequence
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size == 0:
            raise ValueError
    except Exception:
        starter = None

    if starter is None:
        n = 4096
        uniform = np.full(n, 1.0 / n)
        best_seq = uniform.copy()
    else:
        n = starter.size
        best_seq = _project_to_simplex(np.maximum(starter, 0.0))

    L_fft = _next_pow2(2 * n - 1)

    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] initial score={best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (in case the starter was pathological)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    uniform_score = _eval_fast(uniform, L_fft)
    if uniform_score < best_score:
        best_seq = uniform.copy()
        best_score = uniform_score
        print(f"[{time.time() - start_time:.2f}s] uniform improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Fine parametric sweep (geometric × power‑law families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index for power term
    # grids – a bit denser than the previous version
    r_grid = np.logspace(np.log10(0.94), np.log10(0.9999), num=35)
    p_grid = np.linspace(-2.5, 1.5, 41)

    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            shape = geo * (idx ** p)
            s = shape.sum()
            if s <= 0:
                continue
            shape /= s
            # try a few mixtures with the uniform baseline
            for lam in (0.0, 0.25, 0.5, 0.75, 0.9):
                cand = lam * uniform + (1.0 - lam) * shape
                cand = _project_to_simplex(cand)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.3f} λ={lam:.2f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected gradient descent on the exact objective
    # ------------------------------------------------------------------
    # Parameters for the descent
    base_step = 0.12                     # base step size factor
    grad_iterations = 0
    while time.time() < deadline - 2.5:
        grad, max_val, k_max = _subgradient(best_seq, L_fft)
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break

        # initial step size – inversely proportional to the gradient norm
        step = base_step / (grad_norm + 1e-12)

        improved = False
        # simple back‑tracking line‑search (up to 5 reductions)
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            α = step * factor
            cand = best_seq - α * grad
            cand = _project_to_simplex(cand)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_seq = cand
                best_score = sc
                improved = True
                grad_iterations += 1
                print(f"[{time.time() - start_time:.2f}s] GD iter {grad_iterations} α={α:.3e} → {sc:.6f}")
                break
        if not improved:
            # if a gradient step fails, fall back to a small random move
            src = np.random.randint(n)
            if best_seq[src] > 1e-14:
                dst = np.random.randint(n)
                if src != dst:
                    eps = best_seq[src] * np.random.rand() * 0.03  # up to 3 %
                    cand = best_seq.copy()
                    cand[src] -= eps
                    cand[dst] += eps
                    cand = _project_to_simplex(cand)
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_seq = cand
                        best_score = sc
                        print(f"[{time.time() - start_time:.2f}s] random rescue → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Large‑step stochastic hill‑climb (simulated‑annealing style)
    # ------------------------------------------------------------------
    # Annealing schedule – temperature slowly decays to ~0
    T0 = 0.02
    Tmin = 1e-5
    total_time = deadline - start_time
    while time.time() < deadline:
        # linear temperature decay based on elapsed fraction
        elapsed = time.time() - start_time
        T = max(Tmin, T0 * (1.0 - elapsed / total_time))

        src = np.random.randint(n)
        if best_seq[src] <= 1e-14:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # transfer up to a few percent of the source mass
        eps = best_seq[src] * np.random.rand() * 0.07   # ≤7 %
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_to_simplex(cand)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improved → {sc:.6f}")
        else:
            # maybe accept a uphill move with probability exp(-(Δ)/T)
            delta = sc - best_score
            if np.random.rand() < np.exp(-delta / T):
                best_seq = cand
                best_score = sc
                # (no print for accepted uphill moves to keep output tidy)

        # occasional “jitter” restart to escape possible plateaus
        if np.random.rand() < 0.001:
            jitter = best_seq + np.random.normal(scale=0.015, size=n)
            jitter = _project_to_simplex(jitter)
            sc = _eval_fast(jitter, L_fft)
            if sc < best_score:
                best_seq = jitter
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] jitter restart → {sc:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    best_seq = _project_to_simplex(best_seq)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
