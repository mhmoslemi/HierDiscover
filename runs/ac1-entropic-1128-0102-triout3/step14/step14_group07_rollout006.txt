"""
Mini‑search for a low‑score step function.

The original code used a smooth–max Adam optimiser.  Here we replace it
by a *projected sub‑gradient descent* that works directly on the true
objective  f(a)=2·n·max(conv(a,a)) .  For a given vector a the gradient
of a single active convolution entry (the current maximum) is
    ∇ conv_k(a) = 2·shift(a, k) .
We average the gradients over all active indices (those attaining the
maximum) and take a decreasing step size.  After each step we project
the vector back onto the probability simplex ( Σa_i = 1 , a_i ≥ 0 ).

The search proceeds in three phases

1. ** diversified seeding** – many analytic shapes (uniform, power‑ramps,
   symmetric / asymmetric Beta densities) and random Dirichlet vectors.
   The K best (lowest) candidates are retained.

2. ** sub‑gradient refinement** – each retained candidate is refined with
   the projected sub‑gradient method for as long as time permits.

3. ** simulated‑annealing polish** – a short SA run with relatively
   large mass moves gives a final nudge.

All phases respect the 1000 s budget; the algorithm is deliberately
light‑weight (only NumPy and FFTs) and works with a single CPU.  The
returned list sums to 1, contains only non‑negative floats and can be
directly fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT, length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Objective for a probability vector (Σa = 1):
       2·n·max(conv(a,a)).  This equals the value returned by
       evaluate_sequence for a normalised sequence."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {a≥0, Σa=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape factories (all return vectors that already sum to 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p_left: float, p_right: float,
                     plateau_frac: float = 0.0) -> np.ndarray:
    """Symmetric power‑ramp with possibly different left/right exponents."""
    plateau_len = int(round(plateau_frac * n))
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        left = idx ** p_left
        right = idx ** p_right
        shape = np.concatenate([left, np.ones(plateau_len), right[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_sym(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sparse_uniform(n: int, s: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random subset of size s (s≥1)."""
    idx = rng.choice(n, size=min(s, n), replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / idx.size
    return a


def _make_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet vector with a single concentration parameter."""
    a = rng.gamma(concentration, 1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0:
        return _make_uniform(n)
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    Returns the best (vector, score) seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # step‑size schedule – η₀ is tuned experimentally
    eta0 = 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---------- current convolution ----------
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={it:,} "
                  f"best={cur_score:.6f}")

        # ---------- active set (all indices attaining the maximum) ----------
        # a tiny tolerance is used to guard against floating‑point noise
        tol = 1e-12
        active = np.where(conv >= max_conv - tol)[0]

        # ---------- gradient of the smooth max (average over active indices) ----------
        w = np.zeros(2 * n - 1, dtype=np.float64)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # correlation w ⋆ a_rev  (length 3·n‑2)
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a[::-1], n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1:n - 1 + n]          # length n

        # ---------- projected update ----------
        eta = eta0 / math.sqrt(it)                 # classic 1/√t schedule
        a -= eta * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish (larger moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """Very simple SA that swaps a random fraction of mass between two positions."""
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # temperature schedule (exponential decay)
    T_start = 0.2
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        # pick two distinct indices (src, dst) – src chosen proportionally to its weight
        src = rng.choice(n, p=a)
        if a[src] <= 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to max_move_frac) of src's mass to dst
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if (new_score < cur_score) or (rng.random() < math.exp(-(new_score - cur_score) / T)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score

# Tiny helper used during pool construction

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted pool keeping only the best *k*."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])
