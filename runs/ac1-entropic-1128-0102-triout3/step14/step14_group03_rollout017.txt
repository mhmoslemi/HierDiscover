"""
Simulated‑annealing refinement of a step function.

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a)**2) .
For a vector normalised to sum = 1 the score reduces to 2·n·max(conv).

The algorithm proceeds as follows:

1. Initialise a candidate `a` from the global variable `height_sequence_1`
   (if present) or from the uniform distribution.
   The vector is forced onto the probability simplex (non‑negative,
   sum = 1).

2. Compute the full linear convolution `c = np.convolve(a, a)`.  Its maximum
   gives the current score.

3. Repeatedly perform a *mass‑transfer* move:
   - pick a source index `src` (with a non‑zero entry) and a distinct
     destination `dst`;
   - move a small amount `eps = a[src] * rng.random() * max_move_frac`
     from `src` to `dst`.
   - using the linearity of convolution we update the convolution
     incrementally (O(n) work) instead of recomputing it from scratch.

   The move is accepted if it improves the score, otherwise it may be
   accepted with a Metropolis probability that decays with a temperature
   schedule (simulated annealing).

4. Every few thousand iterations a tiny random jitter is added to all
   coordinates followed by a projection onto the simplex – this helps to
   escape plateaus.

5. The best vector seen throughout the run is returned.  The routine respects
   the global time budget (default 1000 s) and prints occasional progress
   messages.

The method works directly on the full vector (no low‑dimensional
parameterisation) and, thanks to the O(n) incremental update, can explore
many millions of proposals within the allowed time, often beating the
previous best score of ≈ 1.508.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Reference: https://doi.org/10.1007/s10107-009-0289-8
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _initial_vector(n: int) -> np.ndarray:
    """Return a good starting vector of length n."""
    # Try to use the globally provided best construction.
    try:
        v = np.asarray(height_sequence_1, dtype=float)
        if v.ndim == 1 and v.size == n:
            v = np.maximum(v, 0.0)
            if v.sum() == 0.0:
                raise ValueError
            v = _project_simplex(v)
            return v
    except Exception:
        pass
    # Fallback: a mildly decreasing exponential shape (works well in practice)
    i = np.arange(n, dtype=float)
    shape = np.exp(-0.0004 * i)          # gentle decay
    return _project_simplex(shape)


# ----------------------------------------------------------------------
# Incremental convolution update for a mass‑transfer move
# ----------------------------------------------------------------------


def _conv_update(conv: np.ndarray, a: np.ndarray,
                 src: int, dst: int, eps: float) -> np.ndarray:
    """
    Return a copy of `conv` after moving `eps` from a[src] to a[dst].
    The update uses the exact linear formula:
        conv' = conv - 2*eps*shift(a,src) + 2*eps*shift(a,dst)
                + eps**2 at 2*src and 2*dst
                - 2*eps**2 at src+dst.
    """
    n = a.shape[0]
    new_conv = conv.copy()
    # shift(a, src) and shift(a, dst) are added/subtracted on contiguous blocks
    new_conv[src:src + n] -= 2.0 * eps * a
    new_conv[dst:dst + n] += 2.0 * eps * a

    eps2 = eps * eps
    new_conv[2 * src] += eps2
    new_conv[2 * dst] += eps2
    new_conv[src + dst] -= 2.0 * eps2
    return new_conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the given
    evaluation function.  The algorithm is a simulated‑annealing
    mass‑transfer search with an O(n) incremental convolution update.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a tiny safety margin for the runtime environment
    deadline = start_time + budget_s - 0.25

    # --------------------------------------------------------------
    # Initialise
    # --------------------------------------------------------------
    # length of the search vector – we keep the same size as the previous
    # best construction (4096) unless a different length is provided.
    n = 4096
    a = _initial_vector(n)                     # sum(a) == 1
    conv = np.convolve(a, a)                    # full linear convolution (len = 2*n-1)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max                # because sum(a)=1
    best_a = a.copy()
    best_score = cur_score
    best_conv = conv.copy()

    print(f"[{time.time() - start_time:.2f}s] initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    init_temp = 0.06               # temperature at the start
    final_temp = 0.0004            # temperature at the end
    init_move = 0.035              # maximal fraction of a[src] moved initially
    final_move = 0.0015            # maximal fraction at the end
    iteration = 0

    while time.time() < deadline:
        # ----- schedule (linear decay) ---------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1
        temp = init_temp * (1.0 - frac) + final_temp * frac
        max_move_frac = init_move * (1.0 - frac) + final_move * frac

        # ----- propose a random source/destination --------------------
        src = rng.integers(n)
        if a[src] < 1e-12:                 # ignore practically empty slots
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1                       # guarantee dst != src

        eps = a[src] * rng.random() * max_move_frac
        if eps <= 0.0:
            continue

        # ----- compute the updated convolution ------------------------
        new_conv = _conv_update(conv, a, src, dst, eps)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # ----- Metropolis acceptance ----------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # climb with a probability that vanishes as the temperature cools
            delta = new_score - cur_score
            if rng.random() < np.exp(-delta / (temp + 1e-12)):
                accept = True

        if accept:
            # apply the move permanently
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            cur_score = new_score
            cur_max = new_max

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:.2f}s] it={iteration:,} "
                      f"new best = {best_score:.6f}")

        # ----- occasional global jitter to avoid stagnation ----------
        if iteration % 20000 == 0 and iteration > 0:
            jitter = 1e-4 * (rng.random(n) - 0.5) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)          # re‑normalise
            conv = np.convolve(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:.2f}s] jitter improved score = {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Final normalisation and sanity checks
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    final_val = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:.2f}s] finished – best score = {final_val:.6f}")

    return [float(v) for v in best_a]
