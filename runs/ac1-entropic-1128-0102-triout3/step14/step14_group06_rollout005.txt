   best = min(pop, key=_objective)
   best_score = _objective(best)

   F = 0.8
  CR = 0.9 (crossover probability)
   while time.time() < deadline:
      # For each target index maybe iterate over all pop
      for i in range(len(pop)):
         if time.time() >= deadline: break
         # mutation
         indices = list(range(len(pop)))
         indices.remove(i)
         a,b,c = rng.choice(indices, size=3, replace=False)
         mutant = pop[a] + F * (pop[b] - pop[c])
         # enforce nonneg
         mutant = np.maximum(mutant, 0)
         # crossover
         cross = rng.random(pop[i].shape) < CR
         if not np.any(cross):
            cross[rng.integers(pop[i].size)] = True
         trial = np.where(cross, mutant, pop[i])
         # project onto simplex
         trial = _project_simplex(trial)
         # evaluate
         trial_score = _objective(trial)
         if trial_score < _objective(pop[i]) - 1e-12:
            pop[i] = trial
            if trial_score < best_score:
               best_score = trial_score
               best = trial.copy()
               print(f"[{time.time() - start_time:.2f}s] DE new best={best_score:.6f}")
   return best, best_score

"""
Step‑function optimiser – population based DE + exponential sub‑gradient.

The optimiser builds a diverse initial pool (uniform, Beta, power‑ramps,
monotone ramps, exponential, sinusoidal, random Dirichlet).  The best few
candidates are kept.  A lightweight Differential‑Evolution (DE) loop
explores the neighbourhood of the pool while always staying on the
probability simplex (Euclidean projection).  The best vector found by DE
is then refined by an exponentiated sub‑gradient descent on the exact
ℓ∞ convolution objective and finally polished with a smooth‑max Adam
descent.  The algorithm adapts the problem size (N) to the allowed
budget and returns a non‑negative list that respects the required format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution / objective utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp_sym(n: int, p: float, plateau: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with optional flat plateau.
    Normalised to sum 1.
    """
    plateau = max(0.0, min(0.5, plateau))
    plateau_len = int(round(plateau * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_monotone_ramp(n: int, p: float, inc: bool = True) -> np.ndarray:
    """Monotone power‑ramp (increase or decrease)."""
    idx = np.arange(1, n + 1, dtype=float) / n
    if inc:
        shape = idx ** p
    else:
        shape = (1.0 - idx) ** p
    shape /= shape.sum()
    return shape


def _make_exponential(n: int, decay: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-decay·i).  decay>0."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-decay * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine(n: int, phase: float = 0.0) -> np.ndarray:
    """Sinusoidal shape shifted to be non‑negative."""
    t = np.arange(n, dtype=float) / (n - 1)
    shape = np.sin(math.pi * t + phase)  # range [-1,1]
    shape = shape - shape.min()           # shift to >=0
    shape += 1e-12                        # avoid pure zeros
    shape /= shape.sum()
    return shape


def _make_random_dirichlet(n: int, concentration: float, rng) -> np.ndarray:
    p = rng.gamma(concentration, 1.0, size=n)
    total = p.sum()
    if total > 0:
        p /= total
    else:
        p = np.full(n, 1.0 / n)
    return p

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (exact ℓ∞ objective)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Exponentiated gradient descent on max‑convolution (exact)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # active set (indices attaining the current maximum, tolerance)
        tol = max(1e-14, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * correlation(w, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # step‑size (slow decay → more aggressive early moves)
        eta0 = 0.6
        eta = eta0 / (1.0 + it / 1500.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – surrogate for max‑conv
# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.07
    T0 = 0.03
    T_end = 5e-5

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule (linear)
        frac = it / (it + 1500.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights on the convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation of w with reversed a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # back to simplex
        a = _project_simplex(a)

    return best_a, best_score

# Differential‑Evolution (simple DE on the simplex)
# ----------------------------------------------------------------------
def _differential_evolution(pop: list,
                            scores: list,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """Very light DE – mutation + binomial crossover, always project onto simplex."""
    P = len(pop)
    n = pop[0].shape[0]
    rng = np.random.default_rng()
    F = 0.8
    CR = 0.9

    # keep track of global best
    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    it = 0
    while time.time() < deadline:
        it += 1
        for i in range(P):
            if time.time() >= deadline:
                break

            # mutation: random distinct indices a,b,c ≠ i
            idxs = list(range(P))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # enforce non‑negativity before projection
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])

            # simplex projection
            trial = _project_simplex(trial)

            trial_score = _objective(trial)
            # greedy replacement
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE iter={it:,} new best={best_score:.6f}")

        # occasional diversification – replace the worst individual
        if it % 100 == 0:
            worst_idx = int(np.argmax(scores))
            # create a fresh random Dirichlet candidate
            conc = 0.3 + 0.7 * rng.random()
            fresh = _make_random_dirichlet(n, conc, rng)
            fresh_score = _objective(fresh)
            if fresh_score < scores[worst_idx]:
                pop[worst_idx] = fresh
                scores[worst_idx] = fresh_score
                if fresh_score < best_score:
                    best_score = fresh_score
                    best_vec = fresh.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE replacement improved best={best_score:.6f}")

    return best_vec, best_score

    """

    # --------------------------------------------------------------

        N = 131072

        N = 65536
    else:

            scores.append(sc)

            if sc < scores[worst]:
                pool[worst] = vec.copy()

    # Uniform baseline
    _add(_make_uniform(N))

    # Symmetric Beta shapes

    # Symmetric power‑ramps

    # Monotone ramps (increasing & decreasing)

            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:

                _add(starter_up)
    except Exception:

    # Ensure we have TOP_K entries (fill with random Dirichlet if needed)

        a = _make_random_dirichlet(N, 0.5, rng)
        _add(a)

    # Current best

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
