"""
Propose a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm is a multi‑stage search:
  1.  Extensive parametric sweep (geometric, power, beta, stretched‑exp,
      logistic, sine‑power).  The best candidate is retained.
  2.  Broad random Dirichlet sampling (many concentration values) to
      diversify the population.
  3.  Smooth‑max Adam optimisation – a temperature‑controlled softmax
      approximation of the max‑convolution, with projection onto the
      probability simplex.
  4.  Exponentiated‑gradient descent on the *exact* max‑convolution
      (sub‑gradient of the L∞ norm).  Step size is adaptive and the
      iterate is always re‑normalised.
  5.  Short final hill‑climb / simulated‑annealing refinement.
All stages respect a global time budget (default 1000 s).  The routine
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# 1️⃣  Parametric families (geometric, power, beta, stretched‑exp,
#                     logistic, sine‑power)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mixture with uniform to avoid extreme spikes
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 61)
    betas = np.linspace(0.05, 4.0, 61)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential -----------------------------------------
    scales = np.linspace(0.25, 1.0, 11)
    p_vals = np.linspace(0.2, 5.0, 41)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} sc={scale:.2f} score={sc:.6f}")

    # ---- logistic -------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.01, 0.2, 11)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power -----------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 41)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Random Dirichlet trials (more intensive)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 2000                     # more draws than the original version
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [0.01, 20]
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate (unchanged)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (same as in the original script)
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
                              np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Exponentiated‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of the L∞‑norm
    of the self‑convolution.  The update is
        a ← normalize( a * exp(-η·grad) )
    where η (step) decays slowly.  Improvements are recorded.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.05            # initial step size
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        # sub‑gradient of the max
        grad = _subgradient_maxconv(a, conv)

        # decay the step size gently
        eta_t = eta / (1.0 + it * 1e-4)

        # exponentiated‑gradient step
        # clip exponent to avoid overflow/underflow
        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)

        # accept if improvement, otherwise keep old a
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")
        else:
            # if not improving, shrink step size a little
            eta *= 0.99
            # occasional random shake to escape plateaus
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing (unchanged)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine follows a five‑stage pipeline and returns a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialisation (uniform or provided starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0    # reserve ~250 s
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more trials)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation of smooth‑max
    # --------------------------------------------------------------
    hill_time = 130.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient descent on exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting EG descent")
        best_seq, best_score = _eg_maxconv_descent(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After EG best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
