"""
Step‑function optimiser (improved version).

The evaluation function to minimise is::

    2·n·max(conv(a,a)) / (sum(a)²)

For a non‑negative vector a of length n.
The theoretical lower bound of this expression is 2 (attained by the uniform
distribution).  The previous reference implementation obtained ≈1.508 for a
large n, apparently exploiting numerical artefacts.  This version follows the
same overall structure (diverse initial pool → simulated‑annealing warm‑start
→ exponentiated sub‑gradient descent → smooth‑max Adam polishing) but
introduces a few changes that give a higher chance of escaping the local
minimum encountered before:

*   a larger and more diverse pool (additional “inverse‑Beta’’ shapes);
*   a two‑phase simulated‑annealing schedule (large moves early, fine moves
    later);
*   an adaptive step‑size for the exponentiated sub‑gradient descent and
    optional restarts from the current best;
*   a more aggressive temperature schedule for the final Adam optimiser;
*   a safety margin that automatically reduces the problem size when the
    remaining time becomes small.

The function returns a plain Python list of non‑negative floats.  All
intermediate vectors are kept normalised to have unit sum (the objective is
scale‑invariant), and the final vector is additionally clipped to `[0,1000]`
to satisfy the checks performed by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT (exact up to floating‑point noise)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (more diverse than the reference code)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat plateau; normalised to sum 1."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.pad(shape, (0, n - shape.size), constant_values=0.0)
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Beta(α,β) density (symmetric when β=None) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    if beta is None:
        beta = alpha
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _make_inverse_beta(n: int, a: float, b: float) -> np.ndarray:
    """1 / Beta(α,β)‑shaped density normalised to sum 1 (produces heavy tails)."""
    # we construct the usual Beta density and then invert it point‑wise.
    beta = _make_beta(n, a, b)
    inv = 1.0 / (beta + 1e-12)
    inv /= inv.sum()
    return inv


# ----------------------------------------------------------------------
# Simple top‑k pool management
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted top‑k pool."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# Optimisers
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated (multiplicative) sub‑gradient descent on the exact
    ℓ∞‑objective.  A diminishing step‑size schedule and occasional
    restarts from the current best are used.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()                         # this equals Σ a_i²
        # active set (indices attaining the max, tolerance to avoid empty set)
        tol = max(1e-14, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # uniform weight over active indices → sub‑gradient of max
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # Correlation a_rev * w gives gradient of max w.r.t. a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # step‑size (slow decay → more exploration)
        eta0 = 0.9
        eta = eta0 / (1.0 + it / 1500.0)

        # multiplicative update, clipped to avoid overflow/underflow
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # re‑project onto simplex (keeps ∑a = 1)
        a /= a.sum()

        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD it={it:,} best={cur:.6f}")

        # very occasional random restart from the best so far
        if it % 20000 == 0 and np.random.rand() < 0.2:
            a = best_a.copy()
            a += np.random.normal(scale=0.001, size=n)   # small jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_a, best_score


def _smoothAdam_descent(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the smooth‑max surrogate of the objective.
    Temperature anneals from 0.02 → 1e‑4, encouraging the solution to become
    progressively more “max‑like’’.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.04
    T0 = 0.02
    T_end = 1e-4

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_c = conv.max()
        cur = 2.0 * n * max_c / (a.sum() ** 2)

        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam it={it:,} best={cur:.6f}")

        # linear temperature decay
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights (stable log‑sum‑exp)
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # gradient of the smooth‑max (correlation of w with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score
