```
def alternate_order(a):
    idx = np.argsort(-a) # descending
    b = np.empty_like(a)
    l=0; r=len(a)-1; toggle=True
    for i in idx:
        if toggle:
            b[l]=a[i]; l+=1
        else:
            b[r]=a[i]; r-=1
        toggle = not toggle
    return b
```

```
def simulated_annealing(a, deadline):
    cur = a.copy()
    cur_score = _objective_score(cur)
    best = cur.copy()
    best_score = cur_score
    T0 = 0.01 * best_score
    T_end = 1e-6 * best_score
    total_time = deadline - time.time()
    while time.time() < deadline:
        # time factor
        t = (time.time() - start) / total_time
        T = T0 * (T_end/T0) ** t
        # propose
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] < 1e-12:
            continue
        # choose delta as fraction of cur[i] with fraction decreasing over time
        max_frac = 0.5
        frac = max_frac * (1 - t) * rng.random()
        delta = cur[i] * frac
        prop = cur.copy()
        prop[i] -= delta
        prop[j] += delta
        # optionally ensure nonnegativity
        if prop[i] < -1e-12:
            continue
        s_new = _objective_score(prop)
        d = s_new - cur_score
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cur = prop
            cur_score = s_new
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] SA best={best_score:.6f}")
    return best, best_score
```
