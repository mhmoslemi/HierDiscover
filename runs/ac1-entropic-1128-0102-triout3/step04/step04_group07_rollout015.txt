    def fast_score(a):
        s = a.sum()
        if s < 1e-12:
            return math.inf
        # use FFT length size_conv
        fft_a = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fft_a * fft_a, size_conv)[:2*n-1]
        max_b = conv.max()
        return 2.0 * n * max_b / (s * s)
    # define projection
    def proj_simplex(v):

        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho

            theta = 0.0

"""
Search for a step function with low self‑convolution
using (i) a broad random‑support scan (including smaller supports),
(ii) a smoothed gradient descent that minimises a soft‑max of the
convolution, and (iii) a final perturbation hill‑climb.
All vectors are projected onto the probability simplex after each update.
"""
import time, math, random, numpy as np

def project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    # implementation as per Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def fast_score(a):
    """Fast version of evaluate_sequence (no safety checks)."""
    s = a.sum()
    if s < 1e-12:
        return math.inf
    n = a.shape[0]
    # length for FFT : power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def smoothed_gradient_descent(a_init, time_limit):
    """Projected gradient descent on a soft‑max approximation of max‑convolution."""
    a = a_init.copy()
    n = a.shape[0]
    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for a*p convolution
    best_a = a.copy()
    best_val = fast_score(a)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # FFT of current vector
        fft_a = np.fft.rfft(a, size_fft)

        # full autocorrelation a*a
        conv_full = np.fft.irfft(fft_a * fft_a, size_fft)
        conv = conv_full[: 2 * n - 1]

        # temperature‑like smoothing parameter (increase slowly)
        alpha = 1.0 + it * 0.001
        # soft‑max weights
        max_c = conv.max()
        # subtract max_c for numerical stability
        exp_vals = np.exp(alpha * (conv - max_c))
        p = exp_vals / exp_vals.sum()

        # gradient = 2 * (p * a)   (convolution, valid part)
        fft_p = np.fft.rfft(p, size_fft)
        grad_full = np.fft.irfft(fft_a * fft_p, size_fft)
        # valid part corresponds to indices n‑1 … n‑1+n‑1
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # simple diminishing step size
        eta = 0.2 / math.sqrt(1.0 + it)

        a_new = a - eta * grad
        a = project_onto_simplex(a_new)

        if it % 500 == 0:
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[gd iter {it}] new best {val:.6f}")
                if val <= 1.503:
                    break
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights whose score is as low as possible."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    n = 1500
    best_seq = None
    best_val = math.inf

    # ----------------------------------------------------------------------
    # 0. use a previously stored construction if it exists
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)[:n]
        if prev.size == n:
            val = fast_score(prev)
            best_seq, best_val = prev, val
            print("[init] previous sequence score", val)
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 1. random‑support scan (including smaller supports)
    # keep the 10 best candidates for later refinement
    candidate_list = []
    max_candidates = 10

    support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700,
                    600, 500, 400, 350, 300, 250, 200, 150, 100]
    attempts_per_size = 2000

    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[scan] support size {m}")
        attempts = 0
        while attempts < attempts_per_size and time.time() < deadline:
            attempts += 1
            S = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / m
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} attempt={attempts} score={val:.6f}")
                if val <= 1.503:
                    print("[success] target reached during scan")
                    return best_seq.tolist()
            # keep a short list of promising candidates
            if len(candidate_list) < max_candidates:
                candidate_list.append((val, a.copy()))
            else:
                # replace worst if current is better
                worst_idx = max(range(len(candidate_list)),
                               key=lambda i: candidate_list[i][0])
                if val < candidate_list[worst_idx][0]:
                    candidate_list[worst_idx] = (val, a.copy())
        print(f"[scan] size {m} finished after {attempts} attempts, best so far {best_val:.6f}")

    # sort candidates by score (ascending)
    candidate_list.sort(key=lambda x: x[0])
    # ----------------------------------------------------------------------
    # 2. refine the few best candidates with smoothed gradient descent
    if candidate_list:
        # allocate part of the remaining budget equally among the candidates
        remaining = deadline - time.time()
        if remaining > 5.0:
            per_candidate = remaining / len(candidate_list)
            print(f"[refine] running gradient descent on {len(candidate_list)} candidates")
            for idx, (_, cand) in enumerate(candidate_list):
                sub_time = min(per_candidate, 10.0)   # cap per run
                cand_best, cand_val = smoothed_gradient_descent(cand, sub_time)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand_best.copy()
                    print(f"[refine] candidate {idx} improved to {cand_val:.6f}")
                    if best_val <= 1.503:
                        print("[success] target reached in refinement")
                        return best_seq.tolist()

    # ----------------------------------------------------------------------
    # 3. global gradient descent from the current best vector
    remaining = deadline - time.time()
    if remaining > 2.0:
        print("[global] performing gradient descent from current best")
        best_seq, best_val = smoothed_gradient_descent(best_seq if best_seq is not None else np.full(n, 1.0 / n), remaining)
        print(f"[global] final grade {best_val:.6f}")

    # ----------------------------------------------------------------------

"""
Search for a step‑function with a low self‑convolution value.

The algorithm works in four stages:
  1. Random‑support scan (including relatively small supports) with a uniform
     distribution on the chosen support.  The 10 best candidates are stored.
  2. Each of those candidates is refined by a projected gradient descent that
     minimises a smooth approximation of the max‑convolution (soft‑max).
  3. A global gradient descent is started from the best vector found so far.
  4. A short hill‑climb (random mass transfers) is performed as a final
     polishing step.

All vectors are projected onto the probability simplex after each update,
so the ℓ₁‑norm stays equal to 1 – the evaluation function is homogeneous and
this projection does not affect the objective value.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = 1}."""
    # Duchi et al. (2008) – O(n log n) implementation.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no safety checks).
    It returns 2·n·max(conv(a,a)) / (sum(a))².
    """
    s = a.sum()
    if s < 1e-12:
        return math.inf
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # power‑of‑two ≥ 2n‑1
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def smoothed_gradient_descent(a_init: np.ndarray,
                              time_limit: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of
    max(conv(a,a)).  The smoothing is performed with a soft‑max
    (log‑sum‑exp) that becomes tighter as the iteration counter grows.
    Returns the best vector found during the allotted time.
    """
    a = a_init.copy()
    n = a.shape[0]
    # FFT size that can handle a * p where len(p)=2n‑1
    size_fft = 1 << ((3 * n - 2).bit_length())
    best_a = a.copy()
    best_val = fast_score(a)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # ---- convolution a*a -------------------------------------------------
        fft_a = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fft_a * fft_a, size_fft)
        conv = conv_full[: 2 * n - 1]

        # ---- smooth max (soft‑max) -------------------------------------------
        alpha = 1.0 + it * 0.001               # slowly tighten the approximation
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))   # stabilised exponentiation
        p = exp_vals / exp_vals.sum()                # soft‑max weights, sum(p)=1

        # ---- gradient = 2·(p * a) -------------------------------------------
        fft_p = np.fft.rfft(p, size_fft)
        grad_full = np.fft.irfft(fft_a * fft_p, size_fft)
        # the “valid” part of the correlation: indices n‑1 … n‑1+n‑1
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ---- step and projection ---------------------------------------------
        eta = 0.2 / math.sqrt(1.0 + it)        # simple diminishing step size
        a_new = a - eta * grad
        a = project_onto_simplex(a_new)

        # ---- occasional logging & best‑track ---------------------------------
        if it % 500 == 0:
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[gd iter {it}] new best {val:.6f}")
                if val <= 1.503:
                    break
        it += 1
    return best_a, best_val
