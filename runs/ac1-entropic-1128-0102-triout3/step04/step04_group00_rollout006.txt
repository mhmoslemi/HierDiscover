"""

`evaluate_sequence`.

Algorithm
---------
1. **Fast primitives** – simplex projection, linear (non‑circular) self‑
   convolution via FFT, sub‑gradient of the max‑convolution.
2. **Exploratory shape sweep** – generate a variety of analytic
   candidates (U‑shaped power laws, exponential ends, Beta‑distributions,
   triangular, uniform, etc.) for the target length `n≈4800`.  Keep the

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# Exact sub‑gradient from all active convolution entries
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are taken as active.
    The contributions are summed (no averaging).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov momentum)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.3,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index
    to a random index.  Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source index chosen proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Analytic shape generators (return a normalised vector of length n)
# ----------------------------------------------------------------------
def _shape_uniform(n):
    return np.full(n, 1.0 / n)


def _shape_triangular(n):
    i = np.arange(n)
    w = np.minimum(i + 1, n - i)
    return w / w.sum()


def _shape_u_powerlaw(n, p):
    """U‑shaped power law: weight ∝ 1 / distance_to_nearest_edge^p."""
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i) + 1.0
    w = 1.0 / (d ** p)
    return w / w.sum()


def _shape_exponential_ends(n, k):
    """Two exponential decays from both ends."""
    i = np.arange(n)
    w = np.exp(-k * i) + np.exp(-k * (n - 1 - i))
    return w / w.sum()


def _shape_beta(n, a, b):
    """Symmetric Beta pdf on {0,…,n‑1} (a,b>0)."""
    x = (np.arange(n) + 0.5) / n  # mid‑point sampling
    w = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    # avoid infinities when a<1 or b<1 (ends become large but finite)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Fast objective (no clipping) – useful for shape sweep
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a)).  a is assumed to sum to 1."""
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # tiny safety margin

    # target length – a compromise between expressiveness and speed
    n_target = 4800

    # ------------------------------------------------------------------
    # Try to initialise from a previously found construction (global variable)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = _upsample(starter, n_target)
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1. Analytic shape sweep – keep a few promising seeds
    # ------------------------------------------------------------------
    shape_candidates = []

    # uniform and triangular (baseline)
    shape_candidates.append(_shape_uniform(n_target))
    shape_candidates.append(_shape_triangular(n_target))

    # U‑shaped power‑laws
    for p in np.linspace(0.2, 3.0, 14):
        shape_candidates.append(_shape_u_powerlaw(n_target, p))

    # Exponential ends
    for k in [0.001, 0.003, 0.006, 0.010, 0.020, 0.040]:
        shape_candidates.append(_shape_exponential_ends(n_target, k))

    # Symmetric Beta distributions (U‑shaped and bell‑shaped)
    for a in [0.3, 0.5, 0.8, 1.2, 2.0]:
        shape_candidates.append(_shape_beta(n_target, a, a))

    # If we have a starter, add it as well
    if starter is not None:
        shape_candidates.append(starter)

    # Quick evaluation of all candidates (fast objective)
    scores = [_fast_objective(seq, n_target) for seq in shape_candidates]

    # Keep the best few for deeper optimisation
    top_k = min(6, len(shape_candidates))
    top_indices = np.argsort(scores)[:top_k]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Gradient descent refinement of the top analytic seeds
    # ------------------------------------------------------------------
    for idx in top_indices:
        if time.time() >= deadline:
            break
        a0 = shape_candidates[idx]
        # allocate a modest time slice for each seed (≈15 % of remaining)
        remaining = deadline - time.time()
        slice_time = max(8.0, 0.15 * remaining)
        slice_deadline = time.time() + slice_time

        a_opt, _ = _run_gradient_descent(
            a0, n_target, slice_deadline, start_time,
            beta=0.90, step0=0.35
        )
        sc = evaluate_sequence(a_opt.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] seed {idx} improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Random‑perturbation + short gradient runs while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 6.0:      # leave a few seconds for the final anneal
        if best_seq is None:
            base = np.random.rand(n_target)
        else:
            base = best_seq
        base = base / base.sum()

        # random Gaussian‑ish perturbation (2 % of total mass)
        eps = 0.02
        pert = base + eps * (rng.random(n_target) - 0.5)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        remaining = deadline - time.time()
        slice_time = max(5.0, 0.12 * remaining)
        slice_deadline = time.time() + slice_time

        a_opt, _ = _run_gradient_descent(
            pert, n_target, slice_deadline, start_time,
            beta=0.92, step0=0.30
        )
        sc = evaluate_sequence(a_opt.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] perturbed run improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing refinement (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, sc_ref = _run_annealing(
            best_seq, n_target,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6,
            steps=200_000
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improved to {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 5. Normalise and return a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n_target, 1.0 / n_target)

    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
