"""
Multi‑stage smooth optimisation for the step‑function problem.

The algorithm proceeds as follows:

1. **Progressive up‑sampling** – start from a modest length (≈800) and
   gradually increase the resolution (2400 → 4800 → 9600).  At each size the
   current best sequence is linearly interpolated to the new length.

2. **Smooth projected Adam descent** – minimise a smooth approximation of
   ``max(conv(a))`` using the soft‑max (log‑sum‑exp) surrogate.
   For a temperature λ the surrogate is
       f_λ(a) = (1/λ)·log∑_k exp(λ·conv_k(a)).
   Its gradient has a closed‑form convolution which we evaluate with FFTs.
   An Adam‑style update is performed and the iterate is projected onto the
   probability simplex (non‑negative, sum = 1).  The temperature λ is
   increased geometrically (0.5 → … → 200) so that the optimisation first
   sees a smoothed landscape and then focuses on the actual maximum.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk that
   transfers a small amount of mass between two random positions.  Downhill
   moves are always accepted; uphill moves are accepted with a temperature
   schedule.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = 1} (O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of  f_λ(a) = (1/λ)·log Σ exp(λ·conv_k(a))  w.r.t. a.
    Returns a vector of length n (same shape as a).
    """
    n = a.shape[0]
    # linear convolution a*a
    conv = _conv_self(a)
    max_c = conv.max()
    # stable soft‑max weights (sum to 1)
    exp_arg = lam * (conv - max_c)          # shift by max to avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w /= w_sum                              # shape (2n‑1,)

    # gradient = 2 * (a_rev * w)_{slice}
    a_rev = a[::-1]
    conv_len = 3 * n - 2                    # length of a_rev * w
    fft_len = 1 << ((conv_len).bit_length())
    Af = np.fft.rfft(a_rev, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    rev = np.fft.irfft(Af * Bf, n=fft_len)[:conv_len]
    # slice indices n‑1 … 2n‑2 (inclusive) give the gradient entries
    grad = 2.0 * rev[(n - 1) : (2 * n - 1)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with smooth surrogate
# ----------------------------------------------------------------------
def _adam_smooth_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lam_schedule=None,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter_per_lambda: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Runs Adam with a geometric increase of the smoothing parameter λ.
    Returns the best point (in terms of the *actual* max‑convolution) and its
    max‑convolution value.
    """
    if lam_schedule is None:
        lam_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    pow_b1 = 1.0
    pow_b2 = 1.0
    it_total = 0

    best_a = a.copy()
    best_max = _conv_self(a).max()

    for lam in lam_schedule:
        it_lambda = 0
        while time.time() < deadline and it_lambda < max_iter_per_lambda:
            it_lambda += 1
            it_total += 1

            # gradient of the smooth surrogate
            grad = _gradient_softmax(a, lam)

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            pow_b1 *= beta1
            pow_b2 *= beta2
            m_hat = m / (1.0 - pow_b1)
            v_hat = v / (1.0 - pow_b2)

            # Adam step + projection
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_simplex(a)

            # evaluate *real* objective (max‑convolution)
            cur_max = _conv_self(a).max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] λ={lam:.1f} "
                    f"it={it_total} max={best_max:.8f}"
                )
        if time.time() >= deadline:
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 500000,
) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2  # move up to 20 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n} iter={it} score={best_score:.6f}"
            )

    return best_seq, best_score

    # ------------------------------------------------------------------

    try:
