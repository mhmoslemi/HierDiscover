```python
from scipy.optimize import linprog

def _solve_lp(f, M):
    n = len(f)
    # Build matrix A_ub shape (2n-1, n)
    # For each k in 0..2n-2, row corresponding to sum_i f_i * g_{k-i} <= M,
    # where g_{j}=0 if j out of range.
    # Equivalent: For each k, the coefficients for g_j are f_{k-j} if 0 <= k-j < n -> i = k-j
    # So row[k][j] = f_{k-j} if 0 <= k-j < n else 0.
    # Build using loops but vectorized.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)  # inclusive
        # i runs from i_start..i_end; j = k-i
        # So for i in range(i_start,i_end+1): A[k, k-i] = f[i]
        # Set A[k, i_start:i_end+1? Actually mapping
        # We'll fill via loop for each k: for i in range(i_start,i_end+1): j=k-i; A[k,j]=f[i]
        # Use vectorized assignment:
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]
    # To avoid constructing dense matrix each time, maybe use sparse matrix but fine.
    # Solve maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

def _lp_refine(f, deadline, max_iters=10, mix=0.2):
    n = len(f)
    best_f = f.copy()
    best_score = 2 * n * _conv_self(f).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(f)
        M = conv.max()
        if not SCIPY_AVAILABLE:
            break
        g0 = _lp_step(f, M)
        if g0 is None:
            break
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g
        # mix
        f_new = (1 - mix) * f + mix * g
        f_new = f_new / f_new.sum()
        f = f_new
        score = 2 * n * _conv_self(f).max()
        if score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP refine iter={it}, score={score:.6f}")
    return best_f, best_score

def _smooth(seq, sigma=1.0):
    # apply Gaussian blur via convolution in frequency domain.
    n = len(seq)
    x = np.arange(-n//2, n//2 + n%2)
    g = np.exp(-0.5 * (x / sigma) ** 2)
    g /= g.sum()
    # compute linear convolution using FFT (circular but we can pad)
    pad_len = len(g) + n - 1
    fft_len = 1 << (pad_len - 1).bit_length()
    seq_f = np.fft.rfft(seq, n=fft_len)
    g_f = np.fft.rfft(g, n=fft_len)
    sm = np.fft.irfft(seq_f * g_f, n=fft_len)[:n]
    sm = np.maximum(sm,0)
    sm = sm / sm.sum()
    return sm

"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Simplex utilities** – projection onto {x≥0, Σx=1}.
2. **Convolution** – linear (non‑circular) self‑convolution via FFT.
3. **Projected gradient descent** – Nesterov‑style momentum, diminishing
   step size.  Works on the *simplex* and is the work‑horse optimiser.
4. **LP refinement** – (optional, requires SciPy) solve a linear programme
   ``max Σ g_i`` subject to ``f * g ≤ M`` (where ``M`` is the current
   maximal self‑convolution).  The solution pushes mass to “quiet’’ places.
   The new vector is mixed with the current one (mixing factor 0.2) and the
   process is repeated a few times.
5. **Multi‑stage scheme**
   * Stage 0 – short gradient runs on a modest size (n≈800) with several
     random restarts.
   * Stage 1 – LP refinement on a medium size (n≈1500) if SciPy is available.
   * Stage 2 – upscale to n≈2400 and run a longer gradient descent.
   * Stage 3 – upscale to n≈4800 and continue gradient descent.
   * Stage 4 – final simulated‑annealing “mass‑transfer’’ moves.
   The time budget (default 1000 s) is split dynamically; each stage stops
   as soon as the global deadline is reached.
6. **Return** – the best sequence found, normalised to sum = 1, as a plain
   Python list of floats.
"""

import time
import math
import numpy as np

# optional SciPy imports (used only for the LP refinement)
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Exact sub‑gradient from all active convolution entries
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are taken as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ------------------------------------------------------------
# Projected gradient descent on the simplex
# ------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.3,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{last_improve - start_time:6.2f}s] grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

        # early stop if we have been stuck for a while
        if time.time() - last_improve > 30.0:
            break

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index to a
    random index.  Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source index sampled proportionally to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Optional smoothing (simple moving average)
# ------------------------------------------------------------
def _smooth(seq: np.ndarray, k: int = 5) -> np.ndarray:
    """Apply a cheap moving‑average smoothing then re‑normalise."""
    if k <= 1:
        return seq
    ker = np.ones(k) / float(k)
    sm = np.convolve(seq, ker, mode='same')
    sm = np.maximum(sm, 0.0)
    sm /= sm.sum()
    return sm


# ------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# ------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float):
    """
    Solve
        max Σ g_i
        s.t. g_i ≥ 0,   (f * g)(k) ≤ M  for all k.
    Returns the optimal g (may be None on failure).
    """
    n = f.shape[0]
    # Build sparse constraint matrix A where A[k, i] = f[k-i]   (zero if out of range)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximize Σ g_i  →  minimize –Σ g_i
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _lp_refine(f: np.ndarray, deadline: float,
               start_time: float,
               max_iters: int = 8,
               mix: float = 0.2) -> tuple[np.ndarray, float]:
    """Iteratively improve ``f`` by LP steps and convex mixing."""
    n = f.shape[0]
    best_f = f.copy()
    best_score = 2.0 * n * _conv_self(f).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(f)
        M = conv.max()
        g0 = _lp_step(f, M)
        if g0 is None:
            break
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g                     # normalise to Σg=1
        f = (1.0 - mix) * f + mix * g
        f = _project_simplex(f)            # maintain feasibility
        cur_score = 2.0 * n * _conv_self(f).max()
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP refine iter={it}, score={cur_score:.6f}")
    return best_f, best_score
