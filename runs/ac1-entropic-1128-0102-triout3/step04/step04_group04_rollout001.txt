"""
Propose a low‑value step function for ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation**
   * Use the previously found construction `height_sequence_1` if it exists.
   * Otherwise start from a power‑law shape (a_i ∝ (i+1)^‑p) and a few
     random Dirichlet draws; keep the best among them.

2. **Fast gradient descent on the max‑convolution**
   * For a probability vector a (sum = 1) the objective is
     f(a)=2·n·max(conv(a,a)).  The sub‑gradient at the active (maximal)
     convolution indices K is
         g_i = 2·∑_{k∈K} a_{k‑i}.
   * We move opposite to the centred sub‑gradient:
         d = – (g – mean(g)).
     The step size α is limited by the non‑negativity constraints,
     and a few candidate α are tried (large to small).
   * After a trial step we project back onto the probability simplex.
   * If none of the α‑steps improves the score we fall back to an
     exponentiated‑gradient (mirror‑descent) update.

3. **Permutation & random‑swap moves**
   * Occasionally (≈1 % of iterations) a random permutation of the
     whole vector is tried – ordering strongly influences the
     convolution peak.
   * Every iteration a cheap random swap of two entries is also tested.

4. **Final cheap hill‑climbing**
   * When the time budget is almost exhausted a classic random
     mass‑transfer hill‑climber (move a ≤ 4 % of the source mass)
     performs the last refinements.

All updates keep the vector non‑negative and normalised, so the final
sequence can be passed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of a with itself and its maximum value."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return indices whose convolution value is within a relative tolerance
    of the current maximum.  If too few are found the globally largest
    entries are added so that we have at most ``max_add`` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of ``max(conv)`` for the active index set K.
    For each k∈K we add 2·a_{k‑i} to g_i if the index is inside the
    valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a non‑negative vector whose entries sum to 1.
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector a (sum=1)."""
    n = a.shape[0]
    _, mx = _conv_and_max(a)
    return float(2.0 * n * mx)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine respects a ~1000 s wall‑clock
    budget and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Load previous construction if available, otherwise create a start.
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s == 0.0:
            raise ValueError
        init = init / s
        n = init.shape[0]
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
              f"(n={n}) → score={best_score:.6f}")
    except Exception:
        # fallback: power‑law + a few random Dirichlet seeds
        n = 5000
        best_seq = None
        best_score = float("inf")
        print(f"[{time.time() - start_time:6.2f}s] No prior construction – "
              f"using power‑law + random starts (n={n})")

        # power‑law grid
        p_vals = np.linspace(0.10, 0.90, 31)
        for p in p_vals:
            if time.time() > deadline:
                break
            i = np.arange(n, dtype=float) + 1.0
            a = i ** (-p)
            a /= a.sum()
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Power p={p:.3f} → "
                      f"score={sc:.6f}")

        # a few random Dirichlet draws (concentration=0.8)
        for _ in range(12):
            if time.time() > deadline:
                break
            a = np.random.gamma(0.8, 1.0, size=n)
            a /= a.sum()
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Random Dirichlet → "
                      f"score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Main optimisation loop (gradient + occasional permutation).
    # --------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1

        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:5d} "
                  f"→ new best score={cur_score:.6f}")

        # ----------------------------------------------------------
        # (a) compute sub‑gradient from the active set
        # ----------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _gradient_active(current, K)

        # ----------------------------------------------------------
        # (b) gradient‑descent step on the simplex
        # ----------------------------------------------------------
        mean_g = g.mean()
        d = -(g - mean_g)                     # descent direction
        # maximum admissible step before any entry would become negative
        if np.any(d < 0):
            limits = -current[d < 0] / d[d < 0]
            max_alpha = limits.min()
        else:
            max_alpha = np.inf

        improved = False
        # try a few decreasing step sizes
        if np.isfinite(max_alpha):
            alphas = [0.5 * max_alpha, 0.2 * max_alpha,
                      0.1 * max_alpha, 0.05 * max_alpha,
                      0.02 * max_alpha, 0.01 * max_alpha]
        else:
            alphas = [1.0, 0.5, 0.2, 0.1]

        for alpha in alphas:
            if alpha <= 0:
                continue
            cand = current + alpha * d
            # projection onto the simplex guarantees feasibility
            cand = _project_simplex(cand)
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score - 1e-12:
                current = cand
                improved = True
                break

        if not improved:
            # ------------------------------------------------------
            # (c) fallback: exponentiated‑gradient (mirror descent)
            # ------------------------------------------------------
            eta = 0.5 / math.sqrt(it + 1.0)
            log_a = np.log(np.maximum(current, 1e-15)) - eta * g
            log_a -= np.max(log_a)           # numerical stabilisation
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            current = a_new

        # ----------------------------------------------------------
        # (d) occasional permutation move (≈1 % of the iterations)
        # ----------------------------------------------------------
        if np.random.rand() < 0.01:
            perm = np.random.permutation(n)
            cand = current[perm]
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Permutation improve "
                      f"→ score={new_score:.6f}")

        # ----------------------------------------------------------
        # (e) cheap random swap (always attempted)
        # ----------------------------------------------------------
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            cand = current.copy()
            cand[i], cand[j] = cand[j], cand[i]
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Swap improve "
                      f"→ score={new_score:.6f}")

    # --------------------------------------------------------------
    # 3. Final cheap random hill‑climbing (mass‑transfer moves).
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 15000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or current[i] == 0.0:
                continue
            eps = np.random.rand() * 0.04          # up to 4 % of source mass
            delta = eps * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                continue
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb "
                      f"→ score={new_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best vector found (plain Python floats).
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
