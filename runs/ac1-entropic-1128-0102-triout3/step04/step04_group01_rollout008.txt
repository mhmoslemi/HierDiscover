"""
Hybrid Adam‑gradient + targeted annealing optimizer.

The objective (2·n·max(a*a)) is scale‑invariant; we therefore work on the
probability simplex (∑aᵢ = 1).  The algorithm proceeds in three phases
for each chosen length n:

1. Warm‑start by up‑sampling the best distribution found so far (or a
   random simplex point if none exists).
2. Adam‑based projected gradient descent on a smooth log‑sum‑exp
   approximation of the ∞‑norm of the self‑convolution.  The smoothing
   parameter λ is gradually reduced, giving a fast early descent and a
   precise late refinement.
3. Targeted simulated‑annealing: at every step we locate the lag k where
   the convolution attains its current maximum and move a small amount
   of mass from an index that contributes heavily to that lag to a
   random destination.  The move is accepted with a Metropolis rule.
   This concentrates the search on the most restrictive constraints and
   typically yields a lower peak than a purely random anneal.

Time is split among a short schedule of increasing lengths
(≈ 4k → 8k → 16k → 32k).  The best vector seen overall is returned.
All helper routines are top‑level, no closures are used and the code
conforms to the 1000 s budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, zero‑padded)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate gradient
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ·log Σ exp(conv/λ)   w.r.t. a.
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a gives dL/da
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimizer on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Adam on the smoothed log‑sum‑exp objective, projected onto simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float('inf')

    # λ‑schedule (initial factor, decay every few thousand iterations)
    lam_factor = 0.40
    lam_decay = 0.65
    lam_decay_interval = 2500

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective because sum(a)=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # λ update
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (diminishing)
        lr = 0.6 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted annealing that concentrates on the current maximal lag
# ----------------------------------------------------------------------


def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-5,
                            max_steps: int = 60000) -> tuple[np.ndarray, float]:
    """
    Move a small ε from an index that heavily contributes to the present
    maximum of the self‑convolution to a random destination.  Accept moves
    with a Metropolis rule.  The schedule is geometric from start_temp to
    end_temp.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    while time.time() < deadline and it < max_steps:
        it += 1
        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # locate the lag with current maximal convolution value
        k_max = int(conv.argmax())

        # contribution of each index to this lag (a_i * a_{k_max-i})
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idxs = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idxs] * a[k_max - idxs]
        total = contrib.sum()

        # choose a source index proportionally to its contribution
        if total > 0.0:
            probs = contrib / total
            src = np.random.choice(idxs, p=probs)
        else:
            src = np.random.randint(n)

        # pick a destination index (uniform, different from src)
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # propose moving a fraction (up to 20 %) of the source mass
        eps = a[src] * np.random.rand() * 0.20
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate new objective (exact, no surrogate)
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            conv = conv_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    Returns a plain Python list of heights.  The routine respects the
    provided time budget (default 1000 s) and combines Adam gradient descent
    with a peak‑focused annealing scheme.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a modest safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available
    # ------------------------------------------------------------------
    starter = None
    try:
        # `height_sequence_1` is injected by the evaluation harness
        starter = np.asarray(globals().get('height_sequence_1'), dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
    except Exception:
        starter = None

    # Length schedule – we grow the problem size gradually.
    lengths = [4096, 8192, 16384, 32768]
    # If time runs out early we may never reach the last entries.
    random.shuffle(lengths)   # avoid systematic bias on early termination

    global_a = starter

    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of the remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - now
        if remaining <= 0.0:
            break
        # larger lengths receive a slightly larger fraction
        fraction = 0.12 / (len(lengths) - idx)
        slice_deadline = time.time() + max(8.0, remaining * fraction)

        # --------------------------------------------------------------
        # Phase 1 – Adam gradient descent on a smooth surrogate
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – targeted annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.25)
            a_ref, score_ref = _run_targeted_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                max_steps=50000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # remember the best distribution for the next up‑sampling step
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed (uniform 200‑point distribution)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Convert to plain Python list – the evaluator will clamp / normalise
    return [float(x) for x in best_seq]
