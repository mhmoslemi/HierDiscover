def lp_optimize(f, M):
    n = len(f)
    # create convolution matrix A (2n-1 x n)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        # i runs over overlapping
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i, j = k - i
        # A_ub[k, j] = f[i]
        # So loop over i
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # b_ub = M (repeat)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum => maximize sum
    # use linprog
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None, None  # fallback
    g = res.x
    val = -res.fun
    return g, val

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> float:
    """Maximum of the auto‑convolution a⊛a (real part only)."""
    conv = np.convolve(a, a)
    return conv.max()


def _build_conv_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build a sparse matrix A such that (A @ g) == f ⊛ g.
    Shape of A: (2·n‑1, n)  where n = len(f).
    """
    n = f.shape[0]
    # rows indices for the full (2n‑1)×n matrix
    row_idx = np.arange(2 * n - 1)[:, None] - np.arange(n)
    mask = (row_idx >= 0) & (row_idx < n)
    # positions where the Toeplitz entry is non‑zero
    rows = np.nonzero(mask)[0]
    cols = np.nonzero(mask)[1]
    data = f[row_idx[mask]]
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_max_sum(f: np.ndarray, M: float, eps: float = 1e-12):
    """
    Solve   max Σ g_j   s.t.   f ⊛ g ≤ M , g ≥ 0 .
    Returns ``g`` (already normalized to sum = 1) and the raw sum before
    normalisation.  If the LP is unbounded or fails, ``None`` is returned.
    """
    n = f.shape[0]
    # Build the constraint matrix once per call.
    A = _build_conv_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n, dtype=float)               # minimise –Σg → maximise Σg
    bounds = [(0, None) for _ in range(n)]

    # HiGHS is fast and can handle sparse matrices.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        # Occasionally the LP is unbounded because f has zero at an edge.
        return None, None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None, None
    g = g / s          # normalise to sum = 1
    return g, s

def _conv_max_and_full(a: np.ndarray):
    """Return (max(conv), full convolution) using FFT for speed."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. ``a``.
    Implements   g_i = 2·Σ_k w_k·a_{k‑i}
    via FFT (convolution of the weight vector with the reversed a).

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # extract the n relevant entries centred around index n‑1
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]

                best_score = cand_j_score
                best_seq = cand_j
                print(f"[{time.time() - start_time:6.2f}s] jitter helped, new score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final Adam refinement (the remaining time budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        n = best_seq.shape[0]
        # very rough estimate: ~1500 Adam iterations per second for n≈2000
        est_iters = int(min(300000, max(50000, remaining * 1500)))
        print(f"[{time.time() - start_time:6.2f}s] Final Adam refinement – "
              f"n={n}, iters≈{est_iters}")

        _, final_best_a, final_best_score = _adam_optimize(
            best_seq.copy(),
            max_iters=est_iters,
            deadline=deadline,
            alpha=0.12,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            temp_start=0.02,
            temp_end=0.0002,
        )
        # Convert the Adam‑score (which is 2·n·max) to the true score
        # (the same expression, because our vectors are already normalised).
        if final_best_score < best_score:
            best_score = final_best_score
            best_seq = final_best_a
            print(f"[{time.time() - start_time:6.2f}s] Adam gave improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. post‑process and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
