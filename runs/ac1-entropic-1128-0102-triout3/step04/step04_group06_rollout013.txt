def conv_fft(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv

"""
Propose a candidate height sequence using fast FFT‑based sub‑gradient descent
on the simplex, occasional random perturbations, and a final stochastic
local‑search phase.  The sub‑gradient of the max‑convolution is obtained via
FFT (no Python loops), which allows many more iterations than the previous
implementation and typically yields a lower evaluation score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of two real vectors using a real FFT of length ``fft_len``.
    ``fft_len`` should be a power of two and >= len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    # keep only the valid part
    return conv[: len(x) + len(y) - 1]


def _auto_conv(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a using the appropriate FFT length."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow_two(conv_len)
    return _conv_fft(a, a, fft_len)


def _active_mask(conv: np.ndarray, rel_eps: float = 1e-3,
                 max_add: int = 12) -> np.ndarray:
    """
    Return a mask (vector of 0/1) of size ``len(conv)`` that marks the
    “active’’ indices driving the sub‑gradient.  All positions whose value
    lies within ``rel_eps`` of the maximum are kept; if fewer than
    ``max_add`` such positions exist the globally largest entries are added.
    """
    max_val = conv.max()
    eps = max_val * rel_eps
    mask = np.zeros_like(conv, dtype=float)
    active = np.where(conv >= max_val - eps)[0]
    if active.size < max_add:
        # take the overall top–max_add entries
        topk = min(max_add, conv.size)
        active = np.unique(np.argpartition(conv, -topk)[-topk:])
    mask[active] = 1.0 / active.size   # uniform weight, normalised
    return mask


def _gradient_via_fft(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv(a,a))`` using the mask that marks the
    active convolution indices.  The gradient is obtained by convolving the
    reversed ``a`` with the mask (both via FFT) and extracting the middle
    slice.  The formula is

        g_i = 2 * sum_{k active} a_{k-i}   (zero padding outside range).

    Returns a vector of length ``len(a)``.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # Length of the convolution a_rev * mask is n + (2n-1) - 1 = 3n-2
    conv_len = 3 * n - 2
    fft_len = _next_pow_two(conv_len)
    conv_rev = _conv_fft(a_rev, mask, fft_len)
    # Extract the slice that corresponds to indices i = 0..n-1
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _score_from_max(n: int, max_val: float) -> float:
    """Objective value when the sum of the sequence is normalised to 1."""
    return 2.0 * n * max_val

    print(f"[{time.time() - start_time:6.2f}s] start: n={target_n}, "
          f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. FFT‑based exponentiated‑gradient descent ----------------------
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.8                # initial step‑size multiplier
    rel_eps = 1e-3            # tolerance for the active set
    max_add = 12              # at most this many active indices
    floor = 1e-12             # minimum entry to keep numerical stability

    while time.time() < deadline:
        it += 1

        # Full auto‑convolution and current objective.
        conv = _auto_conv(a)
        max_val = conv.max()
        cur_score = _score_from_max(target_n, max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d} "
                  f"improved score={cur_score:.6f}")

        # --- compute sub‑gradient for the active indices -----------------
        mask = _active_mask(conv, rel_eps=rel_eps, max_add=max_add)
        g = _gradient_via_fft(a, mask)

        # --- exponentiated (mirror) update on the simplex ---------------
        eta = eta0 / math.sqrt(it)      # diminishing step size
        # safeguard against too‑large updates
        g = np.clip(g, 0.0, 1e3)
        a = a * np.exp(-eta * g)        # multiplicative update (minimise)
        a = np.maximum(a, floor)        # enforce a floor before renorm
        a /= a.sum()

        # --- occasional random perturbation to escape plateaus ----------
        if it % 3000 == 0:
            i, j = np.random.randint(0, target_n, size=2)
            eps = np.random.rand() * 0.04        # up to 4 % of source mass
            delta = eps * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climbing --------------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = target_n
        conv, max_val = _auto_conv(a), None
        cur_score = _score_from_max(n, conv.max())

        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = np.random.rand() * 0.05            # up to 5 % shift
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # keep feasibility
            if a_new[i] < 0:
                continue
            conv_new = _auto_conv(a_new)
            new_score = _score_from_max(n, conv_new.max())
            if new_score < cur_score:
                cur_score = new_score
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"local improve score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence --------------------------------------
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score="
          f"{best_score:.6f}")
    # Convert to plain Python floats for the caller.
    return [float(v) for v in best_seq]
