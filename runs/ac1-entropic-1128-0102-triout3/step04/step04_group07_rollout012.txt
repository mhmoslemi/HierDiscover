def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start=time.time()
    deadline=start+budget_s-0.2
    n=1500
    target_score=1.503
    best_seq=None
    best_score=math.inf

    # Phase 1: random supports of varying size m
    support_sizes = list(range(1300, 400, -100))
    attempts_per_size = 5000  # many attempts
    for m in support_sizes:
        if time.time() > deadline: break
        for attempt in range(attempts_per_size):
            if time.time()>deadline: break
            S = np.random.choice(n, size=m, replace=False)
            a=np.zeros(n)
            a[S]=1.0/m
            val=fast_score(a)
            if val < best_score:
                best_score=val; best_seq=a.copy()
                print(f"[random] m={m} attempt={attempt+1} score={val:.6f}")
                if val <= target_score: return best_seq.tolist()
        print(f"[phase1] after m={m}, best={best_score}")

    # Phase 2: local search on best support size
    # Determine best support from best_seq (nonzero entries)
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_score = fast_score(best_seq)

    # Determine support indices:
    supp = np.where(best_seq>0)[0].tolist()
    m = len(supp)
    # We'll keep same support size; loops using swapping neighbors
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # random swap
        i_out = np.random.choice(supp)
        # choose a candidate not in support
        i_in = np.random.randint(0, n)
        while i_in in supp:  # naive; might be inefficient. Use set
            i_in = np.random.randint(0, n)
        # create new support
        new_supp = supp.copy()
        # replace i_out with i_in
        new_supp.remove(i_out)
        new_supp.append(i_in)
        # create array and evaluate
        a = np.zeros(n)
        a[new_supp]=1.0/m
        val = fast_score(a)
        if val < best_score:
            best_score=val; best_seq=a.copy(); supp = new_supp
            print(f"[local] iter={iteration} new best {val:.6f}")
            if val <= target_score:
                return best_seq.tolist()
        # else maybe accept with small prob if val slightly worse
        # optional acceptance schedule.
    # fallback: use subgradient optimizer from previous
    # (can reuse code)
    # final return
    return best_seq.tolist()

"""
Propose a low‑autocorrelation step function.

The algorithm consists of three stages:

1. **Heavy random search** – many random supports of various sizes are
   sampled and evaluated (uniform weight 1/m).  The best support found so
   far is kept.

2. **Local swap optimisation** – starting from the best support,
   repeatedly try to swap a present index with a missing one.
   The convolution (ordered‑pair counts) is updated incrementally,
   avoiding full FFT recomputation.  Swaps that improve the score are
   accepted; after a long plateau a random restart is performed.

3. **Fallback continuous optimisation** – the current best vector (uniform
   on its support) is refined by the projected sub‑gradient method that
   was used in the reference implementation.

The script respects the 1000 s budget, prints progress, and returns a
list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# Fast evaluation for a binary mask (uniform weight 1/m)
def fast_score_mask(mask: np.ndarray, m: int) -> float:
    """Return the evaluation score for a binary mask with uniform weights."""
    n = mask.shape[0]
    # convolution of the 0/1 mask with itself gives ordered‑pair counts
    conv = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
    maxc = int(conv.max())
    return float(2.0 * n * maxc / (m * m))

# Projected sub‑gradient optimiser (fallback)
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent on the original evaluate_sequence."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score_mask(a > 0, int(np.round(a.sum() * len(a))))  # cheap upper bound
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # --- active convolution index (where max occurs) ---
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # --- sub‑gradient of the active entry ---
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # --- step size and projection onto simplex ---
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex {x≥0, sum x = 1}
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # --- occasional random mass transfer (helps escape plateaux) ---
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta

                # projection again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)

                cand_val = fast_score_mask(cand > 0,
                                            int(np.round(cand.sum() * len(cand))))
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # --- logging ---
        if it % 2000 == 0:
            cur_val = fast_score_mask(a > 0,
                                      int(np.round(a.sum() * len(a))))
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

# Local swap optimisation (incremental convolution updates)
def local_swap_search(mask: np.ndarray,
                     m: int,
                     best_score: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing by swapping a present index with a missing one.
    Only swaps that improve the score are kept.
    """
    n = mask.shape[0]
    # initial convolution (ordered‑pair counts)
    conv = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
    support = np.flatnonzero(mask).astype(np.int32)

    best_mask = mask.copy()
    iter_no = 0
    since_improve = 0
    while time.time() < deadline:
        iter_no += 1
        # pick a random element to remove
        out_idx = int(np.random.choice(support))
        # pick a random element to insert
        in_idx = int(np.random.randint(0, n))
        while mask[in_idx]:
            in_idx = int(np.random.randint(0, n))

        # create a tentative new convolution
        new_conv = conv.copy()

        # --- remove out_idx ---
        sums_out = out_idx + support               # includes self‑pair
        new_conv[sums_out] -= 2
        new_conv[2 * out_idx] += 1                  # correct self‑pair subtraction

        # support after removal (size m‑1)
        new_support = support[support != out_idx]

        # --- add in_idx ---
        sums_in = in_idx + new_support
        new_conv[sums_in] += 2
        new_conv[2 * in_idx] += 1

        # evaluate new score
        maxc = int(new_conv.max())
        new_score = float(2.0 * n * maxc / (m * m))

        if new_score < best_score:
            # accept the swap
            best_score = new_score
            best_mask = np.zeros_like(mask)
            best_mask[in_idx] = True
            best_mask[new_support] = True

            conv = new_conv
            support = np.append(new_support, in_idx).astype(np.int32)

            since_improve = 0
            print(f"[local] iter {iter_no} improved to {new_score:.6f}")
            if new_score <= 1.503:
                break
        else:
            since_improve += 1

        # random restart after a long plateau
        if since_improve >= 20000:
            # fresh random support of the same size
            new_mask = np.zeros(n, dtype=bool)
            idx = np.random.choice(n, size=m, replace=False)
            new_mask[idx] = True
            mask = new_mask
            conv = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
            support = np.flatnonzero(mask).astype(np.int32)
            since_improve = 0
            # recompute score for the new random state
            maxc = int(conv.max())
            best_score = float(2.0 * n * maxc / (m * m))
            best_mask = mask.copy()
            print(f"[local] restart after {iter_no} iterations, new score {best_score:.6f}")

    return best_mask, best_score

        max_attempts = 8000
        while attempts < max_attempts and time.time() < deadline:
            attempts += 1
            mask = np.zeros(n, dtype=bool)
            idx = np.random.choice(n, size=m, replace=False)
            mask[idx] = True
            val = fast_score_mask(mask, m)
            if val < best_score:
                best_score = val
                best_mask = mask.copy()
                print(f"[rand] m={m} att={attempts} score={val:.6f}")
                if val <= target:
                    a = np.zeros(n, dtype=np.float64)
                    a[best_mask] = 1.0 / m
                    return a.tolist()
        print(f"[phase1] size {m} finished, best so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. local swap optimisation
    if best_mask is None:
        # fall back to uniform start
        best_mask = np.full(n, True, dtype=bool)
        m = n
        best_score = fast_score_mask(best_mask, m)
    else:
        m = int(best_mask.sum())

    # leave a few seconds for the final fallback optimiser
    local_deadline = deadline - 5.0
    if time.time() < local_deadline and m > 0:
        best_mask, best_score = local_swap_search(best_mask, m,
                                                  best_score,
                                                  local_deadline)

    # ------------------------------------------------------------------
    # 3. fallback – continuous/sub‑gradient optimisation
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_init = np.zeros(n, dtype=np.float64)
            a_init[best_mask] = 1.0 / m
            print(f"[fallback] running sub‑gradient optimiser (remaining {remaining:.1f}s)")
            a_opt = subgradient_optimize(a_init, remaining)
            final_val = fast_score_mask(a_opt > 0,
                                        int(np.round(a_opt.sum() * len(a_opt))))
            if final_val < best_score:
                best_score = final_val
                best_seq = a_opt
                print(f"[fallback] improved to {best_score:.6f}")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 4. final construction (uniform on the best mask)
    a_final = np.zeros(n, dtype=np.float64)
    if best_mask is not None and m > 0:
        a_final[best_mask] = 1.0 / m
    else:
        a_final = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[final] returning sequence with score {best_score:.6f}")
    return a_final.tolist()
