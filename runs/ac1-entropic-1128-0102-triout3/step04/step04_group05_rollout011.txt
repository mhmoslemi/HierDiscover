import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution (real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k * conv_k  w.r.t. `a`.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of the objective."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best score seen during the run (2·n·max_conv)
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # evaluate current point
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update on the Euclidean space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back to the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that uses the *hard* max
    of the convolution (sub‑gradient of the L∞‑norm).  The update rule is

        a_i ← a_i·exp(−η·g_i)    ,   g = 2·(a * indicator_of_max)

    followed by normalisation to keep Σa_i = 1.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of the (approximately) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the exact max‑norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # decreasing step size (geometric schedule)
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update, with clipping to avoid overflow/underflow
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # exp(±30) ≈ 1e13, safe for double
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises

          score = 2·n·max(conv(a,a)) / (sum(a)²)

    by a multi‑scale combination of Adam (soft‑max) and a hard‑max
    exponential‑gradient refinement.  The function stops before ``budget_s``
    seconds and returns the best sequence found (as a plain list of floats).

    # helpers to build candidate vectors
    # ------------------------------------------------------------------

        s = v.sum()

            return np.full_like(v, 1.0 / v.shape[0])

    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    sizes = []
    n = 1200
    MAX_N = 12000                 # upper bound that still allows many iterations
    while n <= MAX_N:
        sizes.append(n)
        n = int(n * 1.5)

    # ------------------------------------------------------------------
    # main optimisation loop over the size schedule
    # ------------------------------------------------------------------
    for stage, n in enumerate(sizes, 1):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Stage {stage}/{len(sizes)} – n = {n}")

        # --------------------------------------------------------------
        # build a small pool of starting points for this size
        # --------------------------------------------------------------
        pool = []

        # up‑sample the current global best (if it exists)
        if best_seq is not None and best_seq.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, best_seq)
            pool.append(_as_simplex(up))

        # include the previous construction (if any)
        if prev_seq is not None:
            if prev_seq.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, prev_seq.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, prev_seq)
                pool.append(_as_simplex(up))
            else:
                pool.append(prev_seq.copy())

        # deterministic shapes
        pool.append(_as_simplex(np.full(n, 1.0 / n)))                 # uniform
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)                         # symmetric triangle
        pool.append(_as_simplex(tri))
        if n > 1:
            hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))   # raised‑cosine
            pool.append(_as_simplex(hann))

        # a few random Dirichlet draws
        for _ in range(2):
            pool.append(_as_simplex(np.random.dirichlet(np.ones(n))))

        # random sparse spikes (3‑5 non‑zero entries)
        for _ in range(2):
            spikes = np.zeros(n)
            k = np.random.randint(3, 6)               # number of spikes
            pos = np.random.choice(n, size=k, replace=False)
            spikes[pos] = np.random.rand(k) + 0.1      # avoid exact zero
            pool.append(_as_simplex(spikes))

        # --------------------------------------------------------------
        # optimise each candidate
        # --------------------------------------------------------------
        for cid, cand in enumerate(pool, 1):
            if time.time() >= deadline:
                break

            # estimate remaining time and allocate a reasonable number of iters
            remaining = deadline - time.time()
            # rough cost per iteration grows like n·log n; use a simple model
            est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
            max_iters = int(min(180000, max(12000, remaining / est_iter_time)))
            if max_iters <= 0:
                break

            print(f"   [{time.time() - start_time:5.2f}s] cand {cid}/{len(pool)} – max_iters={max_iters}")

            # 1) short Adam refinement (smooth‑max)
            a_adam, score_adam = _adam_optimize(
                cand.copy(),
                max_iters=int(0.45 * max_iters),
                deadline=deadline,
                alpha=0.15,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.07,
                temp_end=0.001,
            )
            if score_adam < best_score:
                best_score = score_adam
                best_seq = a_adam.copy()
                print(f"       NEW GLOBAL BEST (after Adam) – score = {best_score:.6f}")

            # 2) exponential‑gradient (hard‑max) refinement
            a_exp, score_exp = _exp_gradient_optimize(
                a_adam.copy(),
                max_iters=int(0.55 * max_iters),
                deadline=deadline,
                eta_start=0.8,
                eta_end=0.001,
            )
            if score_exp < best_score:
                best_score = score_exp
                best_seq = a_exp.copy()
                print(f"       NEW GLOBAL BEST (after EG) – score = {best_score:.6f}")

        # --------------------------------------------------------------
        # quick extra refinement of the current best at this size
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
            extra_iters = int(min(50000, remaining / est_iter_time))
            if extra_iters > 0:
                print(f"   [{time.time() - start_time:5.2f}s] extra EG refinement (n={n}) – iters={extra_iters}")
                a_ref, score_ref = _exp_gradient_optimize(
                    best_seq.copy(),
                    max_iters=extra_iters,
                    deadline=deadline,
                    eta_start=0.4,
                    eta_end=0.0002,
                )
                if score_ref < best_score:
                    best_score = score_ref
                    best_seq = a_ref.copy()
                    print(f"       IMPROVED AFTER EXTRA EG – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final long refinement on the overall best sequence (hard‑max EG)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(400000, max(80000, remaining / est_iter_time)))
        if final_iters > 0:
            print(f"[{time.time() - start_time:6.2f}s] Final EG refinement – n={n}, iters={final_iters}")
            a_final, score_final = _exp_gradient_optimize(
                best_seq.copy(),
                max_iters=final_iters,
                deadline=deadline,
                eta_start=0.2,
                eta_end=5e-4,
            )
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"   FINAL IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    # ------------------------------------------------------------------

        best_seq = np.full(200, 1.0 / 200)

    # ensure non‑negativity and normalisation before returning

    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    return [float(x) for x in best_seq]

This implementation respects the time budget, uses both Adam (soft‑max) and a hard‑max exponential‑gradient phase, adds diverse initial points (including spikes and up‑sampled previous results), and performs a final long refinement. The hard‑max EG stage focuses directly on the anti‑diagonals that attain the current maximum, which typically yields a tighter reduction of the `2·n·max(conv)` score, helping to breach the target bound of 1.503.
