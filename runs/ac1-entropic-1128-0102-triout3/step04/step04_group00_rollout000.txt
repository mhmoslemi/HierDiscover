def _run_adam(a0, n, deadline, start_time, step0=0.2, beta1=0.9, beta2=0.999, epsilon=1e-8, max_iter=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}")
        grad = _subgradient(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 / math.sqrt(t)   # decayed base step
        update = step * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - update
        a = _project_simplex(a)
    return best_a, best_score
```

def _refine_pairwise(a0, n, deadline, start_time, max_moves=200000):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    rng = np.random.default_rng()
    for move in range(max_moves):
        if time.time()>deadline:
            break
        # compute gradient
        conv = _conv_self(a)
        grad = _subgradient(a, conv)
        # indices
        i = np.argmax(grad)
        j = np.argmin(grad)
        if a[i] < 1e-12:
            continue
        eps = min(a[i]*0.2, 1e-6)   # maybe dynamic: use fraction
        # or random epsilon up to 30% as before
        eps = a[i] * rng.random() * 0.3
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # renorm to ensure sum remains 1 (already done)
        # Accept if better
        new_score = 2.0 * n * _conv_self(a_new).max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm overview
------------------
1. **Simplex projection** – Euclidean projection onto {x≥0, Σx=1}.
2. **Convolution** – linear self‑convolution using a real FFT.
3. **Sub‑gradient** – exact sub‑gradient of `max(conv(a))` obtained from
   all indices whose value is within a tiny relative tolerance of the
   maximum.
4. **Adam‑style projected descent** – adaptive per‑coordinate step sizes
   (β₁=0.9, β₂=0.999) with a 1/√t decay.  After each update the vector is
   re‑projected onto the simplex.
5. **Multi‑stage schedule**  
   *Stage 0* – a few short Adam runs on a moderate size (n≈1500) from
   random restarts (and the optional starter).  
   *Stage 1* – up‑sample the best vector to the target size (n≈4800) and
   run a long Adam optimisation using the remaining budget.  
   *Stage 2* – greedy pairwise mass‑transfer refinement guided by the
   current sub‑gradient (hill‑climbing with optional acceptance).
6. **Return** – the best vector found, normalised to sum = 1 and
   converted to a plain Python list.

The routine respects the supplied time budget (default 1000 s).  All
helper functions are defined at module level and contain no closures.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are taken as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # contribution of a_i * a_j term to ∂/∂a_i and ∂/∂a_j
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              step0: float = 0.25,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with simplex projection.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # bias‑correction accumulators
    b1t = beta1
    b2t = beta2

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient (we want to descend, so later subtract it)
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)

        # learning rate decays like 1/√t
        step = step0 / math.sqrt(it)

        # update (note the minus sign – we minimise)
        delta = step * m_hat / (np.sqrt(v_hat) + eps)
        a = a - delta

        # projection back to the simplex
        a = _project_simplex(a)

        # update bias‑correction factors for next iteration
        b1t *= beta1
        b2t *= beta2

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer refinement (greedy hill‑climbing)
# ----------------------------------------------------------------------
def _run_pairwise_refine(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Greedy refinement: repeatedly transfer a fraction of mass from the
    index with the largest gradient to the index with the smallest.
    Accept moves only if they improve the objective.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    rng = np.random.default_rng()
    move = 0

    while move < max_moves and time.time() < deadline:
        move += 1

        # recompute gradient for the current a
        conv = _conv_self(a)
        grad = _subgradient_full(a, conv)

        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))

        if a[i_src] < 1e-12 or i_src == i_dst:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        eps = a[i_src] * rng.random() * 0.30
        a_new = a.copy()
        a_new[i_src] -= eps
        a_new[i_dst] += eps

        # evaluate
        new_score = 2.0 * n * _conv_self(a_new).max()

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] pairwise improve: n={n}, move={move:,}, score={new_score:.6f}")

        # greedy acceptance
        if new_score < 2.0 * n * conv.max():
            a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
