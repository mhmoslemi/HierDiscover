def _run_descent(a0, n, deadline, start_time):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.6
    lam_factor = 0.5
    lam_decay = 0.7
    lam_decay_interval = 2000
    it = 0
    # Define thresholds for switching to exact gradient
    lam_switch_iter = 20000   # after many iterations
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] descend improve n={n} it={it} score={score:.6f}")

        # update lambda
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-14)

        if it < lam_switch_iter:
            grad = _gradient_from_conv(a, conv, lam)
        else:
            # exact max gradient
            k = conv.argmax()
            grad = _grad_max_at(a, k)   # implement separate function
        # add occasional noise
        if it % 1000 == 0:
            grad += 0.01 * np.random.randn(n)

        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)

        # gradient descent
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a low‑maximum self‑convolution step‑function.

The algorithm works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It combines three stages:

1. **Projected (smooth) gradient descent** – a log‑sum‑exp surrogate for the
   ℓ∞‑norm of the convolution is minimized.  The smoothing parameter λ is
   reduced steadily; after a while the exact sub‑gradient of the true maximum
   is used.
2. **Targeted simulated annealing** – cheap mass‑transfer moves (ε from a
   source index to a destination index) are performed with a geometric
   temperature schedule to escape shallow plateaus.
3. **Multi‑scale warm‑starts** – the best distribution found for one length
   is linearly up‑sampled and used as a warm start for the next, larger,
   length.

The routine respects a total time budget (default 1000 s) and prints
intermediate scores.  It returns a plain Python list of non‑negative heights.

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT (real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate for max(conv).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max over convolution entries
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of the weight vector with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


def _grad_max_at(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) at the entry k (the current argmax).

    For each i, d conv_k / d a_i = 2 * a_{k-i} if k-i in range,
    except when i = k-i (i.e. k even and i = k//2) where the factor is 1.
    """
    n = a.shape[0]
    grad = np.zeros(n, dtype=float)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    if k % 2 == 0:
        c = k // 2
        if 0 <= c < n:
            grad[c] = a[c]                     # correct the double count
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Core optimisation phases

def _run_descent(a0: np.ndarray,
                 n: int,
                 slice_deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a smooth surrogate followed by an exact
    sub‑gradient phase.  Returns the best vector encountered and its exact
    objective value (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')

    beta = 0.9                     # momentum decay
    step0 = 0.6                    # initial step magnitude
    lam_factor = 0.5               # initial smoothing factor (relative)
    lam_decay = 0.7
    lam_decay_interval = 2500
    switch_iter = 20000           # iteration after which we use exact gradient
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        score = 2.0 * n * max_b      # exact objective (sum(a)=1)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] descent improve n={n} it={it} score={score:.6f}")

        # update smoothing λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-14)

        # choose gradient
        if it < switch_iter:
            grad = _gradient_from_conv(a, conv, lam)
        else:
            k = conv.argmax()
            grad = _grad_max_at(a, k)

        # occasional isotropic noise
        if it % 800 == 0:
            grad += 0.02 / math.sqrt(it) * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # projected gradient step
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 3e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer simulated annealing.
    Moves a small ε from a source index to a destination index.
    Returns the (possibly) improved sequence and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue
        eps = a[i] * np.random.rand() * 0.5   # move up to 50 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start modest and climb up; shuffled to avoid bias.
    lengths = [2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Slice a fair share of the remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # later (larger) lengths get a slightly larger share
        slice_frac = max(0.07, remaining * (0.25 / (len(lengths) - idx)))
        slice_deadline = time.time() + max(8.0, remaining * 0.3)

        # --------------------------------------------------------------
        # Smoothed + exact gradient descent
        # --------------------------------------------------------------
        a_opt, score_opt = _run_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after descent n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Optional annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=3e-2,
                end_temp=1e-6,
                steps=200000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256.0)

    # Return plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
