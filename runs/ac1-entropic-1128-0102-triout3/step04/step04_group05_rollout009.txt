def solve_lp_max_sum(f, M):

    # Build matrix A using Toeplitz-like operation
    # For speed, we can use np.zeros((2*n-1, n))

    fft_len = 1 << ((2*n-1).bit_length())

def upsample(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return max(conv(a,a)) and the full convolution (real).
    `a` is assumed to sum to 1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of a weighted max‑convolution objective w.r.t. `a`.
    Implements g_i = 2 * Σ_k w_k * a_{k-i} using FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length while keeping a probability vector."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq = new_seq / s
    return new_seq
