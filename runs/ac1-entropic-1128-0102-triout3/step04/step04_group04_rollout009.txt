def mirror_step(a, g, eta):
    log_a = np.log(np.maximum(a, 1e-15))
    log_a -= eta * g
    # shift for numeric stability
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new = a_new / a_new.sum()
    return a_new

def _weighted_gradient(a, conv, max_val, alpha):
    n = a.shape[0]
    # compute weight vector w of length 2n-1
    diff = max_val - conv  # nonnegative
    # to avoid underflow for large diff * alpha, we can subtract max but it's fine.
    # Use float64
    w_raw = np.exp(-alpha * diff)
    # Normalize
    total = w_raw.sum()
    if total == 0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / total
    # compute correlation via FFT
    m = w.size
    N = 1 << ((n + m - 1).bit_length())
    Af = np.fft.rfft(a, n=N)
    # reverse w
    w_rev = w[::-1]
    W_rev_f = np.fft.rfft(w_rev, n=N)
    conv_rev = np.fft.irfft(Af * W_rev_f, n=N)[: n + m - 1]
    # compute gradient
    base = m - 1  # = 2n-2
    g = 2.0 * conv_rev[base - np.arange(n)]
    return g

"""
Propose a step‑function (non‑negative vector) with a low `evaluate_sequence`
score.

Algorithm
---------
1. **Initialisation** – start from the previously found construction
   (`height_sequence_1`) if it exists, otherwise use a simple power‑law shape.
   The vector is normalised to sum = 1.

2. **Smoothed‑gradient descent on the simplex** – at each iteration
   * compute the auto‑convolution `c = a*a` and its maximum `M`;
   * build a smooth weight vector `w_k ∝ exp(-α·(M‑c_k))` (α grows slowly);
   * the sub‑gradient of the smoothed `max` is  
        `g_i = 2·∑_k w_k·a_{k‑i}` (a correlation that is obtained via FFT);
   * perform a mirror‑descent step on the probability simplex  
        `a_i ← a_i·exp(-η·g_i)`, renormalise, with a step‑size `η≈0.6/√t`.

3. **Exploration** – every few hundred iterations
   * try a cheap random mass‑transfer move;
   * occasionally try a full random permutation of the coordinates.
   Accepted moves that improve the objective are kept.

4. **Final cheap hill‑climbing** – after the time budget is almost exhausted,
   perform small random swaps until the deadline.

The whole search respects a ~1000 s wall‑clock limit and prints progress.
The returned sequence is already normalised; `evaluate_sequence` will
handle the final scaling and security checks.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# --------------------------------------------------------------

def _conv_and_max(a: np.ndarray):
    """
    Compute the full auto‑convolution a*a (linear, not circular) and its
    maximum entry.  Returns (conv, max(conv)).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def _weighted_gradient(a: np.ndarray, conv: np.ndarray,
                       max_val: float, alpha: float) -> np.ndarray:
    """
    Smoothed sub‑gradient of `max(conv)`.

    w_k ∝ exp(-α·(max‑conv_k))   (weights sum to 1)
    g_i = 2·∑_k w_k·a_{k‑i}      (correlation of a with w)

    The correlation is computed by FFT (linear, zero‑padded).
    """
    n = a.shape[0]
    # ----- weight vector -------------------------------------------------
    diff = max_val - conv                     # non‑negative
    w_raw = np.exp(-alpha * diff)             # may under‑flow for large α
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum

    # ----- correlation a ⋆ w (linear, lag 0…n‑1) -------------------------
    m = w.size               # = 2·n‑1
    N = 1 << ((n + m - 1).bit_length())   # enough for linear correlation
    Af = np.fft.rfft(a, n=N)
    # correlation = ifft(conj(Af) * Wf)  where Wf = fft(w)
    Wf = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(np.conj(Af) * Wf, n=N)[: n + m - 1]   # length 3·n‑2

    # For lag i (0 ≤ i < n) we need Σ_t a[t]·w[t+i] = corr[i]
    # The gradient uses exactly this correlation.
    g = 2.0 * corr[:n]   # keep only the first n lags (i = 0…n‑1)
    return g


def _mirror_step(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) update on the probability simplex:
        a_i ← a_i·exp(-η·g_i)
    followed by renormalisation.
    """
    # protect against zero entries
    log_a = np.log(np.maximum(a, 1e-15))
    log_a -= eta * g
    # numerical stabilisation
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


def _objective(a: np.ndarray) -> float:
    """
    Objective used during the search (sum(a)=1 is assumed):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv, mx = _conv_and_max(a)
    return 2.0 * n * mx

# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Load a feasible starting vector
    # ------------------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            raise ValueError
        a0 /= a0.sum()
        n = a0.shape[0]
        best_seq = a0.copy()
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
              f"(n={n}) → score={best_score:.6f}")
    except Exception:
        # fallback – simple power‑law shape (length 5000)
        n = 5000
        i = np.arange(1, n + 1, dtype=float)
        p = 0.55                     # a modest exponent that already works OK
        a0 = i ** (-p)
        a0 /= a0.sum()
        best_seq = a0.copy()
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Init with power‑law "
              f"(n={n}) → score={best_score:.6f}")

    current = best_seq.copy()
    it = 0

    # ------------------------------------------------------------------
    # 2. Smoothed‑gradient descent + occasional exploration
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # current objective
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:

        alpha = 5.0 + 0.02 * math.log1p(it) * 5.0
        g = _weighted_gradient(current, conv, max_val, alpha)

        # ---- mirror‑descent step -----------------------------------------------
        eta = 0.6 / math.sqrt(it)            # decaying step size
        cand = _mirror_step(current, g, eta)

        # evaluate candidate

                if tmp_score < best_score - 1e-12:

                    best_seq = tmp.copy()
                    current = tmp
                    print(f"[{time.time() - start_time:6.2f}s] Random transfer "

        if it % 2000 == 0:
            perm = np.random.permutation(n)
            tmp = current[perm]
            conv_tmp, max_tmp = _conv_and_max(tmp)
            tmp_score = 2.0 * n * max_tmp
            if tmp_score < best_score - 1e-12:
                best_score = tmp_score
                best_seq = tmp.copy()
                current = tmp
                print(f"[{time.time() - start_time:6.2f}s] Permutation improve "
                      f"→ score={tmp_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap hill‑climbing (small swaps) until the deadline hits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or current[i] == 0.0:
            continue
        delta = current[i] * 0.02 * np.random.rand()
        tmp = current.copy()
        tmp[i] -= delta
        tmp[j] += delta
        conv_tmp, max_tmp = _conv_and_max(tmp)
        tmp_score = 2.0 * n * max_tmp
        if tmp_score < best_score - 1e-12:
            best_score = tmp_score
            best_seq = tmp.copy()
            current = tmp
            print(f"[{time.time() - start_time:6.2f}s] Final hill‑climb "
                  f"→ score={tmp_score:.6f}")

    # 4. Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        # safety fallback – uniform distribution of size 100
        best_seq = np.full(100, 1.0 / 100.0)

    return [float(x) for x in best_seq]
