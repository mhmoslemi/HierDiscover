    # Make non-negative and normalise to sum 1.
    init = np.maximum(init, 0.0)
    init_sum = init.sum()

"""
Propose a candidate step‑function using a soft‑max (log‑sum‑exp) surrogate for the
max‑convolution, combined with an exponentiated‑gradient (mirror‑descent) update
on the simplex.  The surrogate temperature is annealed from a smooth to a sharp
approximation, allowing the optimiser to first explore widely and then focus on
the true L∞ objective.  After the gradient phase a cheap stochastic hill‑climb
finishes the search.  The routine runs for at most ``budget_s`` seconds and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """
    Full auto‑convolution of a probability vector a (length n) using FFT.
    Returns the convolution (length 2·n‑1) and its maximum entry.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _gradient_softmax(a: np.ndarray, w: np.ndarray):
    """
    Gradient of the soft‑max weighted convolution objective.
    For a probability vector a (length n) and a weight vector w (length 2·n‑1)
    the i‑th component of the sub‑gradient is

        g_i = 2·∑_k w_k·a_{k‑i}      (indices out of range are ignored)

    This can be written as a cross‑correlation and computed with a single FFT.
    """
    n = a.shape[0]
    # reverse w to turn the cross‑correlation into a convolution
    rev_w = w[::-1]
    # length of the full convolution a * rev_w ( = 3·n‑2 )
    conv_len = n + rev_w.shape[0] - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    A_f = np.fft.rfft(a, n=fft_len)
    W_f = np.fft.rfft(rev_w, n=fft_len)
    full = np.fft.irfft(A_f * W_f, n=fft_len)[:conv_len]
    offset = rev_w.shape[0] - 1          # = 2·n‑2
    # pick the required slice and multiply by 2
    g = 2.0 * full[offset - np.arange(n)]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The algorithm consists of three stages:
      1. initialisation (use the previous best construction if available);
      2. mirror‑descent on a soft‑max surrogate of the L∞ convolution norm;
      3. a final stochastic hill‑climbing phase.
    The best sequence encountered is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 1. initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    target_n = 5000

    try:
        init = np.array(height_sequence_1, dtype=float)   # may not exist
        init = np.maximum(init, 0.0)
        if init.size != target_n:
            # linear interpolation to the required length
            x_old = np.linspace(0.0, 1.0, init.size)
            x_new = np.linspace(0.0, 1.0, target_n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        # fallback: random simplex point
        init = np.random.rand(target_n)

    # Normalise to a probability vector (sum = 1)
    init = np.maximum(init, 0.0)
    tot = init.sum()
    if tot == 0.0:
        a = np.full(target_n, 1.0 / target_n, dtype=float)
    else:
        a = init / tot

    # ------------------------------------------------------------------
    # 2. soft‑max mirror‑descent ---------------------------------------
    # ------------------------------------------------------------------
    # hyper‑parameters
    max_iters = 5_000_000            # hard cap; loop will stop by deadline
    eta0 = 0.7                       # base step size for the exponentiated update
    alpha_start = 1.0                # start temperature for soft‑max
    alpha_end = 30.0                 # final temperature (sharp approximation)

    best_seq = a.copy()
    conv, max_val = _conv_and_max(a)
    best_score = 2.0 * target_n * max_val
    print(f"[{time.time() - start:6.2f}s] start: score={best_score:.6f}")

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and objective
        conv, max_val = _conv_and_max(a)
        cur_score = 2.0 * target_n * max_val

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] iter={it:6d} new best={cur_score:.6f}")

        # anneal the soft‑max temperature
        elapsed = time.time() - start
        frac = min(1.0, elapsed / budget_s)
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # soft‑max weights (numerically stable)
        w_raw = np.exp(alpha * (conv - max_val))   # max entry becomes 1
        w = w_raw / w_raw.sum()                    # normalised probability vector

        # gradient of the surrogate
        g = _gradient_softmax(a, w)

        # exponentiated‑gradient (mirror descent) on the simplex
        eta = eta0 / math.sqrt(it)                 # classical diminishing step
        # keep a strictly positive before taking the logarithm
        a = np.maximum(a, 1e-15)
        log_a = np.log(a) - eta * g
        # numerical stabilisation
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # occasional tiny random mass‑transfer to escape tiny plateaus
        if it % 4000 == 0:
            i, j = np.random.randint(0, target_n, size=2)
            eps = np.random.rand() * 0.025          # up to 2.5 % of source mass
            delta = eps * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 3. final stochastic hill‑climbing --------------------------------
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv, max_val = _conv_and_max(a)
    cur_score = 2.0 * target_n * max_val

    while time.time() < deadline:
        i, j = np.random.randint(0, target_n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.05              # up to 5 % of source mass
        delta = eps * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # guard against negativity caused by rounding
        if a_new[i] < 0:
            continue
        conv_new, max_new = _conv_and_max(a_new)
        new_score = 2.0 * target_n * max_new
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:6.2f}s] hill improve={new_score:.6f}")

    print(f"[{time.time() - start:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
