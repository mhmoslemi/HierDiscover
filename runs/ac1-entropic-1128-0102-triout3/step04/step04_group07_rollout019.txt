def iterative_lp_optimize(init_f, max_iters, time_limit):
    f = init_f.copy()
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    start = time.time()
    for it in range(max_iters):
        # compute current max conv
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # build A matrix
        A = np.empty((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f
        # solve LP
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M* np.ones(2*n-1)]
        obj = cp.Maximize(cp.sum(b))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
        if b.value is None:
            print("LP failed")
            break
        b_val = np.asarray(b.value).flatten()
        # scale to maintain sum = 1 (since our evaluation uses sum=1)
        sum_b = b_val.sum()
        if sum_b == 0:
            break
        # scale to sum 1 (so we remain in same simplex)
        b_scaled = b_val / sum_b
        # update f with step size t
        t = 0.5
        f = (1-t)*f + t*b_scaled
        # reproject onto simplex maybe not needed because convex combination maintains sum=1; also nonnegativity due to b_scaled≥0.
        cur_score = fast_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[iter {it}] new best {best_score:.6f}")
        # break if time exceeded
        if time.time() - start > time_limit:

def project_simplex(v):
    """ Euclidean projection of v onto simplex {x>=0, sum x = 1}. """
    # Use algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for low autoconvolution step functions.
We combine a large random‑support sampling stage with a
smooth‑max gradient descent on the simplex (entropy‑projected
updates).  The gradient of the smoothed objective is
computed via two FFT‑based convolutions, keeping each iteration
fast enough to run thousands of steps within the 1000 s budget.
The algorithm starts from the previously found construction
(if available), explores many random supports, then refines the
best candidate with the gradient descent, finally falling back
to a simple sub‑gradient optimizer if time remains.
"""

import time
import math
import numpy as np

# ----- utilities -------------------------------------------------

def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def random_support_vector(n: int, m: int) -> np.ndarray:
    """Uniform weight on a random support of size m (sum=1)."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a

def random_dirichlet_vector(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet(α) distribution."""
    # generate exponential(1) variates, optionally raise to power 1/conc
    x = np.random.exponential(1.0, size=n)
    if conc != 1.0:
        x = x ** (1.0 / conc)
    a = x / x.sum()
    return a

# ----- smooth‑max gradient descent ------------------------------

def smooth_max_descent(a0: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected gradient descent on a soft‑max approximation of the
    max‑convolution objective.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)

    # pre‑computed FFT lengths
    size_conv = 1 << ((2 * n - 1).bit_length())      # for a*a
    size_tot = 1 << ((3 * n - 2).bit_length())      # for p*a / p*a_rev

    start = time.time()
    it = 0
    eta0 = 0.5          # base step size
    alpha = 20.0        # smoothing parameter (soft‑max temperature)

    while time.time() - start < time_limit:
        # ---- a * a (linear convolution) ---------------------------------
        fft_a = np.fft.rfft(a, size_conv)
        conv_aa = np.fft.irfft(fft_a * fft_a, size_conv)[: 2 * n - 1]

        # ---- soft‑max weights for the max index -------------------------
        max_c = conv_aa.max()
        # compute numerically stable soft‑max
        exp_vals = np.exp(alpha * (conv_aa - max_c))
        p = exp_vals / exp_vals.sum()            # shape (2n‑1,)

        # ---- gradient = 2 * Σ_k p_k * ∂(a*a)_k / ∂a --------------------
        # we need two linear convolutions: p * a and p * rev(a)
        fft_p = np.fft.rfft(p, size_tot)

        conv_pa = np.fft.irfft(fft_p * np.fft.rfft(a, size_tot), size_tot)[: 3 * n - 2]
        conv_par = np.fft.irfft(fft_p * np.fft.rfft(a[::-1], size_tot), size_tot)[: 3 * n - 2]

        grad = 2.0 * (conv_pa[:n] + conv_par[n - 1:n - 1 + n])

        # ---- projected gradient step ------------------------------------
        eta = eta0 / math.sqrt(it + 1.0)
        a_candidate = a - eta * grad
        a = project_simplex(a_candidate)

        # ---- keep the best ------------------------------------------------
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[sgd {it:06d}] new best {best_val:.6f}")

        # ---- cooling schedule for the soft‑max -------------------------
        if it % 2000 == 0 and alpha < 1e5:
            alpha *= 1.2

        it += 1

    return best_a

"""
Improved search for low‑autocorrelation step functions.

The algorithm consists of three phases:

1. **Random‑support / Dirichlet sampling** – many candidates are
   generated (uniform weight on a random support and sparse
   Dirichlet draws).  The best vector found so far is kept.

2. **Smooth‑max gradient descent** – the objective
   `max_k (a*a)[k]` is approximated by a soft‑max
   `log Σ exp(α·(a*a)[k])`.  Its gradient can be written as a
   linear combination of two convolutions, which are computed with
   FFTs.  Each iteration performs a gradient step and projects back
   onto the probability simplex (∑a=1, a≥0).  The smoothing parameter
   `α` is increased slowly, turning the surrogate into the true
   max.  This phase refines the best candidate from step 1.

3. **Fallback sub‑gradient optimizer** – if time permits a short
   sub‑gradient descent (as used in the previous baseline) is run
   to catch any remaining improvement.

All calculations respect the limits of `evaluate_sequence`,
i.e. non‑negative numbers, clipping at 0 / 1000, and scaling‑invariance,
so the returned list can be passed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def random_support_vector(n: int, m: int) -> np.ndarray:
    """Uniform weight on a random support of size m (∑a=1)."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


def random_dirichlet_vector(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet(α) law."""
    x = np.random.exponential(1.0, size=n)
    if conc != 1.0:
        x = x ** (1.0 / conc)
    a = x / x.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (core optimisation)
# ----------------------------------------------------------------------
def smooth_max_descent(a0: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected gradient descent on a soft‑max approximation of the
    `max(a*a)` objective.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)

    # FFT lengths (powers of two)
    size_conv = 1 << ((2 * n - 1).bit_length())      # for a*a
    size_tot = 1 << ((3 * n - 2).bit_length())      # for p*a and p*rev(a)

    start = time.time()
    it = 0
    eta0 = 0.5          # base step size
    alpha = 20.0        # smoothing (soft‑max) parameter

    while time.time() - start < time_limit:
        # a * a  (linear convolution)
        fft_a = np.fft.rfft(a, size_conv)
        conv_aa = np.fft.irfft(fft_a * fft_a, size_conv)[: 2 * n - 1]

        # soft‑max weights over the convolution entries
        max_c = conv_aa.max()
        exp_vals = np.exp(alpha * (conv_aa - max_c))   # stable
        p = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

        # gradients: 2 * Σ_k p_k ∂(a*a)_k / ∂a
        fft_p = np.fft.rfft(p, size_tot)

        conv_pa = np.fft.irfft(fft_p * np.fft.rfft(a, size_tot), size_tot)[: 3 * n - 2]
        conv_par = np.fft.irfft(fft_p * np.fft.rfft(a[::-1], size_tot), size_tot)[: 3 * n - 2]

        grad = 2.0 * (conv_pa[:n] + conv_par[n - 1:n - 1 + n])

        # projected gradient step
        eta = eta0 / math.sqrt(it + 1.0)
        a_candidate = a - eta * grad
        a = project_simplex(a_candidate)

        # keep the best vector seen so far
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[sgd {it:06d}] new best {best_val:.6f}")

        # slowly raise the soft‑max temperature to focus on the true max
        if it % 2000 == 0 and alpha < 1e5:
            alpha *= 1.2

        it += 1

    return best_a

# Fallback sub‑gradient optimiser (short version of the previous baseline)
# ----------------------------------------------------------------------

    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30

    it = 0
    deadline = time.time() + time_limit - 0.1

        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]

        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)

                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")

        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a
