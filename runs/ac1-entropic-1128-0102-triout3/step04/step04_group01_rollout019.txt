def _run_adam(a0, n, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = np.inf
    lam_factor = 0.3   # initial lam scaling factor
    lam_decay = 0.9    # slower decay
    lam_decay_interval = 2000
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve n={n} it={t} score={score:.6f}")
        # update lam
        if t % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)
        # add some isotropic noise occasionally
        if t % 1000 == 0:
            grad += 0.01 * np.random.randn(n)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.01 / (np.sqrt(v_hat) + eps)  # step size decays
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    it = 0
    lam_factor = 0.5   # start scaling factor
    lam_decay = 0.9
    lam_decay_interval = 1500
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")
        # update lambda

            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)

"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑aᵢ = 1).  It
combines three ideas that differ from the previous approach:

1. **Adam‑style projected gradient descent on a log‑sum‑exp surrogate** – 
   the ℓ∞‐norm of the convolution is approximated by a smooth
   soft‑max; Adam’s adaptive step‑size speeds up convergence.
2. **Multi‑scale warm‑starts** – after a length has been optimised,
   its best distribution is linearly interpolated (upsampled) and used
   as the seed for a larger length, allowing the extra degrees of
   freedom to be exploited gradually.
3. **Intensive mass‑transfer simulated annealing** – after each gradient
   phase a cheap annealer moves a small ε of mass from one index to
   another and accepts moves with a temperature schedule that cools
   from ≈5·10⁻² to ≈10⁻⁶.  This polishes the current best vector.

The search is budget‑aware (default 1000 s).  It tries several random
restarts per length, keeps the globally best vector and finally returns
a plain Python list of non‑negative heights ready for evaluate_sequence().
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights ----------------------------------------------------
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # gradient = 2 * (weights *reverse(a)) convolved --------------------
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      slice_deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the log‑sum‑exp surrogate.
    Returns the best vector found and its exact objective value
    (2·n·max(conv) because the vector stays on the simplex).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    lam_factor = 0.35          # initial smoothing relative to current max
    lam_decay = 0.85
    lam_decay_interval = 1500

    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update smoothing parameter (never let it become zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise – helps to escape shallow plateaus
        if it % 2000 == 0:
            grad += 0.02 * np.random.randn(n)

        # Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.05 / (np.sqrt(v_hat) + eps)   # base step size 0.05

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Moves a small ε from a source index to a destination index.
    Returns the (possibly) improved sequence and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
