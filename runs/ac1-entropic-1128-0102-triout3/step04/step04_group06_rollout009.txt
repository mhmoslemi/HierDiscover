"""
Propose a candidate step‑function using an annealed “soft‑max” mirror descent.
The method keeps the vector on the probability simplex via an exponentiated
(update = exp(log a – η·∇)) but replaces the crude sub‑gradient (active‑set)
by the gradient of a smooth approximation of the max‑convolution:
        F(a) = (2·n/τ)·log Σ_k exp(τ·conv_k(a,a)).
The temperature τ grows with the iteration count, so early iterations follow
a gentle landscape and later iterations focus on the true max.
Occasional tiny random mass‑re‑allocations help escape plateaus.
When the time budget is almost exhausted a cheap stochastic hill‑climbing
phase refines the best solution found.
All numbers stay non‑negative and the vector is normalised to sum‑one,
which is invariant for the evaluation score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using FFT.
    Length = len(a)+len(b)-1.
    """
    n = a.size
    m = b.size
    L = _next_pow2(n + m - 1)
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Fa * Fb, n=L)[: n + m - 1]
    return conv


def _full_auto_conv(a: np.ndarray) -> np.ndarray:
    """Return a*a (full auto‑convolution)."""
    return _conv_full(a, a)


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(tau·conv_k)/Σ exp(tau·conv).
    The subtraction of the maximum keeps the exponentials in a safe range.
    """
    # shift by max to avoid overflow
    shift = np.max(tau * conv)
    exps = np.exp(tau * conv - shift)
    return exps / np.sum(exps)


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth max‑convolution objective w.r.t. a.
    Returns the vector g_i = Σ_k w_k·2·a_{k-i}.
    Implemented as a convolution of the reversed a with the
    soft‑max weight vector w.
    """
    n = a.size
    w = _softmax_weights(conv, tau)               # length = 2n‑1
    a_rev = a[::-1]                               # reversed a
    # Convolution a_rev * w (full length = n + (2n‑1) – 1 = 3n‑2)
    grad_full = _conv_full(a_rev, w)              # length 3n‑2
    # Slice the part that corresponds to Σ_k w_k·a_{k-i}
    start = n - 1
    return 2.0 * grad_full[start : start + n]      # factor 2 from d(conv)/da


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (O(n log n)).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest ρ satisfying u_ρ > (cssv_ρ / ρ)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety: return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    target_n = 5000

    # Normalise to the probability simplex (sum = 1)

    # ------------------------------------------------------------------

            if delta <= 0:

            a_new[i] -= delta

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
