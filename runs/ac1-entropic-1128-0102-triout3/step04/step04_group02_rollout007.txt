def _lp_step(f, M):
    # solve LP maximize sum(g) subject to f * g <= M, g>=0
    n = f.shape[0]
    # Build matrix A shape (2n-1, n)
    # Use convolution: for each unit vector e_j (jth basis), conv = f convolve e_j yields shifting f.
    # So conv[k] = f[k-j] if 0 <= k-j < n else 0.
    # For fixed j, conv as vector = f shifted by j: indices k from j to j+n-1 have entries f[0:n] reversed? Actually conv (f convolve e_j) = f shifted by j, basically f padded with zeros left and right: conv[j:j+n] = f. Because e_j is at position j, convolution yields the sequence f shifted to start at j.
    # So for each column j, it's f placed starting at index j.
    # That gives easily A matrix: each column j has f in rows [j,j+n-1].
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j+n, j] = f
    # constraints: A @ g <= M
    # Objective: maximize sum(g) -> minimize -sum(g).
    # Use linprog
    from scipy.optimize import linprog
    c = -np.ones(n)  # minimize -sum = maximize sum
    bounds = [(0, None)] * n
    # Use method 'highs' for speed
    # Could also pass options like disp false
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        g = f.copy()  # fallback
    return g

"""
Hybrid optimisation for the step‑function problem.

The algorithm works on several scales.  For each length *n* it
1. upsamples the best solution found so far,
2. runs an Adam‑style projected gradient descent on a log‑sum‑exp
   surrogate of the max‑convolution,
3. (optional) solves a small linear programme that maximises the total
   mass under the current convolution bound – this often yields a
   shape with a lower peak,
4. performs a short simulated‑annealing sweep that transfers mass
   between two coordinates.

After the multi‑scale passes a fine‑tuning phase uses mirror descent
(exponential updates on the simplex) plus a final annealing polish.
All iterates are kept on the probability simplex, i.e. non‑negative and
unit‑sum, because `evaluate_sequence` is homogeneous (scaling does not
change its value).  The best sequence found before the time budget expires
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of *x* and *y* using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from *len(seq)* to *new_n* and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed ℓ∞ surrogate (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the surrogate   L_λ(a) = λ·log Σ exp(conv/λ) .
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # d/d a_i
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float = 0.0,
                      max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimiser for the log‑sum‑exp surrogate (projection onto simplex)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    step0 = 0.5        # base learning‑rate factor
    lam_factor = 0.4   # initial λ = lam_factor * max(conv)
    lam_decay = 0.5
    decay_interval = 2_000

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # sum(a) = 1 ⇒ objective = 2·n·max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if start_time:
                print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, "
                      f"score={cur_score:.6f}")

        # λ schedule
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        a = a - update
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror descent optimiser (exponential updates on simplex)
# ----------------------------------------------------------------------


def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float = 0.0,
                        max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """Mirror descent with KL‑divergence updates (log‑sum‑exp surrogate)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    lam_factor = 0.4
    lam_decay = 0.5
    decay_interval = 2_000

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if start_time:
                print(f"[{time.time() - start_time:6.2f}s] Mirror improve: n={n}, it={it}, "
                      f"score={cur_score:.6f}")

        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        eta = 0.5 / math.sqrt(it)
        grad_clipped = np.clip(grad, -80.0, 80.0)          # avoid overflow
        a = a * np.exp(-eta * grad_clipped)

        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that transfers a random ε from one coordinate to another.
    Returns the best sequence found during the sweep.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if start_time:
                print(f"[{time.time() - start_time:6.2f}s] Annealing improve: n={n}, it={it}, "
                      f"score={score:.6f}")

    return best_seq, best_score

# Linear‑programming refinement (optional, uses SciPy if available)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_j   s.t. (f * g)_k ≤ M  for all k,
    g_j ≥ 0.
    Returns the optimal *g* (same length as *f*).
    """
    try:
        from scipy.optimize import linprog
    except Exception:                         # SciPy missing – fall back
        return f.copy()

    n = f.shape[0]
    rows = 2 * n - 1
    # Build the (rows × n) convolution matrix A where each column is a shifted copy of f
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)                     # maximize Σ g  ↔  minimize −Σ g
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    # The 'highs' method is fast and robust
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        g = res.x
    else:
        # In the unlikely event of failure, stay with the current shape
        g = f.copy()
    return g

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Warm‑start from the previous run if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Multi‑scale schedule (small → large, shuffled to avoid bias)
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1600, 2400, 3200, 4800]
    np.random.shuffle(lengths)

    # start from the warm‑start or a random point of size 800
    if starter is not None:
        cur_a = starter
    else:
        cur_a = np.random.rand(800)
        cur_a /= cur_a.sum()

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise at current resolution
        if cur_a.shape[0] != n:
            a0 = _upsample(cur_a, n)
        else:
            a0 = cur_a.copy()

        # allocate a time slice for this resolution
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(6.0, 0.13 * remaining)

        # --------------------------------------------------------------
        # 1) Adam gradient descent (main workhorse)
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline,
                                             start_time=start_time,
                                             max_iter=150_000)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Global improve after Adam, "
                  f"n={n}, score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 2) Optional LP refinement (only for moderate sizes)
        # --------------------------------------------------------------
        try:
            import scipy          # quick check – may raise ImportError
            lp_possible = n <= 1200
        except Exception:
            lp_possible = False

        if lp_possible and time.time() < deadline:
            # run a couple of LP‑mix iterations
            for lp_iter in range(2):
                conv = _conv_self(a_opt)
                M = conv.max()
                g = _lp_step(a_opt, M)

                # convex combination and re‑normalisation
                t_mix = 0.35
                a_mix = (1 - t_mix) * a_opt + t_mix * g
                a_mix = _project_simplex(a_mix)

                # a tiny Adam refinement after the mixing
                sub_deadline = time.time() + 2.5
                a_opt, score_opt = _run_adam_descent(a_mix, n, sub_deadline,
                                                     start_time=start_time,
                                                     max_iter=30_000)
                if score_opt < best_score:
                    best_score = score_opt
                    best_seq = a_opt.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Global improve after LP+Adam, "
                          f"n={n}, score={score_opt:.6f}")

        # --------------------------------------------------------------
        # 3) Short simulated‑annealing polish
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(a_opt, n, anneal_deadline,
                                              start_time,
                                              best_score, best_seq,
                                              start_temp=5e-2,
                                              end_temp=1e-5,
                                              steps=20_000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] Global improve after annealing, "
                      f"n={n}, score={score_ref:.6f}")

        # keep this solution as the warm‑start for the next scale
        cur_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fine‑tuning phase (mirror descent + final annealing)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n_cur = best_seq.shape[0]

        # occasional random perturbation to escape shallow local minima
        a0 = best_seq.copy()
        if np.random.rand() < 0.2:
            noise = (np.random.rand(n_cur) - 0.5) * 0.02
            a0 = np.maximum(a0 + noise, 0.0)
            a0 /= a0.sum()

        slice_deadline = time.time() + min(25.0, 0.4 * (deadline - time.time()))
        a_opt, score_opt = _run_mirror_descent(a0, n_cur, slice_deadline,
                                               start_time=start_time,
                                               max_iter=120_000)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Fine‑tune improve (mirror), "
                  f"n={n_cur}, score={score_opt:.6f}")

        # final annealing polish
        if time.time() < deadline:
            anneal_deadline = time.time() + min(10.0, deadline - time.time())
            a_ref, score_ref = _run_annealing(a_opt, n_cur, anneal_deadline,
                                              start_time,
                                              best_score, best_seq,
                                              start_temp=1e-2,
                                              end_temp=1e-6,
                                              steps=12_000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] Fine‑tune improve (anneal), "
                      f"n={n_cur}, score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return as plain Python list
    return [float(x) for x in best_seq]
