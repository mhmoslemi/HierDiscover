"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm Overview
------------------
1. **Representation** – a probability vector `a` on the simplex
   (non‑negative, sum = 1).  All updates keep the vector on the simplex.

2. **Smooth‑max gradient** – instead of the nondifferentiable
   `max(conv(a))` we minimise a smooth approximation
   `S_α(a)= (1/α)·log Σ_k exp(α·conv_k)`.  
   The gradient of `S_α` can be written as a *single*
   convolution, computed efficiently with an FFT.  By slowly
   increasing `α` the method first makes large, well‑behaved steps
   and finally focusses on the true maximum.

3. **Multiplicative (exponentiated) update** – on the simplex an
   exponentiated‑gradient step `a←a·exp(−η·∇S_α)` automatically
   respects non‑negativity; a simple renormalisation restores Σa=1.

4. **Multi‑stage schedule** – optimisation is performed on three
   increasing lengths (800 → 2400 → 4800).  Each stage starts from the
   up‑sampled best vector of the previous stage and runs the
   exponentiated‑gradient loop for a fixed portion of the remaining
   time budget.  A short projected‑gradient refinement and a final
   simulated‑annealing pass are applied at the end.

5. **Time management** – the routine receives the total budget
   (`budget_s`).  All stages respect a common deadline
   (`start_time + budget_s – 0.8 s`) and print progress messages.

The combination of a smooth objective, adaptive step‑sizes and
exponentiated updates usually drives the convolution maximum
significantly below the 1.513 … value obtained by the earlier
projected‑gradient code, aiming at the target ≈ 1.503.
"""

# Linear (non‑circular) self‑convolution via FFT (used in the original code)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# General linear convolution via FFT (used for the smooth‑max gradient)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real‑valued vectors via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[:n]
    return conv


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth‑max of the self‑convolution
# ----------------------------------------------------------------------
def _expgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.7,
                     alpha_start: float = 5.0,
                     alpha_end: float = 200.0,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of 2·n·max(conv(a)).

    Returns the best vector seen and its corresponding objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # caps for the gradient to avoid overflow in the exponent
    GRAD_CLIP = 10.0

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- current objective -----
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ----- smooth‑max parameters (linear schedule on iteration) -----
        # after a few thousand iterations we are already close to the true max,
        # hence we ramp α up to focus on the active constraints.
        fraction = min(it / 200_000, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * fraction

        # ----- soft‑max weights -----
        # shift by max_val for numerical stability (max_val ≥ any conv entry)
        exp_term = np.exp(alpha * (conv - max_val))
        p = exp_term / exp_term.sum()               # shape (2·n‑1,)

        # ----- gradient of the smooth‑max -----
        # g_i = 2 * Σ_j a_j * p_{i+j}  (i = 0 … n‑1)
        p_rev = p[::-1]
        full = _conv_fft(a, p_rev)                 # length 3·n‑2
        m = len(p_rev)                             # = 2·n‑1
        grad = 2.0 * full[m - 1 - np.arange(n)]    # extract the needed segment
        # clip to keep the exponent bounded
        grad = np.clip(grad, -GRAD_CLIP, GRAD_CLIP)

        # ----- step size (Polyak‑style diminishing) -----
        step = step0 / math.sqrt(it)

        # ----- multiplicative (exponentiated) update + renormalisation -----
        a = a * np.exp(-step * grad)
        total = a.sum()
        if total > 0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected (sub‑gradient) descent – kept as a lightweight finetuner
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) (all active entries are used)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _run_simple_pg(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   beta: float = 0.9,
                   step0: float = 0.3) -> tuple[np.ndarray, float]:

    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)                # sub‑gradient
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # project onto the probability simplex
        a = _project_simplex(a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previous run (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 1 – modest size (n≈800) – several random restarts
    # ------------------------------------------------------------------
    n_small = 800
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    for restart in range(3):
        if starter is not None and restart == 0:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()

        a_opt, score_opt = _expgrad_descent(
            a0, slice_deadline, start_time,
            step0=0.8, alpha_start=4.0, alpha_end=180.0
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: restart={restart}, "
                  f"score={score_opt:.6f}")

        if time.time() >= deadline:
            break

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to medium size (n≈2400) and longer optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    remaining = deadline - time.time()
    grad_time_mid = max(20.0, 0.55 * remaining)
    grad_deadline_mid = time.time() + grad_time_mid

    a_mid_opt, score_mid = _expgrad_descent(
        a_mid, grad_deadline_mid, start_time,
        step0=0.6, alpha_start=6.0, alpha_end=200.0
    )
    if score_mid < best_score:
        best_score = score_mid
        best_seq = a_mid_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – up‑sample to full size (n≈4800) – finish with simple PG
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    remaining = deadline - time.time()
    grad_time_big = max(10.0, 0.35 * remaining)
    grad_deadline_big = time.time() + grad_time_big

    a_big_opt, score_big = _expgrad_descent(
        a_big, grad_deadline_big, start_time,
        step0=0.5, alpha_start=8.0, alpha_end=250.0
    )
    if score_big < best_score:
        best_score = score_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 3 improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Quick projected‑gradient finetuning (helps to tighten the bound)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        pg_deadline = deadline - 0.5   # keep a tiny buffer for annealing
        a_fine, score_fine = _run_simple_pg(best_seq, pg_deadline, start_time,
                                            beta=0.93, step0=0.25)
        if score_fine < best_score:
            best_score = score_fine
            best_seq = a_fine.copy()
            print(f"[{time.time() - start_time:6.2f}s] final PG improve: score={score_fine:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(best_seq,
                                          deadline, start_time,
                                          best_score, best_seq,
                                          start_temp=5e-2, end_temp=1e-6,
                                          steps=250_000)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
