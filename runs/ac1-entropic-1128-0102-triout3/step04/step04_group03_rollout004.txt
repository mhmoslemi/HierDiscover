"""
Propose a candidate step‑function.  

The algorithm works in three stages:

1. **Initialization** – load the best known sequence (``height_sequence_1``) or
   start from a random simplex point and up‑sample to a target length
   (default 9600).  The vector is normalised to sum = 1.

2. **Smooth‑max Adam optimisation** – minimise
   ``2·n·max(a∗a)`` on the simplex.  Instead of the non‑smooth max we use
   a temperature‑controlled soft‑max:
   ``w_k ∝ exp(α·(conv_k – max(conv)))``.  The gradient of the smooth
   objective is obtained analytically and evaluated by FFT
   (complexity ≈ O(n log n)).  An Adam step with projection onto the
   probability simplex moves the point.  The temperature parameter α is
   increased during the run so the optimisation first sees a smooth
   landscape and later focuses on the true max.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk
   that transfers a small amount of mass between two positions.  This
   phase cleans up any remaining spikes.

The whole routine respects the supplied time budget (default 1000 s)
and returns the best sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear convolution a*a using FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float, L: int) -> np.ndarray:
    """
    Compute the gradient of  max_{k} conv[k]  where the max is replaced by
    a soft‑max with temperature α.
    Returns a vector of length n (same length as a).
    """
    n = a.shape[0]

    # FFT of a (re‑used for both convolution and gradient)
    Af = np.fft.rfft(a, n=L)

    # ---- self‑convolution (needed for the soft‑max) ----
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # ---- soft‑max weights over the 2·n‑1 entries ----
    # shift by max_val for numerical stability (max exponent = 0)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- gradient of the smooth objective ----
    # gradient = 2 * correlation(a, w)   (see analysis)
    w_rev = w[::-1]                       # reverse for correlation
    W = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(Af * W, n=L)[: 3 * n - 2]   # length 3n‑2
    # extract the slice corresponding to indices n‑1 … 2n‑2 (inclusive)
    grad = 2.0 * grad_full[(n - 1):(2 * n - 1)]
    return grad

# ----------------------------------------------------------------------
# Score (exact, without the denominator because sum = 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a∗a) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Adam optimiser on the simplex with smooth‑max objective
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              deadline: float,
              start_time: float,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best point seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT length that can accommodate both the self‑conv (2n‑1) and the
    # correlation (3n‑2).  Use the larger of the two.
    L = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---- compute exact score (for book‑keeping) ----
        cur_sc = _score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it}, score={cur_sc:.6f}")

        # ---- schedule the soft‑max temperature α ----
        # early iterations: small α (smooth), later: larger α (focus on true max)
        if it < 20000:
            alpha = 10.0
        elif it < 50000:
            alpha = 50.0
        elif it < 100000:
            alpha = 200.0
        else:
            alpha = 500.0

        # ---- gradient of the smooth‑max ----
        grad = _smooth_max_gradient(a, alpha, L)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 300000) -> tuple[np.ndarray, float]:
    """Simple “move ε from i to j’’ annealing."""
    a = a0.copy()
    n = a.shape[0]
    score = _score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Uses Adam optimisation with a smooth‑max
    surrogate followed by a short annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Load a starter if the global variable `height_sequence_1` exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Target length – a larger n gives more freedom.  9600 is a
    # comfortable size for the time budget.
    # ------------------------------------------------------------------
    target_n = 9600

    # initialise a0
    if starter is not None:
        # linearly up‑sample the starter to the target length
        old_n = starter.size
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, target_n)
        a0 = np.interp(new_x, old_x, starter)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(target_n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Adam optimisation phase
    # ------------------------------------------------------------------
    # reserve about 80 % of the time budget for Adam, the rest for annealing
    adam_deadline = start_time + 0.80 * (deadline - start_time)
    best_seq, best_score = _run_adam(a0, adam_deadline, start_time)

    # ------------------------------------------------------------------
    # Final annealing phase (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_annealing(
            best_seq, deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=500000
        )
        print(f"[{time.time() - start_time:6.2f}s] final score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the harness
    return [float(x) for x in best_seq]
