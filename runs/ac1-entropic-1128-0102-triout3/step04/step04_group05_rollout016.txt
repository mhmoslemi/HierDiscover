    print(f"Finished best score {best_score:.6f} length {best_seq.shape[0]}")

"""
Hybrid optimizer for the auto‑convolution max‑norm.

* Phase 1 – several cheap Adam runs from diverse seeds (uniform,
  triangular, Hann, Gaussian, Dirichlet and the previous construction).
* Phase 2 – a long Adam refinement on the current best.
* Phase 3 – “hard‑max” Adam (Adam on the true sub‑gradient of the
  currently maximal convolution entry) to polish the result.
* Phase 4 – a few random‑perturbation restarts.

All updates keep the vector on the probability simplex (sum = 1,
non‑negative).  The routine stops automatically when the 1000 s budget
is exhausted and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) together with the full convolution.
    a is assumed to sum to 1 (simplex point)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _adam_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.05,
    temp_end: float = 0.0005,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation of
    2·n·max(conv(a,a)) on the simplex.

    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = float(2.0 * n * max_val)
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        max_val, conv = _conv_max(a)
        cur_score = float(2.0 * n * max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        # compute g_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
        start = a.shape[0] - 1
        g = 2.0 * conv2[start:start + a.shape[0]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_simplex(a)

    return a, best_a, best_score


def _hard_max_adam(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *true* sub‑gradient of the current maximal
    convolution entry (hard‑max).  Returns the final vector and its
    2·n·max(conv) score.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = float(2.0 * n * max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # index of the (first) maximal convolution entry
        k = int(np.argmax(conv))

        # gradient of conv[k] w.r.t a:
        # g_i = 2 * a_{k-i}  (if the index is in range)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        is_ = k - js
        grad[is_] = 2.0 * a[js]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_simplex(a)

        # occasional reporting
        if it % 40000 == 0:
            max_val, _ = _conv_max(a)
            cur_score = float(2.0 * n * max_val)
            print(f"[{time.time():.2f}s] hard‑max iter {it} score={cur_score:.6f}")

    # final evaluation
    final_score = float(2.0 * a.shape[0] * _conv_max(a)[0])
    return best_a, best_score if best_score < final_score else final_score

            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"  *** after hard‑max polishing – new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – a few jittered restarts (small random perturbations)
    # --------------------------------------------------------------
    for trial in range(5):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        noise = np.random.normal(scale=0.005, size=n)
        cand = best_seq + noise
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        else:
            cand = np.full(n, 1.0 / n)

        max_iters = int(min(60000, remaining / 0.001))
        _, local_best, _ = _adam_optimize(
            cand,
            max_iters=max_iters,
            deadline=deadline,
            alpha=0.1,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            temp_start=0.02,
            temp_end=0.00005,
        )
        max_val, _ = _conv_max(local_best)
        score = float(2.0 * n * max_val)
        if score < best_score:
            best_score = score
            best_seq = local_best.copy()
            print(f"  *** jitter trial {trial} improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final post‑processing
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total

    print(
        f"[{time.time() - start_time:6.2f}s] finished – best 2·n·max(conv) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
