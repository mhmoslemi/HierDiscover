"""
Propose a step‑function (non‑negative list) that minimizes `evaluate_sequence`.

Algorithm
---------
1.  Multi‑scale optimisation.
    * Work on a sequence of lengths [256, 512, 1024, 2048, 4096, 8192,
      16384] (shuffled each run).
    * For each length we take the best sequence from the previous scale,
      up‑sample it by linear interpolation and renormalise to sum = 1.
      If no previous best is available we start from a random simplex point.

2.  Direct sub‑gradient descent on the *ℓ∞* objective  
    f(a) = max_k (a*a)[k]    (self‑convolution).
    * The sub‑gradient at a point a is the average of the gradients of all
      active indices k where the convolution attains its maximum:
        g_i = 2·∑_{k∈K} a_{k‑i}   (indices outside [0,n) are ignored).
      The averaging over the active set K is performed via a convolution
      with a mask that puts weight 1/|K| on each active index.
    * An update step a ← Π_{simplex}(a − η·g) is taken, where Π projects
      onto the probability simplex.  The step size η follows a diminishing
      schedule η₀/√t with back‑tracking line‑search (halving η until the
      objective improves).
    * Every 1000 iterations a cheap “mass‑transfer” perturbation (move a
      small ε from one coordinate to another) is tried; if it improves the
      objective it is kept.

3.  Mirror‑descent refinement.
    After the sub‑gradient phase a few exponentiated‑gradient (mirror)
    steps are performed using the same sub‑gradient; this often smoothes
    out tiny irregularities.

4.  Final simulated‑annealing polish.
    A short annealing sweep (30 000 random mass‑transfer proposals) is run
    on the best sequence found.

All phases respect the global time budget (≈ 1000 s).  The routine keeps
the best sequence seen so far and finally returns it as a plain Python
list.  The algorithm works entirely with NumPy, needs no external
solvers and respects the non‑negativity / sum‑to‑one constraints that
`evaluate_sequence` enforces.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real‑FFT implementation)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length `seq.shape[0]` to `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of the ℓ∞ self‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Return (g, max_val, conv) where
        g   – sub‑gradient of f(a)=max_k (a*a)[k],
        max_val – the maximal convolution value,
        conv – the full self‑convolution vector.
    """
    n = a.shape[0]
    conv = _conv_self(a)                       # length 2n‑1
    max_val = conv.max()
    # mask of active indices (within a tiny tolerance)
    eps = 1e-12 * max_val if max_val > 0 else 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    if mask.sum() == 0.0:
        mask[:] = 1.0
    mask /= mask.sum()                         # average over the active set

    # convolution of the mask with reversed a gives the averaged shifted a
    a_rev = a[::-1]
    full = _conv_fft(mask, a_rev)              # length = (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    g = 2.0 * full[start:start + n]            # sub‑gradient (size n)
    return g, max_val, conv

def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise max self‑convolution via projected sub‑gradient descent.
    Returns the best sequence (sum = 1) and its internal score
    `2·n·max(conv)`.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial objective
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    step0 = 0.8  # initial stepsize constant
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # evaluate current point
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, it={it}, score={cur_score:.6f}")

        # sub‑gradient at the current point
        g, _, _ = _subgradient_max_conv(a)

        # diminishing stepsize
        eta = step0 / math.sqrt(it)

        # back‑tracking line search (max 10 reductions)
        for _ in range(10):
            a_candidate = _project_simplex(a - eta * g)
            new_score = 2.0 * n * _conv_self(a_candidate).max()
            if new_score <= cur_score:
                a = a_candidate

            eta *= 0.5
        # If no improvement after back‑tracking, keep the current a.

                a_tmp[i] -= eps
                a_tmp[j] += eps
                a_tmp = _project_simplex(a_tmp)
                candidate_score = 2.0 * n * _conv_self(a_tmp).max()
                if candidate_score < cur_score:
                    a = a_tmp

    return best_a, best_score

def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iter: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using exponentiated updates.
    Uses the exact sub‑gradient of the ℓ∞ convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        g, _, _ = _subgradient_max_conv(a)

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0:

                a_tmp[i] -= eps
                a_tmp[j] += eps
                a_tmp = _project_simplex(a_tmp)
                a = a_tmp

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing: move a fraction ε from i to j.
    Returns an improved sequence (if any) and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (sum ≈ 1, all entries ≥ 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Warm‑start from the previous run if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule (geometric growth, shuffled)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    np.random.shuffle(lengths)

    # keep the best sequence so far for up‑sampling
    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a0 for the current scale
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a time slice for this scale (roughly proportional to n)
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(6.0, 0.03 * remaining)

        # ------------------------------------------------------------------
        # 1) Sub‑gradient descent (the core optimiser)
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_subgradient_descent(a0, n, slice_deadline,
                                                    start_time, max_iter=150_000)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad: n={n}, score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # 2) Mirror‑descent fine‑tuning (shorter time slice)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            md_deadline = time.time() + max(4.0, 0.02 * (deadline - time.time()))
            a_md, score_md = _run_mirror_descent(best_seq.copy(), n,
                                                 md_deadline, start_time,
                                                 max_iter=80_000)
            if score_md < best_score:
                best_score = score_md
                best_seq = a_md.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after mirror: n={n}, score={score_md:.6f}")

        # ------------------------------------------------------------------
        # 3) Brief annealing sweep
        # ------------------------------------------------------------------
        if time.time() < deadline:
            an_deadline = time.time() + max(4.0, 0.01 * (deadline - time.time()))
            a_an, score_an = _run_annealing(best_seq.copy(), n,
                                            an_deadline, start_time,
                                            best_score, best_seq,
                                            start_temp=5e-2,
                                            end_temp=1e-6,
                                            steps=15_000)
            if score_an < best_score:
                best_score = score_an
                best_seq = a_an.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal: n={n}, score={score_an:.6f}")

        # upscale for the next iteration
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final polishing on the best length (mirror + anneal)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        n_cur = best_seq.shape[0]

        # occasional mild up‑scaling (never exceed 20000)
        if np.random.rand() < 0.15 and n_cur < 20000:
            new_n = min(20000, int(n_cur * 1.5))
            a0 = _upsample(best_seq, new_n)
            n_use = new_n
        else:
            a0 = best_seq.copy()
            n_use = n_cur

        # allocate a short slice (≈ 20 s) for mirror refinement
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(20.0, 0.4 * remaining)

        a_ref, score_ref = _run_mirror_descent(a0, n_use,
                                               slice_deadline, start_time,
                                               max_iter=60_000)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final mirror improve: n={n_use}, score={score_ref:.6f}")

        # final quick annealing polish
        if time.time() < deadline:
            an_deadline = time.time() + min(8.0, deadline - time.time())
            a_fin, score_fin = _run_annealing(best_seq.copy(), n_use,
                                              an_deadline, start_time,
                                              best_score, best_seq,
                                              start_temp=1e-2,
                                              end_temp=1e-7,
                                              steps=12_000)
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"[{time.time() - start_time:6.2f}s] final anneal improve: n={n_use}, score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list – the harness will feed it to evaluate_sequence()
    return [float(x) for x in best_seq]
