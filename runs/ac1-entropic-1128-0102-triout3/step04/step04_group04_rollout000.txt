        large sub‑gradient to indices with small sub‑gradient (several
        source/target pairs and several step sizes);
      – if no greedy improvement is found we perform a **mirror‑descent**
        (exponentiated‑gradient) step on the probability simplex;
      – with a small probability we execute a **local permutation**
        (shuffle a short window) to escape order‑related local minima.

  All candidates are kept on the simplex (non‑negative, sum = 1) by the
  Euclidean simplex projection.

* **Best‑ever tracking** – the globally best vector (over all lengths)
  and its score are stored and printed whenever an improvement occurs.

The function returns the best sequence found as a plain list of Python
floats; `evaluate_sequence` will be called by the outer harness.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Compute the full auto‑convolution a*a using an FFT of size a power of two.
    """
    n = a.shape[0]
    # next power of two >= 2*n-1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_and_max(a: np.ndarray):
    """Return both the convolution array and its maximum."""
    conv = _conv_full(a)
    return conv, conv.max()


def _score_normed(a: np.ndarray) -> float:
    """
    Return the objective value `2·n·max(conv(a,a))` for a *normalized* vector
    (sum(a) == 1).  The code works even if a is not perfectly normalised –
    the vector is first divided by its ℓ₁‑norm.
    """
    s = a.sum()
    if s == 0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    _, max_val = _conv_and_max(a)
    return 2.0 * n * max_val


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30) -> np.ndarray:
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept.
    If fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        # add the largest entries to reach max_add
        topk = min(max_add, conv.size)
        extra = np.argpartition(conv, -topk)[-topk:]
        K = np.unique(np.concatenate((K, extra)))
    return K


def _subgradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv(a,a))` w.r.t. `a` for the active set `K`.
    For each active index k we add 2·a_{k-i} to component i whenever the
    pair (i, k-i) is valid.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        # i ranges over indices where both i and k-i lie in [0,n-1]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries negative – fall back to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core refinement routine
# ----------------------------------------------------------------------


def _refine_sequence(a_init: np.ndarray,
                     deadline: float,
                     start_time: float,
                     verbose: bool = True):
    """
    Gradient‑directed refinement with greedy mass‑transfer, mirror‑descent
    fallback and occasional local permutations.

    Parameters
    ----------
    a_init : ndarray
        Normalised start vector (sum == 1, non‑negative).
    deadline : float
        Absolute wall‑clock time (seconds since epoch) after which the
        routine must stop.
    start_time : float
        Time when the whole search started (used for progress printing).
    verbose : bool
        If True, prints intermediate improvements.

    Returns
    -------
    best_seq, best_score
        The best sequence discovered during the refinement and its
        objective value.
    """
    n = a_init.shape[0]
    current = a_init.copy()
    best_seq = current.copy()
    best_score = _score_normed(best_seq)

    it = 0
    # parameters for the greedy mass‑transfer
    top_k = 5          # how many indices with highest/lowest gradient to consider
    delta_fracs = (0.01, 0.03, 0.07, 0.12)   # candidate step sizes (relative to source mass)
    # probability for a random local permutation
    perm_prob = 0.04

    while time.time() < deadline:
        it += 1

        # full convolution and current score
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            if verbose:
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Iter {it:5d}  new best score={cur_score:.6f}")

        # --------------------------------------------------------------
        # (a) compute active‑set sub‑gradient
        # --------------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _subgradient_active(current, K)

        # --------------------------------------------------------------
        # (b) greedy mass‑transfer between extreme gradient indices
        # --------------------------------------------------------------
        # indices sorted by gradient
        idx_desc = np.argsort(-g)[:top_k]
        idx_asc = np.argsort(g)[:top_k]

        improved = False
        for i_src in idx_desc:
            src_mass = current[i_src]
            if src_mass <= 0:
                continue
            for i_tgt in idx_asc:
                if i_src == i_tgt:
                    continue
                # try a few step sizes
                for frac in delta_fracs:
                    delta = src_mass * frac
                    if delta <= 0:
                        continue
                    cand = current.copy()
                    cand[i_src] -= delta
                    cand[i_tgt] += delta
                    # still on the simplex (mass conserved), just recompute score
                    conv_c, max_c = _conv_and_max(cand)
                    new_score = 2.0 * n * max_c
                    if new_score < cur_score - 1e-12:
                        current = cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            continue  # go to next iteration with the improved vector

        # --------------------------------------------------------------
        # (c) fallback – exponentiated‑gradient (mirror‑descent) step
        # --------------------------------------------------------------
        eta = 0.5 / math.sqrt(it + 1.0)        # decaying step size
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        # stabilise exponentiation
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        current = a_new

        # --------------------------------------------------------------
        # (d) occasional local permutation (simulated‑annealing style)
        # --------------------------------------------------------------
        if random.random() < perm_prob:
            block_len = min(120, n)                     # a modest window
            start = random.randint(0, n - block_len)
            block = current[start:start + block_len]
            perm = np.random.permutation(block_len)
            cand = current.copy()
            cand[start:start + block_len] = block[perm]
            conv_c, max_c = _conv_and_max(cand)
            new_score = 2.0 * n * max_c
            if new_score < cur_score - 1e-12:
                current = cand
                # no immediate restart – continue with the permuted vector

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low `evaluate_sequence`
    score.  The routine respects a≈1000 s wall‑clock budget and prints
    intermediate improvements.

    Returns
    -------
    list[float]
        The best sequence discovered (already normalised to sum = 1);
        the external harness will apply the final scaling and clamping.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # a small safety margin

    # ------------------------------------------------------------------
    # 0. Load previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.sum() == 0.0:
            raise ValueError
        prev = prev / prev.sum()
        prev_len = prev.shape[0]
        best_seq = prev.copy()
        best_score = _score_normed(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Start from height_sequence_1 "
              f"(n={prev_len}) → score={best_score:.6f}")
        length_options = [prev_len]
    except Exception:
        # No previous solution – start with a modest length
        length_options = [5000]
        best_seq = None
        best_score = float("inf")
        print(f"[{time.time() - start_time:6.2f}s] No previous construction – "
              f"using fresh length list {length_options}")

    # ------------------------------------------------------------------
    # 1. Extend the set of lengths (larger n may give a better lower bound)
    # ------------------------------------------------------------------
    # We try a few longer vectors in addition to the initial length.
    # The list is kept modest to stay within the time budget.
    extra_lengths = [8000, 12000, 16000]
    # ensure we do not exceed the budget by adding too many lengths
    for L in extra_lengths:
        if L not in length_options:
            length_options.append(L)

    # ------------------------------------------------------------------
    # 2. Loop over candidate lengths
    # ------------------------------------------------------------------
    for n in length_options:
        if time.time() >= deadline:
            break

        # ---- a) pick an initial vector for this length -----------------
        # If we already have a vector of this length (only possible for the
        # first iteration), reuse it; otherwise generate a power‑law grid.
        if best_seq is not None and best_seq.shape[0] == n:
            a0 = best_seq.copy()
        else:
            # sweep a modest grid of exponents
            p_grid = np.linspace(0.12, 0.85, 25)   # 25 candidates
            a0 = None
            local_best = float("inf")
            for p in p_grid:
                if time.time() >= deadline:
                    break
                idx = np.arange(n, dtype=float) + 1.0
                a = idx ** (-p)
                a_sum = a.sum()
                if a_sum == 0:
                    a = np.full(n, 1.0 / n)
                else:
                    a = a / a_sum
                sc = _score_normed(a)
                if sc < local_best:
                    local_best = sc
                    a0 = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] n={n:5d} "
                          f"p={p: .3f} → init score={sc:.6f}")
            if a0 is None:   # fallback – uniform
                a0 = np.full(n, 1.0 / n)

        # ---- b) random permutation warm‑starts (order matters) ---------
        # A few random shuffles can already improve the score substantially.
        perm_trials = 1500
        for _ in range(perm_trials):
            if time.time() >= deadline:
                break
            perm = np.random.permutation(n)
            a_perm = a0[perm]
            sc = _score_normed(a_perm)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = a_perm.copy()
                print(f"[{time.time() - start_time:6.2f}s] n={n:5d} "
                      f"perm improve → score={sc:.6f}")

        # ---- c) main refinement for this length -----------------------
        if time.time() >= deadline:
            break
        refined_seq, refined_score = _refine_sequence(best_seq,
                                                      deadline,
                                                      start_time,
                                                      verbose=True)
        if refined_score < best_score - 1e-12:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] n={n:5d} "
                  f"refine improve → score={refined_score:.6f}")

        # ---- d) cheap random hill‑climbing on the refined vector -------
        # Quick swaps of small amounts of mass; useful for fine‑tuning.
        if time.time() < deadline:
            hill_iters = 12000
            current = best_seq.copy()
            cur_score = best_score
            for _ in range(hill_iters):
                if time.time() >= deadline:
                    break
                i, j = np.random.randint(0, n, size=2)
                if i == j or current[i] == 0:
                    continue
                eps = np.random.rand() * 0.045   # up to 4.5 % of source mass
                delta = eps * current[i]
                cand = current.copy()
                cand[i] -= delta
                cand[j] += delta
                conv_c, max_c = _conv_and_max(cand)
                new_score = 2.0 * n * max_c
                if new_score < cur_score - 1e-12:
                    cur_score = new_score
                    current = cand
                    best_seq = cand.copy()
                    best_score = new_score
                    print(f"[{time.time() - start_time:6.2f}s] n={n:5d} "
                          f"hill‑climb → score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a short uniform vector
        best_seq = np.full(200, 1.0 / 200)

    # The evaluation function clamps values to [0,1000] and rescales by the
    # sum, so we simply return the normalised vector.
    return [float(x) for x in best_seq]
