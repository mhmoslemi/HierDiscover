def build_sparse_support(n, m, R_max=2, max_trials=10000):
    support = []
    sum_counts = np.zeros(2*n-1, dtype=int)
    # add first element 0
    while len(support) < m:
        # try a few random candidates
        best = None
        best_max = None
        for _ in range(100):
            cand = np.random.randint(0, n)
            if cand in support: continue
            # compute new sum counts if added
            new_counts = sum_counts.copy()
            for i in support:
                new_counts[cand + i] += 2  # ordered pairs (i,cand) and (cand,i)
            new_counts[2*cand] += 1  # self pair i=j=cand
            # compute max count
            cur_max = new_counts.max()
            if cur_max <= R_max:
                # immediate accept
                support.append(cand)
                sum_counts = new_counts
                break
            else:
                # store maybe best if still low
                if best_max is None or cur_max < best_max:
                    best = cand
                    best_max = cur_max
        else:
            # could not find candidate within R_max
            # relax R_max by 1 and continue
            R_max += 1
            continue

"""
Fast Adam‑based optimisation for the step‑function attack.

The optimiser works on the simplex (non‑negative entries that sum to one)
and directly minimises the quantity
    score = 2·n·max(conv)   (conv = a * a, linear convolution)
which is exactly the value returned by ``evaluate_sequence`` when the
sequence is normalised to sum=1.

Key ideas
---------

* **Single‑active‑index sub‑gradient** – the sub‑gradient of ``max(conv)``
  is simply the gradient of the entry where the convolution attains its
  maximum.  This yields an inexpensive O(n) update.

* **Adam optimiser** – an adaptive first‑order method replaces the
  handcrafted diminishing‑step schedule used previously.  Adam’s
  per‑coordinate step‑size automatically balances large and small
  components of the gradient.

* **Simplex projection** – after each update the vector is projected onto
  the probability simplex using an O(n) sort‑based algorithm.

* **Multi‑scale schedule** – optimisation proceeds on lengths
  800 → 2400 → 4800.  For each length several initialisations are tried
  (up‑scaled previous best, a Hann window and a random start) and the
  best result is kept.

* **Final annealing** – a cheap simulated‑annealing walk moves a small
  amount of mass between two positions and refines the solution within
  the remaining budget.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence discovered as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple “window” initialisation (Hann)
# ----------------------------------------------------------------------
def _initial_hann(n: int) -> np.ndarray:
    """Return a Hann window of length n, normalised to sum=1."""
    x = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Up‑sampling (linear interpolation) – keeps total mass = 1
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Adam optimisation on the simplex using the max‑convolution sub‑gradient
# ----------------------------------------------------------------------

                  n: int,

                  start_time: float,
                  step0: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    Returns the best sequence found and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={t}, score={cur_score:.6f}")

        # sub‑gradient of max(conv) – only the index achieving the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        j = k - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[j]                      # g_i = 2 * a_{k-i}

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # element‑wise step size
        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    runs Adam‑based optimisation on three increasingly fine scales,
    and finishes with a short annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter – use the global ``height_sequence_1`` if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definition (progressively larger problem sizes)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800]   # three up‑sampling levels
    anneal_budget = max(30.0, 0.15 * (deadline - start_time))
    grad_budget = (deadline - start_time) - anneal_budget

    total_weight = sum(stage_lengths)
    total_weight = max(total_weight, 1)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a share of the remaining gradient budget to this stage
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # ------------------------------------------------------------------
        # Build a few different initial candidates for this stage
        # ------------------------------------------------------------------
        candidates = []
        if best_seq is not None:
            candidates.append(_upsample(best_seq, n))
        if starter is not None:
            candidates.append(_upsample(starter, n))
        # Hann window is a useful deterministic start
        candidates.append(_initial_hann(n))
        # a pure random start (useful for escaping local minima)
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        candidates.append(rnd)

        # limit the number of candidates so that we stay within the budget
        max_cand = 3
        candidates = candidates[:max_cand]

        # time per candidate (even split)
        per_cand_deadline = time.time() + stage_time / max(len(candidates), 1)

        for a0 in candidates:
            if time.time() >= deadline:
                break
            cand_deadline = min(per_cand_deadline, deadline - 0.1)
            a_opt, score_opt = _run_adam_opt(
                a0, n, cand_deadline, start_time,
                step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8
            )
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works on the final length we ended with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=500_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
