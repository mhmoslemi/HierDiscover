"""
Propose a low‑score step‑function using a multi‑scale symmetric Adam optimiser.

Key ideas
---------
* The evaluation is scale‑invariant → fix the sum of the coefficients to 1.
* Enforce symmetry (a[i] = a[n‑1‑i]); the optimal solution is believed to be
  symmetric, reducing the search space and stabilising gradients.
* Optimise a smooth approximation of the maximum of the auto‑convolution
  using a soft‑max (log‑sum‑exp) with a slowly decreasing temperature.
* The optimiser is Adam with Euclidean projection onto the probability simplex.
* Start from several simple shapes (uniform, triangular, Hann), the previously
  discovered best construction (if any) and a few random Dirichlet draws.
* Run the optimiser on a cascade of lengths (≈1.5× growth).  At each stage the
  best solution is up‑sampled to the new length with linear interpolation.
* After the main loop a final long refinement is performed.  Random jitter
  is occasionally injected to escape shallow local minima.
* All operations are FFT‑based (convolution and the gradient) for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max(conv(a,a)) and the full convolution.
    Assumes sum(a) == 1 (normalised).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k * (a*a)[k] w.r.t. a.
    Implements   g_i = 2 * Σ_j w_{i+j} a_j   using an FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # conv2 length = len(w) + len(a) - 1
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005,
                   symmetrize: bool = True,
                   jitter_schedule: callable | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for 2·n·max(conv) on the simplex.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute FFT lengths (cached for the whole run)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ensure non‑negativity before projection
        a = np.maximum(a, 0.0)

        # optional symmetry enforcement
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # optional jitter (helps escaping local minima)
        if jitter_schedule is not None:
            sigma = jitter_schedule(it, max_iters)
            if sigma > 0.0:
                a += sigma * np.random.randn(n)
                a = np.maximum(a, 0.0)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


def _interpolate(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = old.shape[0]
    if old_len == new_len:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, old)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = np.full(new_len, 1.0 / new_len)
    else:
        up = up / s
    return up
