"""
Propose a non‑negative step‑function minimising `evaluate_sequence`.

This implementation builds on the previous smoothed‑max gradient descent,
but replaces the simple momentum scheme with an Adam optimiser and a more
aggressive multi‑scale schedule.  For each target length `n`:

1. Warm‑start from the best distribution found so far (linear up‑sampling).
2. Run Adam on the log‑sum‑exp surrogate of the ℓ∞‑norm of the
   self‑convolution, projecting onto the probability simplex after each
   update.
3. Refine the result with a cheap mass‑transfer simulated‑annealing
   that moves a fraction of the mass between two positions.
4. Occasionally inject a small random perturbation and restart the Adam
   phase – this helps escape shallow local minima.

The algorithm allocates most of the 1000 s budget to the largest
problem size (up to 32768 coefficients) while keeping a few seconds for
the smaller scales so that a good warm‑start is available.  Progress is
printed to `stdout`; the returned object is a plain Python list of
floats.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# --------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (numerically stable)
    weights = np.exp((conv - max_val) / lam)
    weights_sum = weights.sum()
    if weights_sum == 0:
        # should not happen, but protect against division by zero
        weights = np.full_like(weights, 1.0 / weights.size)
    else:
        weights /= weights_sum
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)               # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]             # ∂L/∂a_i
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) where n = len(a)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      slice_deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate, projected onto the simplex.
    Returns the best vector seen in the time slice and its exact objective.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    best_a = a.copy()
    best_score = float('inf')
    lam_factor = 0.30      # initial factor for λ = lam_factor * max_b
    lam = None

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # Update smoothing parameter λ – never let it become exactly zero
        if lam is None:
            lam = max_b * lam_factor
        else:
            # exponential decay, but keep it at least 1e-12 of max_b
            lam = max(lam * 0.995, max_b * 1e-12)

        # Compute surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # Occasional isotropic noise to shake shallow plateaus
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Learning‑rate schedule – we start relatively aggressive and let it
        # decay as 1/√it via the Adam denominator.
        step = 0.4 * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated‑annealing.
    Returns the possibly improved sequence and its exact objective.
    """
    a = a0.copy()
    cur_score = _objective(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # choose source and destination indices
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1  # ensure j != i

        # move up to 30 % of the source mass (more aggressive early on)
        eps = a[i] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={cur_score:.6f}")

    return best_seq, best_score


def _perturb_and_restart(a: np.ndarray,
                         n: int,
                         budget: float,
                         start_time: float,
                         start_time_global: float) -> tuple[np.ndarray, float]:
    """
    Create a short random perturbation of `a`, run a brief Adam phase and
    return the best result seen within the allocated `budget` seconds.
    This helps to escape from local minima without a full restart.
    """
    # Small Gaussian noise, then re‑project onto simplex
    noise = np.random.randn(n) * 0.02
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0.0)
    if a_pert.sum() == 0.0:
        a_pert = np.full_like(a_pert, 1.0 / n)
    else:
        a_pert /= a_pert.sum()

    slice_deadline = time.time() + budget
    return _run_adam_descent(a_pert, n, slice_deadline, start_time_global)

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm improves on the previous gradient‑descent approach in three
ways:

1. **Adam optimiser** – replaces the simple momentum scheme with the
   adaptive Adam update, which typically converges faster on the
   log‑sum‑exp surrogate of the ℓ∞‑norm of the self‑convolution.
2. **More aggressive multi‑scale schedule** – the best distribution found
   for a size ``n`` is linearly up‑sampled and used as a warm‑start for a
   larger size.  Most of the 1000 s budget is spent on the largest size
   (up to 32768 coefficients) while the smaller sizes provide cheap
   warm‑starts.
3. **Targeted annealing & perturb‑restart** – after the Adam phase a cheap
   mass‑transfer simulated‑annealing refines the solution, and a short
   random perturb‑restart helps escape shallow local minima.

All helper functions are top‑level, the code respects the time budget,
and the final sequence is returned as a plain ``list`` of floats.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# --------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed log‑sum‑exp surrogate of max(conv).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    wsum = weights.sum()
    if wsum == 0:
        weights = np.full_like(weights, 1.0 / weights.size)
    else:
        weights /= wsum
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) where n = len(a)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      slice_deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate, projected onto the simplex.
    Returns the best vector seen in the allotted time slice and its exact objective.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    best_a = a.copy()
    best_score = float('inf')
    lam_factor = 0.30
    lam = None

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # smoothing parameter λ – keep it positive
        if lam is None:
            lam = max_b * lam_factor
        else:
            lam = max(lam * 0.995, max_b * 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise for escaping plateaus
        if it % 500 == 0:
            grad += (0.02 / math.sqrt(it)) * np.random.randn(n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate (0.4 works well in practice)
        step = 0.4 * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer simulated‑annealing.
    Returns the (possibly) improved sequence and its exact objective.
    """
    a = a0.copy()
    cur_score = _objective(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 30 % of the source mass (aggressive early on)
        eps = a[i] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={cur_score:.6f}")

    return best_seq, best_score


def _perturb_and_restart(a: np.ndarray,
                         n: int,
                         budget: float,
                         start_time: float,
                         start_time_global: float) -> tuple[np.ndarray, float]:
    """
    Apply a small random perturbation to ``a`` and run a short Adam phase.
    Returns the best vector found within ``budget`` seconds.
    """
    noise = np.random.randn(n) * 0.02
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 0.0:
        a_pert /= s
    else:
        a_pert = np.full_like(a_pert, 1.0 / n)

    slice_deadline = time.time() + budget
    return _run_adam_descent(a_pert, n, slice_deadline, start_time_global)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_score = float('inf')
    best_seq = None

    # --------------------------------------------------------------
    # Optional warm‑start from the previous run (height_sequence_1)
    # --------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            ss = starter.sum()
            if ss > 0.0:
                starter = starter / ss
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – climb up to 32768.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths[:-1])   # shuffle early lengths, keep largest last

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # --------------------------------------------------------------
        # Allocate a time slice for this length
        # --------------------------------------------------------------
        if idx < len(lengths) - 1:
            slice_time = max(8.0, remaining * 0.10)      # early sizes get ~10 %
        else:
            slice_time = remaining * 0.90                # bulk of the budget for the largest size
        slice_deadline = time.time() + slice_time

        # --------------------------------------------------------------
        # Adam‑based smoothed gradient descent (≈60 % of the slice)
        # --------------------------------------------------------------
        adam_deadline = time.time() + 0.60 * slice_time
        a_opt, score_opt = _run_adam_descent(a0, n, adam_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Annealing refinement (remaining time for this length)
        # --------------------------------------------------------------
        if time.time() < slice_deadline:
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                slice_deadline,
                start_time,
                best_seq,
                best_score,
                start_temp=5e-2,
                end_temp=1e-5,
                max_steps=150000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # --------------------------------------------------------------
        # Light perturb‑restart phase (helps to leave shallow basins)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            restart_budget = min(5.0, remaining * 0.05)   # at most a few seconds
            a_restart, score_restart = _perturb_and_restart(
                best_seq,
                n,
                restart_budget,
                start_time,
                start_time,
            )
            if score_restart < best_score:
                best_score = score_restart
                best_seq = a_restart.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after restart n={n} score={score_restart:.6f}")

        # Keep the best distribution for the next (larger) size.
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
