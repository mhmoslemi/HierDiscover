def _adam_optimize(a0, n, deadline, start_time, max_iter=200000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5  # initial step size

    lam_factor = 0.2
    lam_decay = 0.5
    decay_interval = 2000

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # update lam
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)

        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project to simplex
        a = _project_simplex(a)

    return best_a, best_score
```

def _support_anneal(n, warm_seq, deadline, start_time):
    # Determine K
    K = max(5, int(1.7 * math.sqrt(n)))  # ensure integer >=5.
    # Initialize S
    if warm_seq is not None and warm_seq.shape[0]==n:
        # top K indices
        S = set(np.argsort(-warm_seq)[:K].tolist())
    else:
        S = set(np.random.choice(n, K, replace=False).tolist())

    # indicator array of ints as counts
    indicator = np.zeros(n, dtype=int)
    for i in S:
        indicator[i] = 1
    counts = np.convolve(indicator, indicator)  # length 2n-1
    max_counts = counts.max()
    cur_score = 2.0 * n * max_counts / (K**2)
    best_score = cur_score
    best_S = S.copy()

    # annealing params
    total_iter_est = 200000
    # compute temperature schedule constants
    start_temp = 0.2
    end_temp = 1e-5

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1
        if it > total_iter_est:
            break

        # temperature
        T = start_temp * (end_temp / start_temp) ** (it / total_iter_est)

        # pick swap
        i_out = random.choice(tuple(S))
        # pick candidate not in S
        i_in = random.randint(0, n-1)
        while i_in in S:
            i_in = random.randint(0, n-1)

        # apply removal then addition using diffs list
        diffs = []

        # removal
        for y in list(S):
            s = i_out + y
            delta = -1 if y == i_out else -2
            counts[s] += delta
            diffs.append((s, -delta))  # to revert, add opposite
        S.remove(i_out)

        # addition
        for y in list(S):
            s = i_in + y
            counts[s] += 2
            diffs.append((s, -2))
        # self-pair
        s_self = 2 * i_in
        counts[s_self] += 1
        diffs.append((s_self, -1))
        S.add(i_in)

        new_max_counts = counts.max()
        new_score = 2.0 * n * new_max_counts / (K**2)
        delta_score = new_score - cur_score

        accept = False
        if delta_score < 0:
            accept = True
        else:

                accept = True

        if accept:

                best_score = cur_score
                best_S = S.copy()
                print(f"[{time.time() - start_time:.2f}s] support improve n={n} it={it} score={cur_score:.6f}")
        else:
            # revert changes
            for idx, rev_delta in diffs:
                counts[idx] += rev_delta
            # revert set
            S.remove(i_in)
            S.add(i_out)

        # optionally check if it % 50000 == 0: maybe reduce temp etc.

    # build best sequence
    best_seq = np.zeros(n, dtype=float)
    weight = 1.0 / K
    for i in best_S:
        best_seq[i] = weight

    s = best_seq.sum()

    else:
        best_seq = np.full(n, 1.0/n)
    best_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq)) / (best_seq.sum()**2)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative list of heights) that minimizes
`evaluate_sequence`.  
The search combines three ideas:

1. **Sparse support optimisation** – we look for a set of indices S (size K)
   such that the autocorrelation (the convolution of the indicator of S with
   itself) has a small maximal entry.  The objective for a uniform weight on S
   depends only on ``max_counts/K²``.  An annealing walk swaps one index of S
   with an index outside S while keeping K fixed; the convolution counts are
   updated incrementally (O(K) per swap).

2. **Adam‑style simplex optimisation** – the best sparse vector found above
   is refined by a few hundred thousand iterations of Adam on the
   log‑sum‑exp surrogate of the ℓ∞‑norm of the convolution, with projection
   onto the probability simplex after each step.

3. **Mass‑transfer simulated annealing** – a short final sweep moves a tiny
   amount of probability mass between two coordinates to polish the result.

All phases respect the time budget (default 1000 s).  The code works with a
single CPU, uses only NumPy and the already‑imported ``evaluate_sequence``,
and prints intermediate scores for monitoring.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities (Euclidean projection)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of  λ·log Σ exp(conv/λ)  w.r.t. a.
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex with the smooth ℓ∞ surrogate."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    lam_factor = 0.25
    lam_decay = 0.5
    decay_interval = 2_000

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}"
            )

        # λ schedule
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] annealing improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sparse‑support simulated annealing (binary indicator)
# ----------------------------------------------------------------------
def _support_anneal(
    n: int,
    warm_seq: np.ndarray | None,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Anneal a binary support of size K (uniform weight) to minimise the
    maximal entry of the self‑convolution.  The convolution counts are kept
    incrementally (O(K) per swap) to stay fast.
    """
    # heuristic size of the support
    K = max(5, int(1.7 * math.sqrt(n)))
    # initialise support S
    if warm_seq is not None and warm_seq.shape[0] == n:
        # top‑K indices of the warm start
        S = set(np.argsort(-warm_seq)[:K].tolist())
        if len(S) < K:
            # fill the missing spots at random
            missing = K - len(S)
            extra = set(
                np.random.choice(
                    list(set(range(n)) - S), size=missing, replace=False
                )
            )
            S.update(extra)
    else:
        S = set(np.random.choice(n, K, replace=False).tolist())

    # indicator of current support
    indicator = np.zeros(n, dtype=int)
    for i in S:
        indicator[i] = 1

    # convolution of the indicator with itself – counts of ordered pairs
    counts = np.convolve(indicator, indicator)  # length 2n‑1, dtype=int
    cur_max = counts.max()
    cur_score = 2.0 * n * cur_max / (K ** 2)

    best_S = S.copy()
    best_score = cur_score

    # annealing schedule
    total_iters = 200_000
    start_temp = 0.2
    end_temp = 1e-5

    it = 0
    while True:
        if time.time() >= deadline or it >= total_iters:
            break
        it += 1
        T = start_temp * (end_temp / start_temp) ** (it / total_iters)

        # propose a swap
        i_out = random.choice(tuple(S))
        i_in = random.randint(0, n - 1)
        while i_in in S:
            i_in = random.randint(0, n - 1)

        # --- apply removal of i_out ---------------------------------
        diffs = []  # (index, revert_delta)
        for y in list(S):
            s = i_out + y
            if y == i_out:
                delta = -1
            else:
                delta = -2
            counts[s] += delta
            diffs.append((s, -delta))  # to undo we add the opposite
        S.remove(i_out)

        # --- apply addition of i_in ---------------------------------
        for y in list(S):
            s = i_in + y
            counts[s] += 2
            diffs.append((s, -2))
        s_self = 2 * i_in
        counts[s_self] += 1
        diffs.append((s_self, -1))
        S.add(i_in)

        new_max = counts.max()
        new_score = 2.0 * n * new_max / (K ** 2)
        delta_score = new_score - cur_score

        accept = False
        if delta_score < 0:
            accept = True
        else:
            if math.exp(-delta_score / T) > random.random():
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_S = S.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] support improve n={n} it={it} score={cur_score:.6f}"
                )
        else:
            # revert changes
            for idx, rev in diffs:
                counts[idx] += rev
            S.remove(i_in)
            S.add(i_out)

    # build the probability vector from the best support
    best_seq = np.zeros(n, dtype=float)
    w = 1.0 / K
    for i in best_S:
        best_seq[i] = w
    # normalise just in case of rounding
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full(n, 1.0 / n)

    # final true score (using the exact definition)
    final_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq)) / (best_seq.sum() ** 2)
    return best_seq, final_score
