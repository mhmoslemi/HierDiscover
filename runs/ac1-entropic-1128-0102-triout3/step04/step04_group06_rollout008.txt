"""
Propose a step‑function using smoothed max‑convolution gradients.

The routine works on the simplex (sum = 1, a_i ≥ 0).  At each iteration
a smoothed “log‑sum‑exp” approximation of the max‑convolution is formed:

    F(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

and its gradient is the convolution of the soft‑max weights with the
reversed step‑function.  An exponentiated‑gradient (mirror descent)
update keeps the vector on the simplex.  The smoothing parameter γ
starts small (broad weighting) and is increased gradually, allowing the
algorithm to first move in a smoother landscape and then focus on the
true active constraints.  After the long descent phase a short
simulated‑annealing hill‑climbing phase attempts local mass‑transfer
moves.  The best sequence encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the aperiodic autocorrelation (linear convolution) of ``a``
    with itself and return both the full array and its maximum entry.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv, conv.max()


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    ``w`` are the (normalized) soft‑max weights of length 2n‑1.
    The gradient components are
        g_i = 2·Σ_k w_k·a_{k−i} .
    Computed by a convolution of ``w`` with the reversed ``a``.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)          # length needed for w * a_rev
    W = np.fft.rfft(w, n=L)
    Ar = np.fft.rfft(a_rev, n=L)
    G_full = np.fft.irfft(W * Ar, n=L)[: 3 * n - 2]
    # slice that corresponds to indices i+n‑1, i = 0..n‑1
    return 2.0 * G_full[(n - 1) : (2 * n - 1)]
