"""
Propose‑candidate for the step‑function optimisation.

Algorithm
---------
1. Initialise a probability vector `a` of length `N≈4800`.  
   If the global `height_sequence_1` is available it is up‑sampled,
   otherwise a random Dirichlet sample is used.

2. **Smoothed gradient descent** – minimise a smooth approximation of
   `max(conv(a))` using a soft‑max (log‑sum‑exp) of the convolution.
   The soft‑max temperature (parameter `lam`) is increased during the
   run, so the optimisation starts with a gentle landscape and ends
   with the true non‑smooth objective.  Nesterov‑style momentum and a
   diminishing step size are employed, and the iterate is always
   projected onto the probability simplex.

3. **Greedy hill‑climbing** – with the remaining time a series of
   small mass‑transfer moves (`ε` from a source index to a destination
   index) are tried.  After each move the full convolution is recomputed
   (FFT) and the move is kept only if it improves the objective.
   This fine‑tunes the solution found by the gradient phase.

4. The best vector encountered is normalised to sum = 1 and returned
   as a plain Python list of floats.  All numbers are non‑negative,
   satisfying the security checks of `evaluate_sequence`.

The whole routine respects the supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Upsample a probability vector by linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        fft_len = 1 << ((L - 1).bit_length())
        W = np.fft.rfft(w, n=fft_len)
        A_rev = np.fft.rfft(a_rev, n=fft_len)
        full = np.fft.irfft(W * A_rev, n=fft_len)[:L]
        offset = len(a_rev) - 1                     # = n‑1
        grad = 2.0 * full[offset:offset + n]

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

        # increase smoothing parameter
        lam = min(lam_max, lam + lam_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing with mass‑transfer moves
# ----------------------------------------------------------------------
def _run_hill_climbing(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       cur_best_seq: np.ndarray,
                       cur_best_score: float,
                       eps_factor: float = 0.30,
                       batch: int = 2000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a random fraction of mass from a source index
    (chosen proportionally to its current mass) to a random destination.
    Keep a move only if it improves the objective.
    """
    a = a0.copy()
    score = cur_best_score
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # source ~ current distribution, destination uniform
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * eps_factor
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # full convolution for the trial vector
        conv = _conv_self(a_prop)
        new_score = 2.0 * n * conv.max()

        if new_score < score:
            a = a_prop
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: it={it:,}, score={new_score:.6f}")

        # occasional occasional resampling of ''src'' distribution
        if it % batch == 0:
            # normalise to avoid drift caused by rounding errors
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Uses a smoothed gradient descent followed by greedy hill‑climbing.
    Returns the best sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # for any legacy calls

    start_time = time.time()
    # small safety margin (the harness adds a tiny overhead)
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Starting vector (use the previous best if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # target length – large enough to approach the continuous optimum
    N = 4800

    if starter is not None:
        a0 = _upsample(starter, N)
    else:
        a0 = rng.random(N)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed gradient descent on the full size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        grad_deadline = time.time() + remaining * 0.75   # use ~75 % of the time
        a_opt, score_opt = _run_smooth_gradient_descent(
            a0, N, grad_deadline, start_time,
            step0=0.5, beta=0.9,
            lam_start=10.0, lam_max=2000.0, lam_inc=0.5
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase‑1 improve: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – greedy hill‑climbing (the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_hill_climbing(
            best_seq, N,
            deadline, start_time,
            best_seq, best_score,
            eps_factor=0.30,
            batch=2000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final hill‑climb improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return as a simple Python list of floats
    return [float(x) for x in best_seq]
