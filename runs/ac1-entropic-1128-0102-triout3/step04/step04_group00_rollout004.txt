def _run_exponential_gradient(a0, n, deadline, start_time, eta0=0.5, max_iter=10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it=0
    while time.time() < deadline and it < max_iter:
        it+=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] expgrad improve: n={n}, iter={it}, score={cur_score:.6f}")
        grad = _subgradient_full(a, conv)
        step = eta0 / math.sqrt(it)
        # limit exponent to avoid overflow
        expo = -step * grad
        # Clip expo
        expo = np.clip(expo, -50, 50)
        a = a * np.exp(expo)
        # Renormalize to sum 1
        total = a.sum()
        if total>0:
            a /= total
        else:
            a = np.full_like(a, 1.0/a.size)
    return best_a, best_score

    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp/start_temp)**(it/steps)
        # compute gradient based on current max conv
        conv = _conv_self(a)
        max_val = conv.max()
        active = np.where(conv >= max_val * (1 - 1e-12))[0]

        src = np.argmax(grad)  # high
        # choose dst as index with minimal gradient (or random)
        dst_candidates = np.where(grad == grad.min())[0]
        dst = rng.choice(dst_candidates) if dst_candidates.size>0 else rng.integers(n)
        if src == dst or a[src] < 1e-12:

            best_seq = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm – multi‑stage search
------------------------------

*Stage 1* – many short random restarts on a modest size (n≈600).  For each
restart we run either a projected (Nesterov) gradient descent or an
exponentiated‑gradient descent on the simplex.  The best vector is kept.

*Stage 2* – upscale the best vector to n≈2400 and run a longer
exponentiated‑gradient descent (mirror descent with KL‑projection, i.e.
normalisation).

*Stage 3* – upscale to n≈4800 and continue the exponentiated‑gradient
descent.

*Stage 4* – if time permits, upscale once more to n≈7200 and run another
exponentiated‑gradient phase (more degrees of freedom can lower the
objective).

*Stage 5* – final “targeted” simulated‑annealing refinement.  At each
iteration we move a small amount of mass from the index with the largest
sub‑gradient (most responsible for the current maximal convolution entry)
to an index with the smallest sub‑gradient, accepting downhill moves
always and uphill moves with a temperature schedule.

All stages respect the global time budget (≈1000 s).  The routine returns
the best sequence found, normalised to a unit ℓ₁‑sum, as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (linear, non–circular)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(convolution) w.r.t. a
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) (the active set is taken with a relative
    tolerance).  Returns a vector of the same shape as a.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # contribution of a_j to ∂/∂a_i and vice‑versa
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Projected gradient descent (the original version)
# ----------------------------------------------------------------------
def _run_projected_gradient(a0: np.ndarray, n: int,
                           deadline: float, start_time: float,
                           beta: float = 0.9,
                           step0: float = 0.3,
                           max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              eta0: float = 0.5,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Mirror descent with KL‑divergence: a ← a ⊙ exp(‑η·g) and renormalise.
    Returns the best vector encountered and its score (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        step = eta0 / math.sqrt(it)

        # avoid overflow/underflow in the exponential
        expo = -step * grad
        expo = np.clip(expo, -50.0, 50.0)
        a = a * np.exp(expo)

        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray, n: int,
                            deadline: float, start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-6,
                            steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Move a random fraction (up to 30 %) of mass from the index with the
    largest sub‑gradient to the index with the smallest sub‑gradient.
    Uses a geometric temperature schedule and keeps the global best.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # obtain current sub‑gradient (active set)
        conv = _conv_self(a)
        grad = _subgradient_full(a, conv)

        src = int(np.argmax(grad))
        dst_candidates = np.where(grad == grad.min())[0]
        if dst_candidates.size > 0:
            dst = int(rng.choice(dst_candidates))
        else:
            dst = rng.integers(n)

        if src == dst or a[src] < 1e-12:
            # fallback to a more random move
            src = int(rng.choice(n, p=a / a.sum()))
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation for up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate linearly to a new length and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  Returns the best sequence (as a list of floats)
    found within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – the harness will cut off after ``budget_s`` sec
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global ``height_sequence_1``)
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – many short restarts on a modest size (n≈600)
    # ------------------------------------------------------------------
    n_small = 600
    slice_time = max(20.0, 0.10 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time
    for restart in range(5):
        if starter is not None and restart == 0:
            a0_small = _upsample(starter, n_small)
        else:
            a0_small = np.random.rand(n_small)
            a0_small /= a0_small.sum()

        # Alternate between the two optimisation kernels
        if restart % 2 == 0:
            a_opt, sc = _run_projected_gradient(
                a0_small, n_small, slice_deadline, start_time,
                beta=0.85, step0=0.4
            )
        else:
            a_opt, sc = _run_exponential_gradient(
                a0_small, n_small, slice_deadline, start_time,
                eta0=0.5
            )

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage‑1 improve (restart={restart}) score={sc:.6f}")

        if time.time() >= deadline:
            break

    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 2 – upsample and run a longer exponential‑gradient phase
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    remaining = deadline - time.time()
    grad_time_mid = max(30.0, 0.55 * remaining)
    grad_deadline_mid = time.time() + grad_time_mid

    a_mid_opt, sc_mid = _run_exponential_gradient(
        a_mid, n_mid, grad_deadline_mid, start_time,
        eta0=0.4
    )
    if sc_mid < best_score:
        best_score = sc_mid
        best_seq = a_mid_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage‑2 improve score={sc_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upsample to n≈4800 and continue exponential‑gradient descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    remaining = deadline - time.time()
    grad_time_big = max(30.0, 0.45 * remaining)
    grad_deadline_big = time.time() + grad_time_big

    a_big_opt, sc_big = _run_exponential_gradient(
        a_big, n_big, grad_deadline_big, start_time,
        eta0=0.3
    )
    if sc_big < best_score:
        best_score = sc_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage‑3 improve score={sc_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional extra up‑sampling to n≈7200 if time remains
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # give the remaining budget to a final large‑scale refinement
        n_extra = 7200
        a_extra = _upsample(best_seq, n_extra)
        extra_deadline = deadline
        a_extra_opt, sc_extra = _run_exponential_gradient(
            a_extra, n_extra, extra_deadline, start_time,
            eta0=0.2
        )
        if sc_extra < best_score:
            best_score = sc_extra
            best_seq = a_extra_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage‑4 improve (n={n_extra}) score={sc_extra:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – targeted simulated‑annealing refinement (uses whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Decide how many annealing steps we can afford
        # Roughly 1 µs per step → we aim for a few hundred k steps
        leftover = deadline - time.time()
        steps = int(min(500_000, max(50_000, leftover * 400_000)))  # heuristic
        a_ref, sc_ref = _run_targeted_annealing(
            best_seq, best_seq.shape[0],
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6,
            steps=steps
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain list of Python floats
    return [float(x) for x in best_seq]
