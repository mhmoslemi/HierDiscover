Implementation details for simplex projection:

def simplex_projection(v):
    # projects onto simplex {x >= 0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

"""
Propose a step‑function with a low autocorrelation score.

The algorithm consists of three phases:

1.  Initialise from a previously discovered construction (if available) or from
    a uniform distribution on a random support.
2.  Run a smooth‑max gradient descent on the simplex.
    The non‑smooth evaluation  max(conv)  is replaced by a softmax
    approximation (log‑sum‑exp).  The gradient of the smoothed objective is
    computed analytically (it is a convolution of the current vector with the
    softmax weights).  After each gradient step the vector is projected back
    onto the probability simplex (non‑negative entries that sum to 1).
    A small random mass transfer is added occasionally to escape plateaux.
    The schedule for the softmax temperature (β) and the step size is
    gradually tightened, allowing the optimisation to focus on the true
    maximum as the search proceeds.
3.  If the time budget has not been exhausted and the target score has not
    been reached, a fast random‑support search is performed as a safety net.

All intermediate vectors are kept normalised (ℓ₁‑norm = 1) because the
evaluation function is invariant under scaling.  The routine respects the
1000 s time limit and prints progress information.
"""

import math
import time
import numpy as np
from scipy.optimize import linprog  # only used as a very light fallback
from scipy.sparse import coo_matrix


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` – no safety checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def random_support_vector(n: int, m: int) -> np.ndarray:
    """Uniform weight on a random support of size ``m``."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Core optimiser – smooth‑max gradient descent with occasional random moves
# ----------------------------------------------------------------------
def smooth_max_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_score: float,
) -> tuple[np.ndarray, float]:
    """Run the gradient descent from ``a0`` until time runs out or the target is met.

    Returns the best vector seen and its score.
    """
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    # parameters (tuned empirically)
    step0 = 0.5           # initial step size
    beta = 5.0            # initial soft‑max temperature
    beta_max = 300.0
    beta_mult = 1.5       # increase every 1000 iters
    it = 0

    # pre‑compute size for FFTs (auto‑correlation)
    size_autocorr = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((n + 2 * n - 2).bit_length())  # a (len n) * w (len 2n‑1)

    while time.time() < deadline:
        # ---------- autocorrelation ----------
        fft_a = np.fft.rfft(a, size_autocorr)
        conv = np.fft.irfft(fft_a * fft_a, size_autocorr)[: 2 * n - 1]

        # ---------- smoothed max ----------
        max_c = conv.max()
        # stability for the exponentials
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()               # softmax weights, sum to 1

        # ---------- gradient ----------
        fft_a2 = np.fft.rfft(a, size_grad)
        fft_w = np.fft.rfft(w, size_grad)
        grad_full = np.fft.irfft(fft_a2 * fft_w, size_grad)[:n]    # length n
        grad = 2.0 * grad_full

        # ---------- step ----------
        eta = step0 / np.sqrt(it + 1.0)   # diminishing step size
        a_new = a - eta * grad
        a_new = simplex_projection(a_new)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.02:
            nz = np.where(a_new > 1e-12)[0]
            if nz.size > 0:
                i = np.random.choice(nz)
                j = np.random.randint(n)
                if i != j:
                    delta = 0.05 * a_new[i] * np.random.rand()
                    cand = a_new.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = simplex_projection(cand)
                    cand_val = fast_score(cand)
                    if cand_val < best_val:
                        best_val, best_a = cand_val, cand.copy()
                        a_new = cand
                        print(f"[rand move] iter={it} improved to {cand_val:.6f}")

        # ---------- evaluation ----------
        val = fast_score(a_new)
        if val < best_val - 1e-9:
            best_val, best_a = val, a_new.copy()
            print(f"[grad] iter={it} new best {val:.6f}")
            if val <= target_score:
                print("[goal] target reached inside gradient descent")
                return best_a, best_val

        # prepare next iteration
        a = a_new
        it += 1

        # slowly increase the temperature β (focuses more on the true max)
        if it % 1000 == 0 and beta < beta_max:
            beta = min(beta * beta_mult, beta_max)

    return best_a, best_val

def random_support_search(
    n: int,
    deadline: float,
    target_score: float,
    current_best_seq: np.ndarray,
    current_best_val: float,
) -> tuple[np.ndarray, float]:
    """Explore many random supports with uniform weights."""
    support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
    best_seq = current_best_seq
    best_val = current_best_val

    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[rand sup] trying support size {m}")
        attempts = 0
        while time.time() < deadline and attempts < 2500:
            attempts += 1
            a = random_support_vector(n, m)
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[rand sup] m={m} att={attempts} score={val:.6f}")
                if val <= target_score:
                    print("[goal] target reached in random support search")
                    return best_seq, best_val
        print(f"[rand sup] size {m} finished, best so far {best_val:.6f}")

    return best_seq, best_val
