def _grad_softmax(a, conv, T):
    # conv shape length 2n-1
    # stable softmax
    maxc = conv.max()
    exps = np.exp((conv - maxc) / T)
    w = exps / exps.sum()
    # compute weighted gradient
    rev_a = a[::-1]
    fft_len = 1 << ((len(rev_a) + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, fft_len)
    W = np.fft.rfft(w, fft_len)
    conv_w = np.fft.irfft(A * W, n=fft_len)[:len(rev_a) + len(w) - 1]
    grad = 2.0 * conv_w[len(a)-1:len(a)-1+len(a)]
    return grad

"""
Improved optimisation for the step‑function problem.

Main ideas
~~~~~~~~~
* Use a *smooth* approximation of the max‑convolution (soft‑max) to obtain a
  well‑behaved gradient.  The temperature is annealed during the run, which
  yields a gentle transition from a smooth landscape to the true max.
* Projected stochastic gradient descent with Nesterov momentum on the
  probability simplex.
* Progressive up‑sampling (800 → 2400 → 4800 → 9600) – each stage receives a
  time share proportional to its length.
* A final cheap simulated‑annealing walk reinforces any remaining improvement.
* Multiple starts: if a previous construction (`height_sequence_1`) is
  available it is used as the seed; otherwise a random simplex point is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of  Σ_k w_k(conv_k) * conv_k  with  w_k = exp(conv_k / T) / Σ.
    The factor 2 comes from the derivative of aᵀJ_k a (the symmetric Toeplitz
    matrix for shift k).  The result is already projected onto the simplex
    later – this function only supplies the raw gradient.
    """
    maxc = conv.max()
    # stable soft‑max
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    # weighted gradient: 2 * (rev(a) * w) restricted to the central n entries
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)   # length = n + (2n-1) - 1 = 3n-2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]   # central block corresponds to i‑range
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected (Nesterov) gradient descent with soft‑max gradient
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Performs many iterations of projected gradient descent, using a smooth
    approximation of the max‑convolution.  Returns the best point found and
    its (scaled) evaluation score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # schedule for the temperature (annealing)
    T_start = 2.0e-2
    T_end = 5.0e-4
    # we aim at ~20 000 iterations for the biggest stage; the exponent below
    # automatically adapts when we finish earlier.
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature decays roughly exponentially with iteration count
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient wrt the *smooth* max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1    # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs progressive up‑sampling
    and a soft‑max gradient descent, and finishes with a short annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definitions (length, fraction of remaining time)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]          # progressive enlargement
    total_weight = sum(stage_lengths)

    # Allocate ~15 % of the total budget for the final annealing phase.
    total_time = deadline - start_time
    anneal_budget = max(30.0, 0.15 * total_time)
    grad_budget = total_time - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time share for this stage (proportional to its length)
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for the current size
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run the soft‑max gradient descent on the current length
        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.6, beta=0.9, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works with the final length we ended up with.
    # ------------------------------------------------------------------

            best_score = score_ref

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:

    # final normalisation (tiny drift may happen because of rounding)

    s = best_seq.sum()

        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
