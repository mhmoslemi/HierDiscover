"""
Proposes a non‑negative step‑function (list of heights) that minimises the
`evaluate_sequence` score.

Algorithm
---------
1. **Parameterized warm‑start** – search a one‑parameter family of symmetric
   power‑law shapes `a_i ∝ ((i+0.5)*(N‑0.5‑i))^p` for a modest length (N≈800).
   This quickly yields a shape that already improves on the uniform
   distribution.

2. **Progressive up‑sampling + projected gradient descent** – repeatedly
   linearly up‑sample the current best sequence to larger lengths
   (800 → 2400 → 4800 → 9600).  For each length a few random restarts of a
   projected (Nesterov‑style) gradient method are executed.  The gradient is
   the exact sub‑gradient of the *max* entry of the self‑convolution
   (no averaging over active entries) which provides a sharper descent
   direction.

3. **Final simulated‑annealing refinement** – a long mass‑transfer annealing
   walk (move a small ε from one index to another) finishes the search.
   The annealing schedule is exponential and runs until the remaining time
   budget expires.

All phases respect the global time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the *maximum* convolution entry (no averaging)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of the same shape as a.
    """
    n = a.shape[0]
    k = int(conv.argmax())                 # index of the maximal entry
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g = np.zeros_like(a)
    # contribution a_j to g_i and a_i to g_j (includes diagonal case correctly)
    g[i] += a[j]
    g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov‑style momentum)
# ----------------------------------------------------------------------
def _run_proj_grad_max(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       beta: float = 0.9,
                       step0: float = 0.5,
                       max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent using the exact max‑convolution sub‑gradient.
    Returns the best point found and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _max_subgradient(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size decays a little faster than 1/√t (exponent 0.6)
        step = step0 / ((it) ** 0.6)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Run several random restarts of the gradient method and keep the best
# ----------------------------------------------------------------------
def _run_multiple_restarts(a0: np.ndarray, n: int,
                           deadline: float, start_time: float,
                           restarts: int = 3) -> tuple[np.ndarray, float]:
    """
    Execute `restarts` independent gradient descents (first one from a0, the
    others from fresh random points) inside the time window up to `deadline`.
    Returns the overall best sequence and its scaled score.
    """
    remaining_time = deadline - time.time()
    if remaining_time <= 0.0:
        return None, float("inf")

    best_a = None
    best_score = float("inf")
    for r in range(restarts):
        # allocate a fair share for the remaining restarts
        now = time.time()
        time_left = deadline - now
        if time_left <= 0.0:
            break
        # split remaining time evenly among remaining restarts
        sub_deadline = now + time_left * (1.0 / (restarts - r))

        if r == 0:
            a_init = a0.copy()
        else:
            a_init = np.random.rand(n)
            a_init /= a_init.sum()

        a_opt, sc_opt = _run_proj_grad_max(a_init, n,
                                           sub_deadline, start_time,
                                           beta=0.9, step0=0.5)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 800000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing with a geometric temperature schedule."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]

        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`. The routine respects the given time budget,
    builds a good initial shape via a one‑parameter sweep, refines it by
    progressive up‑sampling with gradient descent, and finishes with a
    long annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    best_seq = None          # numpy array, normalised to sum==1
    best_score = float("inf")   # scaled score = 2·n·max(conv)

    # ------------------------------------------------------------------
    # Helper to update the current global best (scaled score)
    # ------------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score
        if seq is None:
            return
        # the sequence is already normalised; scaled score is 2·n·max(conv)
        n = seq.shape[0]
        conv = _conv_self(seq)
        scaled = 2.0 * n * conv.max()
        if scaled < best_score:
            best_score = scaled
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST: n={n}, score={scaled:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 – inexpensive parametric sweep on a modest size (N=800)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        N0 = 800
        p_vals = np.linspace(-0.90, -0.10, 30)   # negative exponents give a “U’’ shape
        local_best = None
        local_score = float("inf")
        for p in p_vals:
            i = np.arange(N0)
            # symmetric coordinate product, shifted to avoid zero at the borders
            coords = (i + 0.5) * (N0 - 0.5 - i)
            a = np.power(coords + 1e-12, p)       # safe power for negative p
            s = a.sum()
            if s == 0:
                continue
            a /= s
            # scaled score (no need to call evaluate_sequence – sum==1)
            conv = _conv_self(a)
            sc = 2.0 * N0 * conv.max()
            if sc < local_score:
                local_score = sc
                local_best = a.copy()
        # also consider the uniform distribution as a fallback
        a_uni = np.full(N0, 1.0 / N0)
        conv = _conv_self(a_uni)
        sc_uni = 2.0 * N0 * conv.max()
        if sc_uni < local_score:
            local_score = sc_uni
            local_best = a_uni.copy()

        _update_global(local_best)

    # ------------------------------------------------------------------
    # Progressive up‑sampling + projected gradient descent
    # ------------------------------------------------------------------
    # Target lengths (increase roughly by factor 3 each stage)
    stage_lengths = [800, 2400, 4800, 9600]

    # Reserve ~15 % of the total budget for the final annealing phase
    total_time = deadline - start_time
    anneal_time = max(30.0, 0.15 * total_time)
    grad_time_total = total_time - anneal_time
    total_weight = sum(stage_lengths)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time proportional to the size of the stage
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_time_total * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run a few random restarts of the gradient method
        a_opt, sc_opt = _run_multiple_restarts(a0, n, stage_deadline, start_time,
                                               restarts=3)
        if a_opt is not None:
            _update_global(a_opt)

        # update remaining gradient budget (shrink from the front)
        elapsed = time.time() - start_time
        grad_time_total = max(0.0, deadline - start_time - anneal_time - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase (uses whatever length we ended with)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, sc_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6, steps=800000
        )
        _update_global(a_ref)

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list for the harness
    return [float(x) for x in best_seq]
