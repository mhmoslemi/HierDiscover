def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # attempt to load global
    try: prev = np.array(height_sequence_1, dtype=float)
    except: prev = None
    if prev is not None:
        prev = np.maximum(prev, 0)
        if prev.sum()>0:
            prev = prev / prev.sum()
    # schedule
    max_n = 8000
    n = 300
    sizes = []
    while n <= max_n:
        sizes.append(n)
        n = int(n * 1.7)
    # ensure final includes max_n if not exactly
    if sizes[-1] < max_n:
        sizes.append(max_n)
    # iterate sizes
    for stage, n in enumerate(sizes, 1):
        if time.time() >= deadline:
            break
        print(f"[{time.time()-start:.2f}s] Stage {stage}/{len(sizes)} n={n}")

        # generate candidate list
        candidates = []
        # upsample previous best if exists and length different
        if best_seq is not None and best_seq.shape[0] != n:
            x_old = np.linspace(0, 1, best_seq.shape[0])
            x_new = np.linspace(0, 1, n)
            up = np.interp(x_new, x_old, best_seq)
            up = np.maximum(up, 0)
            up_sum = up.sum()
            if up_sum == 0:
                up = np.full(n, 1.0/n)
            else:
                up = up / up_sum
            candidates.append(up)

        # include previous construction if any
        if prev is not None:
            if prev.shape[0] != n:
                x_old = np.linspace(0, 1, prev.shape[0])
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, prev)
                up = np.maximum(up, 0)
                s = up.sum()
                if s == 0:
                    up = np.full(n, 1.0/n)
                else:
                    up = up / s
                candidates.append(up)
            else:
                candidates.append(prev.copy())

        # deterministic shapes
        candidates.append(np.full(n, 1.0/n))  # uniform

        idx = np.arange(n)
        # triangular shape (peaked)
        tri = np.minimum(idx+1, n - idx)
        candidates.append(tri/tri.sum())

        # Hann
        if n>1:
            hann = 0.5*(1 - np.cos(2*np.pi*idx/(n-1)))
            candidates.append(hann/hann.sum())
        else:
            candidates.append(np.full(1, 1.0))

        # U-shaped (inverse product)
        inv = 1.0/((idx+1)*(n-idx))
        candidates.append(inv/inv.sum())

        # sin^2 shape
        sin2 = np.sin(np.pi*idx/(n-1))**2
        candidates.append(sin2/sin2.sum())

        # random Dirichlet draws
        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # optimise each candidate
        for cid, cand in enumerate(candidates, 1):
            if time.time() >= deadline:
                break
            # compute time estimate per iteration for current n
            est_iter_time = 5e-4 * (n/1800.0) + 1e-4

"""
Simple multi‑scale Adam optimiser for the auto‑convolution max‑norm.

We keep the vector on the probability simplex (∑a_i = 1, a_i ≥ 0) and
optimise the smooth approximation of the objective

    2·n·max(conv(a,a))

by Adam.  After each Adam step we **project onto the simplex and enforce
symmetry** (a = rev(a)) – the objective is invariant under reversal, and
symmetrising often helps to flatten the autocorrelation.

The optimiser is run on a schedule of increasing problem sizes:
starting from a modest length we up‑sample the current best shape,
optimise it, and repeat.  A few deterministic shapes (uniform,
triangular, Hann, a U‑shaped inverse‑product, sin²) and a handful of
random Dirichlet starts are tried at every scale.  The remaining time
budget is spent on a long refinement of the globally best sequence.

The function returns a plain list of heights (any scaling would give the
same score because the evaluation function is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # all entries are equal – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Full convolution a*a (real) and its maximum value.  a is assumed to sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(
    a: np.ndarray, w: np.ndarray, fft_len_grad: int
) -> np.ndarray:
    """
    Gradient of Σ_k w_k * (a*a)_k  w.r.t. a.
    Implements   g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT for speed.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize_sym(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.1,
    temp_end: float = 5e-5,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv) on the simplex, with a symmetry
    projection after each step.

    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weighting over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ---- projection onto the simplex ----
        a = _project_simplex(a)

        # ---- enforce symmetry (a_i = a_{n-1-i}) ----
        a = (a + a[::-1]) * 0.5
        # re‑normalise (projected simplex may have moved a tiny bit)
        s = a.sum()
        if s > 0:
            a = a / s

    return a, best_a, best_score
