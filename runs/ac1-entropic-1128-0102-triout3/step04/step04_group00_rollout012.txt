"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Projection** – Euclidean projection onto the simplex {x≥0, Σx=1}.
2. **Convolution** – linear self‑convolution via FFT (O(n log n)).
3. **Smooth‑max gradient** – instead of the hard max we minimise a
   smooth approximation
        Φα(a)= (1/α)·log Σ_k exp(α·(a∗a)[k])
   (α grows from a small value to a large one).  The gradient is
   g = 2·(a⋆w) where w_i ∝ exp(α·(a∗a)[i]) are soft‑max weights.
   The correlation a⋆w is evaluated by an FFT‑based convolution.
4. **Adam update** – adaptive step‑size with bias‑corrected moments,
   followed by simplex projection.
5. **Multi‑stage optimisation** – three stages of increasing length
   (800 → 2400 → 4800).  In each stage we run the smooth‑max descent for
   the whole remaining time budget (leaving a short margin for the final
   simulated‑annealing refinement).
6. **Simulated‑annealing refinement** – mass‑transfer moves
   (transfer a random fraction of a source entry to a random destination)
   with a geometric temperature schedule.
7. **Return** – the best vector found (normalised to Σ=1) as a plain
   Python list.  All numbers are non‑negative, so ``evaluate_sequence``
   will accept the result.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the convolution
# ------------------------------------------------------------
def _grad_smoothmax(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of  Φα(a)= (1/α)·log Σ exp(α·conv) .
    Returns ∇Φα(a) = 2·(a⋆w) where w_i ∝ exp(α·conv_i).
    """
    # soft‑max weights (sum = 1)
    max_val = conv.max()
    # exp(α·(conv – max)) is numerically stable (max term becomes 1)
    exp_part = np.exp(alpha * (conv - max_val))
    w = exp_part / exp_part.sum()          # shape = (2n‑1,)

    # correlation a⋆w via FFT (a ⋆ w)[m] = Σ_j a_j w_{j+m}
    w_rev = w[::-1]                        # reverse the weight vector
    n = a.shape[0]
    L = w_rev.shape[0]                     # = 2n‑1
    fft_len = 1 << ((n + L - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(w_rev, fft_len)
    corr_full = np.fft.irfft(A * W, n=fft_len)[: n + L - 1]

    # indices that give the correlation for lags 0 … n‑1
    idx = (L - 1) - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad


# ------------------------------------------------------------
# Adam‑based smooth‑max descent with simplex projection
# ------------------------------------------------------------
def _run_smoothmax_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    *,
    alpha_start: float = 1.0,
    alpha_end: float = 100.0,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max of the self‑convolution.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smoothmax improve: n={n}, iter={it:,}, "
                f"score={cur_score:.6f}"
            )

        # progress fraction determines the current α
        prog = (time.time() - start_time) / (deadline - start_time)
        prog = min(max(prog, 0.0), 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        # gradient of the smooth‑max
        grad = _grad_smoothmax(a, conv, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    *,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing loop that moves a random ε‑fraction of mass
    from a heavy index to a random destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source chosen proportionally to current mass (bias towards large entries)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score

# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a three‑stage smooth‑max Adam descent,
    followed by a short simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 1 – modest size (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    slice_time = max(30.0, 0.08 * (deadline - time.time()))  # at least 30 s
    slice_deadline = time.time() + slice_time

    for restart in range(3):
        if starter is not None and restart == 0:
            a0_small = _upsample(starter, n_small)
        else:
            a0_small = np.random.rand(n_small)
            a0_small /= a0_small.sum()

        a_small, score_small = _run_smoothmax_descent(
            a0_small,
            n_small,
            slice_deadline,
            start_time,
            alpha_start=1.0,
            alpha_end=12.0,
            lr=0.07,
        )
        if score_small < best_score:
            best_score = score_small
            best_seq = a_small.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage 1 improve: restart={restart}, score={score_small:.6f}"
            )
        if time.time() >= deadline:
            break

    # fallback to uniform if nothing was found
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 2 – medium size (n≈2400)
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    remaining = deadline - time.time()
    # reserve a small chunk for the final annealing
    grad_time_mid = max(20.0, 0.65 * remaining)
    grad_deadline_mid = time.time() + grad_time_mid

    a_mid_opt, score_mid = _run_smoothmax_descent(
        a_mid,
        n_mid,
        grad_deadline_mid,
        start_time,
        alpha_start=5.0,
        alpha_end=50.0,
        lr=0.05,
    )
    if score_mid < best_score:
        best_score = score_mid
        best_seq = a_mid_opt.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_mid:.6f}"
        )

    # ------------------------------------------------------------------
    # Stage 3 – large size (n≈4800)
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    remaining = deadline - time.time()
    # allocate almost all remaining time to gradient descent,
    # keep a few seconds for the annealing step
    grad_time_big = max(10.0, remaining - 8.0)
    grad_deadline_big = time.time() + grad_time_big

    a_big_opt, score_big = _run_smoothmax_descent(
        a_big,
        n_big,
        grad_deadline_big,
        start_time,
        alpha_start=30.0,
        alpha_end=200.0,
        lr=0.03,
    )
    if score_big < best_score:
        best_score = score_big
        best_seq = a_big_opt.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] stage 3 improve: score={score_big:.6f}"
        )

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
