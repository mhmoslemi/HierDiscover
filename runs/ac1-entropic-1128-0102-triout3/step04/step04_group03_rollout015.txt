"""
Fast entropy‑mirror descent for flattening the self‑convolution.

The routine builds a step‑function (a non‑negative vector that sums to 1)
by iteratively minimizing a smooth approximation of the peak of the
linear self‑convolution `a*a`.  The smooth max is realised with a
soft‑max (temperature β) and the simplex constraint is enforced by an
exponentiated‑gradient (entropy‑mirror) update:

    a_i ← a_i · exp(‑η·∇_i) ,      a ← a / Σ a

The gradient of the soft‑max of the convolution can be written as a
single convolution, so each iteration costs only two FFTs (O(n log n)).
The algorithm works on a series of increasingly large lengths
(800 → 2400 → 4800 → 9600).  After the main descent a cheap simulated‑
annealing refinement moves a small amount of mass between two positions
to squeeze out any remaining peaks.

The final list is returned as plain Python floats and should achieve an
evaluation score close to the theoretical lower bound ≈ 1 (the previous
best was ≈ 1.514)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a, length 2·len(a)‑1, using real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Soft‑max (entropy) weights for a smooth maximum of the convolution.
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(beta·b_k)."""
    # shift for numerical stability
    logits = b * beta
    shift = logits.max()
    exp_logits = np.exp(logits - shift)
    w = exp_logits / exp_logits.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth maximum.
# For a fixed temperature β the gradient is
#   g_i = 2·Σ_j a_j·w_{i+j}
# where w are the soft‑max weights of the current convolution.
# This is a single (linear) convolution of a_rev and w.
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Compute gradient of the β‑soft‑max of a*a and also return the raw convolution.
    Returns (grad, conv) where grad has shape a.shape.
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _conv_self_fft(a)                     # length 2n‑1

    # 2) soft‑max weights on the convolution entries
    w = _softmax_weights(conv, beta)             # length 2n‑1, sum 1

    # 3) gradient = 2 * (a_rev * w)  restricted to n entries
    a_rev = a[::-1]
    L = 1 << ((len(a_rev) + len(w) - 1).bit_length())
    Ar = np.fft.rfft(a_rev, n=L)
    Wr = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(Ar * Wr, n=L)        # length len(a_rev)+len(w)-1 = 3n‑2

    start = n - 1                               # index where the dot‑product window aligns
    grad = 2.0 * conv_rev[start:start + n]       # shape (n,)

    return grad, conv


# ----------------------------------------------------------------------
# Entropy‑mirror (exponentiated) update on the simplex.
# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a multiplicative (exponentiated‑gradient) update and re‑normalise.
    The update is a_i ← a_i·exp(‑η·grad_i).
    """
    factor = np.exp(-eta * grad)
    a_new = a * factor
    # protect against underflow – keep a tiny floor before renormalisation
    eps = 1e-300
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Gradient descent on the simplex using the smooth‑max gradient.
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray, n: int,
                     deadline: float, start_time: float,
                     eta0: float = 0.6,
                     beta0: float = 1.0,
                     beta_max: float = 80.0,
                     max_iter: int = 10 ** 7):
    """
    Exponentiated‑gradient descent with increasing soft‑max temperature.
    Returns the best vector found and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # temperature schedule – slowly increase β to sharpen the max
        beta = min(beta_max, beta0 * math.log(it + 2))

        grad, conv = _smooth_gradient(a, beta)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad n={n} iter={it} "
                  f"β={beta:.2f} score={cur_score:.6f}")

        # step‑size schedule (standard 1/√t decay)
        eta = eta0 / math.sqrt(it)

        a = _exp_update(a, grad, eta)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 400000) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass between two positions, accepting uphill moves."""
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.07   # move up to 7 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} iter={it} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a sequence to a new length and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        stage_deadline = time.time() + stage_time

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.full(n, 1.0 / n)                # uniform start

        # run exponentiated‑gradient descent on this size
        a_opt, score_opt = _run_exp_gradient(
            a0, n, stage_deadline, start_time,
            eta0=0.7, beta0=0.8, beta_max=70.0,
            max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improvement: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=600000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improvement: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny numeric drift may have occurred)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # optional logging of the final evaluation (uses the supplied function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final evaluation score = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
