"""
A fast iterative optimizer for the step‑function bound.

The algorithm works in three phases:

1. **Adam on the probability simplex** –  
   The coefficients are represented by soft‑max logits `x`.  
   At each iteration we evaluate the self‑convolution via FFT,
   compute the sub‑gradient of the peak of the convolution,
   back‑propagate it to the logits and perform an Adam update.
   The portfolio of the current best vector is kept.

2. **Progressive up‑sampling** –  
   The best vector from the previous phase is linearly interpolated to a
   larger size and the Adam optimisation is continued.  This lets the
   optimiser first find a good coarse shape and then refine it with more
   degrees of freedom.

3. **Aggressive annealing** –  
   A simple mass‑transfer simulated‑annealing step (move a fraction of the
   mass from a high‑mass index to a random index) explores the remaining
   neighbourhood and often pushes the bound a little further.

All steps respect the supplied time budget (`budget_s`).  The final
sequence is normalised to sum = 1, contains only non‑negative numbers and
is returned as a plain Python list ready for `evaluate_sequence`.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: stable soft‑max (maps logits → a probability vector)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Return soft‑max of x (numerically stable)."""
    x_max = np.max(x)
    e = np.exp(x - x_max)
    s = e.sum()
    if s == 0:
        # fallback to uniform
        return np.full_like(x, 1.0 / x.size)
    return e / s


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a.
    All indices k for which conv[k] >= max(conv)*(1‑rel_tol) are taken as
    active and their contributions summed (no averaging).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    # Loop over active convolution indices
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (via soft‑max parametrisation)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              lr: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              epsilon: float = 1e-8,
              rel_tol: float = 1e-12,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv(a)) with a = softmax(x).
    Returns the best vector (probability distribution) and its score.
    """
    # initialise logits from the starting distribution (avoid log(0))
    x = np.log(a0 + 1e-16)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        a = _softmax(x)                     # a >=0, sum=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad_a = _subgradient_full(a, conv, rel_tol=rel_tol)

        # back‑prop to logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        x = x - step

    if best_a is None:
        # fallback (should never happen)
        best_a = _softmax(x)
        best_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing.
    At each step a fraction of mass is moved from a source index (chosen
    proportionally to its current mass) to a random destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportional to its mass (helps to modify large entries)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        new_score = 2.0 * n * _conv_self(a_new).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_new
            score = new_score

        # record global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n`, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 0.7

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 0 – coarse optimisation on a modest size
    # ------------------------------------------------------------------
    n0 = 2000
    # Allocate roughly 12 % of the total budget (or at least 30 s)
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    num_restarts = 4
    for restart in range(num_restarts):
        if time.time() >= deadline:
            break

        if starter is not None and restart == 0:
            a0 = _upsample(starter, n0)
        else:
            # Dirichlet with concentration <1 to get spiky starts
            a0 = np.random.gamma(shape=0.5, scale=1.0, size=n0)
            a0 /= a0.sum()

        a_opt, score_opt = _run_adam(
            a0, n0, slice_deadline, start_time,
            lr=0.07, beta1=0.9, beta2=0.999, epsilon=1e-8,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={restart}, score={score_opt:.6f}")

    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upscale and continue Adam optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4000
    a1 = _upsample(best_seq, n1)
    remaining = deadline - time.time()
    # give ~35 % of the remaining time to this phase

    a1_opt, score1 = _run_adam(
        a1, n1, grad_deadline_1, start_time,
        lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – further upscale and optimise
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 8000

    remaining = deadline - time.time()
    # allocate ~30 % of the remaining time to gradient, rest to annealing

        a2, n2, grad_deadline_2, start_time,

    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement with the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, best_seq.shape[0],
            anneal_deadline, start_time,
            best_score, best_seq,

            steps=250_000

        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------

    best_seq = np.asarray(best_seq, dtype=float)

    # Optional sanity check (does not affect the returned value)

        print(f"[{time.time() - start_time:6.2f}s] final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
