def _grad_smoothed(a, conv, alpha):
    # conv already computed
    # compute weights
    exp_term = np.exp(alpha * (conv - conv.max()))   # subtract max for numerical stability
    w = exp_term / exp_term.sum()
    # compute gradient: g = 2 * cross_correlation(a, w)
    # cross-correlation equivalent to convolution of a with reversed w.
    n = a.shape[0]
    L = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv_aw = np.fft.irfft(A * W_rev, n=L)[: n + len(w) - 1]
    # cross-correlation values for i = 0..n-1: g_i = 2 * conv_aw[len(w)-1 - i] 
    g = 2.0 * conv_aw[len(w)-1 - np.arange(n)]
    return g

def _run_exponential_gradient(a0, n, deadline, start_time, alpha_schedule, eta0=0.3, max_iter=10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    alpha_index = 0
    total_alphas = len(alpha_schedule)
    # schedule: for each alpha, run certain number of iterations or time.
    # We'll just adapt: we can increase alpha linearly after a certain number of iterations, maybe after every 5000 iterations.
    # Or use gradually increasing: exponent = max(1, int(it / steps_per_alpha)).
    steps_per_alpha = 20000  # maybe adapt
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convex and gradient
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgd improve: n={n}, iter={it}, score={cur_score:.6f}")

        # Determine current alpha
        alpha = alpha_schedule[min(alpha_index, total_alphas-1)]

        # compute gradient of smoothed max
        # weights w
        # subtract max for stability
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)  # shape 2n-1
        w = exp_scaled / exp_scaled.sum()
        # cross-correlation gradient = 2 * cross_corr(a, w)
        # compute with FFT
        # compute a conv w_rev
        w_rev = w[::-1]
        L = 1 << ((n + len(w_rev) - 1).bit_length())
        A = np.fft.rfft(a, n=L)
        W_rev = np.fft.rfft(w_rev, n=L)
        conv_aw = np.fft.irfft(A * W_rev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * conv_aw[len(w_rev)-1 - np.arange(n)]

        # exponentiated update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        sum_a = a.sum()
        if sum_a > 0:
            a = a / sum_a
        else:
            a = np.full_like(a, 1.0 / n)

        # update alpha schedule: after certain iterations increase alpha index
        if it % steps_per_alpha == 0 and alpha_index < total_alphas - 1:
            alpha_index += 1

    return best_a, best_score

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Utility building blocks**
   * Fast linear (non‑circular) self‑convolution with FFT.
   * Exact sub‑gradient of ``max(conv)`` (active‑set).
   * Gradient of a *smoothed* max (log‑sum‑exp) that is fully differentiable.
   * Exponentiated‑gradient (multiplicative) updates – intrinsically stay in the
     probability simplex.

2. **Multi‑stage optimisation**
   * Stage 1 – many short random restarts on a modest size (n≈800) using the
     smoothed‑max gradient and exponentiated updates.
   * Stage 2 – up‑sample the current best vector to n≈2400 and continue the
     same optimisation for a larger time slice.
   * Stage 3 – up‑sample to n≈4800 (the length used by the reference) and run
     the optimiser for the bulk of the remaining budget.
   * Stage 4 – a brief projected‑gradient descent with the exact sub‑gradient
     (to polish the solution) followed by a simple simulated‑annealing
     refinement.

3. **Time management**
   The total running time is bounded by the ``budget_s`` argument (≈1000 s).
   Each stage receives a fraction of the remaining time; the last stage uses
   whatever is left.

All numbers are kept non‑negative and normalised to sum = 1, so the
``evaluate_sequence`` call is always valid.  The routine prints progress
messages that appear in the harness output.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: nearest power of two (FFT length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using all indices where conv is
    within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]

    # weights w_k = exp(α·conv_k) / Σ exp(...)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10**7):
    """
    Multiplicative (exponentiated) updates with a smoothed max objective.
    The schedule ``alpha_schedule`` contains increasing smoothing parameters.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # FFT length sufficient for all convolutions at this stage
    fft_len = _next_pow2(3 * n - 2)       # length needed for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        # self‑convolution
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # gradient of the smoothed max
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # exponentiated (multiplicative) update
        eta = eta0 / math.sqrt(it)
        # protect against overflow in the exponent
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # advance smoothing parameter
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent using the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Classic projected gradient descent with momentum (as in the reference
    implementation).  Used for a short polishing phase.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_full(a, conv)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # simple Euclidean projection onto the simplex
        v = a - step * momentum
        # -------- simplex projection (O(n)) ----------
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)
        # -------------------------------------------

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index (chosen proportionally to its current weight) to a
    uniformly chosen destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) between stages
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
