"""
Propose a non‑negative step function (list of heights) that tries to
minimise ``evaluate_sequence``.

Algorithm
---------
* Warm‑start from ``height_sequence_1`` if it exists, otherwise random.
* Work on a series of lengths (800 → 1 600 → 3 200 → 6 400 → 9 600 → 12 800
  → 16 000).  Each length is obtained by linear up‑sampling the current
  best and re‑normalised to sum = 1.
* For every length run a *max‑gradient descent*:
  – the objective is 2·n·max(conv(a,a)).  We keep the sum fixed to 1 by
    projecting onto the probability simplex after each update.
  – the sub‑gradient of the ``max`` is computed exactly: for the
    (possibly several) indices where the convolution attains its maximum
    we sum the corresponding shifted reverse‑vectors.
  – a Nesterov‑style momentum term and a diminishing stepsize
    (step₀/√t) are used.
  – after each projection we enforce symmetry (a ← (a+a[::-1])/2) because
    the optimal step function is expected to be symmetric.
* A cheap simulated‑annealing sweep (mass transfer between two coordinates)
  follows the descent to escape shallow local minima.
* The best sequence seen at any length is kept.  When time permits we
  upscale further (up to ~20 000) and repeat the descent + annealing.
* Finally the best sequence is returned as a plain Python list.

The whole routine respects the 1000 s budget, prints progress and never
uses closures or file/network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (fast self‑convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear up‑sampling (keeps total mass = 1)
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _grad_max_convolution(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a * a)[k]   w.r.t. a.
    ``conv`` must be ``_conv_self(a)``.
    The gradient is 2 * sum_{k in Argmax} (reverse(a) shifted by k).
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices where the maximum is attained (numerical tolerance)
    idx = np.where(conv >= max_val - 1e-12)[0]

    # weight each active index equally so that the weights sum to 1
    w = np.zeros_like(conv)
    if idx.size == 0:
        return np.zeros_like(a)
    w[idx] = 1.0 / idx.size

    a_rev = a[::-1]
    # convolution of the weight vector with reversed a
    full = _conv_fft(w, a_rev)          # length = n + (2n‑1) – 1 = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]  # central part corresponds to gradient
    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) max‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_max_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 200_000,
                     step0: float = 0.6,
                     beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Minimise  max(conv(a,a))  via projected sub‑gradient descent.
    Returns the best sequence (still normalised to sum=1) and its
    2·n·max_b score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] max‑descent improve: n={n}, it={it}, score={cur_score:.6f}")

        # exact sub‑gradient of the max‑convolution
        grad = _grad_max_convolution(a, conv)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - step * momentum)

        # enforce symmetry (helps the optimisation)
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfer)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that moves a random ε‑fraction of mass from i to j.
    Returns possibly improved sequence and its 2·n·max_b score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2          # move up to 20 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score
