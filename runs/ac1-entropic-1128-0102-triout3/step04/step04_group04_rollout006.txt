"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
1. **Multiple initialisations** – try the existing construction (if any),
   a grid of power‑law shapes, exponential decays and a few random Dirichlet
   vectors for a few different lengths (2000‑8000).  Keep the best one.

2. **Two‑phase smooth optimisation**  
   • *Phase A – flatten* : minimise the ‑L² distance of the auto‑convolution
     to the constant average (the ideal flat spectrum).  The gradient of this
     objective is a simple cross‑correlation and can be obtained with one FFT.
   • *Phase B – reduce the peak* : approximate the max‑norm with a soft‑max
     (log‑sum‑exp) using a temperature β that is slowly increased.  Its
     gradient is also a cross‑correlation, now weighted by the soft‑max
     probabilities.

   The two gradients are linearly blended: early iterations use the L²
   gradient (γ≈0) and later iterations focus on the max‑gradient (γ→1).
   Updates are performed with Adam and the result is projected onto the
   probability simplex (non‑negative entries summing to 1).

3. **Cheap random hill‑climbing** – after the main loop a short
   random‑mass‑transfer phase tries to squeeze a few more improvements.

All operations are O(n log n) using FFTs, so the routine comfortably fits
the 1000 s wall‑clock budget even for n≈8000.  The best sequence found is
returned as a plain list of Python floats.

def _score_of_seq(a: np.ndarray) -> float:
    """
    Score for a normalised vector (sum(a)=1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution and its maximum (both returned)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cross_corr_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_j a_j w_{i+j} for i=0…n‑1.
    This is the sub‑gradient of the (soft‑max) max‑objective.
    """
    n = a.shape[0]
    # w length = 2n-1, reverse it for correlation via convolution
    w_rev = w[::-1]
    fft_len = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(A * W, n=fft_len)
    idx_off = 2 * n - 2                       # len(w_rev)-1
    return 2.0 * full[idx_off - np.arange(n)]


def _cross_corr_grad_l2(a: np.ndarray, r: np.ndarray) -> np.ndarray:
    """
    Gradient of the L2‑error  ∥conv(a,a) - avg∥₂² .
    It equals 4·∑_j a_j r_{i+j}.
    """
    n = a.shape[0]
    r_rev = r[::-1]
    fft_len = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    R = np.fft.rfft(r_rev, n=fft_len)
    full = np.fft.irfft(A * R, n=fft_len)
    idx_off = 2 * n - 2
    return 4.0 * full[idx_off - np.arange(n)]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget, prints progress
    and returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1.   Gather a few initial candidates (different lengths & shapes)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    lengths = [2500, 4000, 5000, 7000, 8000]          # a small set of lengths
    for n in lengths:
        # power‑law grid
        for p in np.linspace(0.15, 0.90, 13):
            if time.time() > deadline:
                break
            i = np.arange(1, n + 1, dtype=float)
            a = i ** (-p)
            a /= a.sum()
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Init power‑law n={n} p={p:.3f} → {sc:.6f}")

        # exponential decay
        for lam in np.linspace(0.0002, 0.0015, 5):
            if time.time() > deadline:
                break
            i = np.arange(n, dtype=float)
            a = np.exp(-lam * i)
            a /= a.sum()
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Init exp‑decay n={n} λ={lam:.5f} → {sc:.6f}")

        # few random Dirichlet draws (concentrated)
        for _ in range(3):
            if time.time() > deadline:
                break
            a = np.random.dirichlet(np.ones(n) * 0.5)   # sparsish
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Init random Dirichlet n={n} → {sc:.6f}")

    # If for some reason nothing was generated, fall back to a uniform vector
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 2.   If a previous construction exists, try it as well
    # ------------------------------------------------------------------
    try:
        prev = np.array(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev /= prev.sum()
            sc = _score_of_seq(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 → {sc:.6f}")
    except Exception:
        pass

    # Ensure we work with a normalised vector
    current = best_seq.copy()
    n = current.shape[0]

    # ------------------------------------------------------------------
    # 3.   Main optimisation loop (smooth‑max + L2 flattening)
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(current)
    v = np.zeros_like(current)

    # soft‑max temperature schedule
    beta_temp = 5.0
    beta_inc = 0.0005          # very gentle increase

    it = 0
    while time.time() < deadline:
        it += 1
        # -------- auto‑convolution & basic data ----------
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:5d} → new best {cur_score:.6f}")

        # -------- gradients ----------
        avg = 1.0 / (2 * n - 1)
        r = conv - avg                                 # residual for L2 term

        # L2‑gradient (flattening)
        g_l2 = _cross_corr_grad_l2(current, r)

        # soft‑max weighted gradient (max‑reduction)
        beta_temp = min(200.0, beta_temp + beta_inc)   # slowly raise temperature
        shifted = conv - max_val
        # stable soft‑max
        exp_arg = beta_temp * shifted
        exp_max = np.max(exp_arg)                     # =0 but keep numeric safety
        w_raw = np.exp(exp_arg - exp_max)
        w = w_raw / w_raw.sum()                       # normalised weights
        g_max = _cross_corr_grad(current, w)

        # blend the two gradients – start with L2, finish with max‑gradient
        blend = min(1.0, it / 25000.0)                 # γ grows to 1 after ~25k iters
        g = (1.0 - blend) * g_l2 + blend * g_max

        # -------- Adam update ----------
        eta = 0.2 / math.sqrt(it)                     # decaying step size
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        cand = current - step                         # gradient descent direction
        cand = _project_simplex(cand)

        # evaluate candidate
        _, cand_max = _conv_and_max(cand)
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score - 1e-12:
            current = cand
            continue        # successful descent, go to next iteration

        # -------- fallback greedy mass transfer ----------
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src != i_dst and current[i_src] > 0:
            delta = min(0.07 * current[i_src], current[i_src])
            tmp = current.copy()
            tmp[i_src] -= delta
            tmp[i_dst] += delta
            _, tmp_max = _conv_and_max(tmp)
            tmp_score = 2.0 * n * tmp_max
            if tmp_score < cur_score - 1e-12:
                current = tmp
                continue

        # -------- tiny random perturbation (escape) ----------
        i, j = np.random.randint(0, n, size=2)
        if i != j and current[i] > 0:
            eps_move = np.random.rand() * 0.03
            delta = eps_move * current[i]
            tmp = current.copy()
            tmp[i] -= delta
            tmp[j] += delta
            _, tmp_max = _conv_and_max(tmp)
            tmp_score = 2.0 * n * tmp_max
            if tmp_score < cur_score - 1e-12:
                current = tmp
                continue

        # if nothing helped, keep the current vector
        # (the loop will try a different random direction next time)

    # ------------------------------------------------------------------
    # 4.   Final cheap random hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 20000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or current[i] == 0:
                continue
            eps_move = np.random.rand() * 0.04
            delta = eps_move * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            _, cand_max = _conv_and_max(cand)
            cand_score = 2.0 * n * cand_max
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # 5.   Return best sequence as plain Python floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
