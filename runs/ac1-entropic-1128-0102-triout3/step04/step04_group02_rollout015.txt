    sums = set()

"""
A fast random‑search / hill‑climbing optimiser for the step‑function
evaluation problem.

Idea
----
* For a fixed length `n` pick a binary support `S ⊂ {0,…,n‑1}`.
  All selected positions receive the same height `1/|S|` so that the
  total mass is 1 (the evaluation function is scale‑invariant).
* If `S` is “well‑spread’’ the self‑convolution `a*a` distributes the
  mass over many lags and the maximum entry is close to the average
  `1/(2n)`.  In that case the objective
      2·n·max(a*a)   (which equals `evaluate_sequence` for sum=1)
  is close to 1, beating the uniform baseline (value 2) and the previous
  best (~1.51).
* The algorithm repeatedly samples random supports of different densities,
  keeps the best one, and finally improves it by a cheap hill‑climbing
  (swap a 1↔0) that accepts only improvements.
* Convolutions are performed with a real FFT (O(n log n)) to stay well
  inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based self‑convolution (linear, not circular)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT length."""
    n = a.shape[0]
    # length needed for a linear convolution of two length‑n arrays
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple hill‑climbing on a binary support
# ----------------------------------------------------------------------


def _hill_climb(mask: np.ndarray,
                n: int,
                current_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Try to improve ``mask`` (boolean array) by swapping a 1 ↔ 0.
    Only swaps that strictly lower the objective are kept.
    """
    w = mask.sum()
    # pre‑compute the float version that satisfies sum = 1
    a = mask.astype(float) / w
    best_mask = mask.copy()
    best_score = current_score

    while time.time() < deadline:
        # pick a random 1‑position and a random 0‑position
        ones = np.flatnonzero(mask)
        zeros = np.flatnonzero(~mask)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # perform the swap
        mask[i] = False
        mask[j] = True

        a_candidate = mask.astype(float) / w
        conv = _conv_self_fft(a_candidate)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_mask = mask.copy()
        else:
            # revert
            mask[i] = True
            mask[j] = False

    return best_mask, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method:
        1. Randomly sample binary supports of various densities.
        2. Keep the support with the smallest objective.
        3. Refine it with a short hill‑climbing phase.
        4. Return the corresponding height vector.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    # a few lengths to try – larger lengths give a finer granularity but
    # cost more time.  2000 is a good compromise for the 1000 s budget.
    lengths = [800, 1600, 3200, 6400]
    # densities (fraction of positions that are non‑zero)
    densities = [0.30, 0.40, 0.50, 0.60, 0.70]

    # ------------------------------------------------------------------
    # Random search
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        for dens in densities:
            if time.time() >= deadline:
                break

            w = max(1, int(round(n * dens)))          # size of the support
            # allocate a slice of time for this (n, dens) pair
            slice_deadline = time.time() + min(20.0,
                                               0.04 * (deadline - time.time()))
            # try many random supports until the slice expires
            while time.time() < slice_deadline:
                # generate a random binary mask with exactly w ones
                mask = np.zeros(n, dtype=bool)
                mask[np.random.choice(n, w, replace=False)] = True

                a = mask.astype(float) / w               # sum = 1
                conv = _conv_self_fft(a)
                max_b = conv.max()
                score = 2.0 * n * max_b

                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"new best (rand) n={n} dens={dens:.2f} "
                          f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climbing refinement on the best binary support found so far
    # ------------------------------------------------------------------
    if best_seq is not None:
        # recover the binary mask from the current best vector
        n = best_seq.size
        # because all non‑zero entries are equal, the mask is simply
        # (value > 0)
        mask = best_seq > 0
        # one short optimisation window
        hill_deadline = time.time() + min(30.0, deadline - time.time())
        mask_refined, score_refined = _hill_climb(mask, n, best_score,
                                                  hill_deadline)

        if score_refined < best_score:
            best_score = score_refined
            best_seq = (mask_refined.astype(float) /
                        mask_refined.sum())   # keep sum = 1
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: "
                  f"n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list – the outer harness will feed it to
    # ``evaluate_sequence``.
    return [float(x) for x in best_seq]
