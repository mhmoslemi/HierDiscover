def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s -2.0

    # set n (maybe 5000)
    n = 5000

    # init sequence: if height_sequence_1 exists, use it else random.
    # Ensure sum=1.

    a = init sequence normalized.

    # compute conv and max using FFT for initial
    conv = np.convolve(a,a)   # could use FFT
    current_max = conv.max()
    best_a = a.copy()
    best_max = current_max
    best_score = 2*n*best_max

    it = 0
    # parameters
    rel_eps = 0.002
    move_fraction = 0.05  # up to 5% mass per move (initially)
    frac_decay = 0.9999   # decays each iteration

    while time.time() < deadline:
        it += 1
        # compute active set
        max_val = current_max
        K = np.where(conv >= max_val * (1 - rel_eps))[0]
        # compute subgradient g for active set
        g = np.zeros(n)
        for k in K:
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            i_idx = np.arange(lo, hi+1)
            j_idx = k - i_idx
            g[i_idx] += 2*a[j_idx]
        # find source and dest candidates
        # exclude indices with a[i] very small
        # choose source among top 10% g values weighted by a
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src_g = g[src_candidates]
        src_idx = src_candidates[np.argmax(src_g)]
        # dest: min g among all indices
        dest_idx = np.argmin(g)
        # compute delta
        max_move = move_fraction * a[src_idx]
        # randomly sample delta between 0 and max_move
        delta = np.random.rand() * max_move

        if delta <= 0:
            continue

        # apply incremental conv update using old a
        conv_new = conv.copy()
        # update conv slices
        # conv_new[i:i+n] -= 2*delta*a
        # conv_new[j:j+n] += 2*delta*a
        # need to store a copy of a because a will change after update; use a_old = a.copy()
        a_old = a
        conv_new[src_idx:src_idx+n] -= 2*delta*a_old
        conv_new[dest_idx:dest_idx+n] += 2*delta*a_old
        conv_new[2*src_idx] += delta*delta
        conv_new[2*dest_idx] += delta*delta
        conv_new[src_idx+dest_idx] -= 2*delta*delta

        new_max = conv_new.max()
        accept = False
        # Accept if new_max lower, else maybe accept with SA prob
        if new_max < current_max:
            accept = True
        else:
            # SA temperature schedule based on iteration count
            T = max(1e-7, (best_max - current_max) * 0.5)  # not too small
            # Actually adapt T as decreasing function
            # Let's use T = 1e-4 * (1 - it/ max_iters)
            # But we don't have max_iters; use time left ratio
            time_left = deadline - time.time()
            total_time = budget_s - 2.0
            T = 0.0001 * (time_left / total_time)
            # Accept with prob exp(-(new_max - current_max)/T)
            if T > 0 and np.random.rand() < np.exp(-(new_max - current_max) / T):
                accept = True

        if accept:
            # update a
            a = a.copy()
            a[src_idx] -= delta
            a[dest_idx] += delta
            # clip very small negatives if any
            if a[src_idx] < 0:
                a[src_idx] = 0.0
            # renormalize only if rounding errors? sum should stay 1.
            # but due to floating error may deviate; renormalize occasionally
            # e.g., every 500 iterations.
            if it % 500 == 0:
                a = np.maximum(a, 0.0)
                total = a.sum()
                if total == 0:
                    a = np.full(n, 1.0/n)
                else:
                    a /= total
                # recompute conv from scratch after renormalization:
                conv = np.convolve(a, a)
                current_max = conv.max()
            else:
                conv = conv_new
                current_max = new_max

            # update best if better
            if current_max < best_max:
                best_max = current_max
                best_a = a.copy()
                best_score = 2.0*n*best_max
                print(f"[{time.time() - start_time:6.2f}s] iter={it} new best score={best_score:.6f}")

        # adapt move_fraction gradually downwards
        move_fraction *= frac_decay
        if move_fraction < 1e-5:
            move_fraction = 1e-5

        # maybe do occasional pure random perturbations (mass transfer between random indices)
        if it % 2000 == 0:
            # large random mass redistribution
            # pick random indices to zero out and reassign mass
            idxs = np.random.choice(n, size=10, replace=False)
            total_mass = a[idxs].sum()
            a[idxs] = 0
            # reallocate to random other indices uniformly
            rand_idx = np.random.randint(0, n)
            a[rand_idx] += total_mass
            # recompute conv from scratch
            conv = np.convolve(a, a)
            current_max = conv.max()
            # ensure normalization
            a = np.maximum(a, 0)
            a /= a.sum()
            # update best if improved
            if current_max < best_max:
                best_max = current_max
                best_a = a.copy()
                best_score = 2.0*n*best_max
                print(f"[{time.time() - start_time:6.2f}s] iter={it} random perturb best score={best_score:.6f}")

    # after loop, maybe do final refinement using exponentiated gradient from best_a for a short time
    # we can reuse earlier code for a few hundred iterations
    # for simplicity we stop here.

    final_score = 2.0*n*best_max
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={final_score:.6f}")
    return [float(x) for x in best_a]

"""
propose_candidate
=================

A deterministic local‑search / simulated‑annealing hybrid for the
``evaluate_sequence`` objective.

The algorithm works on a probability vector `a` (non‑negative, sum = 1)
and keeps the full auto‑convolution `conv = a * a`.  In each iteration
the *active* indices (those whose convolution value is within a tiny
relative band of the current maximum) are used to compute a sub‑gradient
of the maximum.  Mass is transferred from a high‑gradient index to a
low‑gradient index; the convolution is updated *incrementally* in O(n)
time.  Improvements are always taken, and occasional worsening moves are
accepted according to a simple temperature schedule (simulated annealing)
to escape shallow plateaus.  Periodic random perturbations and full
re‑computations prevent drift of floating‑point errors.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence (as a plain list of floats) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full auto‑convolution of ``a`` using an FFT of length
    the next power of two.  Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(Af * Af, n=m)[: 2 * n - 1]
    return conv


def _active_set(conv: np.ndarray, rel_eps: float = 0.001) -> np.ndarray:
    """
    Return the indices of ``conv`` whose value is within ``rel_eps``
    relative distance of the global maximum.
    """
    max_val = conv.max()
    eps = max_val * rel_eps
    return np.where(conv >= max_val - eps)[0]


def _subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the functional ``max(conv(a, a))`` restricted to the
    active indices.  For each active index k we add 2·a_{k‑i} to component i.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _incremental_conv_update(conv: np.ndarray, a: np.ndarray,
                             src: int, dst: int, delta: float) -> np.ndarray:
    """
    Return a *new* convolution array resulting from moving ``delta`` mass
    from index ``src`` to index ``dst`` (both 0‑based).  ``a`` is the
    *current* probability vector (i.e. before the move).  The update is
    performed in O(n) time using the identity

        (a+Δa)*(a+Δa) = a*a + 2*a*Δa + Δa*Δa

    where Δa has only the two non‑zero components ``-δ`` at ``src`` and
    ``+δ`` at ``dst``.
    """
    n = a.shape[0]
    conv_new = conv.copy()

    # linear part 2·a*Δa  (Δa = -δ e_src + δ e_dst)
    # contribution -2δ·a shifted by src
    conv_new[src:src + n] -= 2.0 * delta * a
    # contribution +2δ·a shifted by dst
    conv_new[dst:dst + n] += 2.0 * delta * a

    # quadratic part Δa*Δa
    conv_new[2 * src] += delta * delta          # (‑δ)² at index 2·src
    conv_new[2 * dst] += delta * delta          # (+δ)² at index 2·dst
    conv_new[src + dst] -= 2.0 * delta * delta  # cross terms at src+dst

    return conv_new

            time_left = deadline - time.time()
            total_time = budget_s - 2.0
            T = max(1e-9, 1e-4 * (time_left / total_time))
            if np.random.rand() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            # apply the move permanently
            a = a.copy()
            a[src] -= delta
            a[dst] += delta
            # numerical clean‑up
            a[a < 0] = 0.0
            conv = conv_new
            cur_max = new_max

            # keep the best ever seen
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                print(f"[{time.time() - start_time:6.2f}s] iter={it:6d} "
                      f"new best score={best_score:.6f}")

        # ----------------------------------------------------------
        # f) periodic housekeeping
        # ----------------------------------------------------------
        # decay the step‑size slowly
        delta_frac = max(delta_frac * delta_decay, min_delta_frac)

        # occasional full recalculation to eliminate accumulated round‑off
        if it - last_full_recalc >= full_recalc_interval:
            conv = _full_convolution(a)
            cur_max = conv.max()
            last_full_recalc = it

        # big random perturbation every few thousand iterations to jump out
        if it % 5000 == 0:
            # pick a small set of indices, zero them and dump their mass
            # into a random location – cheap way to explore new configurations
            mask = np.random.rand(n) < 0.02               # ~2 % of entries
            if mask.any():
                mass = a[mask].sum()
                a[mask] = 0.0
                target = np.random.randint(0, n)
                a[target] += mass
                # re‑normalise (tiny floating‑point drift may appear)
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a /= a_sum
                else:
                    a = np.full(n, 1.0 / n)
                conv = _full_convolution(a)
                cur_max = conv.max()

    # --------------------------------------------------------------
    # 3. final reporting --------------------------------------------
    # --------------------------------------------------------------
    final_score = 2.0 * n * best_max
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={final_score:.6f}")
    # Return as a plain Python list of floats (required by the grader)
    return [float(x) for x in best_a]
