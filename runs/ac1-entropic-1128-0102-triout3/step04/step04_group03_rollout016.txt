def _adam_optimizer(a0, n, deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, alpha=20000., max_iter=10**7):
    a = a0.copy()
    # ensure sum=1
    a = a / a.sum()
    # logits
    z = np.log(a + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    Nfft = 1 << ((3*n - 2).bit_length())
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute A once
        A = np.fft.rfft(a, Nfft)
        # conv
        conv = np.fft.irfft(A * A, n=Nfft)[:2*n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it}, score={cur_score:.6f}")
        # compute weights w via softmax with alpha
        # ensure stable:
        # w = exp(alpha * (conv - max_conv))
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a:
        # compute correlation via FFT
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, Nfft)
        c_rev = np.fft.irfft(A * W_rev, n=Nfft)[:len(a) + len(w) - 1]
        start = len(w) - 1
        grad_a = 2.0 * c_rev[start - np.arange(len(a))]
        # gradient w.r.t logits:
        g_bar = np.dot(grad_a, a)
        grad_z = a * (grad_a - g_bar)
        # Adam update:
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        # update a from logits
        # stable softmax:
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()
    return best_a, best_score

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1.  Work on the probability simplex (sum a_i = 1) using a softmax
    parametrisation a_i = exp(z_i)/∑exp(z).
2.  Optimise the smooth approximation of the objective
        f(a) = (1/α)·log∑_k exp(α·(a*a)[k]) .
    The gradient of f w.r.t. a is computed by a cross‑correlation of a
    with the softmax weights w_k (the log‑sum‑exp weights).  All convolutions
    are performed with FFTs for speed.
3.  Adam (first‑order adaptive optimiser) is applied to the logits z.
    After each step a is recomputed by a numerically stable softmax and
    the true evaluation score (2·n·max(a*a)) is recorded.
4.  The optimisation proceeds in several size stages
    (800 → 2400 → 4800 → 9600).  After every stage the best sequence is
    up‑sampled linearly and used as the start for the next, larger, stage.
5.  A final stochastic refinement (mass‑transfer simulated annealing)
    cleans up any remaining peaks.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# Basic helpers

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Length of the FFT that can hold linear convolution of length 3·n‑2."""
    return 1 << ((3 * n - 2).bit_length())


def _conv_self(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution a * a (length 2·n‑1) using FFT of size Nfft."""
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return conv


def _grad_softmax_weighted(a: np.ndarray,
                           conv: np.ndarray,
                           alpha: float,
                           Nfft: int) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) objective w.r.t. a.
    Uses w_k = softmax(alpha·conv_k) and computes
        grad_i = 2·∑_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    # softmax weights (stable)
    w = np.exp(alpha * (conv - conv.max()))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation via FFT (a with reversed w)
    w_rev = w[::-1]
    A = np.fft.rfft(a, Nfft)
    W_rev = np.fft.rfft(w_rev, Nfft)
    c_rev = np.fft.irfft(A * W_rev, n=Nfft)[: n + len(w) - 1]
    start = len(w) - 1  # index of lag 0 in the reversed convolution
    grad = 2.0 * c_rev[start - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (logits parametrisation)
# ----------------------------------------------------------------------
def _run_adam_optimizer(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        lr: float = 0.05,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8,
                        alpha0: float = 5000.0,
                        alpha_ramp: float = 200000.0,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smoothed max‑convolution objective."""
    a = a0.copy()
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)

    # logits that automatically enforce positivity and unit sum
    z = np.log(a + 1e-12)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    Nfft = _fft_len_for_n(n)

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- convolution and current score --------------------------------
        A = np.fft.rfft(a, Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                f"iter={it}, score={cur_score:.6f}"
            )

        # ----- softmax‑weighted gradient ------------------------------------
        alpha = alpha0 * (1.0 + it / max(alpha_ramp, 1.0))
        grad_a = _grad_softmax_weighted(a, conv, alpha, Nfft)

        # ----- gradient w.r.t. logits ---------------------------------------
        g_bar = np.dot(grad_a, a)
        grad_z = a * (grad_a - g_bar)

        # ----- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- back‑to simplex -----------------------------------------------
        a = _softmax(z)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-5,
                   steps: int = 600_000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing on the simplex."""
    a = a0.copy()
    Nfft = _fft_len_for_n(n)
    max_conv = _conv_self(a, Nfft).max()
    score = 2.0 * n * max_conv

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2      # shift up to 20 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        max_conv_prop = _conv_self(a_prop, Nfft).max()
        new_score = 2.0 * n * max_conv_prop
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                f"iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the time budget, progressively refines the sequence,
    and finally returns the best list of heights found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0               # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Staging (coarse → fine)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # increasingly finer grids
    total_time = deadline - start_time
    anneal_budget = max(30.0, 0.15 * total_time)  # ≈15 % for the final walk
    grad_budget = total_time - anneal_budget
    total_weight = sum(stage_lengths)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time proportionally to the size of this stage
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this size
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run Adam optimiser on the simplex
        a_opt, score_opt = _run_adam_optimizer(
            a0, n, stage_deadline, start_time,
            lr=0.05,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            alpha0=5000.0,
            alpha_ramp=200_000,
            max_iter=10 ** 7,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                f"score={score_opt:.6f}"
            )

        # re‑compute remaining gradient budget after this stage
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing refinement (uses the length of the best stage)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        best_seq, best_score = _run_annealing(
            best_seq,
            n_final,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=0.2,
            end_temp=1e-5,
            steps=800_000,
        )
        print(
            f"[{time.time() - start_time:6.2f}s] final anneal best score: "
            f"{best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    # final normalisation (tiny drift may happen because of floating‑point noise)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to a plain Python list for the harness
    return [float(x) for x in best_seq]
