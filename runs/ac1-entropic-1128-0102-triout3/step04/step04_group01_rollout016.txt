"""
Propose a low‑maximum self‑convolution step‑function.

The search is a three‑stage pipeline:

1.  Small‑scale refinement (n ≤ 2048) using a *linear program*.
    For a fixed distribution f we maximise Σg under the linear
    constraints (f * g)[k] ≤ M (where M = max(f*f)).  The LP gives an
    extreme point g that locally improves the ℓ∞‑norm of the
    self‑convolution.  After each LP we mix g back into f and repeat
    a few times.

2.  Multi‑scale up‑sampling: the best distribution from the previous
    stage is linearly interpolated to larger lengths (4096, 8192,
    16384).  On each scale a smoothed log‑sum‑exp gradient descent
    (projected onto the probability simplex) is run for the remaining
    time slice.

3.  Final polishing with a cheap mass‑transfer simulated annealing.

All helpers are top‑level (no closures) and the routine respects the
1000 s budget.  The returned list is a non‑negative vector; the
evaluator normalises it internally, so we keep the sum = 1 throughout.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal – return the uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolutions (FFT based, fast for all n)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (full length) via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of log‑sum‑exp surrogate for the ℓ∞ norm
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the L_λ(a) = λ·log Σ exp(conv/λ) surrogate.
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()                 # soft‑max weights
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)        # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]      # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Log‑sum‑exp gradient descent with momentum and a decreasing λ.
    Returns the best vector seen (sum = 1) and its exact objective
    2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8

    # λ‑schedule parameters
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b               # exact objective (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} "
                  f"it={it} score={cur_score:.6f}")

        # λ update (never zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing that moves a small ε from one index to another.
    Returns an improved sequence (sum = 1) and its exact objective.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} "
                  f"it={it} score={score:.6f}")

    return best_seq, best_score

# Linear‑programming based improvement (small n only)
# ----------------------------------------------------------------------
# Try to import a fast LP solver from SciPy; fall back to “none”.

    _SCIPY_LP = True
except Exception:                # pragma: no cover
    _SCIPY_LP = False


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A of shape (2n‑1, n) such that
    (A @ g)[k] = (f * g)[k].
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Row k receives f[i] for i where 0 ≤ k‑i < n   ⇔   i ∈ [max(0,k‑n+1), min(n‑1,k)]
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        A[k, i_start:i_end + 1] = f[i_start:i_end + 1]
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i   s.t.   f * g ≤ M,   g ≥ 0.
    Returns a normalised probability vector (sum = 1) or None on failure.
    """
    if not _SCIPY_LP:                     # pragma: no cover
        return None

    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise Σg  ⇔ minimise –Σg
    bounds = [(0, None)] * n

    # ‘highs’ is fast and robust for moderate sizes (n ≲ 2000)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s > 0.0:
        g /= s
    else:
        g = np.full_like(g, 1.0 / n)
    return g


def _lp_iterate(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps starting from a0.
    Returns the best vector seen (sum = 1) and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        g = _lp_max_sum(a, M)
        if g is None:
            # LP failed – abort the LP stage
            break

        # Mix the LP solution back, decreasing the step size gradually
        t = 0.5 ** (it + 1)          # 0.5, 0.25, 0.125, …
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} "
                  f"it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear interpolation and up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, keeping sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
