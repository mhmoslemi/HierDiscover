def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0
    # load starter
    ...
    best_seq = None
    best_score = float('inf')
    # stage list
    stage_lengths = [800, 2400, 4800]
    # allocate time
    total_weight = sum(stage_lengths)
    remaining = deadline - start_time
    anneal_time = max(30.0, 0.15 * remaining)
    grad_time = remaining - anneal_time
    # allocate each stage time proportionally
    for n in stage_lengths:
        if time.time() >= deadline:
            break
        # compute allocated time
        t_alloc = max(5.0, grad_time * n / total_weight)
        stage_deadline = time.time() + t_alloc
        # init a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            # try deterministic shape
            a0 = np.random.rand(n)
            a0 /= a0.sum()

            else:
                a0_trial = a0.copy()
            a_opt, score_opt = _run_adam(a0_trial, n, stage_deadline, start_time)

        # update remaining grad_time
        elapsed = time.time() - start_time

    # final annealing
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three phases:

1. **Coarse optimisation** – start with a modest length (800) and apply
   a projected Adam optimiser on the probability simplex.  The Adam
   update uses the sub‑gradient of the *single* most violating convolution
   entry (the shift where the self‑convolution attains its maximum).

2. **Progressive up‑sampling** – repeatedly linearly up‑sample the current
   best vector to a larger size (2400 → 4800).  At each size we run the same
   Adam optimiser for the allocated portion of the time budget; several
   short random restarts are attempted to escape shallow minima.

3. **Local stochastic refinement** – a cheap simulated‑annealing walk
   (move a small fraction of mass between two positions) finishes the
   search within the remaining time.

All heavy work respects the given time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x_i≥0, Σx_i=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex in O(n) time."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal (numerical safety)
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution of a (length 2·n‑1)."""
    n = a.shape[0]
    # next power of two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a,a)) and the sub‑gradient of that maximum.
    The gradient follows from d/da_i sum_{i+j=k} a_i a_j = 2·a_{k-i}
    for the active shift k = argmax(conv).
    """
    conv = _conv_self(a)
    k = int(conv.argmax())
    max_val = float(conv[k])
    n = a.shape[0]
    g = np.zeros_like(a)

    if k < n:
        # indices 0 … k contribute
        # g[0:k+1] = 2 * a[k::-1]
        g[: k + 1] = 2.0 * a[k::-1]
    else:
        # indices k-(n-1) … n-1 contribute
        start = k - (n - 1)
        cnt = n - start
        # g[start:] = 2 * a[:cnt][::-1]
        g[start:] = 2.0 * a[:cnt][::-1]

    return max_val, g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected Adam)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the probability simplex.
    Returns the best point encountered together with its *scaled* score
    `2·n·max(conv)`.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = float("inf")

    while time.time() < deadline:
        t += 1
        max_val, grad = _max_conv_and_grad(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={t}, score={cur_score:.6f}")

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # diminishing step size (sqrt schedule)
        step = lr0 / math.sqrt(t)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
