def simplex_projection(v):
    # v is numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v):
    # Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function using projected Adam on the simplex.
The algorithm repeatedly starts from a random (or known) simplex point,
computes a sub‑gradient of max(conv(a,a)) using a small active set,
applies Adam‑style updates, projects back onto the simplex, injects
occasional random mass‑transfer perturbations and finishes with a fast
stochastic hill‑climbing phase.  Multiple restarts are performed until the
time budget expires.  The best sequence found is returned.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of ``a`` (length n) and its maximum entry."""
    n = a.shape[0]
    # next power‑of‑two for a*a (length 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 20):
    """
    Return a set of indices that are “active’’ for the sub‑gradient.
    All indices within ``rel_eps`` relative distance of ``max_val`` are kept.
    If fewer than ``max_add`` such indices exist, the globally largest ones
    are added so that the set contains at most ``max_add`` entries.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Compute a sub‑gradient of ``max(conv(a,a))`` using the active set ``K``.
    For each active convolution index k the contribution to component i is
    ``2·a_{k‑i}`` (if the index is in range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _simplex_projection(v: np.ndarray):
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v == 0):
        # avoid division by zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a step‑function) that aims to
    minimise the ``evaluate_sequence`` score.  The routine runs for at most
    ``budget_s`` seconds, performs several Adam‑based optimisation runs,
    and finishes with a stochastic hill‑climbing local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # A small safety margin.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000                                   # length of the step function
    L_fft = 1 << ((3 * n - 2).bit_length())   # length for correlation (optional)

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    max_restart_iters = 8000                  # Adam iterations per restart
    max_hc_moves = 30000                      # hill‑climbing moves per restart
    lr = 0.08                                 # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999                 # Adam moments
    eps = 1e-8
    rel_eps = 0.001
    max_add = 20
    perturb_every = 4000                      # iterations between random perturbations
    perturb_frac = 0.03                       # up‑to 3 % of source mass

    best_seq = None
    best_score = np.inf

    restart_id = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  Initial point (random or from a previously known solution)
        # --------------------------------------------------------------
        if restart_id == 0 and 'height_sequence_1' in globals():
            a0 = np.array(globals()['height_sequence_1'], dtype=float)
            a0 = np.clip(a0, 0.0, None)
            if a0.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, a0.size)
                x_new = np.linspace(0.0, 1.0, n)
                a0 = np.interp(x_new, x_old, a0)
        else:
            a0 = np.random.rand(n)

        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a0 / s0

        # small random jitter to avoid perfect symmetry
        jitter = (np.random.rand(n) - 0.5) * 1e-3
        a = np.clip(a + jitter, 0.0, None)
        a = a / a.sum()

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

        # --------------------------------------------------------------
        # 2.  Adam optimisation on the simplex
        # --------------------------------------------------------------
        for it in range(1, max_restart_iters + 1):
            if time.time() >= deadline:
                break

            conv, max_val = _conv_and_max(a)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart={restart_id} iter={it:5d} "
                      f"new best score={cur_score:.6f}")

            # Active set for the sub‑gradient
            K = _active_indices(conv, max_val, rel_eps=rel_eps, max_add=max_add)
            g = _gradient_active(a, K)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step = lr / (np.sqrt(v_hat) + eps)
            a_new = a - step * m_hat

            # Project back onto the simplex
            a = _simplex_projection(a_new)

            # Occasionally inject a tiny random mass transfer to escape plateaus
            if it % perturb_every == 0:
                i, j = np.random.randint(0, n, size=2)
                eps_mass = np.random.rand() * perturb_frac * a[i]
                a[i] -= eps_mass
                a[j] += eps_mass
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # --------------------------------------------------------------
        # 3.  Local stochastic hill‑climbing (fast incremental updates)
        # --------------------------------------------------------------
        # start from the best sequence of this restart
        a_hc = best_seq.copy() if best_seq is not None else a.copy()
        conv_hc, max_val_hc = _conv_and_max(a_hc)
        cur_score = 2.0 * n * max_val_hc

        moves = 0
        while time.time() < deadline and moves < max_hc_moves:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps_move = np.random.rand() * 0.04          # up to 4 % of source mass
            delta = eps_move * a_hc[i]
            if delta <= 0:
                continue
            a_new = a_hc.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # maintain feasibility
            if a_new[i] < 0.0:
                continue
            conv_new, max_val_new = _conv_and_max(a_new)
            new_score = 2.0 * n * max_val_new
            # accept if improves (simple greedy hill‑climb)
            if new_score < cur_score:
                a_hc = a_new
                conv_hc = conv_new
                max_val_hc = max_val_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve score={new_score:.6f}")
            moves += 1

        restart_id += 1

    # ------------------------------------------------------------------
    # 4.  Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    # ensure the final list obeys the checks of evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, None)
    return [float(x) for x in best_seq]
