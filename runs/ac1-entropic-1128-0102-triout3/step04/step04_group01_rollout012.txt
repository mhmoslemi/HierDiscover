"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑a_i = 1) because the
evaluation function is scale‑invariant.  It combines three complementary
ideas that were not used together before:

1. **Parametric warm‑starts** – a short stochastic search over a few
   simple families (power‑product, truncated Gaussian, beta‑shaped)
   quickly finds good “shapes” for a given length.
2. **Projected sub‑gradient descent** – the exact ∞‑norm of the
   self‑convolution is used (no log‑sum‑exp surrogate).  The sub‑gradient
   of the active convolution entry is computed analytically, a diminishing
   step is taken and the iterate is projected back onto the simplex.
3. **Cheap mass‑transfer annealing** – after the gradient phase a few
   hundred thousand cheap moves (transfer a small ε from one index to
   another) are performed with a geometric temperature schedule to
   polish the current best vector.

The schedule starts with modest lengths and progressively upsamples the
best vector for larger lengths, always respecting the global 1000 s time
budget.  The routine prints occasional progress messages and finally
returns a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a*a) for a normalized sequence."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick random warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+0.5)*(n-i-0.5) raised to power p.
    p>0 => centre‑heavy, p<0 => edge‑heavy.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex (exact ∞‑norm)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method where the sub‑gradient is taken from the
    active self‑convolution entry.  Returns the best vector seen and its
    exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8            # base step (will be divided by sqrt(iter))
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # ----- sub‑gradient of the active entry -----
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()                         # a_i term
        grad[mask] += a[j[mask]]                # a_{idx-i} term
        grad *= (2.0 * n)                       # d(2·n·conv_max)/da_i

        # Normalise direction to make the step size schedule independent of n
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # Diminishing step size
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        # Tiny isotropic noise every few hundred steps to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    """Move a small ε from a source index to a destination index."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------

    best_score = float("inf")

    # ------------------------------------------------------------------

    return [float(x) for x in best_seq]
