def proj_simplex(v):
    # from https://github.com/indujihah/nn-function/blob/master/optimize.py
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function using a smooth‐max (log‑sum‑exp) surrogate.
The algorithm keeps the coefficient vector on the simplex and performs a
multiplicative (exponentiated‑gradient) mirror descent driven by the full
gradient of the smoothed objective.
The smoothing parameter α is increased gradually, turning the surrogate
into the true max.  After the main descent phase a cheap stochastic
hill‑climbing refines the best solution found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: auto‑convolution of a real vector (full length 2·n‑1)
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full autocorrelation a*a (length 2·n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# Helper: gradient of Σ_k w_k·(a*a)_k w.r.t. a (a is on the simplex)
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2·∑_k w_k·a_{k‑i}   (indices outside [0,n‑1] are ignored).
    w must have length 2·n‑1 and sum to 1.
    """
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    # reverse w and convolve with a (linear convolution)
    w_rev = w[::-1]
    fft_len = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    Fwr = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(Fa * Fwr, n=fft_len)[: n + m - 1]
    mid = m - 1
    idx = mid - np.arange(n)       # indices that give Σ_j a_j·w[i+j]
    g = 2.0 * conv[idx]
    return g

    ``budget_s`` seconds, prints intermediate scores and returns the
    best vector found.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin so the runner does not get killed
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Initialisation
    # --------------------------------------------------------------
    n = 5000                               # length of the sequence
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.size != n:                 # resample to the required length
            x_old = np.linspace(0.0, 1.0, init.size)
            x_new = np.linspace(0.0, 1.0, n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        init = np.random.rand(n)

    a = np.maximum(init, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    best_seq = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    print(f"[{time.time() - start:6.2f}s] start – score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Smooth‑max mirror‑descent
    # --------------------------------------------------------------
    # hyper‑parameters
    eta0 = 0.9            # initial step‑size for the exponentiated update
    alpha = 5.0           # start smoothing parameter
    alpha_max = 200.0
    alpha_mult = 1.30     # increase factor (applied every 5 000 iters)
    eps = 1e-15           # floor to stay away from log(0)

    max_iters = 300_000

        it += 1

        max_val = conv.max()
        cur_score = 2.0 * n * max_val

            best_score = cur_score
            best_seq = a.copy()

        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        g = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(it)

        if it % 10000 == 0:
            i, j = np.random.randint(0, n, size=2)
            eps_mass = np.random.rand() * 0.04        # up to 4 % of source mass
            delta = eps_mass * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # --------------------------------------------------------------
    # 3. Final stochastic hill‑climbing (uses remaining time)
    # --------------------------------------------------------------
    a = best_seq.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps_move = np.random.rand() * 0.04          # up to 4 % of source mass
        delta = eps_move * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv_new = _conv_self(a_new)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:6.2f}s] local improve score={new_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    print(f"[{time.time() - start:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
