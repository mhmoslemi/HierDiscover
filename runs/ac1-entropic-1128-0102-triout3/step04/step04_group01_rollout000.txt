def _run_adam_descent(a0, n, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.1  # maybe.
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    lam_factor = 0.3
    lam_decay = 0.8
    lam_decay_iter = 2000
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # update lambda
        if it % lam_decay_iter == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)
        # Adam
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / np.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)
        # update
        a_new = a - step
        # project
        a = _project_simplex(a_new)
    return best_a, best_score

"""
Propose a low‑maximum self‑convolution step‑function.

The algorithm works on the probability simplex (sum = 1) because the
evaluation is scale‑invariant.  It combines three ideas that differ from
the previous code:

1. **Adam‑style projected gradient descent** – we use an adaptive
   learning‑rate (Adam) on the Euclidean projection onto the simplex.
   The gradient is the log‑sum‑exp surrogate of the ℓ∞‑norm of the
   self‑convolution, with a slowly decreasing smoothing parameter λ.
2. **Multi‑scale warm‑start** – we start with a modest length (≈1 k)
   and repeatedly up‑sample the current best distribution to larger
   lengths (up to 32768).  The up‑sampled vector is re‑normalised to the
   simplex and serves as the initial point for the next scale.
3. **Heavy simulated‑annealing polishing** – after each Adam phase we
   run a longer Metropolis‑Hastings mass‑transfer annealing that moves
   a fraction of the mass from a source index to a destination index.
   The temperature decays aggressively; occasional swap moves add
   additional exploration.

The search stops after the allotted wall‑clock time (default 1000 s) and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerically stable soft‑max
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


def _gradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the exact max‑norm (no smoothing).
    """
    n = a.shape[0]
    max_b = conv.max()
    mask = (conv == max_b).astype(float)
    if mask.sum() == 0:
        # fallback to uniform weights (should not happen)
        mask = np.ones_like(conv) / conv.size
    else:
        mask /= mask.sum()
    a_rev = a[::-1]
    full = _conv_fft(mask, a_rev)
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      slice_deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the probability simplex using the smoothed
    log‑sum‑exp gradient.  Returns the best vector seen and its exact
    objective (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15                    # base learning rate
    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # λ‑schedule parameters
    lam_factor = 0.6
    lam_decay = 0.85
    lam_decay_interval = 4000

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ (smoothness)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient (smooth surrogate)
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decay with sqrt(it) for stability
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # occasional isotropic noise to escape plateaus
        if it % 600 == 0:
            noise = 0.015 / math.sqrt(it) * np.random.randn(n)
            step += noise

        # project back onto simplex
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Heavy simulated‑annealing polishing (mass transfer)
# ----------------------------------------------------------------------


def _run_annealing_improved(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 2e-2,
                            end_temp: float = 5e-6,
                            max_steps: int = 60000) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis mass‑transfer annealing with a longer schedule.
    Returns (possibly) improved sequence and its exact objective.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # propose a move: transfer a fraction of mass from i to j
        i = np.random.randint(0, n)
        if a[i] <= 1e-15:
            continue
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        # up to 40 % of the source mass
        eps = a[i] * np.random.rand() * 0.4
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

        # occasional swap move (pure permutation, no mass change)
        if it % 3000 == 0:
            i2, j2 = np.random.randint(0, n, size=2)
            if i2 != j2:
                a_swap = a.copy()
                a_swap[i2], a_swap[j2] = a_swap[j2], a_swap[i2]
                new_score2 = 2.0 * n * _conv_self(a_swap).max()
                delta2 = new_score2 - score
                if delta2 < 0.0 or math.exp(-delta2 / max(T, 1e-12)) > np.random.rand():
                    a = a_swap
                    score = new_score2
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] anneal swap improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
