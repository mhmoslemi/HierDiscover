def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 6000

    # init: try to use height_sequence_1 if present else random Dirichlet
    try:
        init = np.array(height_sequence_1, dtype=float)
        # resample to n by interpolation if needed
    except NameError:
        init = None

    if init is not None and init.size>0:
        if init.size != n:
            # linear interpolate
            x_old = np.linspace(0,1, init.shape[0])
            x_new = np.linspace(0,1, n)
            init = np.interp(x_new, x_old, init)
    else:
        init = np.random.rand(n)
    # ensure nonnegative
    init = np.maximum(init, 0.0)
    if init.sum() == 0:
        init = np.full(n,1/n)
    a = init / init.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    # Helper to compute conv and max
    def conv_max(a):
        n = a.shape[0]
        fft_len = 1 << ((2*n - 1).bit_length())
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
        return conv, conv.max()

    # Phase 1: Smooth mirror descent
    max_iters = 150000
    eta0 = 1.0
    tau = 0.001 # smoothing temperature
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv and max for diagnostics
        conv, conv_max_val = conv_max(a)
        cur_score = 2.0 * n * conv_max_val / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] mirror iter {it} new best {best_score:.6f}")

        # compute softmax weights over conv entries
        # weight w_k proportional to exp(conv_k / tau)
        # compute gradient g_i = 2 * sum_{k} w_k * a_{k-i} (with range)
        # To compute efficiently, we can compute convolution of a with w (weights) but w defined over conv length.
        # But n relatively small; we can compute using loops for active indices only.
        # For simplicity, use active set approach as before, weighted.
        max_val = conv_max_val
        eps = max_val * 0.001
        # indices close to maximum
        active = np.where(conv >= max_val - eps)[0]
        if active.size < 12:
            # fallback top indices
            top = min(12, conv.size)
            active = np.argpartition(conv, -top)[-top:]
        # compute softmax for these active indices only to approximate gradient
        # compute weights for active entries only
        # derived from conv values
        act_vals = conv[active]
        w = np.exp(act_vals / tau)
        w = w / w.sum()
        # gradient
        g = np.zeros_like(a)
        for idx, weight in zip(active, w):
            lo = max(0, idx - (n-1))
            hi = min(n-1, idx)
            i_idx = np.arange(lo, hi+1, dtype=int)
            j_idx = idx - i_idx
            g[i_idx] += 2.0 * weight * a[j_idx]

        # exponentiated gradient
        eta = eta0 / math.sqrt(it)
        # avoid zeros
        a = np.maximum(a, 1e-12)
        log_a = np.log(a) - eta * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()

        # occasional random jitter
        if it % 3000 == 0:
            i, j = np.random.randint(0, n, size=2)
            eps_move = np.random.rand() * 0.02
            delta = eps_move * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0)
            a = a / a.sum()

    # Phase 2: Simulated annealing hill climbing
    # Use best_a as starting point
    a = best_a.copy()
    cur_score = best_score
    # schedule: start temperature = 1e-3 maybe relative to score diff
    # We'll adapt temperature = init_temp * (0.5**(elapsed/total))
    total_sa_time = max(0.0, deadline - time.time())
    sa_start = time.time()
    init_temp = 0.001 # small; we need acceptance of small worse moves.
    while time.time() < deadline:
        elapsed = time.time() - sa_start
        frac = elapsed / total_sa_time if total_sa_time>0 else 1.0
        # exponential cooling
        temp = init_temp * (0.01 ** frac)  # final temp 0.01*init_temp? actually 1e-5
        # propose move: move mass from i to j
        i, j = np.random.randint(0, n, size=2)
        if a[i] == 0: continue
        eps_move = np.random.rand() * 0.05  # up to 5%
        delta = eps_move * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        conv_new, max_new = conv_max(a_new)
        new_score = 2.0 * n * max_new / (a_new.sum()**2)
        diff = new_score - cur_score
        if diff < 0 or np.random.rand() < math.exp(-diff / (temp + 1e-12)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] SA improve {best_score:.6f}")

    # Phase 3: Random restart refinement (optional)
    # If time left, we can generate new random seeds and do quick local search.
    while time.time() < deadline:
        # generate random Dirichlet with concentration 0.5 maybe to have sparse distribution
        conc = 0.5
        a_rand = np.random.gamma(conc, 1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        # run few hill climbing steps
        cur_seq = a_rand
        cur_conv, cur_max = conv_max(cur_seq)
        cur_sc = 2.0*n*cur_max/(cur_seq.sum()**2)
        # small local search
        for _ in range(200):
            if time.time() >= deadline: break
            i, j = np.random.randint(0, n, size=2)
            if cur_seq[i] == 0: continue
            eps_move = np.random.rand()*0.05
            delta = eps_move * cur_seq[i]
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            conv_c, max_c = conv_max(cand)
            cand_sc = 2.0*n*max_c/(cand.sum()**2)
            if cand_sc < cur_sc:
                cur_seq = cand
                cur_sc = cand_sc
        if cur_sc < best_score:
            best_score = cur_sc
            best_a = cur_seq.copy()
            print(f"[{time.time() - start:.2f}s] random restart improvement {best_score:.6f}")

    # final output
    print(f"[{time.time() - start:.2f}s] finished best score {best_score:.6f}")
    # return as list of floats
    return [float(x) for x in best_a]

import time, math, numpy as np

def conv_full(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n - 1]
    return conv

def max_conv(a):
    conv = conv_full(a)
    return conv.max()

def score_of(a):
    n = a.shape[0]
    maxv = max_conv(a)
    # sum is roughly 1, but compute general
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * maxv / (s * s)
```

            a[j] += delta

            a /= a.sum()

    # Use the best sequence as a starting point.

propose_candidate –  hybrid search for a low‑∞‑norm auto‑convolution step function.

The routine combines three complementary stages within the allotted time:

1. **Parameter sweep** – a collection of analytically‑motivated families
   (power‑product, power‑distance, Gaussian, exponential decay and random
   Dirichlet) is explored on a grid of parameters.  Each candidate is
   normalised to unit ℓ₁‑mass and its score
       2·n·max(conv(a,a)) / (sum(a)²)
   is evaluated (the same expression used by ``evaluate_sequence``).  The best
   sequence found so far is kept.

2. **Weighted mirror‑descent** – starting from the best candidate,
   an exponentiated‑gradient (mirror‑descent) update is performed on the
   simplex.  The sub‑gradient of the (smoothed) max‑convolution is
   obtained from a small *active* set of indices near the current maximum;
   a soft‑max with temperature ``τ`` yields a weighted gradient.
   The step‑size decays as η₀/√t.  Periodic tiny random mass‑re‑allocation
   helps to escape shallow plateaus.

3. **Simulated‑annealing hill‑climbing** – the best sequence obtained so
   far is refined by stochastic moves that transfer a random fraction of
   mass from one position to another.  Moves are accepted according to a
   temperature schedule, allowing occasional uphill steps.

The whole procedure respects the 1000 s wall‑time limit and returns the
best non‑negative height vector it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of sufficient length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return _full_convolution(a).max()


def _score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence``."""
    n = a.shape[0]
    max_val = _max_convolution(a)
    s = a.sum()
    # ``evaluate_sequence`` treats sums <0.01 as invalid – we avoid that.
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Shape generators ------------------------------------------------------
# ----------------------------------------------------------------------
def _shape_power_product(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n-i))**p   (symmetric, peaks in the centre for p>0)."""
    i = np.arange(n, dtype=float)
    a = ((i + 1) * (n - i)) ** p
    return a


def _shape_power_distance(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (|i‑mid|+1)**p   (controls concentration toward the ends for p<0)."""
    mid = (n - 1) / 2.0
    d = np.abs(np.arange(n, dtype=float) - mid) + 1.0
    a = d ** p
    return a


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """a[i] ∝ exp(‑(i‑mid)² / (2σ²))."""
    mid = (n - 1) / 2.0
    d2 = (np.arange(n, dtype=float) - mid) ** 2
    a = np.exp(-d2 / (2.0 * sigma * sigma))
    return a


def _shape_exponential_decay(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(‑λ·i)   (mass concentrated near the left boundary)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a


def _shape_random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Random Dirichlet draw with concentration ``conc``."""
    a = np.random.gamma(conc, 1.0, size=n)
    return a
