Propose a step‑function sequence that minimizes
`evaluate_sequence`.  The optimisation proceeds in three
phases:

1.  **Progressive simplex gradient descent** – start with a modest
    length and repeatedly up‑sample to larger sizes (800 → 2400 → 4800
    → 9600).  After each projection onto the probability simplex we
    enforce symmetry (a_i = a_{n‑1‑i}) which is natural for the
    self‑convolution problem.  A simple momentum scheme with a
    diminishing step size is used.

2.  **Symmetrised mass‑transfer annealing** – a cheap simulated‑annealing
    walk that moves a small fraction of mass between two positions,
    re‑symmetrises the candidate and accepts downhill or occasional
    uphill moves.  This refines the best sequence found by the gradient
    phase.

3.  **Final normalisation** – the best candidate is returned as a plain
    Python list.  All values are non‑negative and sum to one, which
    guarantees that `evaluate_sequence` returns `2·n·max(conv)`.

The routine respects the supplied time budget (default 1000 s) and
prints periodic progress information.
"""

# Simplex utilities

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (linear, non‑circular) using FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – raw (no averaging)
# ----------------------------------------------------------------------
def _subgradient_raw(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Symmetrisation (a_i = a_{n‑1‑i})
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry; the sum stays unchanged."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up.fill(1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape generators (used as deterministic starters when no prior
# sequence is available).
# ----------------------------------------------------------------------
def _gen_shape(n: int, kind: str = "gaussian", sigma_factor: float = 0.20) -> np.ndarray:
    """Return a symmetric, non‑negative shape of length n."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    if kind == "gaussian":
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    elif kind == "triangular":
        a = 1.0 - np.abs((x - centre) / (n / 2.0))
        a = np.maximum(a, 0.0)
    elif kind == "cosine":
        a = np.cos(np.pi * (x - centre) / n) + 1.0
    else:  # uniform fallback
        a = np.ones_like(x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a.fill(1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected (Nesterov‑style) gradient descent on the simplex.
# ----------------------------------------------------------------------
def _run_proj_grad(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   beta: float = 0.9,
                   step0: float = 0.6,
                   max_iter: int = 10 ** 7) -> Tuple[np.ndarray, float]:
    """
    Perform many projected‑gradient iterations.
    Returns the best sequence encountered and its scaled score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # gradient of the max‑convolution term
        grad = _subgradient_raw(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto the simplex
        a = _project_simplex(a - step * momentum)

        # enforce symmetry (keeps the sum unchanged)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement that respects symmetry.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 300_000) -> Tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealing, keeping the sequence symmetric."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # propose moving a modest fraction of mass from i to j
        eps = a[i] * np.random.rand() * 0.10  # ≤10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # re‑symmetrise the proposal (keeps sum = 1)
        a_prop = _symmetrize(a_prop)

        # compute new score
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the time budget,
    progressively enlarges the problem size, runs projected gradient
    descent (with symmetry) and finishes with a symmetric
    simulated‑annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None          # type: np.ndarray | None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter sequence from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definition (lengths) – larger n gets a proportionally larger
    # slice of the gradient budget.
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]          # progressive up‑sampling
    total_weight = sum(stage_lengths)

    # allocate 85 % of the budget to gradient work, 15 % to final annealing
    total_time = deadline - start_time
    anneal_budget = max(30.0, 0.15 * total_time)
    grad_budget = total_time - anneal_budget

    # ------------------------------------------------------------------
    # Gradient phases
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time allotted for the current stage (proportional to n)
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # --------------------------------------------------------------
        # Initial point for this stage
        # --------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # deterministic deterministic shape when nothing else is available
            a0 = _gen_shape(n, kind="gaussian", sigma_factor=0.18)

        # --------------------------------------------------------------
        # Run projected gradient descent
        # --------------------------------------------------------------
        a_opt, score_opt = _run_proj_grad(
            a0, n, stage_deadline, start_time,
            beta=0.9, step0=0.6, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)
        total_weight = sum(stage_lengths[stage_lengths.index(n) + 1:]) or 1

    # ------------------------------------------------------------------
    # Final annealing phase – works on the last length we ended with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            max_steps=500_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list for the harness
    return [float(x) for x in best_seq]
