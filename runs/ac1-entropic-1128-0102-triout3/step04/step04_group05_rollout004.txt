def isotonic_decreasing(y):
    y = np.asarray(y, dtype=float)
    n = y.shape[0]
    level = y.tolist()
    weight = [1.0] * n
    size = [1] * n
    i = 0
    while i < len(level) - 1:
        if level[i] >= level[i+1] - 1e-14:  # ok
            i += 1
        else:
            # merge i and i+1
            total_weight = weight[i] + weight[i+1]
            total_size = size[i] + size[i+1]
            new_level = (level[i] * weight[i] + level[i+1] * weight[i+1]) / total_weight
            level[i] = new_level
            weight[i] = total_weight
            size[i] = total_size
            # delete i+1
            del level[i+1]
            del weight[i+1]
            del size[i+1]
            # step back if possible
            i = max(i-1, 0)
    # expand
    result = np.empty(n, dtype=float)
    idx = 0
    for lev, sz in zip(level, size):
        result[idx: idx+sz] = lev
        idx += sz
    return result

"""
Monotone‑simplex Adam optimiser.

The evaluation metric is
    score = 2·n·max(conv(a,a)) / (sum(a)^2)

Because the metric is homogeneous we may restrict the search to the
probability simplex (sum(a)=1, a_i≥0).  A classic result (rearrangement
inequality) shows that for a fixed length the convolution is smallest
when the vector is sorted in non‑increasing order, i.e. the shape is
monotone decreasing.  This implementation therefore projects every
iterate onto the *monotone simplex* (non‑increasing, non‑negative,
unit‑sum) after each Adam update.

The optimiser works on several resolutions:
* a short list of hand‑crafted decreasing shapes is up‑sampled to the
  current length,
* a few random Dirichlet draws (sorted decreasing) are added,
* each candidate is refined by Adam with a smooth‑max approximation of
  the max‑norm,
* the best candidate is kept and carried to the next resolution,
* finally a long low‑learning‑rate run is performed on the overall best
  shape.

All work is bounded by the supplied time budget (default 1000 s).  The
returned list is a plain Python list of non‑negative floats whose sum
equals one (the scaling is irrelevant for the score).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_monotone_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the set
        { x | x_0 ≥ x_1 ≥ … ≥ x_{n‑1} ≥ 0 ,  sum_i x_i = 1 }.
    The implementation first performs isotonic regression (PAVA) to enforce
    monotonicity, then clips negatives to zero and finally rescales to unit sum.
    """
    n = v.shape[0]

    # ----- isotonic regression (non‑increasing) -----
    # PAVA for decreasing order
    level = v.astype(float).tolist()
    weight = [1.0] * n
    size = [1] * n      # block sizes (used for final expansion)

    i = 0
    while i < len(level) - 1:
        if level[i] >= level[i + 1] - 1e-14:
            i += 1
        else:
            # merge blocks i and i+1
            total_weight = weight[i] + weight[i + 1]
            total_size = size[i] + size[i + 1]
            new_level = (level[i] * weight[i] + level[i + 1] * weight[i + 1]) / total_weight
            level[i] = new_level
            weight[i] = total_weight
            size[i] = total_size
            # delete i+1
            del level[i + 1]
            del weight[i + 1]
            del size[i + 1]
            i = max(i - 1, 0)

    # expand the block averages back to a full length vector
    proj = np.empty(n, dtype=float)
    pos = 0
    for lev, sz in zip(level, size):
        proj[pos: pos + sz] = lev
        pos += sz

    # ----- enforce non‑negativity -----
    proj = np.maximum(proj, 0.0)

    # ----- renormalise to the simplex -----
    total = proj.sum()
    if total <= 0.0:               # fall back to uniform if everything vanished
        proj[:] = 1.0 / n
    else:
        proj /= total

    return proj


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """max( a * a ) and full convolution (real part).  a is assumed to sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. a.
    Implements   g_i = 2 * Σ_k w_k * a_{k-i}   using FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize_monotone(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.05,
    temp_end: float = 0.0005,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv) on the *monotone simplex*.

    Returns (final_a, best_seen_a, best_score).
    """
    a = _project_monotone_simplex(a0.copy())
    n = a.shape[0]

    # cached FFT lengths (powers of two)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # track best
    cur_max, _ = _conv_max(a)
    best_score = 2.0 * n * cur_max
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- compute smooth‑max weights ----
        cur_max, conv = _conv_max(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max temperature schedule (linear in iteration count)
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)

        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # ---- gradient ----
        g = _gradient_from_weights(a, w, fft_len_grad)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ---- projection onto monotone simplex ----
        a = _project_monotone_simplex(a)

    return a, best_a, best_score


def _initial_candidates(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a small pool of diverse decreasing candidates for length n."""
    cand = []

    # 1. uniform (trivially decreasing)
    cand.append(np.full(n, 1.0 / n))

    # 2. linear decreasing to zero
    idx = np.arange(n)
    lin = (n - idx).astype(float)
    lin /= lin.sum()
    cand.append(lin)

    # 3. exponential decreasing (scale chosen so that tail is not too thin)
    exp = np.exp(-idx / (n / 5.0))
    exp /= exp.sum()
    cand.append(exp)

    # 4. raised‑cosine (Hann) – also decreasing on the left half, we keep the left half only
    hann_full = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    # enforce monotone by taking the decreasing side
    hann = np.maximum.accumulate(hann_full[::-1])[::-1]   # make it non‑increasing
    hann /= hann.sum()
    cand.append(hann)

    # 5‑6. a couple of random Dirichlet draws, sorted decreasing
    for _ in range(2):
        rand = rng.dirichlet(np.ones(n))
        rand.sort()
        rand = rand[::-1]                 # now decreasing
        rand /= rand.sum()
        cand.append(rand)

    # 7. up‑sample previous best if available and of different length
    if previous is not None and previous.shape[0] != n:
        x_old = np.linspace(0.0, 1.0, previous.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, previous)
        up = np.maximum(up, 0.0)
        up_sum = up.sum()
        if up_sum > 0:
            up /= up_sum
        else:
            up = np.full(n, 1.0 / n)
        cand.append(up)

    return cand
