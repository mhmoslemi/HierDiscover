"""
Propose a step‑function that minimizes the supplied `evaluate_sequence`
objective.

Approach
--------
1. **Diverse initialisations** – for a given length we generate a handful of
   classic window functions (uniform, Hann, Hamming, Blackman, Bartlett,
   Kaiser with several β, Gaussian with several σ).  Each candidate is
   normalised to sum = 1 and evaluated; the best one seeds the optimiser.

2. **Projected (sub‑)gradient descent** – the objective for a normalised
   sequence `a` (∑a = 1) is `2·n·max(conv(a))`.  Its sub‑gradient can be taken
   from the index where the convolution attains its maximum:
   for `k = argmax(conv)` we have

        ∂max(conv)/∂a_i = 2·a_{k-i}   if 0 ≤ k-i < n, else 0.

   We move opposite to this gradient, use Nesterov‑style momentum and
   project back onto the probability simplex after each step.
   The step size decays like `step0 / sqrt(t)`.

3. **Progressive up‑sampling** – we start with a modest size (≈800) and
   repeatedly up‑sample the current best sequence (linear interpolation)
   to larger lengths (≈2400 → 4800 → 9600).  After each up‑sampling we run
   the gradient optimiser again, keeping the best sequence seen so far.

4. **Local stochastic refinement** – with the remaining time we run a
   simulated‑annealing walk that moves a small fraction of mass between two
   positions.  Downhill moves are always accepted, uphill moves follow a
   temperature schedule.

All phases respect the global time budget (default 1000 s).  The routine
returns a plain Python list of non‑negative floats (the final sequence).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient from the *single* maximal convolution entry
# ----------------------------------------------------------------------
def _grad_from_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Return the (sub‑)gradient of max(conv) w.r.t. `a`,
    using only the index where conv attains its maximum.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # reflected slice, factor 2
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent using the max‑convolution gradient
# ----------------------------------------------------------------------
def _run_gradient_descent_max(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              beta: float = 0.7,
                              step0: float = 0.8,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Run many iterations of projected gradient descent (Nesterov momentum)
    minimising 2·n·max(conv(a)).  Returns the best point seen and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, it={it}, score={cur_score:.6f}")

        # gradient from the active (max) entry only
        grad = _grad_from_max(conv, a)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # projected gradient step
        a = _project_simplex(a - step * momentum)

        # occasional tiny random jitter if we got stuck for a while
        if it - last_improve > 5000:
            jitter = np.random.rand(n) * 5e-5
            a = _project_simplex(a + jitter)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny “move ε from i to j’’ annealing.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up‑to 12 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis rule
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Window‑function factories (all return a non‑negative vector with sum = 1)
# ----------------------------------------------------------------------
def _window_uniform(n):
    return np.full(n, 1.0 / n)


def _window_hann(n):
    w = np.hanning(n)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _window_hamming(n):
    w = np.hamming(n)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _window_blackman(n):
    w = np.blackman(n)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _window_bartlett(n):
    w = np.bartlett(n)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _window_kaiser(n, beta):
    w = np.kaiser(n, beta)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _window_gaussian(n, sigma):
    x = np.linspace(-0.5, 0.5, n)
    w = np.exp(-0.5 * (x / sigma) ** 2)
    if w.sum() == 0:
        w = _window_uniform(n)
    else:
        w = w / w.sum()
    return w


def _candidate_windows(n):
    """Return a list of diverse initial candidates for length `n`."""
    cand = [_window_uniform(n),
            _window_hann(n),
            _window_hamming(n),
            _window_blackman(n),
            _window_bartlett(n)]

    for beta in (2, 4, 6, 8, 10, 12):
        cand.append(_window_kaiser(n, beta))
    for sigma in (0.12, 0.18, 0.24, 0.30, 0.36):
        cand.append(_window_gaussian(n, sigma))
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    performs multi‑scale projected‑gradient optimisation seeded by a diverse
    set of window functions, and finishes with a short simulated‑annealing
    refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter: reuse a previously found construction if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Define progressive sizes (increase resolution)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # we may stop earlier if time is short

    # Allocate ~80 % of the total budget to gradient phases, the rest to annealing.
    grad_budget = max(20.0, 0.80 * (deadline - start_time))
    anneal_budget = max(10.0, (deadline - start_time) - grad_budget)
    total_weight = sum(stage_lengths)

    for n in stage_lengths:
        if time.time() >= deadline - anneal_budget - 2.0:   # keep a buffer for annealing
            break

        # time slice for this stage (proportional to its size)
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # ------------------------------------------------------------------
        # Initialise a0 : up‑sample previous best if we have one, otherwise pick
        # the best among a small library of window functions.
        # ------------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # evaluate a few hand‑crafted windows and keep the best one
            cand = _candidate_windows(n)
            best_local = None
            best_local_score = float("inf")
            for seq in cand:
                # using evaluate_sequence directly ensures identical handling
                sc = evaluate_sequence(seq.tolist())
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = seq
            a0 = best_local if best_local is not None else np.full(n, 1.0 / n)

        # ------------------------------------------------------------------
        # Run the projected gradient optimiser on this size
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_gradient_descent_max(a0, n,
                                                     stage_deadline,
                                                     start_time,
                                                     beta=0.7,
                                                     step0=0.8,
                                                     max_iter=10**7)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # Update remaining gradient budget after the stage
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works on the largest sequence we obtained.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(best_seq, n_final,
                                          anneal_deadline,
                                          start_time,
                                          best_score,
                                          best_seq,
                                          start_temp=5e-2,
                                          end_temp=1e-5,
                                          steps=400_000)
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never trigger.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
