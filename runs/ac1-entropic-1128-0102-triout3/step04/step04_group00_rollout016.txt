import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (>=0, sum=1)."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(a):
    """Linear convolution of a with itself via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv

def _conv_fft(x, y):
    """Linear convolution of x and y using FFT (both 1‑D arrays)."""
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv

def _smooth_gradient(a, beta):
    """
    Gradient of the smooth approximation
          f_beta(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    where conv = a*a.
    Returns ∇f_beta(a) (shape = a.shape).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n-1
    if beta <= 0.0:
        # Uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        # softmax on conv
        maxc = conv.max()
        # Shifted for numerical stability
        exp_term = np.exp(beta * (conv - maxc))
        w = exp_term / exp_term.sum()
    # gradient = 2 * sum_j a_j * w_{i+j}
    # Compute via convolution of a and w_rev
    w_rev = w[::-1]                           # length 2n-1
    conv2 = _conv_fft(a, w_rev)               # length 3n-2
    L = w_rev.size
    grad = 2.0 * conv2[L - n : L][::-1]        # pick n entries and reverse
    return grad, conv

def _adam_descent(a0, n, deadline, start_time,
                  beta_schedule, lr0=0.2,
                  report_every=5000):
    """
    Adam‑style projected gradient descent on the simplex,
    using the smoothed objective with a schedule of β values.
    Returns (best_a, best_score).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # handle schedule iterator
    for beta, max_it in beta_schedule:
        for _ in range(max_it):
            it += 1
            if time.time() >= deadline:
                return best_a, best_score

            grad, conv = _smooth_gradient(a, beta)
            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr0 / (np.sqrt(v_hat) + eps)

            a_new = a - step * m_hat
            a = _project_simplex(a_new)

            if it % report_every == 0:
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] iter={it:,}, β={beta:.1f}, score={cur_score:.6f}")

    return best_a, best_score

def _anneal_refine(a0, deadline, start_time, initial_temp=5e-2,
                   final_temp=1e-6, steps=200_000, report_every=5000):
    """
    Simple simulated‑annealing refinement that moves a small ε
    from a “source” index to a random “destination” index.
    Returns (best_a, best_score).
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = initial_temp * (final_temp / initial_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % report_every == 0:
                print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,}, score={best_score:.6f}")

    return best_a, best_score

def _upsample(seq, new_n):
    """Linear interpolation from old_n points to new_n points, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    L = w_rev.size
    grad = 2.0 * conv2[L - n : L][::-1]        # pick n entries, reverse order
    return grad, conv


# ------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex
# ------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  beta_schedule,
                  lr0: float = 0.2,
                  report_every: int = 2000):
    """
    Perform Adam‑style projected gradient descent.
    beta_schedule = [(β, max_iter), …] .
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    for beta, max_it in beta_schedule:
        for _ in range(max_it):
            it += 1
            if time.time() >= deadline:
                return best_a, best_score

            grad, conv = _smooth_gradient(a, beta)

            # Adam moments
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr0 / (np.sqrt(v_hat) + eps)

            a_new = a - step * m_hat
            a = _project_simplex(a_new)

            if it % report_every == 0:
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] iter={it:,}, "
                          f"β={beta:.1f}, score={cur_score:.6f}")

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass transfers)
# ------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   initial_temp: float = 5e-2,
                   final_temp: float = 1e-6,
                   steps: int = 150_000,
                   report_every: int = 2000):
    """
    Small‑scale simulated annealing: move a random epsilon from a
    high‑mass index to a random destination.
    Returns (best_seq, best_score).
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = initial_temp * (final_temp / initial_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            if it % report_every == 0:
                print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,}, "
                      f"score={best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Linear interpolation to a new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate seq to length new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising `evaluate_sequence`.
    The procedure follows the scheme described in the module docstring
    and respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # small safety margin

    # ------------------------------------------------------------------
    # 1. Starting vector (reuse previous construction if available)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    n_target = 4800
    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # 2. First Adam descent (coarse β schedule)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    grad_time = max(30.0, 0.60 * remaining)
    grad_deadline = time.time() + grad_time

    beta_schedule = [
        (0.5,   5_000),
        (2.0,  10_000),
        (5.0,  15_000),
        (10.0, 20_000),
        (30.0, 25_000),
        (80.0, 30_000),
    ]

    best_a, best_score = _adam_descent(
        a0, n_target, grad_deadline, start_time,
        beta_schedule, lr0=0.20, report_every=2000
    )

    # ------------------------------------------------------------------
    # 3. Fine‑grained descent with larger β (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time() - 1.0   # leave a second buffer
        if remaining > 5.0:
            fine_deadline = time.time() + remaining
            fine_schedule = [
                (120.0, 20_000),
                (200.0, 20_000),
                (400.0, 20_000),
            ]
            best_a, best_score = _adam_descent(
                best_a, n_target, fine_deadline, start_time,
                fine_schedule, lr0=0.12, report_every=1000
            )

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing refinement (tiny moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _anneal_refine(
            best_a, deadline, start_time,
            initial_temp=3e-2,
            final_temp=5e-6,
            steps=150_000,
            report_every=2000
        )

    # ------------------------------------------------------------------
    # 5. Normalise and convert to plain Python list
    # ------------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    return [float(x) for x in best_a]
