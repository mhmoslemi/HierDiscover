def shape_quad_v(n, r, s):
    idx = np.arange(n)
    d = idx - (n - 1) / 2.0
    scaled = d / (n / 2.0)  # range approx [-1,1]
    shape = 1.0 + r * (scaled ** 2) + s * np.abs(scaled)
    shape = np.maximum(shape, 0.0)
    # normalise to sum 1
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full_like(shape, 1.0 / n)
    return shape

"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works in three stages for a given length `n`:

1. **Low‑dimensional parametric hill‑climbing** – we search a simple
   convex “U‑shaped’’ family `a_i ∝ 1 + r·((i−c)/c)² + s·|i−c|/c`
   (where `c = n/2`).  The family has only two non‑negative parameters
   `(r,s)`.  A fast simulated‑annealing style loop explores the
   (r,s)‑plane and keeps the best candidate (objective = 2·n·max(conv)).
2. **Fine‑grained smoothed log‑sum‑exp gradient descent** – starting from
   the best vector of stage 1 we run a projected gradient descent on the
   probability simplex.  The surrogate loss is the log‑sum‑exp of the
   self‑convolution entries; a decreasing smoothing parameter λ and
   momentum give fast convergence.
3. **Mass‑transfer annealing** – a cheap “move ε from i to j’’ move
   polishes the solution, accepting occasional uphill steps with an
   exponential cooling schedule.

The routine proceeds through an increasing length schedule,
upsampling the current best vector as a warm‑start for the next length.
All work respects the global time budget (`budget_s` seconds).  The
returned list is a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors via FFT (no padding beyond needed)."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Objective and gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a normalised sequence: 2·n·max(conv)."""
    n = len(seq)
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerically stable soft‑max over all convolution entries
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # ∂L/∂a_i
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_smooth_descent(
    a0: np.ndarray,
    n: int,
    slice_deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Smoothed log‑sum‑exp gradient descent with momentum.
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.6                 # slightly smaller than the previous code

    # λ‑schedule: start relatively large, halve every few thousand steps
    lam_factor = 0.5
    lam_decay = 0.70
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective (sum(a)=1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_score:.6f}"
            )

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise
        if it % 400 == 0:
            noise_scale = 0.015 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a small ε from source i to destination j.
    Returns the (possibly) improved sequence and its exact objective.
    """
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2  # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Parametric U‑shape family and its optimiser
# ----------------------------------------------------------------------


def _shape_quad_v(n: int, r: float, s: float) -> np.ndarray:
    """
    Convex “U’’ family a_i ∝ 1 + r·((i−c)/c)² + s·|i−c|/c,  c=n/2.
    Parameters r,s≥0 control convexity; the vector is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    scaled = (idx - c) / (n / 2.0)          # approx [-1,1]
    shape = 1.0 + r * (scaled ** 2) + s * np.abs(scaled)
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full_like(shape, 1.0 / n)
    return shape


def _optimize_parameters(
    n: int,
    slice_deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Low‑dimensional hill‑climbing on the (r,s)‑plane of the U‑shape family.
    Returns the best sequence found and its exact objective value.
    """
    best_r, best_s = 0.0, 0.0
    best_seq = _shape_quad_v(n, best_r, best_s)
    best_score = _objective(best_seq)

    sigma_r, sigma_s = 0.5, 0.5
    it = 0

    while time.time() < slice_deadline:
        it += 1
        # Gaussian proposal (keep parameters non‑negative)
        r = max(0.0, best_r + sigma_r * np.random.randn())
        s = max(0.0, best_s + sigma_s * np.random.randn())
        seq = _shape_quad_v(n, r, s)
        score = _objective(seq)

        if score < best_score:
            best_r, best_s = r, s
            best_seq, best_score = seq, score
            print(
                f"[{time.time() - start_time:6.2f}s] n={n} param improve r={r:.4f} s={s:.4f} score={score:.6f}"
            )
            sigma_r = min(sigma_r * 1.2, 5.0)
            sigma_s = min(sigma_s * 1.2, 5.0)
        else:
            # slowly shrink the search radius
            sigma_r = max(sigma_r * 0.99, 1e-3)
            sigma_s = max(sigma_s * 0.99, 1e-3)

        # occasional random restart of the step sizes
        if it % 1000 == 0 and random.random() < 0.1:
            sigma_r = max(sigma_r, 0.5)
            sigma_s = max(sigma_s, 0.5)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
    except Exception:
        starter = None

    # Length schedule – grow gradually; we stop early if time runs out
    lengths = [4096, 8192, 16384, 32768]
    current_seq = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Allocate a slice of the remaining budget for this length.
        # The final (largest) length gets the biggest share.
        if idx < len(lengths) - 1:
            slice_frac = 0.22
        else:
            slice_frac = 0.5
        slice_deadline = time.time() + max(5.0, remaining * slice_frac)

        # ------------------------------------------------------------------
        # Initialise a0 (warm‑start or uniform)
        # ------------------------------------------------------------------
        if current_seq is not None and current_seq.shape[0] == n:
            a0 = current_seq.copy()
        elif current_seq is not None:
            a0 = _upsample(current_seq, n)
        else:
            a0 = np.full(n, 1.0 / n)

        # ------------------------------------------------------------------
        # Stage 1 – parametric U‑shape hill‑climbing
        # ------------------------------------------------------------------
        seq_param, score_param = _optimize_parameters(n, slice_deadline, start_time)
        if score_param < best_score:
            best_score = score_param
            best_seq = seq_param.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Global improve (param) n={n} score={score_param:.6f}"
            )

        # ------------------------------------------------------------------
        # Stage 2 – smoothed gradient descent (fine tuning)
        # ------------------------------------------------------------------
        # Choose the better of the parametric result and the original a0.
        if score_param <= _objective(a0):
            start_vec = seq_param
        else:
            start_vec = a0

        if time.time() < deadline:
            grad_deadline = time.time() + max(2.0, (deadline - time.time()) * 0.35)
            a_opt, score_opt = _run_smooth_descent(start_vec, n, grad_deadline, start_time)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Global improve (grad) n={n} score={score_opt:.6f}"
                )

        # ------------------------------------------------------------------
        # Stage 3 – annealing polishing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(1.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=20000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Global improve (anneal) n={n} score={score_ref:.6f}"
                )

        # Prepare warm‑start for the next (larger) length
        current_seq = best_seq.copy()

    # ----------------------------------------------------------------------
    # Fallback – ensure a valid output
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256.0)

    # Return a plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
