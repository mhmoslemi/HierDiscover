"""
Low‑maximum self‑convolution search.

The optimizer works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It combines three stages:

1. **Projected smoothed gradient descent** – we minimise a soft‑max
   surrogate of the convolution using a decreasing smoothing parameter
   λ and classic momentum.  The descent is run for a generous time slice
   on each problem size.

2. **Targeted annealing** – after the smooth phase a cheap mass‑transfer
   simulated‑annealing refines the solution by moving a small ε from a
   source index to a destination index.

3. **Exact sub‑gradient polishing** – finally we directly reduce the
   true ℓ∞‑norm of the convolution.  At the current peak index k we
   compute the sub‑gradient ∂‖a∗a‖∞/∂a = 2·a_{k‑i} and perform a
   projected gradient step with a decaying step size.

The routine progressively upsamples the current best distribution to
larger lengths (512 → 1024 → … → 16384).  After the final length a few
random restarts with the three‑stage pipeline are performed in the
remaining budget.  The best sequence found is returned as a plain list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max over convolution entries
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad

# Stage 1 – smoothed gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Smoothed log‑sum‑exp gradient descent with momentum.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')

    beta = 0.9                # momentum decay
    step0 = 0.8               # initial step size (decays as 1/√it)

    # λ‑schedule: start relatively large, halve every few thousand steps
    lam_factor = 0.5          # λ = lam_factor * max(conv)
    lam_decay = 0.5
    lam_decay_interval = 2500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b               # exact objective (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_score:.6f}")

        # update λ (never let it become exactly zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # tiny isotropic noise occasionally (helps escaping very flat plateaus)
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a small ε from i to j.
    Returns the (possibly) improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of the source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – exact sub‑gradient polishing of the true ℓ∞‑norm
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     slice_deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true max‑convolution objective.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    step0 = 0.5                     # initial step size (decays as 1/√it)

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        # index where the max is attained (pick first if several)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑conv at a (vector of size n)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            grad[i] = 2.0 * a[j]          # derivative of (a*a)[k] w.r.t a_i

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * grad)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a plain list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from the previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start modest and climb up to ~16k.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)   # avoid systematic bias when stopping early

    global_a = starter

    # ------------------------------------------------------------------
    # Main multiscale loop
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # initialise a0 (warm start / random)
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a time slice for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # later (larger) lengths get a slightly larger share
        fraction = 0.15 / (len(lengths) - idx)   # ≈15 % split, more for later sizes
        slice_deadline = time.time() + max(8.0, remaining * fraction)

        # ---- smoothed gradient descent ----
        a_opt, score_opt = _run_smooth_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient n={n} score={score_opt:.6f}")

        # ---- annealing refinement (if time permits) ----
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=50000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # ---- exact sub‑gradient polishing (if time permits) ----
        if time.time() < deadline:
            sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_sub, score_sub = _run_subgradient(best_seq, n, sub_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={score_sub:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final random‑restart fine‑tuning on the largest size (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_big = best_seq.shape[0]
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break
            # inject a modest random perturbation
            a0 = best_seq + 0.02 * np.random.randn(n_big)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

            # give each restart a fresh slice (roughly equal share)
            remaining = deadline - time.time()
            slice_deadline = time.time() + max(5.0, remaining / (restarts - r))

            a_opt, score_opt = _run_smooth_descent(a0, n_big, slice_deadline, start_time)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart {r+1} improve after gradient score={score_opt:.6f}")

            if time.time() < deadline:
                anneal_deadline = time.time() + max(2.0, (deadline - time.time()) * 0.3)
                a_ref, score_ref = _run_annealing(
                    a_opt,
                    n_big,
                    anneal_deadline,
                    start_time,
                    best_score,
                    best_seq,
                    start_temp=5e-2,
                    end_temp=1e-5,
                    steps=30000,
                )
                if score_ref < best_score:
                    best_score = score_ref
                    best_seq = a_ref.copy()
                    print(f"[{time.time() - start_time:6.2f}s] restart {r+1} improve after anneal score={score_ref:.6f}")

            if time.time() < deadline:
                sub_deadline = time.time() + max(2.0, (deadline - time.time()) * 0.3)
                a_sub, score_sub = _run_subgradient(best_seq, n_big, sub_deadline, start_time)
                if score_sub < best_score:
                    best_score = score_sub
                    best_seq = a_sub.copy()
                    print(f"[{time.time() - start_time:6.2f}s] restart {r+1} improve after subgrad score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
