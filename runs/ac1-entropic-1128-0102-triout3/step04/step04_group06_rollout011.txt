"""
Propose a step‑function that (hopefully) beats the previous 1.5159 bound.
The algorithm mixes three ideas that are deliberately different from the
reference implementation:

1. **Diverse initialisations** – the best known construction, a uniform
   distribution and several Beta‑shaped densities (including the
   arcsine‑like ½,½ case) are tried.  Random Dirichlet draws give further
   diversity.

2. **Exponentiated‑mirror descent** on the simplex (the same geometry as the
   reference code) but with a larger, adaptive active set and a slightly
   larger initial step size.  The active set is built from all convolution
   entries that lie within a tiny relative tolerance of the current maximum.
   Random perturbations are injected more aggressively.

3. **Gradient‑guided hill‑climbing and a fast simulated‑annealing phase**.
   After the mirror‑descent phase the sub‑gradient (computed from the same
   active set) points to the indices that most contribute to the peak of the
   auto‑convolution; mass is moved deterministically from the “worst’’ to the
   “best’’ index.  When no improvement is found a classic annealing loop is
   entered that accepts occasional uphill moves, allowing the search to leave
   shallow plateaux.

The whole routine respects the 1000 s budget, prints timing information and
returns the best sequence found as a plain list of floats (non‑negative,
already normalised to sum = 1; the `evaluate_sequence` routine is scale‑invariant).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of ``a`` (via FFT) and its maximum entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a compact set of indices that are “active’’ for the sub‑gradient.
    All indices within ``rel_eps`` relative distance of ``max_val`` are kept.
    If fewer than ``max_add`` such indices exist, the globally largest ones
    are added so that the set contains at most ``max_add`` entries.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of the maximum auto‑convolution using the active set ``K``.
    For each active convolution index k the contribution to component i is
    ``2·a_{k‑i}`` (when the index falls inside the array bounds).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _score_from_a(a: np.ndarray, n: int) -> float:
    """Exact value of ``evaluate_sequence`` for a normalised (sum=1) vector."""
    max_val, _ = _conv_and_max(a)
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
def _optimise_sequence(a_init: np.ndarray, n: int,
                       start_time: float, end_time: float):
    """
    Starting from ``a_init`` (already normalised to sum = 1) run a three‑phase
    optimisation: mirror descent, deterministic hill‑climbing and simulated
    annealing.  The routine respects the provided ``end_time`` deadline and
    returns the best sequence encountered together with its score.
    """
    a = a_init.copy()
    best_seq = a.copy()
    best_score = _score_from_a(a, n)

    # ----- 1. long mirror‑descent ------------------------------------
    max_iters = 250_000
    eta0 = 1.2                 # a bit larger than the reference 0.9
    rel_eps = 0.001
    max_add = 30

    it = 0
    while time.time() < end_time and it < max_iters:
        it += 1

        max_val, conv = _conv_and_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] it={it:6d}  MD improve={cur_score:.6f}")

        # active set for the sub‑gradient
        K = _active_indices(conv, max_val, rel_eps=rel_eps, max_add=max_add)
        g = _gradient_active(a, K)

        # exponentiated‑gradient (mirror descent) update
        eta = eta0 / math.sqrt(it)
        a = np.maximum(a, 1e-15)                # avoid log(0)
        log_a = np.log(a) - eta * g
        log_a -= np.max(log_a)                  # numerical stabilisation
        a = np.exp(log_a)

        # tiny floor & renormalisation
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # occasional random shake‑up (helps escape plateaux)
        if it % 5000 == 0:
            i, j = np.random.randint(0, n, size=2)
            eps = np.random.rand() * 0.03        # up to 3 % mass shift
            delta = eps * a[i]
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # ----- 2. deterministic gradient‑guided hill‑climbing -----------
    while time.time() < end_time:
        max_val, conv = _conv_and_max(a)
        cur_score = 2.0 * n * max_val

        # a richer active set gives a more informative gradient
        K = _active_indices(conv, max_val, rel_eps=0.0005, max_add=80)
        g = _gradient_active(a, K)

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-14:
            break

        # move a modest fraction of mass from the “worst’’ to the “best’’ index
        delta = 0.04 * a[i_max]               # 4 % of the source mass
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        max_val_new, _ = _conv_and_max(a_new)
        new_score = 2.0 * n * max_val_new

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill improve={new_score:.6f}")
        else:
            # if the directed move fails, try a tiny random perturbation
            i, j = np.random.randint(0, n, size=2)
            eps = np.random.rand() * 0.015
            d = eps * a[i]
            a[i] -= d
            a[j] += d
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # ----- 3. fast simulated annealing ------------------------------
    # a gentle temperature schedule, accepting occasional uphill moves
    temperature = best_score * 0.01 if best_score > 0 else 1e-3
    while time.time() < end_time:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.04
        delta = eps * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        max_val_new, _ = _conv_and_max(a_new)
        new_score = 2.0 * n * max_val_new

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:6.2f}s] anneal improve={new_score:.6f}")
        else:
            # Metropolis acceptance
            accept_prob = math.exp(-(new_score - best_score) / temperature)
            if random.random() < accept_prob:
                a = a_new
        # cool down slowly
        temperature *= 0.99995
        if temperature < 1e-8:
            temperature = 1e-8

    return best_seq, best_score

    ``budget_s`` seconds, tries a variety of sensible initialisations
    (including Beta‑shaped families) and performs a three‑phase optimisation.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # safety margin to avoid being killed exactly at the limit
    deadline = start_time + budget_s - 2.0

    n = 5000                     # length of the step‑function

    # ------------------------------------------------------------------
    # 1. build a pool of diverse start vectors -------------------------
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best construction, if present
    try:
        h = np.array(height_sequence_1, dtype=float)
        if h.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, h.size)
            x_new = np.linspace(0.0, 1.0, n)
            h = np.interp(x_new, x_old, h)
        h = np.maximum(h, 0.0)
        if h.sum() == 0.0:
            h = np.full(n, 1.0 / n)
        else:
            h = h / h.sum()
        candidates.append(h)
    except Exception:
        pass

    # b) uniform distribution (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # c) Beta‑shaped families – symmetric (α = β) with several exponents
    alphas = [0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]
    x = (np.arange(n) + 0.5) / n          # centre of each bin
    for a in alphas:
        pdf = np.power(x, a - 1) * np.power(1 - x, a - 1)
        pdf = np.maximum(pdf, 1e-30)      # avoid exact zeros
        seq = pdf / pdf.sum()
        candidates.append(seq)

    # d) a few random Dirichlet draws for extra diversity
    for _ in range(3):
        seq = np.random.rand(n)
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq = seq / seq.sum()
        candidates.append(seq)

    # ------------------------------------------------------------------
    # 2. evaluate the raw candidates, keep the best as a seed ----------
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for idx, a in enumerate(candidates):
        sc = _score_from_a(a, n)
        print(f"[{time.time() - start_time:6.2f}s] init #{idx} score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3. main optimisation on the current best seed -------------------
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:                     # keep a few seconds as a safety net
        main_end = time.time() + 0.9 * remaining
        best_seq, best_score = _optimise_sequence(best_seq, n,
                                                  start_time, main_end)

    # ------------------------------------------------------------------
    # 4. quick secondary restarts if any time left --------------------
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # give each short restart about 30 s (or whatever remains)
        now = time.time()
        if now + 30.0 > deadline:
            break
        # random normalised start
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        seq, sc = _optimise_sequence(a0, n, start_time, now + 30.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart improved to {sc:.6f}")

    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    # convert to plain Python list as required by the external harness
    return [float(v) for v in best_seq]
