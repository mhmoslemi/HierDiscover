"""
Propose a candidate step‑function by combining three phases:

1. Fast exponentiated‑gradient descent on the simplex using a sub‑gradient of
   the max‑autoconvolution.  The sub‑gradient is obtained via an FFT‑based
   correlation with a weight vector that marks the currently active (almost
   maximal) convolution indices.

2. Optional “smooth‑max” gradient descent where the non‑smooth max is replaced
   by a log‑sum‑exp (temperature τ) surrogate – this gives a smoother landscape
   and helps to escape shallow plateaus.

3. A simulated‑annealing local search that accepts occasional uphill moves.
   All phases respect the non‑negativity and simplex constraints and run
   within the given time budget.  The best sequence seen is returned.
"""
import time
import math
import numpy as np

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _auto_convolution(a: np.ndarray, L: int, conv_len: int):
    """Return the linear auto‑convolution a*a (length ``conv_len``) using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[:conv_len], af

def _gradient_from_weights(a_fft, a_len, w: np.ndarray, L: int):
    """
    Compute the sub‑gradient of the weighted max‑convolution.
    ``w`` is a weight vector of length ``2*a_len-1``.
    Returns an array of length ``a_len``.
    """
    # FFT of w (real)
    wf = np.fft.rfft(w, n=L)
    # Correlation w ⊛ a = ifft(conj(w_f) * a_f)
    corr_full = np.fft.irfft(np.conj(wf) * a_fft, n=L)
    # Slice the central n entries (lag = 0 … n‑1)
    start = a_len - 1                     # n‑1
    grad = 2.0 * corr_full[start:start + a_len]
    return grad

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights that (hopefully) achieves a low
    ``evaluate_sequence`` score.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    n = 5000
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, init.size)
            x_new = np.linspace(0.0, 1.0, n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        init = np.random.rand(n)

    init_sum = init.sum()
    if init_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = init / init_sum                     # simplex (sum = 1)

    # Pre‑compute FFT length and convolution length
    conv_len = 2 * n - 1
    L = _next_pow_two(3 * n - 2)               # enough for correlation (3n‑2)

    # keep best solution found
    conv, _ = _auto_convolution(a, L, conv_len)
    max_val = conv.max()
    best_score = 2.0 * n * max_val
    best_seq = a.copy()
    print(f"[{time.time() - start_time:6.2f}s] start: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Fast exponentiated‑gradient descent (active‑set) ------------
    # ------------------------------------------------------------------
    eta0 = 0.9
    rel_eps = 0.002        # tolerance for active set (wider than before)
    max_add = 30           # size of active set
    min_floor = 1e-15
    it = 0
    last_improve_it = 0

    while time.time() < deadline:
        it += 1

        conv, a_fft = _auto_convolution(a, L, conv_len)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]

        grad = _gradient_from_weights(a_fft, n, w, L)

        eta = eta0 / math.sqrt(it)

        a_new[i] -= delta
        a_new[j] += delta
        # maintain non‑negativity
        if a_new[i] < 0:
            continue

        # evaluate new candidate
        conv_new, _ = _auto_convolution(a_new, L, conv_len)
        new_score = 2.0 * n * conv_new.max()

        delta_score = new_score - cur_score
        accept = False

            accept = True
        else:

            if T > 0 and np.random.rand() < math.exp(-delta_score / T):
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            # keep best ever seen
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve score={new_score:.6f}")
        # temperature decay
        T *= decay
        # avoid T becoming exactly zero
        if T < 1e-12:
            T = 1e-12

    # ------------------------------------------------------------------
    # 4. Return best sequence ------------------------------------------
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    # ensure explicit Python list of floats (and clip very tiny negatives)
    best_seq = np.maximum(best_seq, 0.0).tolist()
    return [float(x) for x in best_seq]

Propose a candidate step‑function by combining three phases:

1. Fast exponentiated‑gradient descent on the simplex using a sub‑gradient of
   the max‑autoconvolution.  The sub‑gradient is obtained via an FFT‑based
   correlation with a weight vector that marks the currently active (almost
   maximal) convolution indices.

2. Simulated‑annealing fine‑tuning that accepts occasional uphill moves.
   Both phases respect the non‑negativity and simplex constraints and run
   within the given time budget.  The best sequence seen is returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _auto_convolution(a: np.ndarray, L: int, conv_len: int):
    """
    Linear auto‑convolution a*a using FFT length ``L``.
    Returns (conv, a_fft) where ``conv`` has length ``conv_len``.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return conv_full[:conv_len], a_fft


def _gradient_from_weights(a_fft, n: int, w: np.ndarray, L: int):
    """
    Compute the sub‑gradient  g = 2·(w ⊛ a)  (correlation of w with a).
    ``w`` must have length 2·n‑1.
    """
    w_fft = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(np.conj(w_fft) * a_fft, n=L)
    # central slice corresponding to lags 0 … n‑1
    start = n - 1
    return 2.0 * corr_full[start:start + n]
