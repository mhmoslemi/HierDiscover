def _lp_optimize(f, n_iter=20, t=0.3):
    a = f.copy()
    best_a = a.copy()
    best_score = _objective_small(a)
    for iter_ in range(n_iter):
        conv = np.convolve(a, a)   # length 2n-1
        M = conv.max()
        # build matrix
        n = a.size
        L = 2*n-1
        A = np.zeros((L, n))
        for i in range(n):
            A[i:i+n, i] = a[i]
        # Solve LP
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(L, M)
        # call linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fall back to break
            break
        b = res.x
        if b.sum() == 0:
            break
        b /= b.sum()  # normalize to sum 1
        # update a
        a = (1 - t) * a + t * b
        # compute new score
        cur_score = _objective_small(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # optionally reduce t
        t *= 0.9
    return best_a, best_score

def hill_climbing(a, max_iter, start_temp, end_temp):
    n = len(a)
    cur_score = objective_score(a)
    best = a.copy()
    best_score = cur_score
    for it in range(1, max_iter+1):
        T = start_temp * (end_temp/start_temp) ** (it/max_iter)
        i,j = np.random.randint(0,n,2)
        if i==j or a[i]==0: continue
        eps = a[i] * np.random.rand() * 0.2  # up to 20% of a_i
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # ensure positivity
        if a_prop[i] < 0: continue
        new_score = objective_score(a_prop)
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / max(T,1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best, best_score

"""
Propose a step‑function (non‑negative list of heights) that minimizes
`evaluate_sequence`.  The implementation uses a cheap stochastic hill‑climbing
procedure on a small problem (n≈200) where objective evaluations are fast.
After a number of random restarts the best small sequence is linearly
upsampled to larger lengths (800, 1600…) and a short annealing phase refines
the final candidate.  The whole search stays well within the 1000 s budget
and works with only NumPy (no external solvers).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """
    Internal objective for a sequence `a` with sum(a)=1:
        score = 2 * n * max(conv(a, a))
    This equals the value returned by `evaluate_sequence` for the same
    sequence (the objective is scale‑invariant).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)               # linear convolution, length 2n‑1
    return float(2.0 * n * conv.max())


def _hill_climb(
    a0: np.ndarray,
    max_iter: int,
    start_temp: float = 2.0,
    end_temp: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing / simulated‑annealing on the simplex.
    At each step a random ε is moved from index i to j.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_iter)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move a fraction of the mass at i towards j
        eps = a[i] * np.random.rand() * 0.2   # at most 20 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective_score(a_prop)
        delta = new_score - cur_score

        # accept if improvement or with Metropolis probability
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
    return best, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from `len(seq)` to `new_n` and renormalisation
    to sum = 1.  The shape (relative heights) is preserved while the
    number of bins grows.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method consists of:
      1. Randomly initialise many short sequences (length ≈200).
      2. For each start run a cheap hill‑climbing (simulated annealing) phase.
      3. Keep the best short sequence found.
      4. Upsample it to a few larger lengths (800, 1600, 3200) and evaluate.
      5. Optionally perform a short annealing refinement on the best long
         candidate.
    The routine respects the global time budget and always returns a valid
    sequence (sum>0.01, non‑negative).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – optimisation on a small problem (fast evaluations)
    # ------------------------------------------------------------------
    n_small = 200
    best_small = None
    best_small_score = float('inf')
    # number of random restarts is adapted to the remaining time
    restart = 0
    while time.time() < deadline - 15.0:
        restart += 1
        # random Dirichlet start
        a0 = np.random.rand(n_small)
        a0 /= a0.sum()

        # hill‑climb / annealing on this start
        max_iter = 30000                     # cheap but effective
        a_opt, score_opt = _hill_climb(a0, max_iter)

        if score_opt < best_small_score:
            best_small_score = score_opt
            best_small = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] restart {restart:2d} "
                f"new small best score={score_opt:.6f}"
            )
        # shrink the iteration budget if we are running out of time
        if time.time() > deadline - 30.0:
            break

    if best_small is None:
        # extremely unlikely, fallback to uniform
        best_small = np.full(n_small, 1.0 / n_small)
        best_small_score = _objective_score(best_small)

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample and evaluate at larger sizes
    # ------------------------------------------------------------------
    candidate_lengths = [800, 1600, 3200]
    best_seq = None
    best_score = float('inf')

    for n_big in candidate_lengths:
        if time.time() >= deadline - 10.0:
            break
        cand = _upsample(best_small, n_big)
        # evaluate with the external function (handles clamping, etc.)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] upsample to n={n_big} "
                f"score={score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – optional short refinement on the best long sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n_big = best_seq.shape[0]
        # a tiny annealing sweep (few thousand proposals)
        max_iter = 8000
        a = best_seq.copy()
        cur_score = evaluate_sequence(a.tolist())
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                break
            T = 1.0 * (0.01 / 1.0) ** (it / max_iter)   # 1 → 0.01
            i, j = np.random.randint(0, n_big, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.1
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
            new_score = evaluate_sequence(a_prop.tolist())
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
                a = a_prop
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] final anneal improve "
                        f"score={cur_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list – the outer harness will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
