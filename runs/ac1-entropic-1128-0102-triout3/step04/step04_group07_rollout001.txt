"""
Algorithm:
1. Random search over large uniform-support step functions.
   Supports sizes from n-10 down to ~1200. For each size we try many random subsets.
2. If not yet reaching target, apply a simple swap-based local search that keeps support size fixed.
   Randomly swaps an index inside the support with an index outside, accepting improvements.
3. Finally run a projected sub‑gradient descent starting from the best vector found.
The search stops early when the target bound 1.503 is achieved.
"""

import time, math, numpy as np

def fast_score(a: np.ndarray, conv_size: int = None) -> float:
    """Fast evaluation of the score without safety checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    if conv_size is None:
        conv_size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, conv_size)
    conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements algorithm from https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent to minimise the score."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # compute convolution and identify the active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size diminishing
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = simplex_projection(v)

        # occasional random mass transfer to escape plateaus
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # periodic logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

"""
Step‑function optimisation for the bound 2 n·max(conv)/(∑a)².

Algorithm
---------
1. **Random large‑support search** – pick many random subsets S ⊂ {0,…,n‑1}
   with size m close to n (e.g. 1490, 1470, …).  The vector is the uniform
   weight a_i = 1/m for i∈S, 0 otherwise.  Because the expected maximum
   autocorrelation of such a random support is ≈1/(2n), the score is close
   to 1 and usually well below the target 1.503.  The search stops early
   if the target is reached.

2. **Swap‑based local optimisation** – keep the best support found so far,
   repeatedly swap one index inside the support with one outside, accepting
   only improvements.  The support size stays constant, the weights stay
   uniform, and the score is re‑evaluated with a fast FFT routine.

3. **Fallback projected sub‑gradient descent** – if the target has not yet
   been met, run a few seconds of the sub‑gradient optimiser (the same
   routine that was used in the reference solution) starting from the
   best vector obtained so far.

The whole procedure respects the 1000 s time budget and returns a list of
non‑negative floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

def fast_score(a: np.ndarray, conv_size: int = None) -> float:
    """Fast evaluation of the objective (no safety checks).

    Parameters
    ----------
    a : np.ndarray
        Non‑negative coefficient vector.
    conv_size : int, optional
        Length of the zero‑padded FFT; if omitted the next power of two
        ≥ 2·len(a)‑1 is used.

    Returns
    -------
    float
        The value 2·n·max(conv)/(∑a)².
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    if conv_size is None:
        conv_size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, conv_size)
    conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()

    idx = np.arange(n)

    while time.time() < deadline:
        # convolution and active index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = simplex_projection(v)

        # occasional random mass transfer
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # periodic logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} attempt={attempts} score={val:.6f}")
            if val <= target_score:
                print("[success] target reached in random‑support phase")
                return best_seq.tolist()
        print(f"[search] size {m} finished after {attempts} attempts, "
              f"best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Local swap optimisation (fixed support size, uniform weights).
    if best_seq is not None:
        support_set = set(np.where(best_seq > 1e-12)[0].tolist())
    else:
        support_set = set(np.arange(n))
    m = len(support_set)
    if m == 0:                               # safety – should never happen
        support_set = set(np.arange(n))
        m = n
    mask = np.zeros(n, dtype=np.float64)
    mask[list(support_set)] = 1.0

    best_vec = mask / m
    if best_seq is None:
        best_seq = best_vec.copy()
        best_val = fast_score(best_seq, conv_size)

    print(f"[swap] starting local swap optimisation (support size {m})")
    swap_iter = 0
    while time.time() < deadline:
        swap_iter += 1
        # element to remove
        out_idx = np.random.choice(list(support_set))
        # element to insert (must be outside the current support)
        in_idx = np.random.randint(0, n)
        while in_idx in support_set:
            in_idx = np.random.randint(0, n)

        # perform swap in the mask
        mask[out_idx] = 0.0
        mask[in_idx] = 1.0
        a_candidate = mask / m
        cand_val = fast_score(a_candidate, conv_size)

        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_candidate.copy()
            support_set.remove(out_idx)
            support_set.add(in_idx)
            # mask already reflects the new support
            if best_val <= target_score:
                print("[success] target reached in swap phase")
                return best_seq.tolist()
        else:
            # revert the change
            mask[out_idx] = 1.0
            mask[in_idx] = 0.0

        if swap_iter % 5000 == 0:
            print(f"[swap iter {swap_iter}] current best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback – projected sub‑gradient optimisation.
    remaining = deadline - time.time()
    if remaining > 2.0:
        print("[fallback] running sub‑gradient optimiser")
        init = best_seq.copy()
        # re‑normalise to avoid tiny numerical drift
        init = init / init.sum()
        refined = subgradient_optimize(init, remaining)
        refined_val = fast_score(refined, conv_size)
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined.copy()
            print(f"[fallback] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final safety net.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq, conv_size)

    print(f"[final] returning sequence with score {best_val:.6f}")
    return best_seq.tolist()
