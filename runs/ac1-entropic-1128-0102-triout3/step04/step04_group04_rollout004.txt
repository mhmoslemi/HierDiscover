"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm overview
------------------
1. **Initial construction** – start from the previously saved
   `height_sequence_1` if it exists, otherwise generate a family of
   power‑law vectors (a_i ∝ (i+1)^‑p) for a dense grid of exponents,
   keep the best one (according to the normalized objective
   `2·n·max(conv(a,a))`).

2. **Permutation sweep** – for the best exponent found, try a few
   random permutations of the values (the values are kept, only their
   order changes).  The best ordering is retained.

3. **Gradient‑directed mass‑transfer** – similar to the previous
   solution, but with a larger active set and a flexible step size.
   Each iteration moves a fraction of mass from the index with the
   largest sub‑gradient to the index with the smallest one; if the
   greedy move fails a mirror‑descent (exponentiated‑gradient) step is
   taken.  The loop runs until the time budget is almost exhausted.

4. **Permutation annealing** – starting from the vector obtained in
   step 3, perform a simulated‑annealing on the *ordering* of the
   entries: repeatedly swap two positions and accept the swap either
   if it improves the score or with a temperature‑dependent
   probability.  This often reduces the peak of the auto‑convolution
   substantially because high values become spread over the whole
   domain.

5. **Final cheap hill‑climbing** – a last phase of small random
   mass‑transfer moves (up to 4 % of the source mass) is performed to
   polish the solution.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) using Euclidean projection.  The routine respects a 1000 s
wall‑clock budget and prints intermediate scores for monitoring.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector already normalised (sum = 1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 80):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` for the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    (non‑negative entries summing to 1).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Permutation annealing (outside of propose_candidate – no closure)
# ----------------------------------------------------------------------
def _permute_annealing(seq: np.ndarray,
                       best_seq: np.ndarray,
                       best_score: float,
                       start_time: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the ordering of `seq`.
    Swaps two positions, keeps the vector normalised (sum unchanged).
    Returns the best sequence and its score found during the annealing.
    """
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = best_score
    # initial temperature – a fraction of the current score
    T0 = best_score * 0.12
    anneal_start = time.time()
    while time.time() < deadline:
        # simple linear cooling
        elapsed = time.time() - anneal_start
        total = deadline - anneal_start
        T = max(T0 * (1.0 - elapsed / total), 1e-12)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        cand = cur_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]

        conv, max_val = _conv_and_max(cand)
        cand_score = 2.0 * n * max_val

        # accept if better, otherwise with Metropolis probability
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
            cur_seq = cand
            cur_score = cand_score

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] PermAnneal new best = {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin – we stop a few seconds before the formal limit
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose problem size and obtain an initial feasible vector.
    # ------------------------------------------------------------------
    n = 5000                               # problem dimension used in the previous run
    best_seq = None
    best_score = float("inf")

    # try to re‑use a previously discovered construction
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be undefined – caught by except
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0.0:
            init = init / s
            best_seq = init.copy()
            best_score = _score_of_seq(best_seq)
            print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
                  f"(n={n}) → score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Power‑law grid search (executed if we have no good init).
    # ------------------------------------------------------------------
    if best_seq is None:
        # generate a dense grid of exponents; the optimum lies around p≈0.4‑0.6
        p_vals = np.linspace(0.10, 0.90, 33)   # 33 candidates
        for p in p_vals:
            if time.time() > deadline:
                break
            i = np.arange(n, dtype=float) + 1.0
            a = i ** (-p)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum
            sc = _score_of_seq(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Power p={p: .3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Permutation sweep on the best power‑law candidate.
    #    (A few random reorderings – cheap but often beneficial.)
    # ------------------------------------------------------------------
    perm_trials = 3000
    for _ in range(perm_trials):
        if time.time() > deadline:
            break
        perm = np.random.permutation(n)
        a_perm = best_seq[perm]
        sc = _score_of_seq(a_perm)
        if sc < best_score:
            best_score = sc
            best_seq = a_perm.copy()
            print(f"[{time.time() - start_time:6.2f}s] Permutation improvement → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Gradient‑directed mass‑transfer refinement.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # record any improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:6d} → new best = {cur_score:.6f}")

        # --------------------------------------------------------------
        # (a) active‑set sub‑gradient
        # --------------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.0012, max_add=80)
        g = _gradient_active(current, K)

        # --------------------------------------------------------------
        # (b) greedy mass transfer from high‑gradient to low‑gradient
        # --------------------------------------------------------------
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max != i_min and current[i_max] > 0:
            # early iterations use a larger step, later we shrink it
            step_factor = 0.12 if it < 20000 else 0.05
            delta = min(step_factor * current[i_max], current[i_max])
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score - 1e-12:
                current = cand
                continue   # greedy step succeeded – restart loop

        # --------------------------------------------------------------
        # (c) fallback: mirror‑descent (exponentiated‑gradient) step
        # --------------------------------------------------------------
        eta = 0.6 / math.sqrt(it)          # decaying step size
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        # normalise in log‑space for numerical stability
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        current = a_new
        # loop continues

    # ------------------------------------------------------------------
    # 5. Permutation annealing (temperature‑driven re‑ordering).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _permute_annealing(best_seq,
                                                  best_seq,
                                                  best_score,
                                                  start_time,
                                                  deadline)

    # ------------------------------------------------------------------
    # 6. Final cheap random hill‑climbing (small mass swaps).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 18000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or current[i] == 0.0:
                continue
            eps = np.random.rand() * 0.04          # up to 4 % of source mass
            delta = eps * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:          # safety – should never happen
                continue
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb → score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best vector found (plain Python floats).
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(100, 1.0 / 100)

    # The evaluate_sequence routine will internally rescale the vector,
    # therefore we keep the simplex normalisation (sum = 1).
    return [float(x) for x in best_seq]
