"""
Propose a low‑maximum self‑convolution step‑function.

This optimizer improves upon the previous solution by:
  1. Using a longer schedule that gives the largest problem sizes
     (n = 16384) a larger share of the time budget.
  2. Adding a *sub‑gradient* refinement stage that works directly on the
     non‑smooth max‑convolution objective (instead of a smooth log‑sum‑exp
     surrogate).  The sub‑gradient is computed analytically from the active
     convolution entries.
  3. Employing a simple *exponential‑move* annealing that preferentially
     removes mass from large‑gradient positions and adds it to low‑gradient
     positions.
  4. Performing occasional random perturbations of the current best vector
     to escape shallow plateaus.
  5. Keeping the original smoothed gradient descent as a fast “coarse”
     stage, then polishing with the sub‑gradient and annealing.

The algorithm works on the probability simplex (∑a_i = 1) because the
evaluation function is scale‑invariant.  All updates are projected back onto
the simplex with an Euclidean projection.  The code respects a 1000 s
budget and prints progress information.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad : dense gradient (shape n)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max over convolution entries – numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of the weights with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _subgradient_from_conv_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a.
    Returns 2 * (mask * a_rev) indexed to length n.
    """
    n = a.shape[0]
    max_val = conv.max()
    mask = (conv >= max_val - 1e-12).astype(float)   # binary mask of active entries
    a_rev = a[::-1]
    full = _conv_fft(mask, a_rev)                   # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp gradient descent (coarse stage)
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with momentum on the simplex.
    Returns the best vector found and its exact objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')

    beta = 0.9
    step0 = 0.8

    # λ‑schedule parameters
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement (fine stage)
# ----------------------------------------------------------------------


def _run_subgrad_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={score:.6f}")

        grad = _subgradient_from_conv_max(a, conv)

        # step size schedule (decreasing)
        step = 0.5 / math.sqrt(it)

        # move opposite the sub‑gradient (to reduce the max)
        a = a - step * grad
        a = _project_simplex(a)

        # occasional random perturbation to escape plateaus
        if it % 1000 == 0:
            eps = 0.01 * np.random.randn(n)
            a = _project_simplex(a + eps)

    return best_a, best_score

# Targeted annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing that moves a small ε from a source index to a destination.
    The source is chosen proportionally to its current mass (more influential
    entries are altered more frequently).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source with probability ~a_i (avoid zero)
        if a.sum() == 0:
            break
        src = np.searchsorted(np.cumsum(a), np.random.rand())
        dest = np.random.randint(0, n)
        if src == dest or a[src] == 0.0:
            continue

        # move a fraction (up to 20 %) of source mass
        eps = a[src] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dest] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
