"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from the best known construction (global
   variable ``height_sequence_1``) and upscale it by linear interpolation
   to a larger length (≈ 10 000).  When no previous construction exists a
   power‑law vector is used.

2. **Gradient‑directed mass transfer with line search** – at each iteration
   we locate the index with the largest sub‑gradient (most responsible for
   the current peak of the autoconvolution) and the index with the smallest
   sub‑gradient.  A move transfers a variable fraction of the source mass
   to the target; several fractions are tried (binary‑search‑like) and the
   best improvement is kept.  This aggressive move replaces the tiny
   5 % steps of the baseline code.

3. **Simulated‑annealing random swaps** – if the deterministic move does
   not improve the score we fall back to a random mass‑swap that is accepted
   according to a slowly cooling temperature.  This helps escape local
   minima.

4. **Final cheap hill‑climbing** – after the time budget is almost spent we
   continue with inexpensive random swaps, keeping any improvement.

All updates keep the vector non‑negative and re‑normalise it to sum = 1,
so the returned sequence can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper functions (all top level – no closures)
# --------------------------------------------------------------

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a non‑negative vector a using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score_norm(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(a,a)) for a vector that already sums to 1.
    (Exactly the numerator of ``evaluate_sequence`` after normalisation.)
    """
    n = a.shape[0]
    conv = _conv_full(a)
    return float(2.0 * n * conv.max())


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30) -> np.ndarray:
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than ``max_add`` are found, the globally largest entries are
    added to reach ``max_add`` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _subgradient(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv)`` for the active set ``K``.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _line_search_move(a: np.ndarray, i_src: int, i_dst: int,
                      max_delta: float, steps: int = 6) -> tuple[np.ndarray, float]:
    """
    Transfer a variable fraction of mass from ``i_src`` to ``i_dst``.
    Several fractions (evenly spaced) are evaluated; the best (lowest)
    objective is returned together with its score.
    """
    best_seq = a
    best_score = _score_norm(a)

    # fractions 0 < f ≤ 1
    fractions = np.linspace(0.0, 1.0, steps + 1)[1:]   # skip 0
    for f in fractions:
        delta = f * max_delta
        cand = a.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        if cand[i_src] < -1e-12:                 # numerical safety
            continue
        cand = np.maximum(cand, 0.0)
        # re‑normalise to keep the sum exactly 1 (tiny rounding errors)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        sc = _score_norm(cand)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = cand
    return best_seq, best_score


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0                # keep a safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size and initialise a feasible vector.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # may be undefined
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            raise ValueError
    except Exception:
        base = None

    if base is not None:
        n0 = base.shape[0]
        # target length – a bit larger to tighten the theoretical bound
        target_n = min(15000, max(8000, n0 * 2))
        # linear interpolation to the new size
        x_old = np.linspace(0.0, 1.0, n0)
        x_new = np.linspace(0.0, 1.0, target_n)
        seq = np.interp(x_new, x_old, base)
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        n = target_n
        print(f"[{time.time() - start:6.2f}s] Init from height_sequence_1 "
              f"→ length {n}")
    else:
        # power‑law fallback
        n = 8000
        p = 0.40
        i = np.arange(1, n + 1, dtype=float)
        seq = i ** (-p)
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        print(f"[{time.time() - start:6.2f}s] No previous construction – "
              f"using power‑law (n={n})")

    # current sequence (will be updated) and the best ever seen
    current_seq = seq.copy()
    best_seq = current_seq.copy()
    best_score = _score_norm(best_seq)
    print(f"[{time.time() - start:6.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Main optimisation loop – gradient move + simulated annealing.
    # --------------------------------------------------------------
    temperature = 1.0
    decay = 0.99996
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_full(current_seq)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # record a new global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current_seq.copy()
            print(f"[{time.time() - start:6.2f}s] Iter {it:5d} "
                  f"→ new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # (a) active‑set sub‑gradient
        # ----------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _subgradient(current_seq, K)

        # ----------------------------------------------------------
        # (b) deterministic large move (source → sink) with line search
        # ----------------------------------------------------------
        # indices with highest / lowest gradient
        idx_desc = np.argsort(-g)          # descending
        idx_asc = np.argsort(g)            # ascending
        moved = False

        # try a few source/target pairs
        for src in idx_desc[:3]:
            if current_seq[src] <= 0.0:
                continue
            max_delta = current_seq[src]
            for dst in idx_asc[:3]:
                if src == dst:
                    continue
                cand_seq, cand_score = _line_search_move(
                    current_seq, src, dst, max_delta, steps=6)
                if cand_score < cur_score - 1e-12:
                    current_seq = cand_seq
                    moved = True
                    break
            if moved:
                break

        if moved:
            continue

        # ----------------------------------------------------------
        # (c) fallback: random swap with simulated‑annealing acceptance
        # ----------------------------------------------------------
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or current_seq[i] == 0.0:
            continue
        # move up to 20 % of the source mass
        frac = np.random.rand() * 0.20
        delta = frac * current_seq[i]
        cand = current_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_score = _score_norm(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.exp(-delta_score / max(temperature, 1e-12)) > np.random.rand():
            current_seq = cand
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start:6.2f}s] Random anneal → "
                      f"score = {best_score:.6f}")

        temperature *= decay

    # --------------------------------------------------------------
    # 3. Final cheap random hill‑climbing (mass swaps).
    # --------------------------------------------------------------
    while time.time() < deadline:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or current_seq[i] == 0.0:
            continue
        # tiny move – up to 4 % of the source
        eps = np.random.rand() * 0.04
        delta = eps * current_seq[i]
        cand = current_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_score = _score_norm(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            current_seq = cand
            print(f"[{time.time() - start:6.2f}s] Final hill‑climb → "
                  f"score = {best_score:.6f}")

    print(f"[{time.time() - start:6.2f}s] Finished – best score {best_score:.6f}")
    return [float(x) for x in best_seq]
