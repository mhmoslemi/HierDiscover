the infinity‑norm of the auto‑convolution.  The objective
    f(a) = log‑sum‑exp(α·(a*a))

We minimise f using Adam on the simplex (non‑negative entries summing to 1)
and finish with a cheap stochastic hill‑climb.  The method is completely
self‑contained, uses only NumPy and runs for at most the allotted time.
"""

def _conv_full(a: np.ndarray):
    """Full linear auto‑convolution of ``a`` and its maximum entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())            # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _gradient_from_weight(a: np.ndarray, w_rev: np.ndarray):
    """
    Given weights ``w_rev`` (the reverse of the soft‑max weights on the
    convolution), return the gradient of  Σ_k w_k·(a*a)_k  w.r.t. ``a``.
    This is 2·(a ⊛ w_rev) restricted to the original length.
    """
    n = a.shape[0]
    m = w_rev.shape[0]               # = 2·n‑1
    fft_len = 1 << ((n + m - 1).bit_length())   # enough for full conv

    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(A * W, n=fft_len)[: n + m - 1]   # length = 3·n‑2

    # The needed slice is indices n‑1 … 2·n‑2 (inclusive) reversed.
    start = n - 1
    end = start + n
    grad_slice = full[start:end][::-1]   # length n
    return 2.0 * grad_slice


def _project_simplex(v: np.ndarray):
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # guard against numerical drift
    w /= w.sum()
    return w

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max (α grows from 5 to 80)
    alpha_start = 5.0
    alpha_end = 80.0
    max_iters = 300_000                           # hard ceiling

    best_seq = a.copy()
    # current true score (sum a = 1 ⇒ score = 2·n·max_conv)
    conv, max_c = _conv_full(a)
    best_score = 2.0 * target_n * max_c
    print(f"[{time.time() - start_time:6.2f}s] start – n={target_n}, score={best_score:.6f}")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----------------------------------------------------------
        # temperature (soft‑max sharpness)
        # ----------------------------------------------------------
        # linear ramp, slows down after half the budget
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / (max_iters // 2))

        # ----------------------------------------------------------
        # current convolution & true objective
        # ----------------------------------------------------------
        conv, max_c = _conv_full(a)
        cur_score = 2.0 * target_n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d}  score={cur_score:.6f}")

        # ----------------------------------------------------------
        # compute soft‑max weights on the convolution entries
        # ----------------------------------------------------------
        max_val = max_c
        # subtract the maximum to avoid overflow
        exp_terms = np.exp(alpha * (conv - max_val))
        sum_exp = exp_terms.sum()
        w = (alpha * exp_terms) / sum_exp            # Σ w = α

        # ----------------------------------------------------------
        # gradient of the smooth surrogate
        # ----------------------------------------------------------
        w_rev = w[::-1]                               # needed for the correlation
        grad = _gradient_from_weight(a, w_rev)        # shape (n,)

        # ----------------------------------------------------------
        # Adam update on the simplex (multiplicative step is not needed)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----------------------------------------------------------
        # project back onto the probability simplex
        # ----------------------------------------------------------
        a = _project_simplex(a)

        # ----------------------------------------------------------
        # occasional random perturbation to escape shallow plateaus
        # ----------------------------------------------------------
        if it % 15000 == 0:
            i, j = np.random.randint(0, target_n, size=2)
            transfer = 0.015 * a[i]                # move up to 1.5 % of the mass
            a[i] -= transfer
            a[j] += transfer
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3. final cheap stochastic hill‑climbing (uses any leftover time)
    # --------------------------------------------------------------
    a = best_seq.copy()
    conv, max_c = _conv_full(a)
    cur_score = 2.0 * target_n * max_c
    while time.time() < deadline:
        i, j = np.random.randint(0, target_n, size=2)
        if i == j:
            continue
        delta = 0.04 * a[i]               # up‑to 4 % of source mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_simplex(a_new)

        conv_new, max_new = _conv_full(a_new)
        new_score = 2.0 * target_n * max_new
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve score={new_score:.6f}")

    # --------------------------------------------------------------
    # 4. return best sequence found
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
