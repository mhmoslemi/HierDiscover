Thus we want to minimize (2 n * max(conv(a,a))) / (sum(a)^2). Since this is homogeneous: scaling a uniformly multiplies numerator by scaling^2 (since conv max scales by scaling^2) and denominator sum(a)^2 scales by scaling^2, so the scaling cancels. def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """Maximum entry of the linear convolution a*a (a is assumed to sum to 1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())

                           fft_len_grad: int) -> np.ndarray:

    Gradient of the smooth‑max objective w.r.t. `a`.
    Implements g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT for speed.

Optimisation of non‑negative step‑function heights to minimise

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm explores several lengths, builds a diverse pool of
initial candidates (uniform, triangular, Hann, exponential, power‑law
and random Dirichlet), refines each with a projected Adam optimiser
that uses a smooth soft‑max approximation of the max‑norm, then runs
a final long refinement on the best sequence found.  The optimiser
works on the probability simplex (sum = 1) and projects after every
update.  A tiny random‑walk hill‑climb is performed at the very end
to squeeze a bit more performance.
"""

# ----------------------------------------------------------------------
# Helper functions – all are top‑level, no closures.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (a is assumed to sum to 1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. `a`.
    Implements g_i = 2·Σ_k w_k·a_{k‑i} using FFT for speed.
    """

    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]
