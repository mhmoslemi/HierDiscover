"""
Simulated‑annealing + targeted‑move optimiser for the step‑function problem.

The algorithm:
1. Initialise a length‑5000 candidate.  If a previous construction
   (global ``height_sequence_1``) exists it is rescaled to have the
   required ℓ₁‑norm √(2·n)=100, otherwise we start from the uniform
   distribution.
2. Repeatedly propose a small mass transfer between two coordinates.
   The proposal is accepted if it improves the objective, or with a
   Metropolis probability ``exp(-(Δscore)/T)`` where the temperature ``T``
   is slowly cooled.
3. Every few hundred iterations a *targeted* move is performed:
   the index pair that contributes most to the current maximal
   convolution entry is identified and a fraction of mass is moved
   from that index towards the far ends of the vector – this directly
   attacks the current bottleneck.
4. The best sequence seen so far is kept and printed occasionally.
   The loop stops just before the time budget expires and the best
   sequence is returned as a plain Python list of floats.

All helper functions are defined at the top‑level; no closures or lambdas
are used.  The routine respects the 1000‑second limit and works with
only NumPy (no external solvers are required).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """
    Full auto‑convolution of a non‑negative vector `a` using FFT and
    return both the convolution array and its maximum entry.
    """
    n = a.shape[0]
    # Next power of two for an efficient FFT (length >= 2n‑1)
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Find the pair (i, k‑i) that contributes most to the current maximal
    convolution entry and propose moving a fraction of mass from i to a
    far‑edge index.  Returns the new candidate vector (still with the same
    ℓ₁‑norm) and the corresponding new score.
    """
    n = a.shape[0]
    # index where the convolution attains its maximum
    k_max = int(np.argmax(conv))

    # indices that actually participate in conv[k_max]
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    # contribution a_i * a_{k_max-i}
    products = a[idx] * a[k_max - idx]
    i_max = idx[int(np.argmax(products))]

    # pick a destination far from i_max (first or last 20 % of the vector)
    if np.random.rand() < 0.5:
        dest = np.random.randint(0, int(0.2 * n))
    else:
        dest = np.random.randint(int(0.8 * n), n)

    # move a modest fraction of the source mass
    eps = np.random.rand() * 0.12          # up to 12 % of source
    delta = eps * a[i_max]
    if delta <= 0.0:
        return a, None, None               # no move possible

    a_new = a.copy()
    a_new[i_max] -= delta
    a_new[dest] += delta

    # renormalise to keep the total sum exactly 100 (sqrt(2·n))
    total = a_new.sum()
    if total > 0:
        a_new *= (100.0 / total)
    else:
        # should never happen – fall back to the original vector
        a_new = a.copy()
    conv_new, max_new = _conv_and_max(a_new)
    new_score = 2.0 * n * max_new
    return a_new, new_score, conv_new
