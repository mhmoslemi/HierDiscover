def interleave_reorder(a):
    n = len(a)
    order = np.empty(n, dtype=int)
    left = 0; right = n-1; pos=0
    while left <= right:
        order[pos] = left
        pos+=1
        if left != right:
            order[pos] = right
            pos+=1
        left+=1
        right-=1
    b = np.empty_like(a)
    b[order] = np.sort(a)[::-1]  # assign largest to early order positions
    return b

def gradient_softmax(a, conv, eps):
    n = a.shape[0]
    m = conv.size  # 2n-1
    max_val = conv.max()
    # compute softmax probabilities (size m)
    p = np.exp((conv - max_val) / eps)
    p_sum = p.sum()
    if p_sum == 0:
        p = np.full_like(p, 1.0 / m)
    else:
        p = p / p_sum
    # now compute gradient
    # convolution of p_rev and a:
    p_rev = p[::-1]
    N = 1 << ((m + n - 1).bit_length())
    Pf = np.fft.rfft(p_rev, n=N)
    Af = np.fft.rfft(a, n=N)
    conv_rev = np.fft.irfft(Pf * Af, n=N)[: (m + n - 1)]
    # segment
    segment = conv_rev[m - n: m][::-1]  # size n
    g = 2.0 * segment
    return g

def _score_normalized(a):
    """Assuming sum(a)=1, return 2*n*max(conv)."""
    n = a.shape[0]
    m = 2*n-1
    N = 1 << ((m-1).bit_length())
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[:m]
    return float(2.0 * n * conv.max()), conv

def _conv_and_max(a):
    n = a.shape[0]
    m = 2*n-1
    N = 1 << ((m-1).bit_length())
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[:m]
    return conv, conv.max()

def _project_simplex(v):
    v = np.asarray(v, dtype=float)
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax_gradient(a, conv, eps):
    """Gradient of softmax-smoothed max(conv) w.r.t a."""
    n = a.shape[0]
    m = conv.shape[0]  # =2*n-1
    max_val = conv.max()
    # softmax probabilities over conv entries
    # subtract max for numerical stability
    p = np.exp((conv - max_val) / eps)
    s = p.sum()
    if s == 0.0:
        p = np.full_like(p, 1.0 / m)
    else:
        p = p / s
    # g_i = 2 * sum_k p_k * a_{k-i}
    # compute via convolution of p_rev and a
    p_rev = p[::-1]
    N = 1 << ((m + n - 1).bit_length())
    Pf = np.fft.rfft(p_rev, n=N)
    Af = np.fft.rfft(a, n=N)
    conv_rev = np.fft.irfft(Pf * Af, n=N)[: (m + n - 1)]
    # required slice: indices m-n .. m-1 (size n), then reversed
    seg = conv_rev[m - n:m][::-1]
    g = 2.0 * seg
    return g

def _active_gradient(a, conv, rel_eps=0.001, max_add=30):
    """Sub‑gradient based on near‑max convolution entries (as earlier)."""
    n = a.shape[0]

    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g

def _interleave_order(n):
    """Return indices that interleave from the ends towards the centre."""
    order = np.empty(n, dtype=int)
    left = 0
    right = n - 1
    pos = 0
    while left <= right:
        order[pos] = left
        pos += 1
        if left != right:
            order[pos] = right
            pos += 1
        left += 1
        right -= 1
    return order

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]

                best_score = sc
                best_seq = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] Parametric init (n={n}) → "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Main optimisation loop – smooth‑max gradient descent + heuristics
    # ------------------------------------------------------------------
    current = best_seq.copy()
    iteration = 0
    eps_start = 0.02
    eps_min = 0.001
    reorder_interval = 5000   # re‑order once in a while

    while time.time() < deadline:
        iteration += 1

        # current convolution and objective
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # record any improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {iteration:6d} "
                  f"best score={best_score:.6f}")

        # stop early if we already beat the target
        if best_score <= 1.503:
            break

        # --------------------------------------------------------------
        # (a) Smooth‑max gradient descent
        # --------------------------------------------------------------
        eps = max(eps_min, eps_start * (0.995 ** iteration))
        g = _softmax_gradient(current, conv, eps)

        step = 0.5 / math.sqrt(iteration)          # decaying step size
        cand = current - step * g
        cand = _project_simplex(cand)

        conv_cand, max_cand = _conv_and_max(cand)
        cand_score = 2.0 * n * max_cand
        if cand_score < cur_score - 1e-12:
            current = cand
            continue

        # --------------------------------------------------------------
        # (b) Greedy active‑set mass transfer (as in the previous code)
        # --------------------------------------------------------------
        g_active = _active_gradient(current, conv, rel_eps=0.001, max_add=30)
        i_max = int(np.argmax(g_active))
        i_min = int(np.argmin(g_active))
        if i_max != i_min and current[i_max] > 0:
            delta = min(0.07 * current[i_max], current[i_max])   # up to 7 %
            cand2 = current.copy()
            cand2[i_max] -= delta
            cand2[i_min] += delta
            conv2, max2 = _conv_and_max(cand2)
            cand2_score = 2.0 * n * max2
            if cand2_score < cur_score - 1e-12:
                current = cand2
                continue

        # --------------------------------------------------------------
        # (c) Exponentiated‑gradient (mirror‑descent) fallback
        # --------------------------------------------------------------
        eta = 0.2 / math.sqrt(iteration)
        cand3 = current * np.exp(-eta * g_active)
        cand3 /= cand3.sum()
        conv3, max3 = _conv_and_max(cand3)
        cand3_score = 2.0 * n * max3
        if cand3_score < cur_score - 1e-12:
            current = cand3
            continue

        # --------------------------------------------------------------
        # (d) Random index swap – cheap hill‑climb
        # --------------------------------------------------------------
        if iteration % 23 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                cand4 = current.copy()
                cand4[i], cand4[j] = cand4[j], cand4[i]
                conv4, max4 = _conv_and_max(cand4)
                cand4_score = 2.0 * n * max4
                if cand4_score < cur_score - 1e-12:
                    current = cand4
                    continue

        # --------------------------------------------------------------
        # (e) Periodic interleaved re‑ordering of the values
        # --------------------------------------------------------------
        if iteration % reorder_interval == 0:
            order = _interleave_order(n)
            sorted_vals = np.sort(current)[::-1]          # descending
            reordered = np.empty_like(current)
            reordered[order] = sorted_vals
            conv_r, max_r = _conv_and_max(reordered)
            r_score = 2.0 * n * max_r
            if r_score < cur_score - 1e-12:
                current = reordered
                continue

        # --------------------------------------------------------------
        # (f) Tiny random perturbation as a last resort
        # --------------------------------------------------------------
        if iteration % 97 == 0:
            noise = np.random.rand(n) * 1e-5
            cand5 = current + noise
            cand5 = np.clip(cand5, 0.0, None)
            cand5 = _project_simplex(cand5)
            conv5, max5 = _conv_and_max(cand5)
            cand5_score = 2.0 * n * max5
            if cand5_score < cur_score - 1e-12:
                current = cand5
                continue

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
