"""
Simple sub‑gradient optimiser for the max‑convolution objective.

The algorithm works on several scales (800 → 1 600 → 3 200 → 6 400 → 9 600).
For each length it runs a projected (Euclidean) sub‑gradient descent on the
objective   f(a) = max_k (a * a)[k]   under the simplex constraint
∑a_i = 1, a_i ≥ 0.  The sub‑gradient at a point is the (averaged) gradient
of the active convolutions attaining the maximum.  After the descent a short
mass‑transfer simulated‑annealing sweep is applied to escape possible local
minima.  The best sequence seen overall is returned.

All heavy lifting (convolution, projection) uses NumPy/FFT and runs well
within the 1000 s budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns the full convolution (length 2·len(a)-1).
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.ifft(Af * Af).real[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of x and y using a real FFT (full length returned).
    """
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.ifft(X * Y).real[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent for max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean sub‑gradient descent on the simplex for
    f(a) = max_k (a * a)[k].
    Returns the best sequence seen and its internal score (2·n·max_b).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    step0 = 0.8                     # initial step size
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # because sum(a)=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, it={it}, score={cur_score:.6f}")

        # identify all active shifts (within machine tolerance)
        active = np.isclose(conv, max_b, atol=1e-12)
        cnt_active = active.sum()
        if cnt_active == 0:
            break

        # weights for the sub‑gradient (uniform over the active set)
        w = active.astype(float) / cnt_active

        # gradient of the weighted sum of convolutions
        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)               # length = 3n‑2
        start = n - 1
        grad = 2.0 * full[start:start + n]       # d/d a_i

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * grad)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method consists of multi‑scale sub‑gradient descent followed by a
    short annealing sweep.  The best sequence obtained across all scales is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter /= starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – a few geometric steps, shuffled to avoid bias.
    lengths = [800, 1600, 3200, 6400, 9600]
    np.random.shuffle(lengths)

    global_a = starter

    # ------------------------------------------------------------------
    # Coarse multi‑scale optimisation
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a0
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # allocate a time slice for this resolution
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(8.0, 0.12 * remaining)

        # sub‑gradient descent on the simplex
        a_opt, score_opt = _run_subgradient_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad: n={n}, score={score_opt:.6f}")

        # short annealing refinement
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # keep the current best as warm‑start for the next scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fine‑tuning phase (optional extra annealing passes)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n_cur = best_seq.shape[0]
        # occasional up‑sampling to allow finer adjustments
        if np.random.rand() < 0.15 and n_cur < 20000:
            new_n = min(20000, int(n_cur * 1.5))
            a0 = _upsample(best_seq, new_n)
            n = new_n
        else:
            # small random perturbation of the current best
            a0 = best_seq.copy()
            eps = 0.01
            noise = (np.random.rand(n_cur) - 0.5) * eps
            a0 = np.maximum(a0 + noise, 0.0)
            a0 /= a0.sum()
            n = n_cur

        remaining = deadline - time.time()
        slice_deadline = time.time() + min(20.0, 0.5 * remaining)

        a_opt, score_opt = _run_subgradient_descent(a0, n, slice_deadline, start_time, max_iter=100_000)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] fine‑tune improve: n={n}, score={score_opt:.6f}")

        if time.time() < deadline:
            anneal_deadline = time.time() + min(8.0, deadline - time.time())
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=1e-2,
                end_temp=1e-7,
                steps=15_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] fine‑tune anneal improve: n={n}, score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return plain Python list – the outer harness will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
