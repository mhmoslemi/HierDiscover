"""
Propose a candidate step‑function using a smoothed mirror‑descent on the simplex.
The algorithm works with a soft‑max approximation of the `max(conv(a,a))`
objective, which yields a dense sub‑gradient and avoids the very small
active‑set used previously.  The update is performed with exponentiated
( multiplicative ) mirror‑descent on the probability simplex; the
temperature (soft‑max sharpness) is increased slowly while the step size is
decreased as 1/√t.  Random mass‑transfer perturbations and an occasional
restart from the current best solution help escape plateaus.  After the
gradient phase a cheap stochastic hill‑climbing finishes the search.

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence (as a list of floats) that it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, L: int, out_len: int) -> np.ndarray:
    """
    Return the full auto‑convolution of a (length n) using FFT.
    The result is truncated to ``out_len`` (=2*n-1).
    """
    Af = np.fft.rfft(a, L)
    conv = np.fft.irfft(Af * Af, L)[:out_len]
    return conv


def _gradient_via_softmax(a: np.ndarray, conv: np.ndarray,
                         beta: float, L: int, len_w: int) -> np.ndarray:
    """
    Compute a dense sub‑gradient of the soft‑max approximation of
    ``max(conv)``.
    ``conv`` is the full auto‑convolution (length = len_w = 2*n‑1).
    The gradient is ``2 * correlation(a, w)`` where
    ``w_i ∝ exp(beta * (conv_i - max(conv)))`` and the correlation is taken
    over the full range.
    """
    # soft‑max weights (already normalised)
    max_val = conv.max()
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safety
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # FFTs needed for the correlation
    Af = np.fft.rfft(a, L)
    Aw = np.fft.rfft(w, L)

    # correlation = irfft(conj(Af) * Aw)
    corr = np.fft.irfft(np.conj(Af) * Aw, L)

    # correlation indices that correspond to g_j = 2 * sum_i a_i w_{i+j}
    start = len_w - 1                     # index of lag 0 in the full correlation
    g = 2.0 * corr[start:start + a.size]
    return g


def _random_mass_transfer(a: np.ndarray, eps_max: float = 0.05) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    eps = np.random.rand() * eps_max
    delta = eps * a[i]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method combines a smoothed mirror‑descent
    phase with a final stochastic hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    n = 5000                                   # length of the sequence
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.clip(init, 0.0, None)
        if init.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, init.size)
            x_new = np.linspace(0.0, 1.0, n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        init = np.random.rand(n)

    # normalise to a probability vector (sum = 1)
    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = init / s

    # FFT bookkeeping
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    len_w = 2 * n - 1                     # length of a full convolution

    # initial score (exact objective = 2·n·max(conv) because sum(a)=1)
    conv = _full_convolution_fft(a, L, len_w)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[{time.time() - start_time:6.2f}s] start: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed mirror‑descent phase --------------------------------
    # ------------------------------------------------------------------
    # hyper‑parameters (tuned empirically)
    eta0 = 0.40                # initial step size for exponentiated update
    beta0 = 0.10               # start temperature for soft‑max
    beta_max = 8.0             # final temperature
    it = 0
    last_improve = 0
    max_iters = 300_000        # generous hard cap; time will stop us earlier

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---------- evaluate current point ----------
        conv = _full_convolution_fft(a, L, len_w)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d}  best={cur_score:.6f}")

        # ---------- soft‑max temperature ----------
        # increase beta linearly over the first ~20 k iterations
        beta = min(beta_max, beta0 + (beta_max - beta0) * (it / 20000.0))

        # ---------- gradient computation ----------
        g = _gradient_via_softmax(a, conv, beta, L, len_w)

        # ---------- exponentiated (mirror‑descent) update ----------
        eta = eta0 / math.sqrt(it)            # 1/√t schedule
        log_a = np.log(a + 1e-30) - eta * g   # stay on the simplex in log‑space
        log_a -= np.max(log_a)                # numerical stabilisation
        a = np.exp(log_a)
        a = np.maximum(a, 1e-15)              # tiny floor
        a /= a.sum()

        # ---------- occasional random perturbation ----------
        if it % 8_000 == 0:
            a = _random_mass_transfer(a, eps_max=0.04)

        # ---------- restart if stuck ----------
        if it - last_improve > 20_000:
            # perturb the current best and continue from there
            a = best_seq.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] restart at iter={it}")

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climbing --------------------------------
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.04
        delta = eps * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue

        # evaluate candidate
        conv_new = _full_convolution_fft(a_new, L, len_w)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            cur_score = new_score
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found --------------------------------
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
