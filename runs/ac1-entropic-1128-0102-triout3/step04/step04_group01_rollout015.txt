        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # exponentiated update
        a = a * np.exp(-step * m_hat)
        # avoid zero
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
    return best_a, best_score

"""
Propose a low‑maximum self‑convolution step‑function.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1).  The objective
  `evaluate_sequence` is scale‑invariant, so the sum can be fixed.
* For a sequence length n we run an **Adam‑style exponentiated‑gradient
  descent** on a smooth log‑sum‑exp surrogate of the ‑∞‑norm of the
  convolution.  The surrogate temperature λ is tied to the current
  maximum convolution value and is decreased during the run.
  The update is multiplicative (a_i ← a_i·exp(‑η·grad_i)) and a
  renormalisation guarantees feasibility.
* The obtained distribution is **polished by a targeted simulated
  annealing** that moves a small amount of mass from indices that
  contribute most to the current maximal convolution entry to random
  destinations.  Acceptance follows a geometric temperature schedule.
* A multi‑scale schedule is used: we start with modest lengths and
  progressively up‑sample the current best distribution to larger
  sizes, always warming‑starting the optimiser.
* Time is split between the stages; the whole routine respects a
  1000 s budget.

The routine returns a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of arbitrary real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # cross‑correlation of the weight vector with reversed a
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]  # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_adam_exp_gradient(a0: np.ndarray,
                           n: int,
                           slice_deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector by exponentiated‑gradient descent
    with Adam‑style adaptive step sizes on the log‑sum‑exp surrogate.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate schedule
    eta0 = 0.6

    # λ‑schedule (tied to current max)
    lam_factor = 0.5
    lam_decay = 0.7
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # λ schedule
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the smoothed surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # element‑wise step (exponentiated update)
        step = eta0 / (np.sqrt(v_hat) + eps)
        # to avoid overflow/underflow bound the exponent
        inc = -step * m_hat
        inc = np.clip(inc, -50, 50)
        a = a * np.exp(inc)

        # renormalise to the simplex (and keep numerics safe)
        a = np.maximum(a, 1e-30)
        a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------


def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-6,
                            steps: int = 20000) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing that moves a small ε from a source index that
    heavily contributes to the current maximal convolution entry to a
    random destination index.
    Returns the (possibly) improved sequence and its objective value.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current convolution and its maximal index
        conv = _conv_self(a)
        max_b = conv.max()
        k = int(conv.argmax())

        # contributions of each i to the peak entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idxs = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idxs] * a[k - idxs]                     # w_i = a_i * a_{k-i}
        total = contrib.sum()
        if total <= 0:
            src = np.random.randint(0, n)
        else:
            prob = contrib / total
            src = np.random.choice(n, p=prob)

        # pick a different destination uniformly
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1

        # propose moving a fraction of the source mass
        eps = a[src] * np.random.rand() * 0.15   # up to 15 % of source
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the proposal
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Runs for at most `budget_s` seconds and
    returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – modest to fairly large.  Random order avoids bias.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 (warm start / random)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Time slice for the current length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # later (larger) lengths get a larger share
        fraction = 0.15 / (len(lengths) - idx)   # similar to the reference code
        slice_deadline = time.time() + max(8.0, remaining * fraction)

        # ------------------------------------------------------------------
        # Adam‑style exponentiated‑gradient descent
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam_exp_gradient(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after gradient n={n} score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # Targeted annealing refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_targeted_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=20000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # keep the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final polishing on the best length (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        final_deadline = time.time() + max(2.0, deadline - time.time())
        best_seq, best_score = _run_targeted_annealing(
            best_seq,
            n,
            final_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=30000,
        )
        print(f"[{time.time() - start_time:6.2f}s] final polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list (the evaluator will clamp / normalise)
    return [float(x) for x in best_seq]
