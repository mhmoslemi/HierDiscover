"""
Candidate generator for the “step‑function” minimisation problem.

The search consists of three phases:

1. **Linear‑programming refinement (small n)** –  
   Starting from a warm start (the previous best construction if
   available) we repeatedly solve a tiny LP:
     maximise Σ b_j   s.t.   f ∗ b ≤ M  (M = max(f∗f),  b ≥ 0)
   where only the K largest convolution entries are kept as constraints.
   The solution b is renormalised to a probability vector g and we move
   f ← (1‑t)·f + t·g with a geometrically decaying step size t.
   This is exactly the LP‑step described in the problem statement and
   quickly drives the maximum self‑convolution down.

2. **Projected (sub‑)gradient descent** –  
   After the LP phase the vector is up‑sampled to medium (≈2400) and
   large (≈4800) lengths.  For each length we run a long projected
   gradient descent on the simplex using the exact sub‑gradient of the
   max‑convolution (all active indices are summed, no averaging).  A
   Nesterov momentum term and a 1/√t step‑size schedule are used.

3. **Simulated‑annealing fine‑tuning** –  
   Finally a light annealing with mass‑transfer moves cleans up any
   remaining local irregularities.

All phases respect the global time budget (default 1000 s) and the
returned sequence is a plain list of non‑negative floats whose sum is
exactly one, so `evaluate_sequence` will accept it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are taken as active.
    The contributions are summed (no averaging).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of a_i a_j term:
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.3,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index to a
    random index.  Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source index sampled proportionally to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# LP‑based improvement for a modest size
# ----------------------------------------------------------------------
def _run_lp_iterations(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       max_iters: int = 200,
                       k_max: int = 300,
                       eps_start: float = 0.05,
                       eps_decay: float = 0.99,
                       step0: float = 0.5,
                       step_decay: float = 0.97) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny linear program:
        maximise Σ b_j
        s.t.   (a * b)[k] ≤ M  for a selected set of offsets k,
               b ≥ 0

    The solution b is normalised to sum = 1, mixed into a with
    a geometrically decaying step size, and the best vector is kept.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("SciPy not available – skipping LP stage.")
        return a0.copy(), 2.0 * n * _conv_self(a0).max()

    a = a0.copy()
    a /= a.sum()                     # enforce Σa = 1
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    eps = eps_start

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, iter={it}, score={cur_score:.6f}")

        # pick active indices (those within (1‑eps) of the peak)
        active = np.where(conv >= M * (1.0 - eps))[0]
        if active.size > k_max:
            # keep only k_max largest entries
            idx = np.argpartition(-conv, k_max)[:k_max]
            active = idx
        eps *= eps_decay

        K = active.size
        if K == 0:
            break

        # Build constraint matrix A (K × n)
        A = np.zeros((K, n), dtype=float)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1, dtype=int)
            i = k - j
            A[row, j] = a[i]

        # Solve LP: maximise 1ᵀb  ⇔ minimise -1ᵀb
        c = -np.ones(n, dtype=float)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(K, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'disp': False})

        if not res.success:
            # if LP fails, relax the active set a little
            eps = min(1.0, eps * 1.5)
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        g = b / sum_b                # renormalise to Σg = 1

        # mixing step (geometrically decaying)
        step = step0 * (step_decay ** (it - 1))
        a = (1.0 - step) * a + step * g
        a /= a.sum()                 # re‑normalise (numeric safety)

    return best_a, best_score

            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad stage improve (n={n_len}): score={score_opt:.6f}")
        return a_opt

    # ------------------------------------------------------------------
    # Stage 1 – medium size (n≈2400)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_mid = 2400
        a_mid = _upsample(best_seq, n_mid)
        remaining = deadline - time.time()
        grad_time_mid = max(15.0, 0.45 * remaining)   # give ~45 % of the rest to this stage
        a_mid_opt = _maybe_improve(a_mid, n_mid, grad_time_mid,
                                   beta=0.90, step0=0.30)
    else:
        a_mid_opt = best_seq

    # ------------------------------------------------------------------
    # Stage 2 – large size (n≈4800) – gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 4800
        a_big = _upsample(a_mid_opt, n_big)
        remaining = deadline - time.time()
        grad_time_big = max(10.0, 0.30 * remaining)   # ~30 % of what remains
        a_big_opt = _maybe_improve(a_big, n_big, grad_time_big,
                                   beta=0.93, step0=0.25)
    else:
        a_big_opt = best_seq

    # ------------------------------------------------------------------
    # Stage 3 – final simulated annealing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            a_big_opt, a_big_opt.shape[0],
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6,
            steps=250_000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
