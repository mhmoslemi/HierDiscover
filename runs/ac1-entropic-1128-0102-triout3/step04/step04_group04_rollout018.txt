def _active_set(conv: np.ndarray, max_val: float,
                rel_eps: float = 0.001, max_add: int = 80) -> np.ndarray:
    """Indices that are (almost) maximal."""
    eps = max_val * rel_eps

    return K


def _gradient_active(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) for active indices K."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

    i = np.arange(n, dtype=float) + 1.0
    a = i ** (-p)
    a /= a.sum()
    return a

    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _make_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = i ** (-p)
    a /= a.sum()
    return a


def _make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """Return a list of non‑negative heights that (hopefully) lowers `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the outer harness will abort after 1000 s
    deadline = start_time + budget_s - 5.0
    total_budget = deadline - start_time

    # --------------------------------------------------------------
    # 0.  Load a previous construction if it exists.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be undefined
        init = np.maximum(init, 0.0)
        if init.sum() > 0.0:
            init /= init.sum()
            best_seq = init.copy()
            best_score = _score_of_seq(best_seq)
            print(f"[{time.time() - start_time:6.2f}s] Loaded height_sequence_1 (n={len(init)}) → score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Parametric scouting over several lengths and families.
    # --------------------------------------------------------------
    n_choices = [3000, 4000, 5000, 6000, 8000]

    for n in n_choices:
        if time.time() > deadline - 200:      # leave ~200 s for later phases
            break
        # draw a few hundred random shapes for this length
        for _ in range(180):
            if time.time() > deadline - 200:
                break
            typ = np.random.choice(["power", "exp", "beta", "gauss"])
            if typ == "power":
                p = np.random.uniform(0.2, 1.3)
                a = _make_power_law(n, p)
            elif typ == "exp":
                lam = np.random.uniform(0.0002, 0.005)
                a = _make_exponential(n, lam)
            elif typ == "beta":
                alpha = np.random.uniform(0.3, 2.0)
                beta = np.random.uniform(0.3, 2.0)
                a = _make_beta(n, alpha, beta)
            else:   # gaussian
                sigma_frac = np.random.uniform(0.04, 0.15)   # sigma as a fraction of n
                sigma = sigma_frac * n
                a = _make_gaussian(n, sigma)

            sc = _score_of_seq(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Scout n={n:4d} {typ:6s} → score={sc:.6f}")

    if best_seq is None:                     # ultimate fallback
        n = 4000
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Fallback uniform (n={n}) → score={best_score:.6f}")

    n = best_seq.shape[0]
    current = best_seq.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # 2. Simulated annealing (mass moves + swaps).
    # --------------------------------------------------------------
    anneal_deadline = start_time + 0.45 * total_budget
    T0 = 0.02
    Tmin = 1e-6
    it_anneal = 0

    while time.time() < anneal_deadline:
        it_anneal += 1
        # linear temperature decay
        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / total_budget), Tmin)

        if np.random.rand() < 0.7:                     # mass‑transfer move
            i = np.random.randint(n)
            if current[i] < 1e-12:
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            frac = np.random.rand() * 0.10            # up to 10 % of source mass
            delta = frac * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
        else:                                            # pure swap (changes ordering only)
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            cand = current.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand_score = _score_of_seq(cand)
        delta_score = cur_score - cand_score          # positive ⇒ improvement

        if delta_score > 0 or np.exp(delta_score / T) > np.random.rand():
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal iter {it_anneal:6d} → new best {cand_score:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent with momentum.
    # --------------------------------------------------------------
    gd_deadline = start_time + 0.85 * total_budget
    momentum = np.zeros_like(best_seq)
    beta = 0.9
    it_gd = 0

    while time.time() < gd_deadline:
        it_gd += 1
        conv = _conv_full(current)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD iter {it_gd:6d} → new best {cur_score:.6f}")

        # active‑set sub‑gradient
        K = _active_set(conv, max_val, rel_eps=0.0005, max_add=80)
        g = _gradient_active(current, K)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # greedy tiny redistribution (optional)
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max != i_min and current[i_max] > 0:
            delta = min(0.03 * current[i_max], current[i_max])
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            cand = _project_simplex(cand)
            cand_score = _score_of_seq(cand)
            if cand_score < cur_score - 1e-12:
                current = cand
                continue   # accepted greedy step

        # mirror‑descent (exponentiated‑gradient) fallback
        eta = 0.6 / math.sqrt(it_gd)
        log_a = np.log(np.maximum(current, 1e-15)) - eta * momentum
        log_a -= np.max(log_a)          # stabilise exponentiation
        a_new = np.exp(log_a)
        a_new = _project_simplex(a_new)
        current = a_new

    # --------------------------------------------------------------
    # 4. Final cheap random hill‑climbing (tiny swaps).
    # --------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.02           # up to 2 % of source mass
        delta = eps * current[i]
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _score_of_seq(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            current = cand
            print(f"[{time.time() - start_time:6.2f}s] Final hill‑climb → score={cand_score:.6f}")

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score {best_score:.6f}")
    return [float(x) for x in best_seq]
