"""
Improved optimisation for the step‑function bound.

The core idea is to minimise a *smooth* approximation of the infinity‑norm
of the self‑convolution.  For a sequence a (∑a=1) we use

    Fα(a) = (1/α)·log(∑exp(α·(a∗a))),

the log‑sum‑exp (soft‑max) of the convolution entries.
Its gradient w.r.t. a is analytic:

    ∂Fα/∂a_i = 2·∑_j w_{i+j}·a_j,
    where w_k = softmax_k(α·(a∗a)).

With this smooth objective we can run a fast first‑order optimiser
(Adam) on the probability simplex, progressively increasing α so that
the surrogate finally approaches the true max‑norm.  The optimisation is
performed on a sequence of increasing lengths (coarse → fine) and a
final cheap simulated‑annealing walk cleans up any remaining tiny
imperfections.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y (length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:n]
    return conv


# ----------------------------------------------------------------------
# Self‑convolution of a (used for the objective)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights for a given convolution vector
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(alpha·conv_k) (stable)."""
    x = alpha * conv
    max_x = x.max()
    # exponentials are safe because conv ≥ 0 and alpha is moderate
    w = np.exp(x - max_x)
    s = w.sum()
    if s == 0:
        # fallback to uniform weights (should never happen)
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate (∂Fα/∂a)
# ----------------------------------------------------------------------
def _grad_softmax_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a Fα(a) where
        Fα(a) = (1/α)·log Σ_k exp(α·(a∗a)_k).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    w = _softmax_weights(conv, alpha)        # same length
    # gradient = 2 * (w * a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)               # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with a growing soft‑max temperature
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alpha_start: float = 1.0,
                   alpha_end: float = 40.0,
                   lr: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑norm.
    Returns the best sequence seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # linearly increase the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / 5000.0)

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t}, α={alpha:.2f}, score={cur_score:.6f}")

        # gradient of the smooth surrogate
        g = _grad_softmax_max(a, alpha)

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

    # final evaluation of the best point
    final_conv = _conv_self(best_a)
    final_score = 2.0 * n * final_conv.max()
    return best_a, final_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfer)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 400000) -> tuple[np.ndarray, float]:
    """Move a small ε from i→j, Metropolis acceptance, keep the best seen."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # shift up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly from its length to `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget, performs progressive
    up‑sampling, Adam optimisation on a soft‑max surrogate of the
    max‑norm, and finishes with a short simulated‑annealing refinement.
    Returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0           # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter: reuse a previously found construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Progressive stages (coarse → fine)
    # ------------------------------------------------------------------
    stage_lengths = [600, 1200, 2400, 4800, 9600]   # increase roughly by factor 2
    anneal_budget = max(30.0, 0.15 * (deadline - start_time))
    grad_budget_total = (deadline - start_time) - anneal_budget
    total_weight = sum(stage_lengths)

    for n in stage_lengths:
        # stop early if we have already reserved the annealing budget
        if time.time() >= deadline - anneal_budget:
            break

        # remaining time for gradient work (excluding final annealing)
        remaining = deadline - time.time() - anneal_budget
        if remaining <= 0:
            break

        # allocate a share proportional to n
        stage_time = max(5.0, remaining * (n / total_weight))
        stage_deadline = time.time() + stage_time

        # initialise a0: up‑sample the best we have, else use a cosine shape
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # deterministic cosine‑like start (always non‑negative)
            idx = np.arange(n)
            a0 = np.sin(np.pi * (idx + 0.5) / n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # Adam optimisation on the current size
        a_opt, score_opt = _adam_optimize(
            a0, n,
            deadline=stage_deadline,
            start_time=start_time,
            alpha_start=1.0,
            alpha_end=40.0,
            lr=0.2,
            beta1=0.9,
            beta2=0.999
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # shrink the remaining gradient budget for the next stage
        elapsed = time.time() - start_time
        grad_budget_total = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – polish the best sequence found.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        best_seq, best_score = _run_annealing(
            best_seq, n_final,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=500000
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may appear because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
