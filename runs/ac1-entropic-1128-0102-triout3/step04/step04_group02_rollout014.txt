"""
A fast multi‑scale optimiser for the max‑convolution objective.
The method builds a warm‑start from the “Fourier‑square‑root” of a
uniform target, then runs many projected Adam steps on the simplex
while gradually cooling a log‑sum‑exp surrogate of the ℓ∞‑norm.
After each scale a short simulated‑annealing sweep is performed,
and the solution is up‑sampled to a larger resolution.
The whole search respects the 1000 s budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y using a real FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation and re‑normalisation to sum‑one."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ·log Σ_k exp(conv_k / λ)
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Warm‑start via Fourier square‑root of a uniform target
# ----------------------------------------------------------------------


def _fft_sqrt_start(n: int) -> np.ndarray:
    """
    Construct an initial probability vector whose self‑convolution
    is close to uniform by taking the (point‑wise) square‑root of the
    DFT of a uniform target.
    """
    m = 2 * n - 1                      # length of the desired convolution
    size = 1 << ((m - 1).bit_length())  # next power of two
    u = np.full(m, 1.0 / m)            # uniform target
    B = np.fft.rfft(u, n=size)         # DFT of the target
    A = np.sqrt(B)                     # point‑wise square‑root
    a_full = np.fft.irfft(A, n=size)[:n]  # take first n samples
    a_full = np.real(a_full)
    a_full = np.maximum(a_full, 0.0)   # enforce non‑negativity
    total = a_full.sum()
    if total > 0.0:
        a_full /= total
    else:
        a_full = np.full(n, 1.0 / n)
    return a_full


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  max_iter: int = None) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on the simplex.
    The loss is the log‑sum‑exp surrogate of the ℓ∞‑norm of a*a.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base step size
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    # λ‑schedule (starts relatively large then cools)
    lam_factor = 0.5
    lam_decay = 0.9995            # multiplicative decay per iteration

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # sum(a) = 1 → evaluate = 2·n·max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # smoothness parameter λ
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the smoothed objective
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

        # cool λ a little each iteration
        lam_factor = max(lam_factor * lam_decay, 1e-8)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns a possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes `evaluate_sequence`.
    The algorithm consists of three resolution stages (256 → 1 024 → 4 096)
    each performing a long Adam optimisation and a short annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # leave a safety margin

    best_score = float('inf')
    best_seq = None

    # Length schedule – a few geometrically increasing stages.
    # The last stage (≈4 k) is a good compromise between quality and speed.
    lengths = [256, 1024, 4096]

    current_a = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a for this resolution
        if current_a is not None:
            a0 = _upsample(current_a, n)
        else:
            # very first warm‑start – the Fourier‑sqrt method
            a0 = _fft_sqrt_start(n)

        # ensure exact normalisation (sum = 1)
        a0 /= a0.sum()

        # Allocate a slice of the remaining budget for this scale.
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # we keep roughly 70 % of the remaining time for the Adam phase,
        # the rest for a short annealing sweep.
        adam_deadline = time.time() + 0.70 * remaining
        a_opt, score_opt = _run_adam_opt(a0, n, adam_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after Adam: n={n}, score={score_opt:.6f}")

        # final quick annealing refinement (if time permits)
        if time.time() < deadline:
            anneal_deadline = deadline
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=40_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: n={n}, score={score_ref:.6f}")

        # warm‑start the next scale
        current_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return as plain Python list – the outer harness will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
