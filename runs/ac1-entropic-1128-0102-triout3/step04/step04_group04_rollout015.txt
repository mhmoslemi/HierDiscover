def conv_autocorr(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    Af = np.fft.rfft(a, L)
    conv = np.fft.irfft(Af * Af, L)[:2*n-1]
    return conv

def gradient_smooth(a, conv, w):
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    w_fft = np.fft.rfft(w, L)
    a_rev_fft = np.fft.rfft(a_rev, L)
    g_full = np.fft.irfft(w_fft * a_rev_fft, L)
    # g_i = 2 * g_full[i + n-1] for i=0..n-1
    g = 2.0 * g_full[(n-1):(n-1 + n)]
    return g

"""
Search for a non‑negative step‑function minimizing `evaluate_sequence`.
Algorithm:
1. Build several simple parametric families (power‑law, two‑sided power‑law)
   and keep the best as an initial vector.
2. Refine the vector with a deterministic‑annealing style gradient descent.
   The objective `2·n·max(conv(a,a))` (with sum(a)=1) is smoothed by a
   log‑sum‑exp weight on the convolution entries; the resulting gradient
   is computed by FFT‑based convolutions.  After each gradient step the
   vector is projected onto the probability simplex.
3. Finish with a cheap random‑swap / small‑mass‑transfer hill‑climb.
The routine respects a ~1000 s wall‑clock budget and prints progress.
"""

import time
import math
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, L)
    conv = np.fft.irfft(Af * Af, L)[: 2 * n - 1]
    return conv

def _score_of_seq(a: np.ndarray) -> float:
    """Objective for a probability vector (sum=1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max‑convolution.
    a has length n, w has length 2·n‑1 and contains exp(alpha*(conv-max)).
    Returns a vector of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    Wf = np.fft.rfft(w, L)
    Af = np.fft.rfft(a_rev, L)
    g_full = np.fft.irfft(Wf * Af, L)
    # slice n‑1 .. n‑1+n‑1
    g = 2.0 * g_full[(n - 1):(n - 1 + n)]
    return g

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a probability vector) that
    tries to minimise `evaluate_sequence`.  The routine spends the
    allocated time budget on a quick parametric initialization,
    deterministic‑annealing gradient refinement and a final cheap
    random hill‑climbing phase.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Determine problem size & possibly a previously found vector.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be undefined
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s
            n = init.shape[0]
            best_seq = init.copy()
            best_score = _score_of_seq(best_seq)
            print(f"[{time.time()-start:6.2f}s] Loaded height_sequence_1 (n={n}) → score={best_score:.6f}")
        else:
            raise ValueError
    except Exception:
        # fall back to a default length
        n = 5000
        best_seq = None
        best_score = float("inf")
        print(f"[{time.time()-start:6.2f}s] No previous vector – using n={n}")

    # ------------------------------------------------------------------
    # 1. Simple parametric families (power‑law, bi‑power‑law, reverse)
    # ------------------------------------------------------------------
    candidates = []

    if best_seq is not None:
        candidates.append(best_seq)

    # pure power‑law
    for p in np.linspace(0.15, 0.85, 15):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a = a / a.sum()
        candidates.append(a)

    # two‑parameter power‑law (left‑right asymmetry)
    p_vals = np.linspace(0.1, 0.9, 9)
    q_vals = np.linspace(0.0, 0.5, 6)
    i = np.arange(1, n + 1, dtype=float)
    j = np.arange(n, 0, -1, dtype=float)
    for p in p_vals:
        for q in q_vals:
            a = (i ** (-p)) * (j ** (-q))
            a = a / a.sum()
            candidates.append(a)

    # reversed versions (right‑heavy)
    for p in np.linspace(0.15, 0.85, 7):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a = a[::-1]                # flip
        a = a / a.sum()
        candidates.append(a)

    # evaluate candidates (stop early if we run out of budget)
    for a in candidates:
        if time.time() > deadline:
            break
        sc = _score_of_seq(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time()-start:6.2f}s] Init candidate → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Deterministic‑annealing gradient descent on the best vector.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    alpha = 0.5               # smoothing parameter (log‑sum‑exp temperature)
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(current)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()

        # weights w_i = exp(alpha*(conv_i - max_val))  (max weight = 1)
        w = np.exp(alpha * (conv - max_val))

        grad = _gradient_smooth(current, w)

        # step size – diminishing
        eta = 0.3 / math.sqrt(it)

        # projected gradient step
        cand = _project_simplex(current - eta * grad)

        cand_score = _score_of_seq(cand)

        # simple accept‑if‑better rule (with occasional uphill moves)
        if cand_score < cur_score:
            current = cand
        else:
            # simulated‑annealing style acceptance
            temp = max(0.01, 1.0 / math.sqrt(it))
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
                current = cand

        # slowly increase the smoothing parameter (anneal)
        if it % 500 == 0:
            alpha = min(alpha * 1.08, 20.0)

        # safety bail‑out
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3. Final cheap random hill‑climbing (swaps + tiny mass moves).
    # ------------------------------------------------------------------
    current = best_seq.copy()
    while time.time() < deadline:
        # ---- random swap of two entries ----
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            cand = current.copy()
            cand[i], cand[j] = cand[j], cand[i]
            sc = _score_of_seq(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time()-start:6.2f}s] Swap improve → {sc:.6f}")

        # ---- tiny mass transfer (up to 5 % of source) ----
        if np.random.rand() < 0.3:
            i = np.random.randint(0, n)
            if current[i] > 0:
                j = np.random.randint(0, n)
                if i != j:
                    eps = np.random.rand() * 0.05
                    delta = eps * current[i]
                    cand = current.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    sc = _score_of_seq(cand)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_seq = cand.copy()
                        current = cand
                        print(f"[{time.time()-start:6.2f}s] Transfer improve → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence as plain Python floats.
    # ------------------------------------------------------------------
    result = [float(x) for x in best_seq]
    print(f"[{time.time()-start:6.2f}s] Finished – best score {best_score:.6f}")
    return result

"""
Search for a non‑negative step‑function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Parametric warm‑start** – generate a handful of simple families
   (power‑law, left/right‑asymmetric power‑law, reversed versions) and keep
   the best one (or the previously saved `height_sequence_1`).

2. **Deterministic‑annealing gradient descent** – minimise the objective
   `2·n·max(conv(a,a))` under the constraint `∑a_i=1`, `a_i≥0`.  The max
   is smoothed by a log‑sum‑exp (soft‑max) with temperature `α`.  The
   resulting gradient can be written as a convolution and is evaluated by
   FFTs.  After each gradient step `a` is projected onto the probability
   simplex.  The soft‑max temperature `α` is slowly increased (annealing).

3. **Random hill‑climbing** – cheap swaps of two entries and tiny mass‑transfer
   moves (up to 5 % of the source) are tried until the wall‑clock budget
   expires; any improvement is kept.

The routine respects a ~1000 s wall‑clock budget, prints progress messages
and finally returns the best sequence found as a plain Python list.

    conv = np.fft.irfft(Af * Af, L)[: 2 * n - 1]

            a = (i ** (-p)) * (j ** (-q))
            a = a / a.sum()
            candidates.append(a)

    # reversed power‑law (right‑heavy)
    for p in np.linspace(0.15, 0.85, 7):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a = a[::-1]
        a = a / a.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2. Evaluate candidates – keep the best.
    # ------------------------------------------------------------------
    for a in candidates:
        if time.time() > deadline:
            break
        sc = _score_of_seq(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] Init candidate → score={sc:.6f}")

    # Fallback to uniform if everything failed
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 3. Deterministic‑annealing gradient descent.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    alpha = 0.5                      # soft‑max temperature
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(current)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start:6.2f}s] Iter {it:5d} → new best {cur_score:.6f}")

        # ----- soft‑max weights and gradient -----
        w = np.exp(alpha * (conv - max_val))          # max weight = 1
        grad = _gradient_smooth(current, w)

        # diminishing step size
        eta = 0.3 / math.sqrt(it)

        cand = _project_simplex(current - eta * grad)
        cand_score = _score_of_seq(cand)

        # accept if better, otherwise occasional uphill move (SA style)
        if cand_score < cur_score:
            current = cand
        else:
            temp = max(0.01, 1.0 / math.sqrt(it))
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
                current = cand

        # anneal the soft‑max temperature
        if it % 500 == 0:
            alpha = min(alpha * 1.08, 20.0)

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final random hill‑climbing (swaps + tiny transfers).
    # ------------------------------------------------------------------
    current = best_seq.copy()
    while time.time() < deadline:
        # ---- swap two entries ----
        i, j = np.random.randint(0, n, size=2)
        if i != j:
            cand = current.copy()
            cand[i], cand[j] = cand[j], cand[i]
            sc = _score_of_seq(cand)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start:6.2f}s] Swap improve → {sc:.6f}")

        # ---- tiny mass transfer (≤5 % of source) ----
        if np.random.rand() < 0.3:
            i = np.random.randint(0, n)
            if current[i] > 0.0:
                j = np.random.randint(0, n)
                if i != j:
                    eps = np.random.rand() * 0.05
                    delta = eps * current[i]
                    cand = current.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    sc = _score_of_seq(cand)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_seq = cand.copy()
                        current = cand
                        print(f"[{time.time() - start:6.2f}s] Transfer improve → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    result = [float(x) for x in best_seq]
    print(f"[{time.time() - start:6.2f}s] Finished – best score {best_score:.6f}")
    return result
