Pseudo:

def project_onto_simplex(v):
   # projects v onto simplex {x >= 0, sum=1}
   n = v.shape[0]
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(n) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

def _project_onto_simplex(v):
    # Compute the projection onto the probability simplex.
    # Returns vector w such that w>=0, sum(w)=1.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
optimise a step‑function by minimizing 2·n·max(conv(a,a)).

The core is a projected Adam optimiser on a *smooth* approximation of the
max‑operator:
    fβ(a) = (1/β)·log Σ_i exp(β·conv_i(a))
For β→∞, fβ tends to max(conv).  The gradient of fβ is obtained in closed
form via a convolution of the weight vector w_i ∝ exp(β·conv_i) with the
reversed coefficient vector.  After each Adam step the vector is projected
onto the probability simplex (non‑negative, sum = 1).  A small β is used
initially for a smooth landscape and increased linearly to a large value,
so the optimiser first makes wide moves and then focuses on the true
max‑objective.  When the time budget is almost exhausted a cheap stochastic
hill‑climbing phase refines the best vector found.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence of non‑negative heights (as plain ``float`` objects) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the *linear* auto‑convolution of ``a`` (length n) using FFT.
    The result has length 2·n‑1.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed objective.
    w – weight vector (length 2·n‑1) that sums to 1.
    The formula is  g = 2·(w * a_rev)_{n‑1 … n‑1+n-1}, where * denotes linear
    convolution and a_rev = a[::-1].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=fft_len)
    Ar = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(Wf * Ar, n=fft_len)
    start = n - 1
    return 2.0 * conv[start:start + n]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    s = init.sum()
    if s == 0.0:
        a = np.full(target_n, 1.0 / target_n)
    else:
        a = init / s

    n = target_n

    # FFT lengths used repeatedly
    fft_len_conv = _next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w * a_rev

    # --------------------------------------------------------------
    # 2. Adam optimiser on a smoothed max‑convolution
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.002               # base learning rate (will decay as 1/√t)
    # Schedule for the smoothing factor β
    beta_start = 1.0
    beta_max = 200.0
    max_iters = 400_000      # hard iteration limit – the time budget will stop earlier

    m = np.zeros(n)          # first moment estimate
    v = np.zeros(n)          # second moment estimate

    # evaluate the initial score
    conv = _full_convolution_fft(a, fft_len_conv)
    max_val = conv.max()
    best_seq = a.copy()
    best_score = float(2.0 * n * max_val)
    print(f"[{time.time() - start_time:6.2f}s] start, score={best_score:.6f}")

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----------------------------------------------------------
        # compute full convolution and current score
        # ----------------------------------------------------------
        conv = _full_convolution_fft(a, fft_len_conv)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d} improved score={cur_score:.6f}")

        # ----------------------------------------------------------
        # smoothed max: weight vector w_i ∝ exp(β·conv_i)
        # ----------------------------------------------------------
        beta = beta_start + (beta_max - beta_start) * it / max_iters
        # shift by the maximum to keep the exponentials well‑scaled
        shift = conv - max_val          # ≤ 0
        w_unnorm = np.exp(beta * shift)
        w_sum = w_unnorm.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w_unnorm, 1.0 / w_unnorm.size)
        else:
            w = w_unnorm / w_sum

        # ----------------------------------------------------------
        # gradient of the smoothed objective
        # ----------------------------------------------------------
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ----------------------------------------------------------
        # Adam update (projected onto the simplex)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)          # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_onto_simplex(a_candidate)

        # ----------------------------------------------------------
        # occasional random mass transfer to avoid tiny plateaus
        # ----------------------------------------------------------
        if it % 5000 == 0:
            i, j = np.random.randint(0, n, size=2)
            eps_move = np.random.rand() * 0.02  # up to 2 % of source mass
            delta = eps_move * a[i]
            a[i] -= delta
            a[j] += delta
            a = _project_onto_simplex(a)

    # --------------------------------------------------------------
    # 3. final stochastic hill‑climbing (local search)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        conv = _full_convolution_fft(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps_move = np.random.rand() * 0.04
            delta = eps_move * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _project_onto_simplex(a_new)
            conv_new = _full_convolution_fft(a_new, fft_len_conv)
            new_score = 2.0 * n * conv_new.max()
            if new_score < cur_score:
                cur_score = new_score
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve score={new_score:.6f}")

    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
