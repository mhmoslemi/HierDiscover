def _run_subgradient(a0, n, slice_deadline, start_time):
    a = a0.copy()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # choose k_max (first occurrence)

def _solve_lp_max_sum_b(a, M, solver='ECOS', max_iters=2000):
    n = a.shape[0]
    # Build sparse matrix A
    from scipy.sparse import csr_matrix, coo_matrix
    rows = []
    cols = []
    data = []
    for j in range(n):
        # column j contains a shifted version of a
        # for i in range(n):
        #    row = i + j
        #    col = j
        #    val = a[i]
        # we can vectorize: row = np.arange(j, j+n)
        # but need to ensure rows <= 2n-2
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        data.extend(a.tolist())
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    # variable
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver, warm_start=True, max_iters=max_iters, verbose=False)
    if b.value is None:
        # fallback: return a copy of a
        return a.copy()
    b_val = np.maximum(b.value, 0.0)
    return b_val

"""
Low‑maximum self‑convolution step‑function search.

The optimizer works on the probability simplex (∑aᵢ = 1) because the
objective `2·n·max(conv(a,a))` is scale‑invariant.
It combines three components that differ from the previous approach:

1. **Exact sub‑gradient descent** – instead of a smoothed log‑sum‑exp
   surrogate we use the true gradient of the active convolution peak.
   A momentum term and a simple isotropic noise help to escape plateaus.

2. **Linear‑programming refinement** – for modest sizes (≤ 2048) we solve
   the LP
        maximise   ∑bⱼ   subject to   (a * b) ≤ M ,   b ≥ 0
   where `M = max(a*a)`.  The optimum gives a direction that strictly
   increases the feasible mass; after normalising it we mixture‑update
   the current distribution.  This step dramatically reduces the peak
   after only a few LP solves.

3. **Multiscale up‑sampling + annealing** – the best distribution from
   a given size is linearly interpolated to the next larger size,
   providing a warm‑start.  Finally a cheap mass‑transfer simulated
   annealing polishes the solution.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence found as a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact ℓ∞ surrogate
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     slice_deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that uses the true max‑convolution
    index.  Returns the best vector seen and its exact objective
    2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    beta = 0.9               # momentum factor
    step0 = 0.8              # base step size
    it = 0
    max_iters = 200_000      # hard cap; the loop will also stop at time

    while it < max_iters and time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        # identify one maximal index (first occurrence)
        k_max = int(np.argmax(conv))

        # exact gradient of conv[k_max] w.r.t. a
        g = np.zeros_like(a)
        i_min = max(0, k_max - n + 1)
        i_max = min(k_max, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        # occasional isotropic noise (helps escape plateaus)
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            g += noise_scale * np.random.randn(n)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a_candidate = a - step * momentum
        a = _project_simplex(a_candidate)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polishing (mass‑transfer)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Moves a small ε from a source index to a destination index.
    Returns the possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (moderate sizes only)
# ----------------------------------------------------------------------


def _build_conv_matrix(a: np.ndarray):
    """
    Build the (2n‑1) × n matrix A such that (A @ b) = a * b
    (linear convolution).  The matrix is Toeplitz and constructed
    as a sparse diagonal matrix for efficiency.
    """
    n = a.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal contains the whole vector a (truncated automatically)
    data = np.tile(a, (n, 1))                  # shape (n, n)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _solve_lp_max_sum_b(a: np.ndarray, M: float, max_iters: int = 5_000):
    """
    Solve   maximise   sum(b)   subject to   a * b ≤ M ,   b ≥ 0 .
    Returns the optimal b (may be un‑normalised).  If a solver is not
    available the function returns a copy of a (i.e. a no‑op).
    """
    try:
        import cvxpy as cp
    except Exception:
        # LP not available – fall back to identity
        return a.copy()

    n = a.shape[0]
    A = _build_conv_matrix(a)

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=max_iters, warm_start=True, verbose=False)
    except Exception:
        # solver failure – return a copy
        return a.copy()

    if b.value is None:
        return a.copy()
    b_val = np.maximum(b.value, 0.0)
    return b_val
