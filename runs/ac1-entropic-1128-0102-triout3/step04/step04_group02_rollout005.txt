def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    lengths = [256, 512, 1024, 2048, 4096, 8192]

def _adam_optimize(a0, n, deadline, start_time,
                   max_iter=20000,
                   lr0=0.5,
                   beta1=0.9,
                   beta2=0.999,
                   eps=1e-8,
                   lambda_factor_init=0.5,
                   lambda_decay=0.7,
                   decay_interval=2000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    lam_factor = lambda_factor_init

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b # sum a =1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # schedule lambda
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lambda_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        # step size schedule: can use lr = lr0 / sqrt(it)
        lr = lr0 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score

def _anneal_refine(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                   start_temp=5e-2, end_temp=1e-6, steps=20000):
    a = a0.copy()
    # ensure sum 1
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0/n)
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp)**(it/steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue
        eps = a[i] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # renormalize not needed as sum unchanged
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(new_n, 1.0/new_n)
    return up
```

"""
Propose a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The algorithm is a simple multi‑scale stochastic
search that combines:

*   a projected Adam optimiser on the probability simplex (using a
    log‑sum‑exp surrogate of the ℓ∞ convolution norm);
*   an exact‑subgradient phase when the surrogate temperature becomes tiny;
*   short random‑mass‑transfer simulated‑annealing refinements;
*   progressive up‑sampling of the best candidate to larger lengths.

The routine repeatedly tries new restarts at several sequence lengths,
always keeping the best solution seen so far.  All operations respect the
required non‑negativity, normalisation (sum = 1) and runtime budget
(~1000 s).  The implementation uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y via real FFT."""
    size = x.size + y.size - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Optimiser: Projected Adam on the simplex with log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    *,
    max_iter: int = 8000,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lambda_factor_init: float = 0.5,
    lambda_decay: float = 0.85,
    decay_interval: int = 400,
):
    """
    Projected Adam optimiser on the simplex.
    Returns the best sequence seen and its (internal) score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    lam_factor = lambda_factor_init

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b  # sum(a) == 1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}"
            )

        # ------------------------------------------------------------------
        # Temperature schedule for the log‑sum‑exp surrogate
        # ------------------------------------------------------------------
        if it % decay_interval == 0:
            lam_factor = max(lam_factor * lambda_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-12)

        # ------------------------------------------------------------------
        # Compute (smoothed) gradient of max(conv) w.r.t. a.
        # When lam is extremely small we fall back to the exact sub‑gradient.
        # ------------------------------------------------------------------
        if lam_factor < 1e-6:  # use exact sub‑gradient
            # treat all entries within a tiny tolerance as maximal
            tol = 1e-12
            mask = np.abs(conv - max_b) <= tol
            cnt = mask.sum()
            if cnt == 0:
                weights = np.exp((conv - max_b) / lam)
                weights /= weights.sum()
            else:
                weights = mask.astype(float) / cnt
        else:
            max_val = conv.max()
            weights = np.exp((conv - max_val) / lam)
            weights_sum = weights.sum()
            if weights_sum == 0.0:
                weights = np.full_like(weights, 1.0 / weights.size)
            else:
                weights /= weights_sum

        # gradient = 2 * (weights * a_rev) convolved, i.e. cross‑correlation
        a_rev = a[::-1]
        full = _conv_fft(weights, a_rev)  # length = 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # ------------------------------------------------------------------
        # Adam update (with bias correction)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        # simple step‑size schedule, decreasing like 1/√t
        lr = lr0 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _anneal_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    rng: np.random.Generator,
    *,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 20000,
):
    """
    Simple mass‑transfer annealing: move a random ε from i to j.
    Returns the best sequence (and its score) found during the sweep.
    """
    a = a0.copy()
    # ensure normalisation (should already hold)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)

    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * rng.random() * 0.4  # move at most 40 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
