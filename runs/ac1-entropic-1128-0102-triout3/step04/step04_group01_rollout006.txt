"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It uses a *deterministic
sub‑gradient descent* on the ℓ∞‑norm of the self‑convolution with a
simple back‑tracking line search, followed by a cheap mass‑transfer
annealing.  The sub‑gradient of maxᵢ (a∗a)[i] is easy to compute:
if the maximum occurs at index `k` then
∂/∂aⱼ (a∗a)[k] = 2·a_{k‑j} (outside the range the term is zero).
The descent is projected onto the simplex after each step, guaranteeing
validity of the candidate.  A multi‑scale schedule is used – the best
distribution found for a smaller length is linearly up‑sampled and used
as a warm‑start for a larger length.  Several random restarts and a
final annealing phase give an extra polishing pass.

The routine respects a 1000 s budget, prints occasional progress
messages and finally returns a plain Python list of non‑negative
heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT (length 2·n‑1)."""
    n = a.shape[0]
    # choose FFT length (next power of two ≥ 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent on the ℓ∞‑norm of the self‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.

    Parameters
    ----------
    conv : np.ndarray   – a*a (length 2n‑1)
    a    : np.ndarray   – current probability vector (sum = 1)

    Returns
    -------
    grad : np.ndarray   – dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # pick the *first* index attaining the maximum (any works)
    k = int(np.argmax(conv))

    # gradient_i = 2 * a_{k-i} if 0 ≤ k‑i < n, else 0
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad


def _run_subgrad_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Deterministic sub‑gradient descent with back‑tracking line search.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_score = 2.0 * n * max_val

    step = 0.3                # initial step size (tuned empirically)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and its maximum
        conv = _conv_self(a)
        max_val = conv.max()
        # sub‑gradient of the max entry
        grad = _subgradient_max(conv, a)

        # back‑tracking line search (decrease step until improvement)
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_try = step * factor
            a_new = _project_simplex(a - step_try * grad)
            conv_new = _conv_self(a_new)
            max_new = conv_new.max()
            if max_new < max_val:               # strict improvement
                a = a_new
                max_val = max_new
                step = step_try * 1.1           # slightly increase step after success
                accepted = True
                break

        if not accepted:
            step *= 0.5                         # shrink step if no progress

        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] subgrad n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 80_000) -> tuple[np.ndarray, float]:
    """Cheap annealing: move a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the grader
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # A modest schedule of increasing lengths.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    # If the budget is still large after these, we will try one more jump.
    extra_big = 32768

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Warm‑start initial vector
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Time allocation for this size
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # Larger sizes get a slightly larger share of the remaining budget.
        slice_time = max(8.0, remaining * (0.12 if n < 4096 else 0.18))
        slice_deadline = time.time() + min(slice_time, remaining * 0.5)

        # --------------------------------------------------------------
        # Sub‑gradient descent phase (main optimisation)
        # --------------------------------------------------------------
        a_opt, score_opt = _run_subgrad_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after descent n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # A couple of cheap random restarts (they sometimes escape a local basin)
        # --------------------------------------------------------------
        for _ in range(2):
            if time.time() >= slice_deadline:
                break
            a_restart = np.random.rand(n)
            a_restart /= a_restart.sum()
            a_opt_r, score_opt_r = _run_subgrad_descent(a_restart, n, slice_deadline, start_time)
            if score_opt_r < best_score:
                best_score = score_opt_r
                best_seq = a_opt_r.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart improve n={n} score={score_opt_r:.6f}")

        # --------------------------------------------------------------
        # Annealing polishing phase
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # keep the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final push on a larger size if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        try:
            n_big = extra_big
            a_big = _upsample(best_seq, n_big)
            slice_deadline = time.time() + min(30.0, deadline - time.time())
            a_opt_big, score_opt_big = _run_subgrad_descent(a_big, n_big, slice_deadline, start_time)
            if score_opt_big < best_score:
                best_score = score_opt_big
                best_seq = a_opt_big.copy()
                print(f"[{time.time() - start_time:6.2f}s] final upsample improvement n={n_big} score={score_opt_big:.6f}")
        except Exception:
            pass  # any failure is harmless – we keep the best found so far

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list – the evaluator will clamp / normalise as needed.
    return [float(x) for x in best_seq]
