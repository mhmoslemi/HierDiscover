def refine(a, n, deadline):
    best = a.copy()
    best_score = _score_of_seq(a)
    current = a.copy()
    iteration = 0
    while time.time() < deadline:
        iteration +=1
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val
        # update best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = current.copy()
        # compute active set
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _gradient_active(current, K)
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max != i_min and current[i_max] > 0:
            delta = min(0.1 * current[i_max], current[i_max]) # 10% step
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score - 1e-12:

        # fallback: exponentiated gradient

        current = a_new
    return best, best_score

Algorithm

1. Initialise from the historic construction if present, otherwise start

   step on the simplex.

3. Final cheap local search: repeatedly move a small fraction (≤4 %)
   of mass from a random source index to a random target index and keep
   any improvement.

All steps respect the simplex constraint (non‑negative, unit sum) and
use fast FFT‑based convolution.  The routine respects the ~1000 s wall
clock budget and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) for a vector a that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of a and the maximal entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` for the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    (non‑negative entries summing to 1).
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    The routine runs for roughly `budget_s` seconds (default 1000) and
    returns the best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and a first feasible vector
    # ------------------------------------------------------------------
    n = 5000
    best_seq = None
    best_score = float("inf")

    try:
        init = np.array(height_sequence_1, dtype=float)
        if init.ndim == 1 and init.size > 0:
            init = np.maximum(init, 0.0)
            s = init.sum()

                init = init / s
                best_seq = init.copy()
                best_score = _score_of_seq(best_seq)
                print(f"[{time.time() - start_time:6.2f}s] Init from height_sequence_1 "
                      f"(n={n}) → score={best_score:.6f}")
    except Exception:
        pass  # ignore – we will generate our own candidates below

    # ------------------------------------------------------------------
    # 2. Parametric families (power‑law, exponential, Gaussian, Beta)
    # ------------------------------------------------------------------
    families = []

    # Power‑law a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.10, 2.50, 25):
        families.append(('power', p))

    # Exponential decay a_i ∝ exp(-p·i)
    for p in np.linspace(5e-4, 0.020, 20):
        families.append(('exp', p))

    # Gaussian a_i ∝ exp(-0.5*((i-μ)/σ)^2)
    for sf in np.linspace(0.05, 0.25, 15):
        families.append(('gauss', sf))

    # Beta‑type a_i ∝ (i+1)^{α}·(n-i)^{β}
    for a in np.linspace(0.0, 5.0, 6):
        for b in np.linspace(0.0, 5.0, 6):
            families.append(('beta', a, b))

    # A handful of random Dirichlet starts
    for _ in range(30):
        families.append(('rand', None))

    # Scan the families
    for fam in families:
        if time.time() > deadline:
            break

        kind = fam[0]

        if kind == 'power':
            p = fam[1]
            i = np.arange(1, n + 1, dtype=float)
            a = i ** (-p)

        elif kind == 'exp':
            p = fam[1]
            i = np.arange(n, dtype=float)
            a = np.exp(-p * i)

        elif kind == 'gauss':
            sf = fam[1]
            mu = (n - 1) / 2.0
            sigma = sf * n
            i = np.arange(n, dtype=float)
            a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

        elif kind == 'beta':
            alpha = fam[1] + 1e-9   # avoid zero exponent
            beta = fam[2] + 1e-9
            i = np.arange(1, n + 1, dtype=float)
            a = (i ** alpha) * ((n + 1 - i) ** beta)

        elif kind == 'rand':
            a = np.random.exponential(scale=1.0, size=n)

        else:
            continue

        total = a.sum()
        if total == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        sc = _score_of_seq(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best from {kind} "
                  f"{fam[1:] if len(fam) > 1 else ''} → score={sc:.6f}")

        # a few cheap random permutations of the same multiset
        for _ in range(3):
            if time.time() > deadline:
                break
            perm = np.random.permutation(n)
            a_perm = a[perm]
            sc_perm = _score_of_seq(a_perm)
            if sc_perm < best_score:
                best_score = sc_perm
                best_seq = a_perm.copy()
                print(f"[{time.time() - start_time:6.2f}s] Permutation improvement "
                      f"({kind}) → score={sc_perm:.6f}")

    # If everything failed, fall back to the uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_of_seq(best_seq)
        print(f"[{time.time() - start_time:6.2f}s] Fallback uniform → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing + gradient‑descent refinement
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_score = _score_of_seq(current)

    # temperature schedule (linear cooling)
    T_start = best_score * 0.01          # about 1 % of the current score
    T_end = 1e-9

    it = 0
    while time.time() < deadline:
        it += 1
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] SA iter {it:6d} "
                  f"→ new best score={cur_score:.6f}")

        # active set for sub‑gradient
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _gradient_active(current, K)

        # greedy mass transfer (source = max‑gradient, target = min‑gradient)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or current[src] <= 0.0:
            # fallback to a completely random pair
            src, dst = np.random.choice(n, 2, replace=False)

        delta = min(0.07 * current[src], current[src])   # move up to 7 % of source
        cand = current.copy()
        cand[src] -= delta
        cand[dst] += delta
        # safely guard against negative due to rounding
        if cand[src] < 0.0:
            cand[src] = 0.0

        conv2, max2 = _conv_and_max(cand)
        new_score = 2.0 * n * max2

        # acceptance probability
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        T = T_start * (1 - frac) + T_end * frac

        if new_score < cur_score - 1e-12:
            accept = True
        else:
            accept = (T > 0.0) and (np.random.rand() < math.exp(-(new_score - cur_score) / T))

        if accept:
            current = cand
            continue

        # fallback: exponentiated‑gradient (mirror descent) step on the simplex
        eta = 0.5 / math.sqrt(it)
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        log_a -= np.max(log_a)                     # numerical stabilisation
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        current = a_new

    # ------------------------------------------------------------------
    # 4. Final cheap random hill‑climbing
    # ------------------------------------------------------------------
    current = best_seq.copy()
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or current[i] == 0.0:
            continue
        eps = np.random.rand() * 0.04        # up to 4 % of source mass
        delta = eps * current[i]
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta

        conv2, max2 = _conv_and_max(cand)
        new_score = 2.0 * n * max2
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = cand.copy()
            current = cand
            print(f"[{time.time() - start_time:6.2f}s] Final hill‑climb → score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
