    start = time.time()
    deadline = start + budget_s - 0.5

    n = 1500
    a = np.full(n, 1.0/n)  # uniform start

    # maybe start from height_sequence_1 if exists larger improvement
    try: 
         prev = np.array(height_sequence_1, dtype=np.float64)
         if prev.shape[0] == n:
            a = prev / prev.sum()  # normalize to sum 1
    except: pass

    best_a = a.copy()
    best_score = fast_score(a)

    # parameters
    max_iter = 200000  # ensure early break by deadline
    iter = 0
    # step size schedule
    eta0 = 0.5
    # smoothing alpha schedule
    alphas = [5, 10, 20, 40, 80]  # gradually increase
    alpha_index = 0
    next_alpha_iter = 0

    # precompute FFT size
    size = 1 << ((3 * n - 2).bit_length())
    ind = np.arange(n)

    while time.time() < deadline and iter < max_iter:
        # choose current alpha
        if iter >= next_alpha_iter and alpha_index < len(alphas)-1:
            alpha_index += 1
            # increase intensity slowly; set next switch after 20000 iter or time
            next_alpha_iter = iter + 20000

        alpha = alphas[alpha_index]

        # compute conv
        fft_a = np.fft.rfft(a, size)
        # calculate convolution a*a (for evaluating max)
        conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
        max_conv = conv.max()
        # compute score (sum is 1)
        cur_score = 2.0 * n * max_conv  # sum=1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {iter}] new best {best_score:.6f}")

        # compute smoothing distribution p
        # compute log-sum-exp trick
        logits = alpha * conv
        max_logit = logits.max()
        exp_logits = np.exp(logits - max_logit)  # subtract max for stability
        p = exp_logits / exp_logits.sum()
        # gradient: 2 * convolution of a with p as described
        # compute p_rev for gradient
        p_rev = p[::-1]  # length 2n-1
        fft_p_rev = np.fft.rfft(p_rev, size)
        grad_full = np.fft.irfft(fft_a * fft_p_rev, size)
        # grad_i = 2 * grad_full[(2n-2)-i]
        grad = 2.0 * grad_full[(2*n-2) - np.arange(n)]

        # step size
        eta = eta0 / (1.0 + 0.0001 * iter)  # decaying

        # gradient descent step
        v = a - eta * grad

        # project onto simplex
        # Euclidean projection onto simplex of sum=1
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind2 = np.arange(1, n+1)
        cond = u - cssv / ind2 > 0
        if np.any(cond):
            rho = ind2[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # random perturbation occasionally
        if np.random.rand() < 0.005:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:

                a[j] += delta

        iter += 1

    # after loop, final best sequence (maybe convert to list)
    print(f"[final] best score {best_score:.6f} after {iter} iterations")
    # scale to original sum (doesn't matter) - keep as is
    return best_a.tolist()

"""
Gradient‑descent search for a low‑autocorrelation step function.

The evaluation function is invariant to scaling, so we work on the
probability simplex (non‑negative entries that sum to 1).  We minimise a
smooth approximation of the max‑convolution:
    f(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
whose gradient is
    ∇f(a) = 2·a ⊙ p   (cross‑correlation with the soft‑max vector p).
Each iteration performs a projected gradient step onto the simplex,
occasionally adds a random mass transfer to escape plateaus, and
periodically sparsifies the vector (keeps only the largest entries)
to explore the “support‑size” regime that worked well in earlier
experiments.  If the time budget is not exhausted a short
sub‑gradient optimiser is run as a safety net.

The algorithm returns the best sequence found within the 1000 s limit.
"""
import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the O(n) algorithm."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast version of evaluate_sequence (no safety checks) – used internally.
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) / (sum(a))² (sum is usually 1)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient optimiser used as a fallback (simple version).
def subgradient_optimize_with_limit(a_init: np.ndarray,
                                    time_limit: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for a short remaining time budget."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    fft_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        a = simplex_projection(a - eta * grad)

        # occasional random mass transfer
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)
                val = fast_score(cand)
                if val < best_val:
                    best_val, best_a = val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine runs a projected gradient descent on a smoothed
    max‑convolution objective, periodically sparsifies the vector,
    and finally falls back to a sub‑gradient optimiser if time permits.
    """

    fft_size = 1 << ((3 * n - 2).bit_length())   # enough for a * p_rev

    while time.time() < deadline:
        # increase smoothing precision gradually
        if it >= next_alpha_iter and alpha_idx < len(alpha_vals) - 1:

        conv = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_val:
            best_val, best_a = cur_score, a.copy()
            print(f"[iter {it}] new best score {best_val:.6f}")

        # ----- smooth approximation of the max -----
        # log‑sum‑exp in a numerically stable way
        logits = alpha * conv
        mx = logits.max()
        exp_shifted = np.exp(logits - mx)
        p = exp_shifted / exp_shifted.sum()               # length 2n‑1, Σp = 1

        # gradient = 2 * (a cross‑correlated with p)
        p_rev = p[::-1]                                   # reverse for convolution
        fp = np.fft.rfft(p_rev, fft_size)
        grad_full = np.fft.irfft(fa * fp, fft_size)       # length >= 3n‑2
        grad = 2.0 * grad_full[(2 * n - 2) - idx]          # extract needed entries

        # step size (simple decay)
        eta = eta0 / (1.0 + 1e-4 * it)

        # projected gradient step onto the simplex
        a = simplex_projection(a - eta * grad)

        # occasional random mass transfer (helps escape plateaus)
        if np.random.rand() < 0.005:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                a = simplex_projection(a)

        # sparsification every few thousand iterations
        if it % 5000 == 0 and it > 0:
            # target number of non‑zero entries gradually shrinks
            m_target = max(300, int(0.6 * n * (1.0 - it / 50000.0)))
            if m_target < n:
                # keep the m_target largest components
                keep_idx = np.argpartition(a, -m_target)[-m_target:]
                mask = np.zeros_like(a, dtype=bool)
                mask[keep_idx] = True
                a = simplex_projection(np.where(mask, a, 0.0))
                print(f"[iter {it}] sparsified to {m_target} non‑zero entries")

        it += 1

        # early exit if target reached
        if best_val <= 1.503:
            print("[success] target reached, exiting early")
            break

    # ------------------------------------------------------------------
    # 3. Fallback: short sub‑gradient optimisation if still above target
    if best_val > 1.503 and time.time() < deadline - 1.0:
        remaining = deadline - time.time()
        print("[fallback] running short sub‑gradient optimiser")
        best_a, best_val = subgradient_optimize_with_limit(best_a, remaining)
