"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a))**2

(the `evaluate_sequence` routine normalises to a simplex internally).

Algorithm
---------

1. **Simplex initialisation** – start from a random simplex point or, if the
   global variable ``height_sequence_1`` exists, use it (normalised).

2. **Progressive sizes** – optimise first on a modest length (≈800),
   then up‑sample the best vector to larger lengths (2400, 4800, 9600 …).
   Larger n gives a smaller average convolution value, which can lower the
   final score.

3. **Smoothed max‑objective** – the non‑smooth objective
   ``max(conv)`` is approximated by the log‑sum‑exp

        f_α(a) = (1/α)·log( Σ_k exp(α·conv_k) )

   for a sequence of increasing smoothing parameters α (0.1 → 100).
   The gradient of f_α is cheap to obtain in the Fourier domain:
   `∇f_α = 2·a ⊛ p` where `p` are the soft‑max weights of the convolution
   entries.  A projected (Nesterov‑style) gradient step on the probability
   simplex is performed, with a diminishing step size ∝ 1/√t.

4. **Final stochastic refinement** – a tiny simulated‑annealing walk that
   moves a small amount of mass between two coordinates and accepts
   downhill or occasional uphill moves.  This cleans up any remaining
   irregularities.

All phases respect the time budget (default 1000 s).  The best sequence
found is returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection -------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT based convolutions ---------------------------------------------
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a power‑of‑two FFT (real signals)."""
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << (n - 1).bit_length()          # next power of two ≥ n
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a (linear convolution)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# 3.  Linear up‑sampling -------------------------------------------------
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# 4.  Smoothed gradient descent on the simplex -------------------------
# ----------------------------------------------------------------------
def _smooth_grad_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alphas,
    step0: float = 0.6,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of max(conv) using projected
    Nesterov‑style gradient descent.

    Returns the best vector found (normalised to sum = 1) and its
    original score = 2·n·max(conv(a,a)).
    """
    # FFT length large enough for both a*a (2n‑1) and a*rev(p) (3n‑2)
    L = 1 << ((3 * n - 2) - 1).bit_length()

    a = a0.copy()
    a = a / a.sum()
    best = a.copy()
    best_score = float("inf")
    it = 0
    momentum = np.zeros_like(a)

    for alpha in alphas:
        while time.time() < deadline:
            # ----- forward FFT of the current iterate -----
            A = np.fft.rfft(a, n=L)                     # A = FFT(a)

            # ----- convolution a * a (needed for the soft‑max) -----
            conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
            max_conv = conv_full.max()

            # ----- soft‑max weights of the convolution entries -----
            exp_vals = np.exp(alpha * (conv_full - max_conv))
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0:
                p = np.full_like(exp_vals, 1.0 / exp_vals.size)
            else:
                p = exp_vals / sum_exp                     # length 2n‑1

            # ----- gradient = 2·a ⊛ p  (cross‑correlation) -----
            revp = p[::-1]
            P = np.fft.rfft(revp, n=L)                    # FFT of reversed weights
            grad_full = np.fft.irfft(A * P, n=L)[: 3 * n - 2]
            offset = len(p) - 1                           # = 2n‑2
            grad = 2.0 * grad_full[offset : offset + n]   # length n

            # ----- Nesterov momentum & step -----
            momentum = beta * momentum + (1.0 - beta) * grad
            it += 1
            step = step0 / math.sqrt(it)
            a = _project_simplex(a - step * momentum)

            # ----- evaluate the true (non‑smoothed) objective -----
            # we recompute the convolution of the updated a
            A_new = np.fft.rfft(a, n=L)
            conv_new = np.fft.irfft(A_new * A_new, n=L)[: 2 * n - 1]
            cur_score = 2.0 * n * conv_new.max()

            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] smooth n={n} "
                    f"α={alpha:.2f} iter={it} score={cur_score:.6f}"
                )

            # safeguard against extremely long runs
            if it % 20000 == 0 and time.time() >= deadline:
                break

        if time.time() >= deadline:
            break

    best = best / best.sum()
    return best, best_score


# ----------------------------------------------------------------------
# 5.  Tiny simulated‑annealing refinement -------------------------------
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing (move ε from i to j)."""
    a = a0.copy()
    # one‑off computation of the current score
    L = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate proposal
        A_prop = np.fft.rfft(a_prop, n=L)
        conv_prop = np.fft.irfft(A_prop * A_prop, n=L)[: 2 * n - 1]
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n} iter={it} score={score:.6f}"
            )

    return best_seq, best_score
