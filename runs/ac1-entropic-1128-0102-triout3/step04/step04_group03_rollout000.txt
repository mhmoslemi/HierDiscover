def _run_adam(a0, n, stage_deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=1e7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < stage_deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve n={n} iter={it} score={cur_score:.6f}")
        # subgradient
        grad = _subgradient_full(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1.  Progressive up‑sampling: start with a modest length (≈800) and
    repeatedly double the size (2400 → 4800 → 9600).  The current best
    sequence is linearly interpolated to the new size each stage.

2.  For each size run a few *restarts* of projected Adam optimisation
    on the probability simplex.  The objective is
        f(a) = 2·n·max(conv(a,a))
    (the denominator of `evaluate_sequence` is fixed to 1 by keeping
    `∑a = 1`).  The Adam update is followed by a Euclidean projection
    onto the simplex.

3.  A cheap greedy refinement that repeatedly moves a small amount of
    mass away from indices that contribute to the current maximal
    convolution entry.

4.  A final simulated‑annealing phase (mass‑transfer moves) that
    further reduces the peak.

The routine respects the given time budget, prints progress and
returns the best sequence found as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (as in the reference code)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are considered.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each product a_i * a_j appears twice in the sum of gradients
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              lr: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex.
    Returns the best point seen and its score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    last_improve = 0
    patience = 20000  # stop early if no improvement for many iters

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # stop if stagnant for a long while
        if it - last_improve > patience:
            break

        # sub‑gradient of max(conv)
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (down‑hill mass transfers)
# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   max_iters: int = 5000) -> np.ndarray:
    """
    Very cheap downhill refinement:
    repeatedly move a tiny amount of mass from a contributor to the
    maximal convolution entry to a random non‑contributing index.
    """
    a = a0.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        k = int(conv.argmax())           # index of the peak

        # indices that participate in the peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx

        # pick the i with the largest coefficient among the contributors
        i = i_idx[np.argmax(a[i_idx])]
        if a[i] < 1e-12:
            continue

        # propose a move to a random index not in the contributor set
        # try a few candidates; accept the first that strictly reduces the peak
        moved = False
        for _ in range(3):
            p = np.random.randint(0, n)
            if lo <= p <= hi:
                continue
            eps = min(a[i] * 0.05, 1e-6)   # tiny transfer
            if eps <= 0:
                continue
            a[i] -= eps
            a[p] += eps

            new_max = _conv_self(a).max()
            if new_max < max_val - 1e-12:
                moved = True
                break          # keep the move
            # revert and try another p
            a[i] += eps
            a[p] -= eps

        if not moved:
            # nothing helped – give up early
            break

    # re‑normalise (numerical drift)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------

    annealing phase, and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter sequence (if the notebook supplied one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage configuration
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]          # progressively larger sizes
    total_budget = max(0.0, deadline - start_time)

    # Reserve ~20 % of the budget for the final annealing phase
    anneal_budget = max(30.0, 0.20 * total_budget)
    grad_budget = total_budget - anneal_budget

    # ------------------------------------------------------------------
    # Gradient / Adam stages
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
