"""
Propose a low‑score step‑function using three phases:

1.  Initialise from the previously discovered best sequence
    (global variable ``height_sequence_1``) or a uniform start.
2.  Repeatedly solve a linear programme that pushes mass from
    positions that contribute to the current maximal convolution
    towards positions that do not.  The LP maximises the total
    weight of a candidate ``b`` under the constraint ``f * b ≤ M``
    (where ``M`` is the current max‐convolution).  The resulting
    direction is mixed into the current distribution.
3.  Refine the obtained distribution with a short Adam optimisation
    (smooth‑max approximation) and a cheap stochastic hill‑climber.
    All vectors are projected onto the probability simplex after
    each update, guaranteeing non‑negativity and sum = 1.
The routine respects the supplied time budget (default 1000 s)
and returns a plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the (non‑reversed) auto‑convolution of ``a``."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _score_from_vec(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a))  (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv), full convolution) – used by the Adam optimiser."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max objective w.r.t. ``a``."""
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser for 2·n·max(conv) on the simplex.
    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        g = _gradient_from_weights(a, w, fft_len_grad)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step function within ``budget_s`` seconds.
    The method combines a linear‑programming push‑away step,
    a short Adam refinement and a cheap stochastic hill‑climber.
    Returns a plain list of non‑negative floats.
    """

    import time
    from scipy.optimize import linprog

    np.random.seed(seed)

    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        if init_seq.sum() == 0.0:
            raise ValueError
        init_seq = init_seq / init_seq.sum()
    except Exception:
        # fallback: uniform on a modest size
        n0 = 1200

    a = init_seq.copy()

    # ------------------------------------------------------------------
    # 1. Linear‑programming push‑away iterations
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline and lp_iter < 15:
        lp_iter += 1
        # current max‑convolution

        # build Toeplitz‑type constraint matrix A (size (2n‑1) × n)
