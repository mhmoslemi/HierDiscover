"""
Optimisation of a non‑negative step‑function (vector of heights) that
minimises `evaluate_sequence`.  The algorithm works on the probability
simplex (∑a_i = 1) because the objective is homogeneous.
It combines three ideas:

1.  Projected Adam on a smoothed log‑sum‑exp surrogate of the
    ℓ∞‑norm of the self‑convolution.
2.  Multi‑scale warm‑starts: after a run on a small size the best
    vector is up‑sampled (linear interpolation) and used as the initial
    point for a larger size.
3.  Cheap mass‑transfer simulated annealing that moves a small ε from a
    source index to a destination index, polishing the best solution
    found by the gradient phase.

The routine respects the 1000 s budget, prints occasional progress
messages and finally returns a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) = λ log Σ exp(conv/λ).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a  (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad : dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerically stable soft‑max over the convolution entries
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # ∂L/∂a_i
    return grad

# Optimisers
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv) because the vector is kept on the simplex).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                # base learning rate before projection

    # λ‑schedule parameters
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # λ update (never let it become exactly zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # occasional isotropic noise to help escape flat regions
        if it % 1000 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # Adam's moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step before projection
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # clip extreme steps to keep iterates stable
        step = np.clip(step, -0.5, 0.5)

        a = _project_simplex(a - step)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Very cheap mass‑transfer annealing.
    Returns (possibly) improved sequence and its exact objective.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move: transfer a small ε from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Runs for at most `budget_s` seconds and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – we start modest and climb up to 32k (or 64k if time permits)
    lengths = [2048, 4096, 8192, 16384, 32768, 65536]
    # Shuffle the order only mildly; keep roughly increasing order to benefit from warm‑starts
    lengths.sort()
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 (warm start / random)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Allocate a time slice for this length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Give later (larger) lengths a slightly larger share, but keep a minimum.
        # The schedule is heuristic: 0.15 of remaining time per remaining length.
        slice_fraction = 0.15 / (len(lengths) - idx)
        slice_time = max(8.0, remaining * slice_fraction)
        slice_deadline = time.time() + slice_time
        if slice_deadline > deadline:
            slice_deadline = deadline

        # ------------------------------------------------------------------
        # Projected Adam optimisation on the log‑sum‑exp surrogate
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Global improve after Adam n={n} score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # Cheap annealing polish (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=50000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] Global improve after anneal n={n} score={score_ref:.6f}")

        # Store the current best as warm‑start for the next (larger) size
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should almost never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Prune tiny entries (helps with numerical stability) and re‑normalise
    tol = 1e-12
    best_seq = np.where(best_seq < tol, 0.0, best_seq)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq]
