"""
Propose a low‑score step‑function using a focused multi‑scale Adam optimiser.
The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
minimises 2·n·max(conv(a,a)).  It proceeds in three phases:

1.  Start from the previously discovered construction (height_sequence_1)
    if present, otherwise from a uniform vector.
2.  Successively up‑sample the current best shape to larger lengths
    (≈×1.5 each stage, up to a configurable max) and run a long
    projected‑Adam optimisation.  The Adam step uses a soft‑max
    approximation of the max‑convolution with a temperature that cools
    from 0.1 down to 1e‑7, which makes the gradient converge to the
    true sub‑gradient of the max‑operator.
    After each Adam step the vector is forced to be symmetric
    (a = (a + a[::-1]) / 2) – symmetry is known to be beneficial for
    flattening the auto‑convolution – and then projected back onto the
    simplex.
3.  With the remaining budget a cheap stochastic local‑search perturbs two
    coordinates at a time; moves that improve the objective are kept.
    The whole routine respects the global deadline (default 1000 s).

The function returns a plain list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution.  `a` must sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the (soft‑max) max‑convolution objective.
    Implements g = 2 * conv_rev(a, w), where conv_rev is the linear convolution
    of the reversed vector a with the weight vector w.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    *,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.1,
    temp_end: float = 1e-7,
    symmetrize: bool = True,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Projected Adam optimiser for 2·n·max(conv(a,a)).
    Returns (final_a, best_a_seen, best_score).
    """
    n = a0.shape[0]
    a = a0.copy()
    # pre‑compute FFT lengths
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and current score ----
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- temperature schedule (exponential cooling) ----
        tau = temp_start * (temp_end / temp_start) ** (it / max_iters)
        tau = max(tau, 1e-14)

        # ---- soft‑max weights over convolution entries ----
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # ---- gradient of the smooth‑max ----
        g = _gradient_from_weights(a, w, fft_len_grad)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ---- optional symmetrisation ----
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # ---- projection onto the simplex ----
        a = _project_simplex(a)

    return a, best_a, best_score


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0:
        up = np.full(new_len, 1.0 / new_len)
    else:
        up = up / s
    return up


def _estimate_iter_time(n: int) -> float:
    """Empirical estimate of the time of one Adam iteration for length n."""
    # calibrated on the reference implementation: ~6e‑4 s per 1200 entries
    return 6e-4 * (n / 1200.0) + 1e-4

        global_best_seq = np.full(n, 1.0 / n)

    print(
        f"[{time.time() - start_time:6.2f}s] Finished – best 2·n·max(conv) = {global_best_score:.6f}"
    )
    return [float(x) for x in global_best_seq]
