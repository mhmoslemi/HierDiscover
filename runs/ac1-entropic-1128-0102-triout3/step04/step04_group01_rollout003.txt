"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It follows a multi‑scale
hierarchy:

1.  Warm‑start the current best distribution for a larger length by
    linear interpolation (upsampling).
2.  Run a few thousand iterations of *smoothed* log‑sum‑exp gradient
    descent (the same surrogate used previously) – this quickly finds
    a good basin.
3.  Refine the candidate with a *exact sub‑gradient* descent on the
    ℓ∞‑norm of the self‑convolution.  The sub‑gradient of
    maxᵢ (a∗a)ᵢ is 2·(active‑set indicator) ⊛ a_rev, which can be
    computed with a single FFT.
4.  Polish the result with a cheap simulated‑annealing “mass‑transfer”
    scheme that moves a fraction of the mass from one index to another.
5.  The whole process is repeated for a schedule of lengths,
    giving the largest length the majority of the remaining time.

All vectors are projected onto the simplex after every update.
The routine respects a 1000 s budget and returns a plain Python list
of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, power‑of‑two length)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# 1) Smoothed log‑sum‑exp gradient descent (same spirit as the previous code)
# ----------------------------------------------------------------------
def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Smoothed gradient descent with momentum."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    beta = 0.9
    step0 = 0.8

    # λ‑schedule parameters
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_score:.6f}")

        # λ update
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient via log‑sum‑exp
        max_val = conv.max()
        weights = np.exp((conv - max_val) / lam)
        weights /= weights.sum()
        a_rev = a[::-1]
        full = _conv_fft(weights, a_rev)            # length 3n‑2
        start = n - 1
        grad = 2.0 * full[start:start + n]          # dL/da_i

        # occasional isotropic noise (helps escape flat plateaus)
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            grad += noise_scale * np.random.randn(n)

        # momentum & diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             slice_deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the ℓ∞‑norm of the self‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    step0 = 0.6          # initial step scale (empirically good)
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        tol = max_b * 1e-12
        active_idx = np.where(conv >= max_b - tol)[0]

            active_idx = np.array([np.argmax(conv)])

        u = np.zeros_like(conv)

        # gradient = 2 * (u ⊛ a_rev) restricted to positions 0…n‑1
        a_rev = a[::-1]
        full = _conv_fft(u, a_rev)        # length 3n‑2
        start = n - 1
        grad = 2.0 * full[start:start + n]

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * grad)

    return best_a, best_score

# 3) Cheap simulated‑annealing polishing (mass‑transfer)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """Very cheap annealing: move a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of the source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

                steps=50000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # remember the current best for the next up‑sample
        a_prev = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
