def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, numpy as np
    np.random.seed(seed)
    start_time=time.time()
    deadline = start_time + budget_s - 1.0

    # gather possible starting sequence
    try: starter = np.asarray(height_sequence_1, dtype=float)
    except: starter = None
    if starter is not None and starter.ndim==1 and starter.size>0:
        starter = starter / starter.sum()
    else:
        starter = None

    # Stage sizes
    stage_lengths = [800, 2400, 4800, 9600]
    # allocate time: total_grad_time = budget - anneal_time
    total_time = deadline - start_time
    anneal_time = max(30.0, 0.15*total_time)
    grad_time = total_time - anneal_time
    # compute weight sum for dividing time: sum(stage_lengths)
    total_weight = sum(stage_lengths)

    best_seq = None
    best_score = float('inf')

    for n in stage_lengths:
        if time.time() >= deadline: break
        # allocate time
        stage_frac = n / total_weight
        stage_budget = max(5.0, grad_time * stage_frac)
        stage_deadline = time.time() + stage_budget

        # initialise
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run Adam optimization
        a_opt, score_opt = _adam_optimize(a0, n, stage_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update grad_time left
        elapsed = time.time() - start_time
        grad_time = max(0.0, deadline - start_time - anneal_time - elapsed)

    # Final annealing
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(best_seq, n_final,
                                          deadline, start_time,
                                          best_score, best_seq,
                                          start_temp=5e-2, end_temp=1e-6,
                                          steps=500000)
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: {score_ref:.6f}")

    # final normalisation to sum=1
    if best_seq is None:
        best_seq = np.full(100, 1.0/100)

    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0/best_seq.size)

            beta = 30.0

            beta = 150.0

"""
Propose a low‑score step‑function.

The algorithm consists of three phases:

1.  **Progressive up‑sampling** – start from a modest size (800), up‑sample
    the best candidate to larger sizes (2400 → 4800 → optionally 9600).

2.  **Adam optimisation with a smoothed max objective** – minimise
    `max(conv)` where `conv = a*a`.  The max is approximated by a
    log‑sum‑exp with a temperature schedule (β grows from 1 to 200).
    Adam’s adaptive steps and Nesterov‑style momentum give faster
    convergence than the simple sub‑gradient method used before.

3.  **Final stochastic annealing** – tiny mass‑transfer moves
    (up to 50 % of the source weight) are accepted with a Metropolis rule.
    This refines the solution after the deterministic optimisation.

All updates are projected onto the probability simplex
(`a_i ≥ 0 , Σ a_i = 1`).  The routine respects the given time budget
(≈ 1000 s) and returns the best sequence found as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n) algorithm)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution (FFT for speed on large inputs)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b (no circular wrap)."""
    n = a.shape[0]
    m = b.shape[0]
    size = n + m - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)[:size]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return _conv_full(a, a)


# ----------------------------------------------------------------------
# Upsampling (linear interpolation, keeps total mass = 1)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` from its current length to `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation of a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   stage_deadline: float,
                   stage_start: float,
                   beta_start: float = 1.0,
                   beta_end: float = 200.0,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 10**7,
                   overall_start: float = None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max of the self‑convolution.
    Returns the best sequence seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # protect against missing overall start time for logging
    if overall_start is None:
        overall_start = stage_start

    while time.time() < stage_deadline and it < max_iter:
        it += 1

        # self‑convolution
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - overall_start:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # ---- smooth max (log‑sum‑exp) -------------------------------------------------
        # linear schedule for β
        stage_elapsed = time.time() - stage_start
        stage_total = stage_deadline - stage_start
        frac = min(1.0, stage_elapsed / stage_total) if stage_total > 0 else 1.0
        beta = beta_start + frac * (beta_end - beta_start)

        # stable exponentiation
        shift = max_val  # subtract the max to avoid overflow
        exp_arg = beta * (conv - shift)
        weights = np.exp(exp_arg)
        w_sum = weights.sum()
        if w_sum == 0:
            weights = np.full_like(conv, 1.0 / conv.size)
        else:
            weights /= w_sum

        # ---- gradient of the smoothed objective --------------------------------------
        # gradient of Σ w_k * conv_k w.r.t a_i:
        #   g_i = 2 * Σ_p a_p * w_{p+i}
        w_rev = weights[::-1]                         # length 2n‑1
        conv2 = _conv_full(a, w_rev)                 # length 3n‑2
        # central slice contains the needed sums (indices n‑1 … 2n‑2)
        start = n - 1
        mid = conv2[start:start + n]                 # shape (n,)
        grad = 2.0 * mid[::-1]                        # reverse to get g_i

        # ---- Adam update --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)   # enforce non‑negativity and Σa = 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   overall_start: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing with larger possible ε."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 50 % of the source mass (but not more than all of it)
        eps = a[i] * np.random.rand() * 0.5
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - overall_start:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function with minimal `evaluate_sequence` score.
    The routine respects the given time budget, grows the sequence size
    progressively, runs an Adam optimiser on a smoothed max‑convolution,
    and finishes with a short annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Optional starter: use the sequence from a previous run if available
    # ------------------------------------------------------------------
    try:

        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage configuration
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]  # progressively larger problems
    total_len_weight = sum(stage_lengths)

    # Reserve ~15 % of the total budget for the final annealing phase
    total_time = deadline - start_time
    anneal_time = max(30.0, 0.15 * total_time)
    grad_time = total_time - anneal_time

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Gradient (Adam) phases
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time for this stage (proportional to n)
        stage_frac = n / total_len_weight
        stage_budget = max(5.0, grad_time * stage_frac)
        stage_deadline = time.time() + stage_budget
        stage_start = time.time()

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:

        # run Adam optimisation
        a_opt, score_opt = _adam_optimize(
            a0, n,
            stage_deadline=stage_deadline,
            stage_start=stage_start,
            beta_start=1.0,
            beta_end=200.0,
            lr=0.05,
            overall_start=start_time
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # recompute remaining gradient budget
        elapsed = time.time() - start_time
        grad_time = max(0.0, deadline - start_time - anneal_time - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=500_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list for the harness
    return [float(x) for x in best_seq]
