def _adam_descent(a0, n, deadline, start_time, max_iter=10**7, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t=0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        grad = _subgradient_full(a, conv)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.5 / (np.sqrt(v_hat) + eps)  # adjust base step
        a = a - step * m_hat
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm is a lightweight multi‑stage optimiser:

1. **Initial shape** – use a discretised arcsine (Beta(0.5,0.5)) density,
   which is already fairly flat when convolved with itself.
2. **Projected gradient descent** – minimise the max‑convolution using a
   sub‑gradient of the active constraints and a Nesterov‑style momentum.
   The step size decays like 1/√t.
3. **Progressive up‑sampling** – repeatedly linearly interpolate the current
   best sequence to a larger size (800 → 2400 → 4800 → 9600) and continue the
   same optimiser; larger n gives a lower theoretical bound 2 n·max/∑²≈1.
4. **Final stochastic refinement** – a cheap simulated‑annealing walk that
   moves a tiny amount of mass between two positions and accepts downhill or
   uphill moves with a geometric temperature schedule.
5. **Safety fallback** – return a uniform distribution if everything else fails.

All heavy work respects the supplied time budget (default 1000 s).  The best
sequence discovered is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    # Average over the active set – this is a standard sub‑gradient choice.
    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.8,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"gd n={n:5d} iter={it:7d} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"anneal n={n:5d} iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------

    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,

    progressive up‑sampling and a final annealing phase, and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional starter from a previous run (if the global variable is
    #    defined by the harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()

    anneal_budget = max(30.0, 0.20 * (deadline - start_time))

    # ------------------------------------------------------------------
