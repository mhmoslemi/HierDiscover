"""
Propose a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` score.

Algorithm Overview
------------------

1. **Initialisation** – start from the previously discovered best
   construction (`height_sequence_1`) if it exists, otherwise fall back
   to a power‑law shape (length ≈ 5000) and normalise it to sum‑1.

2. **Iterative local optimisation** – while the time budget permits we
   repeatedly

   * compute the full auto‑convolution `c = a*a` and its maximum `M`;
   * build an *active set* `K` of convolution indices that are within
     a tiny relative tolerance of `M`;
   * obtain a sub‑gradient `g` of the objective by summing the
     contributions of all active indices (`g_i = 2 * Σ_{k∈K} a_{k-i}`);
   * select a few *source* indices with the largest gradient entries
     and a few *destination* indices with the smallest ones;
   * for every source–destination pair we determine the optimal amount
     of mass `δ` to transfer (using a cheap ternary‑search on the
     convex function `f(δ)=max(c + 2δ·diff)`) and keep the best improving
     move; if an improvement is found the move is applied.
   * when no improving pair exists we perform a mirror‑descent (exponential
     gradient) step on the probability simplex as a fallback.

   The move size is chosen optimally, not fixed to 5 % as in the
   previous code, which allows deeper reductions of the convolution
   peak.

3. **Random hill‑climbing** – after the main loop a cheap random
   mass‑transfer phase (small swaps of up to 4 % of a source entry)
   is performed until the budget expires.

4. **Return** – the best sequence encountered (as plain Python floats)
   is returned.  The sequence always stays non‑negative and sums to 1,
   therefore the scale‑invariance of `evaluate_sequence` guarantees a
   valid score.

The routine respects a ~1000 s wall‑clock budget, prints intermediate
scores and timestamps and never uses closures or file I/O.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of `a` with itself and the maximum entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` for the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _shifted_array(a: np.ndarray, shift: int) -> np.ndarray:
    """
    Return an array of length 2·len(a)‑1 that contains `a` starting at
    position `shift`.  All other entries are zero.
    """
    n = a.shape[0]
    out = np.zeros(2 * n - 1, dtype=a.dtype)
    out[shift:shift + n] = a
    return out


def _best_delta_for_pair(conv: np.ndarray, diff: np.ndarray,
                         src_mass: float, max_iter: int = 12):
    """
    Find (approximately) the δ∈[0, src_mass] that minimises
    max(conv + 2·δ·diff) by ternary search.
    Returns the best δ and the corresponding new maximum value.
    """
    lo, hi = 0.0, src_mass
    best_val = float('inf')
    best_delta = 0.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        val1 = float((conv + 2.0 * m1 * diff).max())
        val2 = float((conv + 2.0 * m2 * diff).max())
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val = val1
                best_delta = m1
        else:
            lo = m1
            if val2 < best_val:
                best_val = val2
                best_delta = m2
    # final checks at the interval ends
    end_val = float((conv + 2.0 * hi * diff).max())
    if end_val < best_val:
        best_val = end_val
        best_delta = hi
    return best_delta, best_val


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # 1. Acquire an initial feasible vector.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        total = init.sum()
        if total == 0.0:
            raise ValueError
        init = init / total
        n = init.shape[0]
        best_seq = init.copy()
        best_score = 2.0 * n * float(np.convolve(best_seq, best_seq).max())
        print(f"[{time.time() - start_time:6.2f}s] Loaded height_sequence_1 "
              f"(n={n}) → score={best_score:.6f}")
    except Exception:
        # fall back to a simple power‑law distribution (n≈5000)
        n = 5000
        p = 0.5
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a = a / a.sum()
        best_seq = a.copy()
        conv, mx = _conv_and_max(best_seq)
        best_score = 2.0 * n * mx
        print(f"[{time.time() - start_time:6.2f}s] No previous construction – "
              f"using power‑law init (n={n}) → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (gradient‑directed mass transfer).
    # ------------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1

        conv, cur_max = _conv_and_max(current)
        cur_score = 2.0 * n * cur_max

        # record any improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:5d} "
                  f"→ new best score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # a) active‑set sub‑gradient
        # ------------------------------------------------------------------
        K = _active_indices(conv, cur_max, rel_eps=0.001, max_add=30)
        g = _gradient_active(current, K)

        # ------------------------------------------------------------------
        # b) greedy move – try a few source / destination pairs
        # ------------------------------------------------------------------
        src_candidates = np.argsort(-g)[:5]            # largest gradients
        dst_candidates = np.argsort(g)[:5]             # smallest gradients
        move_made = False
        best_pair_val = cur_score
        best_pair = None
        best_delta = 0.0

        # pre‑compute shifted arrays for all candidates to avoid redundancy
        shifted_cache = {}
        for idx in set(src_candidates.tolist() + dst_candidates.tolist()):
            shifted_cache[idx] = _shifted_array(current, idx)

        for src in src_candidates:
            src_mass = float(current[src])
            if src_mass <= 0.0:
                continue
            w_src = shifted_cache[src]
            for dst in dst_candidates:
                if src == dst:
                    continue
                w_dst = shifted_cache[dst]
                diff = w_dst - w_src                         # length 2n‑1
                # ternary search for optimal delta
                delta, new_max = _best_delta_for_pair(conv, diff, src_mass)
                new_score = 2.0 * n * new_max
                if new_score < best_pair_val - 1e-12:
                    best_pair_val = new_score
                    best_pair = (src, dst)
                    best_delta = delta

        if best_pair is not None:
            src, dst = best_pair
            delta = best_delta
            if delta > 0.0:
                # perform the transfer
                current[src] -= delta
                current[dst] += delta
                move_made = True
                # continue to next iteration immediately
                continue

        # ------------------------------------------------------------------
        # c) fallback: mirror‑descent (exponentiated‑gradient) step
        # ------------------------------------------------------------------
        eta = 0.5 / math.sqrt(it)               # decreasing step size
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        log_a -= np.max(log_a)                  # stabilise numerics
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        current = a_new

    # ------------------------------------------------------------------
    # 3. Final cheap random hill‑climbing (small mass swaps).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 15000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or current[i] == 0.0:
                continue
            eps = np.random.rand() * 0.04          # up‑to 4 % move
            delta = eps * current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            conv, mx = _conv_and_max(cand)
            new_score = 2.0 * n * mx
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb "
                      f"→ score={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best vector found (as plain Python floats).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # final sanity check (optional – can be removed to save a tiny bit of time)
    try:
        # evaluate_sequence is assumed to be available in the global namespace
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Final reported score = {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
