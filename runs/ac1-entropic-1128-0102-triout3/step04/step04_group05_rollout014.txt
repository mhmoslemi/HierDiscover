"""
Proposes a step‑function (a non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm has two phases:

1. **Frank‑Wolfe‑style linear‑programming iteration**
   – Keep the vector on the L1‑sphere (∑a = √(2n)).
   – For the current vector `f` compute M = max(conv(f,f)).
   – Solve a linear program:
         maximise   ∑ g_j
         subject to  f * g ≤ M   (component‑wise)
                     g ≥ 0
     The constraints are linear because (f*g)_k = Σ_i f_i·g_{k‑i}.
   – Scale `g` to the same L1‑norm as `f` and replace
         f ← (1‑t)·f + t·g   (t slowly annealed).
   – This step never increases the maximum convolution and typically
     drives it down.

2. **Projected Adam refinement**
   – After the LP phase run a few hundred thousand Adam steps on the
     smooth‑max approximation of the objective, always projecting back
     onto the probability simplex.
   – This fine‑tunes the shape found by the LP phase.

The search is performed on several lengths (≈1800, 3000, 4000).  The
best vector encountered is returned (normalised to sum = 1, which is
compatible with `evaluate_sequence`).  The whole routine respects a
global time‑budget (default 1000 s) and prints progress information.
"""
import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full (real) linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. `a`.
    Implements g_i = 2 * Σ_k w_k * a_{k-i}
    using FFT for speed.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max version of 2·n·max(conv)
    on the simplex.  Returns (final_a, best_a_seen, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of the smooth max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

    return a, best_a, best_score


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g) = conv(f,g).
    Row k contains the coefficients f_{k‑j} for all j where 0≤k‑j<n.
    """
    n = f.shape[0]
    # row indices i+j, column indices j, data = f_i
    i = np.arange(n).reshape(-1, 1)          # (n,1)
    j = np.arange(n).reshape(1, -1)          # (1,n)
    rows = (i + j).ravel()                   # length n*n
    cols = np.tile(np.arange(n), n)          # length n*n
    data = np.repeat(f, n)                   # length n*n
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up = up / s
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step function within the allotted time.
    Uses a Frank‑Wolfe LP phase followed by a projected‑Adam refinement.
    Returns a plain list of heights (normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Length schedule – we start from a moderate size and then go larger.
    # --------------------------------------------------------------
    length_schedule = [1800, 3000, 4000]   # feel free to adjust
    for n in length_schedule:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Stage – n = {n}")

        # ----- build a pool of starting candidates -----------------
        candidates: list[np.ndarray] = []

        # up‑sample the current global best (if any)
        if best_seq is not None:
            candidates.append(_upsample(best_seq, n))

        # previous construction from the global variable, if present
        try:
            prev = np.array(height_sequence_1, dtype=float)
            candidates.append(_upsample(prev, n))
        except Exception:
            pass

        # deterministic shapes
        candidates.append(np.full(n, 1.0 / n))                     # uniform
        idx = np.arange(n)
        tri = np.minimum(idx + 1, n - idx)                      # triangular
        candidates.append(tri / tri.sum())
        if n > 1:
            hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))  # Hann window
            candidates.append(hann / hann.sum())

        # a couple of random Dirichlet starts
        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # ----------------------------------------------------------
        # Process each candidate with the LP‑based Frank‑Wolfe loop
        # ----------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)      # we keep vectors on this L1‑sphere
        for cid, cand in enumerate(candidates, 1):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0:
                cand = np.full(n, target_sum / n)
            else:
                cand = cand * (target_sum / cand.sum())

            f = cand.copy()
            step = 0.4                     # initial convex‑combination weight
            prev_max = None
            stagnation = 0
            outer_limit = 30               # can be increased if time permits

            for outer in range(outer_limit):
                if time.time() >= deadline:
                    break

                max_val, _ = _conv_max(f)

                # early‑stop if the max does not improve for a few rounds
                if prev_max is not None and max_val >= prev_max - 1e-12:
                    stagnation += 1
                    if stagnation >= 5:
                        break
                else:
                    stagnation = 0
                prev_max = max_val

                # build the linear constraints A·g ≤ max_val
                A = _build_conv_matrix(f)
                b = np.full(2 * n - 1, max_val)

                # linear programme: maximise Σ g_j  ⇔ minimise –Σ g_j
                c = -np.ones(n)
                res = opt.linprog(c, A_ub=A, b_ub=b,
                                  bounds=[(0, None)] * n,
                                  method='highs')
                if not res.success:
                    # LP failed – abort this candidate
                    break
                g = res.x
                sum_g = g.sum()
                if sum_g <= 1e-12:
                    break
                # scale g back to the same L1‑norm as f
                g = g * (target_sum / sum_g)

                # convex combination (Frank‑Wolfe update)
                f = (1.0 - step) * f + step * g
                step = max(0.05, step * 0.9)   # anneal the step size

            # ---- evaluate the candidate (normalize to sum = 1) ----
            seq = f / f.sum()
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"   NEW GLOBAL BEST (n={n}) – score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Short Adam refinement for the current length (if time permits)
        # ----------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            # crude estimate of iteration time for this n
            est_it = int(min(80000,
                             max(15000,
                                 remaining / (0.0006 * (n / 1800.0) + 1e-4))))
            if est_it > 0:
                print(f"   Adam fine‑tuning (n={n}) – {est_it} iters")
                a0 = best_seq.copy()
                a0 = a0 / a0.sum()
                _, best_a, _ = _adam_optimize(
                    a0,
                    max_iters=est_it,
                    deadline=deadline,
                    alpha=0.15,
                    beta1=0.9,
                    beta2=0.999,
                    eps=1e-8,
                    temp_start=0.02,
                    temp_end=0.0003,
                )
                try:
                    score = evaluate_sequence(best_a.tolist())
                except Exception:
                    score = float("inf")
                if score < best_score:
                    best_score = score
                    best_seq = best_a.copy()
                    print(f"   IMPROVEMENT AFTER ADAM – score = {best_score:.6f}")

    # Final long Adam refinement on the overall best sequence

    if best_seq is not None and time.time() < deadline:

                a0,
                max_iters=final_iters,
                deadline=deadline,
                alpha=0.1,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.01,
                temp_end=0.0001,
            )
            try:
                final_score = evaluate_sequence(best_a.tolist())

                    best_score = final_score
                    best_seq = best_a.copy()
                    print(f"   FINAL IMPROVEMENT – score = {best_score:.6f}")
            except Exception:
                pass

    # --------------------------------------------------------------
    # Post‑process and return
    # --------------------------------------------------------------
    if best_seq is None:
        # very safe fallback
        best_seq = np.full(100, 1.0 / 100)

    best_seq = np.maximum(best_seq, 0.0)

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {best_score:.6f}")
