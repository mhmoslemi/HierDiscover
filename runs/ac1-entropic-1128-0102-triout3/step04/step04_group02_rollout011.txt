```
def _run_subgradient_descent(a0, n, deadline, start_time,
                             max_iter=200000):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    step0 = 1.0
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2*n*max_b/(a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad improve: n={n}, it={it}, score={cur_score}")
        # find index of max conv
        idx = np.argmax(conv)
        # gradient g_i
        # g = zeros
        g = np.zeros_like(a)
        # i such that idx-i in [0,n)
        i_low = max(0, idx - (n - 1))
```

```
            if time.time() < deadline:

                    best_score = score_ref
                    best_seq = a_ref.copy()

                if score_opt < best_score:
                    best_score = score_opt
                    best_seq = a_opt.copy()
    # final mirror descent fine-tune if time left
    while time.time() < deadline - 5.0:
        n_cur = best_seq.shape[0]
        # produce small perturbation
        a0 = best_seq.copy()
        # add small Gaussian noise and renorm
        noise = (np.random.rand(n_cur) - 0.5) * 0.01
```

It starts from the previous best construction (if available) and from several random
initialisations, then runs a **projected sub‑gradient descent** that directly
optimises the true `max(conv)` (the ℓ∞ term).  After the descent a short
simulated‑annealing sweep is applied to escape possible plateaus.
The whole search is performed on a schedule of increasing lengths, always
respecting the 1000 s budget.

```python
"""
Algorithm: projected sub‑gradient descent + fast annealing

* The objective to minimise is 2·n·max(conv(a)) / (sum(a)^2).  Because the
  denominator is homogeneous we keep sum(a)=1 (simplex) throughout.
* At each iteration we evaluate the full linear convolution a*a by FFT,
  locate the index where the convolution attains its maximum and compute the
  (sub)‑gradient of this max‑value w.r.t. a.  A diminishing step size and a
  Euclidean projection onto the probability simplex keep the iterates feasible.
* The sub‑gradient phase is run on several problem sizes (500 … 4800).  The
  current best sequence is up‑sampled and used as a warm‑start for the next
  size.
* Finally a cheap simulated‑annealing sweep (mass transfer between two
  coordinates) is performed to polish the best solution.
* The routine stops just before the time budget expires, returning the best
  sequence encountered.  All numbers are non‑negative and the list is returned
  as plain Python floats.
"""

import time
import math
import numpy as np
from typing import Tuple

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be projected to the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (output length 2·n‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Scoring helper (exact, no call to evaluate_sequence)
# ----------------------------------------------------------------------


def _compute_score(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (sum(a) is not assumed =1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the simplex (max‑convolution surrogate)
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for 2·n·max(conv)/sum²."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _compute_score(a)

    step0 = 1.0  # initial (large) step – will be scaled down as 1/√t

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (a.sum() * a.sum())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, it={it}, score={cur_score:.6f}")

        # index where the convolution attains its maximum
        idx = int(conv.argmax())

        # gradient of the max‑convolution w.r.t. a
        g = np.zeros_like(a)
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        inds = np.arange(low, high + 1)
        g[inds] = 2.0 * a[idx - inds]

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a -= step * g
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing sweep (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> Tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = _compute_score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _compute_score(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random FFT‑based initialisation (a = |Re(FFT⁻¹(random phases))|)
# ----------------------------------------------------------------------


def _init_random_fft(n: int) -> np.ndarray:
    """Generate a non‑negative vector of length n via a random FFT sketch."""
    # random complex spectrum with conjugate symmetry to obtain a real signal
    rng = np.random.default_rng()
    # generate half spectrum (rfft) with unit magnitude and random phase
    phases = rng.uniform(0.0, 2.0 * np.pi, n // 2 + 1)
    spec = np.exp(1j * phases)
    # enforce real DC and Nyquist components
    spec[0] = rng.uniform(0.5, 1.5)  # keep them real and positive
    if n % 2 == 0:
        spec[-1] = rng.uniform(0.5, 1.5)
    # inverse FFT to obtain a real sequence
    a = np.fft.irfft(spec, n=n)
    a = np.abs(a)  # discard sign
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a
```
