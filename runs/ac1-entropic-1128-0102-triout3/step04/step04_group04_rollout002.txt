def grad_smooth(a, w):
    n = a.shape[0]
    L = 1 << ((3*n-2).bit_length())
    Af = np.fft.rfft(a, n=L)
    w_rev = w[::-1]
    Wf = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)[:3*n-2]
    offset = 2*n-2
    g = 2.0 * conv[offset - np.arange(n)]

"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
* **Multiple problem sizes** – try a handful of lengths (≈3500…5000) to
  exploit the fact that `evaluate_sequence` contains the factor `n`.
* **Rich initialisation** – for each size we start from several seeds:
  the previous best construction (if available), a few random Dirichlet
  draws and simple power‑law shapes.
* **Smooth‑max gradient descent** – we minimise a differentiable
  approximation of `max(conv(a,a))`:
      Φₐ(α) = (1/α)·log Σₖ exp(α·(a*a)ₖ) .
  Its gradient w.r.t. `a` is a weighted self‑convolution that can be
  computed with two FFTs.  After each gradient step we project onto the
  probability simplex.
  The smoothing parameter `α` is increased geometrically (0.5 → 100) so
  that the optimisation first moves easily (small α) and later focuses
  on the true maximum (large α).
* **Periodic random jitter** – every few hundred iterations a small
  random perturbation is added to escape possible plateaus.
* **Final cheap hill‑climb** – after the time budget expires we perform
  many tiny random mass‑transfer moves and keep any improvement.
* All candidates are kept on the simplex (sum = 1) – this makes the
  denominator of `evaluate_sequence` equal to 1, so the objective that
  we actually minimise is `2·n·max(conv)`.

The routine respects the ~1000 s wall‑clock budget and prints progress.
If no valid construction is found a uniform distribution of length 100
is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_and_max(a: np.ndarray):
    """Return (full convolution, its maximum)."""
    conv = _conv_self(a)
    return conv, conv.max()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                         # all negative → uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths to explore.
    # ------------------------------------------------------------------
    # If a previous construction exists we reuse its length; otherwise
    # we explore a few candidates around 5000.
    lengths = []
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=float)
            lengths.append(prev.shape[0])
        except Exception:
            pass
    # add a small grid of lengths (all ≤ 6000 to keep FFT cheap)
    lengths.extend([3500, 4000, 4500, 5000])
    lengths = sorted(set(lengths))

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 2. Outer loop over lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Exploring length n={n}")

        # --------------------------------------------------------------
        #   2.1 Build a list of initial vectors for this length.
        # --------------------------------------------------------------
        init_vectors = []

        # a) previous construction (if same length)
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=float)
                if prev.shape[0] == n:
                    a = np.maximum(prev, 0.0)
                    s = a.sum()
                    if s > 0:
                        init_vectors.append(a / s)
            except Exception:
                pass

        # b) a few random Dirichlet draws
        for _ in range(3):
            a = np.random.rand(n)
            a /= a.sum()
            init_vectors.append(a)

        # c) simple power‑law families
        for p in (0.3, 0.5, 0.7):
            idx = np.arange(1, n + 1, dtype=float)
            a = idx ** (-p)
            a /= a.sum()
            init_vectors.append(a)

        # d) a uniform vector (as a baseline)
        init_vectors.append(np.full(n, 1.0 / n))

        # --------------------------------------------------------------
        #   2.2 Optimise each initial vector.
        # --------------------------------------------------------------
        for start_vec in init_vectors:
            if time.time() > deadline:
                break
            a = start_vec.copy()
            it_global = 0

            # schedule of smoothing parameters (increase gradually)
            alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]

            # pre‑compute the FFT length needed for the gradient step
            L_grad = 1 << ((3 * n - 2).bit_length())

            for alpha in alphas:
                if time.time() > deadline:

                # inner loop – gradient descent with projection
                for _ in range(500):               # a few hundred per α
                    if time.time() > deadline:

                    # ----- 1) evaluate current objective -----
                    conv, max_val = _conv_and_max(a)
                    cur_score = 2.0 * n * max_val

                        best_n = n

                    exp_arg = alpha * conv
                    exp_arg -= exp_arg.max()                # for stability
                    w = np.exp(exp_arg)
                    w /= w.sum()                            # softmax

                    # ----- 3) gradient of smooth objective -----
                    # g_i = 2 * Σ_k w_k * a_{k-i}  (cross‑correlation)
                    Af = np.fft.rfft(a, n=L_grad)
                    w_rev = w[::-1]
                    Wf = np.fft.rfft(w_rev, n=L_grad)
                    cross = np.fft.irfft(Af * Wf, n=L_grad)[: 3 * n - 2]
                    offset = 2 * n - 2
                    g = 2.0 * cross[offset - np.arange(n)]

                    # ----- 4) gradient step with decaying stepsize -----
                    it_global += 1
                    eta = 0.6 / math.sqrt(it_global)       # decreasing stepsize
                    a_new = a - eta * g
                    a = _project_simplex(a_new)

                    # ----- 5) occasional random jitter to escape plateaus -----
                    if it_global % 300 == 0:
                        jitter = 0.02 * np.random.randn(n)   # 2 % Gaussian noise
                        a_j = a + jitter
                        a_j = np.maximum(a_j, 0.0)
                        s = a_j.sum()
                        if s > 0:
                            a = a_j / s

            # end of α‑schedule
        # end of initial‑vector loop
    # end of length loop

    # ------------------------------------------------------------------
    # 3. Final cheap random hill‑climbing around the best vector found.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        for _ in range(15000):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # move a small fraction of mass from i to j
            eps = np.random.rand() * 0.04               # up to 4 %
            delta = eps * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # evaluate
            conv, max_val = _conv_and_max(cand)
            cand_score = 2.0 * n * max_val
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"random hill‑climb → score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence (as plain Python floats).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of modest size
        n = 200
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
