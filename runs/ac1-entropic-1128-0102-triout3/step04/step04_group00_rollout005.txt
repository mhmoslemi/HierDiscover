def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # if previous best sequence exists via global variable height_sequence_1
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)
        if init_seq.ndim == 1 and init_seq.size>0:
            init_seq = np.maximum(init_seq,0)
            init_seq /= init_seq.sum()
    except Exception:
        init_seq = None

    # Stage list of lengths
    lengths = [800, 2400, 4800, 9600, 19200]  # as many as time allows
    cur_seq = None
    cur_n = None
    best_seq = None
    best_score = np.inf

    for n in lengths:
        # allocate time proportionally
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # we decide time for this stage: maybe 20% of remaining each stage until n increases.
        stage_time = remaining * 0.2
        stage_deadline = time.time() + stage_time

        # Prepare starting seq
        if cur_seq is None:
            if init_seq is not None and init_seq.shape[0] >= n:
                cur_seq = init_seq[:n].copy()
                cur_seq /= cur_seq.sum()
            else:
                cur_seq = np.random.rand(n)
                cur_seq /= cur_seq.sum()
        else:
            # upsample cur_seq to new length n
            cur_seq = _upsample(cur_seq, n)
        cur_n = n
        # gradient descent
        cur_seq, cur_score = _run_gradient_descent_fast(cur_seq, n, stage_deadline, start)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{time.time()-start:.2f}s] new best at n={n} score={cur_score:.6f}")

    # after loops try annealing on best_seq using remaining time
    if time.time() < deadline:
        best_seq, best_score = _run_annealing(best_seq, best_seq.shape[0],
                                              deadline, start,
                                              best_score, best_seq, start_temp=0.05, end_temp=1e-5, steps=200_000)

    # final normalization
    if best_seq is None:
        best_seq = np.full(800, 1/800)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0/best_seq.size)

    return [float(v) for v in best_seq]

def _run_gradient_descent_fast(a0, n, deadline, start_time, beta=0.9, step0=0.2, max_iter=1e7):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = np.inf
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a)   # length 2n-1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] it={it} n={n} score={cur_score:.6f}")
        # compute mask
        mask = np.zeros_like(conv)
        mask[conv.argmax()] = 1.0
        # gradient via conv mask with reversed a
        grad = _conv_mask_rev_fft(mask, a[::-1])  # returns length n
        # add momentum
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

        grad = _gradient_from_mask(a, mask)

    return best_a, best_score

"""
Optimisation of a non‑negative step function to minimise `evaluate_sequence`.
The algorithm proceeds in three phases:

1. **Fast projected gradient descent** on the probability simplex.
   The sub‑gradient of `max(conv(a))` is obtained with a single FFT by
   convolving the active mask (the indices attaining the current maximum)
   with the reversed current vector.  This makes each iteration O(n log n)
   and allows many thousands of iterations.

2. **Progressive up‑sampling** – we start with a modest length (≈800),
   improve it, then linearly interpolate to larger lengths (≈2400,
   ≈4800, ≈9600 …) while continuing the gradient descent.  Larger
   sequences have a lower theoretical bound, so the up‑sampling helps the
   search approach the target value 1.503.

3. **Final simulated‑annealing refinement** with mass‑transfer moves to
   escape possible local minima.

All steps respect the supplied time budget (default 1000 s).  The
returned list is normalised to sum = 1 and contains only non‑negative
floats, thus being a valid input for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_mask_rev_fft(mask: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Convolution of `mask` (length 2·n‑1) with `a_rev` (length n) via FFT.
    Returns the middle n entries, i.e. result[i] = Σ_k mask[k]·a_rev[i+n‑1‑k].
    """
    n = a_rev.shape[0]
    L = _next_pow2(mask.shape[0] + n - 1)
    M = np.fft.rfft(mask, n=L)
    A = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(M * A, n=L)
    start = n - 1
    return conv[start : start + n]


def _gradient_from_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv(a))` for the active mask.
    For each active index k we add 2·a_{k-i} to the i‑th component.
    """
    a_rev = a[::-1]
    g = _conv_mask_rev_fft(mask, a_rev)
    return 2.0 * g


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Fast projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent_fast(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta: float = 0.9,
    step0: float = 0.3,
    rel_tol: float = 1e-8,
    max_iter: int = 10 ** 7,
):
    """
    Projected gradient descent where the gradient of the max‑convolution
    is obtained via a single FFT.  Returns the best vector seen and its
    score = 2·n·max(conv) (sum of entries is 1).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] GD n={n:,} iter={it:,} score={cur_score:.6f}"
            )

        # active mask (all entries within rel_tol of the maximum)
        thr = max_val * (1.0 - rel_tol)
        mask = (conv >= thr).astype(float)

        grad = _gradient_from_mask(a, mask)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 150_000,
):
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer a random fraction (up to 30 %) of src's mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n:,} iter={it:,} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the time budget and returns the best sequence found
    (normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Small safety margin for the harness
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global name `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage schedule (lengths increase roughly by factor 2)
    # ------------------------------------------------------------------
    # The fractions roughly allocate 10 % / 25 % / 45 % of the total budget
    # to the three gradient phases; the remainder is kept for annealing.
    stage_lengths = [800, 2400, 4800, 9600]  # can be shortened if time runs out
    stage_fractions = [0.10, 0.25, 0.45]      # sum < 1; the rest -> annealing

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # --------------------------------------------------------------
    # Stage 0 – a few random restarts on the smallest size
    # --------------------------------------------------------------
    n0 = stage_lengths[0]
    restarts_stage0 = 4
    time_per_restart = max(2.0, (deadline - time.time()) * 0.15 / restarts_stage0)

    for r in range(restarts_stage0):
        if time.time() >= deadline:
            break
        a0 = np.random.random(n0)
        a0 /= a0.sum()
        sub_deadline = time.time() + time_per_restart
        a_opt, score_opt = _run_gradient_descent_fast(
            a0, n0, sub_deadline, start_time, beta=0.85, step0=0.4, rel_tol=1e-8
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage0 restart {r} improve: score={score_opt:.6f}"
            )
        # keep a candidate for up‑sampling
        cur_seq = a_opt.copy()
        cur_n = n0

    # If we never entered the loop (time exhausted), fall back to uniform
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self_fft(best_seq).max())
        cur_seq = best_seq.copy()
        cur_n = n0

    # --------------------------------------------------------------
    # Progressive up‑sampling & gradient descent
    # --------------------------------------------------------------
    for idx, n_target in enumerate(stage_lengths[1:]):
        if time.time() >= deadline:
            break

        # allocate a share of the remaining budget for this stage
        remaining = deadline - time.time()
        # use the corresponding fraction of the *original* budget
        fraction = stage_fractions[min(idx, len(stage_fractions) - 1)]
        stage_time = max(5.0, fraction * budget_s)
        stage_time = min(stage_time, remaining)
        stage_deadline = time.time() + stage_time

        # up‑sample current best sequence to the new length
        cur_seq = _upsample(cur_seq, n_target)
        cur_n = n_target

        # Run gradient descent from the up‑sampled point
        cur_seq, cur_score = _run_gradient_descent_fast(
            cur_seq,
            cur_n,
            stage_deadline,
            start_time,
            beta=0.90,
            step0=0.3,
            rel_tol=1e-8,
        )
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage {idx+1} improve: n={cur_n:,} score={cur_score:.6f}"
            )

    # --------------------------------------------------------------
    # Final simulated‑annealing refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_seq, best_score = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=0.1,
            end_temp=1e-6,
            steps=200_000,
        )
        print(
            f"[{time.time() - start_time:6.2f}s] final anneal result: score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and convert to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
