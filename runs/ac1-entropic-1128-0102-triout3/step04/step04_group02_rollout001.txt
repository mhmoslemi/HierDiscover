            n = starter.size
        else:
            raise ValueError
    except Exception:
        # Default length – large enough to be representative but still cheap.
        n = 2000

    # Build the uniform distribution.
    uniform = np.full(n, 1.0 / n, dtype=float)

    # Quick sanity check (optional, can be removed for speed).
    try:
        # The harness will call evaluate_sequence later; we can log the score.
        # Import is assumed to be provided by the harness.
        from __main__ import evaluate_sequence  # type: ignore
        score = evaluate_sequence(uniform.tolist())
        print(f"[propose_candidate] uniform length={n}, evaluate_sequence={score:.6f}")
    except Exception:
        # If evaluate_sequence is not importable (e.g., during static analysis)
        # we simply skip the check.
        pass

    # Return a plain Python list as required by the outer harness.
    return uniform.tolist()
