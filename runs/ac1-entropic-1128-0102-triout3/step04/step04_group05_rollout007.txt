"""
Propose a low‑score step‑function using a multi‑scale Adam optimiser
with monotone‑simplex projection and a rich set of random initial
shapes.  The optimiser works on the probability simplex (sum = 1,
a_i ≥ 0) because the evaluation function is scale‑invariant.
After every Adam step we enforce a non‑increasing ordering (isotonic
projection) by sorting the vector and then projecting onto the simplex.
A schedule of problem sizes (≈1.5× growth) is processed; for each size
a small pool of diverse candidates (uniform, triangular, Hann,
geometric, power‑law, Dirichlet with random concentration and the
current global best up‑sampled) is optimised.  The best sequence found
over all sizes is finally refined with a long Adam run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full (real) convolution.
    a is assumed to sum to 1, but the function works for any non‑negative a."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a))
    w.r.t. a. Implements g = 2 * (w * a_rev) convolution.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # length of conv2 is len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize_monotonic(a0: np.ndarray,
                             max_iters: int,
                             deadline: float,
                             alpha: float = 0.2,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             temp_start: float = 0.05,
                             temp_end: float = 0.0005) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser on the simplex with monotone (non‑increasing) projection.
    Returns (final_a, best_seen_a, best_score).
    The objective is 2·n·max(conv(a,a)) (sum(a) is 1).
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- monotone projection -----
        # enforce non‑increasing order by sorting descending,
        # then project onto the probability simplex.
        a = np.sort(a)[::-1]                     # descending
        a = _project_simplex(a)                  # sum = 1, non‑negative

    return a, best_a, best_score

def _generate_candidates(n: int, best_seq: np.ndarray | None,
                         prev_seq: np.ndarray | None,
                         rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small, diverse pool of start vectors for a given length `n`.
    All candidates are normalised to sum 1.
    """
    cand = []

    # 1) up‑sample current global best (if length differs)
    if best_seq is not None and best_seq.shape[0] != n:
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, best_seq)
        up = np.maximum(up, 0.0)
        up = up / up.sum()
        cand.append(up)

    # 2) previous construction (if any)
    if prev_seq is not None:
        if prev_seq.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, prev_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, prev_seq)
            up = np.maximum(up, 0.0)
            up = up / up.sum()
            cand.append(up)
        else:
            cand.append(prev_seq.copy())

    # 3) deterministic baseline shapes
    cand.append(np.full(n, 1.0 / n))          # uniform

    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)      # triangular (mountain)
    cand.append(tri / tri.sum())
    if n > 1:
        hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))  # raised‑cosine
        cand.append(hann / hann.sum())
    else:
        cand.append(np.full(1, 1.0))

    # 4) geometric decreasing (random rate)
    for _ in range(2):
        r = rng.uniform(0.3, 0.99)
        geo = r ** np.arange(n, dtype=float)
        geo = geo / geo.sum()
        cand.append(geo)

    # 5) power‑law decreasing (random exponent)
    for _ in range(2):
        p = rng.uniform(0.5, 3.0)
        pl = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
        pl = pl / pl.sum()
        cand.append(pl)

    # 6) Dirichlet draws with random concentration
    for _ in range(4):
        alpha = rng.uniform(0.1, 5.0)
        dirich = rng.dirichlet(np.full(n, alpha))
        cand.append(dirich)

    # 7) perturbations of the current best (if we have one of the same length)
    if best_seq is not None and best_seq.shape[0] == n:
        for _ in range(3):
            noise = rng.normal(scale=0.02, size=n)
            pert = best_seq + noise
            pert = np.maximum(pert, 0.0)
            if pert.sum() == 0:
                pert = np.full(n, 1.0 / n)
            else:
                pert = pert / pert.sum()
            cand.append(pert)

    # Remove duplicates (exact equality unlikely but cheap)
    uniq = []
    seen = set()
    for c in cand:
        key = tuple(np.round(c, 12))
        if key not in seen:
            seen.add(key)
            uniq.append(c)
    return uniq

    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(400000, max(80000, remaining / est_iter_time)))
        if final_iters > 0:
            print(
                f"[{time.time() - start_time:6.2f}s] Final refinement – n={n}, iters={final_iters}"
            )
            _, local_best, local_score = _adam_optimize_monotonic(
                best_seq,
                max_iters=final_iters,
                deadline=deadline,
                alpha=0.1,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.01,
                temp_end=0.0001,
            )
            if local_score < best_score:
                best_score = local_score
                best_seq = local_best.copy()
                print(f"   FINAL IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. post‑process and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: modest uniform vector
        best_seq = np.full(200, 1.0 / 200)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    total_time = time.time() - start_time
    print(
        f"[{total_time:6.2f}s] Finished – best 2·n·max(conv) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
