"""
Enhanced random‑support + simulated‑annealing search for a step function
with a low autocorrelation bound.

The evaluation function depends only on
        score = 2·n·max (a*a) / (∑a)² .
For a support mask `M` (M_i∈{0,1}) with uniform weights a_i = M_i / |M|
the denominator is 1 and the score reduces to

        score = 2·n·max_count / |M|²

where `max_count` is the largest number of ordered pairs (i,j)∈M×M whose
indices sum to the same value.  Random masks with a large support (≈0.8 n…0.95 n)
have `max_count` close to its average, giving an expected score near 1.0.
Therefore we concentrate the search on binary masks and use a cheap
simulated‑annealing (swap‑random‑index) optimiser to minimise the score.
If the time budget remains, a projected sub‑gradient optimiser refines the
real‑valued weights on the best mask found.

The whole routine respects the 1000 s limit, prints progress and returns
a list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# fast evaluation helpers (no safety checks)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for a real‑valued vector.
    The L1 norm is *not* normalised; the function returns the exact
    value used by ``evaluate_sequence`` (apart from the clipping steps).
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def fast_score_mask(mask: np.ndarray, n: int, size_fft: int) -> float:
    """
    Evaluate the score for a *binary* mask with uniform weights.
    mask[i] ∈ {0,1}; the actual coefficients are mask/m where m = mask.sum().
    """
    m = mask.sum()
    if m == 0:
        return math.inf
    fft = np.fft.rfft(mask, size_fft)
    conv = np.fft.irfft(fft * fft, size_fft)[: 2 * n - 1]
    max_c = conv.max()
    return float(2.0 * n * max_c / (m * m))

    while time.time() < deadline:
        # --- compute active index of the convolution ---
        fft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # --- sub‑gradient of the active entry (with respect to a) ---
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # --- step size (simple decay) ---
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # --- Euclidean projection onto the simplex (L1‑norm = 1) ---
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # --- occasional random mass‑transfer to escape plateaux ---
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # re‑project
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # --- occasional logging ---
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")

        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a coefficient vector whose evaluation score is ≤ 1.503.
    Strategy:
      1. If a previous construction (global ``height_sequence_1``) exists,
         initialise the best candidate with it.
      2. Random‑support search with large support sizes (≈ 0.8 n … 0.95 n) and
         a very cheap simulated‑annealing swap scheme.  The score for a mask
         is computed analytically via FFT.
      3. If the time budget remains and the target has not been reached,
         run the projected sub‑gradient optimiser starting from the best
         uniform‑weight mask discovered.
    The function respects the global time limit and prints progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin
    n = 1500
    size_fft = 1 << ((2 * n - 1).bit_length())
    target_score = 1.503

    best_seq = None          # will hold the *real* coefficient vector
    best_val = math.inf

    # ------------------------------------------------------------------
    # 0. start from a previously stored construction, if any
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size == n:
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random‑support + Simulated‑Annealing search
    #    We keep the support size *fixed* inside each SA run (swap moves).
    #    The temperature decays slowly; acceptance of uphill moves allows
    #    exploration of the combinatorial landscape.
    attempts = 0
    # give about the first 80 % of the budget to this phase
    anneal_deadline = deadline - 20.0
    while time.time() < anneal_deadline:
        # pick a support size (bias towards larger supports)
        m = np.random.randint(1200, 1501)          # 1200 ≤ m ≤ 1500
        # initialise a random mask of that size
        idx = np.random.choice(n, size=m, replace=False)
        mask = np.zeros(n, dtype=np.float64)
        mask[idx] = 1.0

        cur_score = fast_score_mask(mask, n, size_fft)
        cur_mask = mask.copy()
        cur_support = list(idx)                     # list of positions with 1
        # complement list (positions with 0)
        complement = [i for i in range(n) if cur_mask[i] == 0]
        # temperature schedule
        temp = 0.05
        decay = 0.99996
        # number of swaps for this run – adapts to remaining time
        moves = 4000
        for _ in range(moves):
            if time.time() >= anneal_deadline:
                break
            # pick a random element to remove and a random element to add
            i_pos = np.random.randint(len(cur_support))
            j_pos = np.random.randint(len(complement))
            i = cur_support[i_pos]
            j = complement[j_pos]

            # propose the swap
            cur_mask[i] = 0.0
            cur_mask[j] = 1.0
            new_score = fast_score_mask(cur_mask, n, size_fft)
            delta = new_score - cur_score

            # Metropolis acceptance
            if delta < 0 or np.random.rand() < math.exp(-delta / temp):
                # accept
                cur_score = new_score
                cur_support[i_pos] = j
                complement[j_pos] = i
                if cur_score < best_val:
                    best_val = cur_score
                    best_seq = cur_mask.copy()          # still a mask
                    print(f"[SA] new best {best_val:.6f} (m={m})")
                if cur_score <= target_score:
                    print("[SA] target reached – returning immediately")
                    a = cur_mask / cur_mask.sum()
                    return a.tolist()
            else:
                # reject – revert the swap
                cur_mask[i] = 1.0
                cur_mask[j] = 0.0

            # temperature decay
            temp *= decay

        attempts += 1
        if attempts % 20 == 0:
            print(f"[SA phase] attempts={attempts}, best so far {best_val:.6f}")

    print(f"[random‑support] finished after {attempts} restarts, best score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. If no mask has been found yet, fall back to a uniform vector
    if best_seq is None or best_seq.sum() == 0:
        a_init = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # convert the best binary mask into a uniform‑weight vector
        a_init = best_seq / best_seq.sum()

    # ------------------------------------------------------------------
    # 3. Fallback: projected sub‑gradient optimisation on the real vector
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[fallback] running sub‑gradient optimiser (≈{remaining:.1f}s left)")
        a_opt = subgradient_optimize(a_init, remaining)
        final_val = fast_score(a_opt)
        print(f"[fallback] final score {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_opt.copy()
    else:
        best_seq = a_init.copy()
        best_val = fast_score(best_seq)

    print(f"[final] returning sequence with score {best_val:.6f}")
    return best_seq.tolist()
