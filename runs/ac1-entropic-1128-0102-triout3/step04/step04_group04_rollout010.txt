"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm outline
-----------------
1. Choose several problem sizes (including the length of the previously
   discovered construction if it exists).  For each size we start from a
   simple power‑law shape and optionally a few random Dirichlet draws.
2. Perform a fast projected‑gradient descent on the objective
   ``score(a)=2·n·max(conv(a,a))`` (where `a` is kept on the probability
   simplex).  The sub‑gradient of the ``max`` is obtained from an
   active set of near‑maximal convolution entries.
3. Every few hundred gradient steps (only for modest lengths ``n≤1500``)
   we solve a small linear programme

        max   Σ b_j
        s.t.  conv(f,b) ≤ M   (entry‑wise) ,   b≥0

   where `f` is the current iterate and `M` its maximal convolution
   value.  The solution `b` is projected back onto the simplex and mixed
   with `f`.  This LP step reproduces the “alternating linear‑program
   update” described in the state‑of‑the‑art paper and usually yields a
   noticeable drop in the objective.
4. After the gradient/LP phase we finish with cheap random hill‑climbing
   moves (small mass swaps) until the overall time budget is exhausted.
5. The best sequence encountered (across all tested lengths) is
   returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps·max_val are kept;
    if fewer than `max_add` are found, the globally largest entries
    are added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of the ``max(conv)`` for the active set `K`.
    For each active convolution index k the contribution to component i
    is 2·a_{k-i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    (non‑negative entries summing to 1).
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _lp_step(f: np.ndarray, M: float):
    """
    Solve the LP:

        max   Σ b_j
        s.t.  conv(f, b) ≤ M   (entry‑wise),  b ≥ 0

    Returns the optimal vector `b`.  The LP is solved with SciPy's
    HiGHS interior‑point method using a sparse constraint matrix.
    """
    n = f.shape[0]
    rows, cols, data = [], [], []

    # Build the (2n‑1)×n constraint matrix.
    for k in range(2 * n - 1):
        j_low = max(0, k - (n - 1))
        j_high = min(n - 1, k)
        # coefficient of b_j in row k is f_{k‑j}
        for j in range(j_low, j_high + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])

    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                     # minimise -Σ b  →  maximise Σ b
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    return res.x

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Helper to obtain a starting vector for a given length n.
    # ------------------------------------------------------------------
    def get_initial_vector(n):
        # try to reuse the globally stored construction if it matches size
        global_seq = globals().get("height_sequence_1")

            arr = np.array(global_seq, dtype=float)
            arr = np.maximum(arr, 0.0)
            if arr.size == n:
                arr /= arr.sum()
                return arr
            if arr.size > n:
                # truncate and renormalise
                arr = arr[:n]
                arr /= arr.sum()
                return arr
        # fallback: simple power‑law with a modest exponent grid
        best = None
        best_sc = float("inf")

        # also try a few random Dirichlet draws (small concentration → sparse)

                    pass

            eta = 0.5 / math.sqrt(it)      # decaying step size

    # ------------------------------------------------------------------
