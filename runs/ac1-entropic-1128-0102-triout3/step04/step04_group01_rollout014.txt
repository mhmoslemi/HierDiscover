def _run_adam_descent(a0, n, slice_deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 1.0  # base learning rate
    lam_factor = 0.3
    lam_decay = 0.7
    lam_decay_interval = 2000
    it = 0
    best_a = a.copy()
    best_score = float('inf')
    while time.time() < slice_deadline:
        it += 1
        # compute conv self
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")
        # adjust lambda
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-12)
        # compute weights
        if lam > max_b * 1e-4:
            # log-sum-exp gradient
            weights = np.exp((conv - max_b) / lam)
            weights /= np.sum(weights)
        else:
            # max indicator
            eps_ind = max_b - 1e-12
            mask = conv >= eps_ind
            cnt = mask.sum()
            if cnt == 0:  # fallback
                weights = np.ones_like(conv) / conv.size
            else:
                weights = mask.astype(float) / cnt
        # compute gradient
        a_rev = a[::-1]
        full = _conv_fft(weights, a_rev)
        start = n - 1
        grad = 2.0 * full[start:start + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / math.sqrt(it)  # decreasing schedule
        a = a - step * (m_hat / (np.sqrt(v_hat) + epsilon))
        # projection
        a = _project_simplex(a)
    return best_a, best_score
```

def _run_targeted_swap(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    while time.time() < deadline:
        conv = _conv_self(a)
        max_b = conv.max()
        idx_max = np.argmax(conv)
        # compute contributions
        contributions = np.zeros(n)
        for i in range(n):
            j = idx_max - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        # source index with highest contribution
        src = np.argmax(contributions)
        # select destination index where a is minimal
        dst = np.argmin(a)
        if src == dst:
            # choose random other
            dst = random.randint(0, n-1)
            if dst == src:
                continue
        # move epsilon
        eps = a[src] * 0.1 * random.random()  # up to 10% of src
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        new_score = 2.0 * n * _conv_self(a_new).max()

"""
Propose a low‑maximum self‑convolution step‑function.

Algorithm
---------

* Work on the probability simplex (∑aᵢ = 1) – the objective
  `evaluate_sequence` is scale‑invariant.
* For each problem size `n` we run a projected Adam optimiser that
  minimises a smooth surrogate of the ℓ∞‑norm of the self‑convolution.
  The surrogate is a log‑sum‑exp with a temperature λ that is
  geometrically decreased; when λ becomes very small we fall back to the
  exact sub‑gradient of the max‑operator.
* The iterates are projected onto the simplex after every Adam update.
* After the Adam phase we perform a cheap, directed “targeted swap”
  local search that moves a small amount of mass from indices that
  contribute most to the current maximal convolution entry to the
  currently lightest index.  Improvements are accepted greedily.
* Finally a short simulated‑annealing polishing phase (mass‑transfer)
  is run.
* The best sequence seen across all lengths (a geometric schedule up to
  16384 entries) is returned.

The whole routine respects the 1000 s budget by slicing the remaining
time among the stages and aborts cleanly when the deadline is reached.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a via a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y via FFT."""
    size = x.size + y.size - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient descent with Adam on a smooth max surrogate
# ----------------------------------------------------------------------


def _run_adam_descent(a0: np.ndarray,
                      n: int,
                      slice_deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises 2·n·max(a*a).
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 1.0

    # λ‑schedule (smooth‑max temperature)
    lam_factor = 0.30
    lam_decay = 0.70
    lam_decay_interval = 2000

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < slice_deadline:
        it += 1

        # current convolution & objective
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-12)

        # ---- gradient of the smooth max ---------------------------------
        if lam > max_b * 1e-4:
            # soft‑max (log‑sum‑exp) weights
            w = np.exp((conv - max_b) / lam)
            w /= w.sum()
        else:
            # exact sub‑gradient: uniform weight on all maximal entries
            eps_ind = 1e-12
            mask = conv >= (max_b - eps_ind)
            cnt = mask.sum()
            if cnt == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w = mask.astype(float) / cnt

        # gradient w.r.t. a (see derivation in analysis)
        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)          # length = 3·n‑2
        start = n - 1
        grad = 2.0 * full[start:start + n]

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / math.sqrt(it)          # slowly decaying step size
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted mass‑transfer local search (greedy)
# ----------------------------------------------------------------------


def _run_targeted_swap(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       cur_best_score: float,
                       cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Greedy local search that repeatedly moves a small ε from the index
    that contributes most to the current maximal convolution entry to the
    smallest‑mass index.  Accepts only improving moves.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    while time.time() < deadline:
        conv = _conv_self(a)
        max_b = conv.max()
        idx_max = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        contrib = np.zeros(n)
        for i in range(n):
            j = idx_max - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        src = int(np.argmax(contrib))
        if a[src] <= 0.0:
            # nothing to move – pick a random source
            src = random.randint(0, n - 1)
            if a[src] <= 0.0:
                continue

        # destination: index with smallest current mass (avoid src)
        dst = int(np.argmin(a))
        if dst == src:
            # pick a random different index
            dst = random.randint(0, n - 1)
            if dst == src:
                continue

        eps = a[src] * random.random() * 0.15   # up to 15 % of source mass
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] targeted improve n={n} score={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing polishing (same idea as before)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30000) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing: move a small ε from a source index to a
    destination index.  Accepts worsening moves according to classic
    Metropolis rule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * random.random() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > random.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time budget and
    returns the best sequence (as a plain Python list) that it has found.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – modest start, ramp up to ~16 k.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)   # avoid systematic bias when stopping early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm‑start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # larger lengths receive a slightly larger share
        fraction = 0.15 / (len(lengths) - idx)
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # --------------------------------------------------------------
        # Adam‑based gradient descent
        # --------------------------------------------------------------
        a_opt, score_opt = _run_adam_descent(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve (Adam) n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Targeted greedy local search (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            target_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_target, score_target = _run_targeted_swap(
                a_opt,
                n,
                target_deadline,
                start_time,
                best_score,
                best_seq,
            )
            if score_target < best_score:
                best_score = score_target
                best_seq = a_target.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve (targeted) n={n} score={score_target:.6f}")

        # --------------------------------------------------------------
        # Final annealing polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.1)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve (anneal) n={n} score={score_ref:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256.0)

    # Return plain Python list (the external evaluator will clamp / normalise)
    return [float(x) for x in best_seq]
