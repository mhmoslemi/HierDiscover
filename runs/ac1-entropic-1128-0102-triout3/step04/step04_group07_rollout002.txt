"""
Propose a low‑autocorrelation step function.

Strategy
--------
1. **Random‑support phase** – The score is scale‑invariant, therefore we fix
   the ℓ₁‑norm to 1 (uniform weight 1/m on a chosen support S).
   For a large random support the expected value of the score is
   close to 1, well below the target 1.503.
   We try many random supports (sizes ≈ 0.8‒0.95 · n) until the budget runs
   out or a satisfactory sequence is found.

2. **Local swap improvement** – Starting from the best support found,
   we repeatedly swap a support element with a non‑support element
   (keeping the weight uniform).  Swaps that improve the score are kept;
   this hill‑climbing usually reduces the maximal convolution further.

3. **Fallback projected sub‑gradient descent** – If the random search
   does not reach the target, we polish the current best vector (or a
   uniform vector) with a simple projected sub‑gradient method that
   directly minimizes the maximal convolution entry.

All heavy work uses a fast FFT‑based evaluator `fast_score` (identical to
`evaluate_sequence` except for the safety checks).  The whole routine
keeps track of the remaining time and respects the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluator (identical to the safety‑checked version but without checks)
def fast_score(a: np.ndarray) -> float:
    """Return the objective value 2·n·max(conv) / (sum(a))²."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    # next power of two for FFT (>= 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (fallback)
def _euclidean_simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Simple projected sub‑gradient descent for the max‑convolution objective.
    The algorithm follows the outline from the reference implementation.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30          # initial step size
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # current convolution and arg‑max index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry (2*a shifted)
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size schedule
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = _euclidean_simplex_projection(v)

        # occasional random mass transfer to help escaping plateaus
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _euclidean_simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # periodic logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")

        it += 1

    # final log
    final_val = fast_score(best_a)
    print(f"[subgrad final] best value {final_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with evaluation ≤ 1.503.

    The routine consists of three phases:
      1) massive random‑support sampling (large supports first);
      2) local swap‑based hill climbing on the best support;
      3) optional sub‑gradient polishing if the target is not yet met.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # parameters
    n = 1500                                   # length of the vector
    target_score = 1.503

    # Try to continue from a previously stored construction, if any.
    best_seq = None
    best_val = math.inf
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)   # type: ignore
        if prev.size == n:
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random‑support search
    # Use large supports first; the larger the support, the smaller the expected score.
    support_fractions = [0.95, 0.92, 0.90, 0.87, 0.84, 0.80]   # relative to n
    # Decide how many attempts we can afford overall.
    # Rough budget: ~0.6 · budget for this phase.
    random_phase_budget = (deadline - time.time()) * 0.6
    phase_end = time.time() + random_phase_budget

    for frac in support_fractions:
        if time.time() >= phase_end:
            break
        m = int(frac * n)               # support size
        # dynamic limit: try as many attempts as fit into the remaining time.
        # Empirically ~150 µs per evaluation → 6000 attempts per second.
        # We aim for at most 200 k attempts per size but stop earlier if time runs out.
        max_attempts = 200_000
        attempts = 0
        print(f"[search] support size {m} (fraction {frac:.2f})")
        while attempts < max_attempts and time.time() < phase_end:
            attempts += 1
            # random subset without replacement
            S = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / m
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} attempt={attempts} score={val:.6f}")
                if best_val <= target_score:
                    print("[success] target reached during random search")
                    return best_seq.tolist()
        print(f"[search] size {m} finished after {attempts} attempts, best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Local swap improvement (hill climbing)
    if best_seq is not None:
        support = np.flatnonzero(best_seq > 0)
        m = support.size
        print(f"[swap] starting hill climbing on support size {m}")
        while time.time() < deadline:
            # pick a random element of the current support to remove
            out_idx = np.random.randint(0, m)
            i = support[out_idx]

            # pick a random index not currently in the support
            j = np.random.randint(0, n)
            while best_seq[j] > 0:               # ensure j ∉ support
                j = np.random.randint(0, n)

            # construct the new support
            new_support = support.copy()
            new_support[out_idx] = j

            # evaluate the new candidate (uniform weight 1/m)
            a_new = np.zeros_like(best_seq)
            a_new[new_support] = 1.0 / m
            val_new = fast_score(a_new)

            if val_new < best_val:
                best_val, best_seq = val_new, a_new
                support = new_support
                print(f"[swap] new best {best_val:.6f}")
                if best_val <= target_score:
                    print("[success] target reached during swaps")
                    return best_seq.tolist()
            # optional: stop swapping after a number of iterations without improvement
            # (handled implicitly by the time budget)

    # ------------------------------------------------------------------
    # 3. Fallback – projected sub‑gradient optimisation
    remaining = deadline - time.time()
    if remaining > 5.0:                     # give the optimiser a few seconds
        print("[fallback] starting sub‑gradient optimisation")
        if best_seq is None:
            init = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
        else:
            init = best_seq.copy()
        refined = subgradient_optimize(init, remaining)
        refined_val = fast_score(refined)
        print(f"[fallback] refined score {refined_val:.6f}")
        if refined_val < best_val:
            best_seq, best_val = refined, refined_val

    # ------------------------------------------------------------------
    # 4. Final sanity check
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] returning sequence with score {best_val:.6f}")
    return best_seq.tolist()
