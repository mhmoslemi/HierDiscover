"""
A two‑stage smoothed mirror‑descent optimiser.

Stage 1 works on a short sequence (≈2000 entries) where many
iterations are cheap – it uses a smooth approximation of the
max‑convolution (soft‑max with an annealed temperature) and an
exponentiated‑gradient (mirror‑descent) update on the probability
simplex.

The obtained distribution is interpolated to the final length
(≈5000) and refined in stage 2 with the same algorithm but a
slower learning rate and a higher “temperature’’ (beta) so that the
true max becomes dominant.

A final stochastic hill‑climbing phase makes small mass‑transfer
moves until the time budget expires.  The best sequence seen is
returned.  All numbers are non‑negative and the sequence is
normalised to sum = 1, which does not affect the objective
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear auto‑convolution of ``a`` using FFT of length ``L``.
    Returns the full convolution vector of length ``2·len(a)-1``.
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv_full


def _conv_full_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute full auto‑convolution of ``a`` (via FFT) and its maximum.
    """
    n = a.shape[0]
    # length large enough for the later gradient convolution (3n‑2)
    L = _next_pow2(3 * n - 2)
    conv_full = _conv_full(a, L)
    max_val = float(conv_full.max())
    return conv_full, max_val


def _smooth_gradient(
    a: np.ndarray,
    conv_full: np.ndarray,
    max_val: float,
    beta: float,
) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
        f(a) = (1/β)·log Σ_k exp(β·conv_k)

    Returns the sub‑gradient w.r.t. ``a`` (size ``n``).
    """
    n = a.shape[0]
    # smoothed weights (all ≤ 1 because conv ≤ max_val)
    w = np.exp(beta * (conv_full - max_val))
    p = w / w.sum()                     # probability vector on convolution indices

    # correlation a * p (g_i = 2 * Σ_j a_j·p_{i+j})
    p_rev = p[::-1]                     # reverse for convolution
    L = _next_pow2(3 * n - 2)

    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p_rev, n=L)

    corr = np.fft.irfft(Af * Pf, n=L)    # linear convolution a * p_rev
    # extract the needed slice: g_i = 2·corr[ (2n‑2) – i ]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * corr[idx]
    return g


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) update on the simplex.
    """
    # avoid log(0) – a is always ≥ 1e‑15
    log_a = np.log(a) - eta * grad
    log_a -= np.max(log_a)               # stabilise exponentiation
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-15)     # keep a tiny floor
    a_new /= a_new.sum()
    return a_new


def _random_perturb(a: np.ndarray, eps_max: float = 0.03) -> np.ndarray:
    """
    Move a small random fraction of mass from one coordinate to another.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    eps = np.random.rand() * eps_max
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a[i] -= delta
    a[j] += delta
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _score_from_conv(n: int, max_conv: float) -> float:
    """Objective value used in the optimiser (ignoring the constant 2)."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Optimise a non‑negative step function under the `evaluate_sequence`
    objective.  The routine runs for at most `budget_s` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a couple of seconds for the caller.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Small‑scale optimisation (fast, many iterations)
    # ------------------------------------------------------------------
    n_small = 2000
    a = np.random.rand(n_small).astype(float)
    a /= a.sum()
    best_small = a.copy()
    conv_small, max_small = _conv_full_and_max(a)
    best_score = _score_from_conv(n_small, max_small)

    # Stage‑1 timing allocation (≈45 % of the budget)
    stage1_deadline = start_time + budget_s * 0.45
    it = 0
    while time.time() < stage1_deadline:
        it += 1
        # full conv & max (needed for smooth weights)
        conv_full, max_val = _conv_full_and_max(a)

        # linear schedule for the smoothing parameter β
        # start small (β≈5) → end moderate (β≈200) within the stage
        frac = (time.time() - start_time) / (stage1_deadline - start_time)
        beta = 5.0 + 195.0 * frac

        # gradient of smoothed max
        g = _smooth_gradient(a, conv_full, max_val, beta)

        eta = 0.9 / math.sqrt(it)
        a = _exp_update(a, g, eta)

        # occasional random jitter to escape plateaus
        if it % 3000 == 0:
            a = _random_perturb(a, eps_max=0.025)

        # keep the best seen so far
        cur_score = _score_from_conv(n_small, max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_small = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage1 iter={it:6d} "
                f"improved score={cur_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 2. Upsample to the final length and refine
    # ------------------------------------------------------------------
    n_final = 5000
    # linear interpolation from the small‑scale optimum
    x_old = np.linspace(0.0, 1.0, n_small)
    x_new = np.linspace(0.0, 1.0, n_final)
    a = np.interp(x_new, x_old, best_small)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_final = a.copy()
    conv_full, max_val = _conv_full_and_max(a)
    best_score = _score_from_conv(n_final, max_val)

    stage2_deadline = start_time + budget_s * 0.9   # ~45 % more time
    it = 0
    while time.time() < stage2_deadline:
        it += 1
        conv_full, max_val = _conv_full_and_max(a)

        # smoother at the start, then focus on the true max
        frac = (time.time() - stage1_deadline) / (stage2_deadline - stage1_deadline)
        beta = 50.0 + 1450.0 * frac   # β goes from ~50 to ~1500

        g = _smooth_gradient(a, conv_full, max_val, beta)

        eta = 0.6 / math.sqrt(it)      # a slightly smaller step for fine tuning
        a = _exp_update(a, g, eta)

        if it % 4000 == 0:
            a = _random_perturb(a, eps_max=0.02)

        cur_score = _score_from_conv(n_final, max_val)
        if cur_score < best_score:
            best_score = cur_score
            best_final = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage2 iter={it:6d} "
                f"improved score={cur_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climbing / local search
    # ------------------------------------------------------------------
    a = best_final.copy()
    conv_full, max_val = _conv_full_and_max(a)
    cur_score = _score_from_conv(n_final, max_val)

    hill_deadline = deadline
    while time.time() < hill_deadline:
        i, j = np.random.randint(0, n_final, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.04          # up to 4 % of source mass
        delta = eps * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate new max‑convolution
        conv_new, max_new = _conv_full_and_max(a_new)
        new_score = _score_from_conv(n_final, max_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_final = a_new.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] hill‑climb improved "
                    f"score={new_score:.6f}"
                )

    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")

    # Return as a plain Python list of floats
    return [float(v) for v in best_final]
