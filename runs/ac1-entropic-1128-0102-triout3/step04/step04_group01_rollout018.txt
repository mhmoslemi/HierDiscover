def _run_adam_opt(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_size = 0.5  # base step
    lam_factor = 0.3
    decay_interval = 2000
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={iteration} score={cur_score:.6f}")
        # update λ
        if iteration % decay_interval == 0:
            lam_factor = max(lam_factor * 0.7, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)
        # gradient smoothed
        grad = _gradient_from_conv(a, conv, lam)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = step_size / (np.sqrt(iteration) + 1)  # optional schedule
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
        # optionally symmetrize with 10% probability
        if np.random.rand() < 0.1:
            a = (a + a[::-1]) * 0.5
    return best_a, best_score

# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max over all convolution entries – numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex (smooth ℓ∞ surrogate)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector encountered and its exact objective value:
    2·n·max(conv) (the sequence is kept on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.3          # λ ≈ lam_factor * max(conv)
    lam_decay = 0.7
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective (∑a = 1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # Update λ (never let it become zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # Gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # A diminishing base step (the 1/√it schedule works well)
        base_step = 0.6 / math.sqrt(it + 1.0)

        # Parameter update
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # Optionally enforce symmetry (helps for many lengths)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # Project back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 100000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from i to j.
    Returns an improved sequence (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators (different baselines)
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting vectors from a base.
    The returned vectors are already normalised to sum = 1.
    """
    variants = []

    # 1. the base itself (usually a warm‑start)
    variants.append(base_seq.copy())

    # 2. sinusoidal shape (positive on (0,π))
    n = base_seq.shape[0]
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # all positive
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular (ramp up then down)
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. a random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # Keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Try to reuse the previous best if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start small, grow to 32768.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)   # avoid systematic bias when we stop early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start / several variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # generate a handful of different starting points
        init_candidates = _initial_variants(a0)

        # allocate a slice of remaining time for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # larger lengths get a proportionally larger share
        fraction = 0.15 / (len(lengths) - idx)
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # ------------------------------------------------------------------
        # Run a few Adam optimisations (different starts, sometimes forced symmetry)
        # ------------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # occasionally enforce symmetry (30 % of the runs)
            sym = (np.random.rand() < 0.3)
            a_opt, score_opt = _run_adam_opt(start_vec, n, slice_deadline, start_time, symmetrize=sym)

            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # final annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                best_local_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=150000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
