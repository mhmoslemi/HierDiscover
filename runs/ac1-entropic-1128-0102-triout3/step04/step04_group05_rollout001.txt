"""
Multi‑scale simplex‑projected optimisation with additional “U‑shaped’’ seeds
and a long low‑temperature refinement.

Key ideas
---------

* Initialise from the best known construction (`height_sequence_1`) if it
  exists, otherwise start from a uniform vector.
* Build a schedule of increasing lengths (≈1.5×) up to ≈8000.
  At each length the current best shape is up‑sampled by linear interpolation.
* For each length run a projected Adam optimiser that minimises
  `2·n·max(conv(a,a))`.  The Adam optimiser uses a soft‑max approximation of the
  max with a temperature that is exponentially decreased from 5·10⁻² down to
  1·10⁻⁶, thereby eventually focussing the gradient on the true peak.
* Enrich the pool of initial candidates with “U‑shaped’’ (edge‑peaked) vectors
  generated from inverted triangular and negative‑beta‑like formulas; also add
  a few random Dirichlet starts.
* After the multi‑scale loop, allocate all remaining time to a final long
  refinement on the overall best sequence with a very low temperature and a
  smaller learning‑rate.
* All updates keep the vector on the probability simplex (sum = 1,
  non‑negative) via Euclidean projection.

The routine respects the 1000 s budget, prints progress and finally returns
the best sequence as a plain list of floats.
"""

import time

from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> Tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution (real).  a assumed to sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximated objective w.r.t. a.
    Implements g_i = 2 * Σ_k w_k * a_{k-i} using FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 1e-6) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of `2·n·max(conv)` on the simplex.

    Returns (final_a, best_seen_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current score
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule (exponential decay)
        tau = temp_start * (temp_end / temp_start) ** (it / max_iters)
        tau = max(tau, 1e-14)

        # soft‑max weights over convolution entries
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of the smooth max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

    return a, best_a, best_score

def _make_u_shaped(n: int, power: float = 0.5) -> np.ndarray:
    """
    Produce a deterministic 'U‑shaped' candidate:
    weights are larger near the edges and smaller near the centre.
    """
    idx = np.arange(n, dtype=float)
    t = idx / (n - 1) if n > 1 else np.array([0.0])
    # inverted beta‑like shape: (t*(1-t)+eps)^{-power}
    eps = 1e-12
    w = (t * (1.0 - t) + eps) ** (-power)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else np.full(n, 1.0 / n)
