"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
1. **Multiscale optimisation** – first optimise a modest size (≈2000),
   then up‑sample to the full size (≈4800) and continue.
2. **Smooth surrogate loss** – instead of the non‑smooth max we minimise
   a log‑sum‑exp approximation:
        Lα(a) = (1/α)·log(∑ₖ exp(α·(a*a)[k])),
   whose gradient is the correlation of the soft‑max weights with the
   current vector.  A continuation scheme slowly increases α from 2 to 80.
3. **Adam on the probability simplex** – we keep the iterate on the simplex
   by a Euclidean projection after every Adam update (β₁=0.9, β₂=0.999).
   This gives adaptive step sizes and faster escape from plateaus.
4. **Final mass‑transfer annealing** – a short simulated‑annealing phase
   (moving a random ε from a high‑mass index to a random index) refines
   the best vector found.
5. **Time‑budget aware** – the routine respects the supplied budget
   (default 1000 s).  Progress messages are printed; the best sequence
   encountered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the simplex {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Correlation of w (length 2·n‑1) with a (length n) – “valid” part only.
# Returns an array of length n:  grad_i = Σ_k w_{k+i}·a_k .
# ----------------------------------------------------------------------
def _correlation_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation w ⋆ a (valid part, length = len(a))."""
    n = a.shape[0]
    m = w.shape[0]                 # = 2·n‑1
    size = m + n - 1               # = 3·n‑2
    fft_len = 1 << ((size).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(W * A_rev, n=fft_len)
    start = n - 1                  # first index of the “valid’’ correlation
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
# Linear interpolation + renormalisation (upsampling)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with a smooth log‑sum‑exp surrogate.
# ----------------------------------------------------------------------
def _run_adam_softmax(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_start: float = 2.0,
    alpha_end: float = 80.0,
    lr0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a using Adam on the simplex, minimising a smooth approximation
    of max(conv(a)).  Returns the best vector encountered and its exact
    objective value 2·n·max(conv(a)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    total_time = deadline - start_time

    while time.time() < deadline:
        it += 1

        # ----- exact objective (for bookkeeping) -----
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter {it:,} best={cur_score:.6f}")

        # ----- smooth surrogate -----
        # linearly increase α during the run
        elapsed = time.time() - start_time
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_time)

        # stable soft‑max over the convolution values
        shift = max_conv                     # max(conv) is a safe shift
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()       # normalised weights, Σw=1

        # ----- gradient of the surrogate -----
        # ∂L/∂a = 2 * correlation(w, a)
        grad = 2.0 * _correlation_valid(w, a)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index to a
    random index.  Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source index sampled proportionally to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,} best={score:.6f}")

    return best_seq, best_score
