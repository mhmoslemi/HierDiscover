"""
Propose a good step‑function sequence.

The algorithm consists of three stages:

1. **Random support search** – very quick sampling of uniform‑weight
   vectors on random supports.  The best vector found is used as a warm
   start for the following stage.

2. **Iterated LP improvement** – implement the iterative scheme from the
   paper: given f (∑f=1) compute M = max(f∗f).  Solve a linear program
   which maximises the ℓ₁‑norm of a vector g under the linear constraints
   (f∗g)≤M.  After normalising g to have sum 1 we move a small step towards g.
   The step size is back‑tracked so that the new max‑convolution never
   exceeds the previous value.  Repeating this monotone update quickly
   reduces the evaluation score.

3. **Projected sub‑gradient refinement** – if the LP stage does not reach
   the target score (1.503) we fall back to the sub‑gradient optimiser
   used previously.

All operations stay on the simplex (non‑negative, sum = 1) therefore the
evaluation function is scale‑invariant.  The code respects the 1000 s
budget and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Helper utilities (fast version of evaluate_sequence)
def max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] using an FFT of appropriate length."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(conv.max())


def fast_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence (no safety checks)."""
    s = a.sum()
    if s < 1e-12:
        return math.inf
    n = a.size
    M = max_convolution(a)
    return float(2.0 * n * M / (s * s))


# ----------------------------------------------------------------------
# LP building and solving
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = f∗g
    (linear convolution, zero padding outside the support).
    """
    n = f.size
    # first column of the Toeplitz matrix = [f, 0,…,0] (length 2n‑1)
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row = [f[0], 0,…,0] (length n)
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A = toeplitz(c, r)                     # shape (2n‑1, n)
    return A


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀ g   subject to   f ∗ g ≤ M,   g ≥ 0.
    Returns the optimal g (may be far from normalised) or None on failure.
    """
    n = f.size
    A = build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # linprog minimises, so -1ᵀ g maximises 1ᵀ g
    bounds = [(0.0, None)] * n
    # “highs” is the modern, fast interior‑point solver in SciPy
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

# Projected sub‑gradient optimiser (fallback, copied from the reference)
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent on the original objective."""
    a = a_init.copy()
    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # compute convolution and locate worst index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pair count)
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex (L1=1, non‑negative)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer – helps escaping plateaus
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # re‑project onto simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

# Random‑support fast search (phase 0)
def random_support_search(n: int,
                         support_sizes,
                         attempts_per_size: int,
                         deadline: float,
                         target_score: float) -> tuple[np.ndarray, float]:
    """
    Generate uniform‑weight vectors on random supports.
    Returns the best vector and its score (or (None, inf) if none found).
    """
    best_seq = None
    best_val = math.inf
    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[rand‑search] support size {m}")
        attempts = 0
        while attempts < attempts_per_size and time.time() < deadline:
            attempts += 1
            S = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / m
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"   [found] m={m} att={attempts} score={val:.6f}")
            if val <= target_score:
                print("[rand‑search] target reached")
                return best_seq, best_val
        print(f"   size {m} finished, best so far {best_val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The method combines a quick random‑support sweep,
    an LP‑based monotone improvement and, as a safety net, a sub‑gradient
    optimiser.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    target_score = 1.503
    n = 1500                                        # fixed length used historically

    # ------------------------------------------------------------------
    # Stage 0 – initialise from a previously stored construction if any
    best_seq = None
    best_val = math.inf
    try:
        prev = np.asarray(globals().get('height_sequence_1'), dtype=np.float64)
        if prev.size == n:
            prev = prev / prev.sum()               # normalise to ℓ₁‑norm = 1
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – quick random‑support exploration (helps escape a bad start)
    if time.time() < deadline:
        support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
        rand_seq, rand_val = random_support_search(
            n, support_sizes, attempts_per_size=3000,
            deadline=deadline, target_score=target_score)
        if rand_seq is not None and rand_val < best_val:
            best_seq, best_val = rand_seq, rand_val
            print(f"[stage0] random best score {best_val:.6f}")

    # If nothing was found yet, start from the uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)
        print(f"[stage0] fallback uniform score {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Iterated LP improvement
    f = best_seq.copy()                # current iterate (ℓ₁‑norm = 1)
    cur_val = best_val
    iteration = 0
    max_lp_iters = 30                  # guard against overrunning the budget

    while time.time() < deadline and iteration < max_lp_iters:
        iteration += 1
        M = max_convolution(f)                     # current max autocorrelation
        # Solve LP: maximise ℓ₁‑norm of g under f∗g ≤ M
        g0 = solve_lp_for_g(f, M)
        if g0 is None:
            print("[LP] linear programme failed – exiting LP loop")
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[LP] degenerate solution – exiting LP loop")
            break
        g = g0 / sum_g0                             # normalise to sum = 1

        # Mix f towards g.  Use back‑tracking on step size to keep M non‑increasing.
        t = 0.5                                      # initial step size
        accepted = False
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            M_new = max_convolution(f_new)
            if M_new <= M * (1.0 + 1e-12):           # not larger (allow tiny tolerance)
                accepted = True
                break
            t *= 0.5
        if not accepted:
            # cannot make progress this round – stop LP phase
            print("[LP] could not find a non‑increasing step, terminating LP loop")
            break

        f = f_new
        cur_val = fast_score(f)
        if cur_val < best_val:
            best_val, best_seq = cur_val, f.copy()
            print(f"[LP iter {iteration}] improved score {best_val:.6f} (t={t:.4f})")
        else:
            print(f"[LP iter {iteration}] no improvement (t={t:.4f}) score {cur_val:.6f}")

        if best_val <= target_score:
            print("[LP] target reached – returning")
            break

    # ------------------------------------------------------------------
    # Stage 3 – fallback sub‑gradient optimiser if we are still above target
    if best_val > target_score and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[fallback] running projected sub‑gradient optimiser")
            init = best_seq.copy()
            refined = subgradient_optimize(init, remaining)
            refined_val = fast_score(refined)
            print(f"[fallback] sub‑gradient result score {refined_val:.6f}")
            if refined_val < best_val:
                best_val, best_seq = refined_val, refined.copy()

    # ------------------------------------------------------------------
    # Final sanity check (make sure we return a Python list of plain floats)
    final_seq = best_seq.tolist()
    final_score = fast_score(np.asarray(final_seq, dtype=np.float64))
    print(f"[final] returning sequence length {len(final_seq)} with fast score {final_score:.6f}")

    return final_seq
