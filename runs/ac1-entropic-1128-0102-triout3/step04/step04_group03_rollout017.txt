        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n} iter={it} score={cur_score:.6f}")
        k = conv.argmax()

        # Adam updates

        a = a - step  # move opposite gradient direction
        # simplex projection
        a = _project_simplex(a)
    return best_a, best_score

"""
Improved optimisation for the non‑negative step‑function problem.

The algorithm consists of three phases:

1.  **Adam‑based projected optimisation** – we minimise the maximal
    entry of the self‑convolution on the probability simplex using an
    Adam optimiser (first‑ and second‑moment adaptation).  The gradient
    is taken only from the currently maximal convolution entry, which
    gives a much sharper descent direction than averaging over all
    active entries.

2.  **Progressive up‑sampling** – after optimisation at a modest length
    (≈800) the best distribution is linearly interpolated to a larger
    size (≈2400 → 4800 → 9600) and the Adam optimiser is run again.
    Upsampling preserves the total mass and allows the extra degrees of
    freedom to further lower the peak of the convolution.

3.  **Final simulated‑annealing refinement** – a cheap “move ε from i
    to j’’ walk is performed for many steps, accepting downhill moves
    and occasional uphill moves according to a geometric temperature
    schedule.  This phase escapes tiny local minima and yields the best
    result found before the time budget expires.

All phases respect the supplied time budget (default 1000 s).  The
returned sequence is a plain Python list of non‑negative floats whose
sum is forced to 1 (the evaluation function is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv_k = (a*a)[k] w.r.t. a.
    For each i with 0 ≤ k‑i < n:
        if i == k‑i :  ∂ = a_i
        else          :  ∂ = 2·a_{k‑i}
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    j = k - idx
    same = idx == j
    # diagonal contributions
    grad[idx[same]] = a[idx[same]]
    # off‑diagonal contributions (appear twice in the sum)
    grad[idx[~same]] = 2.0 * a[j[~same]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected Adam)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.05,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the maximal entry of the self‑convolution.
    Returns the best point encountered and its scaled score = 2·n·max(conv).
    """
    a = a0.copy()
    a = a / a.sum()                     # ensure we start on the simplex
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n} iter={it} score={cur_score:.6f}")

        k = conv.argmax()               # index of the maximal entry
        grad = _grad_of_max(a, k)       # sub‑gradient of that entry

        # Adam update (we move opposite to the gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                    # descent step
        a = _project_simplex(a)         # back to the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move ≤10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
