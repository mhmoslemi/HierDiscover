def _targeted_refinement(a0, n, deadline, start_time, best_score, best_seq, epsilon_frac=0.05):
    a = a0.copy()
    best = best_seq.copy()
    best_s = best_score
    while time.time() < deadline:
        # compute convolution and max index
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # Might update global best
        if cur_score < best_s:
            best_s = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] targeted improve: n={n}, score={cur_score:.6f}")

        # find index of max_entry (if multiple choose one)
        k = int(conv.argmax())
        # Build contributions
        # contributions only for i where 0 <= k-i < n
        # contributions[i] = a[i] * a[k-i]
        # Use vectorized
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        # compute contributions
        contribs = np.zeros_like(a)
        contribs[valid] = a[valid] * a[partner[valid]]
        # choose i_star as argmax contrib
        i_star = int(np.argmax(contribs))
        # If a_i_star is zero, break to avoid infinite loop
        if a[i_star] <= 1e-12:
            # maybe random reinitialize?
            # break early
            break

        # choose j_star: index with minimal potential increase to conv[k]
        # potential increase = a[partner_j] if partner_j valid else 0
        partner_j = k - idx
        inc = np.where((partner_j >= 0) & (partner_j < n), a[partner_j], 0.0)
        # We want minimal inc, maybe also consider a_j low.
        # Combine: weight = inc + a_j
        # Set large value for large a_j to avoid moving too much to high mass location
        # Use metric = inc + a
        metric = inc + a
        # prefer lower metric
        j_star = int(np.argmin(metric))
        if i_star == j_star:
            # pick second best
            sorted_idxs = np.argsort(metric)
            j_star = int(sorted_idxs[1]) if len(sorted_idxs)>1 else (j_star+1)%n

        # epsilon move
        eps = min(a[i_star] * epsilon_frac, a[i_star])
        if eps <= 0:
            # can't move
            break
        # propose move
        a_prop = a.copy()
        a_prop[i_star] -= eps
        a_prop[j_star] += eps

        # ensure sum remains exactly same (since we move mass)
        # compute new conv and score
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        # Accept if improves
        if new_score < cur_score:
            a = a_prop
        else:
            # maybe accept with small probability for exploration
            # compute temperature decreasing linearly with time
            t_rem = deadline - time.time()
            # Use simple acceptance: if random < 0.01
            if np.random.rand() < 0.01:
                a = a_prop

        # optionally break after many iterations? We'll continue until deadline.
    return best, best_s

def beta_shape(n, alpha, beta):
    x = np.linspace(0.5/n, 1-0.5/n, n) # midpoints on [0,1]
    a = (x**(alpha-1)) * ((1-x)**(beta-1))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1/n)
    return a

"""
Propose a low‑score step function.

The algorithm consists of three stages:

1. **Parametric seeding** – a short search over symmetric Beta
   distributions (a_i ∝ x^{α‑1}(1‑x)^{α‑1}) on a modest size
   (n≈2000).  The best shape found is up‑sampled to the target size.

2. **Projected gradient descent** – many iterations of a
   Nesterov‑style SGD on the probability simplex, using the exact
   sub‑gradient of the ‑∞‑norm of the self‑convolution.

3. **Targeted refinement** – a greedy mass‑transfer walk that always
   removes a small amount of mass from the index contributing the
   most to the current peak of the convolution and places it at an
   index that hurts the peak the least.  Down‑hill moves are always
   accepted; occasional uphill moves are allowed for escaping local
   minima.

The three stages respect the 1000 s time budget and keep the best
sequence discovered.  The final sequence is returned as a plain list
of non‑negative floats (already normalised to sum 1, which makes the
evaluation function independent of scaling).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over all active entries)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov momentum)
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.8,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point found and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation, preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Objective used internally (independent of the external harness)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·len(a)·max(conv) for a normalised (sum=1) non‑negative array."""
    a = a.astype(float)
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a = a / s
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Generate a symmetric Beta‑distribution shaped sequence
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Return a normalised Beta(α,β) density sampled at n mid‑points."""
    # mid‑point of each bin
    x = (np.arange(n) + 0.5) / n
    # avoid 0^neg or 1^neg by clipping
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta_param - 1))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Targeted greedy refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _targeted_refinement(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         cur_best_score: float,
                         cur_best_seq: np.ndarray,
                         eps_frac: float = 0.05,
                         uphill_prob: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Greedy walk: move a small fraction of mass from the index that
    contributes most to the current peak of the convolution to an index
    that hurts that peak the least.  Accept downhill moves always,
    uphill moves with a tiny probability.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] targeted improve: n={n}, score={cur_score:.6f}")

        # index of a maximal convolution entry
        k = int(conv.argmax())

        # contribution of each a[i] to the peak at k
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[valid] * a[partner[valid]]

        i_star = int(np.argmax(contrib))
        if a[i_star] <= 0.0:
            # nothing to move – abort early
            break

        # choose destination j that adds the least to this peak
        # increase would be a[partner_j] if partner_j is valid, else 0
        inc = np.zeros_like(a)
        inc[valid] = a[partner[valid]]
        # a low‑mass destination that also has low inc is preferred
        metric = inc + a
        j_star = int(np.argmin(metric))
        if j_star == i_star:
            # pick the second best if we accidentally landed on the same index
            order = np.argsort(metric)
            j_star = int(order[1] if order.size > 1 else (j_star + 1) % n)

        eps = min(eps_frac * a[i_star], a[i_star])
        if eps <= 0.0:
            break

        a_prop = a.copy()
        a_prop[i_star] -= eps
        a_prop[j_star] += eps

        # evaluate the proposal
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        # accept downhill always, occasional uphill
        if delta < 0 or (np.random.rand() < uphill_prob):
            a = a_prop
            cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget, builds an initial shape
    from a short Beta‑distribution search, refines it with projected
    gradient descent and a final targeted greedy walk, and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # 1) PARAMETRIC SEEDING (short, cheap search)
    # --------------------------------------------------------------
    # work on a modest size to keep the search cheap
    n_seed = 2000
    best_seed = None
    best_score = float("inf")

    # a tiny time slice for the seed search (≈5 % of total time)
    seed_time = min(30.0, (deadline - start_time) * 0.07)
    seed_deadline = time.time() + seed_time

    # allow a mixture of symmetric and asymmetric Beta shapes
    # – sample a few hundred random (α,β) pairs on a log‑scale
    n_trials = 300
    for _ in range(n_trials):
        if time.time() >= seed_deadline:
            break
        # log‑uniform sampling of parameters in [0.1, 5]
        log_a = np.random.uniform(np.log(0.1), np.log(5.0))
        log_b = np.random.uniform(np.log(0.1), np.log(5.0))
        alpha = np.exp(log_a)
        beta_param = np.exp(log_b)
        a = _beta_shape(n_seed, alpha, beta_param)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seed = a
            print(f"[{time.time() - start_time:6.2f}s] seed improve: α={alpha:.3f}, β={beta_param:.3f}, score={score:.6f}")

    # fallback – plain uniform if something went wrong
    if best_seed is None:
        best_seed = np.full(n_seed, 1.0 / n_seed)

    # --------------------------------------------------------------
    # 2) PROJECTED GRADIENT DESCENT (large scale)
    # --------------------------------------------------------------
    # Target size – a compromise between quality and speed.
    n_target = 4800
    a0 = _upsample(best_seed, n_target)

    # Allocate roughly 55 % of the remaining budget to gradient descent.
    remaining = deadline - time.time()
    grad_budget = remaining * 0.55
    grad_deadline = time.time() + grad_budget

    a_opt, score_opt = _run_gradient_descent_full(
        a0, n_target, grad_deadline, start_time,
        beta=0.9, step0=0.8, max_iter=10**7
    )

    best_seq = a_opt.copy()
    best_score = score_opt
    print(f"[{time.time() - start_time:6.2f}s] post‑grad best: n={n_target}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) TARGETED GREEDY REFINEMENT
    # --------------------------------------------------------------
    # Use the remaining time for the greedy refinement.
    if time.time() < deadline:
        a_ref, score_ref = _targeted_refinement(
            best_seq, n_target,
            deadline, start_time,
            best_score, best_seq,
            eps_frac=0.07,   # a slightly larger transfer fraction
            uphill_prob=0.005
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final refined: score={best_score:.6f}")

    # --------------------------------------------------------------
    # SAFETY FALLBACK
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list – the harness expects this format
    return [float(x) for x in best_seq]
