"""
Hill‑climbing optimizer for the step‑function evaluation problem.

The algorithm works in two phases:

1. **Exploratory random‑support search** – generate many candidates with a
   uniform weight on a randomly chosen support set.  The fast FFT based
   score is used to keep the best candidate found so far.  This phase
   quickly finds a vector with a score around 1.5 (the previous best).

2. **Local hill‑climbing** – starting from the best candidate, repeatedly
   move a small amount of mass from a coordinate that contributes strongly
   to the current maximal convolution entry to a coordinate that contributes
   little.  The convolution is updated incrementally (O(n) work per move)
   using the exact formula

        Δconv[l] = 2·ε·(a[l‑j] – a[l‑i])

   where ε is the transferred mass, i the source index and j the target.
   The step size ε is adapted during the run.  All updates respect the
   non‑negativity constraints and keep the ℓ₁‑norm equal to 1, so the
   evaluation score is invariant to scaling.

The whole search respects the 1000 s time budget.  If the time expires
early the best vector found so far is returned.
"""

import time
import math
import numpy as np

def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no safety checks).
    The vector is assumed to be non‑negative; it is not re‑scaled.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    # length of FFT (next power of two >= 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(a_init: np.ndarray, time_limit: float, n: int) -> np.ndarray:
    """
    Local optimizer that reduces max(conv) by moving small amounts of mass.
    The ℓ₁‑norm stays equal to 1, all entries stay non‑negative.
    """
    a = a_init.copy()
    a /= a.sum()                     # normalize to sum = 1
    conv = np.convolve(a, a)         # exact convolution, length 2n‑1
    L = conv.size

    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    start = time.time()
    deadline = start + time_limit

    step_factor = 0.25               # relative step size (ε = step_factor·a[i])
    min_step = 1e-12
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1

        # occasionally recompute convolution from scratch (numerical drift)
        if iter_cnt % 5000 == 0:
            conv = np.convolve(a, a)

        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # contribution of each coordinate to the current maximal entry
        idxs = max_idx - np.arange(n)
        mask = (idxs >= 0) & (idxs < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idxs[mask]]

        # indices that can donate mass
        donors = np.where((a > min_step) & (contrib > 0))[0]
        if donors.size == 0:
            break

        # pick a donor with relatively large contribution (top‑k random)
        top_k = min(10, donors.size)
        donor_weights = contrib[donors] * a[donors]
        top_idx = np.argpartition(-donor_weights, top_k - 1)[:top_k]
        i = donors[top_idx[np.random.randint(top_k)]]

        # pick a receiver with small contribution (bottom‑k random)
        low_thr = np.percentile(contrib, 30)

                print(f"[hill] iter {iter_cnt} improved score {best_score:.6f}")
        else:

            if eps_small > min_step:

                conv_tmp[i:i + n] -= 2.0 * eps_small * a

                    conv = conv_tmp

                if conv_tmp.max() <= conv.max():
                    a[cand_i] -= eps_rand
                    a[cand_j] += eps_rand
                    conv = conv_tmp

        # keep the vector on the simplex (numerical drift)
        if iter_cnt % 2000 == 0:

        print(f"[search] trying support size {m}")
        attempts = 0
        while attempts < attempts_per_size and time.time() < deadline:
            attempts += 1
            S = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / m                     # uniform weight on the support
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} att={attempts} score={val:.6f}")
            if val <= target_score:
                print("[success] target reached during random search")
                return best_seq.tolist()
        print(f"[search] size {m} done, best so far {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Local hill‑climbing refinement on the best candidate
    # --------------------------------------------------------------
    if best_seq is None:
        # fall back to uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    time_left = deadline - time.time()
    if time_left > 2.0:
        print("[refine] starting hill‑climbing")
        refined = hill_climb(best_seq, time_left, n)
        refined_val = fast_score(refined)
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined.copy()
            print(f"[refine] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. final safety net (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    # final diagnostics
    final_val = fast_score(best_seq)
    print(f"[final] returning sequence of length {n} with score {final_val:.6f}")

    # convert to plain Python list and clip to legal range (0 – 1000)
    out = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    return out
