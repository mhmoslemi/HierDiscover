def _adam_optimize_sym(a, max_iters, deadline,
                       alpha=0.1, beta1=0.9, beta2=0.999,
                       eps=1e-8, temp_start=0.2, temp_end=1e-5):
    n = len(a)
    fft_len_conv = 1 << ((2*n-1).bit_length())
    fft_len_grad = 1 << ((3*n-2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # compute initial best
    max_val, _ = _conv_max_fft(a)
    best = 2.0*n * max_val
    best_a = a.copy()
    for it in range(1, max_iters+1):
        if time.time()>=deadline: break
        max_val, conv = _conv_max_fft(a)
        cur = 2.0*n*max_val
        if cur < best:
            best = cur; best_a = a.copy()
        # temperature schedule
        tau = temp_start * (temp_end/temp_start)**(it/max_iters)  # exponential decay
        # compute weights
        w = np.exp((conv - max_val)/tau)
        w_sum = w.sum()
        if w_sum>0: w /= w_sum
        else: w.fill(1.0/len(w))
        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)
        # maybe symmetrize gradient? Not needed as a sym.
        g = (g + g[::-1]) / 2.0
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = alpha/(np.sqrt(v_hat)+eps)
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
    return best_a, best

def _local_hill_climb(a, max_iters, deadline):
    n = len(a)
    best_a = a.copy()
    max_val, conv = _conv_max_fft(best_a)
    best_score = 2.0*n*max_val
    for it in range(max_iters):
        if time.time() >= deadline: break
        # choose random i and j (i != j)
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i: j += 1   # ensure distinct
        # decide epsilon = random fraction of a[i]
        if best_a[i] == 0: continue
        eps = np.random.rand()*0.1  # up to 10% of a[i]
        delta = eps*best_a[i]
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # enforce non-negativity (already ensured)
        # symmetrize
        new_a = (new_a + new_a[::-1]) / 2.0
        # renorm
        s = new_a.sum()
        if s == 0:
            continue
        new_a /= s
        # compute max conv
        max_val, _ = _conv_max_fft(new_a)
        new_score = 2.0*n*max_val
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
    return best_a, best_score

"""
Symmetric Adam + local hill‑climbing optimizer for the auto‑convolution
max‑norm problem.

The objective (with sum(a)=1) is 2·n·max(conv(a,a)).  The algorithm works
on the probability simplex, enforces symmetry a_i = a_{n‑1‑i} after every
projection, and uses an Adam optimiser on a smooth log‑sum‑exp
approximation of the maximum.  After a long Adam refinement a cheap
random hill‑climbing phase performs tiny mass transfers to squeeze the
score a little further.

The search runs inside the supplied time budget (≈1000 s) and returns the
best sequence found as a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Fast convolution via FFT.
    Returns max(conv(a,a)) and the full convolution (real, length 2n‑1).
    The input a is assumed to sum to 1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smooth‑max w.r.t. a.
    Implements g_i = 2 * Σ_k w_k * a_{k-i} using FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize_sym(a0: np.ndarray,
                       max_iters: int,
                       deadline: float,
                       alpha: float = 0.1,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       temp_start: float = 0.2,
                       temp_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with an enforced symmetry a = a[::-1].
    Returns the best vector seen (still summed to 1) and its score
    (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    # FFT lengths – cached for the whole optimisation
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    # initial best
    max_val, _ = _conv_max_fft(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution & score
        max_val, conv = _conv_max_fft(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule – exponential decay
        tau = temp_start * (temp_end / temp_start) ** (it / max_iters)
        tau = max(tau, 1e-14)

        # soft‑max weights over the convolution entries
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # enforce symmetry of the gradient (helps keep a symmetric)
        g = (g + g[::-1]) * 0.5

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

        # enforce symmetry a_i = a_{n‑1‑i}
        a = (a + a[::-1]) * 0.5

    return best_a, best_score


def _local_hill_climb(a0: np.ndarray,
                      max_iters: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random local search: move a tiny amount of mass from one index
    to another, keep the vector symmetric, renormalise and accept if the
    score improves.
    """
    a = a0.copy()
    n = a.shape[0]
    max_val, _ = _conv_max_fft(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    rng = np.random.default_rng()
    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best_a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 5 %) of the mass at i to j
        eps = rng.random() * 0.05
        delta = eps * best_a[i]

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce symmetry and renormalise
        cand = (cand + cand[::-1]) * 0.5
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        max_val, _ = _conv_max_fft(cand)
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = cand

    return best_a, best_score

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(
        f"[{time.time() - start_time:6.2f}s] Finished – best 2·n·max(conv) = {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
