"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
1.  **Choose a length** (default 8000) and generate a family of
    power‑law vectors a_i ∝ (i+1)^(-p) for a grid of exponents p.
2.  **Order the entries** – for each shape we try three simple
    permutations (original, descending, alternating‑ends) and keep the
    best scoring one.  Afterwards a quick stochastic scan of
    ~5000 random permutations further refines the ordering.
3.  **Iterative optimisation** – starting from the best ordering we
    run a fast gradient‑directed mass‑transfer loop (≈10⁴–10⁵
    iterations).  In each iteration we
       * compute the autocorrelation and its active set,
       * build the sub‑gradient of `max(conv)`,
       * try a greedy transfer of a fraction (≈10 %) of mass from the
         highest‑gradient index to the lowest‑gradient index,
       * if the greedy move fails we perform an exponentiated‑gradient
         (mirror‑descent) step and re‑project onto the simplex.
4.  **Final cheap hill‑climbing** – a handful of random swaps/mass
    transfers are attempted until the time budget expires.
5.  Return the best sequence found (as plain Python floats).

All vectors are normalised to sum = 1, so the internal score
`2·n·max(conv(a,a))` equals the value returned by `evaluate_sequence`.
The routine respects a ~1000 s wall‑clock budget and prints progress
messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) for a vector with sum(a)=1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Full autocorrelation and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """Indices whose value is close to the maximum (active set)."""
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """Sub‑gradient of max(conv) for the active set K."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _alternating_ends_order(a_desc: np.ndarray) -> np.ndarray:
    """
    Place the sorted (descending) entries alternately at the leftmost
    and rightmost free positions, moving inward.  This tends to separate
    large coefficients and lowers the autocorrelation peak.
    """
    n = a_desc.shape[0]
    b = np.empty_like(a_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(a_desc):
        if idx % 2 == 0:
            b[left] = val
            left += 1
        else:
            b[right] = val
            right -= 1
    return b

    """

    deadline = start_time + budget_s - 5.0      # safety margin

    # bound because the maximal autocorrelation can shrink faster than 1/n.
    n = 8000
    print(f"[{time.time() - start_time:6.2f}s] Using length n = {n}")

    # --------------------------------------------------------------
    # 2. Power‑law grid search (shape selection).
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # Explore exponents p in a reasonable range.
    p_vals = np.linspace(0.20, 0.80, 31)   # 31 candidates
    for p in p_vals:
        if time.time() > deadline:
            break
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum                       # normalise to sum = 1

        # three cheap orderings
        for ordering in ("orig", "desc", "alt"):
            if ordering == "orig":
                cand = a.copy()
            elif ordering == "desc":
                cand = np.sort(a)[::-1]
            elif ordering == "alt":
                cand = _alternating_ends_order(np.sort(a)[::-1])

            sc = _score_of_seq(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] p={p: .3f} "
                      f"ord={ordering} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Random permutation scan – cheap exploration of the ordering.
    # --------------------------------------------------------------
    perm_trials = 5000
    for _ in range(perm_trials):
        if time.time() > deadline:
            break
        perm = np.random.permutation(n)
        cand = best_seq[perm]
        sc = _score_of_seq(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] Random perm improve → "
                  f"score={sc:.6f}")

    # --------------------------------------------------------------
    # 4. Gradient‑directed mass‑transfer refinement (core loop).
    # --------------------------------------------------------------
    current = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv, max_val = _conv_and_max(current)
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = current.copy()
            print(f"[{time.time() - start_time:6.2f}s] Iter {it:6d} "
                  f"→ new best score={cur_score:.6f}")

        # ----------------------------------------------------------
        # (a) active‑set sub‑gradient
        # ----------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
        g = _gradient_active(current, K)

        # ----------------------------------------------------------
        # (b) greedy mass transfer from high‑gradient to low‑gradient
        # ----------------------------------------------------------
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max != i_min and current[i_max] > 0.0:
            # transfer up to 10 % of the source mass
            delta = min(0.10 * current[i_max], current[i_max])
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            # evaluate candidate
            _, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score - 1e-12:
                current = cand
                continue   # successful greedy step

        # ----------------------------------------------------------
        # (c) fallback – exponentiated‑gradient / mirror descent
        # ----------------------------------------------------------
        eta = 0.5 / math.sqrt(it)          # decaying step size
        log_a = np.log(np.maximum(current, 1e-15)) - eta * g
        # stabilise the exponentials
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        current = a_new
        # loop continues

    # --------------------------------------------------------------
    # 5. Final cheap random hill‑climbing (swap / small mass moves).
    # --------------------------------------------------------------
    if time.time() < deadline:
        local_iters = 15000
        for _ in range(local_iters):
            if time.time() > deadline:
                break
            # randomly decide between a swap and a tiny mass‑transfer
            if np.random.rand() < 0.5:
                # --- swap two entries -------------------------------------------------
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                cand = current.copy()
                cand[i], cand[j] = cand[j], cand[i]
            else:
                # --- move a small fraction of mass ------------------------------------
                i, j = np.random.randint(0, n, size=2)
                if i == j or current[i] == 0.0:
                    continue
                eps = np.random.rand() * 0.04         # up to 4 %
                delta = eps * current[i]
                cand = current.copy()
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:   # safety
                    continue

            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = cand.copy()
                current = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb "
                      f"→ score={new_score:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best vector found (as plain Python floats).
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(100, 1.0 / 100)

    # Ensure non‑negativity and a tiny safety normalisation
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / s

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
