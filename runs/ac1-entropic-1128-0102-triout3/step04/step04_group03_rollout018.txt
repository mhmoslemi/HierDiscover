"""
Industrial‑strength optimiser for the non‑negative step‑function problem.

The routine works in three phases:

1. **LP‑based mass relocation** (small dimension, e.g. n≈1200).  
   For a current distribution f we solve a linear program that
   maximises the total mass of a new vector g under the linear
   constraints   (f∗g)[k] ≤ max(f∗f) .  The solution concentrates mass
   where f is small (usually near the ends).  Repeating the LP several
   times with a convex combination   f←(1‑α)f+α g   monotonically reduces
   the peak of the self‑convolution.

2. **Progressive up‑sampling** – the refined small‑scale distribution
   is linearly interpolated to a larger size (default n=4800).  This
   preserves the total mass and gives a higher‑resolution candidate.

3. **Fine‑grained projected‑gradient descent + simulated annealing**
   on the large problem.  The gradient is the sub‑gradient of the
   maximum convolution entry; after each step the iterate is projected
   onto the probability simplex.  A final annealing walk moves a tiny
   amount of mass between two positions and accepts occasional uphill
   moves.

All phases respect the supplied time budget (default 1000 s).  The best
sequence found is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# Optional SciPy imports – the LP phase is used only if SciPy is available.
try:
    from scipy.optimize import linprog
    from scipy.sparse import toeplitz
except Exception:  # pragma: no cover
    linprog = None
    toeplitz = None

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum x_i = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Sub‑gradient of the peak of the self‑convolution
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices k with conv[k] ≥ max(conv)·(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g

# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (Nesterov‑style)
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.8,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the simplex.
    Returns the best point found together with its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional status line
            if it % 10000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # Diminishing step size – a gentle schedule that stays useful for many
        # iterations.
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Tiny “move ε from i to j’’ annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if it % 10000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Linear‑programming step (maximise total mass under convolution bound)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Given a current distribution `f` (sum f = 1) and the current peak `M`
    of its self‑convolution, solve

        maximise   Σ g_i
        subject to (f * g)[k] ≤ M   for all k
                   g_i ≥ 0

    Returns the optimal `g`.  If SciPy is unavailable or the LP fails,
    a uniform distribution is returned as a safe fallback.
    """
    if linprog is None or toeplitz is None:
        # fallback – uniform distribution of the same size
        n = f.shape[0]
        return np.full(n, 1.0 / n)

    n = f.shape[0]

    # Build the (2n‑1) × n Toeplitz matrix that yields the linear convolution f * g.
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=float)])
    A = toeplitz(col, row, format='csr')        # shape (2n-1, n)

    # Upper‑bound constraints: A @ g ≤ M
    b = np.full(2 * n - 1, M, dtype=float)

    # Objective: maximise Σ g_i  ⇔ minimise –Σ g_i
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n

    # The HiGHS solver is fast for the modest sizes we use.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        g = res.x
        # Numerical safety – enforce non‑negativity and normalise.
        g = np.maximum(g, 0.0)
        if g.sum() > 0:
            g /= g.sum()
        else:
            g = np.full_like(g, 1.0 / n)
        return g
    else:
        # In the unlikely event of failure, fall back to uniform.
        return np.full(n, 1.0 / n)

# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Linear‑programming step (maximise total mass under convolution bound)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve

        maximise   Σ g_i
        subject to (f * g)[k] ≤ M   (for all k)
                   g_i ≥ 0

    If SciPy is unavailable or the LP fails, a uniform distribution is
    returned as a safe fallback.
    """
    if linprog is None or toeplitz is None:
        # Fallback – uniform distribution.
        n = f.shape[0]
        return np.full(n, 1.0 / n)

    n = f.shape[0]

    # Build the (2n‑1)×n Toeplitz matrix that implements the linear convolution f∗g.
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=float)])
    A = toeplitz(col, row, format='csr')          # shape (2n‑1, n)

    # Upper‑bound constraints: A @ g ≤ M
    b = np.full(2 * n - 1, M, dtype=float)

    # Objective: maximise Σ g_i  ⇔ minimise –Σ g_i
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 0:
            g /= g.sum()
        else:
            g = np.full_like(g, 1.0 / n)
        return g
    else:
        # In the unlikely event of failure, fall back to uniform.
        return np.full(n, 1.0 / n)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    performs a small‑scale LP‑based mass relocation, upsamples, then
    refines with projected gradient descent and a final annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Load a possible starter sequence from the global namespace.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Phase 1 – small‑scale LP refinement
    # ------------------------------------------------------------------
    n_lp = 1200
    if starter is not None:
        if starter.size >= n_lp:
            a_small = starter[:n_lp].copy()
            a_small /= a_small.sum()
        else:
            a_small = _upsample(starter, n_lp)
    else:
        a_small = np.random.rand(n_lp)
        a_small /= a_small.sum()

    lp_iters = 7          # number of LP updates
    alpha = 0.25          # convex‑combination weight
    for it in range(lp_iters):
        if time.time() >= deadline:
            break
        M = _conv_self(a_small).max()
        g = _solve_lp_max_sum(a_small, M)
        # Blend and re‑project onto the simplex.
        a_small = (1 - alpha) * a_small + alpha * g
        a_small = _project_simplex(a_small)
        cur_score = 2.0 * n_lp * M      # sum(a_small) = 1
        print(f"[{time.time() - start_time:6.2f}s] LP step {it+1}/{lp_iters}, score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the final problem size.
    # ------------------------------------------------------------------
    n_big = 4800
    a_big = _upsample(a_small, n_big)
    a_big = _project_simplex(a_big)     # re‑normalise after interpolation

    # ------------------------------------------------------------------
    # Phase 3 – long projected‑gradient descent on the large problem.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining < 0:
        remaining = 0.0
    anneal_fraction = 0.20                 # reserve ~20 % for the annealing phase
    grad_deadline = time.time() + remaining * (1 - anneal_fraction)

    a_opt, best_score = _run_gradient_descent_full(
        a_big, n_big,
        grad_deadline, start_time,
        beta=0.9, step0=0.9, max_iter=10**7
    )
    print(f"[{time.time() - start_time:6.2f}s] Gradient descent finished, best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, best_score = _run_annealing(
            a_opt, n_big,
            deadline, start_time,
            best_score, a_opt,
            start_temp=5e-2, end_temp=1e-6,
            steps=600_000
        )
        print(f"[{time.time() - start_time:6.2f}s] Annealing finished, best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and convert to plain Python list.
    # ------------------------------------------------------------------
    a_opt = np.maximum(a_opt, 0.0)
    tot = a_opt.sum()
    if tot > 0:
        a_opt /= tot
    else:
        a_opt = np.full_like(a_opt, 1.0 / a_opt.size)

    return [float(x) for x in a_opt]
