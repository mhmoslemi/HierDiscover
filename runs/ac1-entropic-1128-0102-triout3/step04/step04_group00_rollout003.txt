"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm Overview
------------------
* **Simplex projection** – O(n) Euclidean projection onto
  {x≥0, Σx=1}.
* **Fast convolution** – linear self‑convolution via FFT (real‑FFT).
* **Exact sub‑gradient** – we use the gradient of the *single* maximal
  convolution entry (the true sub‑gradient when the maximum is unique).
  This is O(n) and much cheaper than enumerating all active indices.
* **Projected gradient descent** – Nesterov‑style momentum, step size
  η₀/√t.  Several random restarts at a modest size are performed.
* **Multi‑stage refinement** – the best vector is up‑sampled (linear
  interpolation) to larger dimensions (≈2 k → 8 k) and further
  optimised.  Larger n allows a finer discretisation of the optimal
  continuous shape.
* **Simulated‑annealing fine‑tuning** – mass‑transfer moves (ε transferred
  from a high‑mass index to a random one) with a geometric temperature
  schedule, limited to the remaining budget.

The whole routine respects a user‑supplied time budget (default 1000 s)
and prints progress messages.  The final sequence is normalised to sum = 1
so that `evaluate_sequence` will accept it.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (using the unique maximum)
# ------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) when the maximum is (approximately)
    attained at a single index.  This is O(n) and sufficient for descent.
    """
    n = a.shape[0]
    kmax = int(np.argmax(conv))          # one index attaining the maximum
    idx = np.arange(n)
    i = kmax - idx
    mask = (i >= 0) & (i < n)

    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]           # contribution from ordered pairs

    # diagonal correction: when i == idx (i.e. 2*idx == kmax) we have
    # counted the term a_i*a_i only once in the convolution, so the
    # derivative should be a_i, not 2*a_i.
    diag = mask & (2 * idx == kmax)
    if np.any(diag):
        g[diag] = a[idx[diag]]
    return g


# ------------------------------------------------------------
# Projected gradient descent on the simplex
# ------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on  F(a)=2·n·max(conv(a))  (sum(a)=1).
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient
        grad = _subgradient_max(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,

    Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source index proportionally to its mass (bias towards large entries)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑stage scheme (random restarts → up‑sampling →
    long gradient descent → annealing) and respects the supplied time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run (height_sequence_1)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈1024) with random restarts
    # ------------------------------------------------------------------
    n0 = 1024
    # allocate about 12 % of the total budget (or at least 30 s)
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 4
    for r in range(restarts):
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # randomise a few hyper‑parameters for diversity
        beta = np.random.uniform(0.80, 0.95)
        step0 = np.random.uniform(0.25, 0.45)

        a_opt, score = _run_gradient_descent(
            a0, n0, slice_deadline, start_time,
            beta=beta,
            step0=step0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={score:.6f}")

        if time.time() >= deadline:
            break

    # fallback in the unlikely case that no vector was produced
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upscale to medium size (n≈4096) and longer gradient descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    # spend about 40 % of the remaining budget on this stage
    grad_time1 = max(20.0, 0.40 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_gradient_descent(
        a1, n1, grad_deadline1, start_time,
        beta=0.92,
        step0=0.28,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a large size (n≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    # allocate roughly 35 % of remaining budget to gradient, rest to annealing
    grad_time2 = max(15.0, 0.35 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_gradient_descent(
        a2, n2, grad_deadline2, start_time,
        beta=0.94,
        step0=0.24,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
