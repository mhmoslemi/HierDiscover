"""
Low‑maximum self‑convolution step‑function search.

The algorithm is a pure simplex optimisation based on the exact
sub‑gradient of the ℓ∞‑norm of the self‑convolution.
* The objective is 2·n·max(conv) when the vector is normalised to sum = 1.
* In each iteration we compute the convolution, identify the indices where
  the maximum is attained and obtain the sub‑gradient
      ∇_i = 2·∑_{k∈K} a_{k‑i}    (K = set of maximising indices).
* A multiplicative (exponentiated‑gradient) update
      a_i ← a_i·exp(‑η·∇_i)  is performed and the vector is renormalised.
  This respects the non‑negativity constraint and avoids an explicit Euclidean
  projection.
* A simple simulated‑annealing “mass‑transfer’’ phase is run after the
  gradient descent to polish the best solution found.
* The search proceeds through a schedule of lengths (512 … 16384).  The best
  distribution for a given length is up‑sampled (linear interpolation) and
  used as a warm‑start for the next, larger length.
* The routine obeys the 1000 s time budget and returns the best sequence it
  discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for all lengths)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (linear) using a power‑of‑two FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simple annealing (mass‑transfer) – used for polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 50000) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer annealing: move a small ε from a source index to a
    destination index.  Returns the (possibly) improved sequence and its
    exact objective value.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2   # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the probability simplex
# ----------------------------------------------------------------------


def _optimize_subgradient(a0: np.ndarray,
                          n: int,
                          slice_deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent using the exact sub‑gradient
    of max(a*a).  Returns the best vector encountered (sum = 1) and its
    exact objective value (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    eta0 = 0.6                 # initial step‑size for the exponentiated update

    while time.time() < slice_deadline:
        it += 1

        # self‑convolution and current objective
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        # mask of maximising indices (exact equality is fine – floating noise is rare)
        mask = (conv == max_b).astype(float)          # length 2n‑1

        # sub‑gradient:  ∇_i = 2·∑_{k∈K} a_{k‑i}
        grad_full = _conv_fft(mask, a[::-1])         # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # shape (n,)

        # step‑size schedule (decreasing like 1/√t)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad)

        # renormalise to the probability simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            # fallback: re‑initialise uniformly
            a = np.full(n, 1.0 / n)

        # occasional tiny jitter to shake the iterate
        if it % 500 == 0:
            a += 1e-6 * np.random.rand(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score

    evaluator).  The strategy consists of:
        1. a hierarchy of lengths (512 … 16384),
        2. sub‑gradient optimisation on the simplex,
        3. a short annealing polishing phase,
        4. warm‑starting each larger length by linear up‑sampling.
    """

    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:

            else:

        else:
            starter = None
    except Exception:
        starter = None

    global_a = starter

        fraction = 0.12 / (len(lengths) - idx)  # growing share for larger n
        slice_time = max(5.0, remaining * fraction)
        slice_deadline = time.time() + slice_time

        # --------------------------------------------------------------
        # sub‑gradient optimisation on the simplex
        # --------------------------------------------------------------
        a_opt, score_opt = _optimize_subgradient(a0, n, slice_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={score_opt:.6f}")

        # --------------------------------------------------------------
        # final annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=50000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={score_ref:.6f}")

        # warm start for the next (larger) length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return a plain Python list; the evaluator will handle clamping/normalisation.
    return [float(x) for x in best_seq]
