*   The current vector `a` is a probability vector (sum = 1, a_i ≥ 0).
*   In each iteration we move a small random amount of mass from one
    index to another, keep the sum constraint, and evaluate the new
    objective using a fast FFT based convolution.
*   Moves that improve the score are always accepted; worsening moves are
    accepted with a temperature‑dependent probability.
*   The temperature starts relatively high (to explore the space) and is
    cooled geometrically.
*   Several random restarts are performed – each restart is a fresh
    Dirichlet sample – and the best sequence seen over the whole run is
    returned.

The routine respects the 1000 s budget, prints occasional progress
information and finally returns the best list of non‑negative heights
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return the maximum entry of the linear auto‑convolution of a.

    The convolution is performed with an FFT of length a power of two
    (fast and numerically stable).  The returned value is a plain Python
    float.
    """
    n = a.shape[0]
    # length needed for a linear convolution of two length‑n vectors
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next power of two
    # forward transform (real → complex)
    A = np.fft.rfft(a, n=fft_len)
    # element‑wise product = convolution in frequency domain
    C = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(C.max())


def _score(a: np.ndarray):
    """Objective used by `evaluate_sequence` (no clamping needed)."""
    n = a.shape[0]
    max_b = _conv_max_fft(a)
    # the vector is already normalised to sum == 1, therefore the
    # denominator in `evaluate_sequence` equals 1.
    return 2.0 * n * max_b


def _random_simplex(num: int, rng: np.random.Generator):
    """Draw a point uniformly from the probability simplex of size `num`."""
    # Dirichlet(1,…,1) is uniform on the simplex.
    return rng.dirichlet(np.ones(num))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a step‑function) that tries to
    minimise ``evaluate_sequence``.  The method runs for at most
    ``budget_s`` seconds and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin (the environment kills the process a few
    # seconds after the deadline)
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem size --------------------------------------------------
    # ------------------------------------------------------------------
    # A longer vector gives a richer search space but makes each
    # convolution slower.  3000 is a good compromise for a 1000 s budget.
    n = 3000

    # Best solution found so far.
    best_a = np.full(n, 1.0 / n)      # uniform distribution (score = 2)
    best_val = _score(best_a)

    print(f"[{time.time() - start:6.2f}s] start – n={n}, initial score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated annealing schedule ---------------------------------
    # ------------------------------------------------------------------
    # Temperature schedule parameters – chosen empirically.
    T0 = 0.2                # start temperature (acceptance probability)
    Tf = 1e-4               # final temperature (almost greedy)
    total_iters = 0

    # We will perform a number of independent restarts.  Each restart runs
    # a fixed budget of iterations (here about 0.25 s per restart, which
    # yields ~4000 restarts overall on a typical machine).  The loop
    # terminates as soon as the global deadline is reached.
    while time.time() < deadline:
        # ---- a fresh start -------------------------------------------------
        a = _random_simplex(n, rng)   # random probability vector
        cur_val = _score(a)

        # If the random start is already better than the global best keep it.
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time() - start:6.2f}s] new global start score={cur_val:.6f}")

        # ---- annealing on this start --------------------------------------
        # We use a geometric cooling that ends at Tf after a fixed number of
        # inner iterations.  The number of inner steps is proportional to `n`
        # (so larger vectors get more steps).
        inner_iters = max(5000, n // 2)   # at least a few thousand steps
        for it in range(inner_iters):
            total_iters += 1
            # update temperature
            tau = it / inner_iters
            T = T0 * (Tf / T0) ** tau   # geometric cooling

            # propose a random mass redistribution
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # choose a fraction of the source mass (the step size shrinks with T)
            frac = rng.random() * min(0.2, T * 10)   # up to 20 % initially
            delta = frac * a[i]   # amount moved from i to j
            if delta <= 0.0:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # Numerical safety – keep non‑negativity
            if a_new[i] < 0.0:
                continue

            new_val = _score(a_new)

            # Acceptance test
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
                a = a_new
                cur_val = new_val
                # record a new global best if we have improved
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    elapsed = time.time() - start
                    print(f"[{elapsed:6.2f}s] iter={total_iters:6d}  global best={best_val:.6f}")

            # keep an eye on the overall time budget
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 3. Final tidy‑up ------------------------------------------------
    # ------------------------------------------------------------------
    print(f"[{time.time() - start:6.2f}s] finished – total iterations={total_iters}"
          f", best score={best_val:.6f}")

    # The evaluator will re‑clip the vector to [0,1000] and normalise the sum,
    # therefore we simply return the raw probability vector.
    return [float(x) for x in best_a]
