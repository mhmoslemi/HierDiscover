def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The routine runs for about 1000 s
and uses a mixture of:

* a few cheap deterministic initialisations (previous best,
  power‑law, random Dirichlet);
* a projected (additive) sub‑gradient descent on the piece‑wise‑linear
  objective `max(conv)`.  The sub‑gradient is computed only on an
  “active set’’ of the largest convolution entries (default 30–50);
* a greedy mass‑transfer step (move a fraction of mass from the index
  with the largest sub‑gradient to the index with the smallest);
* occasional random perturbations for diversification;
* a final cheap hill‑climbing phase that swaps small amounts of mass
  between random positions.

All vectors are kept on the probability simplex (sum = 1) so that the
objective evaluated inside the optimiser equals the value returned by
`evaluate_sequence`.  The algorithm is completely self‑contained,
uses only NumPy and the standard library and respects the 1 CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# low‑level helpers (top‑level – no closures)
# ----------------------------------------------------------------------


def _score_of_seq(a: np.ndarray) -> float:
    """Objective value for a vector that already sums to 1.
    Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray):
    """Linear auto‑convolution and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """Indices whose convolution value is within `rel_eps` of the maximum.
    If fewer than `max_add` are found the largest entries are added."""
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """Sub‑gradient of `max(conv)` restricted to the active set K.
    For each active index k we add the contribution 2·a_{k‑i} to component i."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Local optimiser (projected sub‑gradient descent)
# ----------------------------------------------------------------------


def _local_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float):
    """
    Run a projected sub‑gradient optimisation starting from ``start_vec``.
    Returns the best vector found (sum‑to‑1) and its score.
    """
    a = start_vec.copy()
    best_local = a.copy()
    best_local_score = _score_of_seq(best_local)

    it = 0
    while time.time() < deadline:
        it += 1
        conv, max_val = _conv_and_max(a)
        cur_score = 2.0 * n * max_val

        # record local improvement
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local = a.copy()

        # ----------------------------------------------------------
        # (1) active‑set sub‑gradient
        # ----------------------------------------------------------
        K = _active_indices(conv, max_val, rel_eps=0.001, max_add=50)
        g = _gradient_active(a, K)

        # ----------------------------------------------------------
        # (2) greedy mass transfer (large step)
        # ----------------------------------------------------------
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        moved = False
        if i_max != i_min and a[i_max] > 0:
            # try a few fractions of the source mass
            for frac in (0.10, 0.05, 0.02):
                delta = min(frac * a[i_max], a[i_max])
                cand = a.copy()
                cand[i_max] -= delta
                cand[i_min] += delta
                # re‑normalise (projection onto the simplex)
                cand = _project_simplex(cand)
                conv2, max2 = _conv_and_max(cand)
                new_score = 2.0 * n * max2
                if new_score < cur_score - 1e-12:
                    a = cand
                    moved = True
                    break

        # ----------------------------------------------------------
        # (3) fallback additive step (projected sub‑gradient)
        # ----------------------------------------------------------
        if not moved:
            eta = 0.5 / math.sqrt(it + 1)          # decaying step size
            cand = a - eta * g
            cand = _project_simplex(cand)
            conv2, max2 = _conv_and_max(cand)
            new_score = 2.0 * n * max2
            if new_score < cur_score:
                a = cand

        # ----------------------------------------------------------
        # (4) occasional random perturbation for exploration
        # ----------------------------------------------------------
        if it % 5000 == 0:
            i, j = np.random.randint(0, n, size=2)

                delta = eps * a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)
                conv2, max2 = _conv_and_max(cand)
                new_score = 2.0 * n * max2
                if new_score < cur_score - 1e-12:
                    a = cand

    return best_local, best_local_score

    # --------------------------------------------------------------
