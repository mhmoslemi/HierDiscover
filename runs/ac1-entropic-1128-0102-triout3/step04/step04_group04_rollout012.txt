"""
Propose a low‑value step function for the hidden ``evaluate_sequence``.

Algorithm (≈1000 s budget)

1. **Initialisation** – try a variety of simple parametric families
   (power‑law, exponential, mixed power‑law) and keep the best normalised
   vector.  If the global variable ``height_sequence_1`` exists it is also
   tried as a starting point.

2. **Main optimiser** – repeats the following while time permits
   * compute the auto‑convolution and the active set of (almost) maximal
     entries;
   * obtain a sub‑gradient of the ``max(conv)`` functional;
   * **greedy mass transfer** – move a relatively large chunk (up to 20 %)
     of mass from a high‑gradient index toward a low‑gradient one;
   * if the greedy step fails, perform an **Adam‑style projected
     gradient step** on the simplex (Euclidean projection);
   * occasionally accept a slightly worse move (simulated‑annealing) to
     escape shallow minima.

3. **Random polishing** – cheap random mass‑transfer moves (up to 4 % of
   source mass) are applied until the time budget expires.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the returned list can be fed directly to ``evaluate_sequence``.
Progress and current best scores are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of a (using FFT) and its maximum value."""
    n = a.shape[0]
    # length of FFT – next power of two ≥ 2n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, np.max(conv)


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices that are (almost) maximal.
    All indices with value >= max_val - rel_eps*max_val are kept;
    if fewer than `max_add` are found, the globally largest entries are
    added to reach `max_add` indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of ``max(conv)`` for the active set K.
    For each active convolution index k the contribution to component i is
    2·a_{k‑i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv) for a vector already normalised to sum=1."""
    n = a.shape[0]
    conv, max_val = _conv_and_max(a)
    return float(2.0 * n * max_val)

    score.  The routine respects a ~1000 s wall‑clock budget and prints
    intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # 1. Choose problem size and obtain a first feasible vector.
    # --------------------------------------------------------------
    n = 5000                     # length of the vector (kept constant)

    # --------------------------------------------------------------

        # ----------------------------------------------------------

            if i_min == i_max or current[i_max] <= 0.0:

    # --------------------------------------------------------------

    if time.time() < deadline:
