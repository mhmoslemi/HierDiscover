        fft_a = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[:2*n - 1]
        k_max = int(np.argmax(conv))
        # gradient
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >=0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        # projection onto simplex
        # using standard algorithm:
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n+1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)
        # evaluate
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[subgrad iter {it}] new best {val:.6f}")
        # occasional random mass transfer
        if np.random.rand() < 0.01:
            i, j_idx = np.random.randint(0, n, size=2)
            if a[i] > 1e-9:
                delta = a[i] * 0.1 * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j_idx] += delta
                # projection onto simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_a = cand.copy()
                    a = cand
        it += 1
    return best_a

"""
search for a low‑autocorrelation step function.

The procedure has three stages:

1) **Random‑support exploration** – many candidates with a large support
   (≈ 80‑90 % of the positions) are generated.  The weights on the support
   are either uniform or drawn from a Dirichlet distribution.
   The best candidate seen so far is kept.

2) **Exponentiated‑gradient refinement** – starting from the best candidate,
   a mirror‑descent (multiplicative) update is applied to the entries that
   contribute most to the current maximal convolution value.  The update
   reduces those entries and automatically redistributes the mass,
   keeping the vector on the probability simplex.  Fast FFT‑based
   convolution makes it possible to perform tens of thousands of iterations
   inside the 1000 s budget.

3) **Fallback sub‑gradient optimiser** – if time is still left and the
   target bound (1.503) has not been reached, a classic projected
   sub‑gradient descent is run for a short while.

All arithmetic is performed with double precision; the final list of
floats is returned ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (fast, self‑contained – no safety checks)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Return the value of `evaluate_sequence` without the security checks.
    The vector is assumed to be non‑negative and its ℓ₁‑norm is not zero.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def compute_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x≥0, sum(x)=1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (multiplicative mirror descent)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Refine a candidate with a multiplicative (exponentiated‑gradient) update.
    The algorithm always stays on the simplex.
    """
    a = a_init.copy()
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_val = fast_score(a)

    start = time.time()
    it = 0
    eta0 = 2.0                     # initial step size
    decay = 1e-5                   # decay for the step size

    while time.time() - start < time_limit - 0.1:
        # convolution and most‑violated index
        fft_a = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # gradient of the active quadratic form (only entries that appear)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # multiplicative update
        eta = eta0 / (1.0 + it * decay)
        a = a * np.exp(-eta * grad)

        # re‑normalise to keep sum = 1
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # keep the best ever seen
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[expgrad iter {it}] new best {val:.6f}")

        # occasional random mass transfer to escape tiny plateaux
        if np.random.rand() < 0.005:
            i, jdx = np.random.randint(0, n, size=2)
            if a[i] > 1e-12:
                delta = a[i] * 0.1 * np.random.rand()
                a[i] -= delta
                a[jdx] += delta
                a /= a.sum()

        # early exit if we already beat the target
        if best_val <= 1.503:
            break
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (fallback, same as the reference code)
# ----------------------------------------------------------------------


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Classic projected sub‑gradient descent on the max‑convolution objective.
    """
    a = a_init.copy()
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_val = fast_score(a)

    start = time.time()
    it = 0
    eta0 = 0.30
    eta_decay = 1e-5
    idx = np.arange(n)

    while time.time() - start < time_limit - 0.1:
        # active index of the convolution
        fft_a = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size and Euclidean projection onto the simplex
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = simplex_projection(v)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, jdx = np.random.randint(0, n, size=2)
            if a[i] > 1e-12:
                delta = a[i] * 0.1 * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[jdx] += delta
                cand = simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # keep the best ever seen
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[subgrad iter {it}] new best {cur_val:.6f}")

        # early exit if target reached
        if best_val <= 1.503:
            break

        it += 1

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (length 1500) whose evaluation
    score is as low as possible.  The algorithm mixes a large random‑support
    search with an exponentiated‑gradient refinement and a final sub‑gradient
    fallback.  It is fully deterministic given the supplied seed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = 1500
    target_score = 1.503

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) start from a previously stored construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        if prev.shape[0] == n:
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) random‑support search (uniform or Dirichlet weights)
    # ------------------------------------------------------------------
    support_sizes = [1400, 1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
    attempts_per_size = 3000

    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[search] trying support size {m}")
        attempts = 0
        while attempts < attempts_per_size and time.time() < deadline:
            attempts += 1

            # sample a support set
            S = np.random.choice(n, size=m, replace=False)

            if np.random.rand() < 0.5:
                # uniform weight on the support
                a = np.zeros(n, dtype=np.float64)
                a[S] = 1.0 / m
            else:
                # Dirichlet (gamma) weights on the support
                w = np.random.gamma(shape=1.0, scale=1.0, size=m)
                w /= w.sum()
                a = np.zeros(n, dtype=np.float64)
                a[S] = w

            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} attempt={attempts} score={val:.6f}")

            if val <= target_score:
                print("[success] target reached in random search")
                return best_seq.tolist()

        print(f"[search] size {m} finished after {attempts} attempts, "
              f"best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print(f"[expgrad] starting optimisation, {remaining:.1f}s left")
        if best_seq is None:
            best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = exp_gradient_optimize(best_seq, remaining)
        best_val = fast_score(best_seq)
        print(f"[expgrad] best after refinement {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) fallback sub‑gradient optimiser if still above target
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_val > target_score and remaining > 2.0:
        print(f"[fallback] running sub‑gradient for remaining {remaining:.1f}s")
        if best_seq is None:
            best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = subgradient_optimize(best_seq, remaining)
        best_val = fast_score(best_seq)
        print(f"[fallback] final fallback score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) final safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] returning sequence with score {best_val:.6f}")
    return best_seq.tolist()
