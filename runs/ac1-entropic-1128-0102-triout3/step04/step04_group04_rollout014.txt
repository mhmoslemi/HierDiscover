    best = cur.copy()

        # or use exponential cooling

        i, j = random distinct
        if cur[i] == 0: continue

            cur_score = new_score

    np.random.seed(seed)

    start = time.time()

    best_seq = None
    best_score = float('inf')

    lengths = [4000, 5000, 8000, 10000]
    # allocate equal budget per length
    per_len_budget = (deadline - start) / len(lengths)
    for n in lengths:
        if time.time() > deadline: break
        # prepare initial sequences
        init_candidates = []
        # try power law exponents
        p_grid = np.linspace(0.2, 0.8, 13)  # 13 values

            a = (idx+1) ** (-p)

            init_candidates.append(a)
        # also add uniform

        # random permutations of a base shape
        base = init_candidates[0]
        for _ in range(3):
            perm = np.random.permutation(n)

"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
1. Pick a few problem sizes (n = 3000, 5000, 8000, 10000).  
2. For each size:
   * Initialise with a power‑law shape a_i ∝ (i+1)^(-p) for several p,
     also try a uniform vector and a few Dirichlet samples.
   * Randomly permute the best initial vector a few hundred times and keep the
     best ordering (permutations dramatically affect the convolution peak).
   * Run a long local optimisation loop until the time budget is exhausted:
        – compute the auto‑convolution and its maximal value;
        – build a modest active set of “almost‑maximal’’ indices;
        – obtain a sub‑gradient of the maximal convolution entry;
        – try a greedy mass‑transfer from the index with the largest
          sub‑gradient to the one with the smallest;
        – if greedy fails, perform a projected gradient descent step
          (Euclidean step on the simplex) with a decaying step size;
        – every 1000 iterations also try a large random transfer to escape
          shallow local minima.
   * Keep the best vector seen for the current length.
3. Keep the globally best vector across all lengths and return it.

All vectors are kept on the probability simplex (sum = 1) – the scaling does
not affect the objective `evaluate_sequence`.  The implementation uses FFT‑based
convolution (O(n log n)) and respects the 1000 s wall‑clock limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of a (linear, zero‑padded) and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _score_seq(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – the quantity minimised by `evaluate_sequence`."""
    n = a.shape[0]
    _, mx = _conv_and_max(a)
    return 2.0 * n * mx


def _active_indices(conv: np.ndarray, mx: float,
                    rel_eps: float = 0.001, max_add: int = 30) -> np.ndarray:
    """Indices whose convolution value is within rel_eps of the maximum."""
    eps = mx * rel_eps
    K = np.where(conv >= mx - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _subgradient(a: np.ndarray, K: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) corresponding to the active index set K."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _local_search(initial: np.ndarray, n: int,
                  deadline: float, start_time: float):
    """
    Intensive local optimisation for a fixed length `n` until `deadline`.
    Returns the best vector seen and its score.
    """
    cur = initial.copy()
    cur_score = _score_seq(cur)
    best = cur.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # convex‑like descent step
        # ------------------------------------------------------------------
        conv, mx = _conv_and_max(cur)
        cur_score = 2.0 * n * mx

        if cur_score < best_score - 1e-12:
            best = cur.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:6.2f}s] n={n:5d} it={it:6d} "
                  f"new best score={best_score:.6f}")

        # sub‑gradient from (almost) active constraints
        K = _active_indices(conv, mx, rel_eps=0.001, max_add=30)
        g = _subgradient(cur, K)

        # --------------------------------------------------------------
        # (a) greedy mass transfer from high to low sub‑gradient entry
        # --------------------------------------------------------------
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        improved = False
        if i_src != i_dst and cur[i_src] > 1e-15:
            # move up to 10 % of source mass
            delta = min(0.10 * cur[i_src], cur[i_src])
            cand = cur.copy()
            cand[i_src] -= delta
            cand[i_dst] += delta
            cand_score = _score_seq(cand)
            if cand_score < cur_score - 1e-12:
                cur = cand
                cur_score = cand_score
                improved = True
                continue

        # --------------------------------------------------------------
        # (b) occasional large random move (helps escape plateaus)
        # --------------------------------------------------------------
        if it % 1000 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j and cur[i] > 1e-15:
                frac = 0.30 * np.random.rand()   # up to 30 % of source mass
                delta = min(frac * cur[i], cur[i])
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = _score_seq(cand)
                if cand_score < cur_score - 1e-12:
                    cur = cand
                    cur_score = cand_score
                    improved = True
                    continue

        # --------------------------------------------------------------
        # (c) random swap of two entries (often lowers the peak)
        # --------------------------------------------------------------
        if not improved:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                cand = cur.copy()
                cand[i], cand[j] = cand[j], cand[i]
                cand_score = _score_seq(cand)
                if cand_score < cur_score - 1e-12:

            a_new = _project_simplex(a_new)

        # ------------------------------------------------------------------
        # 2.  Build a collection of cheap initial candidates.
        # ------------------------------------------------------------------
        candidates = []

        # a) power‑law shapes with a sweep of exponents
        p_vals = np.linspace(0.20, 0.80, 13)   # 13 values
        indices = np.arange(n, dtype=float) + 1.0
        for p in p_vals:
            a = indices ** (-p)
            a /= a.sum()
            candidates.append(a)

        # b) uniform vector
        candidates.append(np.full(n, 1.0 / n))

        # c) a few random Dirichlet samples (fast, diverse)
        for _ in range(5):
            a = np.random.dirichlet(np.ones(n))
            candidates.append(a)

        # d) if we already have a global best of the same length, use it
        if global_best_seq is not None and len(global_best_seq) == n:
            candidates.append(global_best_seq.copy())

        # ------------------------------------------------------------------
        # 3.  Pick the best among the cheap candidates.
        # ------------------------------------------------------------------
        cur = None
        cur_score = float('inf')
        for a in candidates:
            sc = _score_seq(a)
            if sc < cur_score:
                cur = a.copy()
                cur_score = sc

        # ------------------------------------------------------------------
        # 4.  Random permutation sweep (convolution is very sensitive to ordering).
        # ------------------------------------------------------------------
        for _ in range(200):
            if time.time() >= deadline:
                break
            perm = np.random.permutation(n)
            a_perm = cur[perm]
            sc = _score_seq(a_perm)
            if sc < cur_score:
                cur = a_perm
                cur_score = sc

        # ------------------------------------------------------------------
        # 5.  Intensive local optimisation for this length.
        # ------------------------------------------------------------------
        best_local, best_local_score = _local_search(cur, n, len_deadline, start_time)

        # ------------------------------------------------------------------
        # 6.  Record a new global best if we have improved.
        # ------------------------------------------------------------------
        if best_local_score < global_best_score:
            global_best_seq = best_local.copy()
            global_best_score = best_local_score
            print(f"[{time.time() - start_time:6.2f}s] ** NEW GLOBAL BEST ** "
                  f"n={n} score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Fallback (should never be needed).
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = np.full(100, 1.0 / 100)

    # Return as plain Python floats (the caller will feed this to evaluate_sequence)
    return [float(x) for x in global_best_seq]
