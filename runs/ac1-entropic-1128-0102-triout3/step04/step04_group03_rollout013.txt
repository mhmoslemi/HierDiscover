"""
Improved search for a low‑peak step‑function.

The algorithm works in four phases:

1.  Initialise from the supplied global `height_sequence_1` (if any) or from a
    random simplex point.  The sequence is always normalised to sum = 1.

2.  Progressive up‑sampling – the current best shape is linearly interpolated
    to larger lengths (800 → 2400 → 4800 → 9600).  Larger `n` gives more
    degrees of freedom and typically lowers the peak of the self‑convolution.

3.  At each length a tiny *active‑set LP* is solved:
        maximise   Σ g_j
        subject to   g_j ≥ 0  and  (f * g)_k ≤ M  for every active index k,
    where `M = max(f * f)`.  Only those convolution entries that are within a
    relative tolerance of the maximum are kept as constraints; any violated
    constraints discovered after a solve are added iteratively.  The resulting
    `g` is mixed with the current `f` (mixing factor 0.2) and projected back to
    the simplex.  This step moves mass from the current peak towards a flatter
    distribution and often yields a sizable drop in the objective.

4.  A Nesterov‑style projected‑gradient descent on the simplex refines the
    shape, followed by a cheap simulated‑annealing walk that swaps small amounts
    of mass between two positions.  The whole routine respects the 1000‑second
    budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# optional imports – SciPy is used for the small LP, but the code will still run
# without it (the LP step will simply be skipped).
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper 1: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1} (O(n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (length len(a)+len(b)-1)."""
    n = a.shape[0]
    m = b.shape[0]
    conv_len = n + m - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Helper 3: Sub‑gradient of the max‑convolution (averaged over active indices)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all active positions)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if act.size:
        g /= float(act.size)
    return g


# ----------------------------------------------------------------------
# Helper 4: Gradient descent on the simplex (Nesterov momentum)
# ----------------------------------------------------------------------
def _run_gradient_descent_full(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              beta: float = 0.9,
                              step0: float = 0.6,
                              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex.
    Returns the best point encountered and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper 5: Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a small ε from one index to another."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper 6: Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper 7: Small active‑set linear programme that pulls mass away from the peak
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               rel_tol: float = 1e-3,
               mix_factor: float = 0.2,
               max_lp_iters: int = 5) -> tuple[np.ndarray, bool]:
    """
    Solve a tiny LP that tries to maximise the total mass of a new vector g
    while keeping (a * g) below the current peak M.
    Returns a new candidate a' and a flag indicating whether an improvement
    (strictly lower max‑convolution) was achieved.
    """
    if not _SCIPY_AVAILABLE:
        return a, False

    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()

    # initial active indices – those close to the current maximum
    active = set(np.where(conv >= M * (1.0 - rel_tol))[0])

    for _ in range(max_lp_iters):
        # Build sparse matrix for the current active set
        rows, cols, data = [], [], []
        act_list = sorted(active)
        for row_idx, k in enumerate(act_list):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_vals = np.arange(lo, hi + 1, dtype=int)
            j_vals = k - i_vals
            rows.extend([row_idx] * len(i_vals))
            cols.extend(j_vals.tolist())
            data.extend(a[i_vals].tolist())

        A = sp.csr_matrix((data, (rows, cols)), shape=(len(act_list), n))
        b = np.full(len(act_list), M, dtype=float)
        c = -np.ones(n)                     # maximise Σ g → minimise -Σ g
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort the refinement
            return a, False

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            return a, False

        # Scale g so that it again sums to 1
        scale = 1.0 / sum_g
        g_scaled = g * scale

        # Mix a and the new direction
        a_new = (1.0 - mix_factor) * a + mix_factor * g_scaled
        a_new = _project_simplex(a_new)

        # Compute the new peak
        new_conv = _conv_self(a_new)
        new_M = new_conv.max()

        if new_M < M - 1e-12:                     # strict improvement
            return a_new, True

        # If not improved check feasibility of the candidate g_scaled
        # (a * g_scaled) must not violate any constraint that we have omitted.
        viol = _conv(a, g_scaled) - M
        bad_idx = np.where(viol > 1e-12)[0]
        if bad_idx.size == 0:
            # feasible but not improving – stop
            return a, False

        # add the violating indices to the active set and repeat
        active.update(bad_idx.tolist())

    return a, False

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided `evaluate_sequence`.  The routine respects the time budget,
    performs progressive up‑sampling, a short active‑set LP polishing,
    projected gradient descent and a final annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:                                   # pragma: no cover
        starter = None

    # ------------------------------------------------------------------
    # Stage definitions – progressively larger lengths.
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # feel free to adjust if time runs out
    # target 15 % of the total budget for final annealing
    anneal_budget = max(30.0, 0.15 * (deadline - start_time))
    remaining_budget = (deadline - start_time) - anneal_budget
    total_n = sum(stage_lengths)
    # allocate time proportionally to n (larger n gets a bit more)
    stage_times = [max(5.0, remaining_budget * (n / total_n)) for n in stage_lengths]

    for n, stage_time in zip(stage_lengths, stage_times):
        if time.time() >= deadline:
            break

        stage_deadline = time.time() + stage_time

        # initialise at this resolution
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # ------------------------------------------------------------------
        # 1️⃣  LP‑based polishing (short, active‑set only)
        # ------------------------------------------------------------------
        lp_end = time.time() + min(0.2 * stage_time, 20.0)   # at most a few seconds
        while time.time() < lp_end:
            a_new, improved = _lp_refine(a)
            if not improved:
                break
            a = a_new
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Gradient descent on the simplex
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_gradient_descent_full(
            a, n, stage_deadline, start_time,
            beta=0.9, step0=0.6, max_iter=10**7
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD stage improve: n={n}, score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # Update the remaining gradient budget for the next stages
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        remaining_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – make a fine‑grained stochastic walk.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        # The remaining time determines how many annealing moves we can afford.
        remaining_seconds = deadline - time.time()
        # Rough conversion: ~2000 moves per second is typical.
        est_steps = max(50000, int(remaining_seconds * 3000))
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=est_steps
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – never return an empty list.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
