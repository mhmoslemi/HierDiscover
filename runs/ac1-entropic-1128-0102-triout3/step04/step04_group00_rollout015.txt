def _run_adam(a0, n, deadline, start_time, 
              lr=0.1, beta1=0.9, beta2=0.999,
              eps=1e-8, max_iter=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t=0
    while time.time() < deadline and t < max_iter:
        t+=1
        grad, max_val = _gradient_maxconv(a)
        grad_obj = 2.0 * n * grad

        m = beta1*m + (1-beta1)*grad_obj
        v = beta2*v + (1-beta2)*(grad_obj**2)

        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # evaluate objective
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: iter={t}, score={score:.6f}")

"""
Propose a step‑function (non‑negative list) that minimises
`evaluate_sequence`.

Algorithm
---------
1. Start from the provided `height_sequence_1` (if available) and
   up‑sample it to a target length `N = 4800`.  If the starter is not
   present we begin with a uniform distribution.

2. **Fast Adam‑style projected optimisation**  
   – In each iteration the full self‑convolution `c = a*a` is obtained
     with an FFT (O(N log N)).  
   – The objective is `F(a)=2·N·max(c)`.  Its sub‑gradient is the sum of
     the gradients of all (almost) active convolution entries.  The
     gradient is computed in O(N) by a single cross‑correlation
     (`np.correlate`).  
   – Adam updates are performed on the unconstrained vector and the
     result is projected onto the probability simplex (`Σa_i=1`,
     `a_i≥0`).  
   – The loop runs as long as the allocated time budget allows
     (≈70 % of the total time).  The best vector seen so far is stored.

3. **Simulated‑annealing fine‑tuning**  
   With the remaining time a cheap mass‑transfer annealing is applied:
   a random amount of probability mass is moved from a source index
   (chosen proportionally to its current weight) to a destination
   index.  Moves are accepted with a Metropolis rule.  The best sequence
   discovered during annealing is kept.

4. The final vector is normalised to sum = 1 and returned as a plain
   Python list of floats.

The whole routine respects the supplied `budget_s` (default 1000 s) and
prints progress information.  It is deliberately self‑contained and
uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – optionally smoothed with a tolerance
# ----------------------------------------------------------------------
def _grad_maxconv(a: np.ndarray, conv: np.ndarray,
                  rel_tol: float = 1e-6) -> np.ndarray:
    """
    Compute the gradient of max(conv) w.r.t. a.
    All convolution entries with value >= max·(1‑rel_tol) are treated as
    active (softened sub‑gradient).  The gradient has O(N) cost.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    # Binary mask of active shifts
    act = (conv >= max_val * (1.0 - rel_tol)).astype(float)

    # Linear cross‑correlation of the mask with a
    # np.correlate returns length len(act)+len(a)-1, with lag = -(len(a)-1)…len(act)-1
    corr = np.correlate(act, a, mode='full')
    offset = n - 1                     # index of lag 0
    corr_i = corr[offset:offset + n]   # correlations for lags 0,…,n‑1

    # Diagonal correction (pairs i=j contribute only once)
    diag = act[2 * np.arange(n)]      # safe because 2*i ≤ 2n‑2

    grad = 2.0 * corr_i - diag * a
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimisation of 2·N·max(conv)  (N = len(a))
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              lr: float = 0.05,
              beta1: float = 0.90,
              beta2: float = 0.999,
              eps: float = 1e-8,
              rel_tol: float = 1e-6,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: iter={it:,}, score={score:.6f}")

        # gradient of the objective (2·n·max(conv))
        grad = _grad_maxconv(a, conv, rel_tol=rel_tol)
        grad_obj = 2.0 * n * grad

        # Adam moments
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing.  Returns the best sequence encountered.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportionally to current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30   # move at most 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate new candidate
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the time budget (`budget_s`) and returns the best
    sequence found (as a list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.6

    # ------------------------------------------------------------------
    # Initialise from the historic construction if possible
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Target length (chosen as the length that gave the best historic bound)
    # ------------------------------------------------------------------
    N = 4800

    if starter is not None:
        a0 = _upsample(starter, N)
    else:
        a0 = np.full(N, 1.0 / N)           # uniform fallback

    best_seq = a0.copy()
    best_score = float('inf')
    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (≈70 % of the time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        grad_deadline = time.time() + 0.70 * remaining
        a_opt, score_opt = _run_adam(
            a0, N,
            grad_deadline, start_time,
            lr=0.05,
            rel_tol=1e-6
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase‑1 improve: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Fine‑grained simulated annealing (rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _run_annealing(
            best_seq, N,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000
        )
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={anneal_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
