"""
Propose a low‑value step function for `evaluate_sequence`.

Algorithm
---------
1. **Initialisation**  
   Use the previously found construction (`height_sequence_1`) when it is
   available, otherwise fall back to a power‑law shape of length 5 000.

2. **Continuation‑gradient descent**  
   Minimise a *smoothed* version of the true objective  
   `f(a) = 2·n·max(conv(a,a))`.  
   The smoothing is a soft‑max with a temperature parameter `α`:
       `softmax_α(conv) = (log Σ exp(α·conv))/α`.  
   For a fixed `α` we compute the exact gradient of the smooth objective
   (the gradient of `softmax_α` is the corresponding soft‑max probability
   vector).  The gradient w.r.t. `a` is obtained by a single FFT‑based
   correlation.  An exponentiated‑gradient (mirror‑descent) update on the
   simplex is performed, with an Adam‑style adaptive step‑size.  The
   smoothing parameter is increased in stages `[5, 20, 80, 200]` – a classic
   continuation method that first moves quickly on a very smooth landscape
   and then refines on the true non‑smooth objective.

3. **Random hill‑climbing**  
   After the continuation phase a cheap local search moves a small
   fraction of mass between random positions and accepts strictly improving
   moves.

All updates keep the vector non‑negative and normalised, so the final
sequence can be passed directly to `evaluate_sequence`.  The routine stops
shortly before the wall‑clock budget expires and prints intermediate
improvements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a non‑negative vector using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _score_of_seq(a: np.ndarray) -> float:
    """True objective 2·n·max(conv(a,a)) for a vector that sums to 1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _smooth_obj_and_grad(a: np.ndarray, alpha: float):
    """
    Returns three items for a given probability vector `a`
    (sum(a)=1) and a smoothing parameter `alpha` (>0):
        * smooth objective value 2·n·softmax_α(conv)
        * gradient w.r.t. `a` of that smooth objective
        * the *true* max‑convolution value (needed for reporting)
    """
    n = a.shape[0]

    # ----- convolution -------------------------------------------------
    conv = _conv_fft(a)                     # length 2n‑1

    # ----- soft‑max over convolution entries ----------------------------
    alpha_conv = alpha * conv
    max_ac = np.max(alpha_conv)
    exp_shifted = np.exp(alpha_conv - max_ac)   # stable
    sum_exp = exp_shifted.sum()
    p = exp_shifted / sum_exp                    # soft‑max probabilities
    approx_max = (np.log(sum_exp) + max_ac) / alpha

    # ----- gradient of soft‑max term ------------------------------------
    # correlation of p with the reversed a (g_i = Σ_j p_{i+j} a_j)
    a_rev = a[::-1]
    len_corr = len(p) + n - 1                     # = 3n‑2
    fft_len = 1 << (len_corr.bit_length())
    P = np.fft.rfft(p, n=fft_len)
    AR = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(P * AR, n=fft_len)[:len_corr]
    # slice = corr[n‑1 : n‑1+n]  (size n)
    slice_vals = corr[n - 1: n - 1 + n]

    # term for the diagonal (k = 2i) appears only once in the true gradient
    idx_2i = 2 * np.arange(n)
    p2i = p[idx_2i]

    # g_i = 2·Σ_j p_{i+j} a_j – p_{2i}·a_i
    g = 2.0 * slice_vals - p2i * a

    # gradient of the *smooth* objective (factor 2·n from the definition)
    grad_a = 2.0 * n * g

    # true max‑convolution (needed for the actual score)
    true_max = conv.max()

    # smooth objective value
    smooth_obj = 2.0 * n * approx_max

    return smooth_obj, grad_a, true_max

                    break

            # ---- Adam‑style adaptive exponentiated‑gradient update ----------
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # update in log‑space (mirror descent on the simplex)
            delta = base_eta * m_hat / (np.sqrt(v_hat) + eps)
            a = a * np.exp(-delta)          # remain non‑negative
            a_sum = a.sum()
            if a_sum == 0.0:                # numerical safeguard
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

        if time.time() > deadline or best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3. Final cheap random hill‑climbing (mass swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        # a very small number of hill‑climb steps keeps the runtime low
        hill_iters = 15000
        for _ in range(hill_iters):
            if time.time() > deadline:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # move up to 4 % of the source mass
            delta = a[i] * np.random.rand() * 0.04
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = _score_of_seq(cand)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:6.2f}s] Random hill‑climb "
                      f"→ score={best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # 4. Return the best found vector as plain Python floats
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
