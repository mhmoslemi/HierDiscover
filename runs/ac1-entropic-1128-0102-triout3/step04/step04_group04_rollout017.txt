def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # initialize best sequence
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.sum() == 0:
            raise ValueError
        # normalize to sum 1
        init = init / init.sum()
        n = init.shape[0]
    except Exception:
        n = 5000
        init = None

    best_seq = None
    best_score = float('inf')
    # stage: candidate generation using various param families
    def try_seq(a):
        nonlocal best_seq, best_score
        sc = _score_of_seq(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start:6.2f}s] New best {sc:.6f}")

    if init is not None:
        try_seq(init)

    # param families
    param_families = []
    # power law
    p_vals = np.concatenate([np.linspace(0.02, 0.5, 25), np.linspace(0.5, 2.0, 31)])
    param_families.append(('power', p_vals))
    # exponential decay
    r_vals = np.linspace(0.90, 0.9999, 25)
    param_families.append(('exp', r_vals))
    # linear decreasing
    # not param

    for name, vals in param_families:
        for val in vals:
            if time.time() > deadline:
                break
            if name == 'power':
                i = np.arange(1, n+1, dtype=float)
                a = i ** (-val)
            elif name == 'exp':
                i = np.arange(n, dtype=float)
                a = (val) ** i
            a = a / a.sum()
            try_seq(a)

    # random permutations of current best
    for _ in range(2000):
        if time.time() > deadline:
            break
        perm = np.random.permutation(n)
        a = best_seq[perm]
        try_seq(a)

    # Simulated annealing
    # start from best_seq
    current = best_seq.copy()
    cur_score = best_score
    # schedule parameters
    total_iters_est = 200000  # approximate until deadline
    # we will compute remaining time and adjust loops
    it = 0
    while time.time() < deadline:
        it += 1
        # temperature linearly decreasing
        elapsed = time.time() - start
        remaining_frac = max(0.0, (deadline - time.time()) / (deadline - start))
        # T scaling: start high ~ 0.01 * cur_score? Use 0.1 * best_score maybe
        T0 = best_score * 0.1  # adjust
        T = T0 * remaining_frac

        # propose move
        move_type = np.random.choice([0,1,2], p=[0.5,0.3,0.2])
        cand = current.copy()
        if move_type == 0:
            # mass transfer
            # pick source index with prob proportional to value (more mass)
            nonzero = np.where(cand > 1e-12)[0]
            if nonzero.size == 0:
                continue
            i = np.random.choice(nonzero)

            j = np.random.randint(0, n)

            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta
        elif move_type == 1:

            noise = np.random.normal(scale=0.01, size=n)
            cand = cand + noise
            cand = np.maximum(cand, 0.0)
            total = cand.sum()
            if total <= 0:

            cand = cand / total

            # subgradient
            K = _active_indices(conv, max_val, rel_eps=0.001, max_add=30)
            g = _gradient_active(best_seq, K)
            i_max = int(np.argmax(g))
            i_min = int(np.argmin(g))
            if i_max != i_min and best_seq[i_max] > 0:
                delta = min(0.1 * best_seq[i_max], best_seq[i_max])
                cand = best_seq.copy()

                cand[i_min] += delta
                new_sc = _score_of_seq(cand)
                if new_sc < best_score:
                    best_score = new_sc
                    best_seq = cand
                    print(f"[{time.time() - start:6.2f}s] Greedy improve {new_sc:.6f}")

    # return best_seq as list of floats
    return [float(x) for x in best_seq]

```python
"""
Search for a low‑value step function for `evaluate_sequence`.

The algorithm combines:
  1. A broad parametric sweep (linear, power‐law, exponential shapes) and
     random Dirichlet draws to obtain a strong initial candidate.
  2. Random permutations of the multiset to exploit the permutation effect.
  3. Simulated‑annealing with three simple moves (mass transfer, swap,
     Gaussian perturbation) that accept uphill moves with a linearly
     decreasing temperature.
  4. A final gradient‑directed local refinement that repeatedly moves a
     small amount of mass from the index with the largest sub‑gradient to
     the one with the smallest, falling back to an exponentiated‑gradient
     step when the greedy move does not improve.

All vectors are kept on the probability simplex (sum = 1, non‑negative).
The objective that is minimised is `2·n·max(conv(a,a))`, which equals the
value returned by `evaluate_sequence` for a sequence with unit sum.

The routine respects a ~1000 s wall‑clock budget, prints progress and
returns the best coefficient list found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Return full auto‑convolution of `a` (linear convolution) and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())

def _score_of_seq(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a vector that already sums to 1."""
    n = a.shape[0]
    _, max_val = _conv_and_max(a)
    return float(2.0 * n * max_val)

def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Modest set of (almost) maximal convolution indices.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K

def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` w.r.t. `a` for the active set `K`.
    For each active convolution index `k` the contribution to component `i`
    is `2·a_{k-i}` whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]

Search for a low‑value step function for `evaluate_sequence`.

The algorithm blends several ideas:

1. **Broad parametric sweep** – a collection of simple families
   (linear, power‑law, exponential, Dirichlet draws) is explored
   and the best candidate is kept.

2. **Permutation scan** – many random reorderings of the multiset of
   the current best vector are tested; the convolution depends heavily
   on the ordering.

3. **Simulated‑annealing (SA)** – three inexpensive moves are used
   *mass transfer* (from the largest entry to the smallest),
   *swap* of two positions and *Gaussian perturbation*.
   An SA schedule with a linearly decreasing temperature accepts
   uphill moves early on, allowing the search to escape local minima.

4. **Final gradient‑directed refinement** – the active‑set sub‑gradient
   of the `max(conv)` functional is computed.  A greedy mass‑transfer
   step is attempted; if it does not improve the objective an
   exponentiated‑gradient (mirror‑descent) step on the simplex is taken.
   This stage tightens the solution after SA.

All vectors are kept on the probability simplex (non‑negative,
sum = 1), so the objective `2·n·max(conv)` equals the value that
`evaluate_sequence` would return for a unit‑sum sequence.
The routine respects a ~1000 s wall‑clock budget, prints progress

def _conv_and_max(a: np.ndarray):
    """Full linear convolution of `a` with itself and its maximum."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())

def _score_of_seq(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a vector that already sums to 1."""
    n = a.shape[0]
    _, max_val = _conv_and_max(a)
    return float(2.0 * n * max_val)

def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 30):
    """
    Return a modest set of indices whose convolution values are
    (almost) maximal.  All entries with value >= max_val − rel_eps·max_val
    are kept; if fewer than `max_add` are found the globally largest
    entries are added to reach `max_add`.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K

def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv)` for the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k‑i} whenever the index lies in the valid range.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]

    ind = np.arange(len(v)) + 1

    theta = cssv[cond][-1] / rho

    return w
