    cur_a = a.copy()

    cur_score = 2 * n * max_val

    best_a = cur_a.copy()

    # Starting temperature maybe 0.02 (like fraction of score)

                best_a = new_a.copy()

"""
Hybrid multi‑restart Adam + simulated‑annealing optimizer.

The objective to minimise is
    score = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the score is invariant under uniform scaling we keep the vector on the
probability simplex (sum = 1, a_i ≥ 0).  The optimiser works in three phases:

1.  Several diverse initial candidates (previous best, uniform, triangular,
    a few random Dirichlet draws) are each refined with a fairly long Adam
    optimisation that uses a soft‑max approximation of the max‑conv term.
2.  The best candidate among the above is refined a second time with a tighter
    schedule and many more Adam iterations.
3.  A cheap simulated‑annealing / random‑move local search is run for the
    remaining time budget to squeeze out the last bits of improvement.

All operations are performed with FFTs for fast convolution and the
Euclidean projection onto the simplex (Duchi et al.).  The code respects a
global time budget (default 1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv(a,a)), full convolution). ``a`` is assumed to sum to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. ``a``.
    Implements   g_i = 2 * Σ_k w_k * a_{k-i}
    using FFTs.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # keep only the central n entries
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser for 2·n·max(conv) on the simplex.

    Returns (final_a, best_seen_a, best_score) where best_score = 2·n·max(conv).
    """
    n = a.shape[0]

    # FFT lengths – cached for the whole run
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights over convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)               # avoid division by zero
        w = np.exp((conv - max_val) / tau)  # w_i ∝ exp((c_i-M)/τ)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

    return a, best_a, best_score


def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple random‑move annealing on the simplex.
    Moves a small amount of mass from one coordinate to another.
    Returns the best vector encountered and its score (2·n·max(conv)).
    """
    n = a.shape[0]
    cur = a.copy()
    max_val, _ = _conv_max(cur)
    cur_score = 2.0 * n * max_val
    best_seq = cur.copy()
    best_score = cur_score

    # temperature starts as a small fraction of the current score
    T = cur_score * 0.02
    it = 0

    while time.time() < deadline:
        it += 1
        # occasional early exit if a max iteration count was supplied
        if max_iters is not None and it > max_iters:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # propose moving a random fraction of the mass at i to j
        delta = np.random.rand() * cur[i] * 0.05   # ≤5 % of the source value
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        max_val_new, _ = _conv_max(new)
        new_score = 2.0 * n * max_val_new

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # exponential cooling
        T *= 0.99997
        if T < 1e-12:
            T = 1e-12

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:

        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, seq)
        up = np.maximum(up, 0.0)

        if total > 0:

        else:
            up = np.full(n, 1.0 / n)
