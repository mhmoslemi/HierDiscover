Implementation details: The simplex projection function:

def simplex_projection(v):
    """Returns vector projected onto probability simplex (sum=1, nonnegative)."""
    # Implementation: sort descending, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0
    w = np.maximum(v - theta, 0)
    return w

"""
Improved search for a low‑autocorrelation step function.

Approach
--------
1. Initialise several candidate vectors on the probability simplex
   (uniform, random Dirichlet, a sine‑shape, and an optional
   previously stored construction).
2. For each candidate run a *smooth* projected gradient descent.
   The objective is a smooth approximation of the max‑convolution:
       f_t(a) = (1/t)·log Σ_k exp(t·(a*a)_k) .
   Its gradient can be expressed with a single FFT‑based correlation,
   allowing fast updates even for n≈1500.
   The schedule slowly raises *t* (starting from 5 up to ≈200) so the
   optimisation first reduces the average autocorrelation and then
   focuses on the peak.
3. After the continuous phase a short random‑mass‑transfer local
   search is performed to escape possible plateaus.
4. The best sequence seen (lowest value of the original
   `evaluate_sequence`) is returned.  All vectors are projected onto
   the non‑negative simplex after every update, guaranteeing a valid
   input for the evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑negative)
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation used during the search (identical to the provided one
# but without the safety checks).
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum(a)²) for a non‑negative array a."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth projected gradient descent (continuation on the temperature t)
def smooth_projected_gradient_descent(
    a_init: np.ndarray, time_budget: float, target: float = 1.503
) -> np.ndarray:
    """
    Minimise a smooth surrogate of the max‑convolution on the simplex.
    Returns the best vector found before the time budget expires.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)

    n = a.size
    # length large enough for linear correlation of lengths n and 2n‑1
    L = 1 << ((3 * n - 2).bit_length())

    # step‑size schedule
    eta0 = 0.5
    decay = 1e-4

    # temperature schedule (smooth → sharp)
    t = 5.0
    t_max = 200.0
    t_inc_every = 5000  # increase after this many iterations

    start = time.time()
    it = 0

    while time.time() - start < time_budget:
        # ----- convolution a*a (full length 2n‑1) -----
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ----- soft‑max weights over the convolution entries -----
        scaled = t * conv_full
        mx = scaled.max()
        exp_scaled = np.exp(scaled - mx)          # stabilise exponentials
        w = exp_scaled / exp_scaled.sum()          # w_i ≥ 0, Σw = 1

        # ----- gradient of the smooth surrogate -----
        W = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(A) * W, n=L)   # linear correlation a ⋆ w
        grad = 2.0 * t * corr_full[:n]                 # ∇f = 2·t·(a ⋆ w)

        # ----- projected gradient step -----
        eta = eta0 / (1.0 + it * decay)
        a_next = a - eta * grad
        a = simplex_projection(a_next)

        # ----- occasional evaluation -----
        if it % 400 == 0:
            val = fast_score(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[pgd iter {it}] new best {best_val:.6f}")
            if best_val <= target:
                print("[pgd] target reached – stopping early")
                break

        # ----- temperature increase (continuation) -----
        if it % t_inc_every == 0 and t < t_max:
            t = min(t * 1.05, t_max)

        it += 1

    print(f"[pgd final] best value {best_val:.6f} after {it} iters, t={t:.2f}")
    return best_a


# ----------------------------------------------------------------------
# Small randomised local search (mass transfer between two coordinates)
def local_search(a: np.ndarray, time_budget: float) -> np.ndarray:
    """Try random pairwise mass transfers for a short time."""
    best = a.copy()
    best_val = fast_score(best)
    n = best.size
    start = time.time()
    while time.time() - start < time_budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] < 1e-12:
            continue
        delta = 0.05 * best[i] * np.random.rand()   # up to 5 % of a_i
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_projection(cand)
        val = fast_score(cand)
        if val < best_val:
            best_val, best = val, cand.copy()
    print(f"[local search] improved to {best_val:.6f}")
    return best

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) whose evaluation score
    is as small as possible.  The routine respects the 1000 s wall‑clock
    limit, explores several initialisations and runs a smooth projected
    gradient descent with a continuation on the temperature parameter.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2            # safety margin

    n = 1500                     # length of the sequence (as used previously)
    target_score = 1.503

    best_seq = None
    best_val = math.inf

    # ------------------------------------------------------------------
    # 0. try to reuse a previously stored construction if present
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.size == n:
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. deterministic sinusoidal start (often gives a fairly flat spectrum)
    x = np.arange(n, dtype=np.float64)
    sine_vec = np.sin(np.pi * (x + 0.5) / n)          # positive on (0, π)
    sine_vec = simplex_projection(sine_vec)         # put on simplex
    val = fast_score(sine_vec)
    if val < best_val:
        best_seq, best_val = sine_vec.copy(), val
        print(f"[init] sine start score {val:.6f}")

    # ------------------------------------------------------------------
    # 2. multiple random restarts with smooth PGD
    # reserve some time for the final local search
    remaining_budget = deadline - time.time()
    if remaining_budget <= 0:
        remaining_budget = 0.0
    # we will use up to 4 restarts, each getting a share of the remaining time
    n_restarts = 4
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # allocate time (a bit less than equal share to keep a margin)
        remaining = deadline - time.time()
        per_restart = max(0.0, remaining / (n_restarts - r) * 0.85)
        if per_restart < 5.0:      # not worth a full run – break early
            break

        # random Dirichlet start (uniform exponential)
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()

        # run the smooth projected gradient descent
        a_opt = smooth_projected_gradient_descent(a0, per_restart, target_score)

        val = fast_score(a_opt)
        print(f"[restart {r}] best value after PGD {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_opt.copy()
            print(f"[global] new best {best_val:.6f}")

        if best_val <= target_score:
            print("[success] target reached – terminating search")
            break

    # ------------------------------------------------------------------
    # 3. final short local search using any leftover time
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 1.0:
        best_seq = local_search(best_seq, time_left * 0.9)
        best_val = fast_score(best_seq)

    # ------------------------------------------------------------------
    # 4. safety net – if nothing was found, fall back to uniform vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] returning sequence (length {n}) with score {best_val:.6f}")
    return best_seq.tolist()
```
