def simplex_proj(v):
    # project v onto probability simplex using sort
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimiser for the step‑function evaluation problem.

The algorithm consists of three stages:

1. **Initialisation** – start from the previously discovered best construction
   (global variable ``height_sequence_1``) if it exists, otherwise from a
   random Dirichlet point; always normalise to a probability vector.

2. **Large‑active‑set exponentiated‑gradient descent** – minimise the quantity
   ``max(conv(a,a))`` while keeping the vector on the simplex.
   An active set of the largest convolution entries (up to ``max_add``) is
   used to compute a sub‑gradient.  The update is performed in log‑space
   (mirror descent with an adaptive step size).  The active set is larger
   than in the reference implementation (``max_add = 120``) which provides a
   more accurate descent direction.

3. **Simulated‑annealing local search** – after the gradient phase the
   remaining time is spent on a temperature‑controlled hill‑climbing that
   moves a small fraction of mass between two randomly chosen positions.
   Moves that increase the score are occasionally accepted, allowing the
   search to escape shallow plateaus.

Both phases keep the vector non‑negative and normalised, so the final
candidate can be returned directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Full auto‑convolution of ``a`` (via FFT) and its maximum entry."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 0.001, max_add: int = 120):
    """
    Return a set of indices that are “active’’ for the sub‑gradient.
    All indices within ``rel_eps`` relative distance of ``max_val`` are kept.
    If fewer than ``max_add`` such indices exist, the globally largest ones
    are added so that the set contains at most ``max_add`` entries.
    """
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    if K.size < max_add:
        # add the globally largest entries
        topk = min(max_add, conv.size)
        K = np.unique(np.argpartition(conv, -topk)[-topk:])
    return K


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of ``max(conv(a,a))`` using the active index set ``K``.
    For each active index k the contribution to a_i is 2·a_{k‑i} (if the index
    is in range).  The result is summed over all active indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a step‑function) that aims to
    minimise the ``evaluate_sequence`` score.  The routine runs for at most
    ``budget_s`` seconds and prints occasional progress messages.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Initialisation
    # --------------------------------------------------------------
    target_n = 5000
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.size != target_n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, init.size)
            x_new = np.linspace(0.0, 1.0, target_n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        # fall back to a random Dirichlet point
        init = np.random.rand(target_n)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        a = np.full(target_n, 1.0 / target_n)
    else:
        a = init / s                       # probability vector on the simplex

    conv, max_val = _conv_and_max(a)
    cur_score = 2.0 * target_n * max_val
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:6.2f}s] start: n={target_n}, score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 2. Exponentiated‑gradient descent with a large active set
    # --------------------------------------------------------------
    max_iters = 400_000                     # generous upper bound
    eta0 = 0.9                               # initial step size
    rel_eps = 0.001
    max_add = 120                            # size of active set
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Full convolution (required for the active set and score)
        conv, max_val = _conv_and_max(a)
        cur_score = 2.0 * target_n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d} improved score={cur_score:.6f}")

        # Active set and sub‑gradient
        K = _active_indices(conv, max_val, rel_eps=rel_eps, max_add=max_add)
        g = _gradient_active(a, K)

        # Adaptive exponentiated update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)               # classic diminishing step
        a = np.maximum(a, 1e-15)                 # avoid log(0)
        log_a = np.log(a) - eta * g
        log_a -= np.max(log_a)                    # stabilise exponentials

        a = np.maximum(a, 1e-15)

    # --------------------------------------------------------------
    if time.time() < deadline:

        cur_score = 2.0 * target_n * max_val

        # simple exponential cooling schedule

        cooling_rate = 0.99996

        while time.time() < deadline:
            i, j = np.random.randint(0, target_n, size=2)
            if i == j:
                continue
            # propose moving a small fraction of mass from i to j
            eps = np.random.rand() * 0.05      # up to 5 % of source mass
            delta = eps * a[i]
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # guard against numerical negativity
            if a_new[i] < 0.0:

            new_score = 2.0 * target_n * max_val_new

            # Metropolis acceptance
            if diff < 0 or np.random.rand() < math.exp(-diff / max(temperature, 1e-12)):

                cur_score = new_score

                    best_seq = a_new.copy()
                    print(f"[{time.time() - start_time:6.2f}s] SA improve score={new_score:.6f}")

            # cool down
            temperature *= cooling_rate
            if temperature < 1e-12:
                temperature = 1e-12

    # --------------------------------------------------------------
    # 4. Return best sequence found
    # --------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    # Convert to plain Python floats as required by the evaluation harness
    return [float(x) for x in best_seq]
