```
def gradient_smooth(a, beta):
    n = len(a)
    # compute conv = a*a via FFT (full)
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    # compute softmax weights
    max_conv = conv.max()
    exp_weights = np.exp(beta * (conv - max_conv))
    p = exp_weights / exp_weights.sum()
    # compute a_rev
    a_rev = a[::-1]
    # compute convolution p * a_rev
    size2 = 1 << (( (2*n-1) + n - 1).bit_length())
    # Actually conv length needed is (2n-1) + n -1 = 3n-2
    size2 = 1 << ((3*n - 2).bit_length())
    fft_p = np.fft.rfft(p, size2)
    fft_a_rev = np.fft.rfft(a_rev, size2)
    conv_pa = np.fft.irfft(fft_p * fft_a_rev, size2)
    # Now gradient = 2 * conv_pa[ n-1 : n-1 + n ]
    grad = 2 * conv_pa[n-1 : n-1 + n]
    return grad, max_conv
```

```python
search for a low‑autocorrelation step function.

The method combines:
  * several random restarts,
  * a smoothed‑max objective (log‑sum‑exp) with an Adam‑style
    projected gradient descent on the probability simplex,
  * a fallback projected‑subgradient optimiser (the previous
    implementation) for the remaining time.

The optimiser works on vectors of length 1500, keeps the  ℓ₁‑norm
to 1 (scaling does not affect the score) and respects the
non‑negativity constraints by Euclidean projection onto the simplex.

# 1. Fast evaluation (identical to evaluate_sequence but without checks)
def fast_score(a: np.ndarray) -> float:
    """Calculate the evaluation score for a vector a (∑a = 1)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 2. Euclidean projection onto the probability simplex
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i >= 0, sum x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

# 3. Gradient of the smoothed‑max objective
def smoothed_max_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is the gradient of the
    smoothed max (log‑sum‑exp) of the convolution a*a
    and max_conv is the true maximum of a*a.
    """
    n = a.size
    # 1) convolution a*a (full length 2n‑1)
    size1 = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fft_a * fft_a, size1)[: 2 * n - 1]

    max_val = conv.max()

    # 2) soft‑max weights (log‑sum‑exp trick)
    exp_w = np.exp(beta * (conv - max_val))
    w = exp_w / exp_w.sum()               # shape (2n‑1,)

    # 3) gradient = 2 * (w * a_rev)  (convolution)
    a_rev = a[::-1]
    size2 = 1 << (( (2 * n - 1) + n - 1).bit_length())   # >= 3n‑2
    fft_w = np.fft.rfft(w, size2)
    fft_rev = np.fft.rfft(a_rev, size2)
    conv_wr = np.fft.irfft(fft_w * fft_rev, size2)       # length 3n‑2

    # slice the needed part: indices n‑1 .. n‑1+n‑1
    grad = 2.0 * conv_wr[n - 1: n - 1 + n]
    return grad, max_val
```
