def _run_adam(a0, n, deadline, start_time, step0=0.01, beta1=0.9, beta2=0.999, eps=1e-8):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[...]: adam improve: n={n}, iter={t}, score={cur_score:.6f}")
        grad = _subgradient_full(a, conv)  # shape (n,)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection
        a = _project_simplex(a_new)
    return best_a, best_score

The program searches for a non‑negative step function `a = (a₀,…,aₙ₋₁)` (a probability vector) that minimises

```
F(a) = 2·n·max(conv(a)) / ( Σ a )²
```

Because `Σ a` is kept equal to 1 the objective reduces to `2·n·max(conv(a))`.  
The algorithm consists of four stages, each respecting the global
time‑budget (`budget_s` seconds).

1. **Exploration (small size)** – a few random starts on `n≈1200`,
   each optimised with an *Adam* style projected gradient method.
   The best vector is kept.

2. **Scale‑up & intensive gradient descent** – the current best vector is
   linearly interpolated to `n≈2400` and then to `n≈4800`.  For each length a
   long Adam optimisation is performed.  Adam gives an adaptive per‑coordinate
   step‑size and converges faster than the plain Nesterov scheme used
   previously.

3. **Final refinement (very large size)** – the best vector is up‑sampled
   to `n≈9600`.  A last round of Adam optimisation is executed using a
   more aggressive learning‑rate schedule.

4. **Directed simulated annealing** – a mass‑transfer metropolis walk that
   exploits the *sub‑gradient* of `max(conv(a))`.  The source of a transfer

"""
Enhanced search for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1.  Simplex projection – O(n) Euclidean projection onto {x ≥ 0, Σx = 1}.
2.  Fast linear convolution – FFT based, O(n log n).
3.  Sub‑gradient of max(conv) – sum of contributions of all active indices.
4.  Adam‑style projected gradient descent – adaptive per‑coordinate steps,
   momentum and a simple learning‑rate schedule.
5.  Multi‑scale optimisation – start at n≈1200, up‑sample to 2400,
   4800 and finally 9600, performing long gradient runs at each scale.
6.  Directed simulated‑annealing – mass transfers from indices with
   large positive gradient to those with large negative gradient.
   Acceptance follows a geometric temperature schedule.
7.  The best feasible vector (normalised to sum = 1) is returned.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# 2.  Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())      # next power‑of‑two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# 3.  Sub‑gradient of max(conv) (all active entries)
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (sum over all active indices)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    act = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # contributions a_j → ∂/∂a_i and a_i → ∂/∂a_j
        g[i] += a[j]
        g[j] += a[i]
    return g


# ------------------------------------------------------------
# 4.  Adam‑style projected gradient descent
# ------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int,
              deadline: float, start_time: float,
              step0: float = 0.02,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_full(a, conv)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step
        step = step0 / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a = _project_simplex(a_new)

    return best_a, best_score


# ------------------------------------------------------------
# 5.  Directed simulated annealing (mass transfers)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   max_steps: int = 300_000) -> Tuple[np.ndarray, float]:
    """
    Mass‑transfer Metropolis walk that biases moves according to the sub‑gradient.
    Returns the best sequence encountered during the walk.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_steps:
        it += 1

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # compute sub‑gradient once every few steps (every 10 steps is enough)
        if it % 10 == 0:
            grad = _subgradient_full(a, conv)

            # split gradient into positive / negative parts
            pos = np.maximum(grad, 0.0)
            neg = np.maximum(-grad, 0.0)
            pos_sum = pos.sum()
            neg_sum = neg.sum()
            if pos_sum == 0 or neg_sum == 0:
                # fall back to uniform probabilities
                prob_src = a / a.sum()
                prob_dst = prob_src
            else:
                prob_src = pos / pos_sum
                prob_dst = neg / neg_sum

        # choose source (mass to take) and destination (mass to give) indices
        src = rng.choice(n, p=prob_src)
        dst = rng.choice(n, p=prob_dst)

        if src == dst or a[src] < 1e-14:
            continue

        # propose moving a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# 6.  Upsampling (linear interpolation) and normalisation
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
