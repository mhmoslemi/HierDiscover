"""
Propose a step‑function (non‑negative list of heights) that minimises
`evaluate_sequence`.

The search combines three ideas that were not used in the previous
implementation:

1. **Bias‑towards‑the‑left initialisation** – samples are drawn from a
   Dirichlet distribution whose concentration parameters decay
   exponentially with the index.  This creates a non‑symmetric shape
   that is known to reduce the peak of the self‑convolution.

2. **Sub‑gradient descent on the true ℓ∞‑objective** – instead of the
   smooth log‑sum‑exp surrogate, we minimise the exact objective
   `max(conv)` using a simple sub‑gradient step.  The gradient for a
   given maximal index `k*` is simply `2*a_{k*-i}` on the feasible
   indices.  After each step we project onto the probability simplex
   (non‑negative, sum = 1).

3. **Targeted simulated‑annealing polish** – after the deterministic
   phase a cheap mass‑transfer annealing sweep (moving a small ε from one
   coordinate to another) is applied to escape possible local minima.

A multi‑scale schedule (800 → 1600 → 3200 → 6400 → 9600) is used; the
best sequence at each scale is up‑sampled (linear interpolation) to
initialise the next scale.  All stages respect the 1000 s time budget,
run on a single CPU and rely only on NumPy (no external solvers).

The returned list is ready for the outer harness to feed to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _grad_max_convolution(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the objective max_k (a*a)_k at the current maximal
    index `k`.  Gradient_i = 2 * a_{k-i} for all i where the partner index
    lies inside the array, otherwise 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the simplex
# ----------------------------------------------------------------------


def _run_subgrad_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true ℓ∞ objective.
    Returns the best vector seen and its (internal) score 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    step0 = 0.8  # initial step size (tuned experimentally)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"it={it}, score={score:.6f}")

        # index of the current maximal convolution entry
        k_star = int(conv.argmax())
        grad = _grad_max_convolution(a, k_star)

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step & simplex projection
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing: move a small ε from i to j.
    Returns possibly improved sequence and its score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm proceeds scale‑by‑scale, uses biased Dirichlet starts,
    deterministic sub‑gradient descent on the exact ℓ∞ objective,
    and finishes with a quick annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Length schedule (few geometric steps, shuffled)
    # ------------------------------------------------------------------
    lengths = [800, 1600, 3200, 6400, 9600]
    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    # Coarse multi‑scale optimisation
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a time slice for this resolution
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(8.0, 0.13 * remaining)

        # --------------------------------------------------------------
        # Several random restarts with left‑biased Dirichlet initialisation
        # --------------------------------------------------------------
        restarts = 3
        for r in range(restarts):
            # exponential bias parameter (larger → stronger front weight)
            bias = np.random.uniform(0.0, 5.0)
            i = np.arange(n, dtype=float)
            # concentration parameters for Dirichlet (>0)
            alpha = np.exp(-bias * i / n) + 0.5
            a0 = np.random.gamma(alpha, 1.0)
            a0 /= a0.sum()                     # project onto simplex

            a_opt, score_opt = _run_subgrad_descent(
                a0, n, slice_deadline, start_time, max_iter=120_000)

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve (subgrad): "
                      f"n={n}, restart={r}, score={score_opt:.6f}")

        # --------------------------------------------------------------
        # Short annealing polish (if time permits)
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, 0.04 * (deadline - time.time()))
            a_ref, score_ref = _run_annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=30_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after annealing: "
                      f"n={n}, score={score_ref:.6f}")

        # use the current best as warm‑start for the next larger scale
        if n != max(lengths):
            next_n = min([m for m in lengths if m > n])
            best_seq = _upsample(best_seq, next_n)

    # ------------------------------------------------------------------
    # Fine‑tuning phase (additional sub‑gradient passes on the final size)
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 5.0:
        n_cur = best_seq.shape[0]

        # allocate a short slice (up to 30 s) for another sub‑gradient pass
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(30.0, 0.5 * remaining)

        a_opt, score_opt = _run_subgrad_descent(
            best_seq, n_cur, slice_deadline, start_time, max_iter=80_000)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] fine‑tune improve: n={n_cur}, score={score_opt:.6f}")

        # one final cheap annealing sweep
        if time.time() < deadline:
            anneal_deadline = time.time() + min(10.0, deadline - time.time())
            a_ref, score_ref = _run_annealing(
                best_seq,
                n_cur,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=1e-2,
                end_temp=1e-6,
                steps=10_000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] final anneal improve: n={n_cur}, score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Return plain Python list – the outer harness will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
