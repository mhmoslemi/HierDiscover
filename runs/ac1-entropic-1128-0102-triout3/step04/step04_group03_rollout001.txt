def _gradient_softmax(a, conv, beta):
    # conv is convolution a*a
    max_val = conv.max()
    # compute stable softmax weights
    # exponent scaling: we may want to avoid overflow; subtract max_val.
    exp_terms = np.exp(beta * (conv - max_val))
    w = exp_terms / exp_terms.sum()  # length 2n-1
    # compute gradient
    a_rev = a[::-1]
    # perform convolution using FFT
    # pad length to next power of 2 > len(a_rev)+len(w)-1.
    conv_len = len(a_rev) + len(w) - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Af = np.fft.rfft(a_rev, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    grad_full = np.fft.irfft(Af * Bf, n=fft_len)[:conv_len]
    # gradient portion corresponds to indices n-1 .. 2n-2 inclusive?
    # slice:
    start = len(w)//2  # (2n-1)//2 = n-1
    grad = 2.0 * grad_full[start:start+len(a)]
    return grad

"""
Propose a step‑function sequence with a low self‑convolution peak.

The algorithm works in three phases:

1.  **Progressive Adam optimisation** – start from a modest size (n≈800) and
    repeatedly run a projected Adam optimiser on the probability simplex.
    The objective is a smooth approximation of the max‑convolution
    (log‑sum‑exp with a gradually increasing sharpness β).  The gradient of
    the smooth max is obtained analytically as a correlation, computed with
    FFTs for speed.

2.  **Upsampling** – after each stage the current best sequence is linearly
    interpolated to a larger length (800 → 2400 → 4800 → optionally 9600)
    while keeping the total mass equal to 1.

3.  **Local stochastic refinement** – a cheap simulated‑annealing walk
    that transfers a small fraction of mass between two random positions.
    The walk accepts downhill moves and, with a geometrically decaying
    temperature, occasional uphill moves.

All steps respect the given time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Compute soft‑max weights w_k ∝ exp(β·conv_k). Numerically stable."""
    max_val = conv.max()
    # conv - max_val ≤ 0, so exp does not overflow
    exp_term = np.exp(beta * (conv - max_val))
    w = exp_term / exp_term.sum()
    return w


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max  f(a)= (1/β)·log Σ exp(β·conv_k)
    w.r.t. a, i.e.  ∂f/∂a_i = 2·Σ_k w_k·a_{k‑i}
    (correlation of a with the soft‑max weights w).
    Computed via FFT in O(n log n).
    """
    n = a.shape[0]
    w = _softmax_weights(conv, beta)                # shape (2n-1,)
    a_rev = a[::-1]                                 # reversed a

    # convolution a_rev * w (full) → length 3n‑2
    conv_len = len(a_rev) + len(w) - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Af = np.fft.rfft(a_rev, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(Af * Bf, n=fft_len)[:conv_len]

    # correlation part we need: indices n‑1 … 2n‑2 (inclusive)
    start = len(w) // 2          # (2n‑1)//2 = n‑1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
#  Adam optimiser on the simplex (soft‑max surrogate of max‑conv)
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray, n: int,
                  deadline: float, start_time: float,
                  beta0: float = 5.0,
                  beta_max: float = 300.0,
                  beta_inc: float = 1.0,
                  lr: float = 0.15,
                  max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser.
    Returns the best sequence discovered and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    beta = beta0

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and (scaled) score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, β={beta:.1f}, score={cur_score:.6f}")

        # gradient of smooth max
        grad = _gradient_softmax(a, conv, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # update and project back onto the simplex
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

        # increase β (sharpen the approximation) – but cap it
        beta = min(beta_max, beta + beta_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Simple linear up‑sampling that preserves total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
#  Simulated‑annealing local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 400000) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score
