def _adam_optimize(a0, n, deadline, start_time, max_iter=10**7, tau_init=0.5, tau_final=20.0):
    # a0 is initial probability vector summing to 1.
    # Convert to z
    eps = 1e-12
    z = np.log(a0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    best_a = a0.copy()
    best_obj = 2.0 * n * np.max(_conv_self(a0))
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = _softmax(z)
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, obj={obj:.6f}")
        # schedule tau
        # linear interpolation based on iteration fraction
        frac = min(1.0, it / 20000.0)  # after 20000 iterations reach tau_final
        tau = tau_init + frac * (tau_final - tau_init)
        # compute softmax weights
        # use numerical stability: subtract max(conv) to avoid overflow
        # conv already has max; we can use conv - max_val
        exp_term = np.exp(tau * (conv - max_val))
        w = exp_term / exp_term.sum()

        # gradient w.r.t a
        grad_a = 2.0 * np.convolve(a[::-1], w)[n-1 : n-1 + n]
        # chain rule to z
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.1 / (1 + 0.00005 * it)  # decaying learning rate
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return best_a, best_obj
```

"""
Algorithm
---------
A smoothed approximation of the objective `max(conv(a))` is minimised.
The max is replaced by a temperature‑controlled soft‑max:
    φ_τ(a) = (1/τ)·log(∑_k exp(τ·(a*a)_k))

With τ→∞ the approximation becomes the true max.  The optimisation is
performed on the *softmax* parametrisation `a = softmax(z)`,
which automatically satisfies the simplex constraints
`a_i≥0, Σa_i=1`.  An Adam optimiser drives `z` while τ is increased
linearly (a homotopy from a smooth to a sharp objective).  After the
gradient phase a short simulated‑annealing refinement moves mass
between coordinates to escape any remaining shallow local minima.

The routine respects the supplied time budget, runs a few restarts,
upsamples progressively (1200 → 3000 → 5600 points) and finally returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using an FFT of length a power of two."""
    n = a.shape[0]
    m = b.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)[: n + m - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·len(a)-1)."""
    return _conv_linear(a, a)


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   tau_init: float = 0.5,
                   tau_final: float = 20.0,
                   max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam on the soft‑max approximation.
    Returns the best vector seen and its (unsmoothed) objective 2·n·max(conv).
    """
    eps = 1e-12
    z = np.log(a0 + eps)                     # initialise softmax parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = a0.copy()
    best_obj = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        a = _softmax(z)

        # ----- full self‑convolution -----
        conv = _conv_self(a)
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, obj={cur_obj:.6f}")

        # ----- temperature schedule -----
        # increase τ linearly over the first 20000 iterations, then keep τ_final
        frac = min(1.0, it / 20000.0)
        tau = tau_init + frac * (tau_final - tau_init)

        # ----- soft‑max weights over convolution indices -----
        # stable softmax: subtract max(conv) before exponentiation
        exp_term = np.exp(tau * (conv - max_val))
        w = exp_term / exp_term.sum()

        # ----- gradient w.r.t. a (2·(a_rev * w))[segment] -----
        grad_a_full = _conv_linear(a[::-1], w)        # length 3n‑2
        grad_a = 2.0 * grad_a_full[n - 1: n - 1 + n]   # slice to length n

        # ----- chain rule to z (softmax Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decays gently
        lr = 0.08 / (1.0 + 5e-5 * it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (same as in the previous solution)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing: move ε from a source index (chosen proportionally
    to its mass) to a random destination.  Accept moves with Metropolis rule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # biased source selection (more mass → higher chance)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Uses a smoothed‑max Adam optimiser followed by a short annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.8

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Multi‑stage optimisation: increasing resolution
    # ------------------------------------------------------------------
    stages = [1200, 3000, 5600]          # progressive sizes
    cur_seq = None

    for stage_idx, n in enumerate(stages):
        if time.time() >= deadline:
            break

        # initialise the vector for this stage
        if cur_seq is None:
            if starter is not None:
                a0 = _upsample(starter, n)
            else:
                a0 = np.random.rand(n)
                a0 /= a0.sum()
        else:
            a0 = _upsample(cur_seq, n)

        # allocate a fraction of the remaining budget for this stage
        remaining = deadline - time.time()
        # use roughly 30 % for the final stage, split the rest equally
        if stage_idx < len(stages) - 1:
            stage_deadline = time.time() + 0.3 * remaining
        else:
            stage_deadline = deadline

        # run the Adam optimiser on the smoothed objective
        a_opt, obj_opt = _adam_optimize(a0, n, stage_deadline, start_time)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage {stage_idx + 1} improve: n={n}, obj={obj_opt:.6f}")

        # keep the result for the next up‑sampling step
        cur_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Final annealing refinement (uses whatever time is left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, obj_ref = _run_annealing(
            best_seq, best_seq.shape[0],
            deadline, start_time,
            best_obj, best_seq,
            start_temp=5e-2, end_temp=1e-6,
            steps=200_000
        )
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: obj={obj_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list of floats
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
