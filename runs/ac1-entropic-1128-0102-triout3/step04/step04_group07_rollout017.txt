def evaluate_swap(counts, S, i, j):
    # counts: np.int64 array
    # S: list of ints (size m)
    # i in S, j not in S.
    # returns new_max, new_counts (maybe just new_max and diff)
    m = len(S)
    # Convert to numpy array:
    S_arr = np.array(S, dtype=np.int32)
    # Removal contributions:
    # compute idx_remove = i + S_arr
    idx_remove = i + S_arr
    # Determine dec vector: dec = np.full(m, 2, dtype=int); dec[np.where(S_arr == i)] = 1
    dec = np.full(m, 2, dtype=np.int32)
    dec[S_arr == i] = 1
    # new counts copy:
    new_counts = counts.copy()
    np.subtract.at(new_counts, idx_remove, dec)
    # After removal we need S_minus = S_arr[S_arr != i]
    mask = S_arr != i
    S_minus = S_arr[mask]   # length m-1
    # addition contributions:
    idx_add = j + S_minus
    np.add.at(new_counts, idx_add, 2)
    # self pair:
    new_counts[2*j] += 1

    new_max = new_counts.max()
    return new_max, new_counts

"""
Proposes a step‑function sequence with a low autocorrelation score.

The algorithm proceeds in three stages:
1. Randomly generate many support sets of a given size m (uniform weights 1/m)
   and keep the best (lowest max‑convolution count).
2. Starting from the best support, perform a simulated‑annealing hill‑climbing
   that repeatedly swaps one element of the support with an element outside.
   The swap is accepted if it does not increase the peak count, or with a
   probability that decays with a temperature parameter.
3. As a safety net, a short projected‑sub‑gradient optimisation (identical to the
   one used in the reference solution) is run on the current vector if time
   permits.

The whole search respects the 1000 s time budget.  The returned list of
floats contains non‑negative entries and can be fed directly into
`evaluate_sequence`.
"""
import time, math, random
import numpy as np

def _uniform_score_from_counts(max_count, m, n):
    """Score for uniform weights a_i = 1/m on a support of size m."""
    return float(2 * n * max_count / (m * m))

def _compute_counts(n, support):
    """Return ordered‑pair count array (convolution of indicator) for given support."""
    indicator = np.zeros(n, dtype=np.int8)
    indicator[support] = 1
    # np.convolve returns length 2n‑1 array
    return np.convolve(indicator, indicator)

def _random_support_search(n, support_sizes, attempts_per_size, deadline):
    """Search many random supports; return best support & its max count."""
    best_support = None
    best_max = None
    best_val = math.inf
    rng = np.random.default_rng()
    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[search] support size {m}")
        for _ in range(attempts_per_size):
            if time.time() > deadline:
                break
            # draw a random subset of size m
            S = rng.choice(n, size=m, replace=False)
            counts = _compute_counts(n, S)
            cur_max = int(counts.max())
            val = _uniform_score_from_counts(cur_max, m, n)
            if val < best_val:
                best_val = val
                best_support = S
                best_max = cur_max
                print(f"[found] m={m} score={val:.6f} max={cur_max}")
                if val <= 1.503:
                    return best_support, best_max, best_val
        print(f"[search] finished size {m}, best so far {best_val:.6f}")
    return best_support, best_max, best_val

def _hill_climb_support(support, n, max_iter, deadline, target_score):
    """
    Simulated‑annealing hill climbing on a fixed‑size support.
    Returns the best support found and its max count.
    """
    m = len(support)
    support_list = list(support)                # mutable copy
    support_set = set(support_list)
    counts = _compute_counts(n, support_list)
    cur_max = int(counts.max())
    best_support = support_list.copy()
    best_max = cur_max

    # temperature schedule
    T = cur_max * 0.5 + 1.0   # avoid zero
    it = 0
    while time.time() < deadline and it < max_iter:
        # pick a random element from the support and a random outsider
        i = random.choice(support_list)
        j = random.randint(0, n - 1)
        while j in support_set:
            j = random.randint(0, n - 1)

        s_arr = np.array(support_list, dtype=np.int32)

        # removal of i
        idx_remove = i + s_arr
        dec = np.where(s_arr == i, 1, 2).astype(np.int32)
        new_counts = counts.copy()
        np.subtract.at(new_counts, idx_remove, dec)

        # addition of j (with support minus i)
        mask = s_arr != i
        s_minus = s_arr[mask]
        idx_add = j + s_minus
        np.add.at(new_counts, idx_add, 2)
        new_counts[2 * j] += 1

        new_max = int(new_counts.max())

        # acceptance rule
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            # simulated‑annealing probability
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-8))
            if random.random() < prob:
                accept = True

        if accept:
            # commit the swap
            support_set.remove(i)
            support_set.add(j)
            # replace i with j in list (O(m) search)
            idx_i = support_list.index(i)
            support_list[idx_i] = j

            counts = new_counts
            cur_max = new_max

            # keep best
            if cur_max < best_max:
                best_max = cur_max
                best_support = support_list.copy()
                cur_val = _uniform_score_from_counts(best_max, m, n)
                print(f"[hill] iter {it} new best val {cur_val:.6f} max={best_max}")

                if cur_val <= target_score:
                    break

        # temperature decay
        it += 1
        if it % 1000 == 0:
            T *= 0.99  # exponential cooling

    return np.array(best_support, dtype=np.int32), best_max

def _subgradient_optimize(a_init, time_limit):
    """Projected sub‑gradient descent (same as reference) – simple copy."""
    a = a_init.copy()
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_val = float('inf')
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random transfer
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project back onto simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                # evaluate
                cand_val = 2.0 * n * np.max(np.convolve(cand, cand)) / (np.sum(cand) ** 2)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = 2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a) ** 2)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad] iter {it} new best {best_val:.6f}")

        it += 1

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function sequence with score ≤ 1.503.

    The function respects the 1000‑second budget.  It first tries many
    random supports, then refines the best one with a swap‑based
    simulated‑annealing hill climb, and finally falls back to a few
    iterations of the reference sub‑gradient optimiser.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    n = 1500
    target_score = 1.503

    # ------------------------------------------------------------------
    # 1. Random‑support phase
    support_sizes = [800, 700, 600, 500, 400, 300]
    # attempts per size – we will stop earlier if time runs out
    attempts_per_size = 20000
    best_support, best_max, best_val = _random_support_search(
        n, support_sizes, attempts_per_size, deadline
    )

    # ------------------------------------------------------------------
    # 2. Local optimisation on the best support
    if best_support is not None and best_val > target_score:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[hill] starting hill‑climb")
            max_iter = 5_000_000   # a large number, loop will stop by time
            best_support, best_max = _hill_climb_support(
                best_support, n, max_iter, deadline, target_score
            )
            best_val = _uniform_score_from_counts(best_max, len(best_support), n)
            print(f"[hill] finished, best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback continuous optimiser if still above target
    if best_val > target_score:
        remaining = deadline - time.time()
        if remaining > 2.0:
            print("[fallback] running sub‑gradient optimiser")
            # initialise with uniform weights on the current support
            m = len(best_support) if best_support is not None else n
            a0 = np.zeros(n, dtype=np.float64)
            if best_support is not None:
                a0[best_support] = 1.0 / m
            else:
                a0[:] = 1.0 / n
            a_opt = _subgradient_optimize(a0, remaining)
            best_val = 2.0 * n * np.max(np.convolve(a_opt, a_opt)) / (np.sum(a_opt) ** 2)
            best_seq = a_opt.tolist()
            print(f"[fallback] final value {best_val:.6f}")
            return best_seq

    # ------------------------------------------------------------------
    # 4. Build final sequence from the best support (uniform weights)
    if best_support is None:
        # as a last resort fall back to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        m = len(best_support)
        best_seq = np.zeros(n, dtype=np.float64)
        best_seq[best_support] = 1.0 / m

    return best_seq.tolist()

"""
Proposes a step‑function sequence with a low autocorrelation score.

Algorithm (≤ 1000 s):
1. Randomly sample many support sets (indices where the step height is non‑zero)
   of several sizes and keep the one with the smallest peak of the
   self‑convolution (uniform weight 1/m on the support).
2. Starting from that set, run a simulated‑annealing hill‑climber that swaps
   one index of the support with an index outside.  Swaps that do not increase
   the maximum count are always accepted; otherwise they are accepted with a
   temperature‑dependent probability.  The best set seen is remembered.
3. If the score is still above the target, a short projected sub‑gradient
   optimisation (the reference fallback) is applied to the full vector.

The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _uniform_score_from_counts(max_count: int, m: int, n: int) -> float:
    """Score for a uniform weight a_i = 1/m on a support of size m."""
    return float(2 * n * max_count / (m * m))


def _compute_counts(n: int, support) -> np.ndarray:
    """Convolution of the 0/1 indicator of ``support`` with itself (ordered pairs)."""
    indicator = np.zeros(n, dtype=np.int8)
    indicator[support] = 1
    # length of the result is 2·n‑1
    return np.convolve(indicator, indicator)


# ----------------------------------------------------------------------
# 1. Random‑support phase
# ----------------------------------------------------------------------


def _random_support_search(
    n: int,
    support_sizes,
    attempts_per_size: int,
    deadline: float,
) -> tuple:
    """Return best support, its max count and the associated score."""
    best_support = None
    best_max = None
    best_val = math.inf
    rng = np.random.default_rng()

    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[search] trying support size {m}")

        for _ in range(attempts_per_size):
            if time.time() > deadline:
                break

            S = rng.choice(n, size=m, replace=False)
            cnt = _compute_counts(n, S)
            cur_max = int(cnt.max())
            val = _uniform_score_from_counts(cur_max, m, n)

            if val < best_val:
                best_val = val
                best_support = S
                best_max = cur_max
                print(f"[found] m={m} score={val:.6f} max={cur_max}")

                if val <= 1.503:
                    return best_support, best_max, best_val

        print(f"[search] size {m} finished, best so far {best_val:.6f}")

    return best_support, best_max, best_val


# ----------------------------------------------------------------------
# 2. Hill‑climbing (simulated annealing) on a fixed‑size support
# ----------------------------------------------------------------------


def _hill_climb_support(
    support,
    n: int,
    max_iter: int,
    deadline: float,
    target_score: float,
) -> tuple:
    """
    Perform a swap‑based annealing search.
    Returns the best support found and its max count.
    """
    m = len(support)
    support_list = list(support)          # mutable
    support_set = set(support_list)
    counts = _compute_counts(n, support_list)
    cur_max = int(counts.max())
    best_support = support_list.copy()
    best_max = cur_max

    T = cur_max * 0.5 + 1.0   # starting temperature
    it = 0

    while time.time() < deadline and it < max_iter:
        # pick a random element of the support and a random outsider
        i = random.choice(support_list)
        j = random.randint(0, n - 1)
        while j in support_set:
            j = random.randint(0, n - 1)

        s_arr = np.array(support_list, dtype=np.int32)

        # --- remove i ----------------------------------------------------
        idx_remove = i + s_arr
        dec = np.where(s_arr == i, 1, 2).astype(np.int32)
        new_counts = counts.copy()
        np.subtract.at(new_counts, idx_remove, dec)

        # --- add j -------------------------------------------------------
        mask = s_arr != i
        s_minus = s_arr[mask]                     # support after removal of i
        idx_add = j + s_minus
        np.add.at(new_counts, idx_add, 2)
        new_counts[2 * j] += 1                    # (j,j) pair

        new_max = int(new_counts.max())

        # --- acceptance test ---------------------------------------------
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-8))
            if random.random() < prob:
                accept = True

        if accept:
            # commit the swap
            support_set.remove(i)
            support_set.add(j)
            idx_i = support_list.index(i)        # O(m), acceptable
            support_list[idx_i] = j

            counts = new_counts
            cur_max = new_max

            # record best ever seen
            if cur_max < best_max:
                best_max = cur_max
                best_support = support_list.copy()
                cur_val = _uniform_score_from_counts(best_max, m, n)
                print(f"[hill] iter {it} new best val {cur_val:.6f} max={best_max}")

                if cur_val <= target_score:
                    break

        # temperature decay
        it += 1
        if it % 1000 == 0:
            T *= 0.99

    return np.array(best_support, dtype=np.int32), best_max


# ----------------------------------------------------------------------
# 3. Reference sub‑gradient optimiser (fallback)
# ----------------------------------------------------------------------


def _subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (same as the reference fallback)."""
    a = a_init.copy()
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_val = float("inf")
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project back onto the simplex
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)

                cand_val = 2.0 * n * np.max(np.convolve(cand, cand)) / (np.sum(cand) ** 2)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = 2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a) ** 2)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad] iter {it} new best {best_val:.6f}")

        it += 1

    return best_a

    """
    Search for a step‑function whose score does not exceed 1.503.
    The algorithm respects the 1000 s time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    n = 1500
    target_score = 1.503

    # ------------------------------------------------------------------
    # 1. Random‑support phase
    # ------------------------------------------------------------------
    support_sizes = [800, 700, 600, 500, 400, 300]
    attempts_per_size = 8000

    best_support, best_max, best_val = _random_support_search(
        n, support_sizes, attempts_per_size, deadline
    )
    print(f"[random] best value after sampling {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Hill‑climbing refinement
    # ------------------------------------------------------------------
    if best_support is not None and best_val > target_score:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[hill] starting hill‑climb")
            max_iter = 5_000_000          # generous upper bound; loop ends by time
            best_support, best_max = _hill_climb_support(
                best_support, n, max_iter, deadline, target_score
            )
            best_val = _uniform_score_from_counts(best_max, len(best_support), n)
            print(f"[hill] finished, best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback sub‑gradient optimisation (if still above target)
    # ------------------------------------------------------------------
    if best_val > target_score:
        remaining = deadline - time.time()
        if remaining > 2.0:
            print("[fallback] running sub‑gradient optimiser")
            m = len(best_support) if best_support is not None else n
            a0 = np.zeros(n, dtype=np.float64)
            if best_support is not None:
                a0[best_support] = 1.0 / m
            else:
                a0[:] = 1.0 / n
            a_opt = _subgradient_optimize(a0, remaining)
            best_val = 2.0 * n * np.max(np.convolve(a_opt, a_opt)) / (np.sum(a_opt) ** 2)
            print(f"[fallback] final value {best_val:.6f}")
            return a_opt.tolist()

    # ------------------------------------------------------------------
    # 4. Build final sequence (uniform weights on the best support)
    # ------------------------------------------------------------------
    if best_support is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        m = len(best_support)
        best_seq = np.zeros(n, dtype=np.float64)
        best_seq[best_support] = 1.0 / m

    print(f"[final] returning sequence with score {best_val:.6f}")
    return best_seq.tolist()
