def project_simplex(v):
    # Project onto simplex {x>=0, sum x = 1}
    # v: np.ndarray.
    # Implementation from Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def conv_and_max(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    return conv, conv.max()
```

def gradient_active(a, active):
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i indexes lo...hi
        i_idx = np.arange(lo, hi+1)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g
```

def active_indices(conv, max_val, rel_eps=1e-4, max_add=30):
    eps = max_val * rel_eps
    K = np.where(conv >= max_val - eps)[0]
    # ensure at least some entries:
    if K.size < max_add:
        # take top max_add indices
        topk = np.argpartition(conv, -max_add)[-max_add:]
        K = np.unique(np.concatenate([K, topk]))
    return K

"""
Propose a candidate step‑function by directly minimising the
∞‑norm of the aperiodic auto‑convolution.
The method uses projected (additive) gradient descent on the simplex:
* the convolution is evaluated by an FFT (O(n log n)),
* the sub‑gradient of the current maximum is formed on a small
  “active’’ set of indices,
* an AdaGrad‑style step with momentum is taken,
* the iterate is projected back onto the probability simplex.
After the long descent a cheap stochastic hill‑climber (mass transfer)
exploits any remaining time.

The algorithm improves on the previous exponentiated‑gradient approach
by using an additive update (often more stable for this quadratic
objective) and by working with a larger active set.  It also scales to
larger lengths (default n = 10 000) – the theoretical lower bound for the
score is ≈ 1, so the hope is to push the current 1.5159‑score further
down.
"""

# ----------------------------------------------------------------------
# Imports
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -----------------------------------------------------

def _conv_and_max(a: np.ndarray):
    """
    Full aperiodic auto‑convolution via FFT and its maximum entry.
    Returns (conv, max_value).
    """
    n = a.shape[0]
    # next power of two ≥ 2n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv, conv.max()


def _active_indices(conv: np.ndarray, max_val: float,
                    rel_eps: float = 1e-4, max_add: int = 30):
    """
    Return a modest set of indices that are “active’’ for the sub‑gradient.
    All indices within `rel_eps` relative distance of `max_val` are kept.
    If fewer than `max_add` such indices exist, the globally largest ones
    are added so that the set size never exceeds `max_add`.
    """
    eps = max_val * rel_eps
    idx = np.where(conv >= max_val - eps)[0]
    if idx.size < max_add:
        # add a few globally biggest entries
        topk = np.argpartition(conv, -max_add)[-max_add:]
        idx = np.unique(np.concatenate([idx, topk]))
    return idx


def _gradient_active(a: np.ndarray, K: np.ndarray):
    """
    Sub‑gradient of `max(conv(a,a))` w.r.t. `a` using the active set `K`.
    For each active convolution index k the contribution to component i is
    2·a_{k‑i} (if the index is in range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    return g


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation follows Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # --------------------------------------------------------------
