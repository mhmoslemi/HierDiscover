def fast_score(a):
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s*s)
```

def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w
```

def smooth_optimize(a_init, deadline):

    best_a = a.copy()
    best_val = fast_score(a)
    n = a.shape[0]
    size_fft = 1 << ((3*n).bit_length())
    iteration = 0
    beta = 1.0
    beta_growth = 1.02  # multiply each iter after certain steps maybe
    eta0 = 0.1
    while time.time() < deadline:
        # compute c = conv(a,a) (full)
        fft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[:2*n-1]
        c = conv
        max_c = c.max()
        # compute softmax probabilities
        # increase beta gradually based on iteration
        if iteration % 1000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_growth)
        # compute p
        shifted = c - max_c  # to avoid overflow
        exp_term = np.exp(beta * shifted)
        sum_exp = exp_term.sum()
        p = exp_term / sum_exp
        # gradient: conv of p and a_rev
        # Compute gradient with FFT
        fft_p = np.fft.rfft(p, size_fft)
        fft_a_rev = np.fft.rfft(a[::-1], size_fft)
        grad_full = np.fft.irfft(fft_p * fft_a_rev, size_fft)
        grad = 2.0 * grad_full[:n][::-1]
        # update via exponentiated gradient (mirrored)
        eta = eta0 / np.sqrt(iteration + 1)
        a = a * np.exp(-eta * grad)   # multiplicative update
        a = simplex_projection(a)  # project onto simplex

                return best_a
        iteration += 1

"""
Optimise a non‑negative step function for the provided evaluation routine.

The algorithm consists of four phases:

1.  **Analytic families** – try symmetric Beta‑shaped vectors
    a_i ∝ x_i^{α‑1}(1‑x_i)^{α‑1} (α∈(0,1)).  These distributions are known
    to flatten the self‑convolution and often beat the naive uniform vector.

2.  **Random‑support search** – pick many random subsets S ⊂ {0,…,n‑1}
    of size m and use the uniform weight 1/m on S.  The score improves
    quickly as the number of trials grows, so we exhaust the remaining
    time on this cheap Monte‑Carlo phase.

3.  **Smooth max optimisation** – minimise a smooth approximation of
    the max‑convolution using an exponentiated‑gradient / mirror‑descent
    scheme.  At iteration t we compute the soft‑max probabilities
    p_k ∝ exp(β·c_k) where c = a*a; the gradient is
    g = 2·corr(p,a) (cross‑correlation).  The update a←a·exp(‑ηg) is
    followed by normalisation (projection onto the probability simplex).

4.  **Fallback (sub‑gradient)** – if the score is still above the target
    we run the simple sub‑gradient optimiser from the original solution
    as a safety net.

The whole search stops after the allotted 1000 s and returns the best
vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())          # power‑of‑two ≥ 2n‑1
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on a discrete grid, normalised to sum = 1."""
    # centre the grid at (i+0.5)/n ∈ (0,1)
    x = (np.arange(n) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


# ----------------------------------------------------------------------
# Phase‑3: smooth‑max optimiser (exponentiated gradient / mirror descent)
# ----------------------------------------------------------------------


def smooth_optimize(a0: np.ndarray, deadline: float, target: float = 1.503) -> np.ndarray:
    """
    minimise the smooth approximation of max_k (a*a)[k] on the simplex.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = fast_score(a)

    # size for FFTs that cover p (len 2n‑1) and a_rev (len n)
    size_fft = 1 << ((3 * n).bit_length())  # ≥ 3n
    it = 0
    # schedule for the soft‑max temperature
    beta = 1.0
    beta_max = 200.0
    beta_growth = 1.02          # multiplicative increase each 500 iterations
    eta0 = 0.5                  # base step size for exponentiated update

    while time.time() < deadline:
        # ---- compute convolution a*a -------------------------------------------------
        fft_a = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]   # linear conv
        c = conv_full                                          # shape (2n‑1,)

        # ---- soft‑max over the convolution entries ------------------------------------
        c_max = c.max()
        shifted = c - c_max                                   # ≤ 0, avoids overflow
        exp_term = np.exp(beta * shifted)                     # max term becomes 1
        sum_exp = exp_term.sum()
        p = exp_term / sum_exp                                 # probabilities, length 2n‑1

        # ---- gradient: g_i = 2·∑_j p_{i+j} a_j = 2·corr(p,a) (valid mode) ----------
        # np.correlate with mode='valid' gives exactly the needed sliding dot‑product.
        grad = 2.0 * np.correlate(p, a, mode='valid')        # shape (n,)

        # ---- exponentiated‑gradient / mirror‑descent update ---------------------------
        eta = eta0 / math.sqrt(it + 1.0)                     # diminishing step size
        a = a * np.exp(-eta * grad)                          # element‑wise multiply
        a_sum = a.sum()
        if a_sum == 0.0:                                     # numerical safeguard
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum                                     # projection onto simplex

        # ---- logging & best‑track ----------------------------------------------------
        if it % 200 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[smooth] iter {it:5d}  new best {best_val:.6f}")
                if best_val <= target:
                    print("[smooth] target reached – early exit")
                    return best_a

        # ---- temperature schedule ----------------------------------------------------
        if (it + 1) % 500 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase‑4: simple sub‑gradient optimiser (fallback, as in the reference code)
# ----------------------------------------------------------------------


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    size_fft = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # active convolution index
        fft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        a = simplex_projection(v)

        # occasional random mass transfer
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it:5d}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

    return best_seq.tolist()
