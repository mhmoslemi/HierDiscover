def _simulated_annealing(a, best_score, deadline, temperature_start=0.1, temperature_end=0.001, max_iters=200000, step_factor=0.2):
    compute n
    cur_a = a.copy()
    cur_score = compute_score(cur_a) (maybe using _conv_max)
    best_a, best_score = cur_a.copy(), cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        T = temperature_start * (temperature_end/temperature_start)**(it/max_iters)  # exponential decay
        # propose move
        i = np.random.randint(n)
        # ensure a_i >0
        if cur_a[i] <= 0:
            continue
        j = np.random.randint(n)
        if i == j: continue
        # delta = random fraction * min(step_factor * cur_a[i], cur_a[i])
        delta = cur_a[i] * step_factor * np.random.rand()
        if delta > cur_a[i]:
            delta = cur_a[i] # but delta computed from cur_a[i] so okay
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # no need to renormalize sum remains unchanged.

        # compute new_score
        new_score = 2 * n * _conv_max(new_a)[0]
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score)/T):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        it +=1

    return best_a, best_score

"""
Optimisation of a non‑negative step function for the objective

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The problem is scale‑invariant, so we keep all candidates on the probability
simplex (sum = 1, a_i ≥ 0).  The search consists of four phases:

1. **Multi‑scale Adam optimisation** – a list of increasing lengths is tried.
   For each length several initial shapes (upsampled best, uniform, triangular,
   Hann, Gaussian, Dirichlet samples) are optimised with a projected Adam
   optimiser that uses a *sub‑max* approximation of the ℓ∞‑norm:
   only the convolution entries within a shrinking relative tolerance of the
   current maximum receive gradient weight.

2. **Extra refinement** – after all candidates of a given length have been
   processed, the current best for that length is refined with more Adam
   steps and a slower learning‑rate schedule.

3. **Simulated annealing** – the globally best vector is then subjected to a
   cheap pairwise mass‑exchange neighbourhood search.  Moves that increase
   the objective are occasionally accepted according to an exponential cooling
   schedule, allowing the search to escape shallow local minima.

4. **Final Adam polish** – a short low‑temperature Adam run cleans up any
   remaining roughness.

All heavy work is performed on the simplex; after optimisation the vector
is renormalised to sum = 1 and returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (max(conv(a,a)), full convolution) for a vector a that sums to 1.
    The convolution is performed via FFT.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k (a*a)_k   with respect to a.
    Implements   g_i = 2 * Σ_k w_k * a_{k-i}
    using FFTs.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize_submax(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha_start: float = 0.25,
    alpha_end: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    adam_eps: float = 1e-8,
    eps_rel_start: float = 0.15,
    eps_rel_end: float = 1e-5,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimiser that minimises 2·n·max(conv) on the simplex.
    The max is approximated by a *sub‑max* weighting:
    indices whose convolution value lies within a shrinking relative tolerance
    of the current maximum receive equal weight.
    Returns (final_a, best_a_seen, best_score).
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment

    # initialise best
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # relative tolerance for sub‑max selection
        eps_rel = eps_rel_start + (eps_rel_end - eps_rel_start) * (it / max_iters)
        thresh = max_val * (1.0 - eps_rel)
        mask = conv >= thresh
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (linear decay)
        lr = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        step = lr / (np.sqrt(v_hat) + adam_eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return a, best_a, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    temp_start: float = 0.015,
    temp_end: float = 0.00015,
    step_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Simple pair‑wise mass‑exchange SA on the simplex.
    Returns (best_a, best_score) where the score is 2·n·max(conv) (sum is 1).
    """
    n = a.shape[0]
    cur_a = a.copy()
    max_val, _ = _conv_max(cur_a)
    cur_score = 2.0 * n * max_val
    best_a = cur_a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        # exponential temperature schedule
        t = temp_start * (temp_end / temp_start) ** (it / max_iters)

        # propose a transfer of mass from i to j
        i = rng.integers(n)
        if cur_a[i] <= 0.0:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        delta = cur_a[i] * step_factor * rng.random()
        if delta <= 0.0:
            it += 1
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # compute new score
        max_val_new, _ = _conv_max(new_a)
        new_score = 2.0 * n * max_val_new

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(t, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

        it += 1

    return best_a, best_score

                cand = np.full(n, 1.0 / n)
            else:
                cand = cand / cand_sum
            candidates.append(cand)

        # ------------------------------------------------------------------
        # 3. optimise each candidate at the current resolution
        # ------------------------------------------------------------------
        for cid, cand in enumerate(candidates, 1):
            if time.time() >= deadline:
                break

            # estimate iteration budget
            est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
            remaining = deadline - time.time()
            max_iters = int(min(150_000, max(15_000, remaining / est_iter_time)))
            if max_iters <= 0:
                break

            print(
                f"    [{time.time() - start_time:5.2f}s] cand {cid}/{len(candidates)} – iters={max_iters}"
            )

            _, local_best_a, local_best_score = _adam_optimize_submax(
                cand.copy(),
                max_iters=max_iters,
                deadline=deadline,
                alpha_start=0.25,
                alpha_end=0.02,
                eps_rel_start=0.15,
                eps_rel_end=1e-5,
            )

            if local_best_score < best_score:
                best_score = local_best_score
                best_seq = local_best_a.copy()
                print(
                    f"        NEW GLOBAL BEST (n={n}) – score = {best_score:.6f}"
                )

        # ------------------------------------------------------------------
        # 4. quick extra refinement of the current best at this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
            extra_iters = int(min(50_000, remaining / est_iter_time))
            if extra_iters > 0:
                print(
                    f"    [{time.time() - start_time:5.2f}s] extra refinement – iters={extra_iters}"
                )
                _, refined_a, refined_score = _adam_optimize_submax(
                    best_seq.copy(),
                    max_iters=extra_iters,
                    deadline=deadline,
                    alpha_start=0.20,
                    alpha_end=0.01,
                    eps_rel_start=0.08,
                    eps_rel_end=1e-5,
                )
                if refined_score < best_score:
                    best_score = refined_score
                    best_seq = refined_a.copy()
                    print(
                        f"        IMPROVEMENT AFTER REFINEMENT – score = {best_score:.6f}"
                    )

    # ------------------------------------------------------------------
    # 5. Simulated annealing on the overall best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
        max_sa_iters = int(min(200_000, remaining / est_iter_time))
        if max_sa_iters > 0:
            print(f"[{time.time() - start_time:6.2f}s] Simulated annealing – n={n}, iters~{max_sa_iters}")
            sa_best_a, sa_best_score = _simulated_annealing(
                best_seq.copy(),
                deadline=deadline,
                max_iters=max_sa_iters,
                temp_start=0.015,
                temp_end=0.00015,
                step_factor=0.25,
            )
            if sa_best_score < best_score:
                best_score = sa_best_score
                best_seq = sa_best_a.copy()
                print(f"        SA IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. final low‑temperature Adam polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 6e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(100_000, max(30_000, remaining / est_iter_time)))
        if final_iters > 0:
            print(f"[{time.time() - start_time:6.2f}s] Final Adam polish – n={n}, iters={final_iters}")
            _, final_a, final_score = _adam_optimize_submax(
                best_seq.copy(),
                max_iters=final_iters,
                deadline=deadline,
                alpha_start=0.10,
                alpha_end=0.002,
                eps_rel_start=0.02,
                eps_rel_end=1e-5,
            )
            if final_score < best_score:
                best_score = final_score
                best_seq = final_a.copy()
                print(f"        FINAL POLISH IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. post‑process & return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: modest uniform vector
        best_seq = np.full(200, 1.0 / 200)

    # enforce non‑negativity and normalise once more (just in case)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])

    # optional sanity check using the provided evaluator (for logging only)
    try:
        from __main__ import evaluate_sequence  # safeguard: maybe already imported
    except Exception:
        evaluate_sequence = None

    if evaluate_sequence is not None:
        eval_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Finished – evaluator score = {eval_score:.6f}")
    else:
        print(f"[{time.time() - start_time:6.2f}s] Finished – best internal score = {best_score:.6f}")

    return [float(x) for x in best_seq]
