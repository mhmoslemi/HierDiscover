"""
Search for a low‑autocorrelation step function.

The method consists of three phases:
1. Random‑support sampling (large number of trials).  Supports are
   binary vectors of a given size m with uniform weight 1/m.  The
   score depends only on the maximum entry of the convolution, which
   is cheap to evaluate with a FFT based routine.
2. Local improvement by swapping one element of the support with a
   zero while updating the convolution counts incrementally.
   Only non‑worsening swaps are kept; the best solution seen is stored.
3. If the target score 1.503 is not met, a projected sub‑gradient
   optimiser (the same one used in the reference solution) is run
   for the remaining time as a safety net.

The implementation respects the 1000 s time limit and never uses
closures or file/network IO.
"""
import time
import math
import numpy as np

def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the score without the safety checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def apply_swap(mask: np.ndarray, counts: np.ndarray, i: int, j: int) -> None:
    """Swap index i (currently in the support) with j (currently not)
    and update the convolution count array in place."""
    # remove i
    idx = np.where(mask)[0]          # includes i
    np.add.at(counts, i + idx, -2)
    counts[2 * i] += 1               # self‑pair was subtracted twice
    mask[i] = False

    # add j
    idx2 = np.where(mask)[0]         # support after removal (size m‑1)
    np.add.at(counts, j + idx2, 2)
    counts[2 * j] += 1
    mask[j] = True

def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # active convolution index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # occasional logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        it += 1

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function whose evaluation score is ≤ 1.503."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    n = 1500
    target_score = 1.503
    best_seq = None
    best_score = math.inf

    try:

            best_seq, best_score = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random‑support search (large number of trials)
    support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
    # allocate roughly 60 % of the total budget to this phase
    random_deadline = start + 0.6 * budget_s
    attempts = 0
    while time.time() < random_deadline:
        m = support_sizes[np.random.randint(len(support_sizes))]
        # draw a random support of size m
        S = np.random.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / m
        val = fast_score(a)
        attempts += 1
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[random] attempt={attempts} m={m} score={val:.6f}")
            if val <= target_score:
                print("[random] target reached")
                return best_seq.tolist()
        # stop after a very large number of attempts just in case
        if attempts >= 500000:
            break

    # ------------------------------------------------------------------
    # 2. Local improvement on the best support found so far
    if best_seq is not None:
        mask = best_seq > 0
        m = int(mask.sum())
        # convolution counts for the current support (ordered pairs)
        counts = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
        cur_score = best_score
        # maintain a dense list of indices belonging to the support
        support_idx = np.where(mask)[0].astype(np.int32)

        # simple hill‑climbing with swaps; stop when time is out
        while time.time() < deadline:
            # propose a swap
            i_pos = np.random.randint(m)
            i = int(support_idx[i_pos])
            j = np.random.randint(n)
            while mask[j]:

            apply_swap(mask, counts, i, j)

            new_score = 2.0 * n * new_max / (m * m)

            if new_score <= cur_score:
                # accept the move
                cur_score = new_score
                support_idx[i_pos] = j
                if new_score < best_score:
                    best_score = new_score
                    best_seq = np.zeros(n, dtype=np.float64)
                    best_seq[mask] = 1.0 / m
                    print(f"[local] new best score {new_score:.6f} (iters)")
                    if new_score <= target_score:
                        print("[local] target reached")
                        return best_seq.tolist()
            else:
                # revert the move
                apply_swap(mask, counts, j, i)

    # ------------------------------------------------------------------
    # 3. Fallback – projected sub‑gradient optimisation (if necessary)
    if best_score > target_score:
        remaining = deadline - time.time()

            init = np.full(n, 1.0 / n, dtype=np.float64)

            best_seq = subgradient_optimize(init, remaining)
            best_score = fast_score(best_seq)
            print(f"[fallback] final score {best_score:.6f}")

    # 4. Final safety net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = fast_score(best_seq)

    print(f"[final] returning sequence with score {best_score:.6f}")
    return best_seq.tolist()

"""
Search for a low‑autocorrelation step function.

The method consists of three phases:
1. Random‑support sampling (many trials).  Supports are binary vectors of a
   given size m with uniform weight 1/m.  The score depends only on the
   maximum entry of the convolution, which is cheap to evaluate with an
   FFT based routine.
2. Local improvement by swapping a support element with a zero while
   updating the convolution counts incrementally.  Only non‑worsening
   swaps are kept; the best solution seen is stored.
3. If the target score 1.503 is not met, a projected sub‑gradient optimiser
   (the same one used in the reference solution) is run for the remaining
   time as a safety net.

All computation respects the 1000 s time limit; the code never uses
closures, file or network I/O.
"""
import time
import math
import numpy as np


def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the score without the safety checks."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def apply_swap(mask: np.ndarray, counts: np.ndarray, i: int, j: int) -> None:
    """Swap index i (present) with j (absent) and update the convolution
    count array in place.  The mask is modified accordingly."""
    # remove i
    idx = np.where(mask)[0]                # includes i
    np.add.at(counts, i + idx, -2)
    counts[2 * i] += 1                     # self‑pair was subtracted twice
    mask[i] = False

    # add j
    idx2 = np.where(mask)[0]               # support after removal (size m‑1)
    np.add.at(counts, j + idx2, 2)
    counts[2 * j] += 1
    mask[j] = True


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # active convolution index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # occasional logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        it += 1

    return best_a


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function whose evaluation score is ≤ 1.503."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    n = 1500
    target_score = 1.503
    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 0. Initialise from a previously stored construction if available
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        if prev.size == n:
            val = fast_score(prev)
            best_seq, best_score = prev.copy(), val
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random‑support search (large number of trials)
    support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
    random_deadline = start + 0.6 * budget_s
    attempts = 0
    while time.time() < random_deadline:
        m = support_sizes[np.random.randint(len(support_sizes))]
        S = np.random.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / m
        val = fast_score(a)
        attempts += 1
        if val < best_score:
            best_score, best_seq = val, a.copy()
            print(f"[random] attempt={attempts} m={m} score={val:.6f}")
            if val <= target_score:
                print("[random] target reached")
                return best_seq.tolist()
        if attempts >= 500_000:
            break

    # ------------------------------------------------------------------
    # 2. Local improvement on the best support found so far
    if best_seq is not None:
        mask = best_seq > 0
        m = int(mask.sum())
        # convolution counts for the current support (ordered pairs)
        counts = np.convolve(mask.astype(np.int64), mask.astype(np.int64))
        cur_score = best_score
        support_idx = np.where(mask)[0].astype(np.int32)

        while time.time() < deadline:
            # propose a swap
            i_pos = np.random.randint(m)
            i = int(support_idx[i_pos])
            j = np.random.randint(n)
            while mask[j]:
                j = np.random.randint(n)

            # execute the swap
            apply_swap(mask, counts, i, j)
            new_max = counts.max()
            new_score = 2.0 * n * new_max / (m * m)

            if new_score <= cur_score:
                # accept move
                cur_score = new_score
                support_idx[i_pos] = j
                if new_score < best_score:
                    best_score = new_score
                    best_seq = np.zeros(n, dtype=np.float64)
                    best_seq[mask] = 1.0 / m
                    print(f"[local] new best score {new_score:.6f}")
                    if new_score <= target_score:
                        print("[local] target reached")
                        return best_seq.tolist()
            else:
                # revert move
                apply_swap(mask, counts, j, i)

    # ------------------------------------------------------------------
    # 3. Fallback – projected sub‑gradient optimiser (if needed)
    if best_score > target_score:
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[fallback] running sub‑gradient optimiser")
            init = np.full(n, 1.0 / n, dtype=np.float64)
            if best_seq is not None:
                init = best_seq.copy()
            best_seq = subgradient_optimize(init, remaining)
            best_score = fast_score(best_seq)
            print(f"[fallback] final score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final safety net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = fast_score(best_seq)

    print(f"[final] returning sequence with score {best_score:.6f}")
    return best_seq.tolist()
