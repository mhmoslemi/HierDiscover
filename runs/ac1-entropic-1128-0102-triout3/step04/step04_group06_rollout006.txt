"""
Propose a candidate step‑function using a *smoothed* mirror‑descent on the
simplex plus a simulated‑annealing hill‑climber.

The method works in two phases:

1. **Smoothed mirror‑descent** – we minimise a smooth approximation of
   the max‑convolution (log‑sum‑exp) using exponentiated updates.
   The smoothing parameter β is increased gradually so that early
   iterations see a gentler landscape (helps escaping plateaux) while
   later iterations focus on the true ∞‑norm.  The gradient of the smooth
   objective is obtained analytically as a correlation between the current
   sequence and the soft‑max weights; both convolutions are computed with
   FFTs (O(n log n)).

2. **Stochastic hill‑climbing** – with the remaining time we perform
   random mass‑transfer moves.  Moves are accepted either if they improve
   the score or with a Metropolis probability that slowly cools, giving
   the algorithm a chance to escape shallow local minima.

The algorithm automatically falls back to a random initial simplex point
or, if available, the previously discovered construction `height_sequence_1`.
It respects the 1000 s budget, prints progress, and returns the best
sequence found (as a list of non‑negative floats)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _autoconv_max(a: np.ndarray, fft_len: int):
    """
    Compute the linear auto‑convolution a*a (length 2n‑1) and its maximum.
    Returns (conv, max_val) where conv is a 1‑d ndarray of length 2n‑1.
    """
    # FFT of a (zero‑padded to fft_len) and point‑wise square.
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)
    n = a.shape[0]
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    return conv, max_val


def _weights_from_conv(conv: np.ndarray, beta: float):
    """
    Build a soft‑max weight vector from the convolution values.
    w_i ∝ exp(beta * (conv_i - max(conv)))  (max shifted to avoid overflow).
    """
    max_c = conv.max()
    # subtract max to keep the exponent ≤ 0
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int):
    """
    Gradient of the smoothed max‑convolution w.r.t. a.
    Implements g = 2 * Corr(a, w) using FFTs.

    The correlation Corr(a,w)[s] = Σ_i a_i * w_{i+s} for s = 0..n‑1.
    """
    n = a.shape[0]
    # reverse w for correlation via convolution
    w_rev = w[::-1]
    # convolution a * w_rev (linear) – length n + (2n‑1) - 1 = 3n‑2
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(w_rev, n=fft_len)
    conv_full = np.fft.irfft(Af * Bf, n=fft_len)
    # slice that corresponds to indices n‑1 … 2n‑2 (inclusive) → length n
    start = n - 1
    g_slice = conv_full[start:start + n]
    # reverse to obtain the proper ordering
    g = 2.0 * g_slice[::-1]
    return g


def _exp_mirror_update(a: np.ndarray, g: np.ndarray, eta: float):
    """
    Exponentiated‑gradient (mirror descent) update on the simplex.
    a ← a·exp(‑η·g)  then re‑normalise.
    """
    # avoid log(0)
    a = np.maximum(a, 1e-15)
    log_a = np.log(a) - eta * g
    # stabilise before exponentiation
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def _random_perturb(a: np.ndarray, magnitude: float = 0.03):
    """Transfer a small random fraction of mass between two random indices."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    eps = np.random.rand() * magnitude
    delta = eps * a[i]
    a[i] -= delta
    a[j] += delta
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _hill_climb_sa(a: np.ndarray,
                   n: int,
                   time_deadline: float,
                   start_score: float,
                   best_seq: np.ndarray,
                   best_score: float):
    """
    Stochastic hill‑climbing with a simple simulated‑annealing schedule.
    Attempts to improve the current best sequence until the deadline.
    """
    cur_score = start_score
    cur_seq = a.copy()
    # Temperature schedule – starts relatively hot and cools linearly.
    T0 = 0.02
    while time.time() < time_deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 0.04  # up‑to‑4 % of source mass
        delta = eps * cur_seq[i]
        if delta <= 0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0:
            continue
        # evaluate new score (sum is kept = 1, so denominator = 1)
        conv, _ = _autoconv_max(new_seq, _next_pow2(3 * n - 2))
        new_score = 2.0 * n * conv.max()
        # acceptance rule
        if new_score < cur_score:
            accept = True
        else:
            # linear cooling
            remaining = max(time_deadline - time.time(), 1e-8)
            frac = remaining / (time_deadline - start_time)
            T = T0 * frac
            accept = np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12))
        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve → {new_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights that (approximately) minimises
    evaluate_sequence.  The routine runs for at most `budget_s` seconds,
    printing occasional progress messages.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin so the runner does not kill us
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1. Problem dimensions & initial point
    # ------------------------------------------------------------------
    target_n = 5000                     # final length
    # try a previously discovered construction
    try:

        init = np.maximum(init, 0.0)
        if init.size != target_n:

            x_new = np.linspace(0.0, 1.0, target_n)
            init = np.interp(x_new, x_old, init)
    except Exception:                  # No previous construction available
        init = np.random.rand(target_n)

    # Normalise to a probability vector (sum = 1)
    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        a = np.full(target_n, 1.0 / target_n)
    else:
        a = init / s

    # Pre‑compute FFT length that works for both convolutions (autoconv and
    # correlation) – the larger of 2n‑1 and 3n‑2.
    fft_len = _next_pow2(3 * target_n - 2)

    # ------------------------------------------------------------------
    # 2. Smoothed mirror‑descent (the heavy lifting)
    # ------------------------------------------------------------------
    max_iters = 300_000                     # hard iteration cap
    eta0 = 0.2                               # initial step size
    beta_start, beta_end = 10.0, 200.0       # smoothing schedule limits
    beta_ramp = 80_000                       # iterations over which β grows
    best_seq = a.copy()
    conv, max_val = _autoconv_max(a, fft_len)
    best_score = 2.0 * target_n * max_val
    print(f"[{time.time() - start_time:6.2f}s] start – n={target_n}, score={best_score:.6f}")

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- smoothing parameter β (linear ramp, then constant) ---
        if it <= beta_ramp:
            beta = beta_start + (beta_end - beta_start) * it / beta_ramp
        else:
            beta = beta_end

        # --- compute convolution and soft‑max weights ---
        conv, _ = _autoconv_max(a, fft_len)
        w = _weights_from_conv(conv, beta)

        # --- gradient of the smoothed max ---
        g = _gradient_from_weights(a, w, fft_len)

        # --- exponentiated mirror‑descent update ---
        eta = eta0 / math.sqrt(it)          # diminishing step size
        a = _exp_mirror_update(a, g, eta)

        # occasional tiny random perturbations to shake off plateaus
        if it % 4000 == 0:
            a = _random_perturb(a, magnitude=0.025)

        # evaluate current objective (exact max, not the smoothed version)
        conv, max_val = _autoconv_max(a, fft_len)
        cur_score = 2.0 * target_n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:6d}  improve → {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        conv, _ = _autoconv_max(a, fft_len)
        cur_score = 2.0 * target_n * conv.max()
        best_seq, best_score = _hill_climb_sa(a,
                                              target_n,
                                              deadline,
                                              cur_score,
                                              best_seq,
                                              best_score)

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] finished – best score={best_score:.6f}")
    return [float(x) for x in best_seq]
