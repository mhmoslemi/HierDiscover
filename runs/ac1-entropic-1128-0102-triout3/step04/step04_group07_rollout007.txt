"""
Propose a low‑autocorrelation step function.

The algorithm works in two stages:

1. **Discrete‑set local search** –  we keep a binary indicator
   `x[i] ∈ {0,1}` (the support of the step function) of size `m`.
   For a given support the score equals

        score = 2·n·max_count / m²,

   where `max_count` is the largest number of ordered pairs
   `(i,j)∈S×S` with `i+j = k`.  Using FFT we evaluate `max_count`
   quickly and perform a biased random‑swap hill‑climb:
        * pick an index that participates in the currently maximal sum,
        * swap it with a random element outside the support,
        * accept the swap if the score improves (or with a tiny
          uphill probability to escape plateaux).

   The search is executed for several support sizes (≈ 500 … 1300)
   while respecting the overall time budget.

2. **Continuous refinement** –  the best binary support found is turned
   into a uniform probability vector `a = x / m` (so that
   `∑a = 1`).  Starting from this vector we run the projected
   sub‑gradient optimiser from the previous reference implementation.
   The optimiser works on the original continuous evaluation function
   and further reduces `max(a*a)`.

The whole procedure respects the 1000 s limit, keeps the best vector
found and returns it as a list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast score for a binary indicator (integer counts)
def _score_from_indicator(indic: np.ndarray):
    """
    Compute the evaluation score from a binary indicator vector.

    Parameters
    ----------
    indic : np.ndarray, shape (n,)
        0/1 array, 1 where the step‑function has a positive height.

    Returns
    -------
    score : float
        The value 2·n·max_count / m²  (where m = Σindic).
    conv  : np.ndarray, shape (2·n‑1,)
        Integer convolution (ordered‑pair counts) – needed for the
        local‑search heuristic.
    """
    n = indic.size
    m = int(indic.sum())
    if m == 0:
        return math.inf, None

    # length of FFT (next power of two ≥ 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_x = np.fft.rfft(indic.astype(np.float64), size)
    conv = np.fft.irfft(fft_x * fft_x, size)[: 2 * n - 1]
    # Round to nearest integer (numerical noise may produce tiny fractions)
    conv_int = np.rint(conv).astype(np.int64)
    max_count = int(conv_int.max())
    score = 2.0 * n * max_count / (m * m)
    return score, conv_int


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (same as the reference code)
def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the original evaluation function.
    The routine stops after *time_limit* seconds.
    """
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        # active convolution index
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        # occasional logging
        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a

# Fast (float) evaluation – identical to the reference implementation
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:

    Search for a step‑function with a low evaluation score.

    The search consists of a combinatorial hill‑climb on binary supports
    (fast integer convolution) followed by a short continuous
    sub‑gradient optimisation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Global parameters
    n = 1500                                     # length of the sequence
    support_sizes = [1300, 1200, 1100, 1000, 900,
                    800, 700, 600, 500]         # larger → smaller expected score
    max_swaps_per_size = 20000                  # swaps per support size (adjusted by time)

    best_score = math.inf
    best_indicator = None                       # binary vector (0/1)

    # ------------------------------------------------------------------

    try:

            best_indicator = (prev > 0).astype(np.int8)
            print(f"[init] previous sequence score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. discrete‑set local search
    for m in support_sizes:
        if time.time() > deadline:
            break

        # initialise a random support of size m
        support = np.random.choice(n, size=m, replace=False)
        indicator = np.zeros(n, dtype=np.int8)
        indicator[support] = 1

        # position lookup: index → position inside `support` array
        pos_of = {int(idx): i for i, idx in enumerate(support)}

        cur_score, conv = _score_from_indicator(indicator)
        # keep global best if this start is already better
        if cur_score < best_score:
            best_score = cur_score
            best_indicator = indicator.copy()
            print(f"[start] m={m} initial score {cur_score:.6f}")

        swaps = 0

            # bias: try to remove an element contributing to the maximal sum
            k_max = int(np.argmax(conv))

            # indices of the current support that have a partner giving the maximal sum
            # i participates iff (k_max - i) is also in the support
            diff = k_max - support
            mask = (diff >= 0) & (diff < n) & (indicator[diff] == 1)
            active_candidates = support[mask]

            if active_candidates.size > 0:
                # pick a victim from the active set
                i_out = int(np.random.choice(active_candidates))
                out_pos = pos_of[i_out]
            else:
                # fall back to a completely random element
                out_pos = np.random.randint(m)
                i_out = int(support[out_pos])

            i_in = np.random.randint(0, n)

            # perform the swap (tentative)
            indicator[i_out] = 0
            indicator[i_in] = 1

            # update the position map

                conv = new_conv
                swaps += 1

                indicator[i_out] = 1
                support[out_pos] = i_out
                del pos_of[i_in]
                pos_of[i_out] = out_pos
                swaps += 1
