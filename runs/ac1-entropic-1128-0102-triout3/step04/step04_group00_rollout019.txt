def _conv_self_fft(a):
    n = len(a)
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _grad_flat(a):
    conv = _conv_self_fft(a)
    target = 1.0 / (2 * len(a) - 1)
    e = conv - target
    a_rev = a[::-1]
    # convolution e * a_rev via FFT:
    n = len(a)
    # zeros pad e and a_rev to length L >= len(e) + n - 1 = (2n-1)+n-1 = 3n-2
    fft_len = 1 << ((3 * n - 2).bit_length())
    Ef = np.fft.rfft(e, n=fft_len)
    Ar = np.fft.rfft(a_rev, n=fft_len)
    conv_ea = np.fft.irfft(Ef * Ar, n=fft_len)[: 3 * n - 2]
    # extract slice n-1 : n-1+n

"""
A fast multi‑stage optimizer for the step‑function problem.

Stage 1 – tiny problem (n≈800) : a few random restarts, short Adam‑style
gradient descent.

Stage 2 – medium size (n≈2400) : up‑sample the best vector and continue the
descent.

Stage 3 – large size (n≈4800) : use the bulk of the time, now the descent
optimises a *combined* sub‑gradient:
    • the true sub‑gradient of max(conv) (active set);
    • a “flattening” gradient that pushes the whole convolution towards the
      constant value 1/(2·n‑1).  Their linear combination yields a
      stronger descent direction.
    Momentum (Nesterov) and a diminishing step size are employed.

Stage 4 – cheap local mass‑transfer hill‑climbing : repeatedly move a small
fraction of mass from a heavy index to a light one; keep any improvement.
All vectors are projected onto the probability simplex after each step.
The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found (normalised to sum = 1) as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All indices with conv[k] ≥ max·(1‑rel_tol) are taken as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thr = max_val * (1.0 - rel_tol)
    act = np.where(conv >= thr)[0]

    g = np.zeros_like(a)
    for k in act:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of a_i a_j wrt a_i adds a_j, and vice‑versa.
        g[i] += a[j]
        g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Gradient that tries to flatten the whole convolution
# ----------------------------------------------------------------------
def _gradient_flat_from_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k (conv_k - c)^2  where  c = 1/(2n‑1).
    Implements  g = 4·(e * a_rev)  with e = conv - c.
    """
    n = a.shape[0]
    c = 1.0 / (2 * n - 1)
    e = conv - c
    a_rev = a[::-1]

    # length needed for linear convolution e * a_rev
    L = 1 << ((len(e) + len(a_rev) - 1).bit_length())
    Ef = np.fft.rfft(e, n=L)
    Ar = np.fft.rfft(a_rev, n=L)
    conv_e_rev = np.fft.irfft(Ef * Ar, n=L)[: len(e) + len(a_rev) - 1]

    # slice the part that corresponds to indices 0 … n‑1
    grad = 4.0 * conv_e_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_opt(a0: np.ndarray, n: int,
             deadline: float, start_time: float,
             beta: float = 0.92,
             step0: float = 0.3,
             flat_weight: float = 0.15,
             max_iter: int = 10 ** 7):
    """
    Optimises the objective 2·n·max(a*a) using a combined gradient.
    Returns the best vector (sum = 1) encountered and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    eps = 1e-12
    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter improve n={n} iter={it:,} score={cur_score:.6f}")

        # combine the two gradients
        g_max = _subgradient_full(a, conv)
        g_flat = _gradient_flat_from_conv(a, conv)
        g = g_max + flat_weight * g_flat

        # Adam moments
        m = beta * m + (1.0 - beta) * g
        v = beta * v + (1.0 - beta) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta ** it)
        v_hat = v / (1.0 - beta ** it)

        # diminishing step size
        step = step0 / math.sqrt(it)

        # update + projection onto simplex
        a = _project_simplex(a - step * m_hat / (np.sqrt(v_hat) + eps))

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and re‑normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate seq to length new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Very cheap greedy mass‑transfer refinement
# ----------------------------------------------------------------------
def _local_improvement(a0: np.ndarray, n: int,
                       deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from a heavy index to a light one.
    Keeps any improvement.  Fast because only one convolution per move.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.30  # up‑to 30 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] local improve: n={n} score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a list of floats (sum ≈ 1).  The routine follows a multi‑stage
    schedule and respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try to start from the global `height_sequence_1` if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 1 – tiny problem (n≈800), a couple of random restarts
    # ------------------------------------------------------------------
    n_small = 800
    time_small = max(20.0, 0.05 * (deadline - time.time()))   # ~5 % of budget
    small_deadline = time.time() + time_small

    # two short restarts (one from starter if possible)
    for r in range(2):
        if starter is not None and r == 0:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()
        a_opt, sc = _run_opt(a0, n_small, small_deadline, start_time,
                             beta=0.85, step0=0.4, flat_weight=0.12)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 improve restart={r} score={sc:.6f}")
        if time.time() >= deadline:
            break

    if best_seq is None:
        # fallback – uniform
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 2 – medium size (n≈2400)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_mid = 2400
        a_mid = _upsample(best_seq, n_mid)
        time_mid = max(30.0, 0.15 * (deadline - time.time()))   # ~15 %
        mid_deadline = time.time() + time_mid
        a_opt, sc = _run_opt(a_mid, n_mid, mid_deadline, start_time,
                             beta=0.90, step0=0.3, flat_weight=0.15)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve score={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – large size (n≈4800) – main optimisation phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 4800
        a_big = _upsample(best_seq, n_big)
        time_big = max(10.0, (deadline - time.time()) - 30.0)   # leave ~30 s for final tweaks
        big_deadline = time.time() + time_big
        a_opt, sc = _run_opt(a_big, n_big, big_deadline, start_time,
                             beta=0.93, step0=0.25, flat_weight=0.1)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve score={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap local hill‑climbing (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, sc = _local_improvement(best_seq, best_seq.shape[0],
                                       deadline, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"Final best score (2·n·max(conv)) = {best_score:.6f}")
    return [float(x) for x in best_seq]
