def project_simplex(v):
    # returns projection onto simplex {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑autocorrelation step function.

The search consists of three stages:

1. **Random‑sparse support** – try many uniform‑weight vectors with a
   randomly chosen support of size `m`.  This very quickly finds
   candidates with score ≲ 1.6.

2. **Continuous optimisation (Adam + soft‑max smoothing)** – starting
   from the best vector found so far, minimise a smooth approximation
   of the objective `max_k (a*a)_k` using Adam.  The smoothing parameter
   `alpha` is increased gradually, so the optimiser first flattens the
   whole convolution and later focuses on the true maximum.  After each
   update the vector is projected onto the probability simplex
   (non‑negative, sum = 1).

3. **Final fallback** – if time remains a simple uniform vector is used.

All operations respect the 1000 s budget; intermediate progress and the
best score are printed.  The returned list can be fed directly to
`evaluate_sequence`.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm from
    # Duchi et al., "Efficient Projections onto the l1‑Ball for
    # Learning in High Dimensions", ICML 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------
def adam_optimize(
    a_init: np.ndarray,
    time_limit: float,
    target_score: float,
    verbose: bool = True,
) -> np.ndarray:
    """
    Minimise `max_k (a*a)_k` by smoothing the max with a soft‑max.
    The optimisation runs for at most `time_limit` seconds and returns
    the best vector seen.
    """
    a = a_init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = fast_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02           # base learning rate
    t = 0

    # soft‑max schedule
    alpha = 1.0         # start with a mild smoothing
    alpha_inc = 0.03    # increase per iteration
    max_alpha = 30.0    # final focus on the true max

    start = time.time()
    deadline = start + time_limit - 0.2   # tiny safety margin

    while time.time() < deadline:
        # 1) convolution a*a
        size = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]

        # 2) soft‑max weights over the convolution entries
        scaled = conv * alpha
        max_s = scaled.max()
        exp_scaled = np.exp(scaled - max_s)   # stabilise
        w = exp_scaled / exp_scaled.sum()    # w_k ≥ 0, Σ w_k = 1

        # 3) gradient of  Σ_k w_k * (a*a)_k  = 2 * corr(w, a)
        #    correlation with 'valid' mode yields exactly
        #    grad_i = 2 * Σ_j w_{i+j} a_j
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # 4) Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step            # we minimise, so go opposite to gradient
        a = project_simplex(a)  # enforce a_i ≥ 0, Σ a_i = 1

        # 5) occasional evaluation
        if t % 200 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                if verbose:
                    print(f"[adam iter {t}] new best {best_val:.6f}")
                if best_val <= target_score:
                    if verbose:
                        print("[adam] target reached – early exit")
                    return best_a

        # 6) increase smoothing (focus more on the true maximum)
        alpha = min(alpha + alpha_inc, max_alpha)

        # occasional tiny random jitter to escape plateaux
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * 0.001
            a = project_simplex(a + jitter)

    return best_a

    best_seq: np.ndarray | None,
    best_val: float,
    time_budget: float,
    target_score: float,
) -> tuple[np.ndarray, float]:
    """
    Exhaustively try random supports of several sizes.  The function stops
    as soon as the target score is reached or the time budget expires.
    """
    start = time.time()
    deadline = start + time_budget
    # support sizes – larger m gives less variance, but interior peak
    # is always about twice the overall average.  Empirically m≈700–900
    # works well for n=1500.
    support_sizes = [1300, 1200, 1100, 1000, 900, 800, 700, 600, 500]
    attempts_per_size = 5000   # many attempts – still cheap thanks to FFT

    for m in support_sizes:
        if time.time() > deadline:
            break
        print(f"[randsearch] trying support size {m}")
        for attempt in range(attempts_per_size):
            if time.time() > deadline:
                break
            # draw a random subset of size m
            S = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[S] = 1.0 / m
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[found] m={m} attempt={attempt+1} score={val:.6f}")
                if best_val <= target_score:
                    print("[randsearch] target reached – returning early")
                    return best_seq, best_val
        print(f"[randsearch] size {m} finished, best so far {best_val:.6f}")

    return best_seq, best_val

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a score ≤ 1.503.
    The routine combines random‑support sampling with a continuous
    Adam‑based optimiser, obeys the 1000 s time limit and returns the
    best sequence found.
    """
    np.random.seed()               # make runs non‑deterministic
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin
    target_score = 1.503

    # ------------------------------------------------------------------
    # Length of the vector – we keep the previously used length 1500,
    # but fallback to any length present in a previously stored solution.
    n = 1500

    best_seq: np.ndarray | None = None
    best_val = math.inf

    # ------------------------------------------------------------------
    # 0) initialise from a previously discovered construction if available
    try:
        # pylint: disable=undefined-variable
        prev = np.array(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            n = int(prev.size)          # respect the historic length
            val = fast_score(prev)
            best_seq, best_val = prev.copy(), val
            print(f"[init] previous sequence (len={n}) score {val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) random‑support search (large portion of the time budget)
    remaining = deadline - time.time()
    if remaining > 2.0:
        # allocate at most half the remaining budget to the random phase
        rand_budget = min(remaining * 0.5, 250.0)   # cap to keep enough time for optimisation
        best_seq, best_val = random_support_search(
            n=n,
            best_seq=best_seq,
            best_val=best_val,
            time_budget=rand_budget,
            target_score=target_score,
        )
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 2) continuous optimisation with Adam (fallback)
    if best_val > target_score and remaining > 5.0:
        init = best_seq if best_seq is not None else np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[adam] optimisation start – {remaining:.1f}s left, initial score {fast_score(init):.6f}")
        opt_seq = adam_optimize(
            a_init=init,
            time_limit=remaining,
            target_score=target_score,
            verbose=True,
        )
        opt_val = fast_score(opt_seq)
        if opt_val < best_val:
            best_val, best_seq = opt_val, opt_seq.copy()
            print(f"[adam] final best score {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) safety net (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = fast_score(best_seq)

    print(f"[final] returning sequence of length {n} with score {best_val:.6f}")
    return best_seq.tolist()
