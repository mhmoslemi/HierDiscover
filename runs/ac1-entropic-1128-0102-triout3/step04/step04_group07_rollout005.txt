def fast_score(a):
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return math.inf
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s*s))

"""
Propose a low‑autocorrelation step‑function.

The algorithm consists of three stages:

1. **Massive random‑support search** –  draws many uniformly‑weighted
   vectors of the form a_i = 1/m on a random set of m indices.
   For a length‑1500 vector the expected score is close to 1, so with
   enough draws we quickly find a candidate below the target 1.503.

2. **Simple hill‑climbing (simulated‑annealing style)** – starting from the
   best random vector, repeatedly moves a small amount of mass from one
   index to another.  The move is accepted if it improves the score or,
   with a temperature‑controlled probability, if it is slightly worse.
   This refines the best candidate found in stage 1.

3. **Fallback sub‑gradient optimiser** – if the target has not been met,
   a projected sub‑gradient descent (the same routine used previously)
   is run for the remaining time.

All evaluations use a fast FFT‑based implementation of the objective
function; the final list of floats is returned directly for the provided
`evaluate_sequence` routine.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only needed for the optional LP fallback


def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return math.inf
    # FFT length: next power of two >= 2n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def subgradient_optimize(a_init: np.ndarray, time_limit: float) -> np.ndarray:
    """Projected sub‑gradient descent (fallback optimiser)."""
    a = a_init.copy()
    best_a = a.copy()
    best_val = fast_score(a)
    n = a.size
    conv_size = 1 << ((2 * n - 1).bit_length())
    eta0 = 0.30
    eta_decay = 1e-5
    it = 0
    deadline = time.time() + time_limit - 0.1
    idx = np.arange(n)

    while time.time() < deadline:
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        grad = np.zeros_like(a)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad

        # Euclidean projection onto the simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
        else:
            theta = 0.0
        a = np.maximum(v - theta, 0.0)

        # occasional random mass transfer (helps escape plateaux)
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.1 * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # project again
                u = np.sort(cand)[::-1]
                cssv = np.cumsum(u) - 1.0
                cond = u - cssv / ind > 0
                if np.any(cond):
                    rho = ind[cond][-1]
                    theta = cssv[cond][-1] / rho
                else:
                    theta = 0.0
                cand = np.maximum(cand - theta, 0.0)
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_a = cand_val, cand.copy()
                    a = cand
                elif np.random.rand() < 0.001:
                    a = cand

        if it % 2000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[subgrad iter {it}] new best {best_val:.6f}")
        it += 1

    print(f"[subgrad final] best value {best_val:.6f}")
    return best_a
