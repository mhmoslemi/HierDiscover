def _exp_optimize(a, max_iters, deadline, eta_start=0.1, eta_end=0.001, temp_start=0.05, temp_end=0.0005):
    n = len(a)
    best_score = 2*n*max_conv(a)
    best_a = a.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        max_val, conv = _conv_max(a)
        cur_score = 2*n*max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute softmax weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        w = np.exp((conv - max_val)/tau)
        w_sum = w.sum()
        if w_sum >0: w = w / w_sum
        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)  # as before
        # step size - decreasing
        eta = eta_start + (eta_end - eta_start) * (it / max_iters)

        # exponentiated update
        a = a * np.exp(-eta * g)  # elementwise multiplication
        # renormalize to simplex
        a = _project_simplex(a)
    return a,best_a,best_score
```

"""
Propose a low‑score step‑function using a multi‑scale
exponentiated‑gradient optimizer (mirror descent on the simplex).
The procedure:

1.  Start from the best known construction (`height_sequence_1`) if it
    exists, otherwise from a uniform vector.
2.  Iterate over a schedule of lengths (≈1.5× growth up to 8000).
    For each length we:
        * up‑sample the current global best,
        * add a few random Dirichlet starts,
        * run a *soft‑max* exponential‑gradient descent (mirror descent)
          on the probability simplex,
        * keep the best candidate.
3.  After the size loop we perform a long refinement of the overall
    best sequence, first with Adam (as a sanity check) and then with
    the exponential‑gradient optimizer with a very slow learning‑rate
    schedule.
4.  Return the final sequence as a plain list of floats.

The optimiser minimises 2·n·max(conv(a,a)) which is exactly the score
returned by ``evaluate_sequence`` for a normalised vector (sum(a)=1).
The use of the Kullback–Leibler mirror map (exponentiated updates)
keeps all entries non‑negative and often finds better minima than
plain projected Adam for this simplex‑constrained, non‑convex problem.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return the maximum entry of the auto‑convolution of `a`
    (assumed to sum to 1) and the full convolution.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. `a`.
    Implements g_i = 2 * Σ_k w_k * a_{k-i} via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _exp_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    eta_start: float = 0.15,
    eta_end: float = 0.001,
    temp_start: float = 0.05,
    temp_end: float = 0.0005,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) optimiser on the simplex.
    Returns (final_a, best_a, best_score), where the score is
    2·n·max(conv(a,a)) (i.e. the quantity evaluated by evaluate_sequence
    for a normalised vector).
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and (raw) score
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # learning‑rate schedule (decreasing)
        eta = eta_start + (eta_end - eta_start) * (it / max_iters)

        # exponentiated update (mirror‑descent with KL divergence)
        # protect against overflow/underflow
        delta = -eta * g
        # clip to avoid extreme exponentials
        delta = np.clip(delta, -30.0, 30.0)
        a = a * np.exp(delta)

        # re‑project onto the simplex
        a = _project_simplex(a)

    return a, best_a, best_score


def _adam_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.05,
    temp_end: float = 0.0005,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv) on the simplex (kept for a final
    polishing pass).  Returns (final_a, best_a, best_score).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        g = _gradient_from_weights(a, w, fft_len_grad)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return a, best_a, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling of a probability vector."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up = up / s
    return up

            prev_seq = None
    except Exception:
        prev_seq = None

    sizes = []
    n = 1200
    MAX_N = 8000
    while n <= MAX_N:

    best_seq = None          # best probability vector (sum = 1)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main multi‑scale loop
    # ------------------------------------------------------------------
    for stage, n in enumerate(sizes, 1):
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] Stage {stage}/{len(sizes)} – n = {n}")

        # --------------------------------------------------------------
        # Build a modest pool of starting candidates for this size
        # --------------------------------------------------------------
        candidates = []
