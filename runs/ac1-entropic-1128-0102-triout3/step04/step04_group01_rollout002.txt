"""
Propose a low‑maximum self‑convolution step‑function.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1).  The evaluation function is
  scale‑invariant, so the objective becomes 2·n·max(a∗a).
* Multi‑scale optimisation: start with a modest length, repeatedly up‑sample
  the current best distribution to a larger size and use it as warm‑start.
* For each size run several *random‑restart* smooth‑gradient descents.
  The smooth surrogate is the log‑sum‑exp of the convolution entries; its
  gradient is obtained analytically via FFTs.  An Adam‑style update with
  projection onto the simplex is used.
* After the gradient phase a cheap simulated‑annealing polishing moves a
  random fraction of mass from one index to another (Metropolis acceptance).
  The temperature decays geometrically.
* The sequence is forced to be symmetric after each update; this halves the
  effective dimensionality and improves the chance of reaching a low peak.
* The whole procedure respects the 1000 s budget, dynamically splitting the
  remaining time between the different lengths and restarts.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry and renormalise to sum = 1."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a via a power‑of‑two FFT (real inputs)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max over convolution entries
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float,
                        step0: float = 0.8,
                        beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Smoothed log‑sum‑exp descent with Adam‑style moment estimation.
    Returns the best vector seen (as a probability vector) and its exact
    objective value 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    eps = 1e-12
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    # λ schedule parameters
    lam_factor = 0.30
    lam_decay = 0.65
    lam_decay_interval = 2500

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_score:.6f}")

        # λ update (gradually tighten the surrogate)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)
        lam = max(max_b * lam_factor, 1e-12)

        # surrogate gradient
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta * m + (1.0 - beta) * grad
        v = 0.999 * v + (1.0 - 0.999) * (grad ** 2)
        m_hat = m / (1.0 - beta ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        # step size slowly decays (1/√it) to keep stability for large it
        current_step = step0 / math.sqrt(it)

        a_new = a - current_step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex and enforce symmetry
        a = _project_simplex(a_new)
        a = _symmetrize(a)

        # occasional tiny isotropic noise to shake out of shallow plateaux
        if it % 500 == 0:
            noise_scale = 0.02 / math.sqrt(it)
            a += noise_scale * np.random.randn(n)
            a = _project_simplex(a)
            a = _symmetrize(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings annealing that transfers a random fraction
    of mass from one index to another.  The temperature decays geometrically.
    Returns the best sequence seen and its exact objective value.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # propose a transfer move
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move up to 30 % of the source mass
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # re‑symmetrise and renormalise
        a_prop = _symmetrize(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
