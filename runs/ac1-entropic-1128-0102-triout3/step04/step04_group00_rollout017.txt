def _run_adam_descent(a0, n, deadline, start_time,
                      step0=0.2,
                      beta1=0.9,
                      beta2=0.999,
                      eps=1e-8,
                      max_iter=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # compute subgradient
        grad = _subgradient_full(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)
    return best_a, best_score

        c = -np.ones(n)

                                    bounds=bounds, method='highs', options={'presolve': True})

        else:
            # if no improvement maybe reduce tau

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  

The algorithm mixes three ideas that complement the gradient‑descent
approach used previously:

1. **Arcsine start** – a distribution that puts most mass near the
   boundaries is a good heuristic for a low autocorrelation peak.
2. **Projected Adam optimisation** – an adaptive‑step, momentum based
   optimiser (Adam) on the simplex gives a faster, more stable descent
   than the plain Nesterov scheme.
3. **Linear‑programming refinement** – for a fixed shape `f` the problem
   `max_i (f * g)_i ≤ M` is linear in `g`.  By maximising `∑ g_i`
   under this constraint we obtain a new candidate `g` that can be mixed
   with the current best vector, guaranteeing a non‑increase of the
   objective.  Only a handful of LP solves are performed (they are cheap
   with HiGHS).

The search proceeds through three sizes (800 → 2400 → 4800), performing
gradient descent on each size, a short LP refinement, and a final
simulated‑annealing polishing phase.  Time is split dynamically so the
whole routine respects the 1000 s budget.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max(conv) (all active entries)
# ------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    All entries with conv[k] ≥ max(conv)*(1‑rel_tol) are taken as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of a_i * a_j term
        g[i] += a[j]
        g[j] += a[i]

    return g


# ------------------------------------------------------------
# Adam optimiser on the simplex
# ------------------------------------------------------------
def _run_adam_descent(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      step0: float = 0.3,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam with simplex projection.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Linear‑programming refinement
# ------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> diags:
    """
    Build the (2n‑1)×n convolution matrix T such that T @ b = conv(a,b).
    Returned as a scipy.sparse.csc_matrix.
    """
    n = a.shape[0]
    data = []
    offsets = []
    for off in range(n):
        length = n - off
        data.append(np.full(length, a[off], dtype=float))
        offsets.append(off)

    T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T


def _run_lp_refine(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   max_lp_steps: int = 3,
                   tau: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements:
    maximise Σ b_i   s.t.  b_i ≥ 0  and  conv(a,b) ≤ M,
    where M = max(conv(a,a)).  After each solve mix b into a.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for step in range(max_lp_steps):
        if time.time() >= deadline:
            break

        T = _conv_matrix(a)
        M = _conv_self(a).max()
        c = -np.ones(n)                     # maximise Σ b_i  ⇔ minimise -Σ b_i
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            # if the LP fails, abort further refinements
            break

        b = res.x
        sb = b.sum()
        if sb == 0.0:
            break

        # normalise the new candidate and mix
        b_norm = b / sb
        a = (1.0 - tau) * a + tau * b_norm
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, "
                f"step={step+1}, score={cur_score:.6f}"
            )
        else:
            # no improvement – shrink the mixing factor
            tau *= 0.5
            if tau < 1e-4:
                break

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-7,
                   steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer SA: move ε from a high‑mass index to a random one.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ------------------------------------------------------------
# Arcsine‑type initial distribution (mass near the ends)
# ------------------------------------------------------------
def _initial_arcsine(n: int) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n) + 0.5          # centre of each bin
    x = i / n                       # in (0,1)
    eps = 1e-12
    a = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), eps))
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.8

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 0 – cheap initial candidate (arcsine)
    # ------------------------------------------------------------------
    n_small = 800
    a0 = _initial_arcsine(n_small)

    # ------------------------------------------------------------------
    # Stage 1 – Adam descent on the small problem
    # ------------------------------------------------------------------
    slice_deadline = min(deadline, time.time() + 40.0)   # ~4 % of total time
    a_small, score_small = _run_adam_descent(
        a0, n_small, slice_deadline, start_time,
        step0=0.30, beta1=0.90, beta2=0.990
    )
    if score_small < best_score:
        best_score = score_small
        best_seq = a_small.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score_small:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to medium size and longer Adam run
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    remaining = deadline - time.time()
    # allocate roughly half of the remaining budget to this
    mid_time = min(remaining * 0.5, 260.0)
    mid_deadline = time.time() + mid_time

    a_mid_opt, score_mid = _run_adam_descent(
        a_mid, n_mid, mid_deadline, start_time,
        step0=0.25, beta1=0.90, beta2=0.995
    )
    if score_mid < best_score:
        best_score = score_mid
        best_seq = a_mid_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 2‑LP – a short LP refinement at medium resolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(remaining * 0.15, 120.0)
        a_mid_lp, score_mid_lp = _run_lp_refine(
            best_seq, n_mid, lp_deadline, start_time,
            max_lp_steps=3, tau=0.15
        )
        if score_mid_lp < best_score:
            best_score = score_mid_lp
            best_seq = a_mid_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2‑LP improve: score={score_mid_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – up‑sample to the full size and Adam optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    remaining = deadline - time.time()
    # keep ~20 % of the remaining time for a final LP, the rest for Adam
    lp_time = min(remaining * 0.20, 120.0)
    gd_time = remaining - lp_time - 30.0                     # 30 s reserved for SA

    # LP refinement on the big problem (very cheap, only a couple of steps)
    lp_deadline = time.time() + lp_time
    a_big_lp, score_big_lp = _run_lp_refine(
        a_big, n_big, lp_deadline, start_time,
        max_lp_steps=2, tau=0.12
    )
    if score_big_lp < best_score:
        best_score = score_big_lp
        best_seq = a_big_lp.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 3‑LP improve: score={score_big_lp:.6f}")

    # Adam descent on the full resolution
    if time.time() < deadline:
        gd_deadline = time.time() + gd_time
        a_big_opt, score_big = _run_adam_descent(
            best_seq, n_big, gd_deadline, start_time,
            step0=0.20, beta1=0.95, beta2=0.998
        )
        if score_big < best_score:
            best_score = score_big
            best_seq = a_big_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, best_seq.shape[0],
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-7,
            steps=800_000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
