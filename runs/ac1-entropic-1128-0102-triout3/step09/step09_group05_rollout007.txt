"""
Hybrid smooth‑max Adam + sub‑gradient + targeted annealing optimiser.

The algorithm has three stages:

1️⃣  Adam optimisation on a smooth (log‑sum‑exp) surrogate of the
    maximal self‑convolution entry.  The surrogate uses a temperature
    parameter α that is increased gradually, and additionally a
    small “flatness” penalty (λ·‖conv−mean‖²) is added to the gradient.
    This spreads the mass more evenly and usually pushes the maximal
    convolution below the previous best.

2️⃣  Projected sub‑gradient descent (as in the reference code) to
    clean up any remaining spikes.

3️⃣  Targeted pair‑annealing that moves mass away from the two indices
    that currently contribute most to the peak of the convolution.

All phases respect the non‑negativity and simplex constraints
(and the evaluate_sequence routine caps values automatically).  The
routine adapts its time allocation: ∼40 % of the budget for the Adam
phase, ∼30 % for sub‑gradient, the rest for annealing, and performs a
few short random restarts if time permits.

Returns a list of non‑negative heights (the step‑function) that
minimises ``evaluate_sequence`` within the provided time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv) (sum of coefficients is assumed 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent (phase 2 of the original pipeline)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Pair‑annealing (phase 3 of the original pipeline)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 180_000):
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct destinations
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

                          starter)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            raise Exception
    except Exception:
        final_n = 4096
        a = np.random.rand(final_n)
        a /= a.sum()

    conv = _conv_self(a, L_self)
    best_score = _score_from_conv(conv, n)
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] init score: {best_score:.6f}")

    # Phase 1 – Adam on a smooth max + flatness penalty.
    # --------------------------------------------------------------
    # allocate ~40 % of remaining budget
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(20.0, 0.40 * remaining)

    # parameters for the Adam loop
    L_grad = _next_pow2(3 * n - 2)           # length needed for a * rev‑vector
    alpha = 1.0
    alpha_max = 80.0
    alpha_inc_iter = 5000
    lam = 0.5                                 # flatness weight λ
    lam_decay = 0.99996                       # per‑iteration decay
    lr0 = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < phase1_deadline:
        it += 1

        # ---------- self‑convolution (full FFT) ----------
        Af = np.fft.rfft(a, n=L_grad)
        conv_full = np.fft.irfft(Af * Af, n=L_grad)
        conv = conv_full[: 2 * n - 1]                # length 2n‑1

        # ---------- smooth‑max weights ----------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # ---------- gradient of smooth max ----------
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr_max = np.fft.irfft(Af * Wrev, n=L_grad)
        idx_base = len(w_rev) - 1                     # = 2n‑2
        grad_max = 2.0 * corr_max[idx_base - np.arange(n)]

        # ---------- flatness gradient ----------
        avg = 1.0 / (2 * n - 1)                        # sum(a)=1 ⇒ mean conv
        diff = conv - avg
        diff_rev = diff[::-1]
        Drev = np.fft.rfft(diff_rev, n=L_grad)
        corr_flat = np.fft.irfft(Af * Drev, n=L_grad)
        grad_flat = 2.0 * corr_flat[idx_base - np.arange(n)]

        # combined gradient
        grad = grad_max + lam * grad_flat

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        lr = lr0 / math.sqrt(it)
        step = lr * m / (np.sqrt(v) + eps)
        a = a - step

        # projection onto the simplex (non‑negative & sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

            if cur_score < best_score:

                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            alpha = min(alpha_max, alpha * 1.5)
        lam *= lam_decay

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (classic)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(10.0, 0.30 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad phase improve: "
                  f"score={score_sub:.6f}")

    # Phase 3 – Targeted pair‑annealing (as in the reference code)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=180_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # --------------------------------------------------------------
    # Optional quick random restarts if time still left
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(30.0, time_left / 2.0)
        restart_deadline = time.time() + block

        pert = best_seq + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short sub‑gradient run
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        if time.time() < deadline:
            pert_seq, pert_score = _run_pair_annealing(
                pert_seq,
                n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={pert_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return as plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # final check (optional, only for logging)
    try:
        import evaluate_sequence  # the function is already available in the environment
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Final evaluate_sequence score: {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
