    n = a.shape[0]

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

        cur_max = conv.max()
        # compute objective reliably:
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # indices achieving max
        # Use tolerance relative to max perhaps 1e-12
        # If multiple indices, sum contributions.
        max_val = cur_max

        for k in ks:

"""
Propose a step‑function that beats the previous bound (~1.5088).

Algorithm
---------
1. **Coarse analytic sweep** –  Try the symmetric “tent’’ family
   a_i ∝ min(i,n‑1‑i)^p for many exponents p∈[0.05,0.9].
   The best of these gives a solid starting point.

2. **Random Dirichlet restarts** –  Generate a few random probability
   vectors (Dirichlet draws) and keep any that improve the bound.

3. **Projected Adam optimisation** –  Starting from the best vector,
   run a projected Adam optimiser on the exact (non‑smooth) objective.
   The sub‑gradient of the max‑convolution is computed exactly
   (including the missing a_i term that the reference code omitted).
   After each update we project back onto the probability simplex.

4. **Secondary Adam run from a perturbed point** –  Add a tiny random
   perturbation to the best iterate and run a second Adam phase,
   allowing us to escape shallow local minima.

5. **Light simulated‑annealing polish** –  Random mass transfers
   (with an exponential cooling schedule) give a final push.

All steps respect the 1000 s time budget, work on a single CPU and
return a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, not circular)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {w≥0, Σw = 1} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected Adam optimiser for the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the true objective.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.20                     # base learning rate
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                f"score={cur_score:.6f}"
            )

        # -------- sub‑gradient of max(conv) ----------
        max_val = cur_max
        # tolerance – the convolution values are exact up to ~1e‑15,
        # therefore a tight absolute tolerance works.
        ks = np.where(np.abs(conv - max_val) < 1e-12)[0]

        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # derivative of conv[k] w.r.t. a_i is a_i + a_{k-i}
            grad[idx] += a[idx] + a[k - idx]

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Projection back onto the simplex (keeps sum = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the reference)
# ----------------------------------------------------------------------
def _anneal_polish(a: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Tiny random mass‑transfer moves with exponential cooling.
    Returns the best sequence found before `deadline`.
    """
    n = a.shape[0]
    current_score = _objective(a)
    best_seq = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    total_steps = 200_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # pick a source proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - current_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"Anneal iter={it:,} best={new_score:.6f}"
                )

    return best_seq, best_score
