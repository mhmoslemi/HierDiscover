"""
Hybrid local‑search for the minimum ℓ∞‑norm of a self‑convolution.

The algorithm combines four loosely‑coupled phases that each respect the
global 1000 s wall‑clock budget:

1️⃣  **Exponentiated‑gradient warm‑up** – maximises a smoothed version of
    the max (log‑sum‑exp) and quickly moves the distribution away from the
    uniform plateau.

2️⃣  **Projected Adam‑style sub‑gradient descent** on the exact,
    non‑smoothed ℓ∞‑norm.  The adaptive step‑sizes of Adam allow a much
    larger number of effective updates than a plain 1/√t schedule.

3️⃣  **Aggressive mass‑shifting refinement** – repeatedly transfers a
    fraction of mass from indices that heavily contribute to the current
    maximal convolution bin to indices that contribute little.  For every
    chosen pair the exact largest admissible step size (solution of a
    1‑D LP) is computed, guaranteeing a monotone decrease of the maximum.

4️⃣  **Final log‑sum‑exp polishing** – a short run of the exponential‑
    gradient with a very large smoothing parameter finishes the search
    and removes remaining tiny peaks.

All phases keep track of the best sequence seen and return it at the end.
The code uses only NumPy and the provided `evaluate_sequence` (indirectly)
and fits comfortably inside the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose value
    lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` to length `new_n` while preserving total mass.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates on the log‑sum‑exp smoothed max.
    Returns the best sequence seen in this phase together with its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ----- smoothed‑max gradient -----
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ----- multiplicative update -----
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected Adam‑style sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray, n: int,
                          deadline: float, start_time: float,
                          step0: float = 0.5,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑like updates on the true ℓ∞‑norm.
    Returns the best sequence discovered in this phase and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam‑subgrad improve: "
                  f"iter={t:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        # Adam update (note: we treat step0 as a global scale)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # Projection onto the simplex (ensures non‑negativity and sum=1)
        a = _project_simplex(a)

    return best_a, best_score

                                   eps_end: float = 1e-4,
                                   rel_improve: float = 1e-12,
                                   recompute_every: int = 4000) -> np.ndarray:
    """
    Repeatedly shift a fraction of mass from a “source’’ index that heavily
    participates in the current maximal convolution bin to a “destination’’
    index that participates little.  For each chosen pair the exact largest
    admissible step size (solution of a 1‑D LP) is computed, guaranteeing the
    biggest safe improvement per iteration.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        # linear schedule for the fraction of source mass we are allowed to move
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # index of the current maximum entry of the convolution
        k_max = int(np.argmax(conv))

        # ----- pick a source index -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fallback – pick a heavy element
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- pick a destination index -----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- linear effect of moving ε mass from src→dst -----
        # s[t] = -2·a[t‑src] + 2·a[t‑dst]   (out‑of‑range entries are zero)
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_k = s[k_max]
        if s_k >= 0.0:
            # the move would not lower the current max – skip
            continue

        # ----- compute maximal admissible ε before another bin overtakes the max -----
        mask = s > s_k                     # slopes steeper than at the max bin
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_k                       # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ----- apply the move analytically (O(n)) -----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - rel_improve:
            # genuine improvement – commit the move
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to curb floating‑point drift
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Phase‑4: final short exponential‑gradient polishing
# ----------------------------------------------------------------------
def _run_final_polish(a0: np.ndarray, n: int,
                     deadline: float, start_time: float) -> np.ndarray:
    """
    A brief high‑alpha exponential‑gradient run that removes any remaining
    tiny spikes.  The learning rate is kept very small so the move is only
    a fine adjustment.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha = 200.0                     # large enough to essentially be the true max
    eta0 = 0.05                       # tiny step size for polishing

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -20.0, 20.0))
        a = a * upd
        a /= a.sum()

    return best_a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The routine follows the four‑phase scheme described at the top
    of the file.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  4800 gave good results historically; a slightly
    # larger size can give a little more flexibility without inflating runtime.
    # ------------------------------------------------------------------
    n = 5400

    # ------------------------------------------------------------------
    # Initial distribution: try to reuse a previously discovered construction
    # if it exists, otherwise fall back to a random Dirichlet sample.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter / starter.sum(), n)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient warm‑up
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + min(30.0, 0.08 * (deadline - time.time()))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam‑style sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve roughly 40 % of the remaining budget for phase 3 and 4
        phase2_deadline = time.time() + max(20.0, 0.55 * remaining)

        a_sub, score_sub = _run_adam_subgradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            step0=0.5,
            beta1=0.9,
            beta2=0.999,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive optimal incremental refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_optimal_incremental_refine(
            best_seq,
            n,
            deadline,
            start_time,
            eps_start=0.15,
            eps_end=1e-4,
            rel_improve=1e-13,
            recompute_every=3000,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing with a high‑alpha exponential‑gradient
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = time.time() + max(5.0, 0.05 * (deadline - time.time()))
        a_final = _run_final_polish(best_seq, n, polish_deadline, start_time)
        conv_final = _conv_self(a_final)
        score_final = 2.0 * n * conv_final.max()
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final polish improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
