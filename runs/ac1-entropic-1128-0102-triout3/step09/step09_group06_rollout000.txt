def propose_candidate(seed=0, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    n = 8000
    rng = np.random.default_rng(seed)

    # keep global best
    best_a = _initial_candidate(n, rng)  # includes prior sequence if exists
    best_score = float(evaluate_sequence(best_a.tolist()))

    # Phase 1: Multiple Adam restarts (50% time)
    adam_deadline = start_time + 0.5*budget_s

    # compute total restarts approximate
    restarts = 5
    for r in range(restarts):
        if time.time() >= adam_deadline:
            break
        # allocate time for this restart
        remaining = adam_deadline - time.time()
        # share equally
        restart_time = remaining / (restarts - r)
        # generate candidate
        if r == 0:
            a0 = best_a.copy()
        else:
            # random Dirichlet with concentration drawn
            alpha = rng.uniform(0.1, 1.0)
            gamma = rng.gamma(shape=alpha, scale=1.0, size=n)
            a0 = gamma / np.sum(gamma)
        # run Adam
        a_opt, score_opt = _adam_optimize_long(a0, time.time() + restart_time, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
    # Phase 2: Subgradient polishing (30% time)
    subgrad_deadline = min(deadline, time.time() + 0.3*budget_s)
    best_a, best_score = _subgrad_optimize(best_a, subgrad_deadline, start_time,

    # Phase 3: Fast greedy refinement (remaining time)
    if time.time() < deadline:

    best_a = np.clip(best_a, 0.0, None)

    if total > 0:

    else:
        best_a = np.full_like(best_a, 1.0 / n)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")
    return best_a.tolist()

def _adam_optimize_long(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3*n-2)
    # schedule extended
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0] 

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self(a)
            # smoothed max
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradient
            grad = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # Adam update
            t += 1
            m = beta1 * m + (1-beta1) * grad
            v = beta2 * v + (1-beta2) * (grad**2)
            m_hat = m / (1-beta1**t)
            v_hat = v / (1-beta2**t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a_new)

            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha}) score={sc:.6f}")

    return best_a, best_score

def _random_local_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = len(a)
    best_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()
    while time.time() < deadline:

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst: continue
        eps = a[src] * rng.uniform(0.001, 0.02)
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # ensure nonnegative

        a_new = np.clip(a_new, 0, None)
        # renormalize
        s = a_new.sum()
        if s <= 0:

        a_new = a_new / s
        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
    return best_a, best_score

"""
Advanced multi‑restart optimiser for the step‑function problem.

Strategy
--------
1. **Initialisation** – use the historic best sequence if present,
   otherwise fall back to a uniform distribution.  Several random
   Dirichlet starts are also generated.

2. **Adam on a smooth max** – perform projected Adam on the smooth
   approximation of the max‑convolution, using an extended schedule of
   smoothing parameters (α up to 8192).  The optimiser is run several
   times (restarts) within the first half of the time budget; after each
   α‑phase a tiny Gaussian perturbation is added to escape plateaus.

3. **Sub‑gradient polishing** – classic projected sub‑gradient descent
   on the true max‑convolution objective with momentum.

4. **Fast greedy refinement** – move a small fraction of the coefficient
   that contributes most to the current peak toward the smallest
   coefficient, updating the convolution incrementally.

5. **Normalisation** – clip negatives, renormalise to sum = 1 and return.

The routine respects the 1000 s budget, prints progress information and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre the correlation to obtain gradient entries 0..n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:

    Prefer the historic best sequence if available, otherwise use uniform.
    """

                new_x = np.linspace(0.0, 1.0, n)

            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:

    except Exception:

    # uniform fallback
    return np.full(n, 1.0 / n)


def _random_dirichlet_candidate(n: int, rng: np.random.Generator,
                               concentration: float = 0.4) -> np.ndarray:
    """
    Return a random probability vector drawn from Dirichlet(α,…,α)
    with the given concentration parameter (α).
    """
    # Gamma draws are a fast way to sample a Dirichlet.
    gamma = rng.gamma(shape=concentration, scale=1.0, size=n)
    if gamma.sum() == 0:
        return np.full(n, 1.0 / n)
    return gamma / gamma.sum()

# Phase 1 – multi‑restart Adam optimisation on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max function.
    The run ends when `deadline` is reached.  Returns the best vector
    (as a probability vector) and its exact evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT length large enough for all convolution‑related transforms
    fft_len = _next_pow2(3 * n - 2)

    # Extended smoothing schedule – larger α approximates the true max ever more closely
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.20                     # relatively aggressive initial learning rate

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        # reset Adam's moments for each smoothing level – helps when α jumps
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        while time.time() < deadline:
            # compute full convolution once per iteration
            conv = _conv_self(a)

            # ---- smooth max value (for logging) ----
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                     # smoothed max
            # gradient of the outer factor 2·n (as in evaluate_sequence)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # ---- Adam update ----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)                 # simple decay with iteration count

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha}) "
                          f"score={sc:.6f}")

            # safety break – the outer while also checks the deadline
            if time.time() >= deadline:
                break

        # after finishing one α‑stage, add a minuscule random jitter
        if time.time() < deadline:
            noise = rng.normal(scale=1e-4, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# Phase 2 – classic sub‑gradient descent (fine‑grained polishing)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.90,
                      step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns best vector and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # --- identify the active set (indices achieving the max) ---
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # --- sub‑gradient (sum of contributions from active convolutions) ---
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each term a[i] * a[j] contributes a[j] to ∂/∂a[i] (and symmetrically)
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless for non‑negative a

        # --- momentum + diminishing step size ---
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase 3 – fast greedy peak reduction (incremental updates)
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Greedy scheme that moves a modest fraction of the coefficient most
    responsible for the current convolution peak toward the smallest
    coefficient.  Convolution updates are performed incrementally (O(n)).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        # indices that participate in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index most responsible for the peak

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # transfer a modest fraction of the source value
        eps = 0.05 * a[src]                     # 5 % of the source mass
        if eps < 1e-14:
            break

        # incremental update of the convolution array
        conv_tmp = conv.copy()
        # linear part: -2·eps·a shifted by src  + 2·eps·a shifted by dst
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        # quadratic part (self‑contributions)
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # strict improvement required
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: iter={it:,}, max={new_max:.6f}")

    return best_a, best_score
