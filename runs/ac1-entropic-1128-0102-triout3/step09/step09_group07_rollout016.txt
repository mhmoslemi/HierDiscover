"""
Improved search for a low‑max self‑convolution step‑function.

The optimiser works in three stages:

1️⃣  **Exponentiated‑gradient warm‑up** – a log‑sum‑exp smoothed
    approximation of the ℓ∞‑norm is minimised with an Adam‑style
    multiplicative update.  The schedule of the smoothing parameter
    α is gradually increased so that the optimisation first sees a
    very smooth landscape and then refines the true max.

2️⃣  **Projected sub‑gradient descent** – the exact ℓ∞‑norm of the
    self‑convolution is used.  Momentum (Polyak averaging) gives a
    fast first‑order routine on the probability‑simplex.

3️⃣  **Deterministic incremental refinement** – each iteration
    chooses a *source* index that contributes most to the current
    maximal convolution bin and a *destination* index that contributes
    the least.  The move “transfer ε mass from src to dst” changes *all*
    convolution entries linearly; the largest feasible ε that still
    reduces the current maximum is obtained analytically (a tiny 1‑D
    linear programme).  Because ε is as large as possible, each step
    yields a substantial improvement while remaining O(n).  The
    procedure runs until the time budget expires, constantly keeping
    the best sequence seen.

All three phases share the same FFT‑based convolution routine; the
overall wall‑clock budget (default 1000 s) is obeyed strictly.  The
function returns the best non‑negative height vector (normalised to sum
= 1) found within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 – exponentiated‑gradient (Adam‑style) on smoothed max
# ----------------------------------------------------------------------
def _run_exp_grad(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  alphas,
                  eta0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8):
    """Adam‑style multiplicative updates on a log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"score={cur_score:.6f}")

        # current smoothing parameter
        alpha = alphas[min(t // 2000, len(alphas) - 1)]

        # smoothed‑max (log‑sum‑exp) gradient w.r.t. a
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        # reversed because convolution with reversed weights gives gradient
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wr = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wr, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam moments (in log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update (ensuring positivity)
        upd = np.exp(-step)
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – projected sub‑gradient descent (exact ℓ∞‑norm)
# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 step0: float = 0.4,
                 beta: float = 0.95):
    """Standard sub‑gradient descent with momentum on the exact max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient (all entries within 1e‑12 of the max)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – deterministic incremental refinement
# ----------------------------------------------------------------------
def _run_det_refine(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    eps_start: float = 0.05,
                    eps_end: float = 0.001,
                    recompute_every: int = 4000):
    """
    Aggressive mass‑transfer moves.
    At each iteration:
        * src  = index with maximal contribution a[i]·a[k_max−i]
        * dst  = index with minimal contribution (excluding src)
    The largest feasible ε that strictly lowers the current maximal
    convolution entry is computed analytically.  The move is applied
    (O(n) work) and the process repeats.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        # linear schedule for the maximum step‑size fraction
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # ---- compute contributions to the current maximal bin ----
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_arr = np.arange(i_low, i_high + 1, dtype=int)
        j_arr = k_max - i_arr
        contrib = a[i_arr] * a[j_arr]

        # source: index with largest contribution (and non‑zero mass)
        src_candidates = i_arr[contrib > 0]
        if src_candidates.size == 0:
            # fallback – any positive entry
            pos = np.where(a > 0)[0]
            if pos.size == 0:
                break
            src = rng.choice(pos)
        else:
            src = src_candidates[np.argmax(contrib[contrib > 0])]

        # destination: index with the smallest contribution (excluding src)
        # we consider the whole vector – even indices that do not
        # contribute to k_max (their contribution is zero) are perfect
        # destinations.
        dst = int(np.argmin(a))          # smallest height overall
        if dst == src:
            # pick a second smallest if needed
            cand = np.argsort(a)
            dst = int(cand[1] if cand[0] == src else cand[0])

        # ---- linear effect of moving ε from src to dst ----
        # s[t] = -2·a[t‑src] + 2·a[t‑dst]   (out‑of‑range = 0)
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # the chosen pair cannot reduce the current maximum – try a random perturb
            continue

        # ---- maximal admissible ε before another entry overtakes the max ----
        mask = s > s_max                     # slopes strictly larger than s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ---- apply the move analytically (still O(n)) ----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - 1e-12:                     # genuine improvement
            a[src] -= epsilon
            a[dst] += epsilon
            conv = conv_new
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] det‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid drift
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
