def _simulated_annealing(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Scale initial temperature based on objective
    T0 = max(0.01, best_score * 0.1)
    T = T0
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Choose two indices i, j (i != j)
        i, j = rng.choice(n, size=2, replace=False)
        # Transfer a fraction epsilon from i to j
        eps = a[i] * rng.uniform(0.0, 0.1)  # up to 10% of source value
        if eps < 1e-12:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # evaluate
        score_new = evaluate_sequence(a_new.tolist())
        delta = score_new - best_score
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # accept with probability exp(-delta/T)
            if rng.random() < math.exp(-delta / T):
                a = a_new
        # schedule temperature decay
        if iteration % 1000 == 0:
            # linear cooling
            T = T0 * (1 - (time.time() - start_time) / (deadline - start_time))
            if T < 1e-6:
                T = 1e-6
            # optional: occasional random restart from best_a
    return best_a, best_score

"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm Overview
------------------
* **Rich initial pool** – a collection of handcrafted starts (uniform,
  exponential, power‑law, linear, Hann/Blackman windows,
  Dirichlet draws with various concentrations and the previous best
  sequence if available).

* **Heavy Adam warm‑up** – a long Adam optimisation of a smooth
  log‑sum‑exp surrogate of the maximal self‑convolution.
  The smoothing parameter α is increased gradually up to 4096 and
  many iterations (thousands per α) are performed, keeping the best
  candidate seen.

* **Fine‑grained exponentiated‑gradient polishing** – a multiplicative
  mirror‑descent on the *true* objective, using the gradient of the
  currently active shift (exact sub‑gradient).  The step size decays as
  η∝1/√t.  This phase runs for the bulk of the time budget.

* **Optional simulated‑annealing fine‑tuning** – a cheap random‑move
  search that perturbs the current best vector while a temperature
  schedule allows occasional uphill moves.  It works on the fast
  FFT‑based objective and stops when the remaining time is exhausted.

* **Deterministic greedy peak reduction** – the same deterministic move
  used previously, applied at the very end for a last strict improvement.

All vectors are kept on the probability simplex (sum = 1) and clipped
to ``[0, 1000]`` only at the final return, as required by the judge.
The routine respects a user‑supplied ``budget_s`` (default 1000 s) and
will always return the best sequence discovered before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _fast_objective(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the exact objective (no safety checks)."""
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    return 2.0 * a.shape[0] * max_b / (s * s)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1) Adam optimisation of a soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   alpha_schedule=None,
                   iters_per_alpha=3000,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best vector (probability vector) and its exact score.
    """
    if alpha_schedule is None:
        alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 1024.0, 2048.0, 4096.0]

    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))          # logits for the soft‑max

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector (always sums to 1)
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smooth surrogate value and gradient
            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # true objective (used only for logging)
            true_obj = 2.0 * n * conv.max()

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            mmax = scaled.max()
            w = np.exp(scaled - mmax)
            w /= w.sum()                               # soft‑max weights over shifts
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max (log‑softmax)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on the logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.0f}, score={sc:.6f}"
                    )

    return best_a, best_score

# 2) Exponentiated‑gradient optimisation of the true objective
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on f(a)=max_k (a*a)[k].
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # gradient from the *single* shift attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]                     # sub‑gradient for the active shift

        # step size (diminishing)
        eta = eta0 / math.sqrt(t)

        # multiplicative update – move opposite to the gradient
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation (guard against drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] ExpGrad exact eval: score={sc:.6f}"
                )

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,

    """
    Simple SA that moves a random fraction of mass from one index to another.
    Uses the fast FFT‑based objective for speed.
    """
    rng = np.random.default_rng(12345)
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _fast_objective(a, fft_len)

    # temperature schedule – start relatively high, cool linearly
    T0 = max(0.1, best_score * 0.2)
    T = T0

    it = 0
    while time.time() < deadline:
        it += 1

        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 1e-12:
            continue
        # move a random fraction (up to 15 %) from i to j
        eps = a[i] * rng.uniform(0.0, 0.15)
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # quick evaluation
        score_prop = _fast_objective(a_prop, fft_len)
        delta = score_prop - best_score

        if score_prop < best_score:
            best_score = score_prop
            best_a = a_prop.copy()
            a = a_prop
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-delta / max(T, 1e-12)):
                a = a_prop

        # linear cooling (every 1000 attempts)
        if it % 1000 == 0:
            fraction = (time.time() - start_time) / (deadline - start_time)
            T = T0 * (1.0 - fraction)
            if T < 1e-6:
                T = 1e-6

    return best_a, best_score


# ----------------------------------------------------------------------
# 4) Greedy deterministic peak reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Transfer a modest fraction from the coefficient that most
    contributes to the current convolution peak towards the smallest
    coefficient.  Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]                 # worst contributor

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]   # cheapest place

        eps = 0.25 * a[src]                               # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# 5) Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _powerlaw_start(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    w = idx ** (-p)
    return w / w.sum()


def _hann_window(n: int) -> np.ndarray:
    """Standard Hann (raised‑cosine) window, normalised."""
    x = np.arange(n)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    return w / w.sum()


def _blackman_window(n: int) -> np.ndarray:
    """Standard Blackman window, normalised."""
    x = np.arange(n)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    )
    return w / w.sum()


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of a diverse warm‑up, a long Adam phase,
    exponentiated‑gradient polishing, a final simulated‑annealing
    refinement and a greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4      # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (keep the same n as the previous best for fair comparison)
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # ------------------------------------------------------------------
    starts = []

    # 1) previous construction, if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear up‑sampling preserving the total sum
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            seq_sum = seq.sum()
            if seq_sum > 0:
                seq = seq / seq_sum
                starts.append(seq)
    except Exception:
        pass

    # 2) Uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) Exponential decays
    for r in [0.9999, 0.9995, 0.999, 0.995, 0.99, 0.985]:
        starts.append(_exponential_start(n, r))

    # 4) Power‑law families
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        starts.append(_powerlaw_start(n, p))

    # 5) Linear decreasing (triangular shape)
    tri = np.arange(n, 0, -1, dtype=float)
    starts.append(tri / tri.sum())

    # 6) Hann and Blackman windows
    starts.append(_hann_window(n))
    starts.append(_blackman_window(n))

    # 7) Dirichlet draws with a few fixed concentrations
    for conc in [0.3, 0.7, 1.5, 3.0]:
        starts.append(_dirichlet_start(n, conc, rng))

    # 8) Random Dirichlet draws (different concentrations)
    for _ in range(5):
        conc = 10 ** rng.uniform(-2, 2)          # log‑uniform between 0.01 and 100
        starts.append(_dirichlet_start(n, conc, rng))

    # ------------------------------------------------------------------
    # Evaluate all raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:02d} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – brief Adam warm‑up on each start (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.10
    for idx, a0 in enumerate(starts):
        if time.time() >= phase1_deadline:
            break
        sub_deadline = phase1_deadline
        cand_seq, cand_score = _adam_optimize(
            a0,
            sub_deadline,
            start_time,
            fft_len,
            iters_per_alpha=2500,
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam (start {idx+1}) "
                f"score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient polishing (≈70 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.80
    if phase2_deadline > deadline:
        phase2_deadline = deadline
    polished_seq, polished_score = _expgrad_optimize(
        best_seq,
        phase2_deadline,
        start_time,
        fft_len,
        eta0=0.6,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after ExpGrad score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _simulated_annealing(
            best_seq, deadline, start_time, fft_len
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after SA score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 4 – deterministic greedy peak‑reduction (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)          # remove possible negatives
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final reported score (for the log)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return as plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
