"""
Propose a step‑function (non‑negative list) with a low value of

    2·n·max(conv(a,a)) / (sum(a)**2)

The method follows three ideas:

1.  Use a fairly large number of points (n = 8192) – the optimum
    of the continuous problem is 1.5, and the discrete optimum approaches
    this limit as n grows.

2.  Create a diverse set of starter sequences (uniform, linearly
    decreasing, symmetric “tent” with several exponents and random
    decreasing vectors).  Keep the one with the smallest objective.

3.  Refine the best starter by projected sub‑gradient descent with
    momentum (the same idea as the reference code, but with a cheaper
    projection onto the simplex).  The loop runs until the time budget
    is exhausted, continuously remembering the best iterate.

All heavy work (convolution, gradient) is done with FFTs, so the
algorithm easily fits into the 1000 s budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (linear, not circular)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (exact same as evaluate_sequence, but faster)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) – assumes non‑negative seq."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Simple (O(n)) projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to ≥0 and renormalise to sum = 1 (cheap but sufficient)."""
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:                               # fallback – uniform vector
        w = np.full_like(w, 1.0 / w.size)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (uses the true nonsmooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a0 until `deadline`.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum parameters
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑convolution (see reference)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto the simplex (cheap O(n) version)
        a = _project_to_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence.
    Returns a plain Python list of non‑negative numbers.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a better asymptotic bound.
    # --------------------------------------------------------------
    n = 8192            # 2 × the previous best size; still cheap for FFT

    # --------------------------------------------------------------
    # 1) Build a variety of starter candidates.
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # linearly decreasing (large mass on the left side)
    dec = np.arange(n, 0, -1, dtype=float)
    candidates.append(dec / dec.sum())

    # linearly increasing (mirror of the above)
    inc = np.arange(1, n + 1, dtype=float)
    candidates.append(inc / inc.sum())

    # symmetric tent with several exponents
    x = np.arange(n, dtype=float) / (n - 1)       # points in [0,1]
    for p in [0.5, 0.75, 1.0, 1.2, 1.5, 2.0]:
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() > 0:
            candidates.append(w / w.sum())

    for _ in range(12):

        rnd.sort()                     # ascending → reverse to make decreasing
        rnd = rnd[::-1]
        candidates.append(rnd / rnd.sum())

    # optional: interpolate the global starter if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)

    # 4) Very light final annealing (tiny mass transfers) – optional.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        current_score = _objective(a)
        best_local = a.copy()
        best_local_score = current_score
        rng = np.random.default_rng()
        total_steps = 25000
        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break
            # simple linear cooling
            T = 0.02 * (1.0 - it / total_steps)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-16:
                continue
            eps = a[src] * rng.random() * 0.25
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _objective(a_prop)
            delta = new_score - current_score
            if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
                current_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final score={_objective(best_seq):.6f}")
    return [float(x) for x in best_seq]
