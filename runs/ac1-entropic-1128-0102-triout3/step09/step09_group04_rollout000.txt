    if not isinstance(sequence, list):

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # Determine n and initial a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        if a.ndim != 1:
            a = np.full(2048, 1/2048)
    else:
        n=2048
        a = np.full(2048, 1/2048)
    # Normalize sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full_like(a, 1.0 / a.size)
    n = a.shape[0]

    # Compute current best
    best_a = a.copy()
    current_score = 2 * n * _conv_self(a).max()
    best_score = current_score
    print(f"[{time.time()-start:.2f}s] initial score {best_score:.6f}")

    # hyperparameters
    step_scale = 0.5  # proportion of source value to move (initial)
    min_step = 1e-4
    decay = 0.9995  # decay per iteration
    temp = 0.001  # initial temperature for SA
    temp_decay = 0.9999

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # choose indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to move
        delta = a[i] * step_scale * np.random.rand()
        if delta <= 0:
            continue
        # create new candidate
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # compute score
        conv = _conv_self(cand)
        cand_score = 2 * n * conv.max()
        if cand_score < current_score:
            a = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time()-start:.2f}s] iter {iteration} improve → {best_score:.6f}")
        else:
            # simulated annealing acceptance
            # accept with probability exp(-(cand_score-current_score)/temp)
            if temp > 0:
                prob = math.exp(-(cand_score - current_score) / temp)
                if np.random.rand() < prob:
                    a = cand
                    current_score = cand_score
        # decay step scale and temperature
        step_scale = max(min_step, step_scale * decay)
        temp = max(1e-8, temp * temp_decay)

    # final normalization (sum to 1)
    best_a = np.maximum(best_a, 0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time()-start:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]

"""
stochastic hill‑climbing optimizer for the self‑convolution bound

The evaluation objective for a non‑negative vector a (sum a = 1) is
    score = 2·n·max(conv(a,a))

which is scale‑invariant.  Starting from the best known construction
(if the global ``height_sequence_1`` exists) we perform a long
simulated‑annealing / hill‑climbing run:

* at each iteration move a small amount of probability mass from a
  randomly chosen “source” index i to a randomly chosen “target” j;
* accept the move if it improves the objective, otherwise accept it
  with a temperature‑controlled probability;
* the step size and temperature decay exponentially, allowing coarse
  exploration early on and fine tuning later;
* the current best vector (and its score) is kept throughout the run;
* the whole search respects the supplied time budget (default 1000 s).

The routine returns a plain Python list that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·len(a)-1) using a real FFT.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# -------------------------------------------------------------
# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic hill‑climbing / simulated‑annealing optimizer for the
    self‑convolution L∞ bound.  Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin so we always finish before the hard limit
    deadline = start_time + budget_s - 0.1

    # -----------------------------------------------------------------
    # Initialise a starting vector.
    # Prefer the globally offered construction; otherwise fall back to uniform.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
            if a0.ndim != 1 or a0.size == 0:
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = 2048
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        # ensure non‑negative, normalise to sum 1, keep original length
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
        a = a0.astype(float)

    n = a.shape[0]

    # -----------------------------------------------------------------
    # Baseline score
    # -----------------------------------------------------------------
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Hyper‑parameters for the stochastic search
    # -----------------------------------------------------------------
    # initial proportion of source mass to move (as fraction of a[i])
    step_scale = 0.5
    min_step = 1e-4          # never shrink below this proportion
    step_decay = 0.9995      # geometric decay per iteration

    # simulated‑annealing temperature (absolute scale of score differences)
    temperature = cur_score * 0.001
    temp_decay = 0.9999

    # Counters for occasional progress reports
    report_interval = 2000
    last_report = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # ---------------------------------------------------------
        # pick two distinct indices, source i and target j
        # ---------------------------------------------------------
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1     # ensure j != i

        # ---------------------------------------------------------
        # amount of probability mass to shift from i to j
        # ---------------------------------------------------------
        # we move at most step_scale·a[i] (if a[i] is tiny we move less)
        delta = a[i] * step_scale * rng.random()
        if delta <= 0.0:
            continue

        # ---------------------------------------------------------
        # construct candidate vector
        # ---------------------------------------------------------
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # ---------------------------------------------------------
        # evaluate candidate
        # ---------------------------------------------------------
        cand_score = _score(cand)

        # ---------------------------------------------------------
        # acceptance test (hill‑climbing + annealing)
        # ---------------------------------------------------------
        accepted = False
        if cand_score < cur_score:
            accepted = True
        else:
            # accept worse move with probability exp(-(Δ)/T)
            if temperature > 0.0:
                prob = math.exp(-(cand_score - cur_score) / temperature)
                if rng.random() < prob:
                    accepted = True

        if accepted:
            a = cand
            cur_score = cand_score
            # keep track of the best ever seen
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] iter {it:<7d} improve → {best_score:.6f}")

        # ---------------------------------------------------------
        # decay step size and temperature
        # ---------------------------------------------------------
        step_scale = max(min_step, step_scale * step_decay)
        temperature = max(1e-12, temperature * temp_decay)

        # occasional lightweight progress report (to avoid flooding output)
        if it - last_report >= report_interval:
            print(f"[{time.time() - start_time:6.2f}s] iter {it:<7d} cur {cur_score:.6f} best {best_score:.6f}")
            last_report = it

    # -----------------------------------------------------------------
    # Final normalisation (guard against numerical drift)
    # -----------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
