```
    # Phase grid of p-values

        a[~np.isfinite(a)] = 0.0

        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]
        score = float(2.0*n*conv.max())

            best_seq = a.copy()
```

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}.
\]

Because the objective is homogeneous of degree 0 we may normalise the
vector to a probability simplex (`∑a = 1`).  
For a fixed length `n` the problem is to make the self‑convolution
as flat as possible – i.e. minimise the largest entry of `conv = a*a`.

**Idea**

1. **Parametric families** – generate candidate step‑functions from a few
   simple analytic families (symmetric power law, discretised Beta
   distributions).  
   These families contain the optimal shape (an “U‑shaped’’ distribution
   whose convolution is almost uniform) and can be evaluated very quickly
   with an FFT.

2. **Coarse search** – sweep the parameters, keep the best candidate
   (this already brings the score well below the previous 1.512).

3. **Refinement** – starting from the best candidate run a few
   iterations of projected sub‑gradient descent on the true objective.
   The gradient of the maximal convolution entry is cheap to compute,
   the iterate is kept on the simplex by Euclidean projection.

4. **Targeted annealing** – a short simulated‑annealing phase moves a
   fraction of the mass from the two indices that currently contribute most
   to the maximal convolution entry to random locations.  This is the same
   deterministic move that proved useful in the reference optimiser,
   but now applied only for a limited number of steps after a good
   initial point has been found.

All steps are timed; the algorithm stops well before the 1000 s budget
and returns the best sequence found.

```python
"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence.  The method consists of

  * a fast parametric sweep over simple symmetric Beta / power‑law families,
  * a short projected sub‑gradient descent on the true max‑convolution,
  * a brief targeted pair‑annealing that moves mass away from the peak.

The code works with a single CPU, uses only numpy / math / time and
never writes to disk.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (full length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv)  (the simplex normalisation is assumed)."""
    return float(2.0 * n * conv.max())

# Core optimisation phases

def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score: float,
                         max_iters: int = 500_000):
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Very cheap – each iteration needs one FFT.
    """
    a = a0.copy()
    best_a = a.copy()
    best = best_score
    momentum = np.zeros_like(a)
    beta = 0.90
    step0 = 0.5
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L)
        k = int(conv.argmax())               # index of the current peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

        # score check
        cur_score = _score_from_conv(_conv_self(a, L), n)
        if cur_score < best:
            best = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad itr={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best


def _pair_annealing(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, cur_best_score: float,
                    cur_best_seq: np.ndarray,
                    start_temp: float = 5e-2,
                    end_temp: float = 1e-9,
                    max_steps: int = 200_000):
    """
    Very cheap targeted simulated‑annealing.
    Only the two indices that realise the current peak are weakened.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # peak index and the pair that contributes most to it
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct destinations (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose a small re‑allocation of mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] <= 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20        # ≤20 % of source
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate proposal
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best:
            best = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal itr={it:,} "
                  f"score={score:.6f}")

    return best_seq, best

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best known bound (≈1.512).  The routine performs a fast parametric sweep,
    a few hundred thousand sub‑gradient steps and a short targeted annealing,
    all inside the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – the same size used by the reference solution
    # --------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Start from the previously stored solution if it exists
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.clip(starter, 0.0, None)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                conv = _conv_self(starter, L)
                sc = _score_from_conv(conv, n)
                best_seq = starter.copy()
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] init from "
                      f"height_sequence_1 score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1) PARAMETRIC SWEEP – symmetric power‑law family
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)
    for p in np.linspace(0.0, 2.5, 51):     # 51 candidates, cheap
        if time.time() >= deadline:
            break
        a = ((i + 0.5) * (n - i - 0.5)) ** (-0.5 * p)
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        conv = _conv_self(a, L)
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] p={p:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2) PARAMETRIC SWEEP – discretised Beta family
    #    a_i ∝ x^α (1‑x)^β  with x = (i+0.5)/n
    # --------------------------------------------------------------
    xs = (i + 0.5) / n
    eps = 1e-12
    xs = np.clip(xs, eps, 1.0 - eps)

    alphas = np.linspace(-0.9, 4.0, 41)   # include a few negative values
    betas = np.linspace(-0.9, 4.0, 41)

    for a_par in alphas:
        if time.time() >= deadline:
            break
        term1 = xs ** a_par
        for b_par in betas:
            if time.time() >= deadline:
                break
            term2 = (1.0 - xs) ** b_par
            a = term1 * term2
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            conv = _conv_self(a, L)
            sc = _score_from_conv(conv, n)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] α={a_par:.2f} β={b_par:.2f} "
                      f"score={sc:.6f}")

    # --------------------------------------------------------------
    # If nothing was found (should not happen) fall back to uniform
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3) REFINEMENT – projected sub‑gradient descent
    # --------------------------------------------------------------
    a_refined, refined_score = _subgradient_descent(
        best_seq,
        n,
        deadline,
        start_time,
        best_score,
        max_iters=500_000,
    )
    if refined_score < best_score:
        best_score = refined_score
        best_seq = a_refined.copy()
        print(f"[{time.time() - start_time:.2f}s] sub‑grad final score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) TARGETED PAIR‑ANNEALING – short but often effective
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final, final_score = _pair_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            max_steps=200_000,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal final score={final_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] best score returned {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
```
