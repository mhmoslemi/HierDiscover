def _random_local_search(a0, deadline, start_time, fft_len):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, fft_len)
    n = len(a)
    # parameters
    max_step_frac = 0.2  # fraction of source value
    T0 = best_score * 0.01  # starting temperature
    T_final = 1e-6
    total_iters = 0
    while time.time() < deadline:
        # compute temperature based on progress
        # We can schedule T = max(T_final, T0 * (deadline - time.time()) / (deadline - start_time))
        # simpler linear decay
        remaining = deadline - time.time()
        total = deadline - start_time
        T = T0 * (remaining / total)
        i, j = rng.choice(n, size=2, replace=False)
        # ensure we have enough mass to move
        step = min(a[i], max_step_frac * a[i])
        if step <= 0:
            continue
        if a[j] + step > 1000:
            step = max(0, 1000 - a[j])
            if step <= 0:
                continue
        a_new = a.copy()
        a_new[i] -= step
        a_new[j] += step
        # compute score
        cur_score = _score(a_new, fft_len)
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / (T + 1e-12)):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] random local improve: score={cur_score:.6f}")
    return best_a, best_score

"""
Candidate generator for the “step‑function” challenge.

Strategy
--------
1. **Diverse seeds** – uniform, several exponential decays, a linear
   ramp, a quadratic ramp, power‑law and Dirichlet draws.  The best
   seed (including the previously‑known `height_sequence_1`) is kept
   as the starting point.

2. **Adam warm‑up** – few Adam steps on a smooth (log‑sum‑exp) surrogate
   of the max‑convolution.  This quickly leaves the poorest regions.

3. **Top‑K exponentiated‑gradient polishing** – a mirror‑descent step that
   uses the gradients of the *K* largest convolution entries (instead of
   only the absolute maximum).  This yields a richer descent direction
   and typically squeezes a few percents out of the objective.

4. **Random local search (simulated annealing)** – after the gradient
   phase we perturb the vector by moving a small fraction of mass
   between two random positions.  Moves that improve the exact score are
   always accepted; worsening moves are accepted with a temperature‑
   dependent probability, allowing us to escape shallow local minima.

5. **Greedy peak reduction** – a deterministic fine‑tuner identical to
   the one used previously, applied for the last sliver of time.

All phases respect the supplied time budget (default 1000 s).  The
returned sequence is clipped to the admissible interval `[0,1000]`,
renormalised to a probability vector and finally converted to a plain
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (full linear self‑convolution via FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (avoids the heavy security checks of
# evaluate_sequence – used while searching)
# ----------------------------------------------------------------------


def _exact_score(a: np.ndarray, fft_len: int) -> float:
    """Exact value of evaluate_sequence (without clipping / checks)."""
    n = a.shape[0]
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of max(convolution)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam steps on the smoothed objective.
    Returns the best vector seen (probability) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))          # work in log‑space

    # smoothing schedule – start smooth, finish sharp
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 1500                     # a few thousand total

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = _exact_score(a, fft_len)

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smoothed surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact check
            if t % 250 == 0:
                sc = _exact_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing using the *K* largest shifts
# ----------------------------------------------------------------------


def _expgrad_topk_optimize(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           fft_len: int,
                           eta0: float = 0.5,
                           K: int = 3) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) that uses the gradients of the
    K most‑violating convolution entries.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGradTopK improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ---- top‑K gradients -------------------------------------------------
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]          # unsorted K largest

        g = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]

        # average over the K active shifts
        g = g / float(len(top_idx))

        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact check (cheap)
        if t % 500 == 0:
            sc = _exact_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _exact_score(a, fft_len)

    total_time = deadline - start_time
    max_frac = 0.20        # start with up‑to‑20 % moves
    min_frac = 0.0005      # end with very tiny moves

    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear cooling of the move size
        frac = min_frac + (max_frac - min_frac) * (remaining / total_time)

        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue

        # limit step so we never exceed the hard 1000‑cap
        step = min(a[i] * frac, 1000.0 - a[j])
        if step <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= step
        a_new[j] += step

        cur_score = _exact_score(a_new, fft_len)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
            print(
                f"[{time.time() - start_time:6.2f}s] Random local improve: score={cur_score:.6f}"
            )
        else:
            # simple annealing acceptance
            temperature = frac * 0.1
            if np.random.rand() < math.exp(-(cur_score - best_score) / (temperature + 1e-12)):
                a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak reduction (identical to the previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Move a modest fraction of the coefficient that most contributes
    to the current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]                     # modest move
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

# Random start generators (additional families)
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _linear_decreasing_start(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (strictly decreasing linear ramp)."""
    vec = np.arange(n, 0, -1, dtype=float)
    return vec / vec.sum()


def _quadratic_decreasing_start(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)²."""
    vec = np.arange(n, 0, -1, dtype=float) ** 2
    return vec / vec.sum()


def _powerlaw_start(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ (i+1)^‑p (heavier tail for smaller p)."""
    idx = np.arange(n, dtype=float) + 1.0
    vec = idx ** (-p)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that (almost) minimises
    `evaluate_sequence`.  The routine respects the allotted time budget
    and returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # --------------------------------------------------------------
    # Problem size (kept from the previous best)
    # --------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    starts = []

    # 1) previous construction (if present)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # handcrafted families
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))    # gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_gaussian_start(n))                 # bell curve
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))
    starts.append(_linear_decreasing_start(n))
    starts.append(_quadratic_decreasing_start(n))
    starts.append(_powerlaw_start(n, p=0.5))
    starts.append(_powerlaw_start(n, p=1.0))
    starts.append(_powerlaw_start(n, p=2.0))

    # evaluate raw starts and keep the best
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _exact_score(a0, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 1 – short Adam warm‑up (≈10 % of time)
    # --------------------------------------------------------------
    adam_deadline = start_time + budget_s * 0.1
    cand_seq, cand_score = _adam_optimize(
        best_seq, adam_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – top‑K exponentiated‑gradient polishing (≈70 % of time)
    # --------------------------------------------------------------
    expgrad_deadline = start_time + budget_s * 0.8
    polished_seq, polished_score = _expgrad_topk_optimize(
        best_seq, expgrad_deadline, start_time, fft_len, eta0=0.75, K=5
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Top‑K polishing score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 3 – stochastic local search (≈10 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        random_deadline = start_time + budget_s * 0.94
        final_seq, final_score = _random_local_search(
            best_seq, random_deadline, start_time, fft_len
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after random search score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 4 – deterministic greedy refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Final safety checks – clip, renormalise, report
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)   # ensure non‑negative
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = _exact_score(best_seq, fft_len)
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return plain Python list as required by the evaluation harness
    return [float(x) for x in best_seq.tolist()]
