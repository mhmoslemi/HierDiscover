    starts.append(_gaussian_start(n))

    # random Dirichlet with random conc
    for _ in range(5):
        conc = 10**rng.uniform(-1, 1)   # between 0.1 and 10
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))
    # random normalized Gaussian shape with random sigma
    for _ in range(3):
        sigma = rng.uniform(n/12, n/3)   # vary
        xs = np.arange(n)
        g = np.exp(-0.5*((xs-(n-1)/2)/sigma)**2)
        starts.append(g/g.sum())
    # evaluate
    best_seq = None; best_score = float('inf')
    for i,a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc<best_score:
            best_score=sc; best_seq=a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] init #{i+1} score={sc:.6f}")
    # Phase 1: short Adam
    explore_deadline = start_time + budget_s*0.07  # 7%
    cand_seq, cand_score = _adam_optimize(best_seq, explore_deadline, start_time, fft_len)
    if cand_score < best_score:
        best_score = cand_score; best_seq = cand_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] New best after Adam {best_score:.6f}")

    # Phase 2: expgrad
    eg_deadline = start_time + budget_s*0.8  # 80%
    polished_seq, polished_score = _expgrad_optimize(best_seq, eg_deadline, start_time, fft_len, eta0=0.9)
    if polished_score < best_score:
        best_score = polished_score; best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] New best after expgrad {best_score:.6f}")

    # Phase 3: stochastic fine tuning
    if time.time() < deadline:
        fine_seq, fine_score = _stochastic_fine_tune(best_seq, deadline, start_time, rng)
        if fine_score < best_score:
            best_score = fine_score; best_seq = fine_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best after fine-tune {best_score:.6f}")

    # final clip & normalise
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0/n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]

"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------

1. **Diverse initial pool**
   * Uniform, exponential, Gaussian, Dirichlet and a few *power‑law*
     families (a_i ∝ (i+1)^‑p) with several exponents.
   * The previous best sequence (``height_sequence_1``) is also used
     if it is available.

2. **Short Adam warm‑up**
   * A few Adam iterations on a *soft‑max* surrogate of the
     max‑convolution (log‑sum‑exp).  The surrogate becomes tighter
     as the schedule progresses, giving a quick escape from clearly
     bad regions.

3. **Long exponentiated‑gradient polishing**
   * Mirror‑descent (multiplicative updates) on the *true* objective.
     Only the shift that currently attains the maximum contributes
     to the sub‑gradient, which keeps each iteration cheap.
   * A relatively large initial step‑size (``eta0``) is used and then
     decays like 1/√t.

4. **Stochastic fine‑tuning**
   * A simple hill‑climbing / simulated‑annealing phase.
     At each iteration we move a small fraction of mass from a
     coefficient that contributes most to the active peak towards a
     coefficient that is currently smallest.  The move is kept only
     when it strictly reduces the maximal convolution value.
   * The phase runs for the remaining wall‑clock budget.

All three phases respect the supplied time budget (default 1000 s).
The final vector is clipped to ``[0,∞)`` and renormalised to sum to
one before being returned.
"""

# ----------------------------------------------------------------------
# Imports and low‑level helpers
# ----------------------------------------------------------------------
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Soft‑max surrogate of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------
def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of max(conv). Returns value and conv."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv

# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    # work in log‑space (softmax logits)
    y = np.log(np.maximum(a, eps_log))

    # schedule: increasingly tight smoothness
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 3000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # current soft‑max vector (valid probability)
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}")

    return best_a, best_score

# Exponentiated‑gradient optimisation on the true objective
# ----------------------------------------------------------------------
def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient from the active shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # multiplicative update & re‑normalisation
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact check
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

# Stochastic fine‑tuning (hill‑climbing with small mass transfers)
# ----------------------------------------------------------------------
def _stochastic_fine_tune(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    conv = _conv_self(a, fft_len)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # recompute convolution (required after every accepted move)
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]

        # pick a source among the biggest contributors (top‑5)
        top_k = min(5, len(i_idx))
        src_candidates = i_idx[np.argpartition(contrib, -top_k)[-top_k:]]
        src = rng.choice(src_candidates)

        # pick a destination among the smallest coefficients (bottom‑5)
        low_k = min(5, n)
        dst_candidates = np.argpartition(a, low_k)[:low_k]
        dst = rng.choice(dst_candidates)

        if src == dst or a[src] <= 0.0:
            continue

        # move a modest random fraction of the source mass
        eps = a[src] * rng.uniform(0.005, 0.03)   # 0.5 % – 3 % of src
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # evaluate new max‑convolution
        conv_new = _conv_self(a_new, fft_len)
        new_max = conv_new.max()
        if new_max < cur_max:               # accept only strict improvements
            a = a_new
            cur_max = new_max
            cur_score = 2.0 * n * new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] fine‑tune improve: iter={iteration:,}, score={cur_score:.6f}")

    return best_a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()

def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()

def _powerlaw_start(n: int, exponent: float, offset: float = 0.0) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0 + offset
    a = i ** (-exponent)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
