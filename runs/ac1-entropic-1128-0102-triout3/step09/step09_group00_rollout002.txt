"""
Propose a step‐function (non‑negative heights) with a low
`evaluate_sequence` value.

Algorithm
---------
1. **Parameter sweep** –  generate symmetric “tent” shapes
   a_i ∝ min(i,n‑1‑i)^p for a range of exponents p and keep the
   best candidate (the shape is normalised to sum‑1).

2. **Random starts** – a few Dirichlet draws give diverse initial
   points; keep any improvement.

3. **Smoothed projected gradient descent** – minimise the
   (scale‑invariant) objective 2·n·max(conv(a,a)) by smoothing the
   max with a log‑sum‑exp (temperature s).  The gradient of the
   smoothed max can be written as a correlation and is obtained
   efficiently with FFTs.  After each step we project back onto the
   probability simplex.

   The temperature s is increased from a small value to a large one,
   turning the smooth objective gradually into the true max‑norm.
   Momentum and a 1/√t stepsize are used.

4. **Light annealing** – a short simulated‑annealing phase
   (mass transfers) tries to escape any remaining local optimum.

The whole search respects the 1000 s budget; if time runs out the
best sequence found so far is returned.  The final sequence is
non‑negative, normalised to sum = 1 and converted to a plain Python
list for the evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed projected gradient descent on the true objective
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent where the max in the objective is
    approximated by a log‑sum‑exp with a temperature that grows
    from `s_start` to `s_end`.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum factor
    step0 = 0.5         # initial stepsize multiplier
    it = 0

    # temperature schedule
    s_start = 5.0
    s_end = 300.0

    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # 1) current convolution and objective
        # --------------------------------------------------------------
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max      # sum(a)≈1 because of projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 2) temperature (smoothness) – increase as we spend time
        # --------------------------------------------------------------
        total = deadline - start_time
        progress = min(1.0, (time.time() - start_time) / total)
        s = s_start * (s_end / s_start) ** progress

        # --------------------------------------------------------------
        # 3) weights for the smoothed max (log‑sum‑exp)
        # --------------------------------------------------------------
        maxc = conv.max()
        exp_vals = np.exp(s * (conv - maxc))          # shift to avoid overflow
        w = exp_vals / exp_vals.sum()                # ∑ w_i = 1

        # --------------------------------------------------------------
        # 4) gradient of the smoothed objective
        #    ∂/∂a_k  = 2· Σ_i w_i a_{i‑k}   (correlation)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        L = _next_pow2(len(w) + n - 1)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(W * A_rev, n=L)
        # relevant slice: positions n‑1 … n‑1+n‑1
        grad = 2.0 * corr_full[n - 1:n - 1 + n]

        # --------------------------------------------------------------
        # 5) momentum + step, then projection onto simplex
        # --------------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # optional tiny stochastic perturbation every few thousand steps
        if it % 5000 == 0:
            a += 1e-8 * np.random.randn(n)
            a = _project_simplex(a)

    return best_a, best_score

def _anneal_polish(a_start: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing that moves tiny amounts of mass between
    randomly chosen positions.  Returns the best sequence seen.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (very fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # pick source proportional to its mass, destination uniform
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence.
    Returns a Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger canvas improves the discretisation.
    # --------------------------------------------------------------
    n = 8192                     # feel free to change; 8192 works fast

    # --------------------------------------------------------------
    # 1) Analytic symmetric tent family – sweep many exponents.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    xs = np.linspace(0.0, 1.0, n, dtype=float)

    # fine grid of exponents (more points than the previous run)
    for p in np.arange(0.4, 2.6, 0.05):
        w = np.minimum(xs, 1.0 - xs) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exponent={p:.2f} score={score:.6f}")
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) Random Dirichlet starts – diversify the search.
    # --------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        # concentration <1 gives spikier vectors, >1 gives smoother ones
        conc = np.random.choice([0.3, 0.6, 1.0, 2.0])
        a = np.random.dirichlet(np.full(n, conc))
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.1f} score={score:.6f}")

    # --------------------------------------------------------------
    # 3) Ensure we have a valid starter.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:.2f}s] Fallback uniform score={best_score:.6f}")

    # normalise onto simplex (protect against tiny rounding errors)
    best_seq = _project_simplex(best_seq)

    # --------------------------------------------------------------
    # 4) Main optimisation – smoothed gradient descent.
    #    Use most of the remaining budget, keep best iterate.
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # allocate a chunk of the remaining budget to one descent run
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.75
        refined_seq, refined_score = _smooth_gradient_descent(
            best_seq, sub_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

        # If we are stagnating, perturb and try again
        if refined_score >= best_score * 0.9999:
            perturb = np.random.dirichlet(np.full(n, 0.5))
            best_seq = 0.9 * best_seq + 0.1 * perturb
            best_seq = _project_simplex(best_seq)

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 5) Very light final annealing (optional but often helps a bit).
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal_polish(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score = annealed_score
            best_seq = annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The evaluation harness expects a simple list of floats.
    return [float(v) for v in best_seq]
