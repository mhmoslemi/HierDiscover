def _simulated_annealing_improve(a0, conv0, deadline, start_time, init_temp=1e-3, min_temp=1e-8, cooling=0.9995, max_iters=500_000):

Pseudo:

a = a0.copy()
conv = conv0.copy()
n = a.shape[0]
cur_max = conv.max()
best_score = 2 * n * cur_max
best_a = a.copy()
best_max = cur_max

temp = init_temp
iters = 0

while time.time() < deadline and iters < max_iters and temp > min_temp:
    iters += 1
    # choose src: compute contributions to current peak positions
    k_max = conv.argmax()
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    i_idx = np.arange(lo, hi + 1, dtype=int)
    j_idx = k_max - i_idx
    contrib = a[i_idx] * a[j_idx]
    src = i_idx[contrib.argmax()] # index with max contribution

    # choose dst: index with smallest a (tie broken random)
    # maybe also choose random index from low values.
    dst_candidates = np.delete(np.arange(n), src)
    # small probability to choose random; majority choose min
    if rng.random() < 0.2:
        dst = rng.choice(dst_candidates)
    else:
        dst = dst_candidates[a[dst_candidates].argmin()]

    # epsilon: dynamic fraction of src; degrade over time
    # we set epsilon = min(0.5 * a[src], 0.01) * (temp / init_temp) ???.
    # We'll adapt: eps = min(a[src] * 0.5, 0.001) maybe.
    eps = min(a[src] * 0.5, 1e-3)
    if eps <= 1e-12:
        continue

    # propose update
    # compute new conv via copy and incremental update low cost
    conv_tmp = conv.copy()
    # linear term
    conv_tmp[src:src + n] -= 2.0 * eps * a
    conv_tmp[dst:dst + n] += 2.0 * eps * a
    # quadratic terms
    idx_src_sq = 2 * src
    idx_dst_sq = 2 * dst
    conv_tmp[idx_src_sq] += eps * eps
    conv_tmp[idx_dst_sq] += eps * eps
    conv_tmp[src + dst] -= 2.0 * eps * eps

    new_max = conv_tmp.max()
    delta = new_max - cur_max

    accept = False
    if delta < -1e-12:
        accept = True
    else:
        prob = math.exp(-delta / (temp + 1e-12))
        if rng.random() < prob:
            accept = True

    if accept:
        # apply changes
        a[src] -= eps
        a[dst] += eps
        conv = conv_tmp
        cur_max = new_max
        # update best
        if 2 * n * cur_max < best_score:
            best_score = 2 * n * cur_max
            best_a = a.copy()
            best_max = cur_max
            print(f"[{time.time() - start_time:6.2f}s] SA improve: iter={iters:,}, score={best_score:.6f}")

    # temperature schedule
    temp *= cooling

Return best_a, best_score.

"""
search for a non‑negative step function that minimises ``evaluate_sequence``.

Approach
--------
1. **Initialisation** – use the historic best (``height_sequence_1``) if
   available, otherwise a uniform distribution.  The vector is normalised
   to have L1‑norm 1 (the objective is scale‑invariant).

2. **Projected Adam on a smooth max** – minimise a smooth approximation of
   the L∞‑norm of the self‑convolution while staying on the probability
   simplex.  The smoothing parameter ``α`` is increased geometrically during
   the run (0.5 → 2 → 8 → … → 8192) so that the optimisation gradually
   focuses on the true maximum.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method
   (with momentum) that works directly on the true objective
   ``max_k (a*a)[k]``.  This refines the Adam solution.

4. **Simulated‑annealing style local moves** – repeatedly move a small
   amount of mass from the index that contributes most to the current
   convolution peak to a low‑value index.  Moves that worsen the peak are
   occasionally accepted according to a temperature schedule, allowing the
   search to escape shallow local minima.

5. **Final normalisation and evaluation** – clip to non‑negative,
   re‑normalise, and return the best sequence found.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed‑max function

        fα(a) = (1/α)·log Σ_k exp(α·conv_k)

    where conv = a*a.  Returns ∂fα/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre of the correlation corresponds to indices 0..n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# -------------------------------------------------------------
# Phase 1 – Adam optimiser on the smoothed maximum
# -------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max.  The smoothing parameter α
    starts at 0.5 and is multiplied by 2 every ``steps_per_alpha`` steps
    (up to a maximum of 8192).  The best exact score seen so far is
    returned together with the corresponding vector.
    """
    a = a0.copy()
    n = a.shape[0]

    fft_len = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    α = 0.5
    α_max = 8192.0
    steps_per_alpha = 2000   # after this many Adam steps we double α
    steps_at_current_alpha = 0

    while time.time() < deadline:
        # ---- compute convolution and smooth‑max ----
        conv = _conv_self(a)
        scaled = α * conv
        max_scaled = scaled.max()
        lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
        f_smooth = lse / α
        obj_smooth = 2.0 * n * f_smooth

        # gradient of the *smooth* objective (includes outer factor 2·n)
        grad = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_t = lr0 / np.sqrt(t)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a_new)

        # occasional exact evaluation
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f}) "
                      f"score={sc:.6f}")

        # increase α periodically
        steps_at_current_alpha += 1
        if steps_at_current_alpha >= steps_per_alpha and α < α_max:
            α = min(α * 2.0, α_max)
            steps_at_current_alpha = 0
            print(f"[{time.time() - start_time:6.2f}s] ↑ increasing α to {α:.0f}")

    return best_a, best_score


# -------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent (exact max)
# -------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Uses momentum and a diminishing step size.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # ---- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# -------------------------------------------------------------
# Phase 3 – Simulated‑annealing style local improvement
# -------------------------------------------------------------


def _simulated_annealing_improve(a0: np.ndarray,
                                 conv0: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 rng: np.random.Generator,
                                 init_temp: float = 1e-3,
                                 cooling: float = 0.9997,
                                 min_temp: float = 1e-9,
                                 max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Local moves: transfer a small fraction of mass from the index that
    contributes most to the current convolution peak to a low‑value index.
    Moves that increase the peak are sometimes accepted according to a
    temperature schedule (simulated annealing).
    """
    a = a0.copy()
    conv = conv0.copy()
    n = a.shape[0]
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    temp = init_temp
    it = 0

    while time.time() < deadline and it < max_iters and temp > min_temp:
        it += 1

        # peak location
        k_max = conv.argmax()

        # indices that take part in the peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index most responsible

        # choose destination – usually the smallest coefficient,
        # occasionally a random one to increase exploration
        dst_candidates = np.delete(np.arange(n), src)
        if rng.random() < 0.15:
            dst = rng.choice(dst_candidates)
        else:
            dst = dst_candidates[a[dst_candidates].argmin()]

        # amount to move – 5 % of the source value, bounded away from zero
        eps = 0.05 * a[src]
        if eps < 1e-14:
            continue

        # incremental update of the convolution
        conv_tmp = conv.copy()
        # linear part (source loses, destination gains)
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        # quadratic self‑terms
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        # cross term between src and dst
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        delta = new_max - cur_max

        # acceptance rule
        accept = False
        if delta < -1e-15:
            accept = True
        else:
            prob = math.exp(-delta / (temp + 1e-18))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_tmp
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: iter={it:,}, score={best_score:.6f}")

        # temperature schedule
        temp *= cooling

    return best_a, best_score

def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a normalised (L1 = 1) non‑negative vector.
    Preference is given to the historic best ``height_sequence_1``.
    """
    # historic best
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation preserving the total mass
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass

    # fallback: uniform distribution
    return np.full(n, 1.0 / n)
