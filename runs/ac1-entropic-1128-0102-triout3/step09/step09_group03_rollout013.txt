"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm Overview
-----------------
* **Multiple handcrafted starts** – uniform, exponential, Gaussian, Dirichlet and the
  previously known best sequence (if any) are normalised to a probability vector.
* **p‑norm Adam warm‑up** – the smooth objective
  ``f_p(a) = ( Σ_k (a*a)[k]^p )^{1/p}`` (p = 2,4,…,1024) is minimised with Adam
  on the logits ``log a``.  The gradient of ``f_p`` is obtained by a single FFT‑based
  convolution, i.e. the whole step costs only a few FFTs per iteration.
* **Exponentiated‑gradient polishing** – a mirror‑descent step on the *true*
  max‑convolution objective.  Instead of using the gradient of a single
  active shift we take the *average* over all shifts whose value lies within a
  tiny tolerance of the current maximum.  The average gradient is again computed
  with one FFT.
* **Simulated‑annealing fine‑tuning** – to escape possible local minima we perform
  a temperature‑controlled random mass transfer between two coordinates,
  accepting deteriorations with a probability that decays over time.
* **Greedy peak reduction** – a deterministic local move that moves a fraction
  of the coefficient that contributes most to the current peak towards the
  smallest coefficient (strict improvements only).

The whole routine respects the supplied ``budget_s`` (default 1000 s).  It
prints intermediate scores and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_general(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc`` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full(n, 1.0 / n)


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    return up / s if s > 0 else np.full_like(up, 1.0 / new_n)


# ----------------------------------------------------------------------
# p‑norm Adam warm‑up (smooth approximation of max)
# ----------------------------------------------------------------------


def _adam_pnorm_optimize(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         fft_len_self: int,
                         eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth p‑norm objective
    f_p(a) = ( Σ (a*a)[k]^p )^{1/p}
    with a schedule of increasing ``p``.
    """
    a = a0.copy()
    n = a.shape[0]
    # work in log‑space (logits)
    y = np.log(np.maximum(a, eps_log))

    # p‑schedule (doubling)
    p_sched = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    iters_per_p = 2000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.07
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for p in p_sched:
        for _ in range(iters_per_p):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            conv = _conv_self(a, fft_len_self)                # a*a
            conv_p = conv ** p
            S = conv_p.sum()
            if S == 0.0:
                smooth_val = 0.0
                w_scaled = np.zeros_like(conv)
            else:
                smooth_val = S ** (1.0 / p)
                # weights w_k = conv[k]^{p‑1} / S^{(p‑1)/p}
                factor = S ** (-(p - 1.0) / p)
                w_scaled = (conv ** (p - 1.0)) * factor

            # gradient of the smooth objective:
            # g = 2 * Σ_k w_k * a_{k-i}
            # implement via one FFT (cross‑correlation)
            rev_w = w_scaled[::-1]                     # length 2n‑1
            fft_len_grad = _next_pow2(n + rev_w.shape[0] - 1)
            Af = np.fft.rfft(a, n=fft_len_grad)
            Wf = np.fft.rfft(rev_w, n=fft_len_grad)
            corr = np.fft.irfft(Af * Wf, n=fft_len_grad)
            # g_i = 2 * corr[len(w)-1 - i]
            g = 2.0 * corr[rev_w.shape[0] - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (g - np.dot(g, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam(p={p}) improve: score={sc:.6f}"
                    )

    # final projection onto simplex (just in case)
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full_like(a, 1.0 / n)
    return a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing with *all* active shifts
# ----------------------------------------------------------------------


def _expgrad_active_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len_self: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    The gradient is the average over all shifts whose value is within
    ``eps`` of the current maximum.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    eps = 1e-12

    # pre‑compute FFT length for the gradient convolution (a * rev(w))
    fft_len_grad = _next_pow2(3 * n)          # ≥ n + (2n‑1) – 1

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len_self)          # a*a, length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad active improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ---- gradient from *all* active shifts ----
        active_idx = np.where(conv >= cur_max - eps)[0]
        if active_idx.size == 0:                 # should never happen
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active_idx] = 1.0 / active_idx.size     # uniform average

        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        rev_w = w[::-1]
        Af = np.fft.rfft(a, n=fft_len_grad)
        Wf = np.fft.rfft(rev_w, n=fft_len_grad)
        corr = np.fft.irfft(Af * Wf, n=fft_len_grad)
        g = 2.0 * corr[w.shape[0] - 1 - np.arange(n)]

        # multiplicative (exponentiated) update
        eta = 0.5 / math.sqrt(t)                     # diminishing step size
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         n: int,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves accepted with a temperature schedule.
    Used as a final local‑search phase to escape shallow minima.
    """
    a = a0.copy()
    cur_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (exponential decay)
    T_start = 0.015
    T_final = 5e-5
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (T_final / T_start) ** frac   # decays towards T_final

        # propose a move: transfer a small fraction from a larger entry to a smaller one
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.choice(np.delete(np.arange(n), src))

        # fraction of the source mass to move (up to 20 % of src)
        max_delta = a[src] * 0.20
        delta = rng.random() * max_delta
        if delta == 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        sc = evaluate_sequence(a_prop.tolist())

        # Metropolis acceptance
        if sc < cur_score or rng.random() < math.exp(-(sc - cur_score) / max(T, 1e-12)):
            a = a_prop
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] SA improve: iter={it:,}, score={sc:.6f}"
                )

    return best_a, best_score

def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Move a fraction of the coefficient that most contributes to the peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # modest transfer
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score
