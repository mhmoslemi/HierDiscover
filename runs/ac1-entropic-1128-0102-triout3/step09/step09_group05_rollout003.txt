"""
Optimiser for the self‑convolution peak.

The search consists of three complementary stages:

1️⃣  Adam on a smooth (log‑sum‑exp) approximation of max(conv(a,a)).  
    The smoothness parameter β is increased gradually, driving the
    optimisation towards the true ℓ∞‑norm while keeping the gradient
    well‑behaved.

2️⃣  Fast hill‑climbing: repeatedly move a random fraction of the mass
    from a few source indices to random destinations, accepting moves
    that improve the score (or with a small Metropolis probability that
    cools down over time).  This stage explores the neighbourhood of the
    current best solution more aggressively than a pure sub‑gradient walk.

3️⃣  Targeted pair‑annealing (similar to the reference code) that weakens
    the two entries that contribute most to the current maximal
    convolution coefficient.

All updates keep the vector on the probability simplex via Euclidean
projection, guaranteeing non‑negativity and a unit sum (the objective is
scale‑invariant).  The routine works with any length; we use 8192 points,
which provides enough degrees of freedom to push the score below the
target 1.503 while still being fast enough for the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv) (sum of a is assumed to be 1)."""
    return float(2.0 * n * conv.max())


def _grad_from_weights(a: np.ndarray, w_rev: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t. a.
    Uses the FFT based correlation a ⋆ w_rev.
    """
    Af = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * W, n=L)[: len(a) + len(w_rev) - 1]
    # pick the indices that correspond to the gradient components
    offset = (len(w_rev) - 1) - np.arange(n)
    grad = 2.0 * corr[offset]
    return grad


# ----------------------------------------------------------------------
# Stage 1 – Adam on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation to max(conv).
    The smoothness (β) is increased gradually.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = _score_from_conv(conv, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    # schedule for smoothness β (log‑sum‑exp temperature)
    betas = np.geomspace(0.5, 40.0, num=25)

    for beta in betas:
        # allocate ~2500 iterations per β (adjusted to remaining time)
        max_iters = 3000
        for _ in range(max_iters):
            if time.time() >= deadline:
                return best, best_score
            it += 1

            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve (β={beta:.2f}): "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth‑max weights (softmax) -----
            scaled = beta * conv
            max_s = scaled.max()
            w = np.exp(scaled - max_s)          # subtract max for stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ----- gradient -----
            grad = _grad_from_weights(a, w_rev, n, L)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = 0.05 / (np.sqrt(v_hat) + eps)   # base step size
            a = a - step * m_hat

            # projection onto the simplex
            a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Stage 2 – Fast hill‑climbing with random mass moves
# ----------------------------------------------------------------------


def _run_hill_climb(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Randomly relocate a fraction of the mass.  Accept improving moves
    and, with a slowly cooling temperature, occasionally accept worse ones.
    """
    a = a0.copy()
    best = a.copy()
    best_score = cur_best_score
    L = _next_pow2(2 * n - 1)

    # cooling schedule
    start_temp = 3e-2
    end_temp = 5e-6

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling factor
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (end_temp / start_temp) ** (elapsed / total)

        a_prop = a.copy()

        # perform a small number of source→dest moves (2‑3 moves)
        moves = 3
        for _ in range(moves):
            src = np.random.randint(n)
            if a_prop[src] < 1e-14:
                continue
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            eps = a_prop[src] * np.random.rand() * 0.12   # ≤12 % of source mass
            a_prop[src] -= eps
            a_prop[dst] += eps

        # projection (keeps sum = 1, enforces non‑negativity)
        a_prop = _project_simplex(a_prop)

        conv = _conv_self(a_prop, L)
        cur_score = _score_from_conv(conv, n)
        delta = cur_score - best_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            if cur_score < best_score:
                best_score = cur_score
                best = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Stage 3 – Targeted pair‑annealing (spike‑flattening)
# ----------------------------------------------------------------------


def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly weakens the two entries that
    contribute most to the current maximal convolution coefficient.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # indices that can participate in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i to the peak (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max

        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.18   # ≤18 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] pair‑anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
