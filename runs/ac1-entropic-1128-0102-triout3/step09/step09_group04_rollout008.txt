def local_search(a, time_limit):
    start = time.time()
    best = a.copy()
    best_score = compute_score(best)
    n = len(a)
    while time.time() - start < time_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)

        epsilon = best[i] * 0.1 * np.random.rand()   # up to 10% of source
        if epsilon <= 0: continue
        cand = best.copy()
        cand[i] -= epsilon
        cand[j] += epsilon
        # ensure nonnegative
        if cand[i] < 0: continue
        new_score = compute_score(cand)
        if new_score < best_score:
            best = cand
            best_score = new_score
    return best, best_score

"""
A fast three‑stage optimizer for the step‑function benchmark.

*Stage 0* – initialise from the previous best construction (if any) and
  add a few random Dirichlet starts.

*Stage 1* – minimise a smooth approximation of the max‑convolution
  using exponentiated–gradient / projected‑gradient steps on the
  probability simplex.  The smooth max is
      S_α(a)=log‑sum‑exp(α·(a*a))/α,
  whose gradient can be written as a correlation of the soft‑weights
  with the current vector.  The correlation is evaluated with a single
  FFT, so each iteration costs O(n log n).

*Stage 2* – polish the solution with a classic sub‑gradient descent on
  the true nonsmooth objective (the maximum convolution entry).  An
  exponential‑gradient (mirror descent) variant is run afterwards.

All steps respect the supplied time budget (default 1000 s) and keep the
vector on the simplex (non‑negative, sum = 1), which is safe because the
objective is scale‑invariant.  The routine returns a plain ``list`` that
can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (no external dependencies)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based transforms."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    Averaged over all indices attaining (approximately) the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _smooth_max_grad(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of the max‑convolution and its gradient:

        s(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Gradient:  ∂s/∂a_i = 2·Σ_k w_k·a_{k−i},
    where w_k = exp(α·c_k)/Σ_j exp(α·c_j) and c_k = (a*a)_k.
    The correlation Σ_k w_k·a_{k−i} is computed with a single FFT.
    """
    n = a.shape[0]
    conv = _conv_self(a)               # length 2n‑1
    max_val = conv.max()

    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (conv - max_val))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp

    # correlation w ⋆ a   (np.correlate would be O(n²))
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)[: len(w) + n - 1]

    # extract the n relevant entries (shifts 0 … n‑1)
    grad = 2.0 * corr[n - 1: n - 1 + n]

    # smooth value = max + log-sum-exp/α
    smooth_val = max_val + np.log(sum_exp) / alpha
    return smooth_val, grad


def _score(a: np.ndarray) -> float:
    """Exact objective value (scale invariant – assumes sum(a)=1)."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine runs three stages
    (random starts, smooth‑max gradient descent, sub‑gradient polishing)
    while respecting the global time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # -------------------------------------------------------------
    # Problem size.  4096 gives a noticeably lower theoretical bound
    # while still being cheap enough for many FFT‑based iterations.
    # -------------------------------------------------------------
    n = 4096

    # -------------------------------------------------------------
    # Initialise – reuse the previously discovered construction if it
    # exists, otherwise start from the uniform distribution.
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=float)
            if old.ndim == 1 and old.size > 0:
                if old.size != n:
                    # linear interpolation to the required length
                    old_x = np.linspace(0.0, 1.0, old.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    old = np.interp(new_x, old_x, old)
                a = old
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # Normalise to the simplex (sum = 1)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # -------------------------------------------------------------
    # Stage 0 – a few random Dirichlet starts to escape a bad basin.
    # -------------------------------------------------------------
    for _ in range(12):
        if time.time() >= deadline:
            break
        cand = np.random.exponential(1.0, size=n)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] random start improve → {sc:.6f}")

    # Adopt the best start for the subsequent phases.
    a = best_seq.copy()

    # -------------------------------------------------------------
    # Stage 1 – smooth‑max gradient descent (FFT‑based correlation).
    # -------------------------------------------------------------
    it = 0
    while time.time() < deadline - 5.0:   # leave a few seconds for polishing
        it += 1
        # adapt α slowly – start soft, finish hard
        if it < 5000:
            alpha = 0.5
        elif it < 15000:
            alpha = 5.0
        else:
            alpha = 20.0

        # smooth objective and gradient
        smooth_val, grad = _smooth_max_grad(a, alpha)

        # step size schedule (simple 1/√t decay)
        step = 0.3 / math.sqrt(it)

        # projected gradient step
        a = _project_simplex(a - step * grad)

        # evaluate true (nonsmooth) score occasionally
        if it % 250 == 0:
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth‑grad iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

    # -------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the exact max.
    # -------------------------------------------------------------
    sub_deadline = deadline - 2.0
    momentum = np.zeros_like(a)
    it_sub = 0
    while time.time() < sub_deadline:
        it_sub += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it_sub:<6d} "
                  f"improve → {cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_max(a, conv)

        # momentum + diminishing step
        momentum = 0.94 * momentum + 0.06 * grad
        step = 0.12 / math.sqrt(it_sub)
        a = _project_simplex(a - step * momentum)

    # -------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient (mirror descent) polishing.
    # -------------------------------------------------------------
    it_eg = 0
    while time.time() < deadline:
        it_eg += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG iter {it_eg:<6d} "
                  f"improve → {cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        eta = 0.18 / math.sqrt(it_eg)
        a = a * np.exp(-eta * grad)
        # renormalise onto the simplex (exactly sums to 1)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    # -----------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
