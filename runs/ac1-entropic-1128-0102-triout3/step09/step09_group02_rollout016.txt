    L = _next_pow2(2*n-1)

    score = 2.0 * n * conv.max()

        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.1  # up to 10% of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

"""
A multi‑phase optimiser for the “max self‑convolution” problem.

The algorithm builds a good initial guess (an arcsine‑shaped distribution,
which is known to give a flatter self‑convolution than the uniform step)
and then refines it with:

1. Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
2. Exponentiated‑gradient (mirror descent) on the true max‑convolution.
3. Targeted simulated‑annealing with a cooling schedule.
4. A deterministic greedy‐reduction that moves mass from the most
   contributing index of the current peak to a low‑impact location.
5. A short stochastic hill‑climb for final polishing.

All phases respect the 1000 s overall budget; intermediate progress is
printed so the runner can see the current best bound.  The returned
sequence is a plain Python list of non‑negative floats that sum to 1,
which is the normalisation required for the evaluation function.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a) for a normalised to sum‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 30000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            m_scaled = scaled.max()
            w = np.exp(scaled - m_scaled)        # stay numerically stable
            w /= w.sum()
            w_rev = w[::-1]                     # because of convolution symmetry

            # ----- gradient of the smoothed max -----
            Af = rfft(a, n=L)
            Wrev = rfft(w_rev, n=L)
            corr = irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the (non‑smooth) max – only the active sum index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            deadline: float,
                            start_time: float,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current peak index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]

        # source is chosen proportionally to its contribution to the peak
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.20    # up to 20 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    Lfft = _next_pow2(2 * n - 1)

    conv = _conv_self(a, Lfft)                 # full convolution (length 2n‑1)
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (used as a target)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices participating in the current peak
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and a candidate destination
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # choose a destination that currently has the smallest partner
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0

        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            # fallback – move a tiny amount
            eps = min(a[src], 0.02 * a[src])
        else:
            # aim to bring the peak down close to the second‑largest value
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update (O(n)) -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Small stochastic hill‑climb used after the main pipeline
# ----------------------------------------------------------------------
def _run_hill_climb(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.05   # move at most 5 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            conv = conv_new
            score = new_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve: "
                  f"score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) shape – heavy at the ends, flat sum."""
    i = np.arange(n)
    # add a tiny epsilon to avoid division by zero at the boundaries
    eps = 1e-12
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    w /= w.sum()
    return w


def _linear_edge_distribution(n: int) -> np.ndarray:
    """Linear weight that is largest at the two edges and smallest in the centre."""
    i = np.arange(n)
    w = np.maximum(i, n - 1 - i).astype(float) + 1.0   # +1 avoids exact zeros
    w /= w.sum()
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine follows the multi‑phase strategy
    described in the module doc‑string and respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a good deterministic start – arcsine shape is a strong baseline.
    # If a previous construction exists in the global variable
    # `height_sequence_1`, we interpolate it; otherwise we fall back to arcsine.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate to the working dimension
            final_n = 4096
            a0 = np.interp(np.linspace(0, 1, final_n),
                           np.linspace(0, 1, starter.size), starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            raise ValueError
    except Exception:
        final_n = 4096
        # three simple deterministic initialisers – pick the best after a quick test
        cand = [
            _arcsine_distribution(final_n),
            _linear_edge_distribution(final_n),
            np.random.rand(final_n)   # pure random (will be normalised)
        ]
        best_tmp = None
        best_tmp_score = float("inf")
        for c in cand:
            c = np.maximum(c, 0.0)
            c /= c.sum()
            s = _score(c)
            if s < best_tmp_score:
                best_tmp_score = s
                best_tmp = c
        a0 = best_tmp

    best_seq = a0.copy()
    best_score = _score(a0)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed max (≈10 % of the remaining budget)
    # ------------------------------------------------------------------

        phase1_deadline = time.time() + max(8.0, 0.10 * (deadline - time.time()))
        a_adam, score_adam = _run_adam_smoothing(

            lr0=0.6,

        if score_adam < best_score:

    # ------------------------------------------------------------------

        a_exp, score_exp = _run_exp_gradient(

            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted annealing (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:

        a_ann, score_ann = _run_targeted_annealing(

    return [float(x) for x in best_seq]
