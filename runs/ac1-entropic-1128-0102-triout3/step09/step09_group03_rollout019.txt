    # a) Uniform (baseline)
    _try_candidate(np.full(n, 1.0 / n))

    # b) Polynomial a_i ∝ (i+1)^p  (p may be negative)
    for p in np.linspace(-2.0, 3.0, 101):
        if time.time() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=np.float64)
        weights = np.power(i, p)
        if weights.sum() == 0:
            continue
        _try_candidate(weights)

    # c) “U‑shaped’’ a_i ∝ ((i+1)*(n-i))^p  (p < 0 puts mass at the ends)
    for p in np.linspace(-2.0, -0.1, 81):
        if time.time() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=np.float64)
        prod = i * (n + 1 - i)          # (i+1)*(n-i)  with 1‑based indexing
        weights = np.power(prod, p)
        if weights.sum() == 0:
            continue
        _try_candidate(weights)

    # d) Beta‑like a_i ∝ i^{α‑1}·(n+1‑i)^{β‑1}
    alphas = np.linspace(0.2, 5.0, 41)
    betas = np.linspace(0.2, 5.0, 41)
    for α in alphas:
        if time.time() >= deadline:
            break
        for β in betas:
            if time.time() >= deadline:
                break
            i = np.arange(1, n + 1, dtype=np.float64)
            weights = np.power(i, α - 1.0) * np.power(n + 1 - i, β - 1.0)
            if weights.sum() == 0:
                continue
            _try_candidate(weights)

    # e) Geometric decay a_i ∝ r^i
    for r in np.linspace(0.99, 0.9999, 101):
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        weights = np.power(r, i)
        _try_candidate(weights)

    # ------------------------------------------------------------------
    # 2) Exponentiated‑gradient polishing of the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = np.maximum(best_seq, 1e-15)
        a = a / a.sum()
        t = 0
        while time.time() < deadline:
            t += 1
            # full convolution
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] POLISH improved to {cur_score:.6f}")

            # gradient from the *single* shift attaining the maximum
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = k - i_idx
            g = np.zeros_like(a)
            g[i_idx] = 2.0 * a[j_idx]

            # diminishing step size
            eta = 0.5 / np.sqrt(t + 1.0)

            # exponentiated (multiplicative) update
            a = a * np.exp(-eta * g)

            # re‑normalise onto the simplex
            s = a.sum()
            if s == 0.0 or np.isnan(s) or np.isinf(s):
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

    # ------------------------------------------------------------------
    # 3) Final safety checks and report
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return a plain Python list (as required by the judging harness)
    return best_seq.tolist()
