"""
Propose a low‑score step‑function via a three‑phase hybrid optimiser.

Phase 1 – short Adam on a smooth max (quickly reduces the peak).
Phase 2 – sub‑gradient descent with momentum (fine‑grained polishing).
Phase 3 – Frank–Wolfe style LP refinement.
         A linear program maximises the total mass of a new candidate `g`
         while respecting the active convolution constraints (those
         currently attaining the maximal value).  The convex combination
         a←(1‑t)a+t g is accepted if it lowers the true objective.
         Only the active constraints are used, so the LP is tiny (≈10–30
         rows) even for a few thousand variables and can be solved many
         times within the 1000 s budget.

All vectors are kept on the probability simplex (sum = 1, non‑negative),
hence scaling does not affect the evaluated score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – short Adam optimisation on the smooth max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth maximum of the convolution."""
    a = a0.copy()
    n = a.shape[0]

    fft_len = _next_pow2(3 * n - 2)

    # a short schedule – we stay at a moderate smoothing level
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                   # smooth max
            obj = 2.0 * n * f_val             # same scale as evaluate_sequence

            # gradient of the smooth max (see original code)
            # soft‑max weights
            w = np.exp(scaled - max_scaled)
            w /= w.sum()

            # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            grad_a = grad * (2.0 * n)         # outer factor 2·n

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve α={α:.0f} "
                          f"score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic sub‑gradient polishing
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve iter={it:,} "
                  f"score={cur_score:.6f}")

        # active set (indices that are (almost) maximal)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]     # diagonal counted twice – harmless

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Frank‑Wolfe refinement using a tiny LP
# ----------------------------------------------------------------------


def _frank_wolfe_refine(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style improvement.
    At each iteration we solve a small LP:
        maximise    Σ b_i
        subject to  (a * b)[k] ≤ M   for all k in the active set,
                    b_i ≥ 0.
    The solution `b` is rescaled to a probability vector `g`,
    and a line search on the segment a↔g is performed.
    """
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("SciPy not available – skipping Frank‑Wolfe phase.")
        return a0, float(evaluate_sequence(a0.tolist()))

    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # initialise best_score with the true objective (sum(a)=1)
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # active set – indices whose value is within a small relative window
        rel_tol = 1e-4
        active_idx = np.where(conv >= M * (1 - rel_tol))[0]

        # limit the active set size (solving a very small LP is cheap)
        if active_idx.size > 30:
            # keep the largest values only
            order = np.argsort(-conv[active_idx])
            active_idx = active_idx[order[:30]]

        m_active = active_idx.size
        # build the constraint matrix A (m_active × n)
        A = np.zeros((m_active, n))
        for r, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            A[r, i] = a[k - i]           # coefficient a_{k‑i}

        # linear programme: maximise Σ b_i  ⇔ minimise –Σ b_i
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(m_active, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True, 'disp': False})
        if not res.success:
            # if the LP fails we abort the refinement
            print(f"[{time.time() - start_time:6.2f}s] LP failure – terminating FW.")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b < 1e-12:
            # nothing useful has been found
            break
        g = b / sum_b                     # now Σ g_i = 1

        # line search on t ∈ (0,1] – start with a moderate step
        t = 0.5
        improved = False
        while t > 1e-5:
            a_candidate = (1.0 - t) * a + t * g
            conv_new = _conv_self(a_candidate)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                print(f"[{time.time() - start_time:6.2f}s] FW improve iter={it} "
                      f"t={t:.4f} score={best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # no improvement possible – quit early
            break

    return best_a, best_score

# Helper to build the initial candidate
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised start vector (historical best if present,
    otherwise a uniform vector)."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # fallback – uniform distribution
    return np.full(n, 1.0 / n)

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # problem size – a few thousand entries keep FFT cheap
    n = 4800

    # ------------------------------------------------------------------
    # 1) initial candidate
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------
    # 2) short Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.30 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start short Adam optimisation")
    best_a, best_score = _adam_optimize_long(a0, adam_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) sub‑gradient polishing (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = min(deadline, time.time() + 0.30 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, best_score = _subgrad_optimize(best_a, subgrad_deadline,
                                               start_time,
                                               beta=0.94,

        print(f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Frank‑Wolfe refinement (use the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start Frank‑Wolfe refinement")
        best_a, best_score = _frank_wolfe_refine(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] Frank‑Wolfe phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:

    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
