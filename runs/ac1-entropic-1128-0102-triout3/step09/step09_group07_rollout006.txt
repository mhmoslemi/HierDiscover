def _run_adam(a0, n, deadline, start_time, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    best_a = a.copy()
    best_max = np.inf
    while time.time() < deadline:
        t_iter += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print("[{:.2f}s] adam improve: iter={}, max={:.8f}".format(time.time()-start_time, t_iter, cur_max))
        grad = _subgradient_full(a, conv)  # gradient direction for increase
        # descent direction
        g = -grad
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        lr = lr0 / np.sqrt(t_iter)   # decay
        a = a + lr * m_hat / (np.sqrt(v_hat) + eps)  # step
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_max

def _run_incremental_refine(a0, n, deadline, start_time, eps_start=0.2, eps_end=0.005, recompute_every=5000):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    total_time = deadline - start_time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linearly decay eps_frac
        eps_frac = max(eps_end, eps_start * (1 - elapsed / total_time))
        # find max bin
        k_max = int(np.argmax(conv))
        # find source with largest contribution to the max bin
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        jdx = k_max - idx
        contrib = a[idx] * a[jdx]
        src = idx[np.argmax(contrib)]
        if a[src] < 1e-15:
            # fallback random source
            src = rng.choice(n, p=a / a.sum())
        # choose destination with minimal contribution to max bin (prefer zero)
        pos = np.arange(n)
        valid = (k_max - pos >= 0) & (k_max - pos < n)
        a_contrib = np.zeros(n)
        a_contrib[valid] = a[k_max - pos[valid]]
        # prefer zero contribution
        zero_idxs = np.where(a_contrib == 0)[0]
        if zero_idxs.size > 0:
            # avoid src
            zero_idxs = zero_idxs[zero_idxs != src]
            if zero_idxs.size > 0:
                dst = rng.choice(zero_idxs)
            else:
                # fallback
                dst = rng.choice(np.delete(np.arange(n), src))
        else:
            # all contributions positive, pick smallest
            dst_vals = a_contrib.copy()
            dst_vals[src] = np.inf
            dst = int(np.argmin(dst_vals))
        # compute s vector
        s = np.zeros(2 * n - 1, dtype=float)
        # subtract src contribution
        s[src:src + n] -= 2.0 * a
        # add dst contribution
        s[dst:dst + n] += 2.0 * a
        s_max = s[k_max]
        if s_max >= 0:
            # move would not reduce current max; try random alternative
            continue
        # compute eps_star
        mask = s > s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]
            delta_s = s[mask] - s_max
            # avoid division by zero but we know delta_s>0
            eps_candidates = delta_conv / delta_s
            # keep positive
            eps_candidates = eps_candidates[eps_candidates > 0]
            eps_star = eps_candidates.min() if eps_candidates.size > 0 else np.inf
        else:
            eps_star = np.inf
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)
        if epsilon <= 0:
            continue
        # apply move
        conv = conv + epsilon * s
        a[src] -= epsilon
        a[dst] += epsilon
        max_val = conv.max()
        if max_val < best_max:
            best_max = max_val
            best_a = a.copy()
            print("[{:.2f}s] inc-refine improve: max={:.8f}".format(time.time() - start_time, best_max))
        # periodically recompute conv to avoid drift
        if int(np.random.rand() * recompute_every) == 0:
            conv = _conv_self(a)
            max_val = conv.max()
    return best_a, best_max

"""
Improved solution for the “min‑max self‑convolution” challenge.

Algorithm overview
-------------------
1. **Initialisation** – start from the published construction
   (`height_sequence_1`) if available, otherwise from a random
   probability vector.  The vector is up‑sampled to a fixed length
   `N = 4800` and slightly perturbed to escape tiny local minima.

2. **Adam optimisation** – minimise the exact ℓ∞‑norm of the
   self‑convolution (`max(conv)`) on the probability simplex.
   The sub‑gradient of `max(conv)` is computed analytically
   (all active convolution entries are taken into account).  After each
   step the iterate is projected back onto the simplex.  The best
   iterate seen is remembered.

3. **Greedy incremental refinement** – a large mass transfer is
   performed in each iteration.  For the current maximal convolution
   bin `k*` we choose a *source* index that contributes most to this
   bin and a *destination* index that does **not** contribute at all.
   The effect of moving an amount `ε` from source to destination on the
   whole convolution is linear; a one‑dimensional linear programme
   yields the maximal admissible `ε` that still reduces the current
   maximum.  This gives huge per‑iteration progress while each move is
   O(N).  The procedure runs until the time budget is exhausted,
   keeping the best sequence observed.

The whole routine respects a 1000 s wall‑clock limit (a 0.5 s safety
margin) and returns the best list of non‑negative heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – Adam optimiser on the exact max‑convolution
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    Returns the best vector seen and the corresponding maximum
    convolution entry.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_max = np.inf
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={it:,}, max={cur_max:.8f}")

        # sub‑gradient of the *increase* of the max; we walk opposite
        grad = _subgradient_full(a, conv)
        g = -grad

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decaying step size
        lr = lr0 / math.sqrt(it)

        a = a + lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)          # enforce non‑negativity and sum=1

    return best_a, best_max


# ----------------------------------------------------------------------
# Stage 2 – Greedy incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------


def _run_incremental_refine(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            eps_start: float = 0.30,
                            eps_end: float = 0.005,
                            recompute_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shift mass from a source index (large contribution to the
    current maximal convolution bin) to a destination index that does
    not contribute at all.  The maximum admissible step `ε` is computed
    analytically (a 1‑D LP).  Each iteration costs O(n) and yields a
    guaranteed reduction of the current maximum.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal bin
        k_max = int(np.argmax(conv))

        # ----- source: index with largest contribution to the maximal bin -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx
        contrib = a[idx] * a[jdx]
        src = idx[np.argmax(contrib)]
        if a[src] < 1e-15:                     # fall back to a random source
            src = rng.choice(n, p=a / a.sum())

        # ----- destination: index that does NOT appear in the max bin -----
        pos = np.arange(n)
        valid = (k_max - pos >= 0) & (k_max - pos < n)
        a_contrib = np.zeros(n)
        a_contrib[valid] = a[k_max - pos[valid]]

        zero_idxs = np.where(a_contrib == 0)[0]
        zero_idxs = zero_idxs[zero_idxs != src]    # avoid self‑move
        if zero_idxs.size > 0:
            dst = rng.choice(zero_idxs)
        else:
            # all positions appear – pick the one with the smallest contribution
            tmp = a_contrib.copy()
            tmp[src] = np.inf
            dst = int(np.argmin(tmp))

        # ----- linear effect of moving ε from src to dst -----
        # s[t] = -2·a[t‑src] + 2·a[t‑dst]   (out‑of‑range entries are 0)
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # this move would not lower the current maximum – try another pair
            continue

        # ----- maximal admissible ε before another entry overtakes the maximum -----
        mask = s > s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ----- apply the move analytically (O(n)) -----
        conv = conv + epsilon * s
        a[src] -= epsilon
        a[dst] += epsilon
        max_val = conv.max()
        accepted += 1
        it += 1

        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep numerical errors in check
        if accepted % recompute_every == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of half a second to guarantee a return before timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – start from the published construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    N = 4800                      # length of the step function (can be changed)
    if starter is not None:
        a0 = _upsample(starter, N)
    else:
        a0 = np.random.rand(N)
        a0 /= a0.sum()

    # add a tiny random perturbation to avoid being stuck in a flat region
    a0 = 0.9 * a0 + 0.1 * np.random.rand(N)
    a0 /= a0.sum()

    best_seq = a0.copy()
    best_max = _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] initial max={best_max:.8f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimiser (takes roughly 15‑20 % of the budget)
    # ------------------------------------------------------------------
    stage1_deadline = start_time + min(200.0, 0.18 * (deadline - start_time))
    a_adam, max_adam = _run_adam(best_seq, N, stage1_deadline, start_time)
    if max_adam < best_max:
        best_max = max_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] after Adam: max={best_max:.8f}")

    # ------------------------------------------------------------------
    # Stage 2 – Greedy incremental refinement (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined, max_refined = _run_incremental_refine(best_seq, N, deadline, start_time)
        if max_refined < best_max:
            best_max = max_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refined max={best_max:.8f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    return [float(x) for x in best_seq]
