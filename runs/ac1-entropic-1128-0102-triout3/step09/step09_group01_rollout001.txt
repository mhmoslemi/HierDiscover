"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

under a ~1000 s wall‑clock budget.

Algorithm (high‑level)
----------------------
1️⃣  **Fast parametric sweep** – generate a large variety of analytic families
    (geometric, reverse‑geometric, double‑exponential, power, Beta‑type,
    random Dirichlet draws and mixtures with the uniform vector).  Every
    candidate is normalised to unit sum and evaluated with an FFT‑based
    convolution.  The best of this cheap sweep seeds the next phases.

2️⃣  **Smoothed‑gradient descent** – minimise a smooth approximation of the
    “max” using a soft‑max (log‑sum‑exp) with a temperature that is
    gradually increased.  The gradient of the soft‑max is a weighted
    cross‑correlation of the current sequence and the soft‑max weights.
    An Euclidean step with a decreasing step‑size is taken and the iterate
    is projected onto the probability simplex.  Momentum is added
    (Adam‑style) for faster convergence.

3️⃣  **Annealing polish** – a lightweight mass‑transfer simulated annealer
    (move a small fraction of mass from one coordinate to another) runs
    until the deadline, accepting downhill moves and occasional uphill
    moves according to a geometric temperature schedule.

All three stages keep track of the best sequence ever seen.  The final
sequence is re‑normalised to sum to 1 and returned as a plain list of
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / projection helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv) )
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : smoothed gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             init_score: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style gradient descent on the smoothed max:
        f(a) = logsumexp(alpha * conv(a,a)) / alpha

    The gradient is a weighted cross‑correlation of a and the soft‑max
    weights.  The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    # Adam parameters
    mt = np.zeros_like(a)
    vt = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # FFT length large enough for both self‑conv (2n‑1) and cross‑corr (n+2n‑2)
    L = _next_pow2(3 * n - 2)

    it = 0
    # temperature (smoothing) schedule
    alpha0 = 1.0
    alpha_max = 150.0

    while time.time() < deadline:
        it += 1
        # ----- self‑convolution -----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑grad it={it:,} score={cur_score:.6f}")

        # ----- smoothing parameter -----
        # increase alpha linearly to alpha_max over the first half of the budget
        elapsed = time.time() - start_time
        total = deadline - start_time
        if elapsed < total * 0.5:
            alpha = alpha0 + (alpha_max - alpha0) * (elapsed / (total * 0.5))
        else:
            alpha = alpha_max

        # ----- soft‑max weights -----
        # numerical stability: subtract max before exponentiation
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fall back to a single hot weight at the max location
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient (cross‑correlation) -----
        Wf = np.fft.rfft(w, n=L)
        # cross‑correlation = ifft( Af * conj(Wf) )
        grad_complex = np.fft.irfft(Af * np.conj(Wf), n=L)[:n]
        grad = 2.0 * grad_complex.real

        # ----- Adam update -----
        mt = beta1 * mt + (1.0 - beta1) * grad
        vt = beta2 * vt + (1.0 - beta2) * (grad ** 2)
        mt_hat = mt / (1.0 - beta1 ** it)
        vt_hat = vt / (1.0 - beta2 ** it)

        # step size schedule (decreasing)
        step0 = 0.15
        step = step0 / (np.sqrt(it) + 1.0)

        a = a - step * mt_hat / (np.sqrt(vt_hat) + eps)

        # ----- simplex projection -----
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 400_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (move a small fraction of mass from one
    coordinate to another)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ---- random Dirichlet draws (sparse shapes) ----
    rng = np.random.default_rng()
    for _ in range(80):
        if time.time() >= deadline - 30:
            break
        # concentration < 1 favours sparsity
        alpha = rng.uniform(0.1, 1.0)
        rnd = rng.dirichlet(np.full(n, alpha))
        sc = _eval_fast(rnd, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = rnd.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={alpha:.3f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent (bulk of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve a few seconds for the final annealing polish
    if remaining > 30.0:
        grad_deadline = deadline - 10.0
        a_smooth, sc_smooth = _smooth_gradient_descent(best_seq, n,
                                                        grad_deadline,
                                                        start_time,
                                                        best_score)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑grad final improve score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2,
                                       end_temp=1e-7,
                                       steps=500_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Return as plain Python list
    return [float(x) for x in best_seq]
