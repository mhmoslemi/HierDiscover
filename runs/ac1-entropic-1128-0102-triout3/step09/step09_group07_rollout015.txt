    L = next_pow2(3*n - 2)   # length for convolution of w (2n-1) with a_rev (n)

"""
Propose a step‑function with a low self‑convolution maximum.

Algorithm
---------
1. Initialise the sequence from the provided `height_sequence_1`
   (if any) or from a random point on the simplex.
2. Warm‑up a few seconds with the exponentiated‑gradient method
   from the reference implementation (very cheap and gives a decent
   start).
3. Optimise the *smooth* max of the self‑convolution by Adam on the
   log‑probability (soft‑max) parametrisation.
   The smooth max is a log‑sum‑exp with a temperature β that is
   increased linearly during the run, so the optimisation smoothly
   turns into a true max‑optimisation.
   The gradient of the smooth max w.r.t. the probabilities is obtained
   analytically using one FFT‑based convolution (O(n log n)).
4. A final “large‑move” refinement step (the same incremental‑mass‑shift
   routine used previously) removes the last few bits of sub‑optimality.
5. Return the best sequence (sum normalised to 1) as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (weights w) w.r.t. a.
    Implements g_i = 2 Σ_k w_k a_{k-i} using one FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    L = _next_pow2(3 * n - 2)           # length for convolution w * a_rev
    A_rev = np.fft.rfft(a_rev, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(A_rev * W, n=L)[: 3 * n - 2]
    # slice indices n‑1 … n‑1+n‑1
    grad = 2.0 * corr[(n - 1):(n - 1 + n)]
    return grad


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – exponentiated‑gradient warm‑up (optional, cheap)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam on a smooth approximation of the max
# ----------------------------------------------------------------------
def _run_adam_optimize(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       beta_start: float = 5.0,
                       beta_end: float = 150.0,
                       lr: float = 0.2,
                       b1: float = 0.9,
                       b2: float = 0.999,
                       eps: float = 1e-8):
    """
    Optimises the exact max of the self‑convolution by Adam on the
    smooth log‑sum‑exp approximation.
    """
    # start from logits (log‑probabilities)
    y = np.log(a0 + 1e-20)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a0.copy()
    best_score = float("inf")
    it = 0

    total_time = deadline - start_time
    while time.time() < deadline:
        it += 1
        a = _softmax(y)                 # stays on the simplex
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # linear schedule for β
        elapsed = time.time() - start_time
        beta = beta_start + (beta_end - beta_start) * min(elapsed / total_time, 1.0)

        # smooth‑max (log‑sum‑exp) weights
        scaled = beta * conv
        max_s = scaled.max()
        exp_scaled = np.exp(scaled - max_s)
        sum_exp = exp_scaled.sum()
        w = exp_scaled / sum_exp                # Σ w = 1

        # gradient w.r.t. a
        grad_a = _grad_from_weights(a, w)

        # gradient w.r.t. logits (chain rule through softmax)
        dot_ag = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot_ag)

        # Adam update
        m = b1 * m + (1.0 - b1) * grad_y
        v = b2 * v + (1.0 - b2) * (grad_y ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – optimal incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    recompute_every: int = 5000):
    """
    Perform many mass‑shifts.  For a chosen source `src` and destination `dst`
    the effect on the whole convolution is linear in ε.
    The algorithm computes the largest ε that still reduces the current
    maximal entry (solution of a 1‑D linear programme).  This yields
    far larger progress per move while keeping the per‑iteration cost O(n).
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal index
        k_max = int(np.argmax(conv))

        # source: pick an index that really contributes to the max bin
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination: prefer indices that do not appear much in the max bin
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # per‑unit‑ε effect on the whole convolution
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            continue    # this move would not lower the current max

        # maximal admissible ε before another entry overtakes the max
        mask = s > s_max                     # slopes strictly larger than s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # apply the move analytically (O(n))
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        # ensure genuine improvement (avoid tiny numerical noise)
        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional re‑normalisation / recomputation
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
