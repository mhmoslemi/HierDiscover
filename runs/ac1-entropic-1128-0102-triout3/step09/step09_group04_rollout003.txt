"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The routine follows three complementary phases:

1. **Iterated LP balancing** – repeatedly solve the linear programme
   maximise   Σ b_j    s.t.   b≥0  and  (a∗b)(k) ≤ M  (M = max(a∗a)).
   The optimiser b pushes probability mass into positions where the
   current a has little influence.  A convex combination
   a←(1‑t)a+t b (t decreasing) is performed and the result is projected
   onto the probability simplex.  The loop is continued until the
   time budget for this phase is exhausted; unlike the reference code the
   iteration limit is removed so the LP can keep driving the peak down.

2. **Adam‑style projected gradient descent** – we minimise the true
   nonsmooth objective using the sub‑gradient of the peak of the
   autocorrelation (computed by ``_subgradient_max``).  Adam’s adaptive
   learning‑rate scheme together with Euclidean projection onto the simplex
   gives fast, stable convergence.

3. **Entropy‑mirror (exponentiated‑gradient) polishing** – a short EG
   phase refines the solution after the Adam step; it is essentially the
   same update as in the reference but with a schedule that decays more
   gently.

All phases respect a global time budget (default 1000 s) and keep the best
vector found.  The final vector is normalised to sum to 1 before being
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / optimisation helpers (no external dependencies)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    For all k attaining the maximum we add the contribution of a to the
    gradient; the result is averaged over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _build_T(vec: np.ndarray) -> np.ndarray:
    """
    Build the convolution matrix T such that (a∗b) = T @ b.
    Shape is (2·n‑1, n) and each column i contains a shifted copy of ``vec``.
    """
    n = vec.shape[0]
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = vec
    return T


def _score(a: np.ndarray, n: int) -> float:
    """Convenient wrapper returning the objective value (with sum(a)=1)."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

            if tmp.ndim == 1 and tmp.size > 0:

            else:
                a = np.full(n, 1.0 / n)
        except Exception:

    else:

    # Normalise to the simplex
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0.0:

    else:

    best_seq = a.copy()
    best_score = _score(a, n)
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – iterated LP balancing (if cvxpy is available)
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        # allocate roughly half of the remaining time to the LP phase
        lp_deadline = time.time() + (deadline - time.time()) * 0.45
        lp_iter = 0
        t_factor = 0.6                     # initial convex‑combination weight

        while time.time() < lp_deadline:

            grad = _subgradient_max(a, conv_a)

            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = lr0 / math.sqrt(it)   # slowly decaying base learning rate
            step = lr / (np.sqrt(v_hat) + eps)

            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase 3 – Entropy‑mirror descent (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:

        it = 0

            it += 1
            conv_a = _conv_self(a)
            cur_score = _score(a, n)

                best_seq = a.copy()

                      f"improve → {cur_score:.6f}")

            a = a * np.exp(-eta * grad)   # multiplicative update
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
            else:
                a = np.full_like(a, 1.0 / n)

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_score = _score(best_seq, n)
    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
