"""
Propose a step‑function with a very low evaluation score.

The implementation follows a simple but effective strategy:

1. **Problem size** – If a global starter ``height_sequence_1`` exists we
   reuse its length, otherwise we fall back to ``n = 4096`` (the size used
   by the reference solution).

2. **Multiple random restarts** – For the bulk of the time budget we
   generate many random probability vectors (using a Dirichlet
   distribution) and run a fast projected gradient descent on a smooth
   approximation of the max‑convolution.  The smooth surrogate is the
   Log‑Sum‑Exp (LSE) of the convolution entries; its gradient can be
   written analytically and projected back onto the probability simplex.

3. **Fine‑tuning** – The best iterate from the random‑restart phase is
   further polished with a cheap simulated‑annealing routine that moves
   tiny amounts of mass between coordinates.

4. **Return** – The final vector is projected onto the simplex (non‑negative,
   sum = 1) and returned as a plain Python list, which is exactly the
   format expected by ``evaluate_sequence``.

The code is written for a single CPU, respects the 1000 s budget,
and contains only top‑level functions (no closures, no lambdas).  It
uses only the standard library and NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using an FFT.
    ``L`` can be supplied to reuse a pre‑computed transform length.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of the vector ``v`` onto the simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate for max(conv) – Log‑Sum‑Exp with temperature α
# ----------------------------------------------------------------------


def _lse_conv(a: np.ndarray, alpha: float) -> float:
    """
    Compute the smooth approximation  (1/α) * log Σ exp(α·c_i)
    where c_i are the entries of the self‑convolution of ``a``.
    """
    conv = _conv_self(a)
    # use the classic log‑sum‑exp trick for numerical stability
    max_c = conv.max()
    if alpha * max_c > 700:   # avoid overflow of exp
        return max_c
    scaled = alpha * (conv - max_c)
    return (max_c + (1.0 / alpha) * np.log(np.exp(scaled).sum()))


def _grad_lse_conv(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. ``a``.
    The derivative of a single convolution entry c_k = Σ_i a_i a_{k-i}
    is 2·[a shifted version].
    The gradient of the LSE combines these with softmax weights.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    scaled = alpha * (conv - max_c)
    # softmax weights, normalised
    weights = np.exp(scaled)
    Z = weights.sum()
    weights /= Z

    # gradient contribution from each shift k:
    # grad_k = Σ_i ( d c_i / d a_k ) * w_i
    # d c_i / d a_k = a_{i-k} + a_{i-k'}?? actually for linear convolution:
    # for a fixed k, the derivative of conv_i = Σ_j a_j a_{i-j}
    # w.r.t. a_k gives a_{i-k} + a_{i-k} (since a appears twice when j=k)
    # which is 2 * a_{i-k} when indices are in range, else 0.
    # Hence grad = 2 * Σ_i w_i * a_{i-k}
    # This is a convolution of the weight vector with a, multiplied by 2.
    # We can compute via FFT for speed.
    w = weights
    # compute convolution w (reversed) * a
    # Since conv is linear (not circular), we pad appropriately.
    # Simple direct implementation works because n is moderate.
    grad = np.zeros_like(a)
    for k in range(n):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx] * w[k]   # careful, w index is shift k
    # The above double loop is O(n^2) – too slow.
    # Instead use FFT:
    #   grad = 2 * (ifft(fft(w) * fft(a))).real  (circular convolution)
    # But we need linear convolution of w with a, mirrored properly.
    # A robust and fast implementation:
    L = _next_pow2(2 * n - 1)
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a, n=L)
    conv_lin = np.fft.irfft(Wf * Af, n=L)[: 2 * n - 1]
    # The contribution to grad for index k comes from conv_lin[n-1 + k]
    grad = 2.0 * conv_lin[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent on the smooth surrogate
# ----------------------------------------------------------------------


def _pgd_smooth(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the Log‑Sum‑Exp surrogate.
    Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    beta = 0.9
    momentum = np.zeros_like(a)

    # schedule for the temperature α – start with a large value (smooth)
    # and gradually increase (more like the true max)
    alpha0 = 1.0
    alpha_max = 50.0

    while time.time() < deadline:
        it += 1
        # linearly increase α
        alpha = alpha0 + (alpha_max - alpha0) * (it / 50000)
        if alpha > alpha_max:
            alpha = alpha_max

        # gradient of smooth objective
        grad = _grad_lse_conv(a, alpha)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size – diminishing
        step = 0.5 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD iter={it:,} α={alpha:.2f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing that moves tiny masses between
    coordinates – useful for a final polish after gradient descent.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_steps = 30_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)               # choose source proportional to its mass
        dst = rng.integers(n)                  # destination uniformly
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30    # transfer up to 30% of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.

    The routine:
      * chooses a problem size (reuse ``height_sequence_1`` if available),
      * runs several random restarts of projected gradient descent on a smooth
        surrogate of the max‑convolution,
      * refines the best candidate with a short simulated‑annealing polish,
      * normalises the final vector onto the probability simplex and returns it.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 4096

    # --------------------------------------------------------------
    # Helper to generate a random probability vector (Dirichlet)
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Start with a uniform vector (the theoretical optimum of the
    # continuous relaxation).  It also gives a solid baseline.
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective(best_seq)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Random‑restart projected gradient descent (smooth surrogate)
    # --------------------------------------------------------------
    # Allocate roughly 70% of the remaining budget to this phase.
    remaining = deadline - time.time()
    if remaining > 30.0:
        grad_deadline = time.time() + 0.70 * remaining
        # we will perform a small number of restarts (e.g. 5) – each gets
        # an equal share of the allocated time.
        n_restarts = 5
        per_restart = (grad_deadline - time.time()) / n_restarts
        for r in range(n_restarts):
            if time.time() >= grad_deadline:
                break
            # Dirichlet(1) == uniform over the simplex
            a0 = rng.dirichlet(np.ones(n))
            restart_deadline = time.time() + per_restart
            seq, score = _pgd_smooth(a0, restart_deadline, start_time)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Restart {r+1}/{n_restarts} improved score={score:.6f}")

    # --------------------------------------------------------------
    # Final annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        seq, score = _anneal(best_seq, anneal_deadline, start_time)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved final score={score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation onto the simplex (non‑negative, sum = 1)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list as required by the evaluation harness
    return [float(x) for x in best_seq]
