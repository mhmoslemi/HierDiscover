"""
A fast three‑stage optimiser for the “max‑auto‑convolution” problem.

Stage 1 – ultra‑fast parametric sweep.
    We generate a variety of analytic shapes (geometric, power, beta,
    sinusoidal and logistic families) on the simplex and keep the best
    candidate.

Stage 2 – projected Adam sub‑gradient descent.
    For the active convolution index we have the exact sub‑gradient
    g_i = 2·a_{k‑i} (zero outside the overlapping interval).  An Adam
    optimiser with a decreasing learning‑rate is applied, always
    projecting back onto the probability simplex.

Stage 3 – long mass‑transfer annealing.
    Tiny random transfers of mass between two positions are accepted
    with a Metropolis rule; the temperature decays exponentially.
    This polish is run for the rest of the time budget.

The routine respects the 1000 s wall‑clock limit, prints progress and
returns a plain Python list ready for ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (convolution and power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w≥0, Σw=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum normalised to 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam sub‑gradient descent on f(a)=2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the exact sub‑gradient of the active
    convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    L_grad = _next_pow2(3 * n - 2)   # length for the gradient FFTs

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # active index (any max is fine)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-8,
               steps: int = 400_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (move a fraction of mass from src→dst)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its mass and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns a plain
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # small safety margin

    n = 4096                                     # problem size used by the reference
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Uniform baseline (objective = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if ``height_sequence_1`` exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep – richer families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric (decreasing) + mix with uniform
    eps_grid = np.logspace(-4, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family (a_i ∝ (i+1)^p)
    p_grid = np.linspace(-1.5, 3.0, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family a_i ∝ (i+1)^p * (n-i)^q
    beta_params = np.linspace(0.0, 2.5, 6)            # more granularity
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # sinusoidal family (increasing then decreasing)
    for p in np.linspace(0.5, 4.0, 8):
        seq = np.sin(np.pi * (idx + 1) / (n + 1)) ** p
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sin p={p:.2f} score={sc:.6f}")

    # logistic (sigmoid) family
    for steep in np.linspace(2.0, 12.0, 6):
        seq = 1.0 / (1.0 + np.exp(-steep * (idx - n / 2) / n))
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Logistic steep={steep:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refine with Adam sub‑gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 60.0:               # keep a few seconds for polish
        adam_deadline = deadline - 10.0
        a_adam, sc_adam = _adam_descent(best_seq, n, adam_deadline,
                                        start_time, best_score)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam final improve score={sc_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline,
                                       start_time,
                                       best_score, best_seq,
                                       start_temp=2e-2,
                                       end_temp=1e-8,
                                       steps=500_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
