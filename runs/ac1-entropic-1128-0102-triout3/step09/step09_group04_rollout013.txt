"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The method combines three complementary phases and respects a global
time budget (default 1000 s).

1️⃣  **Initialisation** – start from the best known construction if it
    exists (global variable ``height_sequence_1``), otherwise from the
    uniform distribution.  The vector is always normalised to sum = 1
    (the objective is scale‑invariant).

2️⃣  **Smoothed‑Lᵖ descent** – we minimise a smooth approximation of the
    L∞‑norm.  For a fixed exponent *p* we set
        w_i ∝ (conv_i)^{p‑1},
    which concentrates the gradient on the largest convolution entries.
    The gradient of the resulting “soft‑max” is
        g = 2·corr(w, a)                     (corr = valid correlation)
    and is updated with an Adam‑style rule.  The exponent *p* is
    increased gradually (≈4 → 64) so that the optimisation first sees a
    smooth landscape and then focuses on the true peak.

3️⃣  **Local random‑perturbation hill‑climbing** – after the smooth
    phase we perform fast stochastic moves (transfer a tiny amount of
    mass between a few coordinates) and keep any improving move.  The
    remaining time budget is spent here; it is cheap and often squeezes
    out the last bits of improvement.

All vectors are projected onto the probability simplex after each
update (Euclidean projection).  The routine returns a plain ``list`` of
floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Simplex projection (Euclidean)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# FFT‑based self‑convolution (real input)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two (≥ x)."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
# Objective (scale‑invariant, assumes sum(a)=1)
# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum = 1."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()

# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows the three‑stage
    strategy described in the module docstring and respects the supplied
    time budget (in seconds).  The returned value is a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # Choose a problem size.  2048 works well for the time budget.
    # ---------------------------------------------------------
    n = 2048

    # Initialise from the best known construction if it exists.
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():

            if tmp.ndim == 1 and tmp.size > 0:

                    new_x = np.linspace(0.0, 1.0, n)

    else:

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0.0:

    else:
        a = np.full(n, 1.0 / n)

    best_seq = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # Helper for the smooth‑Lᵖ phase
    # ---------------------------------------------------------
    def _smooth_lp_step(a_vec: np.ndarray, p: float,
                        m: np.ndarray, v: np.ndarray,
                        it: int) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        Perform one Adam‑style update with the Lᵖ‑smoothed objective.
        Returns (new_a, new_m, new_v).
        """
        # convolution of the current vector
        conv = _conv_self(a_vec)

        eps = 1e-14

        return a_new, m, v

    # ---------------------------------------------------------
    # Phase 1 – Smoothed Lᵖ descent (≈ 60 % of the budget)
    # ---------------------------------------------------------
    lp_deadline = start_time + 0.60 * (deadline - start_time)
    m = np.zeros_like(a)

        a, m, v = _smooth_lp_step(a, p, m, v, it)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-9:

    # ---------------------------------------------------------
    while time.time() < deadline:

            if i == j:

            amount = delta * cand[i]
            # keep non‑negativity
            amount = max(-cand[j], min(cand[i], amount))
            cand[i] -= amount
            cand[j] += amount

        # re‑normalise (project onto simplex)
        cand = _project_simplex(cand)

        cur_score = _objective(cand)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] random perturb improve → {cur_score:.6f}")

    # ---------------------------------------------------------
    # Final normalisation (the vector is already on the simplex)
    # ---------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
