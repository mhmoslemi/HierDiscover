    it = 0
    stagnant = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # recompute conv each iteration? we have conv maybe from previous iteration.
        # Let's recompute each iteration for safety.
        conv = _conv_self(a, L)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute second max
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()
        # range contributing indices
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[max_idx - idx_range]
        src = idx_range[np.argmax(contrib)]
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        out_mask = np.ones(n, dtype=bool)
        out_mask[lo:hi+1] = False
        if np.any(out_mask):
            dst_candidates = np.where(out_mask)[0]
            # pick destination with smallest a
            dst = dst_candidates[np.argmin(a[dst_candidates])]
        else:
            # fallback: pick index with minimal contribution
            dst = idx_range[np.argmin(contrib)]

"""
Hybrid optimisation for the max‑self‑convolution objective.

The routine works in three stages within the 1000 s timeout:

1. **Exponentiated‑gradient (mirror descent)** – a fast stage that spreads
   the mass away from the current peak of the convolution while staying on
   the probability simplex.

2. **Deterministic peak‑reduction** – repeatedly identifies the index that
   contributes most to the current maximal convolution value and moves a
   carefully chosen amount of mass to an index that contributes the least.
   The step size is chosen analytically so that the peak is driven just
   below the second–largest value, guaranteeing monotonic improvement.

3. **Random local search** – when the deterministic stage stalls we perform
   tiny random transfers of mass and keep any improvement.

All stages respect the global time budget, print occasional progress
messages and finally return the best sequence (a plain ``list[float]``)
found.  The sequence is normalised to sum to 1, all entries are ≥ 0,
and the evaluation function from the competition can be called on the
result.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length L (power of two).
    Returns a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient (mirror descent) on the true max‑conv.
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> Tuple[np.ndarray, float]:
    """Mirror‑descent on the active (max) convolution index."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index of the current maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # gradient of conv[k] w.r.t. a (2·a_{k-i} except at the centre)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy peak reduction
# ----------------------------------------------------------------------
def _run_peak_reduction(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        max_iters: int = 500_000) -> Tuple[np.ndarray, float]:
    """
    Repeatedly transfer a small amount of mass from the index that
    contributes most to the current maximal convolution value to an
    index that contributes the least.  The step size is chosen so that
    the peak is driven just below the second‑largest value.
    """
    a = a0.copy()
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    L = _next_pow2(2 * n - 1)

    # keep a current full convolution to avoid recomputing on every
    # iteration when a move is rejected.
    conv = _conv_self(a, L)
    cur_score = 2.0 * n * conv.max()
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # recompute the convolution ( cheap enough for n≈4k )
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # range of indices that actually appear in this maximal sum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1)

        # contributions a[i] * a[max_idx - i]
        contrib = a[idx_range] * a[max_idx - idx_range]
        src = idx_range[np.argmax(contrib)]
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # choose a destination outside the contributing range (partner = 0)
        out_mask = np.ones(n, dtype=bool)
        out_mask[lo:hi + 1] = False
        if np.any(out_mask):
            dst_candidates = np.where(out_mask)[0]
            dst = dst_candidates[np.argmin(a[dst_candidates])]
            partner_dst = 0.0
        else:
            # fallback: pick the index with the smallest contribution
            dst = idx_range[np.argmin(contrib)]
            partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0

        delta = partner_src - partner_dst
        if delta <= 0.0:
            # no useful direction – make a tiny random move
            dst = np.random.randint(n)
            while dst == src:
                dst = np.random.randint(n)
            eps = min(a[src], 0.01 * a[src])
        else:
            # we would like max after the move to be just below second_max
            needed = max_val - second_max
            eps_target = needed / (2.0 * delta + 1e-15) * 0.9  # safety factor
            eps = min(a[src], max(eps_target, 0.0))
            if eps < 1e-12:
                eps = min(a[src], 0.01 * a[src])

        if eps <= 0.0:
            continue

        # perform the transfer
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new[a_new < 0.0] = 0.0        # numerical safety

        # new score
        conv_new = _conv_self(a_new, L)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score - 1e-14:     # monotone improvement
            a = a_new
            conv = conv_new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] peak‑red improve: "
                      f"iter={it:,}, score={new_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2000:
                # perform a small random perturbation to escape a plateau
                src2 = np.random.randint(n)
                if a[src2] > 1e-12:
                    dst2 = np.random.randint(n)
                    while dst2 == src2:
                        dst2 = np.random.randint(n)
                    eps2 = min(a[src2], 0.001 * a[src2])
                    a[src2] -= eps2
                    a[dst2] += eps2
                no_improve = 0
                # force a recomputation of the convolution after the perturb
                conv = _conv_self(a, L)
                cur_score = 2.0 * n * conv.max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – cheap random local search (optional)
# ----------------------------------------------------------------------
def _run_random_search(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       cur_best_seq: np.ndarray,
                       cur_best_score: float) -> Tuple[np.ndarray, float]:
    """Tiny random mass transfers; keep any improvement."""
    a = a0.copy()
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        src = np.random.randint(n)
        if a[src] < 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        eps = a[src] * np.random.rand() * 0.02   # up to 2 %
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv = _conv_self(a_prop, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] random improve: "
                  f"score={score:.6f}")

    return best_a, best_score

# Main entry point expected by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # --------------------------------------------------------------
    # Initialise – use the previously discovered construction if present.
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is a global variable created by prior runs.
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate to the working length
            final_n = 4096
            a0 = np.interp(
                np.linspace(0, 1, final_n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            raise Exception
    except Exception:                          # fallback to uniform
        final_n = 4096
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    # keep track of the best solution found so far
    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – Exponentiated gradient (fast global shaping)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + min(80.0, (deadline - time.time()) * 0.30)
    a_exp, score_exp = _run_exp_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        eta0=0.6,
    )
    if score_exp < best_score:
        best_score = score_exp
        best_seq = a_exp.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
              f"score={score_exp:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic greedy peak reduction
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = deadline - 30.0   # reserve a little time for phase 3
        a_red, score_red = _run_peak_reduction(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=800_000,
        )
        if score_red < best_score:
            best_score = score_red
            best_seq = a_red.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_red:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Random local search (polish remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_rand, score_rand = _run_random_search(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_seq,
            best_score,
        )
        if score_rand < best_score:
            best_score = score_rand
            best_seq = a_rand.copy()
            print(f"[{time.time() - start_time:.2f}s] final improve: "
                  f"score={score_rand:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # ``evaluate_sequence`` will be called by the surrounding harness,
    # but we sanity‑check the value here for debugging purposes.
    # (Comment out the line below if you prefer not to invoke it.)
    # print("final evaluate_sequence:", evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
