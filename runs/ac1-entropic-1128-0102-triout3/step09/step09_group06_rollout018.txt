"""
Search for a non‑negative step function minimizing evaluate_sequence.
The algorithm consists of three phases:

1. Projected Adam on a smoothed max (fast global search).
2. Short sub‑gradient descent on the exact max (fine‑tuning).
3. Aggressive hill‑climbing with incremental convolution updates
   that moves small amounts of mass from the most responsible
   coefficient of the current peak to the smallest coefficient.
   A back‑tracking line‑search on the step size is performed.
The whole routine respects the 1000 s time budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    # Soft‑max over the entries of conv
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    # gradient = 2 * (a ⋆ w)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]
    grad = 2.0 * corr[w_rev.shape[0] - 1 - np.arange(a.shape[0])]
    return grad

def _adam_optimize(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters (increasingly tight)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    for alpha in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self_fft(a)
            # smooth max value (optional for logging)
            # gradient
            grad = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)
            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve α={alpha:.0f} score={sc:.6f}")

    return best_a

def _subgrad_optimize(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0
    beta = 0.94
    step0 = 0.05
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve iter={it:,} score={cur_score:.6f}")
        # sub‑gradient corresponding to the current peak
        k = conv.argmax()
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] += a[j]
        g[j] += a[i]   # diagonal counted twice – correct for derivative of a_i a_j
        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a

def _hill_climb(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Aggressive local search that transfers a fraction of mass from the
    coefficient that contributes most to the current peak towards the
    smallest coefficient.  A simple back‑tracking line search on the step
    size is performed.  The convolution array is updated incrementally,
    so each trial costs O(n).  The routine keeps the overall best vector.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self_fft(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    rng = np.random.default_rng(0)

    # small function to recompute convolution from scratch (for stability)
    def recompute():
        return _conv_self_fft(a)

    while time.time() < deadline:
        it += 1
        cur_max = conv.max()
        k = conv.argmax()
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]   # the most responsible index

        # destination – smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # try decreasing fractions of src until an improvement is obtained
        delta = a[src] * 0.2   # start with 20 % of the source mass

        while delta > 1e-14:

            conv_tmp = conv.copy()
            conv_tmp[src:src + n] -= 2.0 * delta * a
            conv_tmp[dst:dst + n] += 2.0 * delta * a
            conv_tmp[2 * src] += delta * delta
            conv_tmp[2 * dst] += delta * delta
            conv_tmp[src + dst] -= 2.0 * delta * delta
            new_max = conv_tmp.max()
            if new_max < cur_max - 1e-15:
                # accept
                conv = conv_tmp
                a[src] -= delta
                a[dst] += delta
                cur_max = new_max
                score = 2.0 * n * cur_max
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hill‑climb iter={it:,} score={score:.6f}")
                improved = True
                break
            else:
                delta *= 0.5   # try a smaller step

        if not improved:

            src = rng.integers(0, n)
            if a[src] < 1e-14:

            dst = dst_candidates[a[dst_candidates].argmin()]
            delta = a[src] * 0.05
            conv_tmp = conv.copy()
            conv_tmp[src:src + n] -= 2.0 * delta * a
            conv_tmp[dst:dst + n] += 2.0 * delta * a
            conv_tmp[2 * src] += delta * delta
            conv_tmp[2 * dst] += delta * delta
            conv_tmp[src + dst] -= 2.0 * delta * delta
            new_max = conv_tmp.max()
            if new_max < cur_max - 1e-15:
                conv = conv_tmp
                a[src] -= delta
                a[dst] += delta
                cur_max = new_max
                score = 2.0 * n * cur_max
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm has three phases:

1. Projected Adam on a smoothed‑max objective.  This quickly finds a good
   region of the feasible simplex.

2. Short sub‑gradient descent on the exact max (the true objective) to
   polish the solution.

3. Aggressive hill‑climbing that transfers a fraction of the mass of the
   coefficient most responsible for the current convolution peak to the
   smallest coefficient.  A simple back‑tracking line‑search on the step
   size is performed, the convolution is updated incrementally (O(n)),
   and the best vector seen so far is kept.

The whole routine respects the 1000 s time budget and finally returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                     # soft‑max weights

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]
    grad = 2.0 * corr[w_rev.shape[0] - 1 - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> np.ndarray:
    """
    Projected Adam on the smoothed‑max objective.
    Returns the best vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α approaches the true max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self_fft(a)
            # smooth max value (only for logging)
            if t % 400 == 0:
                scaled = alpha * conv
                max_scaled = scaled.max()
                lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
                smooth_val = lse / alpha
                # (no need to use it further)

            # gradient of the smoothed max (including outer factor 2·n)
            grad = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # Adam update on the raw vector, then project onto the simplex
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)               # simple decay
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 800 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve α={alpha:.0f} "
                          f"score={sc:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – short sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    Performs a few thousand iterations with a decreasing step size.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0
    beta = 0.94
    step0 = 0.05

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient for the (single) peak
        k = conv.argmax()
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] += a[j]
        g[j] += a[i]          # diagonal counted twice – correct

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – aggressive hill‑climbing with incremental updates
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> np.ndarray:
    """
    Repeatedly moves a fraction of the coefficient that most contributes
    to the current peak towards the smallest coefficient.
    A back‑tracking line‑search on the step size is performed.
    The convolution array is updated incrementally (O(n)).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self_fft(a)

    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    rng = np.random.default_rng(42)

    # helper to recompute the full convolution (used periodically)
    def recompute():
        return _conv_self_fft(a)

    while time.time() < deadline:
        it += 1
        cur_max = conv.max()
        k = conv.argmax()

        # indices that participate in the current peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]

        # most responsible coefficient
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient different from src
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # try to move a fraction of src to dst; halve the fraction until an
        # improvement is observed.
        delta = a[src] * 0.20        # start with 20 % of the source mass
        improved = False
        while delta > 1e-14:
            # incremental update (copy‑on‑write)
            conv_tmp = conv.copy()
            conv_tmp[src:src + n] -= 2.0 * delta * a
            conv_tmp[dst:dst + n] += 2.0 * delta * a
            conv_tmp[2 * src] += delta * delta
            conv_tmp[2 * dst] += delta * delta
            conv_tmp[src + dst] -= 2.0 * delta * delta

            new_max = conv_tmp.max()
            if new_max < cur_max - 1e-15:
                # accept the move
                conv = conv_tmp
                a[src] -= delta
                a[dst] += delta
                cur_max = new_max
                score = 2.0 * n * cur_max
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hill‑climb iter={it:,} "
                          f"score={score:.6f}")
                improved = True
                break
            delta *= 0.5

        if not improved:
            # occasional random trial to escape possible stagnation
            src = rng.integers(0, n)
            if a[src] > 1e-14:
                dst = dst_candidates[a[dst_candidates].argmin()]
                delta = a[src] * 0.05
                conv_tmp = conv.copy()
                conv_tmp[src:src + n] -= 2.0 * delta * a
                conv_tmp[dst:dst + n] += 2.0 * delta * a
                conv_tmp[2 * src] += delta * delta
                conv_tmp[2 * dst] += delta * delta
                conv_tmp[src + dst] -= 2.0 * delta * delta
                new_max = conv_tmp.max()
                if new_max < cur_max - 1e-15:
                    conv = conv_tmp
                    a[src] -= delta
                    a[dst] += delta
                    cur_max = new_max
                    score = 2.0 * n * cur_max
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] hill‑climb random iter={it:,} "
                              f"score={score:.6f}")

        # periodic full recomputation to avoid drift from rounding errors
        if it % 5000 == 0:
            conv = recompute()

    return best_a

def _initial_candidate(n: int, previous: np.ndarray | None = None) -> np.ndarray:
    """Return a probability vector of length n (sum = 1)."""
    if previous is not None and previous.ndim == 1 and previous.size > 0:
        seq = previous.astype(float)
        if seq.shape[0] != n:
            # linear interpolation preserving total mass

            new_x = np.linspace(0.0, 1.0, n)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s > 0:
            return seq / s
    # fallback – uniform
    return np.full(n, 1.0 / n)
