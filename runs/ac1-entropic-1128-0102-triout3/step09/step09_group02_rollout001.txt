"""
Greedy‑only optimiser
~~~~~~~~~~~~~~~~~~~~~

The evaluator is scale‐invariant, therefore we fix the total mass to 1.
Starting from the best known construction (``height_sequence_1`` if it
exists, otherwise a uniform vector) we repeatedly perform a *targeted*
mass transfer:

* Find the index ``k`` where the self‑convolution is maximal.
* Choose a source index ``src`` that contributes the most to this peak.
* Choose a destination index ``dst`` that contributes the least to this
  peak (the “partner’’ of ``src`` in the reversed view of the peak).
* Move a small amount ``eps`` from ``src`` to ``dst`` – the amount is
  chosen analytically so that the current peak would (ideally) be reduced
  to the second‑largest convolution value.
* Update the convolution incrementally (O(n)) and recompute it exactly
  every few thousand moves to avoid drift.

The loop runs until the time budget (≈1000 s) is exhausted, keeping the
best sequence seen.  When the optimisation stagnates we add a tiny
random perturbation to escape shallow local minima.  The final vector is
renormalised to sum to 1 and returned as a plain ``list`` of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full a*a convolution using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Greedy reduction of the convolution peak
# ----------------------------------------------------------------------
def _greedy_reduce(a: np.ndarray,
                   deadline: float,
                   start_time: float,
                   recompute_every: int = 2000,
                   max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the element that
    contributes most to the current peak to an element that contributes
    almost nothing.  The convolution is updated incrementally; an exact
    recomputation is performed every ``recompute_every`` steps.
    Returns the best vector seen and its evaluation score.
    """
    n = a.shape[0]
    Lfft = _next_pow2(2 * n - 1)

    # exact convolution
    conv = _conv_self(a, Lfft)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    rng = np.random.default_rng()

    # helpers for “second largest’’ value
    def _second_max(arr: np.ndarray, idx_max: int) -> float:
        # set a copy to -inf at the maximum location and take max
        tmp = arr.copy()
        tmp[idx_max] = -np.inf
        return tmp.max()

    while time.time() < deadline and it < max_iters:
        it += 1

        # current peak
        k = int(conv.argmax())
        max_val = conv[k]

        # second peak (used for target epsilon)
        second_val = _second_max(conv, k)

        # indices that actually appear in this peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # contributions a[i] * a[k-i]
        contrib = a[idx] * a[k - idx]
        src = idx[np.argmax(contrib)]

        # partner values for each position in the peak (a[k-i])
        a_rev = np.zeros_like(a)
        a_rev[idx] = a[k - idx]

        # choose destination with the smallest partner value,
        # but avoid picking src itself
        cand = np.where(a_rev == a_rev.min())[0]
        cand = cand[cand != src]
        if cand.size == 0:
            # fallback: any index different from src
            dst = rng.integers(0, n - 1)
            while dst == src:
                dst = rng.integers(0, n - 1)
        else:
            dst = rng.choice(cand)

        # partner values for src and dst (may be outside range)
        partner_src = a[k - src] if 0 <= k - src < n else 0.0
        partner_dst = a[k - dst] if 0 <= k - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide how much mass to move
        if delta_a <= 0.0:
            # no guaranteed improvement – move a tiny fraction
            eps = min(a[src], 0.02 * a[src])
        else:
            # target to bring the peak down to the second peak
            eps_target = (max_val - second_val) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            # safeguard against too small steps
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # incremental update of the full convolution (ignoring O(eps²) term)
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old

        # update the vector itself
        a[src] -= eps
        a[dst] += eps
        # numerical safety
        a[a < 0] = 0.0

        # exact recomputation to remove accumulated drift
        if it % recompute_every == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # occasional random shake if we have been stuck for a while
        if it % 200_000 == 0:
            # small multiplicative noise, then renormalise
            noise = 0.01 * rng.standard_normal(n)
            a = np.maximum(a * (1.0 + noise), 0.0)
            total = a.sum()
            if total > 0:
                a /= total
            else:
                a = np.full(n, 1.0 / n)
            conv = _conv_self(a, Lfft)   # recompute after shake

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the allotted time.
    The optimiser consists mainly of a long greedy reduction of the
    self‑convolution peak (see the docstring above for details).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a short safety margin – the platform may kill the process a
    # little after the deadline.
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise the sequence.
    # Prefer the previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter.copy()
        else:
            a0 = None
    except Exception:
        a0 = None

    # target length – 4096 is a good trade‑off between quality and speed.
    n = 4096
    if a0 is not None:
        # interpolate the existing construction to the chosen size
        a0 = np.interp(
            np.linspace(0.0, 1.0, n),
            np.linspace(0.0, 1.0, a0.size),
            a0,
        )
    else:
        a0 = np.random.rand(n)

    # enforce non‑negativity and normalise to sum = 1 (the evaluation
    # function is scale‑invariant, so any normalisation works).
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Greedy reduction – the bulk of the budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, score_greedy = _greedy_reduce(
            best_seq,
            deadline,
            start_time,
            recompute_every=2000,
            max_iters=int(1e9),   # effectively “as many as time permits’’
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy best: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Convert to a plain Python list as required by the platform.
    return [float(x) for x in best_seq]
