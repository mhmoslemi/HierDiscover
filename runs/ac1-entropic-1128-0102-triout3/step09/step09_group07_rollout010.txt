"""
Improved step‑function optimisation.

The search proceeds in several stages:

1. **Warm‑up** – a short exponentiated‑gradient run on a smooth
   approximation of the max (log‑sum‑exp).  This quickly produces a
   decent distribution.

2. **Projected sub‑gradient descent** – uses the exact sub‑gradient of the
   ℓ∞‑norm of the self‑convolution (only the current maximising shift is
   considered, which makes the iteration cheap).  Momentum and a
   decreasing step‑size are employed.

3. **Large mass‑shifts** – an O(n) “optimal incremental refinement’’ that
   moves a fraction of mass from a source index (contributing heavily to the
   current maximal convolution entry) to a destination that contributes
   little.  The step size is chosen analytically as the biggest ε that
   still reduces the current maximum.

4. **Greedy refinement** – after the bulk optimisation we run a short
   greedy pass that examines a small set of promising source/destination
   pairs, evaluates the exact admissible ε for each pair and applies the
   best improvement.  Each greedy move is also O(n) but yields noticeably
   larger drops per iteration.

Multiple independent runs (different random seeds) are performed while
time permits; the best sequence seen is kept.  Finally the greedy pass is
run on the overall best sequence for the remaining time.

The function ``propose_candidate`` returns the final list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage‑1 : exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray, n: int,
                              deadline: float, start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    """Exponentiated‑gradient on a log‑sum‑exp approximation of the max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ---- smoothed‑max gradient ----
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑2 : projected sub‑gradient descent (exact max, single active index)
# ----------------------------------------------------------------------
def _subgradient_one_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a using the *single* maximal shift."""
    n = a.shape[0]
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g = np.zeros_like(a)
    g[i] += a[j]
    g[j] += a[i]
    return g


def _run_subgradient_descent(a0: np.ndarray, n: int,
                             deadline: float, start_time: float,
                             step0: float = 0.5,
                             beta: float = 0.94) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_one_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑3 : optimal incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray, n: int,
                                   deadline: float, start_time: float,
                                   eps_start: float = 0.15,
                                   eps_end: float = 0.001,
                                   rel_improve: float = 1e-12,
                                   recompute_every: int = 5000) -> np.ndarray:
    """
    Huge number of mass‑shifts. For a chosen source `src` and destination `dst`
    the effect on the whole convolution is linear in ε:
        Δconv = ε·s ,   s[t] = 2·(a[t‑dst] – a[t‑src]).
    The algorithm computes the maximal ε that still reduces the present
    maximum (solution of a tiny 1‑D LP) and applies the move.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end,
                       eps_start * (1.0 - elapsed / total_time))

        # ---- current maximal index ----
        k_max = int(np.argmax(conv))

        # ---- source index (high contribution to the maximal bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- destination index (prefer indices that appear little in the max bin) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- slope vector for the whole convolution ----
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # move would not decrease the current maximum
            continue

        # ---- largest admissible ε before another entry overtakes the max ----
        mask = s > s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                      # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we are allowed to take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ---- apply the move analytically (O(n)) ----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep numerical errors in check
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Stage‑4 : Greedy refinement (small exhaustive search of promising pairs)
# ----------------------------------------------------------------------
def _run_greedy_refinement(a0: np.ndarray, n: int,
                           deadline: float, start_time: float,
                           top_k: int = 10,
                           bottom_k: int = 10,
                           eps_fraction: float = 0.05,
                           rel_improve: float = 1e-12) -> np.ndarray:
    """
    Greedy pass that examines a limited set of source/destination pairs,
    computes the exact admissible ε for each pair and applies the best
    improvement.  Each iteration is O(k²·n) with k≈10, inexpensive enough
    for a few thousand moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        # current maximal bin
        k_max = int(np.argmax(conv))

        # source candidates: indices that heavily contribute to the max bin
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        if i_range.size > top_k:
            idx_top = np.argpartition(-contrib, top_k - 1)[:top_k]
            src_cands = i_range[idx_top]
        else:
            src_cands = i_range

        # destination candidates: smallest entries (avoid moving to itself)
        if n > bottom_k:
            idx_bot = np.argpartition(a, bottom_k)[:bottom_k]
            dst_cands = idx_bot
        else:
            dst_cands = np.arange(n)

        # pre‑compute the part of the slope that corresponds to removing mass from src
        best_pair = None
        best_new_max = max_val
        best_eps = 0.0

        for src in src_cands:
            if a[src] < 1e-12:
                continue
            # slope component for removing from src
            s_src = np.zeros(2 * n - 1, dtype=float)
            s_src[src:src + n] -= 2.0 * a

            for dst in dst_cands:
                if dst == src:
                    continue
                # full slope for this (src,dst) pair
                s = s_src.copy()
                s[dst:dst + n] += 2.0 * a

                s_max = s[k_max]
                if s_max >= 0.0:
                    continue  # would not lower the current maximum

                # compute admissible ε (same logic as in incremental refinement)
                mask = s > s_max
                if np.any(mask):
                    delta_conv = conv[k_max] - conv[mask]
                    delta_s = s[mask] - s_max
                    eps_cands = delta_conv / delta_s
                    eps_cands = eps_cands[eps_cands > 0.0]
                    eps_star = eps_cands.min() if eps_cands.size else np.inf
                else:
                    eps_star = np.inf

                eps_limit = a[src] * eps_fraction
                epsilon = min(eps_star, eps_limit)
                if epsilon <= 0.0:
                    continue

                new_max = (conv + epsilon * s).max()
                if new_max < best_new_max - rel_improve:
                    best_new_max = new_max
                    best_pair = (src, dst, s, epsilon)

        if best_pair is None:
            # nothing found – fall back to a tiny random move
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] < 1e-12:
                continue
            # a minimal move just to shake the system
            epsilon = min(a[src] * 0.001, 1e-6)
            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a
            conv += epsilon * s
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = conv.max()
            continue

        src, dst, s, epsilon = best_pair
        conv = conv + epsilon * s
        a[src] -= epsilon
        a[dst] += epsilon
        max_val = best_new_max

        if max_val < best_max - rel_improve:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: max={best_max:.8f}")

    return best_a


# ----------------------------------------------------------------------
# Whole‑search driver (multiple independent attempts)
# ----------------------------------------------------------------------
def _run_full_search(seed: int, n: int,
                     time_budget: float,
                     start_time: float) -> np.ndarray:
    """
    Execute the three main stages (warm‑up, sub‑gradient, incremental)
    within ``time_budget`` seconds and return the best sequence obtained.
    """
    rng = np.random.default_rng(seed)

    # initialise – either from the global starter or random
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter / starter.sum(), n)
        else:
            raise Exception
    except Exception:
        a0 = rng.random(n)
        a0 /= a0.sum()

    deadline = time.time() + time_budget

    # ------------------------------------------------------------------
    # Stage 1 – short exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    stage1_deadline = min(deadline, time.time() + min(15.0, 0.06 * time_budget))
    a_warm, _ = _run_exponential_gradient(
        a0,
        n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=4000,
    )
    a_curr = a_warm.copy()

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (most of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        stage2_deadline = time.time() + max(20.0, 0.70 * (deadline - time.time()))
        a_sub, _ = _run_subgradient_descent(
            a_curr,
            n,
            stage2_deadline,
            start_time,
            step0=0.5,
            beta=0.94,
        )
        a_curr = a_sub.copy()

    # ------------------------------------------------------------------
    # Stage 3 – optimal incremental refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_ref = _run_optimal_incremental_refine(
            a_curr,
            n,
            deadline,
            start_time,
            eps_start=0.15,
            eps_end=0.001,
        )
        a_curr = a_ref.copy()

    # final normalisation
    a_curr = np.maximum(a_curr, 0.0)
    s = a_curr.sum()
    if s > 0:
        a_curr /= s
    else:
        a_curr = np.full(n, 1.0 / n)

    return a_curr.tolist()


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    Multiple independent searches are performed while time permits;
    the best result is further polished by a greedy refinement pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee we return before timeout
    deadline = start_time + budget_s - 0.5

    # problem size – 4800 works well, but we keep it as a parameter for flexibility
    final_n = 4800

    best_seq = None
    best_score = float("inf")
    attempt = 0

    # ------------------------------------------------------------------
    # Run several independent attempts while we have time.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # keep ~30 s for final greedy pass
        remaining = deadline - time.time()
        # each attempt gets at most 300 s or 70 % of the remaining time
        attempt_budget = min(300.0, 0.70 * remaining)
        seq = _run_full_search(seed + attempt, final_n, attempt_budget, start_time)

        # quick evaluation (exact objective, no extra caps)
        a_arr = np.asarray(seq, dtype=float)
        conv = _conv_self(a_arr)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[{time.time() - start_time:.2f}s] attempt {attempt} new best: "
                  f"score={score:.6f}")

        attempt += 1

    # ------------------------------------------------------------------
    # Final greedy refinement on the overall best sequence.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        leftover = deadline - time.time()
        if leftover > 5.0:
            a_arr = np.asarray(best_seq, dtype=float)
            a_refined = _run_greedy_refinement(
                a_arr,
                final_n,
                deadline,
                start_time,
                top_k=12,
                bottom_k=12,
                eps_fraction=0.05,
            )
            conv = _conv_self(a_refined)
            score = 2.0 * final_n * conv.max()
            if score < best_score:
                best_seq = a_refined.tolist()
                best_score = score
                print(f"[{time.time() - start_time:.2f}s] greedy final improve: "
                      f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution
        best_seq = [1.0 / final_n] * final_n
    else:
        best_seq = [max(0.0, float(x)) for x in best_seq]
        total = sum(best_seq)
        if total > 0:
            best_seq = [x / total for x in best_seq]
        else:
            best_seq = [1.0 / final_n] * final_n

    return best_seq
