Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The code follows the successful three‑stage strategy of the reference
solution but makes three systematic changes aimed at breaking the
current ~1.512 bound:

1. **Extended LP balancing** – the linear‑programming phase is
   iterated until convergence (or the time budget runs out) instead of
   stopping after a fixed small number of iterations.  After each LP
   solve we perform a line‑search over the mixing coefficient `t` to
   obtain the best convex combination of the current vector and the
   LP‑partner.

2. **More aggressive sub‑gradient phase** – the momentum decay and step‑size
   schedules are tuned and the loop runs as long as time permits
   (instead of a hard cut after a few seconds).

3. **Random hill‑climbing polish** – a simple stochastic local‑search
   that repeatedly moves a tiny amount of mass from one coordinate to
   another and accepts the move only if it improves the objective.
   This often finds tiny refinements that gradient‑based methods miss.

All stages respect the global time budget (default 1000 s).  The
returned list is a probability vector (sum = 1) and therefore can be
fed directly to ``evaluate_sequence``.

import time
import math
import random
import numpy as np

# -------------------------------------------------------------
# Linear‑algebra helpers (no external dependencies)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two (used for FFT‑based convolution)."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    For all k attaining the maximum we add the contribution of a to the
    gradient; the result is averaged over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _score(a: np.ndarray) -> float:
    """Objective value for a (assumes sum(a)=1)."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())

                print("[LP] zero‑sum partner – exiting LP stage.")
                break

            g = b_vec / sum_b          # partner normalised to the simplex

            # -------------------------------------------------
            # line‑search for the mixing coefficient t ∈ [0,1]
            # -------------------------------------------------
            candidates = []
            for t in np.linspace(0.0, 1.0, 11):
                cand = (1.0 - t) * a + t * g
                cand = _project_simplex(cand)
                cand_score = _score(cand)
                candidates.append((cand_score, cand, t))

            # also try a few random t’s for extra robustness
            for _ in range(5):
                t = random.random()
                cand = (1.0 - t) * a + t * g
                cand = _project_simplex(cand)
                cand_score = _score(cand)
                candidates.append((cand_score, cand, t))

            cand_score, cand_vec, best_t = min(candidates, key=lambda x: x[0])

            if cand_score < best_score - 1e-7:
                a = cand_vec
                best_score = cand_score
                best_seq = a.copy()
                no_improve_counter = 0
                print(f"[{time.time() - start_time:6.2f}s] LP iter {iteration:<3d} "
                      f"t={best_t:.3f} improve → {cand_score:.6f}")
            else:
                no_improve_counter += 1

            # stop early if no improvement for 5 consecutive LP solves
            if no_improve_counter >= 5:
                break

    # =========================================================
    # Stage 2 – projected sub‑gradient descent (momentum)
    # =========================================================
    sub_deadline = deadline - 2.0          # keep a couple of seconds for polishing
    if time.time() < sub_deadline:
        a = best_seq.copy()
        momentum = np.zeros_like(a)
        it = 0
        while time.time() < sub_deadline:
            it += 1
            conv_a = _conv_self(a)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

            grad = _subgradient_max(a, conv_a)
            # momentum decay a little faster than the reference code
            momentum = 0.90 * momentum + 0.10 * grad
            step = 0.12 / math.sqrt(it)         # slightly larger step at the start
            a = _project_simplex(a - step * momentum)

    # =========================================================
    # Stage 3 – exponentiated‑gradient (mirror descent)
    # =========================================================
    if time.time() < deadline:
        a = best_seq.copy()
        it = 0
        while time.time() < deadline:
            it += 1
            conv_a = _conv_self(a)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

            grad = _subgradient_max(a, conv_a)
            eta = 0.18 / math.sqrt(it)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    # =========================================================
    # Stage 4 – random hill‑climbing polish (optional, time‑limited)
    # =========================================================
    polish_deadline = deadline - 0.1   # leave a tiny buffer before return
    if time.time() < polish_deadline:
        a = best_seq.copy()
        it = 0
        while time.time() < polish_deadline:
            it += 1
            # pick two distinct indices
            i, j = random.sample(range(n), 2)
            # propose moving a small chunk from i to j
            eps = min(a[i], 0.001 * random.random())
            if eps <= 0.0:
                continue
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps
            # projection is not needed (mass is conserved and non‑negative)
            cand_score = _score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                # occasional debug output (once per thousand accepted moves)
                if it % 1000 == 0:
                    print(f"[{time.time() - start_time:6.2f}s] polish iter {it:<6d} "
                          f"improve → {cand_score:.6f}")

    # -----------------------------------------------------------------
    # Final normalisation (ensuring sum = 1) and return a plain list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
