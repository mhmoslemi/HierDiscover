"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of three stages

1. **Diverse starts** – a few handcrafted vectors (uniform, exponential,
   Gaussian, Dirichlet) and the previously discovered best sequence
   (``height_sequence_1`` if it exists).  Each start is normalised to a
   simplex (sum = 1).

2. **Exponentiated‑gradient polishing** – a fast mirror‑descent on the
   true objective 2·n·max(a*a).  Only the active shift (the one attaining
   the current maximum) contributes to the sub‑gradient, which makes the
   update cheap and allows millions of iterations in the allotted time.

3. **Simulated‑annealing refinement** – a stochastic local‑search that
   repeatedly transfers a random fraction of mass from a “source” index
   to a “target’’ index.  The move is accepted if it improves the score,
   otherwise it is accepted with probability exp(Δ/T).  The temperature
   T and the transfer fraction are cooled exponentially, giving a good
   balance between exploration and exploitation.

Both optimisation phases respect the global time budget (default 1000 s)
and keep the best vector seen so far.  The final result is clipped to
non‑negative values and re‑normalised before being returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution a*a via FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _max_conv(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _conv_self(a, fft_len).max()


def _score_from_vec(a: np.ndarray, fft_len: int) -> float:
    """Exact objective used by ``evaluate_sequence`` (sum is 1)."""
    # a is already non‑negative and its sum equals 1
    n = a.shape[0]
    return 2.0 * n * _max_conv(a, fft_len)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on f(a)=max_k (a*a)[k] using the sub‑gradient from the
    single shift attaining the maximum.  The iterate stays on the
    probability simplex by a multiplicative update followed by normalisation.
    """
    a = np.maximum(a0, 1e-15)          # avoid exact zeros for the log‑step
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score_from_vec(a, fft_len)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # index of the active shift
        k = int(conv.argmax())

        # Determine the indices that contribute to this shift
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i

        # Sub‑gradient for the active shift (only the terms that appear)
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]          # each term a_i*a_j appears twice (i↔j)

        # Diminishing step size
        eta = eta0 / math.sqrt(t)

        # Exponentiated (multiplicative) update
        a = a * np.exp(-eta * g)

        # Project back onto the simplex (simple normalisation works because a>=0)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Periodic exact evaluation (cheap because we already have the score)
        if t % 500 == 0:
            cur_score = _score_from_vec(a, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _sa_optimize(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 fft_len: int,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that moves a random fraction of mass from one
    index to another.  Acceptance follows a temperature schedule that
    decays exponentially.
    """
    cur_a = a0.copy()
    n = cur_a.shape[0]
    cur_score = _score_from_vec(cur_a, fft_len)

    best_a = cur_a.copy()
    best_score = cur_score

    total_time = deadline - start_time
    # temperature schedule
    T_start = 2e-2
    T_end = 1e-7
    # transfer fraction schedule (large moves early, fine tuning later)
    frac_start = 0.4
    frac_end = 0.01

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * ((T_end / T_start) ** frac)               # exponential cooling
        max_transfer_frac = frac_end + (frac_start - frac_end) * (1.0 - frac)

        # bias source selection to large entries (prob proportional to a_i)
        src = rng.choice(n, p=cur_a)                # larger values are more likely to donate
        # uniform destination different from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # amount to transfer
        max_transfer = cur_a[src] * max_transfer_frac
        if max_transfer < 1e-15:
            continue
        delta = rng.random() * max_transfer

        # propose new vector
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # evaluate
        new_score = _score_from_vec(new_a, fft_len)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA new best: score={new_score:.6f}")

        it += 1

    return best_a, best_score

# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector of length n with concentration ``conc``."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6), normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation that preserves the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    previous best bound.  The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a small pool of diverse initial points
    # ------------------------------------------------------------------
    starts = []

    # 1) use the previously known best if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted starts
    starts.append(np.full(n, 1.0 / n))                     # uniform
    starts.append(_exponential_start(n, r=0.9995))        # very mild decay
    starts.append(_exponential_start(n, r=0.985))         # stronger decay
    starts.append(_gaussian_start(n))                     # bell‑shaped
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))  # moderate concentration
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))  # spiky

    # pick the best raw start
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _score_from_vec(a0, fft_len)   # exact score (sum=1)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient polishing (≈30 % of time)
    # ------------------------------------------------------------------
    eg_deadline = start_time + budget_s * 0.30
    best_seq, eg_score = _expgrad_optimize(best_seq,
                                           eg_deadline,
                                           start_time,
                                           fft_len,
                                           eta0=0.6)
    if eg_score < best_score:
        best_score = eg_score
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after EG score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement (remainder of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, sa_score = _sa_optimize(best_seq,
                                          deadline,
                                          start_time,
                                          fft_len,
                                          rng)
        if sa_score < best_score:
            best_score = sa_score
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after SA score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final reported score (for logging – uses the official checker)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
