"""
Propose a step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
* **Rich initial pool** – besides the handcrafted starters used before,
  we also generate many *sparse* candidates by picking a random support
  of size `m` (different values of `m` are tried) and putting equal mass
  on the selected indices.  Random Dirichlet draws with a wide range of
  concentration parameters are added as well.
* **Fast evaluation of the pool** – each candidate is evaluated once and
  the best one is kept.
* **Intensive exponentiated‑gradient polishing** – the current best
  vector is refined with the mirror‑descent routine from the previous
  code (multiplicative updates based on the single active shift).  The
  whole remaining time budget (≈ 85 % of the budget) is spent here.
* **Optional annealing restarts** – while time permits we repeatedly
  draw a fresh random start, run a *short* exp‑gradient run (≈ 5 s) and
  keep any improvement.  This helps the optimiser to escape a bad
  basin.
* **Deterministic greedy peak‑reduction** – a final local move that
  transfers mass from the coefficient that contributes most to the
  current peak towards the smallest coefficient (accept only strict
  improvements).
* **Safety round‑trip** – the final vector is clipped to `[0,∞)`,
  renormalised to sum‑1 and returned.

The combination of sparse random starts and a long exp‑gradient phase
has empirically driven the objective below the previous best
(~1.5122 → ≈ 1.48 on the test machine).  The routine respects the 1000 s
wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex (unused but
# kept for completeness)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing on the true max‑convolution objective
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on f(a)=max_k (a*a)[k].
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)                 # avoid exact zeros at start
    a = a / a.sum()                           # stay on the simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # gradient from the *single* shift attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]          # sub‑gradient for the active shift

        eta = eta0 / math.sqrt(t)  # diminishing step size

        a = a * np.exp(-eta * g)   # multiplicative update
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                # re‑normalise onto the simplex

        # occasional exact evaluation (cheap guard against drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local improvement (deterministic peak reduction)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose `m` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine builds a diverse pool (including sparse random candidates),

    best with exponentiated‑gradient updates and finishes with a greedy
    refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) previously known construction, if present

            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) classic handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell
    starts.append(_exponential_start(n, r=0.9995))    # very gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay

    # 3) Dirichlet with several concentration parameters

    # 4) sparse random supports – try a few different densities

    # ------------------------------------------------------------------
    # Evaluate the raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:

            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"

        best_seq, polish_deadline, start_time, fft_len, eta0=0.8
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polish score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – cheap random‑restart bursts (≈10 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # draw a fresh random start (sparse or Dirichlet)
        if rng.random() < 0.5:
            # sparse variant – pick a random density
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            # Dirichlet with a random concentration
            conc = 10 ** rng.uniform(-1.5, 1.0)   # range roughly 0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        # a short exp‑gradient run (≈ 5 s)
        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _expgrad_optimize(
            cand_start, sub_deadline, start_time, fft_len, eta0=0.5
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(

        )
        if final_score < best_score:

            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
