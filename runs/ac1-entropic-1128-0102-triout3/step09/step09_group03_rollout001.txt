def _fine_greedy_refine(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    sum_a = a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len).max() / (sum_a**2)
    step_mult = 0.02  # fraction initial

    while time.time() < deadline:
        conv = _conv_self(a, fft_len)
        idx_max = conv.argmax()
        cur_max = conv[idx_max]
        # get indices that contribute
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = idx_max - i_idx
        prod = a[i_idx] * a[j_idx]
        src = i_idx[prod.argmax()]
        # choose destination index with minimal a (excluded src)
        dst = np.argmin(a)
        if dst == src:
            # find second smallest
            order = np.argsort(a)
            for d in order:
                if d != src:
                    dst = d
                    break
        # epsilon fraction
        eps = step_mult * a[src]
        if eps <= 1e-12:
            # reduce step_mult perhaps
            step_mult *= 0.5
            continue
        # try move
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # compute new max using conv (maybe update approx)
        new_conv = _conv_self(a_new, fft_len)
        new_max = new_conv.max()
        # compute score
        new_score = 2.0 * n * new_max / (sum_a**2)  # sum unchanged
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] fine greedy improved to {best_score:.6f}")
        if new_score < (2.0 * n * cur_max / (sum_a ** 2)) - 1e-12:
            # accept improvement
            a = a_new
            # possibly increase step_mult gradually
            step_mult = min(step_mult * 1.05, 0.2)
        else:
            # reject: reduce epsilon
            step_mult *= 0.7
            if step_mult < 1e-5:
                step_mult = 0.02
        # optional: after some iterations, maybe random jitter to escape local plateau
    return best_a, best_score

"""
search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm proceeds in three stages:

1. **Diverse starts** – uniform, exponential, Gaussian and Dirichlet draws,
   plus the previously known best (``height_sequence_1``) if available.
2. **Coarse optimisation** – a few Adam iterations on a smooth
   log‑sum‑exp surrogate of the max‑convolution (quickly leaves bad regions).
3. **Fine‑grained greedy refinement** – deterministic moves that transfer a
   small fraction of the coefficient contributing most to the current peak
   towards the smallest coefficient.  The step size adapts automatically and
   the loop runs until the time budget expires.

All work is done on the probability simplex (sum = 1); because the objective
is scale‑invariant this does not affect the final score.  The routine respects
the supplied ``budget_s`` (default 1000 s) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_a(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast (exact) objective value for a probability vector *a*."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Adam optimisation of a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of max(conv). Returns (value, conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """A short Adam run on the smoothed surrogate."""
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule – progressively larger α (more exact)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 1200

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate (exactly as in the original code)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 150 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score

# Exponentiated‑gradient optimisation on the true max‑convolution
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the exact objective."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # sub‑gradient from the active shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = a / a.sum()   # stay on the simplex

        # occasional exact evaluation guard
        if t % 1000 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def _fine_greedy_refine(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Very small deterministic moves that continuously shave the peak.
    Each iteration moves a fraction of the coefficient that most contributes
    to the current max‑convolution towards the smallest coefficient.
    The step‑size adapts automatically.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    # initialise best record
    cur_score = _score_from_a(a, n, fft_len)
    best_a = a.copy()
    best_score = cur_score

    # initial step fraction (relative to the source entry)
    step_frac = 0.02

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, fft_len)
        k_max = int(conv.argmax())
        cur_max = conv.max()
        # indices contributing to this shift
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination – smallest coefficient (different from src)
        dst = np.argmin(a)
        if dst == src:
            # pick the second smallest
            order = np.argsort(a)
            for d in order:
                if d != src:
                    dst = d
                    break

        eps = step_frac * a[src]
        if eps < 1e-15:
            # nothing sensible to move – shrink step and continue
            step_frac = max(step_frac * 0.5, 1e-4)
            continue

        # propose the move
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # compute new score (sum unchanged, so we reuse the helper)
        new_score = _score_from_a(a_new, n, fft_len)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] fine greedy iter {it:,} → best {best_score:.6f}"
            )

        if new_score < cur_score - 1e-15:
            # accept improvement and modestly increase step size
            a = a_new
            cur_score = new_score
            step_frac = min(step_frac * 1.03, 0.2)
        else:
            # reject – shrink the step fraction
            step_frac = max(step_frac * 0.7, 1e-5)

        # occasional random jitter to escape tiny plateaus
        if it % 5000 == 0:
            sigma = 0.001 * a.max()
            jitter = np.random.default_rng().normal(0.0, sigma, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            cur_score = _score_from_a(a, n, fft_len)

    return best_a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration *conc*."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑1."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    return up / s if s > 0 else np.full_like(up, 1.0 / new_n)
