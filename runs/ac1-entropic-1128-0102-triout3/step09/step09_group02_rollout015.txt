"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The optimiser works in three stages:

1.  Initialise from a few deterministic shapes (uniform, linear,
    sqrt‑law, exponential) and keep the one with the smallest
    self‑convolution peak after a very short mirror‑descent run.

2.  Long mirror‑descent (exponentiated‑gradient) on the true
    max‑convolution.  This directly reduces the current peak
    ``max(a*a)`` while keeping the vector on the probability simplex.

3.  Deterministic greedy “peak‑flattening”.  Repeatedly move a small
    amount of mass from the index that contributes most to the current
    maximum to an index that contributes least, updating the convolution
    incrementally (O(n) per move).  This stage is able to shave the last
    few 10⁻³ off the objective.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback – shouldn't happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that directly minimises
    ``max(conv)``.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # --- gradient of the active (max) convolution index ---
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # d/d a_i of conv[k]

        # --- mirror‑descent update (multiplicative) ---
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # --- renormalise onto the simplex (exact sum = 1) ---
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least,
    updating the full convolution incrementally.  A full recomputation
    is performed every few thousand steps to remove numerical drift.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between flexibility and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initialise – try a few simple deterministic shapes and keep the best.
    # ------------------------------------------------------------------
    initial_candidates = []

    # 1) previously discovered construction (if available)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            interp = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            initial_candidates.append(interp)
    except Exception:
        pass

    # 2) uniform distribution
    initial_candidates.append(np.full(n, 1.0 / n))

    # 3) linear decreasing (a_i ∝ n‑i)
    lin = np.arange(n, 0, -1, dtype=float)
    initial_candidates.append(lin / lin.sum())

    # 4) sqrt‑law (a_i ∝ 1/√(i+1))
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    initial_candidates.append(sqrt_shape / sqrt_shape.sum())

    # 5) exponential decay (q close to 1)
    q = 0.9995
    exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
    initial_candidates.append(exp_shape / exp_shape.sum())

    best_seq = None
    best_score = float("inf")

    # very short exp‑gradient runs to pick a promising start
    for cand in initial_candidates:
        if time.time() >= deadline:
            break
        temp_deadline = time.time() + min(4.0, (deadline - time.time()) * 0.04)
        seq, scr = _run_exp_gradient(cand, n, temp_deadline, start_time, eta0=0.8)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init improve: score={scr:.6f}")

    # fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (≈60 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.60
        seq, scr = _run_exp_gradient(best_seq, n, phase2_deadline, start_time,
                                     eta0=0.5)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] long exp‑grad improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy reduction (use all time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=10_000_000,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny random‑restart polishing if a few seconds remain.
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # small Gaussian perturbation
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()

        # short mirror‑descent burst
        tmp_deadline = time.time() + 2.0
        seq, scr = _run_exp_gradient(pert, n, tmp_deadline, start_time,
                                     eta0=0.6)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the objective is homogeneous, any scaling works)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(x) for x in best_seq]
