import scipy.optimize

def _lp_step(a, M, time_limit):
    n = len(a)
    # Build sparse matrix A
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((2*n - 1, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = a  # vector assignment
    A = A.tocsr()
    c = -np.ones(n)  # minimize -sum => maximize sum
    bounds = [(0, None)] * n
    # set tolerance, maybe use high precision
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                                    bounds=bounds, method='highs',
                                    options={'disp': False, 'time_limit': time_limit})
    except Exception as e:
        print("LP solve error:", e)
        return None
    if res.success:
        return res.x
    else:
        print("LP not successful:", res.message)
        return None
```

"""
Hybrid linear‑program (LP) equalisation → up‑sample → sub‑gradient → pair‑annealing.

The idea:
1. Work on a modest size (e.g. 512).  Given a current step‑function a,
   we compute M = max(conv(a,a)).  Then we solve a * b ≤ M (entrywise)
   with b ≥ 0 and maximise Σ bᵢ.  The LP pushes mass into the “gaps’’ of
   the convolution without raising the peak.  Repeating a convex
   combination a←(1‑t)·a + t·b (t≈0.5) quickly flattens the self‑convolution.
2. Interpolate the coarse result to the target length 4096.
3. Polish the up‑sampled vector with a classic projected sub‑gradient
   descent (same as the reference code) and a targeted simulated‑annealing
   step that moves mass away from the dominant convolution entry.
4. Return the normalised non‑negative vector as a plain Python list.

All helper routines are top‑level; the only external dependency beyond
NumPy is SciPy (linprog & sparse matrices), which is available in the
execution environment.  The algorithm respects the 1000 s budget and
prints intermediate improvements.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Linear‑algebra helpers (identical to the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The normalisation Σa=1 is assumed."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent (from the reference code)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a simple momentum scheme and a 1/√t step‑size decay.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the index achieving the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Pair‑annealing (from the reference code)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 180_000):
    """
    Simulated annealing that repeatedly attacks the two indices that
    contribute most to the current maximal convolution coefficient.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # maximal convolution index
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each admissible i
        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose a small mass shift
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)                     # maximise Σb → minimise -Σb
        bounds = [(0, None)] * n

        # Solve the LP (highs is fast for this size)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'max_iter': 2000})
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP exception: {e}")
            break

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP not successful: {res.message}")
            break

        b = res.x
        if b.sum() > 0:
            b = b / b.sum()
        else:
            b = np.full_like(b, 1.0 / n)

        # convex combination (t=0.5 works well in practice)
        t = 0.5
        a = (1.0 - t) * a + t * b
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        # track improvement on the coarse problem
        conv_new = _conv_self(a, L)
        score = _score_from_conv(conv_new, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP improve: iter={lp_iter}, "
                  f"score={score:.6f}")

        lp_iter += 1
        if lp_iter >= 30:                      # enough for a good coarse shape
            break

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to the target length
    # ------------------------------------------------------------------
    final_n = 4096
    a_up = np.interp(np.linspace(0, 1, final_n),
                     np.linspace(0, 1, coarse_n),
                     best_a)
    a_up = np.maximum(a_up, 0.0)
    a_up = a_up / a_up.sum()

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent on the full problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # recompute the exact score for the up‑sampled vector
        L_full = _next_pow2(2 * final_n - 1)
        cur_score = _score_from_conv(_conv_self(a_up, L_full), final_n)

        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.6 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            a_up,
            final_n,
            phase2_deadline,
            start_time,
            best_score_initial=cur_score,
            max_iters=None)

        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve after LP: "
                  f"score={score_sub:.6f}")
        else:
            best_seq = a_up.copy()
    else:
        best_seq = a_up.copy()

    # ------------------------------------------------------------------
    # 4️⃣  Targeted pair‑annealing for the final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=200_000)

        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Optional diagnostics (can be removed in the final submission)
    try:
        # evaluate_sequence is supplied in the execution environment
        from __main__ import evaluate_sequence            # type: ignore
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
