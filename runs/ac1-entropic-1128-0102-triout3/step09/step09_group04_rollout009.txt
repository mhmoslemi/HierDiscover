def _adam_optimize(a0, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2=0.999

    t = 0

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a hybrid of three simple ideas that run within the
global time budget:

1.  Start from the best known construction (or the uniform vector) and
    normalise it to the probability simplex.

2.  Run a long Adam‑style projected optimisation on the true
    non‑smooth objective.  The sub‑gradient of the peak of the
    self‑convolution is computed analytically; after each Adam step the
    iterate is projected back onto the simplex.

3.  When the time budget is almost exhausted a cheap stochastic‑annealing
    “mass‑transfer” neighbourhood search is performed.  It repeatedly
    moves a tiny amount of probability mass from one coordinate to
    another and accepts the move if it improves the objective (or with
    a small Metropolis probability that decreases with time).  This
    phase helps to escape the plateaux that Adam alone often stalls on.

The routine respects the 1000 s budget, prints progress, and finally
returns a plain list of floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (top‑level – no closures)
# -------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    n = v.shape[0]
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All components are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) via real FFT.
    The result is a 1‑D ``np.ndarray`` of dtype float.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of  max(conv)  w.r.t. a (under the constraint a ≥ 0).
    For each index k attaining the maximum we add the contribution of a
    to the gradient; the result is averaged over the number of active
    indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    # vector of all indices once, then we reuse it
    idx = np.arange(n)

    for k in max_inds:
        # For a given k the contributing pairs satisfy i + j = k
        # i ranges over max(0, k-(n-1)) … min(n-1, k)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i = idx[i_low:i_high + 1]
        j = k - i
        # accumulate 2 * a_j for every i in the active set
        grad[i] += 2.0 * a[j]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (scale‑invariant version):
    2 * n * max(conv(a,a))   with ``a`` already on the simplex.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _random_simplex(n: int) -> np.ndarray:
    """
    Return a random point on the probability simplex using the
    exponential (Gamma(1,1)) distribution.
    """
    x = -np.log(np.random.rand(n) + 1e-12)   # avoid log(0)
    return x / x.sum()


# -------------------------------------------------------------
# Optimisation primitives
# -------------------------------------------------------------


def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the simplex.
    Returns the best vector found and its objective value.
    """
    a = start_a.copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                     # initial learning‑rate constant

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        grad = _subgradient_max(a, conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays like 1/√t (similar to the original code)
        lr = lr0 / math.sqrt(it)

        # Adam step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # Project back onto the simplex
        a = _project_simplex(a)

        # Evaluate
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter {it:<6d} → {cur_score:.6f}")

    return best_a, best_score


def _simulated_annealing(start_a: np.ndarray,
                         start_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap neighbourhood search that repeatedly transfers a tiny
    amount of probability mass between two coordinates.
    """
    a = start_a.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    it = 0
    # Initial temperature proportional to the current objective
    T0 = 0.02 * start_score

    while time.time() < deadline:
        it += 1
        # pick two distinct indices uniformly at random
        i, j = np.random.choice(n, size=2, replace=False)

        # amount to move – a random fraction (0–10 %) of the donor mass
        eps = a[i] * (np.random.rand() * 0.1)
        if eps == 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # sum is preserved; all entries stay non‑negative by construction

        new_score = _objective(a_new)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            # temperature schedule – roughly T ∝ 1/√it
            T = T0 / math.sqrt(it)
            accept = np.random.rand() < math.exp(-(new_score - cur_score) / T)

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA  iter {it:<6d} → {new_score:.6f}")

    return best_a, best_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows the three‑stage
    hybrid (Adam + simulated annealing) strategy described in the module
    docstring and respects the supplied time budget (seconds).  The
    returned value is a plain ``list`` of ``float``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin (the caller will also reserve ~0.5 s)
    deadline = start_time + budget_s - 0.5

    # ---------------------------------------------------------
    # Choose the problem size.  2048 is a good compromise between
    # expressiveness and speed of the FFT‑based convolution.
    # ---------------------------------------------------------
    n = 2048

    # ---------------------------------------------------------
    # Initialise -----------------------------------------------------------------
    # Start from the previously discovered construction if it exists,
    # otherwise use the uniform distribution.
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                if tmp.size != n:
                    # Linear interpolation to the required length
                    old_x = np.linspace(0.0, 1.0, tmp.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    tmp = np.interp(new_x, old_x, tmp)
                a0 = tmp
            else:
                a0 = np.full(n, 1.0 / n)
        except Exception:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # Project onto the simplex (the optimiser works in this space)
    a0 = _project_simplex(a0)

    best_seq = a0.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # ---------------------------------------------------------
    # Stage 1 – Adam optimisation (most of the budget)
    # ---------------------------------------------------------
    # Allocate roughly 70 % of the remaining time to Adam.
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.70 * remaining
    a_opt, score_opt = _adam_optimize(best_seq, adam_deadline, start_time)

    if score_opt < best_score:
        best_score = score_opt
        best_seq = a_opt.copy()

    # ---------------------------------------------------------
    # Stage 2 – Stochastic annealing (use the rest of the time)
    # ---------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = deadline
        a_sa, score_sa = _simulated_annealing(best_seq,
                                              best_score,
                                              sa_deadline,
                                              start_time)
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()

    # ---------------------------------------------------------
    # Final normalisation – protect against numerical drift
    # ---------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")

    # Return a plain Python list (exactly the format expected by the
    # evaluation harness).
    return [float(x) for x in best_seq]
