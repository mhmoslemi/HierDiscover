"""
Propose a non‑negative step function (a probability vector) that minimises
``evaluate_sequence``.  

The algorithm consists of three distinct phases

1️⃣  **Parameter sweep** – we generate a few hundred simple families
    (truncated exponentials, power laws, linear ramps) and keep the
    best one (the evaluation is cheap because ``evaluate_sequence`` is
    O(n²) with n=4800).

2️⃣  **Linear‑program refinement** – for a current candidate *f* we
    compute M = max (f*f).  The LP

        maximise   Σ g_i
        subject to (f*g)[k] ≤ M  for all shifts k
                    g_i ≥ 0

    is linear in *g* (the convolution matrix is a banded Toeplitz
    matrix).  Solving it yields a vector *g* with a larger ℓ₁‑norm
    while respecting the same peak *M*.  After normalising *g* we mix
    it with *f* (f←(1‑t)f+t g, t≈0.1).  Repeating a handful of times
    drives the peak down dramatically.

3️⃣  **Mirror‑descent polishing** – we finish with the exponentiated‑gradient
    routine from the previous state‑of‑the‑art (a few hundred thousand
    iterations, decreasing step‑size).  This removes any remaining
    irregularities.

All phases respect the overall 1000 s budget.  The final returned list
contains non‑negative numbers; the sum is 1 (the scale does not affect
the score because ``evaluate_sequence`` is homogeneous).
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Families of parametrised starts
# ----------------------------------------------------------------------


def _exp_family(n: int, r: float) -> np.ndarray:
    """Truncated geometric distribution (∑ a_i = 1)."""
    i = np.arange(n, dtype=float)
    # a_i ∝ r**i, normalised to sum 1
    a = r ** i
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def _power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def _linear_ramp(n: int) -> np.ndarray:
    """Decreasing linear ramp (a_i = n-i)."""
    i = np.arange(n, dtype=float)
    a = n - i
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eps_log: float = 1e-16,
) -> tuple[np.ndarray, float]:
    """Few Adam steps on the smoothed max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # schedule – slowly make the surrogate sharper
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    iters_per_alpha = 1500

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # map logits to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, _ = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * _conv_self(a, fft_len)
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score

def _expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on f(a)=max_k (a*a)[k]."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = a / a.sum()                      # projection onto simplex
        if t % 500 == 0:                     # occasional exact check
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (convex combination step)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]   (linear convolution, zero padding)
    Every column i contains the constant value f[i] on a diagonal of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=float)
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        # column i contributes fi to rows i … i+n‑1
        A[i : i + n, i] = fi
    return A.tocsr()


def _lp_refine(
    f: np.ndarray,
    fft_len: int,
    deadline: float,
    start_time: float,
    max_iters: int = 15,
    mix_frac: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement steps.
    Returns the best sequence seen and its exact score.
    """
    n = f.shape[0]
    best_f = f.copy()
    conv = _conv_self(f, fft_len)
    best_M = conv.max()
    best_score = 2.0 * n * best_M

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # matrix for convolution f * g
        A = _build_conv_matrix(f)
        b = np.full(2 * n - 1, best_M)

        # maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(n)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # fallback: stop refining
            print(f"[{time.time() - start_time:6.2f}s] LP failed at iter {it+1}")
            break

        g0 = res.x
        if g0.sum() == 0.0:
            break
        g = g0 / g0.sum()          # normalise to the simplex

        # convex combination
        f = (1.0 - mix_frac) * f + mix_frac * g
        f = _project_simplex(f)    # numerical safety

        conv = _conv_self(f, fft_len)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_f = f.copy()
            best_M = M
            print(
                f"[{time.time() - start_time:6.2f}s] LP improve iter {it+1}, score={score:.6f}"
            )
        else:
            # no improvement – stop early
            break

    return best_f, best_score
