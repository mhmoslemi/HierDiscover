    n = len(f)
    # Build A_ub for constraints f*b <= M
    rows = []
    cols = []
    data = []

"""
Minimum‑peak self‑convolution optimiser.

The algorithm builds a non‑negative step function a (∑a = 1) that
minimises  2·n·max(conv(a,a)).  It combines several complementary
phases:

1️⃣  Exponentiated‑gradient on a soft‑max surrogate (many short
    passes over a range of temperature parameters).

2️⃣  Gradient descent on a log‑sum‑exp (“smooth‑max”) surrogate with a
    geometric temperature schedule, which drives the distribution toward a
    flatter autocorrelation.

3️⃣  Projected sub‑gradient descent on the exact objective (as in the
    reference implementation).

4️⃣  Targeted pair‑annealing that repeatedly weakens the two indices that
    contribute most to the current maximal convolution entry.

5️⃣  Short random‑restart bursts that perturb the current best solution
    and re‑run phases 2–4.

All phases respect the simplex constraint (non‑negative, sum = 1) by
projecting onto the probability simplex after each update.
The routine adapts the work allocated to every phase to the remaining
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv) (∑a has been normalised to 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on a soft‑max surrogate
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.4,
                      iters_per_alpha: int = 12_000):
    """Exponentiated‑gradient descent on the log‑sum‑exp of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, _next_pow2(2 * n - 1)), n)

    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: iter={it:,} "
                      f"score={cur_score:.6f}")

            # soft‑max weights (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smooth max (2·corr)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # exponentiated update
            eta = eta0 / math.sqrt(it)
            grad = np.clip(grad, 0.0, 10.0)          # avoid overflow
            a = a * np.exp(-eta * grad)

            # simplex projection
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Gradient descent on a smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _run_smooth_max_grad(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 40_000,
                         T_start: float = 5e-2,
                         T_end: float = 5e-4):
    """
    Gradient descent on f_T(a)=T·log∑exp(conv(a,a)/T).
    Temperature T decays geometrically from T_start to T_end.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    best_score = _score_from_conv(_conv_self(a, L), n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution
        conv = _conv_self(a, L)

        # temperature schedule (geometric)
        frac = it / max_iters
        T = T_start * ((T_end / T_start) ** frac)

        # soft‑max weights w_i ∝ exp(conv_i / T)
        scaled = conv / T
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of f_T : 2·(a * w_rev)   (cross‑correlation)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # simple step‑size schedule
        step = 0.6 / math.sqrt(it)

        a = a - step * grad
        a = _project_simplex(a)

        cur_score = _score_from_conv(_conv_self(a, L), n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑max improve: iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient from the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + simplex projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Targeted pair‑annealing
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-9,
                        steps: int = 300_000):
    """Simulated annealing that constantly attacks the two indices forming the peak."""
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current worst convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each admissible i to the peak
        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose a move: shift up to 40 % of each source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.40   # ≤40 % of source
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights (step function) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the previous best if available, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4096
    if starter is not None:
        a0 = np.interp(
            np.linspace(0, 1, final_n),
            np.linspace(0, 1, starter.size),
            starter,
        )
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _score_from_conv(_conv_self(a0, _next_pow2(2 * final_n - 1)), final_n)

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.10 * remaining)

    a_exp, score_exp = _run_exp_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.5,
        iters_per_alpha=12_000,
    )
    if score_exp < best_score:
        best_score = score_exp
        best_seq = a_exp.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(10.0, 0.15 * remaining)

        a_smooth, score_smooth = _run_smooth_max_grad(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            max_iters=40_000,
            T_start=5e-2,
            T_end=5e-4,
        )
        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent (the bulk of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase3_deadline = time.time() + max(20.0, 0.45 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase3_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted pair‑annealing (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Optional random‑restart bursts (if time still left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(30.0, time_left / 2.0)
        restart_deadline = time.time() + block

        # perturb current best by a small Gaussian noise
        pert = best_seq + 0.02 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # a quick sub‑gradient pass
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        if time.time() < deadline:
            # finish with a short annealing run
            pert_seq, pert_score = _run_pair_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
