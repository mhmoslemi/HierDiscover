"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The routine works under a single time budget (default 1000 s).

Strategy
--------
1. **Initialisation** – If a previous construction (`height_sequence_1`)
   is available it is interpolated to a working size `n` (default 4096)
   and normalised to sum = 1.  Otherwise a uniform vector is used.

2. **Adam optimisation of a smooth max** –  
   The non‑smooth objective `max(conv)` is approximated by a
   soft‑max (log‑sum‑exp) with a temperature `α`.  The gradient of the
   smooth max is cheap to compute via two FFT‑based convolutions.
   Adam (adaptive moment estimation) is used together with a Euclidean
   projection onto the probability simplex.  The temperature grows
   slowly so that the optimisation first moves the point far from the
   symmetric uniform region and later concentrates on the true maximum.

3. **Entropy mirror descent (exponentiated gradient)** –  
   Using the same smooth‑max gradient we perform a multiplicative
   update `a ← a·exp(−η·grad)` with a decaying learning rate `η`.
   After each step the vector is re‑normalised (which automatically
   projects onto the simplex).

4. **Local random refinements** –  
   A cheap hill‑climbing stage that moves a tiny amount of mass from a
   random source index to a random target index.  Changes that improve
   the exact objective are kept; occasional uphill moves are allowed
   with a simple temperature schedule.

All stages respect the global deadline; the best vector found is
returned as a plain Python list.
"""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv) with respect
    to the non‑negative vector a.

    The soft‑max weights are w_k ∝ exp(α·conv_k).  The gradient is
        g = 2 * (w * a_rev)[n-1 : n-1+n]
    where `*` denotes linear convolution and `a_rev` is a reflected copy
    of a.
    """
    n = a.shape[0]
    # soft‑max weights (avoid overflow by subtracting max)
    w = np.exp(alpha * (conv - conv.max()))
    w /= w.sum()
    a_rev = a[::-1]

    fft_len = _next_pow2(2 * n - 1 + n - 1)           # = _next_pow2(3*n-2)
    W_fft = np.fft.rfft(w, n=fft_len)
    A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)

    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective used by the evaluation function (sum(a) assumed >0)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget (in seconds) and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    safety = 0.6                      # keep a small margin for finalisation
    deadline = start_time + budget_s - safety

    # -------------------------------------------------------------
    # Choose a problem size.  4096 gives a finer discretisation than
    # the historic 2048 while keeping the FFT cheap.
    # -------------------------------------------------------------
    n_target = 4096

    # -------------------------------------------------------------
    # Initialise from a previously discovered construction if possible.
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim != 1 or tmp.size == 0:
                raise ValueError
            n0 = tmp.size
            if n0 != n_target:
                # linear interpolation to the required length
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n_target)
                tmp = np.interp(new_x, old_x, tmp)
            a = tmp.copy()
        except Exception:
            a = np.full(n_target, 1.0 / n_target)
    else:
        a = np.full(n_target, 1.0 / n_target)

    # normalise to the simplex (sum = 1)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()

    best_seq = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Stage 2 – Adam optimisation of a smooth max approximation.
    # -----------------------------------------------------------------
    # Parameters – can be tuned
    alpha_start = 5.0                 # initial temperature for soft‑max
    alpha_inc = 0.5                   # how fast temperature grows
    alpha_cap = 150.0                 # maximal temperature
    lr0 = 0.8                         # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    m = np.zeros_like(a)              # first moment
    v = np.zeros_like(a)              # second moment
    it = 0
    # allocate roughly 60 % of the remaining time to this stage
    stage_deadline = start_time + (deadline - start_time) * 0.60

    while time.time() < stage_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()   # sum(a)=1, identical to _objective
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter {it:<6d}  improve → {cur_score:.6f}")

        # temperature schedule (slowly increase)
        alpha = min(alpha_cap, alpha_start + alpha_inc * (it // 500))

        # smooth‑max gradient
        grad = _smooth_max_grad(a, conv, alpha)

        # Adam update (gradient descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex
        a = _project_simplex(a_new)

    # -----------------------------------------------------------------
    # Stage 3 – Exponentiated‑gradient (entropy mirror descent) polishing.
    # -----------------------------------------------------------------
    it = 0
    eta0 = 0.6                        # larger than the original 0.20
    alpha_start = 30.0                # a higher temperature for precise max
    alpha_cap = 200.0
    while time.time() < deadline - 4.0:   # keep ~4 s for the final stage
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG iter {it:<6d}  improve → {cur_score:.6f}")

        # slowly increase temperature to focus on the true maximum
        alpha = min(alpha_cap, alpha_start + 0.05 * it)

        grad = _smooth_max_grad(a, conv, alpha)

        eta = eta0 / math.sqrt(it)   # decaying learning rate
        a = a * np.exp(-eta * grad)   # multiplicative update
        # re‑normalise (projects onto the simplex)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)

    # -----------------------------------------------------------------
    # Stage 4 – Simple random local refinements (hill‑climbing).
    # -----------------------------------------------------------------
    it = 0
    temperature = 1e-3                 # initial acceptance temperature
    temp_decay = 0.9999                # geometric cooling
    max_local_iter = 200000
    while time.time() < deadline and it < max_local_iter:
        it += 1
        # pick a source index (biased towards larger entries)
        i = np.random.choice(a.shape[0], p=a / a.sum())
        j = np.random.randint(0, a.shape[0])
        if i == j:
            continue
        # move a tiny amount of mass
        delta = min(a[i] * 0.03, 1e-4 * random.random())
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # re‑project to avoid negative rounding errors
        a_candidate = _project_simplex(a_candidate)

        # exact objective (fast via FFT)
        conv_cand = _conv_self(a_candidate)
        cand_score = 2.0 * a.shape[0] * conv_cand.max()

        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:6.2f}s] local move {it:<6d} improve → {cand_score:.6f}")
            temperature = max(temperature * 0.9, 1e-6)   # a bit more greedy after success
        else:
            # occasional uphill acceptance (simulated annealing)
            prob = math.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if random.random() < prob:
                a = a_candidate
                temperature *= temp_decay

    # -----------------------------------------------------------------
    # Final normalisation and conversion to plain Python list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
