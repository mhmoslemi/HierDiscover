"""
Heuristic search for a step‑function (non‑negative list of heights) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.  Three stages are performed inside the 1000 s wall‑clock
budget:

1️⃣  Fast parametric sweep over several analytic families
    (geometric, stretched‑exponential, power‑geometric, beta,
    mixtures of two geometries and a few random Dirichlet draws).  The
    best candidate of this cheap sweep is retained.

2️⃣  Several independent runs of projected sub‑gradient descent with
    Nesterov momentum and an adaptive 1/t⁰·⁶ step‑size.  Each run starts
    from a slightly perturbed version of the best candidate found so far.
    The best sequence among all runs is kept.

3️⃣  A light mass‑transfer simulated‑annealing polish that moves a
    small fraction of the probability mass from indices that contribute
    most to the current maximal convolution entry toward random destinations.
    Temperature decays exponentially.  The final best sequence is normalised
    to unit sum and returned as a plain list of floats.

All helper functions are defined at module level (no closures, no lambdas)
and the routine never exceeds the allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv) )
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    # denominator is ≈1 (we keep it for safety)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         seed_id: int = 0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses Nesterov momentum and a step size η₀ / t^{α} with α≈0.6.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # momentum vector
    velocity = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    eta0 = 0.5          # initial step size (larger than before)
    alpha = 0.6         # exponent governing decay
    it = 0
    L_grad = _next_pow2(3 * n - 2)   # length for FFT gradients

    # occasional random jitter to avoid stagnation
    jitter_every = 500_000

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SG run{seed_id} it={it:,} score={cur_score:.6f}")

        # gradient of the active entry (if several indices tie we pick one)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # ∂conv[k]/∂a_i

        # Nesterov‑style update
        velocity = beta * velocity + (1.0 - beta) * grad
        step = eta0 / ((it + 1) ** alpha)
        a = a - step * velocity
        a = _project_simplex(a)

        # occasional jitter to escape flat regions
        if it % jitter_every == 0:
            eps = 1e-4
            rnd = np.random.default_rng(seed=seed_id + it).random(a.shape)
            a = a + eps * (rnd - 0.5)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------


def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer.
    At each step a fraction of the mass of a “source’’ index is moved to a
    “destination’’ index.  The source is chosen among those that contribute
    most to the current maximal convolution entry.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    # pre‑allocate a conv array that will be refreshed when needed
    conv = _conv_self(a)

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # index where the convolution attains its current maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # candidate sources are those in the active window with large a_i
        active_window = np.arange(lo, hi + 1, dtype=int)
        probs = a[active_window]
        if probs.sum() == 0:
            src = rng.choice(active_window)
        else:
            src = rng.choice(active_window, p=probs / probs.sum())

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # recompute convolution for the proposal
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (multiple restarts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # keep at least 30 s for the final annealing stage
    if remaining > 40:
        # allocate about 2/3 of the rest to sub‑gradient runs
        sg_budget = remaining * 0.65
        # number of independent runs (2–4 depending on time)
        n_runs = max(1, int(sg_budget // 120))   # ≈2 min per run
        sg_per_run = sg_budget / n_runs
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient phase: {n_runs} runs, ~{sg_per_run:.1f}s each")

        for run_id in range(n_runs):
            run_deadline = time.time() + sg_per_run
            # start from the best so far, add a tiny random perturbation
            a0 = best_seq.copy()
            eps = 5e-3
            a0 += eps * (rng.random(n) - 0.5)
            a0 = _project_simplex(a0)

            a_opt, sc_opt = _subgradient_descent(a0, n, run_deadline,
                                                start_time,
                                                best_score,
                                                seed_id=run_id)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad run{run_id} improved to {sc_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        steps = int(min(600_000, remaining * 5000))   # heuristic: ~5k steps per sec
        print(f"[{time.time() - start_time:.2f}s] Annealing phase: {steps} steps")
        a_final, sc_final = _annealing(best_seq, n, deadline,
                                       start_time,
                                       best_score,
                                       best_seq,
                                       start_temp=1e-2,
                                       end_temp=1e-7,
                                       max_steps=steps)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Return as plain Python list
    return [float(x) for x in best_seq]
