"""
optimal_step_search.py
---------------------

Search for a non‑negative step function that minimizes the
``evaluate_sequence`` metric.

The algorithm combines several ideas that complement each other:

1. **Warm start** – reuse the best historic sequence (``height_sequence_1``)
   if it exists, otherwise start from the uniform distribution.

2. **Projected Adam on a smooth approximation of the max‑convolution**.
   The smoothed objective is
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)[k])
   with an increasing schedule for α.  After each Adam step the iterate
   is projected onto the probability simplex (Σa = 1, a≥0).

3. **Fine‑grained sub‑gradient descent** on the true max‑convolution.
   The sub‑gradient is obtained from all near‑active convolution entries.
   A small momentum term and a 1/√t step schedule provide a stable polish.

4. **Aggressive greedy peak‑reduction** – repeatedly move a modest
   fraction of the coefficient that contributes most to the current peak
   towards the smallest coefficient (incremental O(n) update of the
   convolution).

5. **Light‑weight simulated‑annealing refinements** – random pairwise
   mass transfers are accepted if they improve the exact score or with a
   probability that decays with temperature.  This helps escape tiny
   local minima that the deterministic phases may miss.

The whole pipeline respects the 1000 s wall‑clock limit; each phase stops
as soon as the remaining budget is exhausted.  The best sequence found
throughout the run is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog   # only used for a tiny optional LP check

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx = 1, x≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _smooth_max_grad(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max:
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _exact_score(a: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence for a (assumes Σa=1)."""
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Projected Adam on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth approximation of max (a∗a).
    Returns the best vector (probability simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # α schedule (geometric progression, ends a little beyond the previous code)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0, 32768.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = _exact_score(a)

    for alpha in alpha_schedule:
        # run Adam for as long as the global deadline permits
        while time.time() < deadline:
            conv = _conv_self(a)
            # smooth max value (for monitoring only)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            _ = 2.0 * n * f_val   # dummy to keep same scale as final score

            # gradient w.r.t. a (outer factor 2·n)
            grad = _smooth_max_grad(a, conv, alpha, fft_len) * (2.0 * n)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 700 == 0:
                sc = _exact_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha:.0f}) "
                          f"score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a∗a)[k] .
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _exact_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # active set (indices whose value is within 1e-12 relative of the max)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # build sub‑gradient: for each active k, add the contribution of all (i,j) with i+j=k
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless for sub‑grad

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Fast greedy peak reduction
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shift a modest fraction of the coefficient that most
    contributes to the current convolution peak towards the smallest
    coefficient (incremental O(n) update).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    # decay schedule for the transferred fraction
    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index that most fuels the peak

        # choose a destination that is currently small (but not the source)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # transfer fraction; the fraction slowly shrinks as we iterate
        frac = 0.08 * (0.999 ** (it // 5000))   # start ~8 %, decay slowly
        eps = max(frac * a[src], 1e-15)
        if eps < 1e-15:
            break

        # incremental convolution update (exact)
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it:,}, max={new_max:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Light simulated annealing (random pairwise mass moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         init_temp: float = 1e-3,
                         decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Perform a simple annealing loop: pick random source/destination,
    move a small amount of mass and accept if the score improves, or
    with a probability exp(-Δ/T) otherwise.  The temperature decays.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _exact_score(a)
    cur_score = best_score
    T = init_temp
    it = 0

    while time.time() < deadline and T > 1e-12:
        it += 1
        # draw two distinct indices
        src, dst = np.random.choice(n, size=2, replace=False)
        if a[src] < 1e-12:
            continue

        # propose moving a fraction of src to dst
        delta = min(0.03 * a[src], 1e-12)   # start with at most 3 % of src
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # exact evaluation (fast via FFT)
        new_score = _exact_score(a_new)

        # acceptance test
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={new_score:.6f}")

        # temperature schedule
        T *= decay

    return best_a, best_score

# Utilities for initialisation and final normalisation
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n.
    Prefer the historic best (``height_sequence_1``) when available,
    otherwise fall back to the uniform distribution.
    """
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linearly interpolate to the desired length (preserving total mass)
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)

def propose_candidate(seed: int = 0,

    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 4) Fast greedy peak reduction (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        greedy_deadline = min(deadline, time.time() + 0.15 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start fast greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, greedy_deadline, start_time)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

        best_a, best_score = _simulated_annealing(best_a, anneal_deadline, start_time)

    # ------------------------------------------------------------------
    # 6) Final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()

    else:

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
