def _run_greedy_refine(a0, n, deadline, start_time, cur_best_score, cur_best_seq, max_iters=3000000):
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    recompute_interval = 1000
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # find peak
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # compute second max by temporary masking
        orig = conv[max_idx]
        conv[max_idx] = -np.inf
        second_max = conv.max()
        conv[max_idx] = orig

        # find contributing indices
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # destination: global minimal a (excluding src)
        dst_candidates = np.where(a == a.min())[0]
        # ensure src not in candidates
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            # take random other index
            dst = (src + n // 2) % n  # far away
        else:
            dst = np.random.choice(dst_candidates)

        # partner values in current max sum
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a) if delta_a != 0 else 0.0
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        a_old = a.copy()
        # incremental update
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score
```

"""
Propose a non‑negative step function (probability vector) with a small
self‑convolution peak.  The search consists of three stages:

1.  Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
    This quickly finds a good “shape”.

2.  Exponentiated‑gradient (mirror descent on the simplex) for a longer
    period – cheap because each iteration needs only one FFT‑based
    convolution.

3.  A deterministic greedy “peak‑reduction” phase that updates the
    convolution incrementally (O(n) per move).  It repeatedly moves a
    small amount of probability from the biggest contributor to the
    current maximum towards a globally minimal entry, recomputing the
    full convolution only occasionally.  This phase can run for
    millions of cheap moves and typically shaves the last few 10⁻³ off
    the objective.

Optionally short random‑restart cycles are performed while time permits.
The routine respects a 1000 s wall‑clock budget and finally returns the
best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a using FFT (returns length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_seq(seq: np.ndarray) -> float:
    """Exact objective (2·n·max(conv)) for a normalised sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()

# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:

    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ------- smooth max (log‑sum‑exp) -------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)   # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ------- gradient of the smoothed max -------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ------- Adam update -------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score

# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (exactly sum‑to‑1)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score

# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_refine(a0: np.ndarray,
                       n: int,
                       deadline: float,

                       cur_best_seq: np.ndarray,

    """
    Repeatedly move a small amount of mass from the biggest contributor
    to the current maximal convolution entry towards a globally minimal
    entry.  Updates of the full convolution are performed incrementally;
    a full recomputation is done every `recompute_interval` steps to keep
    numerical errors under control.
    """

    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)               # full convolution (len 2n‑1)
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1

        # current peak
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest value (ignore the current maximum)
        orig = conv[max_idx]
        conv[max_idx] = -np.inf
        second_max = conv.max()
        conv[max_idx] = orig

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # destination – pick a globally minimal index (different from src)
        min_val = a.min()
        cand = np.where(a == min_val)[0]
        cand = cand[cand != src]
        if cand.size == 0:
            dst = (src + n // 2) % n          # fallback far away index
        else:
            dst = np.random.choice(cand)

        # partner values for src/dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide how much probability to move
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            # try to bring the peak down to (just below) the second maximum
            eps_target = (max_val - second_max) / (2.0 * delta_a) if delta_a != 0 else 0.0
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        a_old = a.copy()

        # ----- incremental update of the convolution -----
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        # numerical clean‑up
        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score
