def make_sequence_from_blocks(blocks, n):
    m = len(blocks)
    repeats = n // m
    extra = n % m
    seq = np.repeat(blocks, repeats)
    if extra > 0:
        seq = np.concatenate([seq, blocks[:extra]])
    # now seq length = n (since repeats*m + extra = n)
    return seq
```

def simulated_annealing(start_blocks, n, deadline, start_time, best_seq, best_score):
    # start_blocks: np.ndarray length m > 0, positive numbers
    # convert to log scale
    log_blocks = np.log(start_blocks + 1e-12)
    current_blocks = np.exp(log_blocks)
    # build sequence
    a = make_sequence_from_blocks(current_blocks, n)
    a = a / a.sum()
    cur_score = _objective(a)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(...)
    # parameters
    T = cur_score * 0.1 + 1e-6   # initial temperature
    sigma = 0.1
    iter = 0
    while time.time() < deadline:
        iter +=1
        # propose new log blocks
        # choose random subset of indices to perturb
        idx = rng.integers(0, len(log_blocks), size=2) # moving between two?

        new_log = log_blocks + rng.normal(0, sigma, size=log_blocks.shape)
        # or random choose subset
        new_blocks = np.exp(new_log)
        # build seq
        a_prop = make_sequence_from_blocks(new_blocks, n)
        a_prop = a_prop / a_prop.sum()
        new_score = _objective(a_prop)
        # acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score)/T):
            log_blocks = new_log

                best_seq = a_prop.copy()
                print(f"... SA improve: iter={iter}, score={new_score:.6f}")
        # decay temperature and sigma
        T *= 0.9999
        sigma *= 0.99995
        # optional early break if T small
    return best_seq, best_score

"""
Search for a non‑negative step function minimizing ``evaluate_sequence``.

The algorithm consists of four stages:

1. **Diverse handcrafted starts** – uniform, exponential, Gaussian,
   Dirichlet and the previously known best sequence (if available).

2. **Simulated‑annealing on a low‑dimensional block representation**.
   The full vector (n=4800) is represented by ``m`` constant blocks.
   A simple Metropolis scheme with multiplicative (log‑space) proposals
   explores the landscape and keeps the best candidate found.

3. **Exponentiated‑gradient polishing** – projected mirror‑descent on the
   true objective (max‑convolution) applied to the full‑resolution vector.
   A larger initial step size (``eta0=0.8``) helps escape shallow minima.

4. **Greedy peak reduction** – deterministic local moves that transfer a
   fraction of the coefficient contributing most to the current peak to
   the smallest coefficient, accepting only strict improvements.

All stages respect the supplied time budget (default 1000 s).  The final
sequence is clipped to ``[0,∞)``, renormalised to sum 1 and returned as
a plain ``list[float]``.

import time

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

# Fast objective (the evaluation function for a normalised vector)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Returns the exact value of ``evaluate_sequence`` for a vector that
    already satisfies ``sum(a)=1`` (the usual normalisation).
    """
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max())

def _make_sequence_from_blocks(blocks: np.ndarray, n: int) -> np.ndarray:
    """Repeat each block value to obtain a length‑``n`` vector."""
    m = blocks.shape[0]
    repeats = n // m
    extra = n % m
    seq = np.repeat(blocks, repeats)
    if extra:
        seq = np.concatenate([seq, blocks[:extra]])
    return seq


def _downsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` onto ``new_n`` points."""
    old_n = seq.shape[0]
    if new_n >= old_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _simulated_annealing(init_blocks: np.ndarray,

    Simple Metropolis algorithm on the log‑space of the block variables.
    ``init_blocks`` must be positive; the routine returns the best full‑length
    vector discovered.
    """

    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators (used for the initial pool)
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of a brief simulated‑annealing phase on a
    low‑dimensional block representation, an extensive exponentiated‑gradient
    polishing, and a final greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of handcrafted starts (including the previous best)
    # ------------------------------------------------------------------
    starts = []

    # 1) previous construction, if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential (soft decay)
    starts.append(_exponential_start(n, r=0.9995))
    # 4) exponential (stronger decay)
    starts.append(_exponential_start(n, r=0.985))

    # 5) Gaussian bell
    starts.append(_gaussian_start(n))

    # 6) Dirichlet – moderate concentration
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 7) Dirichlet – low concentration (spiky)
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _objective(a0, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – simulated‑annealing on a low‑dimensional block model
    # ------------------------------------------------------------------
    m_blocks = 30                                     # number of blocks
    sa_deadline = start_time + budget_s * 0.25        # ~25 % of total budget

    # down‑sample the current best to obtain a sensible starting block vector
    init_blocks = _downsample(best_seq, m_blocks)

    # keep launching independent SA runs while time permits
    while time.time() < sa_deadline:
        # 30 % of the runs start from the down‑sampled vector, the rest from random Dirichlet
        if rng.random() < 0.3:
            start_blocks = init_blocks.copy()
        else:
            start_blocks = _dirichlet_start(m_blocks, conc=1.0, rng=rng)

        best_seq, best_score = _simulated_annealing(
            start_blocks, n, fft_len, sa_deadline, start_time, best_seq, best_score, rng
        )
        if time.time() >= sa_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient polishing (majority of remaining time)
    # ------------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.9        # leave a small tail for greedy step
    polished_seq, polished_score = _expgrad_optimize(
        best_seq, polish_deadline, start_time, fft_len, eta0=0.8
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final reported score (for logging – uses the provided evaluate_sequence)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
