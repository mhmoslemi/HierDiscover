"""
Propose a step‑function that beats the previous bound (≈1.5088).

Algorithm
---------
1. **Large canvas** – use a longer sequence (default n = 8192).  The
   theoretical continuous optimum (max (conv) = 3/4·1/n) becomes
   closer to the discrete optimum as n grows.

2. **Analytic sweep** – generate a few families of symmetric non‑negative
   shapes:
        • power‑tent   a_i ∝ min(x_i,1−x_i)^p
        • sine‑tent    a_i ∝ sin(π·x_i)^p   (p>0)
   where x_i = (i+0.5)/n.  A dense grid of exponents (step 0.02) is tried
   and the best candidate (lowest exact objective) is kept.

3. **Projected sub‑gradient descent** – starting from the best analytic
   candidate we minimise the true non‑smooth objective
   `2·n·max(conv(a,a))/sum(a)^2` by a momentum‑accelerated projected
   sub‑gradient method.  The iterate is projected onto the probability
   simplex after every step, preserving non‑negativity and unit sum.

4. **Light simulated‑annealing polish** – a few‑ten‑thousand random
   mass‑transfer moves are performed to escape possible shallow minima.

5. **Return** – the final vector is clipped to be non‑negative,
   normalised to sum = 1 and converted to a plain Python list; the harness
   evaluates it with `evaluate_sequence`.

The whole search respects the 1000 s budget (a small safety margin is kept).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# FFT‑based linear convolution (fast, O(n log n))
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)−1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The sequence is assumed to be non‑negative; scaling does not matter.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# --------------------------------------------------------------
# Euclidean projection onto the probability simplex Σw=1, w≥0
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Return the Euclidean projection of v onto {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Projected sub‑gradient descent on the exact (non‑smooth) objective
# --------------------------------------------------------------

                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:

    Momentum‑accelerated projected sub‑gradient descent.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.15        # initial step size (tuned empirically)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"subgrad iter={it:,} score={cur_score:.6f}")

        # index where the convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see reference implementation)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum update + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection back onto the simplex (non‑negative, sum = 1)
        a = _project_simplex(a)

    return best_a, best_score

# Very light simulated‑annealing polish
# --------------------------------------------------------------
def _anneal_polish(a: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with an exponential cooling schedule.
    """
    n = a.shape[0]
    curr = a.copy()
    curr_score = _objective(curr)
    best = curr.copy()
    best_score = curr_score

    rng = np.random.default_rng()
    total_steps = 300_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.008 * (1e-6 / 0.008) ** (it / total_steps)

        # pick a source index proportionally to its mass
        src = rng.choice(n, p=curr)
        dst = rng.integers(n)
        if src == dst or curr[src] < 1e-15:
            continue

        eps = curr[src] * rng.random() * 0.30  # transfer up to 30 % of src mass
        prop = curr.copy()
        prop[src] -= eps
        prop[dst] += eps

        new_score = _objective(prop)
        delta = new_score - curr_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            curr = prop
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = curr.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best, best_score

# Main entry point required by the competition harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Problem size – a larger canvas gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8192                     # ≈ double the previous best length

    # ------------------------------------------------------------------
    # 2) Analytic families – dense sweep of a few simple parametric shapes.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # grid points representing the centre of each cell
    x = (np.arange(n) + 0.5) / n   # values in (0,1)

    # ----- power‑tent family ------------------------------------------------
    exponents = np.arange(0.40, 1.21, 0.02)   # 0.40,0.42,…,1.20
    for p in exponents:
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] power‑tent p={p:.2f} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ----- sine‑tent family -------------------------------------------------
    exponents_sin = np.arange(1.0, 2.51, 0.05)   # 1.00 … 2.50
    for p in exponents_sin:
        w = np.sin(np.pi * x) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] sine‑tent p={p:.2f} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # 3) Fallback: uniform distribution (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:.2f}s] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Local refinement with projected sub‑gradient descent.
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 12.0:                      # keep a few seconds for polishing
        sub_deadline = now + remaining * 0.85   # ~85 % of what remains
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final annealing polish (optional, cheap).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal_polish(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final objective = {_objective(best_seq):.6f}")
    return [float(v) for v in best_seq]
