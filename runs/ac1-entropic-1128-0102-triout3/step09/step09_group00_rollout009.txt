def _adam_opt(a0, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    step_alpha = 0.2
    it = 0
    best_a = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # compute subgradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a = a - step_alpha * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _lp_update(f, M, t):
    # find active constraints indices
    conv_f = _conv_self(f)
    max_val = conv_f.max()
    active_idx = np.where(conv_f >= max_val * 0.9999999)[0] # tolerance
    # Build A_ub
    n = f.shape[0]
    rows = []
    for k in active_idx:
        row = np.zeros(n)
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        row[idx] = f[k - idx]
        rows.append(row)
    A_ub = np.vstack(rows) if rows else np.empty((0, n))
    b_ub = np.full(A_ub.shape[0], M)
    # Solve BP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f
    g = res.x
    # Normalize g to sum 1
    g_sum = g.sum()
    if g_sum == 0:
        return f
    g = g / g_sum
    new_f = (1 - t) * f + t * g
    # ensure nonnegative and sum to 1 via projection
    new_f = _project_simplex(new_f)
    return new_f
```

"""
Hybrid LP‑Adam optimisation for the step‑function bound.

The programme builds several simple candidate shapes (uniform, power‑law,
exponential, random Dirichlet) and keeps the few with the smallest
objective.  For each of these candidates it performs two phases:

1.  LP‑refinement – a linear programme is solved that maximises the total
    mass of a new distribution `g` while keeping the most critical
    convolution entries `(f * g)(k)` below the current peak `M`.  The new
    iterate is a convex combination `f←(1‑α)f+αg` (α≈0.2) and is projected
    back onto the probability simplex.

2.  Adam‑sub‑gradient descent – a projected Adam optimiser is run on the
    true (non‑smooth) objective `2·n·max(conv(f,f))/sum(f)^2` to polish
    the LP result.

A very light simulated‑annealing pass is performed with the remaining
budget.  The best sequence found before the deadline is returned as a
plain Python list of non‑negative floats (the evaluator will normalise
the sum automatically).

The whole search respects a 1000 s wall‑time budget and prints progress
messages for reproducibility.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a with itself using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the true (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the exact objective.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.25            # initial step‑size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # sub‑gradient of max(conv) (same as in the reference code)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_update(a: np.ndarray, M: float, top_k: int = 10) -> np.ndarray:
    """
    One LP refinement step.
    - a : current distribution (already normalised, sum=1)
    - M : current maximal convolution value
    - top_k : number of largest convolution entries that become constraints
    Returns a new distribution (convex combination of a and the LP solution)
    """
    n = a.shape[0]
    conv = _conv_self(a)

    # indices of the K largest convolution entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    rows = []
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1, dtype=int)
        row = np.zeros(n)
        row[cols] = a[k - cols]
        rows.append(row)

    A_ub = np.vstack(rows) if rows else np.empty((0, n))
    b_ub = np.full(A_ub.shape[0], M)

    # maximise Σ g_i  ⇔ minimise -Σ g_i
    c = -np.ones(n)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n,
                  method="highs", options={"presolve": True})

    if not res.success:
        # LP failed – fall back to the original vector
        return a

    g = res.x
    s = g.sum()
    if s == 0.0:
        return a
    g = g / s

    # convex mix (α determines aggressiveness; 0.2 works well)
    alpha = 0.2
    new_a = (1.0 - alpha) * a + alpha * g
    new_a = _project_simplex(new_a)
    return new_a


def _lp_refine(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP refinement (a few hundred iterations at most).
    Returns the best vector and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()

        a = _lp_update(a, M, top_k=12)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter={it:,} score={cur_score:.6f}")

        # stop after a few hundred cheap iterations if we are close to deadline
        if it >= 400:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(a: np.ndarray, cur_score: float,
            deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_seq = a.copy()
    best_local_score = cur_score
    rng = np.random.default_rng()
    total_steps = 30_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling schedule
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_seq, best_local_score
