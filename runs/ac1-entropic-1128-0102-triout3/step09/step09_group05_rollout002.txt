"""
Propose a candidate step‐function for minimise evaluate_sequence.

Algorithm
---------
1. **Coarse block optimisation** – the sequence (length n=4096) is forced to be
   piecewise constant on m=256 blocks (block size = 16).  We run an
   exponentiated‑gradient descent on a *smooth* (log‑sum‑exp) surrogate of
   the max‑convolution.  The gradient is computed in the Fourier domain,
   then aggregated to the block variables and projected onto the simplex.
   This quickly removes large spikes while keeping the search space small.

2. **Fine‑grained exponentiated‑gradient** – starting from the expanded
   block solution we run the original EG routine (smooth max) on the full
   vector.  A collection of alphas and a decaying learning rate are used.

3. **Projected sub‑gradient descent** – we refine the solution on the true
   objective  f(a)=2·n·max(conv(a,a)).  Momentum and 1/√t step‑size are
   employed.

4. **Targeted pair‑annealing** – a simulated‑annealing phase that repeatedly
   weakens the two indices that currently contribute most to the peak of the
   self‑convolution and moves a fraction of their mass to random locations.
   The temperature follows an exponential schedule.

If any time remains we perform short random restarts (small perturbations of
the best solution) combined with a few extra sub‑gradient/annealing cycles.

All helper routines are top‑level, no closures or lambdas are used and the
total runtime respects the supplied budget (default 1000 s).  The returned
list is a non‑negative sequence normalised to sum 1, ready for
evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  Assumes Σa = 1."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 0 – block‑level exponentiated‑gradient (coarse optimisation)
# ----------------------------------------------------------------------
def _run_exp_gradient_block(p0: np.ndarray,
                            n: int,
                            m: int,
                            deadline: float,
                            start_time: float,
                            alphas,
                            eta0: float = 0.4,
                            iters_per_alpha: int = 5000):
    """
    Exponentiated‑gradient on a smooth approximation of max(conv) where
    the vector a is forced to be piecewise constant on m blocks.
    Returns the best full‑length vector a (as np.ndarray) and its true score.
    """
    block = m
    block_sz = n // block
    # initial block distribution (already on simplex)
    p = p0.copy()
    best_a = None
    best_score = np.inf
    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a if best_a is not None else _expand_block(p, n, block_sz), best_score

            it += 1

            # expand to full length
            a = np.repeat(p / block_sz, block_sz)

            # convolution and true score
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] BLOCK EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # -------- smooth‑max weights (log‑sum‑exp) ----------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # -------- gradient wrt a (same as full EG) ----------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # -------- aggregate to block gradient ----------
            grad_p = np.add.reduceat(grad_a, np.arange(0, n, block_sz)) / block_sz

            # -------- multiplicative update & simplex projection ----------
            eta = eta0 / math.sqrt(it)
            # Clip extreme values to keep the exponent stable
            grad_p = np.clip(grad_p, -10.0, 10.0)
            p = p * np.exp(-eta * grad_p)
            p = _project_simplex(p)

    # final expansion
    final_a = np.repeat(p / block_sz, block_sz)
    return final_a, best_score if best_score < np.inf else _score_from_conv(_conv_self(final_a, L), n)


def _expand_block(p: np.ndarray, n: int, block_sz: int) -> np.ndarray:
    """Utility – expand block vector p to full length n."""
    return np.repeat(p / block_sz, block_sz)


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on the smooth max (full resolution)
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.3,
                      iters_per_alpha: int = 8000):
    """
    Same as in the reference implementation – optimiser on the smooth
    log‑sum‑exp surrogate of the maximum convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)
    best_score = _score_from_conv(_conv_self(a, L), n)

    it = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # smooth‑max weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # exponentiated update
            eta = eta0 / math.sqrt(it)
            grad = np.clip(grad, 0.0, 10.0)
            a = a * np.exp(-eta * grad)

            # simplex projection
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient from the maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 180_000):
    """
    Simulated annealing that always attacks the two indices that contribute
    most to the current maximal convolution coefficient.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # peak index
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        contrib = a[idx_range] * a[k - idx_range]

        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max

        src = [i_max] if i_max == j_max else [i_max, j_max]

        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.55 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

        # a short, aggressive sub‑gradient run
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                final_n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted pair‑annealing (finishing touch)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=240_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Optional random restarts (if any time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # keep a small perturbation of the current best and run a quick
        # sub‑gradient + annealing combo
        time_left = deadline - time.time()
        block_time = min(30.0, time_left / 2.0)
        restart_deadline = time.time() + block_time

        pert = best_seq + 0.02 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=100_000,
        )
        if time.time() < deadline:
            pert_seq, pert_score = _run_pair_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] DONE – final score approx {best_score:.6f}")
    return [float(x) for x in best_seq]
