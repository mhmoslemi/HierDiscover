def _lp_refine(a0, deadline, start_time, max_iter=5, t=0.2):
    a = a0.copy()
    n = len(a)
    # ensure no zero entries
    eps_min = 1e-8
    a = np.maximum(a, eps_min)
    a = a / a.sum()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        # create sparse matrix
        rows = []
        cols = []
        data = []
        for k in range(2*n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # i range
            for i in range(lo, hi+1):
                rows.append(k)
                cols.append(i)
                data.append(a[k - i])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
        b_ub = np.full(2*n - 1, M)
        c = -np.ones(n)  # maximize sum
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                         bounds=(0, None), method='highs')
        except Exception as e:
            print("LP failed:", e)
            break
        if not res.success:
            print("LP not successful:", res.message)
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b = b / sum_b
        # mix
        a = (1 - t) * a + t * b
        # re-project onto simplex (should stay sum 1)
        a = np.maximum(a, 0)
        a = a / a.sum()
        # optionally adjust t decreasing
        t *= 0.8
        cur_score = evaluate_sequence(a.tolist())
        print(f"[{time.time() - start_time:6.2f}s] LP refine iter={it+1}, score={cur_score:.6f}")
    return a
```

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.
The algorithm combines three ideas:

1. **Projected Adam on a smooth approximation** of the max‑convolution.
   This quickly finds a good region of the search space.

2. **Linear‑programming refinement** (the “alternating LP” trick).
   For the current sequence a we solve
        maximise   Σ b_i
        subject to (a * b)_k ≤ M    (M = max_k (a*a)_k),  b ≥ 0
   The solution b is rescaled to a probability vector and mixed with a.
   A few LP steps sharply lower the peak of the self‑convolution.

3. **Fast greedy peak reduction** (the same O(n) move used before)
   to polish the result.

All phases respect the 1000 s time budget, keep the sequence on the
simplex (sum = 1) and never produce negative values.  The final
sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – long Adam optimisation on the smooth max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max function.
    Runs as many iterations as time permits.
    Returns the best vector (probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α → tighter approximation of max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            # log‑sum‑exp in a numerically stable way
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                        # smoothed max
            obj = 2.0 * n * f_val                  # same scale as evaluate_sequence

            # gradient (including outer factor 2·n)
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f})"
                          f" score={sc:.6f}")

            # early stop for this α if time is up
            if time.time() >= deadline:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic sub‑gradient descent (fine polishing)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_k (a*a)[k].
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}"
                  f" score={cur_score:.6f}")

        # active set (indices where conv is (almost) maximal)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – LP‑based refinement (alternating optimisation)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               max_iter: int = 5,
               mix_factor: float = 0.3) -> np.ndarray:
    """
    Repeatedly solve a linear program:
        maximise Σ b_i  s.t. (a*b)_k ≤ M, b ≥ 0,
    where M = max_k (a*a)_k.
    The solution b is normalised to sum = 1 and mixed into a.
    """
    a = a0.copy()
    n = len(a)

    # enforce a strictly positive to avoid unbounded LP
    eps_min = 1e-8
    a = np.maximum(a, eps_min)
    a /= a.sum()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # Build the (2n‑1) × n constraint matrix A where
        #    (A @ b)_k = Σ_i a_{k-i} b_i = (a*b)_k
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                rows.append(k)
                cols.append(i)
                data.append(a[k - i])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)

        # maximise Σ b_i  → minimise -Σ b_i
        c = -np.ones(n)

        try:
            res = scipy.optimize.linprog(c,
                                         A_ub=A,
                                         b_ub=b_ub,
                                         bounds=(0, None),
                                         method='highs')
        except Exception as e:
            print(f"[{time.time() - start_time:6.2f}s] LP failed: {e}")
            break

        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP not successful: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # Normalise b to a probability vector and mix
        b = b / sum_b
        a = (1.0 - mix_factor) * a + mix_factor * b
        a = np.maximum(a, eps_min)
        a /= a.sum()

        # Report progress
        cur_score = evaluate_sequence(a.tolist())
        print(f"[{time.time() - start_time:6.2f}s] LP refine iter={it+1} score={cur_score:.6f}")

        # Decrease mixing factor to stabilise
        mix_factor *= 0.8

    return a


# ----------------------------------------------------------------------
# Phase 4 – fast greedy peak reduction (as in the reference code)
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of the coefficient that most contributes to the current peak
    towards the smallest coefficient.  Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.05 * a[src]                     # transfer 5 % of source
        if eps < 1e-14:
            break

        # incremental update of convolution
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept improvement
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: iter={it:,}"
                  f" max={new_max:.6f}")

    return best_a, best_score

def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Use a historic best sequence if available, otherwise fall back to uniform."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation to the new length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)

# Main driver

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 2500

    # ------------------------------------------------------------------
    # 1) initialise
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smooth max (≈60 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.55 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    best_a, best_score = _adam_optimize_long(a0, explore_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Sub‑gradient polishing (≈25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = min(deadline, time.time() + 0.25 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, _ = _subgrad_optimize(best_a, subgrad_deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished")

    # ------------------------------------------------------------------
    # 4) LP‑based refinement (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = min(deadline, time.time() + 0.10 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start LP refinement")
        best_a = _lp_refine(best_a, lp_deadline, start_time, max_iter=6, mix_factor=0.35)
        print(f"[{time.time() - start_time:6.2f}s] LP refinement finished")

    # ------------------------------------------------------------------
    # 5) Fast greedy peak reduction (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start fast greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] greedy phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
