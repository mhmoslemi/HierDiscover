"""
Propose a step‑function (non‑negative list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

for an arbitrary length n (default 4096).  

The algorithm consists of three stages:

1️⃣ **Fast parametric sweep** – simple analytic families (geometric,
   power, Beta‑type and centred Gaussian) are generated, normalised to
   unit sum and evaluated with an FFT‑based convolution.  The best
   candidate is kept.

2️⃣ **Exponentiated‑gradient optimisation with a smoothed max** – starting
   from the best candidate, we minimise the exact objective by using a
   smooth approximation of the max (soft‑max with a temperature τ that
   anneals to zero).  The gradient of the smoothed max is obtained
   analytically and evaluated by a second FFT.  The update
   ``a ← a·exp(−η·grad)`` (mirror descent on the simplex) automatically
   respects non‑negativity; the vector is renormalised after each step.
   The step size η decays as 1/√t and τ decays exponentially from
   5·10⁻² to 5·10⁻⁴.

3️⃣ **Light mass‑transfer simulated‑annealing polish** – a short
   “move‑mass‑between‑coordinates’’ annealer (identical to the one used
   in the reference solution) finishes the search.

All helpers are top‑level, no closures or lambdas are used.
The routine respects a 1000 s wall‑clock budget and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (unit‑sum → 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (must be non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed max and its gradient (used by exponentiated‑gradient stage)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray,
                         tau: float,
                         L_self: int,
                         L_grad: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where
        conv = a * a   (full linear convolution, length 2·n−1)
        grad = ∇_a (soft‑max_τ(conv)), i.e. the gradient of the
               smoothed maximum of the convolution.
    """
    n = a.shape[0]

    # 1. self‑convolution (exact)
    conv = _conv_self(a, L_self)                 # shape 2·n−1

    # 2. soft‑max weights (stable)
    scaled = conv / tau
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / np.sum(exp_vals)              # weights, sum to 1

    # 3. gradient = 2 * (w  ⋆  a_rev)  (⋆ = linear convolution)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_grad)
    A_rev = np.fft.rfft(a_rev, n=L_grad)
    conv_rev_full = np.fft.irfft(W * A_rev, n=L_grad)
    grad = 2.0 * conv_rev_full[n - 1 : n - 1 + n]   # slice length n

    return conv, grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (smooth max, simplex mirror descent)
# ----------------------------------------------------------------------
def _eg_optimize(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using the smoothed max.
    The temperature τ is annealed exponentially from 5e‑2 to 5e‑4.
    Step size η decays as η0 / sqrt(t) (η0≈5.0).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    it = 0
    eta0 = 5.0                         # initial step size for EG
    tau_start = 5e-2
    tau_end = 5e-4

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        progress = (time.time() - start_time) / (deadline - start_time)
        tau = tau_start * ((tau_end / tau_start) ** progress)

        conv, grad = _smooth_max_and_grad(a, tau, L_self, L_grad)

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)  # sum(a) ≈ 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it:,} τ={tau:.5e} score={cur_score:.6f}")

        # 1/√t step scaling
        eta = eta0 / math.sqrt(it)

        # mirror‑descent update (exponentiated gradient)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional tiny random perturbation to escape plateaus
        if it % 10000 == 0:
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Light mass‑transfer simulated‑annealing polish (same as reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves, Metropolis acceptance)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportional to its mass, and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects a wall‑clock budget (default 1000 s)
    and returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid hard timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size (same as the reference solution)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # try to use the global starter if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric, power, Beta, Gaussian)
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # --- geometric family ------------------------------------------------
    eps_grid = np.logspace(-4, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --- power family ---------------------------------------------------
    p_grid = np.linspace(-1.0, 2.0, 61)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # --- Beta‑type family -----------------------------------------------
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # --- centred Gaussian family ----------------------------------------
    centre = (n - 1) / 2.0
    sigma_grid = np.logspace(np.log10(30.0), np.log10(2000.0), 30)
    for sigma in sigma_grid:
        if time.time() >= deadline - 30:
            break
        gauss = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        if gauss.sum() == 0:
            continue
        gauss /= gauss.sum()
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exponentiated‑gradient optimisation (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:      # keep a few seconds for polishing
        sub_deadline = deadline - 10.0
        a_opt, sc_opt = _eg_optimize(best_seq, n, sub_deadline,
                                      start_time, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] EG final improve score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline,
                                       start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
