"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The implementation follows a *continuation* strategy:

1.  Initialise a on the probability simplex (use the previously found
    ``height_sequence_1`` if available, otherwise uniform).

2.  Repeatedly minimise a **smoothed** version of the max‑convolution
    using Adam with Euclidean projection onto the simplex.
    The smoothed max is a log‑sum‑exp of the autocorrelation
    ``conv = a*a`` with a temperature ``τ``.
    ``τ`` is slowly increased (continuation) so the optimisation
    first sees a smooth landscape and finally concentrates on the true
    ``max``.

3.  When the time budget is almost exhausted a short sub‑gradient
    phase (exact max) fine‑tunes the best vector.

All linear algebra uses FFT‑based convolutions, so each iteration is
~O(n log n).  The routine respects the global 1000 s budget and returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray,
                   fft_len_self: int,
                   len_self: int) -> np.ndarray:
    """Linear convolution a * a using FFT (real input)."""
    A = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(A * A, n=fft_len_self)[:len_self]
    return conv


def _conv_w_a_fft(w: np.ndarray,
                  a: np.ndarray,
                  fft_len_conv: int,
                  len_conv: int,
                  n: int) -> np.ndarray:
    """
    Compute the gradient term 2 * (w * a)_{k} where * is linear convolution.
    Returns the slice of length n corresponding to indices k = 0..n-1.
    """
    W = np.fft.rfft(w, n=fft_len_conv)
    A = np.fft.rfft(a, n=fft_len_conv)
    full = np.fft.irfft(W * A, n=fft_len_conv)[:len_conv]
    # The needed slice is centred at index n‑1
    grad = full[n - 1:n - 1 + n]
    return grad


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Stable soft‑max for the smoothed maximum."""
    max_c = conv.max()
    # shift by max to avoid overflow/underflow
    shifted = tau * (conv - max_c)
    # use np.exp safely
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:                       # numerically impossible, but guard
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


# -------------------------------------------------------------
# Main optimisation routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.

    The method uses a smoothed‑max Adam optimisation with a continuation
    on the temperature parameter τ and a final sub‑gradient polishing step.
    The function respects the supplied time budget (seconds) and returns
    a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee returning before the external timeout
    deadline = start_time + budget_s - 0.3

    # -----------------------------------------------------------------
    # Problem size – 2048 works well for the time budget
    # -----------------------------------------------------------------
    n = 2048

    # -----------------------------------------------------------------
    # Initialise a on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                if tmp.size != n:
                    # linear interpolation to required length
                    old_x = np.linspace(0.0, 1.0, tmp.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    tmp = np.interp(new_x, old_x, tmp)
                a = tmp.copy()
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # Normalise to sum = 1 (the objective is scale invariant)
    a = a.astype(float)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    # -----------------------------------------------------------------
    # Pre‑compute FFT lengths
    # -----------------------------------------------------------------
    len_self = 2 * n - 1
    fft_len_self = _next_pow2(len_self)

    len_conv = len_self + n - 1            # = 3·n‑2
    fft_len_conv = _next_pow2(len_conv)

    # -----------------------------------------------------------------
    # Book‑keeping
    # -----------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len_self, len_self)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate (adjusted by Adam)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    # Continuation parameters for the temperature τ
    tau = 1.0
    tau_max = 80.0
    tau_inc_every = 2500          # increase τ every this many iterations
    tau_factor = 1.10

    it = 0
    # -----------------------------------------------------------------
    # Stage 1 – Smoothed‑max Adam optimisation (continuation on τ)
    # -----------------------------------------------------------------
    while time.time() < deadline - 1.5:        # keep some time for polishing
        it += 1

        # --- convolution and true objective ---------------------------------
        conv = _conv_self_fft(a, fft_len_self, len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter {it:<6d} "
                  f"improve → {cur_score:.6f}")

        # --- smoothed max (log‑sum‑exp) ------------------------------------
        w = _softmax_weights(conv, tau)          # weights sum to 1

        # --- gradient of the smoothed objective ----------------------------
        # g = 2 * (w * a)  (convolution)  – slice of length n
        grad = 2.0 * _conv_w_a_fft(w, a, fft_len_conv, len_conv, n)

        # --- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection onto the simplex (ensures a ≥ 0, sum = 1)
        a = _project_simplex(a_new)

        # --- continuation on τ ---------------------------------------------
        if it % tau_inc_every == 0:
            tau = min(tau * tau_factor, tau_max)

        # update power terms for bias correction
        beta1_pow *= beta1
        beta2_pow *= beta2

        # quick time check
        if time.time() >= deadline - 1.5:
            break

    # -----------------------------------------------------------------
    # Stage 2 – Exact max sub‑gradient polishing (momentum, decreasing step)
    # -----------------------------------------------------------------
    # Use a simple momentum sub‑gradient scheme with a diminishing step size.
    # This stage is cheap and can squeeze out the last bits of improvement.
    momentum = np.zeros_like(a)
    polish_deadline = deadline - 0.2
    polish_it = 0
    while time.time() < polish_deadline:
        polish_it += 1
        conv = _conv_self_fft(a, fft_len_self, len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] polish {polish_it:<6d} "
                  f"improve → {cur_score:.6f}")

        # sub‑gradient of max(conv) (averaged over active indices)
        max_val = cur_max
        eps_sub = max(1e-12 * max_val, 0.0)
        active = np.where(conv >= max_val - eps_sub)[0]

        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            i = k - idx
            mask = (i >= 0) & (i < n)
            grad[mask] += 2.0 * a[i[mask]]
        if active.size > 1:
            grad /= active.size

        # momentum + diminishing step
        momentum = 0.95 * momentum + 0.05 * grad
        step = 0.10 / math.sqrt(polish_it + 1.0)
        a = _project_simplex(a - step * momentum)

    # -----------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    if best_seq.sum() > 0.0:
        best_seq /= best_seq.sum()
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_score = 2.0 * n * _conv_self_fft(best_seq, fft_len_self, len_self).max()
    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
