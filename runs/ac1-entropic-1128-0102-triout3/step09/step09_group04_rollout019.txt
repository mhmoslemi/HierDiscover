"""
A fast “deterministic + stochastic” optimizer for the max‑convolution objective.

The function `propose_candidate` builds a non‑negative step function `a`
(normalised to sum = 1) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

which is exactly the value returned by ``evaluate_sequence`` on a
list of floats.  The optimiser works in three phases, fitting comfortably
into the 1000 s budget:

1. **Optional LP balancing** – if ``cvxpy`` is available we solve a single
   linear programme that pushes mass onto positions that are “inactive”
   for the current peak.  A short line‑search finds the best convex
   combination of the current vector and the LP solution.

2. **Projected sub‑gradient descent** – we minimise the true non‑smooth
   objective using Euclidean sub‑gradients and Euclidean projection onto
   the probability simplex.  A diminishing step size (∝ 1/√t) yields fast
   descent in the first few seconds.

3. **Simulated‑annealing mass‑swap** – the remaining time is spent on a
   simple Markov‑Chain Monte‑Carlo search.  In each iteration a random
   amount of mass is moved from one index to another; the move is always
   accepted if it improves the score, otherwise with a temperature‑controlled
   probability.  The temperature and the maximal transferred fraction decay
   linearly, allowing large exploratory jumps early on and fine‑grained
   moves near the end.  The best vector seen throughout the whole run is
   returned.

All operations are vectorised with NumPy; the convolution is evaluated
with ``np.convolve`` (fast for the modest sizes used here).  The code
conforms to the required API and prints progress messages.
"""
import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length) using the C‑implemented np.convolve."""
    return np.convolve(a, a)


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a (a ≥ 0).
    For every index attaining the current maximum the contribution
    2·a_{k−i} is added to the gradient entry i.  The result is averaged
    over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation function.  The routine respects the supplied time budget
    (seconds) and prints a short progress log.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – we must finish before the external timeout.
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # Choose problem size and initialise the vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                a = tmp.copy()
                # Normalise to the probability simplex.
                s = a.sum()
                if s > 0:
                    a /= s
                else:
                    a = np.full_like(a, 1.0 / a.size)
            else:
                raise ValueError
        except Exception:
            a = np.full(2048, 1.0 / 2048)
    else:
        a = np.full(2048, 1.0 / 2048)

    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    print(f"[{time.time() - start_time:6.2f}s] initial score {cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – optional LP balancing (single cheap iteration)
    # --------------------------------------------------------------
    try:
        import cvxpy as cp
        lp_time = (deadline - time.time()) * 0.1   # give it ~10 % of remaining budget
        lp_deadline = time.time() + lp_time

        if time.time() < lp_deadline:
            # Build convolution matrix T (2n‑1)×n
            T = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                T[i:i + n, i] = a

            M = conv.max()
            b_var = cp.Variable(n, nonneg=True)
            constraints = [T @ b_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=cp.ECOS, max_iters=300, warm_start=True)

            if b_var.value is not None:
                b = np.maximum(b_var.value, 0.0)
                if b.sum() > 0:
                    g = b / b.sum()
                    # simple line search on t∈[0,1]
                    best_local = a
                    best_local_score = cur_score
                    for t in np.linspace(0.0, 1.0, 11):
                        cand = (1.0 - t) * a + t * g
                        cand = _project_simplex(cand)
                        cand_score = _score(cand)
                        if cand_score < best_local_score:
                            best_local_score = cand_score
                            best_local = cand
                    a = best_local
                    conv = _conv_self(a)
                    cur_score = _score(a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] LP improvement → {cur_score:.6f}")
    except Exception:
        # cvxpy not present – simply skip the LP stage.
        pass

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (momentum, 20 % budget)
    # --------------------------------------------------------------
    sub_budget = (deadline - time.time()) * 0.2
    sub_deadline = time.time() + sub_budget
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < sub_deadline:
        it += 1
        grad = _subgradient_max(a, conv)
        momentum = 0.94 * momentum + 0.06 * grad
        step = 0.15 / math.sqrt(it + 1.0)          # diminishing step
        a_new = a - step * momentum
        a_new = _project_simplex(a_new)
        conv_new = _conv_self(a_new)
        new_score = _score(a_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it:<6d} improve → {new_score:.6f}")

        a = a_new
        conv = conv_new
        cur_score = new_score

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing random mass swaps (rest of time)
    # --------------------------------------------------------------
    # Temperature parameters – they shrink linearly with remaining time.
    total_remaining = deadline - time.time()
    T0 = best_score * 0.005  # start with a tiny temperature (tuned empirically)

    it = 0
    while time.time() < deadline:
        it += 1
        # remaining fraction controls both temperature and maximal jump size
        now = time.time()
        remaining_frac = (deadline - now) / (deadline - start_time)

        # temperature decays linearly
        T = T0 * remaining_frac

        # choose two distinct positions
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # maximal transferable fraction (larger early, tiny later)
        max_frac = 0.20 * remaining_frac + 0.001  # stays >0
        delta = max_frac * np.random.rand() * a[i]

        if delta <= 0.0:
            continue

        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # Exact convolution for the candidate (precision matters)
        conv_cand = _conv_self(a_cand)
        cand_score = _score(a_cand)

        # Acceptance rule (Metropolis)
        if cand_score < cur_score:
            accept = True
        else:
            # probability = exp(-(Δ)/T)
            prob = math.exp(-(cand_score - cur_score) / (T + 1e-16))
            accept = np.random.rand() < prob

        if accept:
            a = a_cand
            conv = conv_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{now - start_time:6.2f}s] SA iter {it:<6d} improve → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (ensuring sum=1) and conversion to plain list.
    # ------------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0.0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
