"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of four phases:
1. Warm‑start from the historic best (or uniform) and a few random
   Dirichlet restarts.
2. Long Adam optimisation on a smooth approximation of the max‑convolution.
3. Projected sub‑gradient polishing.
4. Fast greedy refinement that transfers a large fraction of the most
   responsible coefficient to a low‑impact destination, using a back‑tracking
   line‑search on the convolution max.

All phases respect the 1000 s total budget.  The final sequence is
normalised to sum = 1 before the final evaluation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of th​e smoothed max
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on smooth max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max objective.
    Runs until the supplied deadline.
    Returns the best simplex vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # the schedule gradually sharpens the smooth‑max approximation
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0,
                      512.0, 2048.0, 8192.0, 32768.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.14
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        while time.time() < deadline:
            # -----------------------------------------------------------
            # compute convolution, smooth max value and gradient
            # -----------------------------------------------------------
            conv = _conv_self(a)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                     # smoothed max
            # gradient of the *raw* smooth max (without outer 2·n factor)
            grad_raw = _grad_smooth_max(a, conv, α, fft_len)
            # full gradient of the true objective 2·n·max(...)
            grad = grad_raw * (2.0 * n)

            # -----------------------------------------------------------
            # Adam update and projection onto simplex
            # -----------------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f}) "
                          f"score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic sub‑gradient polishing
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective max_k (a*a)[k].
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,} "
                  f"score={cur_score:.6f}")

        # active set (values within 1e-12 relative tolerance of the peak)
        thresh = cur_max * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the active set (averaged)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – fast greedy refinement with line‑search
# ----------------------------------------------------------------------


def _greedy_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a sizeable fraction of the coefficient that contributes
    most to the current convolution peak towards a low‑impact destination.
    A back‑tracking line search on ε guarantees a strict reduction of the
    peak.  The convolution array is updated incrementally.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    accepted_moves = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = int(conv.argmax())
        cur_max = conv[k_max]

        # indices that participate in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        prod = a[i_idx] * a[j_idx]
        src = i_idx[prod.argmax()]

        # gradient‑style impact of each index on the current peak
        impact = np.zeros(n)
        impact[i_idx] = a[j_idx]               # ∂conv_k/∂a_i = a_{k-i}
        impact[src] = np.inf                    # exclude the source temporarily
        dst = np.argmin(impact)                 # destination with smallest impact

        # try to move a large fraction of src; fallback to halving if not improving
        eps = a[src] * 0.5
        if eps < 1e-14:
            break

        accepted = False
        eps_try = eps
        while eps_try > 1e-14:
            # incremental convolution update (linear + quadratic terms)
            conv_tmp = conv.copy()
            conv_tmp[src:src + n] -= 2.0 * eps_try * a
            conv_tmp[dst:dst + n] += 2.0 * eps_try * a
            conv_tmp[2 * src] += eps_try * eps_try
            conv_tmp[2 * dst] += eps_try * eps_try
            conv_tmp[src + dst] -= 2.0 * eps_try * eps_try

            new_max = conv_tmp.max()
            if new_max < cur_max - 1e-12:
                # accept the move
                conv = conv_tmp
                a[src] -= eps_try
                a[dst] += eps_try
                accepted_moves += 1
                new_score = 2.0 * n * new_max
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it:,} "
                          f"eps={eps_try:.6e} max={new_max:.6f}")
                accepted = True
                break
            else:
                eps_try *= 0.5

        if not accepted:
            # if no ε works, give up on this pair and continue
            if it % 1000 == 0:
                pass  # just an occasional checkpoint

        # every few hundred accepted moves recompute the full convolution
        # to avoid drift from rounding errors
        if accepted_moves and accepted_moves % 800 == 0:
            conv = _conv_self(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to produce an initial candidate
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Use the historic best if available, otherwise uniform."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.clip(seq, 0.0, None)
            s = seq.sum()
            if s > 0.0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main driver – obey the 1000 s budget
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to avoid exceeding the limit
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 4800

    # ------------------------------------------------------------------
    # 0) warm start / random restarts
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # occasional random Dirichlet restarts to escape local minima
    best_a = a0.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    restart_deadline = start_time + 0.20 * budget_s  # first 20 % of time
    num_restarts = 5
    for r in range(num_restarts):
        if time.time() >= restart_deadline:
            break
        # Dirichlet with random concentration (sparser or denser)
        conc = rng.uniform(0.2, 2.0)
        a_rand = rng.gamma(conc, 1.0, size=n)
        a_rand /= a_rand.sum()
        # very short Adam run (≈5 s)
        short_deadline = min(restart_deadline, time.time() + 5.0)
        a_opt, sc_opt = _adam_optimize_long(a_rand, short_deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart {r} improved to {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 1) long Adam optimisation (≈30 % of budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.30 * budget_s
    best_a, best_score = _adam_optimize_long(best_a, adam_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] after Adam best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) sub‑gradient polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = min(deadline, start_time + 0.65 * budget_s)
        best_a, sub_score = _subgrad_optimize(best_a, sub_deadline,
                                              start_time,
                                              beta=0.94,
                                              step0=0.05)
        if sub_score < best_score:
            best_score = sub_score
            best_a = best_a.copy()
        print(f"[{time.time() - start_time:6.2f}s] after sub‑grad best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fast greedy refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start greedy refinement")
        best_a, gre_score = _greedy_refine(best_a, deadline, start_time)
        if gre_score < best_score:
            best_score = gre_score
            best_a = best_a.copy()
        print(f"[{time.time() - start_time:6.2f}s] after greedy best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
