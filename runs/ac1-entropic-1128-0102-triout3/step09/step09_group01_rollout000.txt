"""
Propose a 4096‑long non‑negative step function that (approximately) minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The algorithm consists of three stages:

1️⃣  **Extensive parametric sweep** – many simple analytic families
    (geometric, power, Beta‑type, Gaussian, raised‑cosine, triangular)
    are generated, mixed with a uniform component, normalised to a
    probability vector and evaluated with a fast FFT‑based convolution.
    The best‑scoring candidate is kept.

2️⃣  **Adam‑style projected sub‑gradient descent** – from the best
    candidate we run an adaptive first‑order method on the exact
    objective.  The gradient is the sub‑gradient of the active
    convolution diagonal(s); the iterate is always projected onto the
    probability simplex.  A few random restarts with small noise are
    performed if time permits.

3️⃣  **Light simulated‑annealing polish** – a mass‑transfer annealer
    makes tiny random moves between coordinates, accepting worsening
    moves with a Metropolis probability.  This final stage cleans up
    any remaining irregularities.

The routine respects a ~1000 s wall‑clock budget, prints progress
messages and finally returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    The result is a real 1‑d array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv) )
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient (Adam) descent on the exact objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, n: int, deadline: float,
                  start_time: float, best_score_initial: float,
                  max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on f(a)=2·n·max(conv(a,a)).
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam parameters
    alpha = 0.07          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    L_grad = _next_pow2(3 * n - 2)   # safe FFT length for gradient (overkill but cheap)

    it = 0
    t_start = time.time()

    # tolerance for multiple active diagonals
    act_tol = 1e-12

    while time.time() < deadline and it < max_iter:
        it += 1

        # --------------------------------------------------------------
        # compute self‑convolution and identify active diagonals
        # --------------------------------------------------------------
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # sum(a) == 1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # indices where convolution is (almost) maximal
        act = np.where(conv >= cur_max - act_tol)[0]
        # --------------------------------------------------------------
        # sub‑gradient of max(conv) : average over active diagonals
        # --------------------------------------------------------------
        grad = np.zeros_like(a)
        for k in act:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # contribution: 2 * a[k - idx]
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(act))        # average if several active indices

        # --------------------------------------------------------------
        # Adam update  (objective is linear in a after projection)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # ----------------------------------------------------------------------
        # occasional random perturbation to escape flat regions (once per 5000 it.)
        # ----------------------------------------------------------------------
        if it % 5000 == 0:
            # add tiny Gaussian noise, keep on simplex
            noise = np.random.default_rng().normal(scale=0.001, size=n)
            a = a + noise
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray, n: int, deadline: float,
               start_time: float, cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer; moves a small fraction of mass from src → dst."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # choose source proportional to its mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of src mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

def _geometric_family(n, eps):
    """Geometric decay a_i ∝ r^i, r = 1 - eps."""
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)
    seq_sum = seq.sum()
    if seq_sum == 0:
        return None
    return seq / seq_sum


def _power_family(n, p):
    """Power law a_i ∝ (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(idx, p)
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


def _beta_family(n, p, q):
    """Discrete Beta‑type a_i ∝ (i+1)^p * (n-i)^q."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(idx + 1.0, p) * np.power(n - idx, q)
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


def _gaussian_family(n, sigma):
    """Discrete Gaussian centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


def _raised_cosine_family(n, radius, power):
    """Raised‑cosine taper: (0.5+0.5*cos(...))^power inside radius, zero outside."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    dist = np.abs(idx - mu)
    inside = dist <= radius
    # cos argument between 0 and pi
    cos_arg = np.pi * dist[inside] / radius
    seq = np.zeros_like(idx)
    seq[inside] = (0.5 + 0.5 * np.cos(cos_arg)) ** power
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


def _triangular_family(n, radius):
    """Triangular shape with base width 2·radius centred."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    dist = np.abs(idx - mu)
    seq = np.maximum(0.0, 1.0 - dist / radius)
    s = seq.sum()
    if s == 0:
        return None
    return seq / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a 4096‑dimensional non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm follows the three‑stage
    strategy described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overrun
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise: uniform distribution (score = 2.0) and any global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # Use the provided global variable ``height_sequence_1`` if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (extended families)
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # Helper to mix a candidate with uniform (lam in [0,1])
    def _mix_with_uniform(base, lam):
        mixed = lam * uniform + (1.0 - lam) * base
        mixed = np.maximum(mixed, 0.0)
        mixed /= mixed.sum()
        return mixed

    # Geometric family (eps from 1e-5 to 5e-1)
    eps_grid = np.logspace(-5, -0.3, 140)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        geo = _geometric_family(n, eps)
        if geo is None:
            continue
        for lam in np.linspace(0.0, 0.6, 7):
            cand = _mix_with_uniform(geo, lam)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # Power family (p from -2.5 to 5.0)
    p_grid = np.linspace(-2.5, 5.0, 151)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pw = _power_family(n, p)
        if pw is None:
            continue
        sc = _eval_fast(pw, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pw.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # Beta‑type family (p,q up to 6)
    beta_vals = np.arange(0.0, 6.5, 0.5)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = _beta_family(n, p, q)
            if beta_seq is None:
                continue
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # Gaussian family (sigma from 0.04·n to 0.5·n)
    sigma_factors = np.linspace(0.04, 0.5, 13)
    for sf in sigma_factors:
        if time.time() >= deadline - 30:
            break
        sigma = sf * n
        gauss = _gaussian_family(n, sigma)
        if gauss is None:
            continue
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian sigma={sigma:.0f} score={sc:.6f}")

    # Raised‑cosine family (radius from 0.04·n to 0.5·n, power 1–4)
    rad_factors = np.linspace(0.04, 0.5, 12)
    pow_vals = [1, 2, 3, 4]
    for radf in rad_factors:
        radius = radf * n
        for pw in pow_vals:
            if time.time() >= deadline - 30:
                break
            rc = _raised_cosine_family(n, radius, pw)
            if rc is None:
                continue
            sc = _eval_fast(rc, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = rc.copy()
                print(f"[{time.time() - start_time:.2f}s] RaisedCos rad={radius:.0f} pow={pw} score={sc:.6f}")

    # Triangular family (radius from 0.04·n to 0.5·n)
    for radf in rad_factors:
        if time.time() >= deadline - 30:
            break
        radius = radf * n
        tri = _triangular_family(n, radius)
        if tri is None:
            continue
        sc = _eval_fast(tri, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = tri.copy()
            print(f"[{time.time() - start_time:.2f}s] Triangular rad={radius:.0f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style projected sub‑gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:  # keep a few seconds for final polish
        sub_deadline = deadline - 10.0
        a_adam, sc_adam = _adam_descent(best_seq, n, sub_deadline,
                                        start_time, best_score)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam final improve score={sc_adam:.6f}")

        # Optional quick restart with a little noise if time permits
        if time.time() < sub_deadline - 5.0:
            rng = np.random.default_rng()
            for restart in range(3):
                if time.time() >= sub_deadline - 1.0:
                    break
                pert = best_seq + rng.normal(scale=0.005, size=n)
                pert = _project_simplex(pert)
                a_restart, sc_restart = _adam_descent(pert, n, sub_deadline,
                                                      start_time, best_score,
                                                      max_iter=500_000)
                if sc_restart < best_score:
                    best_score = sc_restart
                    best_seq = a_restart.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam restart {restart+1} improve score={sc_restart:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (use the leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=2e-2,
                                       end_temp=1e-7,
                                       max_steps=350_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time, ensure non‑negativity and return plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
