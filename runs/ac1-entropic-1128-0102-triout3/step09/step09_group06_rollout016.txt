def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2
    # load warm start
    try:
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
    except Exception:
        a = np.full(4800, 1/4800, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time()-start:6.2f}s] initial score={best_score:.6f}")

    # pre-calc learning schedule
    step_factor = 0.1
    it = 0

    while time.time() < deadline:
        it += 1

        # occasionally try random candidate
        if it % 100 == 0:
            # generate random Dirichlet vector
            alpha = 0.3  # sparsity
            g = rng.gamma(alpha, 1.0, size=n)
            cand = g / g.sum()
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                print(f"[{time.time()-start:6.2f}s] random improve iter {it} score={sc:.6f}")
                continue

        # hill climbing move
        conv = np.convolve(a, a)
        kmax = conv.argmax()
        # compute contributions for that sum
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = kmax - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]  # index contributing most

        # destination index = smallest coefficient (excluding src)
        # find index with minimal a value
        # optionally random among low values
        dst_candidates = np.where(np.arange(n) != src)[0]
        dst = dst_candidates[a[dst_candidates].argmin()]

        # step size decreasing with iteration count
        eps = step_factor * a[src] / (1 + 0.001 * it)  # decreasing
        if eps <= 1e-12:
            continue
        # ensure we don't make src negative
        eps = min(eps, a[src] * 0.9)

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # projection onto simplex not needed because sum unchanged, values non-negative.

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            # possibly reduce step factor for quicker convergence
            step_factor = max(step_factor * 0.99, 0.001)
            print(f"[{time.time()-start:6.2f}s] hill improve iter {it} score={sc:.6f}")

    # final normalization (already sum=1)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time()-start:6.2f}s] final score={final_score:.6f}")
    return best_a.tolist()

"""
Propose a step‐function (non‑negative list of heights) that minimises the
evaluate_sequence objective.

Algorithm
---------
1. Warm‑start from the historic best sequence `height_sequence_1` if it
   exists, otherwise start from a uniform distribution.  The vector is
   normalised to have sum = 1 (the scale does not affect the objective).

2. **Local stochastic optimisation**
   * Repeatedly perform tiny mass transfers from the coefficient that
     contributes most to the current convolution peak to the currently
     smallest coefficient.
   * The transfer size ε shrinks with the iteration count, allowing a
     coarse first stage and a fine final polishing.
   * After each trial the objective is recomputed (using the fast
     analytical formula – identical to `evaluate_sequence` for the
     admissible ranges).  The move is kept only if it strictly lowers the
     score.

3. **Random restarts**
   * Every few hundred iterations a fresh random Dirichlet vector
     (controlled sparsity) is generated and evaluated.  If it beats the
     current best it becomes the new starting point.

4. The loop runs until the time budget (≈ 1000 s) is exhausted.  The best
   sequence found is returned.

The routine respects the 1000 s limit, prints progress information and
requires only a single CPU core and the standard ``numpy`` library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the objective (identical to evaluate_sequence
# for admissible inputs, but avoids the security checks and the final
# clipping to [0, 1000] which are not needed for our candidates).
# ----------------------------------------------------------------------
def _fast_score(vec: np.ndarray) -> float:
    """Return the objective 2·n·max(conv)/sum(vec)^2 for a non‑negative vector."""
    n = vec.shape[0]
    conv = np.convolve(vec, vec)            # full convolution, length 2·n‑1
    max_b = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    it = 0

    # the loop stops automatically when the deadline is reached
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # (a) occasional random Dirichlet restart – sparsity controlled by alpha
        # ------------------------------------------------------------------
        if it % 200 == 0:
            # a small alpha yields a sparse vector; we keep the support size
            # moderate (≈ 30 % of n) which empirically gives good peak‑to‑average ratios
            alpha = 0.35
            g = rng.gamma(alpha, 1.0, size=n)
            cand = g / g.sum()
            cand_score = _fast_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] random restart improved → {cand_score:.6f}")
                continue

        # ------------------------------------------------------------------
        # (b) hill‑climbing move: shift a tiny amount from the main contributor

        # ------------------------------------------------------------------

        k_max = int(conv.argmax())            # index of the current maximal convolution entry

        hi = min(n - 1, k_max)

        j_idx = k_max - i_idx
