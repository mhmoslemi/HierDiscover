    alpha = 0.2  # base step size
    t = 0

        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()

        t += 1

def _adam_optimize(a, deadline, max_iter=2000):

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_alpha = 0.5  # step size base
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        grad = _subgradient_max(a, conv)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_alpha / np.sqrt(t)  # decaying
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)
        a = _symmetrize(a)
    return best_a, best_score

def _coordinate_optimize(a, deadline, max_iter=5000):
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        conv = _conv_self(a)
        cur_score = 2 * n * conv.max()
        grad = _subgradient_max(a, conv)
        # pick indices
        i_max = np.argmax(grad)
        # pick i_min with smallest grad (or minimal a to fill)
        i_min = np.argmin(grad)
        # compute delta as fraction of a[i_max], ensure > 0
        if a[i_max] <= 0:
            continue
        # delta fraction randomly between 0.01 and 0.2 of a[i_max]
        delta = a[i_max] * np.random.uniform(0.02, 0.3)
        # ensure not too big
        delta = min(delta, 0.05)  # bound absolute delta
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # after move project onto simplex and symmetrize
        a_new = _project_simplex(a_new)
        a_new = _symmetrize(a_new)
        # compute new score
        conv_new = _conv_self(a_new)
        new_score = 2 * n * conv_new.max()
        if new_score < cur_score - 1e-12:  # improvement
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        else:

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a simple stochastic meta‑heuristic:

1.  Load the best known construction (``height_sequence_1``) if it exists,
    otherwise start from the uniform vector.
2.  Refine the vector locally with an Adam‑style projected sub‑gradient
    descent (the sub‑gradient of the max‑convolution is cheap to compute).
3.  Perform a lightweight coordinate‑wise mass‑transfer optimisation
    (move a fraction of mass from the point that contributes most to the
    current peak to the point that contributes least).
4.  Repeatedly launch short “restarts’’:
    – sample a random Dirichlet start,
    – mix it with the current best,
    – run a short Adam + coordinate phase,
    – keep any improvement.
5.  All iterates are kept on the probability simplex and symmetrised
    (the objective is symmetric, so restricting to symmetric vectors
    reduces the search space).

The routine respects the supplied time budget (default 1000 s) and returns
a plain list of floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (no external dependencies)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    The gradient is an average over all indices attaining the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force the vector to be symmetric (a[i] = a[n‑1‑i])."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a_init: np.ndarray,
                   deadline: float,
                   max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the non‑smooth objective (sub‑gradient of max‑conv).
    Returns the best vector seen and its score.
    """
    a = a_init.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.5          # base learning rate
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        grad = _subgradient_max(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_alpha / math.sqrt(t + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


def _coordinate_optimize(a_init: np.ndarray,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple coordinate‑wise mass‑transfer: move a fraction of the mass from
    the index that contributes most to the current peak to the index that
    contributes the least.  Accept only improving moves.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        grad = _subgradient_max(a, conv)

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if a[i_max] <= 0:
            continue

        # random fraction of the mass at i_max, limited to keep steps modest
        delta = a[i_max] * np.random.uniform(0.02, 0.25)
        delta = min(delta, 0.05)          # absolute cap

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        a_new = _project_simplex(a_new)
        a_new = _symmetrize(a_new)

        conv_new = _conv_self(a_new)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score - 1e-12:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # occasional random jitter to escape flat regions
            if np.random.rand() < 0.001:
                a = a + np.random.normal(scale=0.001, size=n)
                a = np.clip(a, 0, None)
                a = _project_simplex(a)
                a = _symmetrize(a)

    return best_a, best_score
