"""
A fast gradient‑based optimizer for the step‑function objective.

The algorithm works in three phases inside the global time budget:

1. **Initialisation** – start from several plausible vectors:
   the previously discovered construction (if any),
   the uniform distribution and a couple of random Dirichlet/exponential draws.
   All vectors are interpolated to a common size `n` and projected onto the
   probability simplex (non‑negative, sum = 1).

2. **Adam on a smoothed max** – minimise a smooth surrogate
   `f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)`.
   Its gradient can be written as a single weighted convolution and is
   evaluated efficiently with the FFT.  The smoothing parameter `α`
   is increased geometrically (starting from 5) so that the optimisation
   gradually focuses on the true peak of the autocorrelation.
   Adam’s adaptive learning rate together with Euclidean projection
   quickly drives the sequence toward a flat convolution.

3. **Polishing with a sub‑gradient step** – after the smoothed phase,
   a few iterations of the true sub‑gradient descent (as used in the
   reference code) clean up the remaining tiny peaks.

The best vector found over all starting points is returned as a plain
Python list, ready to be handed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (no external dependencies)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    For all k attaining the maximum we add the contribution of a to the
    gradient; the result is averaged over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _weighted_convolution(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the vector   g_j = Σ_i w_{i+j}·a_i   (j = 0 … n‑1)
    efficiently via a single FFT convolution.
    """
    n = a.shape[0]
    len_w = w.shape[0]          # = 2·n‑1
    fft_len = _next_pow2(n + len_w - 1)

    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w[::-1], n=fft_len)      # reversed w
    full = np.fft.irfft(A * W, n=fft_len)

    offset = len_w - 1
    idx = offset - np.arange(n)               # indices [offset, offset‑1, …]
    return full[idx]


def _smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    with respect to a.  The gradient is
        2·W·a / Σ_k w_k,
    where w_k = exp(α·(a∗a)_k) and (W·a)_j = Σ_i w_{i+j}·a_i.
    """
    conv = _conv_self(a)
    max_conv = conv.max()
    # numerical stability – subtract the maximum before exponentiation
    w = np.exp(alpha * (conv - max_conv))
    Z = w.sum()
    if Z == 0 or np.isclose(Z, 0.0):
        # fall back to the non‑smoothed sub‑gradient
        return _subgradient_max(a, conv)

    term = _weighted_convolution(a, w)
    grad = 2.0 * term / Z
    return grad


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   global_start: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective, with a geometric
    increase of the smoothing parameter α.  Returns the best vector and
    its true objective value (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    t = 0
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    # smoothing schedule
    alpha = 5.0
    alpha_factor = 1.5
    change_every = 2000  # increase α every this many iterations

    while time.time() < deadline:
        t += 1
        if t % change_every == 0:
            alpha = min(alpha * alpha_factor, 500.0)

        grad = _smooth_max_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # Evaluate the true objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - global_start:6.2f}s] ADAM iter {t:<6d} improve → {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Optional polishing with the true (non‑smoothed) sub‑gradient.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - global_start:6.2f}s] sub‑grad polish {it:<6d} improve → {cur_score:.6f}")

            grad = _subgradient_max(a, conv)
            step = 0.2 / np.sqrt(it)
            a = a - step * grad
            a = _project_simplex(a)

    return best_a, best_score


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine explores several
    initialisations and runs a fast Adam optimiser that minimises a
    smooth approximation of the peak of the autocorrelation, followed
    by a brief sub‑gradient polishing phase.  The total runtime never
    exceeds ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -------------------------------------------------------------
    # Problem size – 4096 provides enough degrees of freedom without
    # making each iteration expensive.
    # -------------------------------------------------------------
    n = 4096

    # -------------------------------------------------------------
    # Build a list of starting vectors.
    # -------------------------------------------------------------
    starts = []

    # 1. previous construction (if available)
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=float)
            if old.ndim == 1 and old.size > 0:
                if old.size != n:
                    old_x = np.linspace(0.0, 1.0, old.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    old = np.interp(new_x, old_x, old)
                old = np.maximum(old, 0.0)
                if old.sum() == 0.0:
                    old = np.full(n, 1.0 / n)
                else:
                    old = old / old.sum()
                starts.append(old)
        except Exception:
            pass

    # 2. uniform distribution
    starts.append(np.full(n, 1.0 / n))

    # 3. random Dirichlet (actually uniform on simplex)
    rng = np.random.default_rng(seed + 1)
    rand_dir = rng.random(n)
    if rand_dir.sum() == 0.0:
        rand_dir = np.full(n, 1.0 / n)
    else:
        rand_dir = rand_dir / rand_dir.sum()
    starts.append(rand_dir)

    # 4. random exponential (heavy‑tailed)
    rng2 = np.random.default_rng(seed + 2)
    rand_exp = rng2.exponential(scale=1.0, size=n)
    if rand_exp.sum() == 0.0:
        rand_exp = np.full(n, 1.0 / n)
    else:
        rand_exp = rand_exp / rand_exp.sum()
    starts.append(rand_exp)

    # -----------------------------------------------------------------
    # Global bookkeeping
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    remaining_starts = len(starts)

    for i, a0 in enumerate(starts, 1):
        # recompute remaining time and allocate equally among remaining starts
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        per_start_deadline = time.time() + time_left / remaining_starts
        remaining_starts -= 1

        print(f"[{time.time() - start_time:6.2f}s] Start {i}/{len(starts)} – "
              f"allocated {time_left/remaining_starts if remaining_starts>0 else time_left:.2f}s")

        # Run the optimiser on this start
        a_opt, score_opt = _optimize_adam(a0, per_start_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] New global best → {best_score:.6f}")

    # -----------------------------------------------------------------
    # Fallback in the (unlikely) event that nothing succeeded.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # -----------------------------------------------------------------
    # Normalise once more (to be safe) and return as a plain list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
