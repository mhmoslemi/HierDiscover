def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096

    # grid points at centers:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    best_seq = None
    best_score = float('inf')
    # parameter sweep for exponent alpha
    alphas = np.linspace(0.2, 0.6, 21)  # includes arcsine 0.5
    for alpha in alphas:
        # compute weights (x*(1-x))**(-alpha)
        w = (x * (1.0 - x)) ** (-alpha)
        # normalise to sum 1
        a = w / w.sum()
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] alpha={alpha:.3f} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # add fallback to height_sequence_1 if defined
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
    except Exception:
        pass

    # if we didn't get a seq (unlikely) fallback uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_score = _objective(best_seq)

    # Local refinement: projected subgradient descent for remaining time
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate 80% of remaining time for refinement
        sub_deadline = time.time() + 0.8 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # optional short annealing for final polish if time remains
    if time.time() < deadline:
        a = best_seq.copy()
        curr = _objective(a)
        best_local = a.copy()
        best_local_score = curr
        rng = np.random.default_rng()
        steps = 20000
        for it in range(steps):
            if time.time() >= deadline:
                break
            # propose a small transfer
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = a[src] * rng.random() * 0.1  # small transfer
            a2 = a.copy()
            a2[src] -= eps
            a2[dst] += eps
            new = _objective(a2)
            if new < curr or rng.random() < 0.001:
                a = a2
                curr = new
                if new < best_local_score:
                    best_local_score = new
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Anneal iter={it} best={new:.6f}")
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # Ensure nonnegative and normalise to sum 1
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)

    # Return Python list
    return [float(v) for v in best_seq]
