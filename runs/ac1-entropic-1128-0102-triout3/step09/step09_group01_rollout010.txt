def _smooth_descent(a0, n, deadline, start_time, best_seq, best_score, L_conv, L_corr, final_time=30.0):
    a = a0.copy()
    best_a = best_seq.copy()
    best_sc = best_score
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0
    # schedule for α (smoothness)
    while time.time() < deadline - final_time:
        it += 1
        conv = _conv_self(a, L=L_conv)
        M = conv.max()
        sc = 2.0 * n * M / (a.sum()**2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={sc:.6f}")

        # gradient (softmax weighting)
        # Use a moderate alpha schedule
        if it < 5000:
            alpha = 50.0
        elif it < 20000:
            alpha = 200.0
        elif it < 80000:
            alpha = 500.0
        else:
            alpha = 1000.0

        # compute softmax weights
        # conv may be up to M, subtract M for stability.
        # w = np.exp(alpha * (conv - M))
        # small numbers maybe vanish; okay.
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to subgradient: use only max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        p = w / w_sum

        # correlation gradient via FFT
        # compute real FFT; note rfft returns N/2+1 complex values.
        # however correlation using rfft may need conj; we can use np.fft.rfft.
        Fa = np.fft.rfft(a, n=L_corr)
        Fp = np.fft.rfft(p, n=L_corr)
        g_full = np.fft.irfft(Fa * np.conj(Fp), n=L_corr)
        grad = 2.0 * g_full[:n]  # first n entries correspond to needed gradient

        # momentum update and projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_sc

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution of the vector `a`.
The algorithm consists of three phases:

1️⃣  **Quick parametric sweep** – a few analytic families (geometric,
    power, and Beta‑type with possibly negative exponents) are generated,
    normalised to unit sum and evaluated with an FFT based convolution.
    The best candidate is kept as a warm start.

2️⃣  **Smoothed gradient descent** – we minimise a smooth approximation
    of the max‑convolution:
        fα(a) = (2·n/α)·log( Σ exp(α·conv_i) )
    The gradient of fα is a weighted sum of the individual
    sub‑gradients and can be obtained by a single FFT‑based cross‑correlation.
    A momentum‑accelerated projected gradient step (onto the probability
    simplex) is performed.  The smoothing parameter α is increased during the
    run (homotopy) and the step size decays as 1/√t.

3️⃣  **Fine polishing** – a short projected sub‑gradient descent (using the
    exact max‑convolution) followed by a tiny mass‑transfer simulated‑annealing
    run finishes the optimisation.

The whole routine respects a 1000 s wall‑clock budget and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    # real FFT, multiply pointwise, inverse FFT
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w≥0, Σw=1}.
    Returns a new array w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective evaluation (assumes Σa = 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : smoothed gradient descent (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    L_fft: int,
                    L_corr: int,
                    reserve: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate
        fα(a) = (2n/α)·log Σ exp(α·conv_i).
    The smoothing parameter α is increased during the run
    (homotopy).  Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = best_seq.copy()
    best_sc = best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline - reserve:
        it += 1

        # --- exact convolution (needed for the objective and for α‑schedule) ---
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_sc:
            best_sc = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={cur_score:.6f}")

        # --- smoothing parameter α (start gentle, end sharp) ---
        if it < 5_000:
            alpha = 30.0
        elif it < 30_000:
            alpha = 150.0
        elif it < 150_000:
            alpha = 500.0
        else:
            alpha = 1200.0

        # --- soft‑max weights over the convolution entries ---
        # stability: subtract the max before exponentiation
        w = np.exp(alpha * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to pure sub‑gradient (only the maximal index)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        p = w / w_sum                     # length 2n‑1, Σp = 1

        # --- gradient = 2 * cross‑correlation(a, p) ---
        # use real FFTs (rfft) for speed
        Fa = np.fft.rfft(a, n=L_corr)
        Fp = np.fft.rfft(p, n=L_corr)
        corr_full = np.fft.irfft(Fa * np.conj(Fp), n=L_corr)
        grad = 2.0 * corr_full[:n]        # first n entries correspond to the needed lags

        # --- momentum + projected step ---
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)           # keep a on the simplex

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑2‑b : exact sub‑gradient descent (same as the reference code)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a simple momentum scheme and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index of the max entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer; moves a small fraction of mass from one index to another."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
