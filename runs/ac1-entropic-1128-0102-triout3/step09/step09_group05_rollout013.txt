    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n_large = 4096
    n_small = 1024  # smaller dimension for quick search.

    # Stage 1: small dimension optimization
    a_small = np.random.rand(n_small)
    a_small /= a_small.sum()
    best_small, best_score_small = _optimize(a_small, n_small, deadline, early_stop=True)
    # end stage maybe have time left

    # Stage 2: upsample to large dimension
    a_large = np.interp(np.linspace(0, 1, n_large),
                        np.linspace(0, 1, n_small),
                        best_small)
    a_large = np.maximum(a_large,0)
    a_large /= a_large.sum()
    # Continue optimizing large dimension with remaining time
    best_large, best_score_large = _optimize(a_large, n_large, deadline)

    # Final rounding and return best
    best_large = np.maximum(best_large,0)
    total = best_large.sum()
    if total > 0:
        best_large /= total
    else:
        best_large = np.full(n_large, 1.0 / n_large)

    return [float(x) for x in best_large]

"""
A fast, Adam‑based optimiser for the self‑convolution objective.

The algorithm proceeds in three stages:

1️⃣  **Adam on a smooth surrogate** – we minimise the log‑sum‑exp
    approximation  `softmax_alpha(conv(a,a))`  while keeping `a`
    on the probability simplex (non‑negative, sum = 1).  Adam’s
    adaptive learning rates give a stable descent even when the
    surrogate becomes very sharp (α grows during the run).

2️⃣  **Targeted pair‑annealing** – after the Adam phase we attack the
    two indices that contribute most to the current maximal
    convolution entry, moving a fraction of their mass to random
    locations.  The schedule is a long exponential cooling schedule;
    the Metropolis rule accepts improving moves always and worsening
    moves with probability `exp(-Δ/T)`.

3️⃣  **Local convex‑combination polishing** – the best vector from
    stage 2 is mixed with (a) the uniform distribution and (b) its
    mirror image.  A fine grid on the mixing weight is scanned; the
    best mixture (projected back onto the simplex) is kept.

All heavy work uses only NumPy (FFT‑based convolutions) and runs
inside the allotted 1000 s budget.  The implementation is fully
self‑contained and returns a plain Python list of non‑negative
heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length L (≥2·len(a)-1).
    Returns the first 2·len(a)-1 entries.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv) (assumes Σa = 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimisation on the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # parameters for the surrogate
    alpha = 4.0          # start temperature for soft‑max
    alpha_inc = 1.05     # geometric increase of alpha
    alpha_max = 200.0

    best_score = _score_from_conv(_conv_self(a, L), n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact score for bookkeeping
        conv = _conv_self(a, L)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ---- log‑sum‑exp surrogate ----
        # increase sharpness gradually
        if it % 2000 == 0:
            alpha = min(alpha * alpha_inc, alpha_max)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)          # avoid overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of Σ w_k * conv_k w.r.t a
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ---- Adam update (additive) ----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning‑rate schedule (≈1/√it)
        lr = 0.12 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_simplex(a)

    # final exact evaluation of the best vector
    best_score = _score_from_conv(_conv_self(best_a, L), n)
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted pair‑annealing (same flavour as the reference implementation)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-9,
                        steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that always attacks the two indices that
    contribute most to the current maximal convolution entry.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # maximal convolution index
        k = int(conv.argmax())

        # all i that can take part in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i to the peak (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # pair with largest product
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose a move: shift up to 20 % of the source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small local search mixing the current best with uniform / reversed
# ----------------------------------------------------------------------
def _polish_by_mixing(best_a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Scan convex combinations with the uniform vector and with the mirror
    of the current best.  Keep the best mixture that improves the exact
    score.  The routine stops promptly if the global time budget is
    exhausted.
    """
    uniform = np.full_like(best_a, 1.0 / n)
    rev = best_a[::-1]
    L = _next_pow2(2 * n - 1)

    candidates = []
    # mix with uniform
    for t in np.linspace(0.0, 1.0, 51):
        cand = (1 - t) * best_a + t * uniform
        candidates.append(cand)
    # mix with reversal
    for t in np.linspace(0.0, 1.0, 51):
        cand = (1 - t) * best_a + t * rev
        candidates.append(cand)

    best_score = _score_from_conv(_conv_self(best_a, L), n)
    best_seq = best_a.copy()

    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = _project_simplex(cand)   # keep Σ=1 and non‑negative
        cur_score = _score_from_conv(_conv_self(cand, L), n)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] mixing improve: "
                  f"score={cur_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Heuristic optimiser for the step‑function objective used by the
    competition.  Returns a list of non‑negative heights (a probability
    vector) that minimises ``evaluate_sequence`` within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse the previously discovered construction if
    # available, otherwise start from a slightly perturbed uniform vector.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4096
    if starter is not None:
        a0 = np.interp(np.linspace(0, 1, final_n),
                       np.linspace(0, 1, starter.size),
                       starter)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.full(final_n, 1.0 / final_n)   # uniform start

    # ------------------------------------------------------------------
    # Stage 1 – Adam on the smooth surrogate (≈15 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_deadline = time.time() + max(5.0, 0.15 * remaining)
    a_adam, score_adam = _run_adam(a0, final_n, stage1_deadline, start_time)
    best_seq = a_adam.copy()
    best_score = score_adam
    print(f"[{time.time() - start_time:.2f}s] stage 1 complete: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Pair‑annealing on the remainder of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_anneal, score_anneal = _run_pair_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=6e-2,
            end_temp=1e-9,
            steps=300_000,
        )
        if score_anneal < best_score:
            best_seq = a_anneal.copy()
            best_score = score_anneal
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_anneal:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Local mixing polish (tiny fraction of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_mix, score_mix = _polish_by_mixing(best_seq, final_n, deadline, start_time)
        if score_mix < best_score:
            best_seq = a_mix.copy()
            best_score = score_mix
            print(f"[{time.time() - start_time:.2f}s] stage 3 improve: score={score_mix:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] finished – final score = "
          f"{_score_from_conv(_conv_self(best_seq, _next_pow2(2 * final_n - 1)), final_n):.6f}")

    return [float(x) for x in best_seq]
