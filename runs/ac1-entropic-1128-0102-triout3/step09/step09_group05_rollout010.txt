"""
A fast “Adam‑gradient + Kaiser‑initialisation + targeted annealing” search
for a step‑function that minimises the provided evaluation function.

The algorithm proceeds in three stages

1️⃣  Initialise a smooth symmetric window (Kaiser) with several β‑values,
    pick the best by the exact objective 2·n·max(conv) (sum is forced to 1).

2️⃣  Refine the shape with two Adam‑optimised descent phases:
    – Exponentiated‑gradient on a smooth (log‑sum‑exp) surrogate of the max.
    – Projected Adam on the true sub‑gradient of the max‑convolution.
    Both phases keep the vector on the probability simplex.

3️⃣  Finish with a lightweight pair‑annealing that repeatedly moves a
    fraction of the mass of the two entries that currently give the
    maximal convolution entry to random locations (Metropolis rule).

All work is performed inside the supplied time budget (default 1000 s).
The returned list is a normalised non‑negative step‑function ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The vector is assumed to be normalised."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max utilities (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max loss.
    Implements 2·corr(a, w_rev) evaluated at the appropriate offsets.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (both additive and multiplicative versions)
# ----------------------------------------------------------------------
def _adam_update_additive(a, grad, m, v, t, step0,
                         beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with an additive update followed by simplex projection."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = step0 / (np.sqrt(v_hat) + eps)
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a_new, 1.0 / a_new.size)
    return a_new, m, v


def _adam_update_multiplicative(a, grad, m, v, t, step0,
                               beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step with a multiplicative (exponentiated‑gradient) update."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = step0 / (np.sqrt(v_hat) + eps)
    a_new = a * np.exp(-step * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a_new, 1.0 / a_new.size)
    return a_new, m, v


# ----------------------------------------------------------------------
# Phase 1 – Adam exponentiated‑gradient on a smooth max surrogate
# ----------------------------------------------------------------------
def _run_adam_exp_gradient(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           alphas,
                           step0: float = 0.15,
                           max_iters_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, _next_pow2(2 * n - 1)), n)
    L = _next_pow2(2 * n - 1)
    t = 0

    for alpha in alphas:
        # fresh Adam moments for each alpha
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        for _ in range(max_iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam‑EG improve: "
                      f"iter={t:,}, alpha={alpha:.1f}, score={cur_score:.6f}")

            w = _smooth_weights(conv, alpha)
            grad = _gradient_smooth_max(a, w)
            a, m, v = _adam_update_multiplicative(a, grad, m, v, t, step0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Adam projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.20,
                          max_iters: int | None = None):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, _next_pow2(2 * n - 1)), n)
    L = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    it = 0

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        t += 1
        conv = _conv_self(a, L)
        cur_score = _score_from_conv(conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the active max entry

        a, m, v = _adam_update_additive(a, grad, m, v, t, step0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing (same spirit as the reference code)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 150_000):
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contributions of each i to the peak
        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # random distinct destinations
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose a move
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] <= 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.30   # up‑to 30 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Window initialiser (Kaiser – a smooth tapered window)
# ----------------------------------------------------------------------
def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Return a Kaiser window of length n (β≥0)."""
    try:
        from scipy.special import i0
    except Exception:
        # fall back to a simple cosine‑type window if SciPy is missing
        return 0.5 * (1 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    idx = np.arange(n)
    alpha = (n - 1) / 2.0
    w = i0(beta * np.sqrt(1.0 - ((idx - alpha) / alpha) ** 2))
    w = w / i0(beta)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns a normalised
    list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length – 4096 works well with the reference code.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a family of Kaiser windows (several β values).
    # ------------------------------------------------------------------
    betas = [2, 4, 6, 8, 10, 12]
    best_a = None
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)

    for beta in betas:
        a = _kaiser_window(n, beta).astype(float)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # quick exact score (no clipping performed by evaluate_sequence)
        conv = _conv_self(a, L)
        score = _score_from_conv(conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"[{time.time() - start_time:.2f}s] init β={beta:.1f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Adam‑exponentiated‑gradient on the smooth max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase_deadline = time.time() + max(5.0, 0.12 * remaining)

        a_exp, score_exp = _run_adam_exp_gradient(
            best_a,
            n,
            phase_deadline,
            start_time,
            alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0],
            step0=0.12,
            max_iters_per_alpha=6000,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_a = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑1 improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam projected sub‑gradient descent on the true max‑conv.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase_deadline = time.time() + max(20.0, 0.65 * remaining)

        a_sub, score_sub = _run_adam_subgradient(
            best_a,
            n,
            phase_deadline,
            start_time,
            step0=0.22,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 improve: score={score_sub:.6f}")

        # a quick extra run with a tighter step‑size
        if time.time() < deadline:
            extra_deadline = time.time() + min(12.0, deadline - time.time())
            a_sub2, score_sub2 = _run_adam_subgradient(
                best_a,
                n,
                extra_deadline,
                start_time,
                step0=0.12,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_a = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted pair‑annealing (use the rest of the budget).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_a,
            n,
            deadline,
            start_time,
            best_score,
            best_a,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=180_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Optional short random‑restart loops (if any budget left).
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        block = min(30.0, (deadline - time.time()) / 2.0)
        restart_deadline = time.time() + block

        # small perturbation of the current best
        pert = best_a + 0.02 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        pert_seq, pert_score = _run_adam_subgradient(
            pert,
            n,
            restart_deadline,
            start_time,
            step0=0.20,
            max_iters=120_000,
        )
        if time.time() < deadline:
            pert_seq, pert_score = _run_pair_annealing(
                pert_seq,
                n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_a = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return a plain Python list.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    # The external judge will call `evaluate_sequence` on the returned list.
    return [float(x) for x in best_a]
