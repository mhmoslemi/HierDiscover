    n = len(u) + len(v) - 1

    Uf = np.fft.rfft(u, n=L)

    conv = np.fft.irfft(Uf * Vf, n=L)[:n]

The objective  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

is minimised on the probability simplex (we normalise the vector to sum 1, then
\(F(a)=2n\max_k(a*a)_k\)).  
The previous implementation used a pure sub‑gradient method.  
The new code adds a *smooth‑max* optimisation layer:

* for a given temperature \(\alpha\) define the smooth approximation  

\[
\phi_\alpha(a)=\sum_k w_k(\alpha,a)\,(a*a)_k,
\qquad 
w_k(\alpha,a)=\frac{e^{\alpha (a*a)_k}}{\sum_j e^{\alpha (a*a)_j}} .
\]

* when \(\alpha\to\infty\) we have \(\phi_\alpha(a)\to\max_k (a*a)_k\).  

* the gradient of \(\phi_\alpha\) can be written **exactly** as a linear
convolution, which is computed with FFTs:
\[
\nabla\phi_\alpha(a)=2\; w\;\star\;a^{\mathrm{rev}},
\]
where \(a^{\mathrm{rev}}\) is \(a\) reversed.

The optimisation proceeds as follows:

1. **Initialisation** – use the best known vector (`height_sequence_1` if it
   exists, otherwise uniform).
2. **Smooth‑max descent** – start with a small \(\alpha\) and increase it
   geometrically.  For each \(\alpha\) we perform many mirror‑descent steps
   with the KL‑divergence (exponentiated‑gradient) update  

   \[
   a_i \leftarrow a_i\;e^{-\eta\,(\nabla\phi_\alpha)_i},
   \qquad a \leftarrow a/\sum a .
   \]

   The step size \(\eta\) decays like \(1/\sqrt{t}\).  
   All operations are FFT‑based, so a full iteration costs only a handful of
   milliseconds even for \(n=4096\).
3. **Projected sub‑gradient refinement** – the cheap smooth‑max phase
   quickly reaches a good basin; a few thousand exact sub‑gradient steps
   (as in the reference code) polish the solution.
4. **Optional simulated‑annealing polish** – if time remains a tiny
   mass‑transfer annealer is run.

The whole routine respects the 1000 s wall‑clock limit and returns a plain
list of floats ready for `evaluate_sequence`.

```python
"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.  

The algorithm consists of

1️⃣  initialisation (global starter or uniform),
2️⃣  a smooth‑max mirror‑descent phase (exponentiated‑gradient updates,
    gradients computed via FFT‑based linear convolution),
3️⃣  a short projected sub‑gradient descent (as in the reference solution),
4️⃣  a final annealing polish (mass‑transfer moves).

All helpers are top‑level functions, the code watches the wall‑clock and never
exceeds the given budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution, not a circular one.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum vectors → the exact objective)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : exact max – projected sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (might be several, we pick the first)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish (identical to reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (new idea)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of the maximum convolution.
    Uses exponentiated‑gradient updates, gradients obtained via FFT.
    The temperature α is increased geometrically.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # geometric schedule for the smoothing parameter
    alphas = np.geomspace(0.5, 200.0, num=30)

    it = 0
    # total iterations cap – safety guard
    max_iterations = 250_000

    for alpha in alphas:
        # inner loop works until time runs out or iteration cap is hit
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution (a*a)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) softmax weights w_k ∝ exp(alpha * conv_k)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability shift
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step (mirror descent on simplex)
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            # protect against exact zeros
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Load a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (core of the new method)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        smooth_deadline = deadline - 8.0
        best_seq, best_score = _smooth_descent(best_seq, n, smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short projected sub‑gradient refinement (as in the reference)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Light annealing polish (optional, uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
