"""
Propose a low‑max self‑convolution step function.

The search consists of three phases:

1️⃣  Warm‑up – a short exponentiated‑gradient (multiplicative) descent on a
    smooth approximation of the max (identical to the reference implementation).

2️⃣  Projected sub‑gradient descent – a few hundred iterations to move quickly
    into a good basin while respecting the simplex constraint.

3️⃣  Aggressive incremental refinement – each iteration moves a *maximal*
    amount of mass from a source index (large contribution to the current
    maximal convolution bin) to a destination index (little or no contribution)
    and analytically computes the largest feasible step size ε (solution of a
    1‑D linear programme).  Several candidate source/destination pairs are
    inspected and the pair with the biggest guaranteed reduction of the current
    maximum is applied.  The ε‑schedule starts large (≈ 0.5 of the source mass)
    and decays to 0.001, allowing big jumps early on and fine‑tuning later.
    The whole algorithm respects a 1000 s wall‑clock budget and returns the
    best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: aggressive incremental refinement
# ----------------------------------------------------------------------


def _run_optimal_incremental_refine_improved(a0: np.ndarray,
                                             n: int,
                                             deadline: float,
                                             start_time: float,
                                             eps_start: float = 0.5,
                                             eps_end: float = 0.001,
                                             src_cand: int = 3,
                                             dst_cand: int = 3,
                                             recompute_every: int = 2000,
                                             rel_improve: float = 1e-12):
    """
    Improved mass‑transfer refinement.  In each iteration a handful of
    source/destination pairs are examined, the pair promising the largest
    guaranteed reduction of the current maximum is applied, and the step size
    ε is chosen maximally (solution of a 1‑D LP).  ε starts large and decays.
    """
    a = a0.copy()
    conv = _conv_self(a)          # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        # linear decay of the allowed fraction of the source mass
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # indices that actually appear in the current maximal bin
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]

        # ----------- source candidates (largest contribution) -------------
        if contrib.sum() > 0:
            order = np.argsort(-contrib)[:src_cand]
            src_candidates = i_range[order]
        else:
            src_candidates = rng.choice(i_range, size=src_cand, replace=False)

        # ----------- destination candidates (ideally outside the max bin) ---
        out_left = np.arange(0, i_start, dtype=int)
        out_right = np.arange(i_end + 1, n, dtype=int)
        out_of_range = np.concatenate((out_left, out_right))
        if out_of_range.size >= dst_cand:
            dst_candidates = rng.choice(out_of_range,
                                        size=dst_cand,
                                        replace=False)
        else:
            # fill the shortage with indices that have the smallest partner weight
            partner = np.full(n, np.inf)
            partner[i_range] = a[j_range]
            low_inside = np.argsort(partner)[: dst_cand - out_of_range.size]
            dst_candidates = np.concatenate((out_of_range, low_inside))
        if dst_candidates.size == 0:
            dst_candidates = rng.choice(n, size=dst_cand, replace=False)

        best_impr = 0.0
        best_pair = None
        best_eps = None
        best_s = None

        # ---- evaluate a small set of (src,dst) pairs --------------------
        for src in src_candidates:
            if a[src] < 1e-15:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue

                # linear effect on the whole convolution
                s = np.zeros(2 * n - 1, dtype=float)
                s[src:src + n] -= 2.0 * a
                s[dst:dst + n] += 2.0 * a
                s_max = s[k_max]
                if s_max >= 0.0:
                    continue

                # maximal ε before another entry overtakes the current max
                mask = s > s_max
                if np.any(mask):
                    delta_conv = conv[k_max] - conv[mask]
                    delta_s = s[mask] - s_max
                    eps_candidates = delta_conv / delta_s
                    eps_candidates = eps_candidates[eps_candidates > 0.0]
                    eps_star = eps_candidates.min() if eps_candidates.size else np.inf
                else:
                    eps_star = np.inf

                eps_limit = a[src] * eps_frac
                epsilon = min(eps_star, eps_limit)
                if epsilon <= 0.0:
                    continue

                improvement = -epsilon * s_max
                if improvement > best_impr:
                    best_impr = improvement
                    best_pair = (src, dst)
                    best_eps = epsilon
                    best_s = s

        # if none of the guided candidates worked, fall back to a random pair
        if best_pair is None:
            # pick a random source from the active range
            src = rng.choice(i_range)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == src:
                continue
            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a
            s_max = s[k_max]
            if s_max >= 0.0:
                continue
            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[k_max] - conv[mask]
                delta_s = s[mask] - s_max
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf
            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)
            if epsilon <= 0.0:
                continue
            best_pair = (src, dst)
            best_eps = epsilon
            best_s = s

        if best_pair is None:
            # nothing feasible – give up
            break

        src, dst = best_pair
        epsilon = best_eps
        s = best_s

        # apply the move analytically
        conv_new = conv + epsilon * s
        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep floating‑point errors in check
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within a 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of half a second to guarantee a return before timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – use the previous best construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                     # dimension of the optimisation
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – short warm‑up (exponential gradient on a smoothed max)
    # ------------------------------------------------------------------
    stage1_deadline = time.time() + min(30.0, 0.07 * (deadline - time.time()))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – brief projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        stage2_deadline = time.time() + min(150.0, 0.2 * (deadline - time.time()))
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive incremental refinement (majority of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_optimal_incremental_refine_improved(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.5,
            eps_end=0.001,
            src_cand=3,
            dst_cand=3,
            recompute_every=2000,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
