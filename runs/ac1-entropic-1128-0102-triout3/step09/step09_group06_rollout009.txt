def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 6000

    a_best = _initial_candidate(n, rng)
    best_score = _objective(a_best)

    # param search
    param_deadline = start + 0.05 * budget_s
    for p in np.linspace(0.0, 0.02, 11):
        a = np.exp(-p * np.arange(n))
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score, a_best = score, a
    # symmetric expo
    for p in np.linspace(0.0, 0.02, 11):
        d = np.arange(n)
        idx = np.minimum(d, n-1-d) # distance to nearest edge
        a = np.exp(-p * idx)
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score, a_best = score, a

    # sine power
    for p in np.linspace(0.5, 4.0, 8):
        a = np.sin(np.pi * (np.arange(n)+0.5) / n) ** p
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score, a_best = score, a

    # start Adam
    explore_deadline = start + 0.2 * budget_s
    a_opt, score_opt = _adam_optimize_long(a_best, explore_deadline, start)
    if score_opt < best_score:
        best_score, a_best = score_opt, a_opt

    # subgradient polishing (most time)
    subgrad_deadline = deadline - 0.1 * budget_s
    a_opt, score_opt = _subgrad_optimize(a_best, subgrad_deadline, start, beta=0.95, step0=0.03)
    if score_opt < best_score:
        best_score, a_best = score_opt, a_opt

    # random moves annealing
    if time.time() < deadline:
        a_opt, score_opt = _random_annealing(a_best, deadline, start)
        if score_opt < best_score:
            best_score, a_best = score_opt, a_opt

    # final
    a_best = np.clip(a_best, 0.0, None)
    total = a_best.sum()
    if total>0:
        a_best = a_best / total
    final_score = evaluate_sequence(a_best.tolist())
    print(f"[{time.time()-start:.2f}s] final score {final_score:.6f}")
    return a_best.tolist()

def _random_annealing(a0, deadline, start_time, max_iters=200000, init_temp=1e-3, decay=0.9999):
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2 * n * cur_max

    temp = init_temp
    it=0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick source and destination
        src = rng.integers(n)
        if a[src]==0: continue
        dst = rng.integers(n)
        if src==dst: continue
        # epsilon adaptation depends on temperature
        eps = min(a[src], temp * a[src])
        if eps<1e-12: continue

        # update conv incrementally similar to greedy but using generic indices
        conv_tmp = conv.copy()
        # Subtract contribution of src:
        # For each i, b_{src+i} decreases by 2 epsilon a_i (if i != src) and b_{2*src} decreases by epsilon^2
        # But we have for general move from src to dst:
        # Equivalent to adding epsilon to dst and subtracting from src.
        # We can reuse the formula from _fast_greedy_improve.

        # For simplicity compute conv from scratch after move (cost O(n log n)), but this may be heavy.
        # We'll compute incremental update as per earlier.

        # We need to handle both contributions
        # Use vectorized update:
        # Subtract 2*eps*a from conv[src:src+n]
        # Add 2*eps*a to conv[dst:dst+n]
        # Quadratic terms: conv[2*src] += eps*eps, conv[2*dst] += eps*eps, conv[src+dst] -= 2*eps*eps

        conv_tmp[src:src+n] -= 2*eps*a
        conv_tmp[dst:dst+n] += 2*eps*a
        conv_tmp[2*src] += eps*eps
        conv_tmp[2*dst] += eps*eps
        conv_tmp[src+dst] -= 2*eps*eps

        new_max = conv_tmp.max()
        # Accept if improvement or with probability e^(-(new-max-cur_max)/temp)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis
            delta = new_max - cur_max
            if np.random.rand() < math.exp(-delta / temp):
                accept = True
        if accept:
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            if 2*n*cur_max < best_score:
                best_score = 2*n*cur_max
                best_a = a.copy()
                # print progress occasionally
                if it % 10000 == 0:
                    print(f"[{time.time()-start_time:.2f}s] anneal best score {best_score:.6f} iter {it}")
        # cooling
        temp *= decay
    return best_a, best_score

"""
Proposes a non‑negative step function (list of heights) that tries to
minimise ``evaluate_sequence``.

The method mixes three ideas that were not used together before:

1. **Fast parametric sweep** – a few simple families (exponential,
   symmetric exponential, powered‑sine) are evaluated on the full
   convolution.  This gives a good deterministic starting point without
   any gradient steps.

2. **Projected Adam on a smooth‑max surrogate** – a long Adam run on the
   smooth approximation of the max‑convolution (log‑sum‑exp) with a
   schedule of increasing smoothing parameters.  The iterate is projected
   onto the probability simplex after each step.

3. **Heavy sub‑gradient polishing + stochastic annealing** – the best
   candidate from (2) is refined with many projected sub‑gradient steps,
   followed by a cheap Metropolis‑type random‑move loop that transfers
   a small amount of mass from one index to another while updating the
   convolution incrementally.  The total wall‑clock budget is respected
   (default 1000 s).

The routine finally normalises the vector to sum = 1, evaluates the
true objective with ``evaluate_sequence`` and returns the list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact value of the quantity returned by ``evaluate_sequence`` for a
    normalised (sum = 1) vector a."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def _parametric_sweep(n: int) -> np.ndarray:
    """
    Generate a few deterministic candidates from simple families,
    return the best (lowest objective) among them.
    """
    best_a = np.full(n, 1.0 / n)            # uniform fallback
    best_score = _objective(best_a)

    # 1. symmetric exponential decay a_i = exp(-p * min(i, n-1-i))
    for p in np.linspace(0.0, 0.02, 11):
        idx = np.arange(n)
        d = np.minimum(idx, n - 1 - idx)
        a = np.exp(-p * d)
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    # 2. pure exponential (not symmetric) – sometimes useful after interpolation
    for p in np.linspace(0.0, 0.02, 11):
        a = np.exp(-p * np.arange(n))
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    # 3. powered sine window a_i = sin(pi*(i+0.5)/n) ** p
    for p in np.linspace(0.5, 4.0, 8):
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()

    return best_a


# ----------------------------------------------------------------------
# Phase 1 – projected Adam on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                     # soft‑max weights

    # gradient of Σ w_k * (a*a)_k  w.r.t. a is 2·(a ⋆ w)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre of the correlation corresponds to indices 0..n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Long projected Adam on the smoothed‑max function.
    Returns the best point found and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters – larger α → tighter approximation
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = _objective(a)

    for alpha in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self(a)
            # smoothed max
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                     # smooth max value
            # gradient of the *objective* (2·n·smooth_max)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            if t % 800 == 0:
                sc = _objective(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha:.0f}) score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic projected sub‑gradient descent
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  max_k (a*a)[k] (i.e. the true max).
    The routine runs until *deadline* is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # define the active set (indices whose value is within 1e‑12 relative tolerance)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the max (summing contributions of all active convolutions)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair (i,j) appears twice in the convolution, diagonal once – the double counting is exactly the gradient we need
            g[i] += a[j]
            g[j] += a[i]

        # momentum + decreasing step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – cheap Monte‑Carlo annealing (random mass moves)
# ----------------------------------------------------------------------


def _random_annealing(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      rng: np.random.Generator,
                      max_iters: int = 500_000,
                      init_temp: float = 1e-3,
                      decay: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Metropolis‑type annealing: repeatedly move a small fraction of
    probability mass from a random source index to a random destination.
    The convolution array is updated incrementally (O(n) per move).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    temp = init_temp
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        src = rng.integers(n)
        if a[src] == 0.0:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # size of the transfer – proportional to temperature
        eps = min(a[src], temp * a[src])
        if eps < 1e-14:
            continue

        # incremental update of convolution (identical to the fast‑greedy routine)
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis acceptance
            delta = new_max - cur_max
            if rng.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                if it % 20000 == 0:
                    print(f"[{time.time() - start_time:6.2f}s] anneal best: iter={it:,}, score={cur_score:.6f}")

        temp *= decay

    return best_a, best_score

def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised (sum = 1) starting vector.  Prefer the historic
    best sequence ``height_sequence_1`` if it exists, otherwise fall back
    to a uniform vector."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation preserving the total sum
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, prints progress
    information and finally returns the best sequence as a plain Python
    list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # small safety margin

    # problem size – a few thousand entries give a good balance between
    # expressive power and speed of FFT‑based convolutions.
    n = 6000

    # ------------------------------------------------------------------
    # 0) cheap deterministic families
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] start parametric sweep (n={n})")
    a_best = _parametric_sweep(n)
    best_score = _objective(a_best)
    print(f"[{time.time() - start_time:6.2f}s] sweep best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Adam optimisation on smooth‑max (≈20 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.20 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    a_opt, sc_opt = _adam_optimize_long(a_best, adam_deadline, start_time)
    if sc_opt < best_score:
        best_score, a_best = sc_opt, a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] Adam improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient polishing – main chunk of time (≈65 % of budget)
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.85 * budget_s   # leave ~15 % for annealing
    print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
    a_opt, sc_opt = _subgrad_optimize(a_best, subgrad_deadline, start_time,
                                      beta=0.95, step0=0.03)
    if sc_opt < best_score:
        best_score, a_best = sc_opt, a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] sub‑grad improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Random annealing / greedy peak reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start random annealing")
        a_opt, sc_opt = _random_annealing(a_best, deadline, start_time, rng)
        if sc_opt < best_score:
            best_score, a_best = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation & evaluation
    # ------------------------------------------------------------------
    a_best = np.clip(a_best, 0.0, None)
    total = a_best.sum()
    if total > 0:
        a_best = a_best / total
    else:
        a_best = np.full_like(a_best, 1.0 / n)

    final_score = evaluate_sequence(a_best.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in a_best.tolist()]
