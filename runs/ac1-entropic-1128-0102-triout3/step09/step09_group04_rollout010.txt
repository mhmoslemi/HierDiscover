"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The routine combines three complementary phases that respect a global
time budget (default 1000 s).

1. **Smooth‑max Adam descent** – the hard max is replaced by a
   soft‑max (log‑sum‑exp) with an increasing temperature.  The gradient
   of the smooth objective is obtained analytically; an Adam‑style
   update with Euclidean projection onto the probability simplex moves
   the vector quickly away from plateaus.

2. **Projected sub‑gradient descent** – after the smooth phase the true
   non‑smooth objective is tackled with a momentum sub‑gradient method
   (the same method used in the reference implementation).  Smaller
   steps allow fine tuning.

3. **Random hill‑climbing** – a short stochastic polishing stage makes
   small Gaussian perturbations, projects back onto the simplex and
   accepts any improvement.  This helps to escape tiny local traps.

The algorithm works on the simplex (∑a = 1), because the objective is
scale‑invariant.  The best vector found during the run is returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """ Small utility used by the FFT based convolution. """
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """ Full linear convolution a * a (length 2·len(a)‑1) via real FFT. """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    For all k attaining the maximum we add the contribution of a to the
    gradient; the result is averaged over the number of active indices.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _interpolate(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_len = seq.size
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows a three‑stage
    strategy described in the module docstring and respects the supplied
    time budget (in seconds).  The returned value is a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the caller
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # If a previous construction exists we keep its length; otherwise
    # we fall back to 2048 which offers a good trade‑off.
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            n = max(2048, tmp.size)            # at least 2048 entries
        except Exception:
            n = 2048
    else:
        n = 2048

    # --------------------------------------------------------------
    # Initial vector (simplex, sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                tmp = _interpolate(tmp, n)
                a = tmp / tmp.sum()
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # bookkeeping
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam descent (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.60
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.12                     # base learning rate (after bias‑correction)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the soft‑max temperature
    alpha_start = 1.0
    alpha_end = 200.0
    ramp_iters = 8000            # after this many iterations we stay at alpha_end

    it = 0
    while time.time() < phase1_deadline:
        it += 1

        # convolution of the current vector
        conv = np.convolve(a, a)             # length 2·n‑1
        # temperature for the smooth max
        if it < ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * it / ramp_iters
        else:
            alpha = alpha_end

        # soft‑max weights (stable implementation)
        X = alpha * conv
        max_X = X.max()
        exp_X = np.exp(X - max_X)
        w = exp_X / exp_X.sum()               # shape (2·n‑1,)

        # gradient of the smooth objective:
        #   grad = 4·n· (a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        conv_rev = np.convolve(a_rev, w, mode='full')
        grad = 4.0 * n * conv_rev[n - 1:n - 1 + n]

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # keep the vector on the simplex
        a = _project_simplex(a)

        # evaluate the true (non‑smooth) objective
        conv_new = np.convolve(a, a)
        cur_score = 2.0 * n * conv_new.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase‑1 iter {it:5d} "
                  f"improve → {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + (deadline - start_time) * 0.90
    if time.time() < phase2_deadline:
        a = best_seq.copy()
        momentum = np.zeros_like(a)
        sub_it = 0
        while time.time() < phase2_deadline:
            sub_it += 1
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad iter {sub_it:5d} "
                      f"improve → {cur_score:.6f}")

            # sub‑gradient of the hard max
            grad = _subgradient_max(a, conv)
            # momentum update (the same parameters as the reference code)
            momentum = 0.94 * momentum + 0.06 * grad
            step = 0.15 / math.sqrt(sub_it)
            a = _project_simplex(a - step * momentum)

    # ------------------------------------------------------------------
    # Phase 3 – random hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        hill_it = 0
        # annealing schedule for the random perturbation amplitude
        while time.time() < deadline:
            hill_it += 1
            # the amplitude shrinks linearly with the remaining time
            remaining = deadline - time.time()
            total = deadline - start_time
            sigma = 0.05 * (remaining / total) + 1e-4

            cand = a + sigma * np.random.randn(n)
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            conv_cand = np.convolve(cand, cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb iter {hill_it:5d} "
                      f"improve → {cand_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
