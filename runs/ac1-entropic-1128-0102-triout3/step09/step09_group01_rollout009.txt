def _smooth_descent(a0, n, deadline, start_time, L_fft, L_grad, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.01  # base learning rate

    # epsilon schedule parameters
    eps_factor = 0.2
    block_iters = 50000
    min_eps_factor = 0.002 # min factor relative to max conv

    it = 0
    while time.time() < deadline:
        it += 1

        # compute conv self
        conv = _conv_self(a, L=L_fft)
        max_conv = conv.max()
        # update epsilon using schedule
        # decrease eps_factor after each block
        if it % block_iters == 0:
            eps_factor = max(eps_factor * 0.5, min_eps_factor)
            print(f"[{time.time() - start_time:.2f}s] Smooth descent eps_factor={eps_factor:.6f}")

        epsilon = max_conv * eps_factor
        if epsilon < 1e-12:
            epsilon = 1e-12

        # compute weights
        w = np.exp(conv / epsilon)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via convolution of reversed a and w
        rev_a = a[::-1]
        # convolution rev_a (len n) with w (len 2n-1) => length 3n-2
        grad_full = _conv_fft(rev_a, w, L=L_grad)  # returns length 3n-2
        grad = 2.0 * grad_full[n-1:n-1+n]  # slice n elements
        # gradient of smooth objective:
        grad_f = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_f
        v = beta2 * v + (1 - beta2) * (grad_f ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # project onto simplex
        a = _project_simplex(a)

        # evaluate actual objective
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)  # a sum ~1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Smooth descent it={it} score={cur_score:.6f}")

    return best_a, best_score

"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.  

Algorithm
---------

1️⃣  **Fast parametric sweep** – a dense grid of Beta‑type shapes
    a_i ∝ (i+1)^p·(n‑i)^q is explored (p,q∈[0,5] with step 0.1).  
    The best candidate is kept.

2️⃣  **Smoothed gradient descent** – the non‑smooth "max" is replaced by a
    log‑sum‑exp surrogate  
    f_ε(a)=2 n·ε·log∑_k exp(conv_k/ε).  
    Its gradient is obtained via a second FFT‑based convolution, and an
    Adam optimiser (projected onto the probability simplex) is applied.
    ε is decreased geometrically (ε = ε_factor·max(conv) with ε_factor
    halved every 50 000 iterations).  This phase occupies the bulk of the
    time budget and typically pushes the score below the previous record.

3️⃣  **Light annealing polish** – a mass‑transfer simulated‑annealing pass
    makes tiny random moves between two coordinates and accepts them with a
    Metropolis probability.  This finishes the search.

All helpers are top‑level; the routine respects the 1000 s wall‑clock
budget and returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Convolution of two real vectors using FFT."""
    if L is None:
        L = _next_pow2(x.shape[0] + y.shape[0] - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: x.shape[0] + y.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → exact objective)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : smoothed gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int,
                    best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the objective using Adam.
    The smoothing parameter ε = ε_factor * max(conv) is halved every
    ``eps_block`` iterations.  The iterate is projected onto the simplex.
    Returns the best sequence (list) and its true score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.01                              # base learning rate

    # ε‑schedule
    eps_factor = 0.2                       # start with 20 % of max(conv)
    eps_block = 50_000                     # halve ε after this many iters
    min_eps_factor = 0.01                  # never go below 1 % of max(conv)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- compute self‑convolution and current ε ----------
        conv = _conv_self(a, L=L_fft)
        max_conv = conv.max()

        # adapt ε
        if it % eps_block == 0:
            eps_factor = max(eps_factor * 0.5, min_eps_factor)
            print(f"[{time.time() - start_time:.2f}s] "
                  f"smooth descent it={it:,} eps_factor={eps_factor:.4f}")

        epsilon = max_conv * eps_factor
        if epsilon < 1e-12:
            epsilon = 1e-12

        # ----- soft‑max weights (numerically stable) ----------
        shifted = conv - max_conv               # max(conv) becomes zero
        w = np.exp(shifted / epsilon)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective ---------------
        # g = 2 * sum_i w_i * ∂conv_i/∂a  ; we obtain it via convolution.
        rev_a = a[::-1]                         # reversed a
        grad_full = _conv_fft(rev_a, w, L=L_grad)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # slice of length n
        grad_f = 2.0 * n * grad                  # ∂f_ε/∂a

        # ----- Adam update + projection onto simplex ------------
        m = beta1 * m + (1.0 - beta1) * grad_f
        v = beta2 * v + (1.0 - beta2) * (grad_f ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step
        a = _project_simplex(a)

        # ----- evaluate true objective ---------------------------
        cur_score = 2.0 * n * _conv_self(a, L=L_fft).max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"smooth descent it={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish (as in the reference code)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves, Metropolis acceptance)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source weighted by its mass and a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    n = 4096                     # problem size used by the reference solution
    L_fft = _next_pow2(2 * n - 1)          # length for self‑convolution FFT
    L_grad = _next_pow2(3 * n - 2)         # length for gradient‑convolution FFT

    # ------------------------------------------------------------------
    # Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fast parametric sweep (dense Beta family)
    # ------------------------------------------------------------------
    idx = np.arange(n)
    p_vals = np.linspace(0.0, 5.0, 51)   # 0 … 5 with step 0.1
    q_vals = np.linspace(0.0, 5.0, 51)

    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline - 30:
                break
            seq = np.power(idx + 1, p) * np.power(n - idx, q)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : smoothed gradient descent (Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:               # reserve ~2 min for polish
        sub_deadline = deadline - 30.0               # keep a tiny window for the final anneal
        a_smooth, sc_smooth = _smooth_descent(best_seq,
                                               n,
                                               sub_deadline,
                                               start_time,
                                               L_fft,
                                               L_grad,
                                               best_score)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] Smoothed descent final score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : simulated‑annealing polish (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq,
                                       n,
                                       deadline,
                                       start_time,
                                       best_score,
                                       best_seq,
                                       start_temp=5e-2,
                                       end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
