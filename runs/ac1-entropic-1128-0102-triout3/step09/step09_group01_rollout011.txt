"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1.  **Quick parametric sweep** – a few families (geometric, power, Beta‑type)
    are normalised to sum 1 and evaluated with an FFT‑based self‑convolution.
    The best of these cheap candidates seeds the optimisation.

2.  **Smoothed projected gradient descent** – we minimise a smooth
    approximation of the ℓ∞‑norm:
        f_λ(a) = (1/λ)·log∑_k exp(λ·conv_k(a))
    (λ grows from ≈10 to ≈150 during the run).  The gradient is the
    cross‑correlation of *a* with the soft‑max weights, computed with FFTs.
    An Adam‑style update is projected onto the probability simplex.

3.  **Mass‑transfer simulated annealing** – with the remaining time we
    perform a classic annealer that moves a random fraction of mass from one
    coordinate to another, accepting moves that improve the exact objective
    or that satisfy a Metropolis criterion with an exponential cooling schedule.

All heavy work is performed with FFTs, the whole routine respects the 1000 s
wall‑clock budget and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Cross‑correlation (used for the smoothed gradient)
# ----------------------------------------------------------------------
def _cross_corr(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Linear cross‑correlation r[i] = Σ_j a[j]·p[i+j] for i=0..len(a)-1.
    Implemented via FFT (r = 2·corr(a,p)).
    """
    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p, n=L)
    R = np.fft.irfft(np.conj(Af) * Pf, n=L)
    offset = p.shape[0] - 1            # lag 0 is at this index
    # gradient of the smoothed max is 2 * r
    return 2.0 * R[offset : offset + a.shape[0]]


# ----------------------------------------------------------------------
# Smoothed projected gradient descent (Adam + λ continuation)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) with Adam on the simplex.
    λ is increased linearly from 10 to 150.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5

    # FFT lengths (self‑conv and cross‑corr)
    L_self = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- exact objective (used only for bookkeeping) -----
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={cur_score:.6f}")

        # ----- λ‑continuation -----
        # start at λ=10, finish near 150 (linear in iteration count)
        lam = 10.0 + min(140.0, 0.03 * it)   # reaches ≈150 after ~5000 iters

        # ----- soft‑max weights -----
        shifted = conv - conv.max()          # for numerical stability
        exp_vals = np.exp(lam * shifted)
        p = exp_vals / exp_vals.sum()        # length 2n‑1, sum = 1

        # ----- gradient of the smoothed objective -----
        grad = _cross_corr(a, p, L_corr)     # shape (n,)

        # ----- Adam update (step size ∝ 1/√it) -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that tweaks the exact objective."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # choose a source weighted by its mass, a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, _next_pow2(2 * n - 1)).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
