import numpy as np
import time, math

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _objective(a):
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_conv / (sum_a**2))

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad iter={it:,} score={cur_score:.6f}")
        # get index achieving max
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
        # safety: ensure nonnegative
        a = np.maximum(a, 0.0)
    return best_a, best_score

def _annealing(a0, deadline, start_time):
    a = a0.copy()
    current_score = _objective(a)
    best_a = a.copy()
    best_score = current_score
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    # compute total remaining steps approximately based on average iteration time later? We'll just loop
    # Determine a temperature schedule based on fraction of time elapsed.
    total_time = deadline - start_time
    while time.time() < deadline:
        it += 1
        # fraction of elapsed time (0 at start, 1 at end)
        frac = (time.time() - start_time) / total_time
        # geometric cooling: T goes from T0 to Tf
        T0 = 0.04
        Tf = 1e-6
        T = T0 * (Tf/T0) ** frac
        # choose source proportionally to mass (to move mass from dense areas)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        # magnitude of transfer
        max_transfer = 0.4
        eps = a[src] * rng.random() * max_transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # project just to keep non-negativity (already)
        new_score = _objective(a_prop)
        delta = new_score - current_score
        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a low value of
    ``2·n·max(conv(a,a))/(sum a)^2``.
    The method combines:
      1. a sweep over simple power‑tent families,
      2. projected sub‑gradient descent,
      3. a long simulated‑annealing polish,
      4. a final short sub‑gradient refinement.
    All vectors are kept on the probability simplex (sum = 1).
    """
    import numpy as np, time, math
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) generate a few analytic starters (power‑tent)
    # ------------------------------------------------------------------
    candidates = []
    x = np.arange(n, dtype=float) / (n - 1)
    exponents = np.linspace(0.45, 2.2, 25)   # fine sweep
    for p in exponents:
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2) include the globally supplied starter if present
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        arr = np.asarray(height_sequence_1, dtype=float)
        if arr.ndim == 1 and arr.size > 0:
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                candidates.append(arr / s)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) pick the best among those starters using the fast objective
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for a in candidates:
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[{time.time()-start_time:.2f}s] starter exponent/arr score={val:.6f}")

    # If nothing found, fallback to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # 4) main sub‑gradient descent phase (≈ 60 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_deadline = time.time() + remaining * 0.6
        seq_sgd, val_sgd = _subgradient_descent(best_seq, sub_deadline, start_time)
        if val_sgd < best_val:
            best_val = val_sgd
            best_seq = seq_sgd.copy()
            print(f"[{time.time()-start_time:.2f}s] After SGD best={val_sgd:.6f}")

    # ------------------------------------------------------------------
    # 5) simulated‑annealing polish (≈ 30 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        ann_deadline = time.time() + remaining * 0.8
        seq_ann, val_ann = _annealing(best_seq, ann_deadline, start_time)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[{time.time()-start_time:.2f}s] After anneal best={val_ann:.6f}")

    # ------------------------------------------------------------------
    # 6) short final sub‑gradient refinement (use any leftover time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        fin_deadline = time.time() + remaining * 0.9
        seq_fin, val_fin = _subgradient_descent(best_seq, fin_deadline, start_time)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = seq_fin.copy()
            print(f"[{time.time()-start_time:.2f}s] Final refinement best={val_fin:.6f}")

    # ------------------------------------------------------------------
    # 7) final normalisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # return plain list
    # ------------------------------------------------------------------
    print(f"[{time.time()-start_time:.2f}s] Final objective { _objective(best_seq):.6f}")
    return [float(v) for v in best_seq]

"""
Propose a step‑function that improves the current bound (≈1.5088).

Algorithm
---------
1. **Generate many analytic starters** – power‑tents, weighted tents,
   symmetric Beta densities and narrow Gaussians.  All are placed on the
   probability simplex (sum = 1) and the best few are kept.
2. **Pairwise mixing** – every convex combination of the top‑K starters is
   evaluated; this cheap search often discovers a shape that beats each
   individual starter.
3. **Projected sub‑gradient descent** – starting from the current best
   vector we minimise the exact nonsmooth objective
   `2·n·max(conv(a,a))/sum(a)²`.  Momentum and a diminishing step size are
   used; the iterate is always projected onto the simplex.
4. **Long simulated‑annealing polish** – mass is transferred between random
   indices; uphill moves are allowed with a geometric temperature schedule.
5. **A final short sub‑gradient refinement** – makes use of any remaining
   time budget.
All vectors are kept non‑negative and normalised; the final list is
returned in plain Python format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 0. Low‑level utilities (FFT convolution, simplex projection, objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/(sum(a)²).  Assumes a≥0."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1. Projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, start_time: float
                         ) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum decay
    step0 = 0.25        # initial step (divided by sqrt(it))
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see the reference code)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)
        a = np.maximum(a, 0.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray, deadline: float, start_time: float
               ) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing with a geometric temperature schedule."""
    a = a0.copy()
    current_score = _objective(a)
    best_a = a.copy()
    best_score = current_score
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    # total time available for this phase
    total_time = deadline - start_time
    T0 = 4e-2           # initial temperature
    Tf = 1e-6           # final temperature

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(elapsed / total_time, 1.0)
        T = T0 * (Tf / T0) ** frac  # geometric cooling

        # choose a source index proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction of the source mass
        eps = a[src] * rng.random() * 0.40   # up to 40 % of src in one move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        new_score = _objective(a_prop)
        delta = new_score - current_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

def _generate_candidates(n: int) -> list[np.ndarray]:
    """Return a list of promising analytic vectors (already normalised)."""
    cand = []
    x = np.arange(n, dtype=float) / (n - 1)          # points in [0,1]

    # (a) simple power‑tents  min(x,1‑x)^p
    for p in np.linspace(0.45, 2.2, 25):
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0:
            continue
        cand.append(w / w.sum())

    # (b) weighted tents  min(x,1‑x)^p * (1‑c·min(x,1‑x))
    for p in np.linspace(0.6, 2.0, 12):
        base = np.minimum(x, 1.0 - x)
        power = base ** p
        for c in np.linspace(0.0, 0.8, 5):
            w = power * (1.0 - c * base)
            w = np.maximum(w, 0.0)
            if w.sum() == 0:
                continue
            cand.append(w / w.sum())

    # (c) symmetric Beta densities  (x·(1‑x))^a
    for a in np.linspace(0.2, 3.0, 20):
        w = (x * (1.0 - x)) ** a
        if w.sum() == 0:
            continue
        cand.append(w / w.sum())

    # (d) narrow Gaussians (forced to zero at the borders)
    for sigma in (0.02, 0.04, 0.06, 0.08):
        w = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        w[0] = w[-1] = 0.0
        if w.sum() == 0:
            continue
        cand.append(w / w.sum())

    return cand


def _mix_top_candidates(candidates: list[np.ndarray],
                        top_k: int = 10) -> tuple[np.ndarray, float]:
    """Mix the top‑K candidates pairwise and keep the best mixture."""
    scores = [_objective(c) for c in candidates]
    ordered = sorted(zip(candidates, scores), key=lambda p: p[1])
    top = [c for c, _ in ordered[:top_k]]
    best_seq = top[0].copy()
    best_val = _objective(best_seq)

    for i in range(len(top)):
        for j in range(i + 1, len(top)):
            for t in np.linspace(0.0, 1.0, 11):
                mix = (1.0 - t) * top[i] + t * top[j]
                mix = np.maximum(mix, 0.0)
                s = mix.sum()
                if s == 0:
                    continue
                mix /= s
                val = _objective(mix)
                if val < best_val:
                    best_val = val
                    best_seq = mix.copy()
                    print(f"[mix] pair {i}-{j} t={t:.2f} score={val:.6f}")
    return best_seq, best_val

    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size (the same canvas as the previous best solution)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 0) collect analytic starters
    # --------------------------------------------------------------
    candidates = _generate_candidates(n)

    # --------------------------------------------------------------
    # 1) optionally add the global starter supplied by the harness
    # --------------------------------------------------------------
    try:
        global height_sequence_1
        arr = np.asarray(height_sequence_1, dtype=float)
        if arr.ndim == 1 and arr.size > 0:
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 0:
                candidates.append(arr / s)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2) keep the best singleton starter (cheap evaluation)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for a in candidates:
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[init] candidate score={val:.6f}")

    # --------------------------------------------------------------
    # 3) pairwise mixing of the top few starters
    # --------------------------------------------------------------
    mixed_seq, mixed_val = _mix_top_candidates(candidates, top_k=10)
    if mixed_val < best_val:
        best_val = mixed_val
        best_seq = mixed_seq.copy()
        print(f"[mix] best mixed score={mixed_val:.6f}")

    # --------------------------------------------------------------
    # 4) main sub‑gradient descent (≈ 60 % of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_deadline = time.time() + remaining * 0.60
        seq_sgd, val_sgd = _subgradient_descent(best_seq, sub_deadline,
                                                start_time)
        if val_sgd < best_val:
            best_val = val_sgd
            best_seq = seq_sgd.copy()
            print(f"[sgd] improved score={val_sgd:.6f}")

    # --------------------------------------------------------------
    # 5) simulated‑annealing polish (≈ 30 % of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        ann_deadline = time.time() + remaining * 0.80
        seq_ann, val_ann = _annealing(best_seq, ann_deadline, start_time)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = seq_ann.copy()
            print(f"[anneal] improved score={val_ann:.6f}")

    # --------------------------------------------------------------
    # 6) final short sub‑gradient refinement (use any leftover time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        fin_deadline = time.time() + remaining * 0.90
        seq_fin, val_fin = _subgradient_descent(best_seq, fin_deadline,
                                                start_time)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = seq_fin.copy()
            print(f"[final] improved score={val_fin:.6f}")

    # --------------------------------------------------------------
    # 7) final normalisation and conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[done] final objective = {_objective(best_seq):.6f}")
    return [float(v) for v in best_seq]
