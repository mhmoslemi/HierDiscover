"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence().

Algorithm
---------
1. Warm‑start – try to reuse the historic best sequence
   (global variable ``height_sequence_1``); otherwise fall back
   to a uniform distribution.

2. Phase‑0 – Projected Adam on a smooth approximation of the
   ∞‑norm (log‑sum‑exp).  The smoothing parameter α is increased
   gradually up to 8 192.  The Adam iterate is always projected
   onto the probability simplex.

3. Phase‑1 – Several independent sub‑gradient runs (restarts).  Each
   run starts from the current best candidate with a small random
   perturbation, uses momentum and a decreasing step size
   (step₀ / √t).  The best point amongst all restarts is kept.

4. Phase‑2 – Fast greedy peak reduction (the same O(n) move as the
   reference implementation) followed by a lightweight simulated
   annealing that occasionally accepts uphill moves to escape local
   minima.

5. Normalise the final vector to sum = 1 and return it.

The routine respects the 1000 s time budget, prints progress messages
and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre the correlation to obtain the gradient entries
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

# Phase‑0 – Adam on smoothed max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max.
    Returns the best vector (probability simplex) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    fft_len = _next_pow2(3 * n - 2)          # length for cross‑correlation
    # extended α schedule – larger α yields a tighter approximation of max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0,
                      2048.0, 8192.0, 32768.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        # keep iterating with the current α while time permits
        while time.time() < deadline:
            conv = _conv_self(a)
            # smoothed max value (log‑sum‑exp)
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                     # smoothed max
            obj = 2.0 * n * f_val               # same scale as evaluate_sequence

            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f}) "
                          f"score={sc:.6f}")

            # break if we exceed the global deadline
            if time.time() >= deadline:
                break

    return best_a, best_score

# Phase‑1 – Sub‑gradient descent with restarts

def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        #--- sub‑gradient of the active peak(s) -------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # double‑count diagonal – harmless for a ≥ 0

        # momentum + decreasing step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Greedy O(n) moves that shift a small fraction from the most
    responsible coefficient to the smallest one, accepting only
    strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # coefficient that contributes most

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.05 * a[src]                     # transfer 5 % of source value
        if eps < 1e-14:
            break

        # incremental update of convolution
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept only strict improvement
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it:,}, max={new_max:.6f}")

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Random mass transfers that occasionally accept uphill moves.
    The temperature decays linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # initial temperature – 2 % of the current objective
    temp0 = best_score * 0.02
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        t_elap = time.time() - start_time
        t_tot = deadline - start_time
        temp = max(temp0 * (1.0 - t_elap / t_tot), 1e-12)

        # pick source (biased towards large coefficients) and destination
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # propose a transfer amount (up to max_move_frac of source)
        eps = rng.random() * max_move_frac * a[src]
        if eps < 1e-14:
            continue

        # incremental update of convolution (same pattern as greedy)
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_score = 2.0 * n * conv_tmp.max()
        delta = new_score - best_score

        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if np.exp(-delta / temp) > rng.random():
                accept = True

        if accept:
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: score={best_score:.6f}")

    return best_a, best_score

    if remaining <= 0:
        remaining = 0.0
    subgrad_deadline = time.time() + 0.55 * remaining   # ~55 % of what is left
    print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient phase (budget {subgrad_deadline - time.time():.1f}s)")

    # number of independent sub‑gradient runs
    n_restarts = 4
    per_restart = (subgrad_deadline - time.time()) / n_restarts

    for r in range(n_restarts):
        # perturb the current best a little before restarting
        a_start = best_a.copy()
        noise = rng.normal(scale=0.02, size=n)      # modest Gaussian noise
        a_start = np.clip(a_start + noise, 0.0, None)
        s = a_start.sum()
        if s > 0:
            a_start /= s
        else:
            a_start = np.full(n, 1.0 / n)

        cand_dead = time.time() + per_restart
        a_opt, sc = _subgrad_optimize(a_start, cand_dead, start_time,
                                      beta=0.94, step0=0.15)
        sc = evaluate_sequence(a_opt.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] restart {r} sub‑grad: score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Greedy refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start fast greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] greedy finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing local search (uses any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start simulated annealing refinement")
        best_a, best_score = _simulated_annealing(best_a, deadline, start_time, rng)
        print(f"[{time.time() - start_time:6.2f}s] SA finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return a plain Python list of floats
    return [float(x) for x in best_a.tolist()]
