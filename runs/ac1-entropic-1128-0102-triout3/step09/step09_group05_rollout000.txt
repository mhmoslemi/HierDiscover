"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence() using a multi‑scale, hybrid optimisation:

1️⃣  Initialise from the previous best (`height_sequence_1`) if it exists,
    otherwise from a random point.  Normalise to a probability vector.

2️⃣  Phase 1 – Exponentiated‑gradient on a smooth (log‑sum‑exp) approximation
    of the max‑convolution.  Several learning‑rate schedules (α) are swept.

3️⃣  Phase 2 – Projected sub‑gradient descent on the true objective
    2·n·max(conv).  Momentum and a 1/√t step decay are used.

4️⃣  Phase 3 – Targeted pair‑annealing (Metropolis) that repeatedly attacks
    the two entries that contribute most to the current peak of the
    self‑convolution.

5️⃣  Upsample the best vector to a larger size (×2) and repeat phases 2‑3
    on the higher‑resolution problem – larger n usually allows a lower
    maximal convolution value.

6️⃣  A short random‑restart loop that perturbs the current best and
    re‑optimises if time permits.

All loops stop as soon as the global deadline (≈budget_s seconds) is
reached.  The final vector is projected onto the simplex and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv); assumes Σa = 1."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on a smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.4,
                      iters_per_alpha: int = 12000):
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    best_score = _score_from_conv(conv, n)

    it = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # current convolution and score
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve "
                      f"(α={alpha}) iter={it:,} score={cur_score:.6f}")

            # smooth‑max weights (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # gradient of the smooth max (2·corr)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            # map correlation to gradient indices (centered)
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # exponentiated update
            eta = eta0 / math.sqrt(it)
            grad_clipped = np.clip(grad, 0.0, 20.0)   # avoid overflow
            a = a * np.exp(-eta * grad_clipped)

            # projection onto the simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.30
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve "
                  f"iter={it:,} score={cur_score:.6f}")

        # sub‑gradient from the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing (Metropolis)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-9,
                        steps: int = 200_000):
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # feasible i‑range for this k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contributions a[i] * a[k-i]
        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max

        # select distinct destination positions (avoid sources)
        src = [i_max] if i_max == j_max else [i_max, j_max]
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction (≤15%) of each source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.15
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate proposal
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector to a new length (linear interp)
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, new_len: int) -> np.ndarray:
    old_len = vec.shape[0]
    if old_len == new_len:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(new_len, 1.0 / new_len)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` within the allocated time budget.
    The algorithm works on a base size of 4096, refines it, then
    up‑samples to 8192 and refines again.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Initialise – use previous construction if available.
    # ------------------------------------------------------------------
    base_n = 4096
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = np.full_like(starter, 1.0 / starter.size)
        else:
            starter = None
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, base_n)
    else:
        a0 = np.random.rand(base_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _score_from_conv(_conv_self(best_seq,
                                            _next_pow2(2 * base_n - 1)),
                                  base_n)
    print(f"[{time.time() - start_time:.2f}s] init score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Phase 1 – exponentiated‑gradient (≈10 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.10 * remaining)

    a_exp, score_exp = _run_exp_gradient(
        best_seq,
        base_n,
        phase1_deadline,
        start_time,
        alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        iters_per_alpha=12000,
    )
    if score_exp < best_score:
        best_score = score_exp
        best_seq = a_exp.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: {score_exp:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Phase 2 – sub‑gradient descent (major chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.60 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            base_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: {score_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Phase 3 – targeted annealing on the base size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_anneal, score_anneal = _run_pair_annealing(
            best_seq,
            base_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=180_000,
        )
        if score_anneal < best_score:
            best_score = score_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] base anneal improve: {score_anneal:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Upsample to a larger problem (2×) and refine further
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        large_n = base_n * 2  # 8192
        up_seq = _upsample(best_seq, large_n)

        # quick sub‑gradient warm‑up (small budget)
        warm_deadline = time.time() + min(12.0, (deadline - time.time()) * 0.15)
        up_seq, warm_score = _run_subgradient_descent(
            up_seq,
            large_n,
            warm_deadline,
            start_time,
            best_score,
            max_iters=100_000,
        )
        if warm_score < best_score:
            best_score = warm_score
            best_seq = up_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] up‑sample warm‑up improve: {warm_score:.6f}")

        # intensive annealing on the larger size
        if time.time() < deadline - 5.0:
            anneal_deadline = deadline
            up_seq, up_score = _run_pair_annealing(
                up_seq,
                large_n,
                anneal_deadline,
                start_time,
                best_score,
                up_seq,
                start_temp=4e-2,
                end_temp=5e-10,
                steps=250_000,
            )
            if up_score < best_score:
                best_score = up_score
                best_seq = up_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] up‑sample anneal improve: {up_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Random‑restart loop (short perturbations) while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        time_left = deadline - time.time()
        block = min(25.0, time_left * 0.4)
        restart_deadline = time.time() + block

        # perturb the current best (Gaussian noise, then project)
        pert = best_seq + 0.03 * np.random.randn(best_seq.size)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # quick sub‑gradient stage
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            best_seq.size,
            restart_deadline,
            start_time,
            best_score,
            max_iters=80_000,
        )
        # short annealing after sub‑gradient
        if time.time() < deadline:
            pert_seq, pert_score = _run_pair_annealing(
                pert_seq,
                best_seq.size,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=100_000,
            )

        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: {pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] final score (approx): {best_score:.6f}")
    return [float(x) for x in best_seq]
