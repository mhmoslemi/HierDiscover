def _run_adam_subgradient(a0, n, deadline, start_time, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):

    m = np.zeros_like(a)

    t = 0
    best_a = a.copy()
    best_score = float('inf')
    while time.time()<deadline:
        t += 1
        conv=_conv_self(a)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] adam improve: iter={t:,} score={cur_score:.6f}")
        # compute subgradient
        grad = _subgradient_full(a, conv)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)

    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (soft‑max)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: Adam‑style projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.4,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam‑subgrad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: Targeted incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _move_candidate(a: np.ndarray,
                    conv: np.ndarray,
                    src: int,
                    dst: int,
                    eps_frac: float,
                    k_max: int) -> tuple[float, np.ndarray] | None:
    """
    Return (epsilon, s) for the move src→dst if it yields a genuine
    reduction of the current maximum, otherwise None.
    """
    n = a.shape[0]
    # construct directional change s
    s = np.zeros(2 * n - 1, dtype=float)
    s[src:src + n] -= 2.0 * a
    s[dst:dst + n] += 2.0 * a

    s_kmax = s[k_max]
    if s_kmax >= 0.0:
        return None

    # admissible step size before another entry catches up
    mask = s > s_kmax
    if np.any(mask):
        delta_conv = conv[k_max] - conv[mask]
        delta_s = s[mask] - s_kmax
        eps_cand = delta_conv / delta_s
        eps_cand = eps_cand[eps_cand > 0.0]
        eps_star = eps_cand.min() if eps_cand.size else np.inf
    else:
        eps_star = np.inf

    eps_limit = a[src] * eps_frac
    epsilon = min(eps_star, eps_limit)

    if epsilon <= 0.0:
        return None
    return epsilon, s


def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    src_k: int = 8,
                                    dst_k: int = 8,
                                    recompute_every: int = 3000,
                                    random_restart_interval: int = 15000):
    """
    Perform a huge number of targeted mass‑shifts.
    *src_k* / *dst_k* control the size of the candidate sets.
    Occasionally a small random perturbation is applied to escape
    stagnation.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    accepted = 0
    no_improve = 0   # counter for stagnation

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # ---- source candidates ------------------------------------------------
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.sum() > 0:
            # take the `src_k` indices with largest contribution
            top_idx = np.argpartition(-contrib, src_k - 1)[:src_k]
            src_candidates = i_range[top_idx]
        else:
            # fallback: any index with non‑zero mass
            src_candidates = rng.choice(np.where(a > 1e-12)[0],
                                        size=src_k, replace=False)

        # ---- destination candidates -------------------------------------------
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        # smallest contributions → best destinations
        low_idx = np.argpartition(dest_contrib, dst_k - 1)[:dst_k]
        dst_candidates = low_idx[low_idx != -1]  # just in case

        # ensure we have at least two distinct indices
        improved = False
        best_impr = -np.inf
        best_move = None

        for src in src_candidates:
            if a[src] < 1e-14:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue
                mv = _move_candidate(a, conv, int(src), int(dst), eps_frac, k_max)
                if mv is None:
                    continue
                epsilon, s = mv
                improvement = -epsilon * s[k_max]      # positive number
                if improvement > best_impr:
                    best_impr = improvement
                    best_move = (src, dst, epsilon, s)

        if best_move is not None:
            src, dst, epsilon, s = best_move
            conv = conv + epsilon * s
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = conv.max()
            accepted += 1
            it += 1
            no_improve = 0
            if max_val < best_max - 1e-15:
                best_max = max_val
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")
        else:
            # ---- stagnation handling -----------------------------------------
            no_improve += 1
            if no_improve >= random_restart_interval // src_k:
                # small random perturbation to shake the system
                sigma = 0.02 * eps_frac          # scale with current epsilon
                perturb = rng.normal(scale=sigma, size=n)
                a_pert = a + perturb
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = _project_simplex(a_pert)
                conv = _conv_self(a_pert)
                a = a_pert
                max_val = conv.max()
                no_improve = 0
                # keep the best seen so far
                print(f"[{time.time() - start_time:.2f}s] random perturb applied")

        # occasional full recomputation to keep numerical errors in check
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    # ---- timing ---------------------------------------------------------
    start_time = time.time()
    # safety margin of half a second
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed)

    # ---- problem size ---------------------------------------------------
    # Larger n gives more freedom; 8000 is a good compromise.
    n = 8000

    # ---- initialisation -------------------------------------------------
    try:
        # try to use the previously discovered construction
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, n)
    else:
        a0 = rng.random(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exponential‑gradient warm‑up (smooth max)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 20.0:
        phase1_deadline = time.time() + min(40.0, 0.07 * time_left)
        a_warm, score_warm = _run_exponential_gradient(
            a0,
            n,
            phase1_deadline,
            start_time,
            alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
            eta0=0.4,
            steps_per_alpha=8000,
        )
        if score_warm < best_score:
            best_score = score_warm
            best_seq = a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑1 best: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 100.0:
        phase2_deadline = time.time() + max(30.0, 0.65 * time_left)
        a_sub, score_sub = _run_adam_subgradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            lr=0.5,
            beta1=0.9,
            beta2=0.999,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 best: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted incremental refinement (mass shifts)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 20.0:
        a_ref = _run_optimal_incremental_refine(
            best_seq,
            n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.001,
            src_k=10,
            dst_k=10,
            recompute_every=2500,
            random_restart_interval=15000,
        )
        conv_ref = _conv_self(a_ref)
        score_ref = 2.0 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] final best: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as a plain list of floats
    return [float(x) for x in best_seq]
