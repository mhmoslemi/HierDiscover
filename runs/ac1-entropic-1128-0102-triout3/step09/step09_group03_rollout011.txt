def _expgrad_multi(a0, deadline, start_time, fft_len, eta0=0.5, top_k=5):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * len(a) * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # select top_k indices
        top_idxs = np.argpartition(conv, -top_k)[-top_k:]
        weights = conv[top_idxs] / conv[top_idxs].sum()  # normalize as weight distribution (maybe softmax)
        # compute gradient as weighted sum of per-shift gradients
        g = np.zeros_like(a)
        n = a.shape[0]
        for w_idx, k in zip(weights, top_idxs):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += w_idx * 2.0 * a[j]
        # step size
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
* Build a diverse pool of start vectors (uniform, exponential, Gaussian,
  Dirichlet, “U‑shaped” double‑exponential and the previously best known).
* A short Adam phase (soft‑max surrogate) quickly leaves very bad regions.
* Main optimisation: exponentiated‑gradient descent where the update direction
  is a *weighted* sum of the gradients of the *top‑k* convolution peaks.
  The weighted gradient is obtained by a single FFT‑based correlation, so the
  cost per iteration is comparable to the original single‑peak version but
  flattens several peaks at once.
* Random restarts with small Gaussian perturbations are performed while time
  permits – each restart runs the multi‑peak EG for a few seconds.
* Finally a deterministic greedy “peak‑reduction’’ move is applied.
* The resulting vector is normalised, clipped to the allowed range and
  returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of a log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 250 == 0:
                cur = evaluate_sequence(a.tolist())
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={cur:.6f}"
                    )

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation using the *top‑k* convolution peaks
# ----------------------------------------------------------------------


def _expgrad_multi(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eta0: float = 0.5,
                   top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) where the update direction is a
    weighted sum of the gradients of the `top_k` largest convolution entries.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad‑multi improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ---- build a weight vector that concentrates on the top‑k peaks ----
        w = np.zeros_like(conv)
        if top_k >= conv.size:
            w[:] = 1.0 / conv.size
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            # proportional to the peak heights (soft‑maxish) – normalise to 1
            weights = conv[idxs]
            total = weights.sum()
            if total > 0:
                w[idxs] = weights / total
            else:
                w[idxs] = 1.0 / top_k

        # ---- gradient of the weighted sum  (FFT‑based correlation) ----
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ---- exponentiated‑gradient step ----
        eta = eta0 / math.sqrt(t)
        # clipping the gradient avoids overflow in the exponent
        grad = np.clip(grad, -1000.0, 1000.0)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation (helps with drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] ExpGrad‑multi exact: iter={t:,}, score={sc:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local improvement (same as the previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Transfer a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _double_exponential_start(n: int, lam: float = 0.0008) -> np.ndarray:
    """U‑shaped distribution: high at the two ends, low in the centre."""
    xs = np.arange(n)
    left = np.exp(-lam * xs)
    right = np.exp(-lam * (n - 1 - xs))
    vec = left + right
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of:
        1. a pool of diverse starts,
        2. a brief Adam warm‑up,
        3. an extensive multi‑peak exponentiated‑gradient phase with random
           restarts, and
        4. a final greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build diverse initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 0) previous construction, if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 1) uniform
    starts.append(np.full(n, 1.0 / n))

    # 2) exponential – gentle decay
    starts.append(_exponential_start(n, r=0.9995))

    # 3) exponential – stronger decay
    starts.append(_exponential_start(n, r=0.985))

    # 4) Gaussian bell
    starts.append(_gaussian_start(n))

    # 5) Dirichlet – moderate concentration
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 6) Dirichlet – low concentration (spikier)
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 7) double‑exponential (U‑shaped)
    starts.append(_double_exponential_start(n, lam=0.0008))

    # a few random Dirichlet draws with random concentration
    for _ in range(3):
        conc = rng.uniform(0.1, 2.0)
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # Evaluate raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – Adam warm‑up (≈10 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.1
    cand_seq, cand_score = _adam_optimize(
        best_seq, explore_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – Multi‑peak exponentiated‑gradient with random restarts
    # ------------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.85   # leave ~5 % for greedy
    while time.time() < polish_deadline:
        # create a perturbed start from the current best
        noise = rng.normal(0.0, 0.01, size=n)
        a_start = best_seq + noise
        a_start = np.maximum(a_start, 1e-12)
        a_start /= a_start.sum()

        # give each run a short personal deadline (≈20 s or whatever is left)
        sub_deadline = min(polish_deadline, time.time() + 18.0)

        seq, sc = _expgrad_multi(
            a_start,
            sub_deadline,
            start_time,
            fft_len,
            eta0=0.5,
            top_k=7,
        )
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after multi‑EG score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic greedy peak‑reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety checks and normalisation
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
