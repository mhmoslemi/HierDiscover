```
def _expgrad_softmax_optimize(a0, deadline, start_time, n, fft_len_self, alpha=64.0, eta0=0.5):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] SoftMG improve: iter={t:,}, score={cur_score:.6f}")
        # compute softmax weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum
        # compute gradient
        grad = _grad_from_weights(a, w)
        # multiplicative update
        eta = eta0 / math.sqrt(t)
        # Clip grad magnitude to avoid overflow
        max_exp_arg = 100.0
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -max_exp_arg, max_exp_arg)
        a = a * np.exp(exp_arg)
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # occasional exact eval
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score
```

The scoring function  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2}
\]

depends only on the *shape* of the coefficients, because the scaling factor
cancels out.  After normalising the vector to a probability distribution
\(p_i\ge 0,\ \sum_i p_i=1\) the problem becomes

\[
\min_{p\ge 0,\; \mathbf 1^{\!\top}p=1}\; 2n\;\max_k (p*p)[k] .
\]

The convolution \(c=p*p\) is the auto‑correlation of the distribution.
A perfect solution would make this auto‑correlation completely flat
(the theoretical lower bound is \(2n/(2n-1)\approx 1\)).
The previous code reaches a score of **1.512 …** – the maximum of the
convolution is still far from the flat optimum.

The new search routine improves the previous pipeline in three ways

1. **Soft‑max gradient descent** – instead of using only the single peak
   we take a *soft* maximum (log‑sum‑exp) with a fixed temperature.
   The gradient is the (weighted) sum of the gradients of *all* large
   peaks, which makes the optimisation much less prone to get stuck.

2. **More aggressive multi‑phase schedule** – a short Adam warm‑up,
   a long soft‑max exponentiated‑gradient phase, a deterministic greedy
   peak‑reduction phase and finally a lightweight simulated‑annealing
   local search.  Each phase uses a dedicated time slice of the 1000 s
   budget.

3. **Vector‑wise updates** – all convolutions are performed with FFTs,
   the gradient for the soft‑max phase is obtained by a single extra
   convolution, and the updates stay on the probability simplex by a
   multiplicative (exponentiated‑gradient) step followed by a normalisation.

The whole algorithm respects the 1000 s wall‑clock limit, prints
intermediate scores and finally returns the best sequence it has found.

```python
"""
Search for a non‑negative step function that (almost) minimises
evaluate_sequence.  The method consists of four phases:

* Adam warm‑up on a log‑sum‑exp surrogate of the max‑convolution.
* Long soft‑max exponentiated‑gradient optimisation – the gradient is the
  weighted sum of *all* almost‑maximal convolution shifts, not only the
  single peak.
* Deterministic greedy peak‑reduction (same idea as the reference code).
* A tiny simulated‑annealing local search for the final few seconds.

All phases respect the supplied time budget and the final vector is
projected onto the probability simplex (non‑negative, sum‑to‑one)."""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT helpers – full linear convolution
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using FFT.  Returns an array of length
    len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    fft_len = _next_pow2(n + m - 1)
    Af = rfft(a, n=fft_len)
    Bf = rfft(b, n=fft_len)
    conv = irfft(Af * Bf, n=fft_len)
    return conv[: n + m - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    Af = rfft(a, n=fft_len)
    conv = irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Gradient from a soft‑max weight vector
# ----------------------------------------------------------------------


def _grad_from_weights(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   max_w  (p * p)   w.r.t. p when w is a weight vector of
    length 2·n‑1 (the same length as a self‑convolution).  The gradient is
    2·(p * w_rev) taken at the positions of p.
    """
    w_rev = w[::-1]
    full = _conv_full_fft(p, w_rev)                # length 3·n‑2
    offset = len(w_rev) - 1                         # = 2·n‑2
    grad = 2.0 * full[offset - np.arange(p.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 0‑dimensional start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 0 – short Adam warm‑up on a smooth surrogate
# ----------------------------------------------------------------------


def _smooth_objective(p: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of the max‑convolution."""
    conv = _conv_self(p, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(p0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int) -> tuple[np.ndarray, float]:
    """A few Adam iterations on the smoothed max‑convolution objective."""
    p = p0.copy()
    n = p.shape[0]

    # work in the log‑space of the soft‑max (logits)
    y = np.log(np.maximum(p, 1e-16))

    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_p = p.copy()
    best_score = float(evaluate_sequence(p.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_p, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            p = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(p, alpha, fft_len)

            # true objective (for printing only)
            obj = 2.0 * n * f_val
            # gradient of the smooth surrogate w.r.t. p  (see paper)
            scaled = alpha * conv
            max_s = scaled.max()
            w = np.exp(scaled - max_s)
            w /= w.sum()
            w_rev = w[::-1]
            grad = _grad_from_weights(p, w)

            # back‑prop through the soft‑max
            grad_y = p * (grad - np.dot(grad, p))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y -= lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasionally evaluate the exact objective
            if t % 250 == 0:
                sc = evaluate_sequence(p.tolist())
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}")

    return best_p, best_score


# ----------------------------------------------------------------------
# Phase 1 – soft‑max exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _expgrad_softmax_optimize(p0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              n: int,
                              fft_len_self: int,
                              alpha: float = 64.0,
                              eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *soft* max‑convolution.
    The weight vector w is a soft‑max (log‑sum‑exp) with temperature 1/α.
    """
    p = np.maximum(p0, 1e-15)
    p = p / p.sum()
    best_p = p.copy()
    best_score = float('inf')
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(p, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[{time.time() - start_time:6.2f}s] Soft‑grad improve: iter={t:,}, score={cur_score:.6f}")

        # soft‑max weights over all shifts
        scaled = alpha * conv
        m = scaled.max()
        w = np.exp(scaled - m)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient (sum of the individual gradients weighted by w)
        grad = _grad_from_weights(p, w)

        # multiplicative update – move opposite to the gradient
        eta = eta0 / math.sqrt(t)
        # protect against overflow in the exponent
        max_exp = 100.0
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -max_exp, max_exp)
        p = p * np.exp(exp_arg)

        # re‑normalise onto the simplex
        s = p.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            p = np.full_like(p, 1.0 / n)
        else:
            p = p / s

        # occasional exact evaluation for safety
        if t % 500 == 0:
            sc = evaluate_sequence(p.tolist())
            if sc < best_score:
                best_score = sc
                best_p = p.copy()

    return best_p, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy peak reduction (same as before)
# ----------------------------------------------------------------------


def _greedy_local_improve(p0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of the coefficient that contributes most to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly improving moves.
    """
    p = p0.copy()
    n = p.shape[0]
    best_score = 2.0 * n * _conv_self(p, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(p, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = p[i_idx] * p[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[p[dst_candidates].argmin()]

        eps = 0.25 * p[src]               # modest fraction
        if eps <= 1e-14:
            break

        p_new = p.copy()
        p_new[src] -= eps
        p_new[dst] += eps

        new_max = _conv_self(p_new, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            p = p_new
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return p, best_score


# ----------------------------------------------------------------------
# Phase 3 – light simulated annealing for the last seconds
# ----------------------------------------------------------------------


def _anneal_optimize(p0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     init_temp: float = 0.02,
                     decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style local search.
    Small mass transfers between two random positions are accepted
    according to a decreasing temperature schedule.
    """
    p = p0.copy()
    current_score = evaluate_sequence(p.tolist())
    best_p = p.copy()
    best_score = current_score
    n = p.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        # move a random fraction (up to half) of the source mass to the target
        if p[i] <= 0.0:
            continue
        eps = rng.random() * min(p[i], 0.5 * p[i])
        p_new = p.copy()
        p_new[i] -= eps
        p_new[j] += eps

        new_score = evaluate_sequence(p_new.tolist())
        delta = new_score - current_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            p = p_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={new_score:.6f}")

        T *= decay
        if T < 1e-8:
            T = 1e-8

    return best_p, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of:
        1. a short Adam warm‑up,
        2. a long soft‑max exponentiated‑gradient phase,
        3. deterministic greedy peak reduction,
        4. a tiny simulated‑annealing local search.
    The best vector found is returned as a list of ``float``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size – keep the same length that has been used before
    # --------------------------------------------------------------
    n = 4800
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build diverse starting points (including the previous best)
    # --------------------------------------------------------------
    starts = []

    # 1) previously best sequence, if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) gentle exponential decay
    starts.append(_exponential_start(n, r=0.9995))

    # 4) stronger exponential decay
    starts.append(_exponential_start(n, r=0.985))

    # 5) Gaussian bell
    starts.append(_gaussian_start(n))

    # 6) Dirichlet – moderate concentration
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 7) Dirichlet – low concentration (spikier)
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 8) a few extra Dirichlet draws
    for conc in (0.1, 5.0, 10.0):
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # --------------------------------------------------------------
    # Pick the best raw start
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – short Adam exploration (≈8 % of the total budget)
    # --------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.08
    cand_seq, cand_score = _adam_optimize(best_seq, explore_deadline, start_time, fft_len_self)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – soft‑max exponentiated‑gradient polishing (≈78 % of budget)
    # --------------------------------------------------------------
    soft_deadline = start_time + budget_s * 0.86   # 8 % + 78 % = 86 %
    if time.time() < soft_deadline:
        polished_seq, polished_score = _expgrad_softmax_optimize(
            best_seq,
            soft_deadline,
            start_time,
            n,
            fft_len_self,
            alpha=64.0,
            eta0=0.7,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after soft‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic greedy peak‑reduction (≈10 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=8000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – very cheap simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal_optimize(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation / safety checks
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
```
