"""
Propose a low‑max self‑convolution step‑function.

The algorithm is a multi‑start convex optimisation routine
following three ideas:

1️⃣  Initialise several symmetric candidates (the known construction,
    a uniform distribution and a triangular shape) and keep the best.

2️⃣  For each start perform
      * an exponentiated‑gradient “soft‑max’’ warm‑up,
      * a projected Adam‑style sub‑gradient descent on the exact
        ℓ∞‑norm of the convolution (the objective is convex).
    After every update the vector is symmetrised and projected onto
    the simplex (∑a_i = 1, a_i ≥ 0).

3️⃣  Finally run a fast “mass‑transfer’’ refinement that moves a
    fraction of mass from a source index (which contributes heavily
    to the current maximal convolution entry) to a destination index
    (which contributes little).  The move size ε is chosen analytically
    as the largest step that still reduces the current maximum, and the
    move is performed simultaneously on the symmetric counterpart to
    preserve symmetry.

All three phases respect the overall 1000 s wall‑clock budget.
The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT helpers, simplex projection, up‑sampling, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry (v[i] = v[n-1-i]) and renormalise to sum 1."""
    v = (v + v[::-1]) * 0.5
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v[:] = 1.0 / v.shape[0]
    return v


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient warm‑up (soft‑max smoothing)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 8000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (soft‑max with temperature 1/alpha)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative mirror‑descent update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _symmetrize(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected Adam sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.15,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑subgrad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv, rel_tol=1e-12)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _symmetrize(a)        # keep symmetry & normalisation

    return best_a, best_score

# Phase 3 – Symmetric optimal incremental refinement
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    cand_per_iter: int = 5,
                                    recompute_every: int = 2000):
    """
    Large‑step mass‑shifts that preserve symmetry.
    Each iteration samples a few destination candidates and picks the
    one giving the greatest reduction of the current maximal convolution entry.
    """
    a = a0.copy()
    a = _symmetrize(a)

    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0
    accepted = 0

    while time.time() < deadline:
        it += 1
        # linear schedule for allowable fraction of source mass
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # ---- source index (contributes to the current max bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- destination candidates (prefer low contribution) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()

        cand_dst = rng.choice(n, size=cand_per_iter, replace=False, p=prob_dst)

        best_eps = 0.0
        best_new_max = max_val
        best_dst = None
        best_s = None

        for dst in cand_dst:
            if dst == src:
                continue

            # ----- symmetric move vectors -----
            src_sym = n - 1 - src
            dst_sym = n - 1 - dst

            # compute s = contribution of moving epsilon from src->dst and src_sym->dst_sym
            s = np.zeros(2 * n - 1, dtype=float)
            # first pair
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a
            # symmetric pair (avoid double counting when src==src_sym or dst==dst_sym)
            if src_sym != src:
                s[src_sym:src_sym + n] -= 2.0 * a
            if dst_sym != dst:
                s[dst_sym:dst_sym + n] += 2.0 * a

            s_max = s[k_max]
            if s_max >= 0.0:
                continue    # this direction cannot lower the current max

            # maximal admissible ε before another entry overtakes the max
            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[k_max] - conv[mask]          # >=0
                delta_s = s[mask] - s_max                     # >0
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf

            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)

            if epsilon <= 0.0:
                continue

            new_max = (conv + epsilon * s).max()
            if new_max < best_new_max - rel_improve:
                best_new_max = new_max
                best_eps = epsilon
                best_dst = dst
                best_s = s

        if best_dst is None or best_eps <= 0.0:
            continue

        # ---- apply the best move ----
        dst = best_dst
        epsilon = best_eps
        src_sym = n - 1 - src
        dst_sym = n - 1 - dst

        # update a (maintaining symmetry)
        a[src] -= epsilon
        a[dst] += epsilon
        if src_sym != src:
            a[src_sym] -= epsilon
        if dst_sym != dst:
            a[dst_sym] += epsilon

        # update convolution linearly
        conv = conv + epsilon * best_s
        max_val = best_new_max
        accepted += 1

        if max_val < best_max - rel_improve:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                  f"iter={accepted:,}, max={best_max:.8f}")

        # Re‑compute convolution occasionally to avoid drift
        if accepted % recompute_every == 0:
            a = _symmetrize(a)
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a

    def _make_initials(final_n):

        initials.append(np.full(final_n, 1.0 / final_n))

        tri /= tri.sum()
        initials.append(tri)

        # 4. Random Dirichlet (more exploration)
        rand_vec = np.random.rand(final_n)
        rand_vec /= rand_vec.sum()
        initials.append(rand_vec)

        # enforce symmetry on all
        sym_inits = [_symmetrize(v) for v in initials]
        return sym_inits

    # ------------------------------------------------------------------
    # Problem size (you may tune this; larger n gives more freedom)
    # ------------------------------------------------------------------
    final_n = 4800

    # ------------------------------------------------------------------
    # Multi‑start optimisation
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    initial_vectors = _make_initials(final_n)

    # allocate roughly equal time to each start (except the last which gets the remainder)
    per_start_budget = (deadline - time.time()) * 0.7 / len(initial_vectors)
    for idx, a0 in enumerate(initial_vectors):
        start_deadline = time.time() + per_start_budget
