def _gradient_descent_softmax(initial_a, n, deadline, start_time, best_score):
    a = initial_a.copy()
    y = np.log(a + 1e-30)
    L = _next_pow2(3*n)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9; beta2=0.999; eps=1e-8
    step0=0.1
    it=0
    best_a = a.copy(); best_s = best_score
    while time.time() < deadline:
        it+=1
        # compute a via softmax
        exp_y = np.exp(y - np.max(y))
        a = exp_y / exp_y.sum()
        # compute convolution
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # denominator is 1
        if cur_score < best_s:
            best_s = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Softmax GD it={it} score={cur_score:.6f}")
        # softmax weight
        max_c = cur_max # for numeric stability may use max
        # choose alpha as schedule
        # Linear schedule: alpha = min(alpha_max, alpha_start + (alpha_max - alpha_start) * it / max_iters)
        # but we don't know max_iters; we can set based on elapsed time fraction.
        t_frac = (time.time() - start_time) / (deadline - start_time)
        alpha = 5.0 + 200.0 * t_frac  # start with 5, end 205
        # compute weight vector w
        w_raw = np.exp(alpha * (conv - max_c))
        w = w_raw / w_raw.sum()
        # compute gradient via correlation
        W = np.fft.rfft(w, n=L)
        grad = 2.0 * np.fft.irfft(np.conj(A) * W, n=L)[:n]
        # project gradient to sum zero
        grad = grad - np.mean(grad)
        # adam updates on y
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps)
        y = y - step * m_hat
    return best_a, best_s

"""
search for a low‑value step‑function.

The strategy is a multi‑stage optimiser:

1️⃣  Fast parametric sweep over several analytic families (geometric,
    power, Beta‑type, classic window functions).  The best candidate
    of this cheap sweep is kept.

2️⃣  Tempered soft‑max gradient descent:  minimise a smooth
    approximation of  max(conv(a,a))  by using the gradient of the
    soft‑max (log‑sum‑exp) of the convolution values.  The update is
    performed on the log‑its (exponential map) so that the iterate
    always stays on the probability simplex.  An Adam‑style adaptive
    step‑size is used and the temperature (α) is increased
    linearly, interpolating between the average and the true max.

3️⃣  A projected sub‑gradient descent (as in the reference solution)
    refines the solution further.

4️⃣  A light simulated‑annealing “mass‑transfer’’ polish removes any
    remaining tiny peaks.

The whole pipeline respects the 1000 s wall‑clock budget and returns
a plain Python list of non‑negative heights ready for ``evaluate_sequence``.
"""

import math
import numpy as np

# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Cross‑correlation of a (length n) with w (length 2·n‑1).
    Returns the first n entries (lags 0 … n‑1).
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(A) * W, n=L)[: a.shape[0]]
    return corr


# ----------------------------------------------------------------------
# Simplex projection (used only for the fallback sub‑gradient phase)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation for a unit‑sum candidate
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 2️⃣  Tempered soft‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              best_initial_score: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth surrogate of max(conv(a,a)) while staying on the
    probability simplex.  The surrogate uses a temperature α that is
    increased linearly from 5 to 205 throughout the run.
    An Adam‑style update is applied to log‑its (y) so that positivity
    is automatic.
    """
    # initialise log‑its
    y = np.log(a0 + 1e-30)

    L = _next_pow2(3 * n)               # length for all FFTs
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.12                         # initial learning rate
    it = 0

    best_a = a0.copy()
    best_score = best_initial_score

    while time.time() < deadline:
        it += 1

        # ----- current probability vector -----
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # ----- convolution of a with itself -----
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # denominator is 1 (sum a = 1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Softmax‑GD it={it:,} score={cur_score:.6f}")

        # ----- temperature schedule -----
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        alpha = 5.0 + 200.0 * elapsed_frac      # grows from 5 → 205

        # ----- soft‑max weights over all lags -----
        # shift by current maximum for numerical stability
        w_raw = np.exp(alpha * (conv - cur_max))
        w = w_raw / w_raw.sum()                # sum(w)=1

        # ----- gradient of the smooth surrogate -----
        grad = 2.0 * _cross_corr(a, w, L)       # length n
        # enforce zero‑sum (tangent to the simplex)
        grad -= np.mean(grad)

        # ----- Adam update on log‑its -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        y = y - step * m_hat

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient descent (same spirit as the reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient method using the active lag.
    Momentum (β≈0.9) and a 1/√t step‑size are employed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active lag
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # gradient of aᵀ Sₖ a is 2·a_{k‑i}
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Light simulated‑annealing polish (mass transfer)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealer."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # baseline – uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # try to use the global starter if present
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time()-start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Fast parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = start_time + 120.0  # first two minutes for the sweep
    idx = np.arange(n)

    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            return
        seq = seq / s
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time()-start_time:.2f}s] New best score={sc:.6f}")

    # geometric family
    for r in np.linspace(0.10, 0.999, 120):
        if time.time() >= sweep_deadline:
            break
        geo = r ** idx
        _maybe_update(geo)
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            _maybe_update(seq)

    # power family
    for p in np.linspace(-1.5, 3.0, 200):
        if time.time() >= sweep_deadline:
            break
        pow_seq = (idx + 1) ** p
        _maybe_update(pow_seq)

    # Beta‑type family (discrete analogue of a Beta density)
    beta_grid = np.linspace(-0.5, 3.0, 15)
    for p in beta_grid:
        for q in beta_grid:
            if time.time() >= sweep_deadline:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            _maybe_update(beta_seq)

    # classic window functions
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * idx / (n - 1))
    _maybe_update(hann)
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * idx / (n - 1))
    _maybe_update(hamming)
    blackman = (0.42 - 0.5 * np.cos(2 * np.pi * idx / (n - 1))
                + 0.08 * np.cos(4 * np.pi * idx / (n - 1)))
    _maybe_update(blackman)

    # sinusoidal power
    for p in np.linspace(0.5, 3.0, 30):
        if time.time() >= sweep_deadline:
            break
        sin_seq = np.sin(np.pi * idx / (n - 1)) ** p
        _maybe_update(sin_seq)

    # --------------------------------------------------------------
    # 2️⃣  Tempered soft‑max gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_time = remaining * 0.55
        gd_deadline = time.time() + gd_time
        best_seq_gd, best_score_gd = _softmax_gradient_descent(
            best_seq, n, gd_deadline, start_time, best_score)
        if best_score_gd < best_score:
            best_score = best_score_gd
            best_seq = best_seq_gd.copy()
            print(f"[{time.time()-start_time:.2f}s] Softmax‑GD final best score={best_score_gd:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_time = (deadline - time.time()) * 0.35
        sub_deadline = time.time() + sub_time
        a_sub, sc_sub = _subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time()-start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Light annealing polish (uses the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_fin, sc_fin = _annealing(
            best_seq, n, anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-7, steps=250_000)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"[{time.time()-start_time:.2f}s] Anneal final improve score={sc_fin:.6f}")

    # --------------------------------------------------------------
    # final normalisation and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = uniform

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
