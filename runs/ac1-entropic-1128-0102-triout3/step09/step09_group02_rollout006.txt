"""
Propose a step‑function with a low self‑convolution peak.

The algorithm consists of four lightweight phases that run within the
budget (default 1000 s):

1. **Adam on a smoothed max** – the max of the convolution is replaced
   by a log‑sum‑exp with a decreasing smoothing factor `alpha`.  The
   gradient is obtained analytically via FFT and Adam updates keep the
   vector on the probability simplex.

2. **Exponentiated‑gradient descent on the true max** – after the first
   phase the exact max is used.  A mirror‑descent (multiplicative) step
   moves probability mass away from the indices that contribute most to
   the current peak.

3. **Simple mixing / refinement** – the current best vector is mixed
   with the uniform distribution and with its reversed copy; a small
   line‑search over the mixing weight often reduces the peak further.

4. **Random local hill‑climbing** – whenever time remains we perform
   tiny mass‑transfer moves that are accepted only if they improve the
   true objective.  This final polishing can shave the last few
   10⁻⁴ off the score.

All phases respect the 0.5 s safety margin before the deadline.
The returned list contains non‑negative numbers that sum to one, ready
to be fed to `evaluate_sequence`.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (assumes a ≥ 0)."""
    n = a.shape[0]
    if n == 0:
        return float("inf")
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 alphas,
                 iters_per_alpha: int = 12000,
                 lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam minimising a log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            # full convolution
            conv = _conv_self(a)

            # keep track of true objective
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w_sum = w.sum()
            w = w / w_sum
            w_rev = w[::-1]

            # ----- gradient of the smooth max (via FFT) -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project onto simplex
            a = _project_simplex(a)

    # final true objective for the best we have seen
    final_score = _objective(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative) update on the true max."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index of the current maximum of the convolution
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # gradient of the active max entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise (exactly onto simplex)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simple mixing with uniform / reversed copy
# ----------------------------------------------------------------------
def _mix_refine(a: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Search a 1‑D mixing line between `a` and a few simple companions."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    reversed_a = a[::-1]

    candidates = [(uniform, "uniform"), (reversed_a, "reversed")]
    best_a = a.copy()
    best_score = _objective(a)

    # try 21 uniformly spaced mixing weights in [0, 0.2]
    for lam in np.linspace(0.0, 0.20, 21):
        if time.time() >= deadline:
            break
        for comp, name in candidates:
            mixed = (1.0 - lam) * a + lam * comp
            mixed = np.maximum(mixed, 0.0)
            mixed /= mixed.sum()
            sc = _objective(mixed)
            if sc < best_score:
                best_score = sc
                best_a = mixed.copy()
                print(f"[{time.time() - start_time:.2f}s] mix ({name}) lam={lam:.3f} score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Random local hill‑climbing (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                max_trials: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    rng = np.random.default_rng()
    it = 0

    # pre‑compute a list of indices that currently hold non‑zero mass
    while time.time() < deadline and it < max_trials:
        it += 1

        # pick a source index proportional to its current mass
        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-15:
            continue

        # destination is chosen uniformly
        dst = rng.integers(n)
        if dst == src:
            continue

        # transfer a small fraction (up to 5 %) of the source mass
        eps = a[src] * rng.random() * 0.05
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # maintain non‑negativity (tiny numerical noise)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()   # renormalise (keeps sum == 1)

        sc = _objective(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improve: "
                  f"iter={it:,}, score={sc:.6f}")
        # otherwise discard the move (keep a unchanged)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Initialise – use the supplied construction if present,
    # otherwise start from a random simplex point.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # may be undefined
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    final_n = 4096
    if starter is not None:
        # interpolate the previous construction onto the target size
        a0 = np.interp(np.linspace(0.0, 1.0, final_n),
                       np.linspace(0.0, 1.0, starter.size),
                       starter)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = rng.random(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective(best_seq)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smoothed max (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + max(8.0, 0.15 * (deadline - time.time()))
    alphas = [40.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2]
    a_adam, score_adam = _adam_smooth(
        best_seq,
        phase1_deadline,
        start_time,
        alphas,
        iters_per_alpha=12000,
        lr0=0.5,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent on the true max (≈25 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(20.0, 0.25 * (deadline - time.time()))
        a_exp, score_exp = _exp_gradient(
            best_seq,
            phase2_deadline,
            start_time,
            eta0=0.6,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Mixing with uniform / reversed copy (≈10 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + max(8.0, 0.10 * (deadline - time.time()))
        a_mix, score_mix = _mix_refine(best_seq, phase3_deadline, start_time)
        if score_mix < best_score:
            best_score = score_mix
            best_seq = a_mix.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_mix:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
