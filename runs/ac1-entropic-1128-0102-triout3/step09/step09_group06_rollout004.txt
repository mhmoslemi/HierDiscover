def _random_hill_climb(a0, conv0, deadline, start_time):
    a = a0.copy()
    conv = conv0.copy()
    best_a = a.copy()
    best_score = 2*n*conv.max()
    it=0
    while time.time() < deadline:
        it+=1
        # choose src weighted by contribution maybe from conv peak
        src = np.random.randint(0,n)
        if a[src] <= 1e-12:
            continue
        # choose dst random
        dst = np.random.randint(0,n)
        if dst == src:
            continue
        # choose fraction f randomly in [0.01,0.2] maybe also variable
        f = np.random.uniform(0.01, 0.2)  # fraction of src mass
        eps = min(f * a[src], a[src])
        # apply update to conv
        conv_new = conv.copy()
        conv_new[src:src+n] -= 2*eps*a
        conv_new[dst:dst+n] += 2*eps*a
        # quadratic terms
        conv_new[2*src] += eps*eps
        conv_new[2*dst] += eps*eps
        conv_new[src+dst] -= 2*eps*eps
        new_max = conv_new.max()
        if new_max < conv.max() - 1e-12:  # improvement
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = 2*n*new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] hillclimb iter={it:,} score={cur_score:.6f}")
        # else reject
    return best_a, best_score

"""
Propose a low‐convolution step function.

Strategy
--------
1. Warm‑start from the historic best (`height_sequence_1`) and normalize
   it to a probability vector of length *n* (default n=5000).  If no
   historic data is available fall back to a uniform distribution.

2. **Adam on a smooth approximation** of the max‑convolution.
   We minimise the log‑sum‑exp surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
   with an increasing schedule of α.  After each Adam step the vector
   is projected onto the probability simplex (guaranteeing non‑negativity
   and Σa = 1).  The best iterate (according to the exact objective)
   is kept.

3. **Projected sub‑gradient descent** on the true objective
        max_k (a*a)[k]
   with momentum and a slowly decreasing step size.  This gives a fine
   polishing of the Adam solution.

4. **Random hill‑climbing** in the remaining time.
   A small amount of mass is moved from a “source’’ index to a “destination’’
   index; the convolution is updated incrementally (O(n)) and the move is
   accepted only if the peak of the convolution strictly decreases.
   The source is chosen among the indices that contribute most to the
   current maximal convolution entry, while the destination is the current
   smallest coefficient.  This phase explores moves that the deterministic
   greedy step may miss.

5. Finally the candidate is clipped to [0,∞), renormalised to sum = 1 and
   evaluated with the provided ``evaluate_sequence`` function.

The whole routine respects the 1000 s time limit, prints intermediate
progress, and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function

        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)

    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Adam on the smooth max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max.
    Returns the best vector found (as a probability vector) and its exact
    score according to ``evaluate_sequence``.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α → tighter approximation of the true max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0,
                      512.0, 2048.0, 8192.0, 16384.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.10
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        # inner loop runs until either the global deadline or a short per‑α cap
        per_alpha_deadline = min(deadline, time.time() + 0.09 * (deadline - start_time))
        while time.time() < per_alpha_deadline:
            conv = _conv_self(a)                     # exact convolution
            scaled = α * conv
            max_scaled = scaled.max()
            # log‑sum‑exp (stable)
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                           # smooth max
            obj = 2.0 * n * f_val                     # same scaling as evaluate_sequence

            # gradient of the *unsmoothed* objective = 2·n·∂f_α/∂a
            grad_a = _grad_smooth_max(a, conv, α, fft_len) * (2.0 * n)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={α:.0f}) "
                          f"score={sc:.6f}")

            # early stop if total budget exhausted
            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent (fine polishing)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,} "
                  f"score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – random hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------


def _random_hill_climb(a0: np.ndarray,
                       conv0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from a high‑contributing index
    to the current smallest coefficient.  Moves are accepted only if the
    peak of the convolution strictly decreases.
    """
    a = a0.copy()
    conv = conv0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0

    # pre‑compute the index of the smallest coefficient – updates later
    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) pick the index that contributes most to the current peak
        k_max = conv.argmax()
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # 2) destination = smallest coefficient (different from src)
        if a[src] <= 1e-15:
            continue
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # 3) transfer a modest fraction of src to dst
        eps = 0.03 * a[src]                     # 3 % of source mass
        eps = min(eps, a[src] - 1e-15)          # stay non‑negative

        # 4) incremental update of the convolution
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        # quadratic corrections
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        if new_max < conv.max() - 1e-15:                # strict improvement
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = 2.0 * n * new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb iter={it:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score

# Helper to obtain a warm start
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build an initial candidate.
    If the historic best ``height_sequence_1`` is available, interpolate it
    to length n and normalise it to sum = 1.  Otherwise start from a uniform
    distribution.
    """
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation preserving total mass
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 5000

    # ------------------------------------------------------------------
    # 0) Warm start
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------
    # 1) Adam optimisation (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.45 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    best_a, best_score = _adam_optimize(a0, adam_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient polishing (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = start_time + 0.30 * budget_s
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, best_score = _subgrad_optimize(best_a, subgrad_deadline,
                                               start_time,
                                               beta=0.94,
                                               step0=0.05)
        print(f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Random hill‑climbing (use the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start random hill‑climb")
        conv0 = _conv_self(best_a)
        best_a, best_score = _random_hill_climb(best_a, conv0,
                                                deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] hill‑climb phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
