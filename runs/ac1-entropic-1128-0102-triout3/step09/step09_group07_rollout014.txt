"""
Propose a low‑max self‑convolution step‑function.

The search consists of several independent “runs’’ that each
(start from a different seed) perform three stages:

1️⃣  Exponentiated‑gradient warm‑up on a smoothed max (log‑sum‑exp).
2️⃣  Projected sub‑gradient descent on the exact ℓ∞‑norm.
3️⃣  Aggressive incremental refinement: each iteration selects a
    pair (src,dst) from the current maximal convolution bin,
    computes the largest admissible mass transfer ε (solution of a
    tiny 1‑D LP) and applies the move.  This version explores the
    best pair among a few promising candidates rather than a random
    pair, and it allows ε up to the theoretical limit (no artificial
    4 % cap).

The whole procedure respects the supplied wall‑clock budget
(≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All entries whose value lies within `rel_tol` of the maximum are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror‑descent on simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Stage 2 – projected sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Stage 3 – aggressive incremental refinement (pair selection)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.5,
                                    eps_end: float = 0.001,
                                    recompute_every: int = 2000):
    """
    Repeatedly moves mass from a “source’’ index that heavily contributes
    to the current maximal convolution bin to a “destination’’ index that
    contributes little.  For the selected pair we solve the 1‑D LP that
    yields the largest admissible step ε (no artificial 4 % cap).  Among
    a few promising candidates we pick the one that promises the biggest
    reduction of the current max.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    accepted = 0
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        # linearly anneal the allowed fraction of source mass (not a hard cap)
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # choose a handful of source candidates (largest contribution)
        # --------------------------------------------------------------
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.sum() > 0:
            # keep up to 5 best sources
            src_idxs = i_range[np.argsort(-contrib)[:5]]
        else:
            src_idxs = rng.choice(n, size=5, replace=False)

        # --------------------------------------------------------------
        # choose destination candidates (indices that appear little)
        # --------------------------------------------------------------
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.full(n, np.inf)
        dest_contrib[valid] = a[k_max - idx[valid]]
        dst_idxs = np.argsort(dest_contrib)[:5]   # smallest contribution

        best_improve = 0.0
        best_pair = None
        best_eps = None
        best_s = None

        # -----------------------------------------------------------------
        # evaluate each (src,dst) pair
        # -----------------------------------------------------------------
        for src in src_idxs:
            if a[src] <= 1e-14:
                continue
            for dst in dst_idxs:
                if src == dst:
                    continue
                # slope at the current max bin
                idx_src = k_max - src
                idx_dst = k_max - dst
                a_src = a[idx_src] if 0 <= idx_src < n else 0.0
                a_dst = a[idx_dst] if 0 <= idx_dst < n else 0.0
                s_max = 2.0 * (a_dst - a_src)          # negative means improvement
                if s_max >= -1e-14:
                    continue

                # full slope vector (only two shifted copies of a are needed)
                s = np.zeros(2 * n - 1, dtype=float)
                s[src:src + n] -= 2.0 * a
                s[dst:dst + n] += 2.0 * a

                # admissible ε before another entry overtakes the current max
                mask = s > s_max
                if np.any(mask):
                    delta_conv = max_val - conv[mask]          # ≥0
                    delta_s = s[mask] - s_max                 # >0
                    eps_candidate = delta_conv / delta_s
                    eps_star = eps_candidate[eps_candidate > 0].min()
                else:
                    eps_star = np.inf

                # we may not want to move all the mass at once – respect eps_frac
                eps_limit = a[src] * eps_frac
                epsilon = min(eps_star, eps_limit, a[src])

                if epsilon <= 0.0:
                    continue

                improve = -s_max * epsilon          # reduction of the current max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (src, dst)
                    best_eps = epsilon
                    best_s = s

        if best_pair is None:
            # fallback: random pair (as in the original code)
            src = rng.choice(np.where(a > 1e-15)[0])
            dst = rng.integers(0, n)
            if src == dst:
                continue
            # compute s, eps_star etc. (same as above) – here we just use a tiny step
            eps = a[src] * eps_frac
            if eps <= 0:
                continue
            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a
            conv_new = conv + eps * s
            new_max = conv_new.max()
            if new_max < max_val - 1e-12:
                conv = conv_new
                a[src] -= eps
                a[dst] += eps
                max_val = new_max
                accepted += 1
                it += 1
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                          f"iter={it:,}, max={best_max:.8f}")
            continue

        # --------------------------------------------------------------
        # apply the best move found
        # --------------------------------------------------------------
        src, dst = best_pair
        epsilon = best_eps
        conv = conv + epsilon * best_s
        a[src] -= epsilon
        a[dst] += epsilon
        max_val = conv.max()
        accepted += 1
        it += 1

        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                  f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep numerical errors in check
        if accepted % recompute_every == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
