Improved search for a low‑max self‑convolution step‑function.

The algorithm works on a fixed length `n` (default 4800) and iteratively
reduces the quantity

    2·n·max(a*a)      ( = evaluate_sequence(a)  for a normalised to sum = 1 )

by a *projected sub‑gradient* method with an *exact line search*.
For the current point `a` we

* compute the exact sub‑gradient of the ℓ∞‑norm of the self‑convolution,
* build a feasible direction `d` that preserves the simplex constraint,
* compute the maximal admissible step length `α` that keeps every convolution
  entry below the current maximum (i.e. solves a 1‑D linear programme exactly),
* update `a ← a + α·d` and re‑project onto the probability simplex.

If progress stalls a cheap random perturbation is performed and the search
continues until the time budget expires.  The best sequence seen is returned.

The whole routine respects a 1000 s wall‑clock budget and never produces
negative numbers, NaNs or infinities, so it is always a valid answer for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of two 1‑D arrays using FFT (full length)."""
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    c = np.fft.irfft(Af * Bf, n=L)
    return c[: a.shape[0] + b.shape[0] - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    return _conv_full_fft(a, a, L)


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All convolution indices whose value is within `rel_tol` of the maximum
    are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Core optimisation loop (projected sub‑gradient with line search)
# ----------------------------------------------------------------------
def _optimise(a: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Reduce 2·n·max(a*a) by a projected sub‑gradient method with exact line search.
    Returns the best sequence encountered before `deadline`.
    """
    n = a.shape[0]
    # pre‑allocated FFT length for convolutions of length up to 2·n‑1
    L_conv = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    stagnation = 0          # number of iterations without improvement

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and its maximum
        conv = _conv_self(a)
        max_val = conv.max()
        # compute exact sub‑gradient of the max‑convolution
        g = _subgradient_full(a, conv, rel_tol=1e-12)

        # direction that preserves the simplex (sum zero) and points downhill
        avg_g = np.mean(g)
        d = -g + avg_g          # Σ d_i = 0

        # if direction is (near) zero, fall back to a tiny random shuffle
        if np.linalg.norm(d) < 1e-16:
            src, dst = np.random.choice(n, 2, replace=False)
            eps = min(a[src] * 0.2, 0.001)
            a[src] -= eps
            a[dst] += eps
            a = _project_simplex(a)
            continue

        # slopes of every convolution entry when moving along d
        s = 2.0 * _conv_full_fft(a, d, L_conv)[: 2 * n - 1]

        # index of the current maximal bin
        k_max = int(np.argmax(conv))
        s_max = s[k_max]

        # if the slope of the maximal bin is not negative we cannot decrease it:
        if s_max >= -1e-12:
            # simple heuristic: move mass from the index with largest gradient
            # to the index with smallest gradient
            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src != dst:
                eps = min(a[src] * 0.5, 0.01)
                a[src] -= eps
                a[dst] += eps
                a = _project_simplex(a)
            continue

        # --------------------------------------------------------------
        # compute maximal admissible step size (feasibility + crossing)
        # --------------------------------------------------------------
        # (i) non‑negativity of the probabilities
        neg = d < 0
        if np.any(neg):
            alpha_nonneg = np.min(a[neg] / (-d[neg]))
        else:
            alpha_nonneg = np.inf

        # (ii) crossing of another entry with the maximal one
        # we need α such that for any t with s[t] > s_max:
        #   conv[t] + α·s[t]  ≤  conv[k_max] + α·s_max
        # => α ≤ (conv[k_max] - conv[t]) / (s[t] - s_max)
        mask = s > s_max + 1e-18
        if np.any(mask):
            numer = max_val - conv[mask]                # ≥ 0
            denom = s[mask] - s_max                     # > 0
            alphas = numer / denom
            alphas = alphas[alphas > 0]
            alpha_cross = alphas.min() if alphas.size else np.inf
        else:
            alpha_cross = np.inf

        # safe step (a tiny margin keeps us strictly feasible)
        alpha = min(alpha_nonneg, alpha_cross) * 0.9999

        if alpha <= 0.0 or not np.isfinite(alpha):
            # fallback to a small random move
            src, dst = np.random.choice(n, 2, replace=False)
            eps = min(a[src] * 0.2, 0.001)
            a[src] -= eps
            a[dst] += eps
            a = _project_simplex(a)
            continue

        # --------------------------------------------------------------
        # apply the move
        # --------------------------------------------------------------
        a = a + alpha * d
        a = np.maximum(a, 0.0)          # numerical safety
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a[:] = 1.0 / n

        # evaluate current score (exact, no call to evaluate_sequence)
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] iter {it:,} "
                  f"new best {best_score:.6f}")
        else:
            stagnation += 1

        # occasional random perturbation if we have been stuck for a while
        if stagnation > 5000:
            a = best_a.copy()
            # add a tiny amount of noise and re‑project
            noise = np.random.randn(n) * 0.001
            a = np.clip(a + noise, 0, None)
            a = _project_simplex(a)
            stagnation = 0

    return best_a


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise – use the previously discovered construction if present.
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined globally by the runner.
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # keep the same length if possible, otherwise up‑sample
            final_n = 4800
            if starter.size != final_n:
                a0 = _upsample(starter, final_n)
            else:
                a0 = starter.copy()
        else:
            raise ValueError
    except Exception:
        # fall back to a uniform distribution
        final_n = 4800
        a0 = np.full(final_n, 1.0 / final_n)

    # normalise to a probability simplex (sum = 1)
    a0 = np.clip(a0, 0, None)
    s = a0.sum()
    if s > 0:
        a0 /= s
    else:
        a0[:] = 1.0 / final_n

    # --------------------------------------------------------------
    # Run the optimiser
    # --------------------------------------------------------------
    best_seq = _optimise(a0, deadline, start_time)

    # final normalisation (protect against accidental drift)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list of floats
    return [float(x) for x in best_seq]
