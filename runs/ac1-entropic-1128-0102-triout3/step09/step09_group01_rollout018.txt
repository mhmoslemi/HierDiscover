def _grad_smooth(a, beta, L_fft):
    conv = _conv_self(a, L=L_fft)  # length 2n-1
    max_c = conv.max()
    # Compute w
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # correlation: linear correlation (a ⋆ w)
    # Compute using FFT
    # Pad both to L_fft
    A = np.fft.rfft(a, n=L_fft)
    W = np.fft.rfft(w, n=L_fft)
    # correlation in time domain: c = ifft(A * np.conj(W))
    corr = np.fft.irfft(A * np.conj(W), n=L_fft)[: 2 * len(a) - 1]
    # gradient (for each index 0..n-1) = 2 * correlation[n-1 : n-1 + n]
    grad = 2.0 * corr[len(a)-1 : len(a)-1 + len(a)]
    return grad

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_init
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    alpha0 = 0.1
    beta = 1.0
    beta_factor = 1.001
    beta_max = 200.0

    L_fft = _next_pow2(3*n - 2) # maybe bigger for correlation but current conv length 2n-1; correlation also same length. Using 2n-1 padded to pow2.

    while time.time() < deadline:
        it += 1
        # Compute FFT of a
        A = np.fft.rfft(a, n=L_fft)
        # conv = ifft(A * A)
        conv = np.fft.irfft(A * A, n=L_fft)[: 2*n - 1]
        cur_max = conv.max()
        # objective
        s = a.sum()
        if s < 1e-12:
            break
        cur_score = 2.0 * n * cur_max / (s*s)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")

        # Softmax weights
        max_c = cur_max
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute correlation via FFT
        W = np.fft.rfft(w, n=L_fft)
        corr = np.fft.irfft(A * np.conj(W), n=L_fft)[: 2*n - 1]
        grad = 2.0 * corr[n-1:n-1+n]  # length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection onto simplex
        a = _project_simplex(a)

        # update smoothing beta
        if beta < beta_max:
            beta = beta * beta_factor
            if beta > beta_max:
                beta = beta_max

    return best_a, best_score

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the expression is homogeneous we may normalise the vector to a
probability simplex (`∑a = 1`).  
The optimisation problem is then

* minimise the peak of the self‑convolution,  
* keep the vector on the simplex.

The programme works in three phases

1. **Fast parametric sweep** – a collection of analytic families
   (geometric, power, beta, Gaussian‑valley, …) is sampled, normalised
   and evaluated with a cheap FFT based convolution.  
   The best candidate is kept.

2. **Smoothed Adam optimisation** – the non‑smooth “max’’ is
   approximated by a soft‑max  
   \[
   \mathrm{M}_\beta(a)=\frac{1}{\beta}\log\!\sum_k e^{\beta\,(a*a)_k}
   \]
   whose gradient can be written as a correlation
   \[
   \nabla \mathrm{M}_\beta(a)=2\;(a\star w),\qquad
   w_k=\frac{e^{\beta\,(a*a)_k}}{\sum_j e^{\beta\,(a*a)_j}} .
   \]
   An Adam optimiser (per‑coordinate adaptive steps) is applied, each
   iterate being projected back onto the simplex.  The smoothing
   parameter `β` is increased slowly (continuation method) so that the
   optimiser finally behaves like a sub‑gradient method but benefits
   from the smooth early stage.

3. **Mass‑transfer simulated annealing** – the best vector from phase 2
   is polished with a tiny random move that transfers a fraction of the
   mass from one coordinate to another.  Moves are accepted with the
   Metropolis rule; the temperature is decreased exponentially.

All phases respect the 1000 s wall‑clock budget.  The final sequence is
returned as a plain Python list ready for `evaluate_sequence`.

```python
"""Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a fast parametric sweep, a smoothed Adam optimiser on the simplex,
and a short simulated‑annealing polish.  The code fits the 1000 s budget
and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution and simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact objective for a sum‑1 vector)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2 : smoothed Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_score_init: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smoothed max of a*a.
    The smoothing parameter β starts small and is increased
    (continuation) until it reaches beta_max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_init

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.04          # base learning rate

    # smoothing schedule
    beta_smooth = 1.0
    beta_factor = 1.001
    beta_max = 200.0

    L_fft = _next_pow2(2 * n - 1)

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- convolution and current score -----
        A = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(A * A, n=L_fft)[: 2 * n - 1]
        cur_max = conv.max()
        s = a.sum()
        cur_score = 2.0 * n * cur_max / (s * s)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"best={best_score:.6f}")

        # ----- soft‑max weights -----
        # w_k ∝ exp(β·conv_k)  (numerically stable)
        exp_vals = np.exp(beta_smooth * (conv - cur_max))
        w = exp_vals / exp_vals.sum()

        # ----- gradient = 2 * correlation(a, w) -----
        W = np.fft.rfft(w, n=L_fft)
        corr = np.fft.irfft(A * np.conj(W), n=L_fft)[: 2 * n - 1]
        grad = 2.0 * corr[n - 1:n - 1 + n]   # length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # ----- increase smoothing -----
        if beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * beta_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer – tiny random moves between two coordinates."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)                 # source proportional to mass
        dst = rng.integers(n)                    # arbitrary destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30      # move up to 30 % of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} "
                  f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Fast parametric sweep (various analytic families)
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int, deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Explore a few analytic families and return the best candidate."""
    idx = np.arange(n)
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)

    # ----- geometric (mixed with uniform) -----
    eps_grid = np.logspace(-4, -1, 180)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} "
                      f"lam={lam:.2f} score={sc:.6f}")

    # ----- power law -----
    p_grid = np.linspace(-1.2, 2.0, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p, dtype=np.float64)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} "
                  f"score={sc:.6f}")

    # ----- beta‑type (including negative exponents) -----
    exps = np.linspace(-2.0, 2.0, 41)        # p and q values
    for p in exps:
        for q in exps:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1.0, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} "
                      f"q={q:.2f} score={sc:.6f}")

    # ----- Gaussian valley (low centre, high edges) -----
    centre = (n - 1) / 2.0
    sigma_vals = np.linspace(0.08 * n, 0.45 * n, 12)
    for sigma in sigma_vals:
        if time.time() >= deadline - 30:
            break
        gauss = 1.0 - np.exp(- ((idx - centre) ** 2) / (2.0 * sigma ** 2))
        gauss_sum = gauss.sum()
        if gauss_sum == 0:
            continue
        gauss /= gauss_sum
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gauss σ={sigma:.1f} "
                  f"score={sc:.6f}")

    # ----- mixture of a valley with uniform (to keep positivity) -----
    for lam in np.linspace(0.0, 1.0, 9):
        seq = lam * uniform + (1.0 - lam) * best_seq
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix lam={lam:.2f} "
                  f"score={sc:.6f}")

    return best_seq, best_score
```
