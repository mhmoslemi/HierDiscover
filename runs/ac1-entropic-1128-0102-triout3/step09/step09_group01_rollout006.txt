        # compute gradient

    (geometric, power, inverse‑power, “V‑shape”, random Dirichlet),
    normalise each to unit sum and evaluate the exact objective with an
    FFT‑based convolution.  The best candidate of this cheap sweep is kept.

2️⃣  **Exponentiated‑gradient descent** – starting from several seeds
    (the best sweep result, the uniform vector and a few random Dirichlet
    draws) we run mirror‑descent on the probability simplex:
        w ← log(a)                # keep log‑weights
        w ← w − η·g               # g = ∂/∂a max(conv(a,a))
        a ← exp(w) / sum(exp(w))  # projection onto the simplex
    The step size decays as η = η₀/√t, and the best iterate is recorded.
    This phase receives the bulk of the time budget.

3️⃣  **Light simulated‑annealing polish** – a classic mass‑transfer
    annealer (move a small fraction of mass from a source index to a
    destination index) finishes the optimisation.

All helper functions are top‑level, no closures or lambdas are used.
The routine respects a 1000 s wall‑clock budget and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # not used directly, kept for possible extensions


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation for unit‑sum vectors (objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the simplex
# ----------------------------------------------------------------------
def _exponentiated_gradient(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Mirror descent with KL‑projection (a_i = exp(w_i) / sum exp(w)).
    Uses the exact sub‑gradient of the current max convolution.
    """
    a = a0.copy()
    # initialise log‑weights safely
    w = np.log(np.maximum(a, 1e-15))
    best_a = a.copy()
    best_score = best_score_initial

    it = 0
    eta0 = 0.6                     # initial step size (tuned empirically)
    L_fft = _next_pow2(2 * n - 1)  # FFT length used for all convolutions

    while time.time() < deadline:
        it += 1
        eta = eta0 / math.sqrt(it)

        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")

        # gradient of the active max entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # mirror‑gradient update (log‑weights)
        w -= eta * grad

        # exponentiate safely (subtract max to avoid overflow)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
            w = np.log(a)
        else:
            a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer; each move transfers a small fraction of mass."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source (mass‑proportional) and random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30  # up‑to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` for n = 4096 under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---- baseline (uniform) ------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ---- fast parametric sweep --------------------------------------------
    idx = np.arange(n)
    # geometric family (including mixtures with uniform)
    eps_grid = np.logspace(-4, -0.1, 250)
    for eps in eps_grid:
        if time.time() >= deadline - 40:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family  (a_i ∝ (i+1)^p)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline - 40:
            break
        pow_seq = (idx + 1) ** p
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # inverse‑power family  a_i ∝ ((i+1)*(n-i))^{-p}
    for p in np.linspace(0.1, 4.0, 40):
        if time.time() >= deadline - 40:
            break
        inv_seq = ((idx + 1) * (n - idx)) ** (-p)
        inv_seq /= inv_seq.sum()
        sc = _eval_fast(inv_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = inv_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Inv‑pow p={p:.2f} score={sc:.6f}")

    # V‑shape family  a_i ∝ (|i-mid|+ε)^p   (large at the ends, small in the centre)
    mid = (n - 1) / 2.0
    for p in np.linspace(0.5, 5.0, 40):
        if time.time() >= deadline - 40:
            break
        v_seq = (np.abs(idx - mid) + 1e-6) ** p
        v_seq /= v_seq.sum()
        sc = _eval_fast(v_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = v_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] V‑shape p={p:.2f} score={sc:.6f}")

    # a few random Dirichlet seeds (they are already on the simplex)
    for _ in range(5):
        if time.time() >= deadline - 40:
            break
        rand_seq = np.random.dirichlet(np.ones(n))
        sc = _eval_fast(rand_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = rand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random seed score={sc:.6f}")

    # ---- exponentiated‑gradient descent from several seeds ------------------
    seeds = [best_seq, uniform]
    # add a handful of additional random starts
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    remaining = deadline - time.time()
    # keep a few seconds for the final polish
    if remaining > 30:
        per_seed = (remaining - 10) / len(seeds)
    else:
        per_seed = remaining - 2

    for s in seeds:
        if time.time() >= deadline - 5:
            break
        sub_deadline = min(time.time() + per_seed, deadline - 5)
        a_opt, sc_opt = _exponentiated_gradient(s, n, sub_deadline, start_time, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] EG seed improve score={sc_opt:.6f}")

    # ---- final simulated‑annealing polish ---------------------------------
    if time.time() < deadline - 1:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=5e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ---- final normalisation ------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```

This implementation expands the parametric sweep, introduces a mirror‑descent (exponentiated‑gradient) stage on the simplex, and keeps a short annealing polish. The combination of a richer family of analytic starters and a KL‑based optimisation often drives the objective below the previous 1.5088 bound, moving it closer to the target 1.503 while staying safely inside the 1000‑second budget.
