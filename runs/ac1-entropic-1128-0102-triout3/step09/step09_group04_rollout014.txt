"""
Propose a low‑peak step‑function.

The objective 2·n·max(conv(a,a)) / (sum(a)²) depends only on the shape of
a after normalisation (the scaling cancels).  For a vector of length n the
maximum of the convolution is attained either at the centre (the overlap
of a with its reverse) or at the left‑most entry a₀².  Choosing a
monotonically decreasing power law

    a_i ∝ (n−i)ᵖ   ,   i = 0,…,n−1

gives a very small convolution peak: the centre contributes
C(p)/n with C(p) = (p+1)²·(p!)²/(2p+1)!, while the edge contributes
(p+1)²/n².  For n≈2000 the optimum exponent is p=5, where the edge term
dominates but is still O(1/n) → the overall score is about
2·(p+1)²/n ≈ 0.035, far better than the previous 1.5‑ish bound.

The routine tries a handful of exponents (1…6), evaluates the true
objective with ``evaluate_sequence`` and returns the best normalised
vector.  It runs in a few milliseconds, well within the 1000 s budget.
"""

import time
import numpy as np

def _make_power_sequence(n: int, p: int) -> np.ndarray:
    """Return the normalised descending power sequence (n‑i)ᵖ."""
    # descending integers n, n‑1, …, 1
    idx = np.arange(n, 0, -1, dtype=np.float64)
    seq = idx ** p
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return seq / total


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function that minimises the evaluation objective.
    The algorithm enumerates a few decreasing power‑law shapes, picks the
    one with the lowest score (as reported by ``evaluate_sequence``) and
    returns it as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the outer runner
    deadline = start + budget_s - 0.1

    # -----------------------------------------------------------------
    # Choose a length.  If a previous construction is available reuse its
    # length; otherwise fall back to a moderate size (2048).
    # -----------------------------------------------------------------
    n = 2048
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                n = int(tmp.size)
        except Exception:
            pass

    best_seq = None
    best_score = float("inf")
    # exponents to explore – p=5 is theoretically optimal for this n
    exponents = [1, 2, 3, 4, 5, 6]

    for p in exponents:
        if time.time() > deadline:
            break

        seq = _make_power_sequence(n, p)
        # evaluate using the provided (imported) function
        score = evaluate_sequence(seq.tolist())
        print(f"[{time.time() - start:6.2f}s] exponent {p:2d} score {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # fallback – should never happen, but guard against an empty result
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start:6.2f}s] fallback uniform score {best_score:.6f}")

    print(f"[{time.time() - start:6.2f}s] final best score {best_score:.6f}")

    # ``evaluate_sequence`` works with plain Python lists
    return best_seq.tolist()
