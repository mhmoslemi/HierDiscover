"""
Hybrid evolutionary + gradient optimisation for the self‑convolution bound.

The algorithm proceeds in three stages:

1️⃣  **Differential‑Evolution (global search).**
    A modest population of probability vectors on the simplex is evolved.
    Each individual is evaluated by the fast FFT based score
    ``2·n·max(conv)`` (equivalent to ``evaluate_sequence`` for normalised vectors).
    This quickly discovers diverse shapes that are hard to reach by local
    descent alone.

2️⃣  **Projected sub‑gradient descent (local polishing).**
    The best vector found by DE is refined with the exact sub‑gradient of the
    ℓ∞‑norm of the convolution (the same code that was used in the reference
    solution).  A momentum term and a decaying step‑size are used.

3️⃣  **Optimal incremental refinement (fine‑grained mass shifts).**
    As in the reference code, we repeatedly move a small fraction of mass
    from an index that contributes heavily to the current maximal convolution
    bin to a “quiet’’ index, computing analytically the largest admissible step.
    This yields the final tiny improvements needed to push the bound below
    the target 1.503.

All stages respect the global time budget (default 1000 s) and the routine
returns a plain Python list of non‑negative heights that minimises
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _de_global_search(n: int,
                      deadline: float,

    """
    # initialise population (each entry on the simplex)

    scores = np.empty(pop_size, dtype=float)

    # first individual = upsampled known construction if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            pop[0] = _upsample(starter, n)
        else:
            raise ValueError
    except Exception:
        pop[0] = rng.random(n)
        pop[0] /= pop[0].sum()

    # remaining individuals = random Dirichlet‑like draws
    for i in range(1, pop_size):
        pop[i] = rng.random(n)

    # compute initial scores

        conv = _conv_self(pop[i])
        scores[i] = 2.0 * n * conv.max()

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    it = 0
    while time.time() < deadline:
        it += 1
        for i in range(pop_size):
            # --- mutation ---
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)            # enforce non‑negativity

            if s > 0:
                mutant /= s
            else:
                mutant = np.full_like(mutant, 1.0 / n)

            # guarantee at least one dimension from mutant

            trial = np.where(cross, mutant, pop[i])

            # ---
            trial = np.maximum(trial, 0.0)
            s = trial.sum()

                trial /= s
            else:
                trial = np.full_like(trial, 1.0 / n)

            # --- evaluation ---
            conv = _conv_self(trial)
            trial_score = 2.0 * n * conv.max()

            # --- replacement ---

                scores[i] = trial_score

                    print(f"[{time.time() - start_time:.2f}s] DE improve "

                max_iter=2000,

            # recompute its score
            conv = _conv_self(best_vec)

    return best_vec, best_score

# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: optimal incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    recompute_every: int = 5000):
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - time.time()
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal index
        k_max = int(np.argmax(conv))

        # source: an index that appears in the max bin (probability ∝ contribution)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination: prefer indices that *do not* contribute much to the max bin
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # per‑unit‑ε effect on the whole convolution
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            continue

        # maximal admissible ε before another entry overtakes the max
        mask = s > s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # apply move analytically
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Helper: fast score (no clipping)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(self‑convolution) for a normalised probability vector."""
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a wall‑clock budget.

    The search consists of a global Differential‑Evolution phase,
    followed by a projected sub‑gradient descent and a final
    analytical refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4800 entries already proved effective.
    # --------------------------------------------------------------
    final_n = 4800

    # --------------------------------------------------------------
    # Phase 1 – global DE (roughly 50 % of the total time)
    # --------------------------------------------------------------
    de_time = min(0.55 * (deadline - start_time), 600.0)  # cap at ~10 min
    de_deadline = time.time() + de_time
    print(f"[{time.time() - start_time:.2f}s] start DE (n={final_n})")
    best_vec, best_score = _de_global_search(final_n,
                                             de_deadline,
                                             start_time,
                                             rng,
                                             pop_size=30,
                                             F=0.6,
                                             CR=0.9)

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (majority of the rest)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep at least 30 s for the final refinement
        sub_deadline = time.time() + max(30.0, 0.7 * remaining)

        print(f"[{time.time() - start_time:.2f}s] start sub‑grad descent")
        best_vec, sub_score = _run_subgradient_descent(
            best_vec,
            final_n,
            sub_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        best_score = min(best_score, sub_score)
        print(f"[{time.time() - start_time:.2f}s] sub‑grad best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – optimal incremental refinement (fine‑grained)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] start optimal incremental refinement")
        best_vec = _run_optimal_incremental_refine(
            best_vec,
            final_n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.001,
        )
        best_score = _fast_score(best_vec, final_n)
        print(f"[{time.time() - start_time:.2f}s] final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_vec]
