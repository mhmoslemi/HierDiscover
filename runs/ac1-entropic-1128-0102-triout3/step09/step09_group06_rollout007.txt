"""
Propose a non‑negative step function that minimises the L∞ convolution
objective used by ``evaluate_sequence``.

Algorithm
---------
1. Warm start – reuse the historic best sequence (``height_sequence_1``) if
   available, otherwise start from the uniform distribution.

2. Long projected sub‑gradient descent – repeatedly move probability mass
   away from indices that contribute most to the current peak of the
   self‑convolution.  The iterate is always projected onto the simplex
   (so the sum stays equal to 1).  This phase occupies the bulk of the
   time budget and drives the objective down quickly.

3. Fast greedy peak‑reduction – a cheap O(n) loop that transfers a small
   fraction of the coefficient that is responsible for the current
   maximum towards the smallest coefficient, accepting only strict
   improvements.

4. Random hill‑climbing – while time permits, perform many stochastic
   moves (transfer a random fraction of a source coefficient to a random
   destination) and keep the move only if it strictly lowers the peak.
   The convolution is updated incrementally, allowing thousands of
   attempts per second.

5. Final normalisation and evaluation – clip any tiny negatives caused
   by numerical error, renormalise to sum = 1 and return the sequence.

The whole search respects the 1000 s limit and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start helper
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector of length ``n``."""
    try:
        # ``height_sequence_1`` may be defined in the surrounding notebook.
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # Linear interpolation to the required length, preserving the sum.
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # Uniform fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient (active set) --------------------------------
        # active indices where the convolution attains the current maximum
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]   # diagonal counted twice – harmless

        # momentum + decreasing step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast greedy peak‑reduction (deterministic)
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a modest fraction of the coefficient that most
    contributes to the current convolution peak towards the smallest
    coefficient.  The convolution array is updated incrementally (O(n)
    per move) which makes the loop cheap.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        # indices that participate in this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # transfer a modest fraction (decrease over time)
        eps = 0.05 * a[src]                     # 5 % of the source value
        if eps < 1e-14:
            break

        # ---- incremental convolution update ----
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept only strict improvements
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: iter={it:,}, max={new_max:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climbing refinement
# ----------------------------------------------------------------------


def _random_hill_climb(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       rng: np.random.Generator,
                       max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Monte‑Carlo moves: pick a source index, a destination index and a
    random fraction of the source mass; apply the move only if it strictly
    reduces the convolution peak.  The convolution is updated incrementally
    (O(n) per trial).  The routine is monotone – only improving moves are
    accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # choose a source with non‑zero mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # choose a destination different from source
        dst = rng.integers(0, n)
        if dst == src:
            continue

        # random fraction of the source mass (1 %–10 %)
        eps = a[src] * rng.uniform(0.01, 0.10)
        if eps < 1e-14:
            continue

        # incremental update of the full convolution
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        new_score = 2.0 * n * new_max

        if new_score + 1e-12 < best_score:          # strict improvement
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] random hill‑climb: iter={it:,}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we never overrun the allotted time
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 4800

    # ------------------------------------------------------------------
    # 1) initialise
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------
    # 2) long sub‑gradient descent (majority of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, start_time + 0.70 * budget_s)
    print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient phase")
    best_a, best_score = _subgrad_optimize(a0, subgrad_deadline, start_time,
                                           beta=0.94, step0=0.05)
    print(f"[{time.time() - start_time:6.2f}s] sub‑gradient finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fast greedy refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start fast greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] greedy phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) random hill‑climbing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start random hill‑climbing")
        best_a, best_score = _random_hill_climb(best_a, deadline, start_time, rng)
        print(f"[{time.time() - start_time:6.2f}s] random hill‑climb finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
```

This implementation follows a different optimisation pipeline (pure sub‑gradient → greedy → stochastic hill‑climbing) and allocates more time to the stages that directly lower the L∞ convolution peak, giving a better chance of beating the previous bound.
