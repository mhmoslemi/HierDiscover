"""
Propose a step‑function that (hopefully) beats the current best bound ≈1.5088.
--------------------------------------------------------------------------

Idea
----
1. **Large canvas** – use longer sequences (up to ~16000 points).  The bound
   2·n·max(conv)/sum² tends to 1.5 as n→∞, so a larger n gives a better
   theoretical limit.

2. **Analytic starter family** – the discrete analogue of a “tent’’ shape:
   a_i ∝ min(i, n‑1‑i)^p   (p>0).  For each length we sweep a modest grid of
   exponents and keep the best (smallest objective value).

3. **Projected sub‑gradient descent** – the true objective is non‑smooth
   because of the max‑operator.  We use the exact sub‑gradient (as in the
   reference implementation) together with Euclidean projection onto the
   probability simplex.  A few hundred‑thousand iterations are cheap for
   n≈8000 and usually improve the bound by a few 10⁻⁴.

4. **Very light simulated‑annealing polish** – after the gradient phase we
   perform a cheap random‑mass‑transfer walk that can escape tiny local
   plateaus.

The code respects the 1000 s time budget, prints progress, and finally
returns a plain Python list of non‑negative numbers (the clipping in
`evaluate_sequence` is never triggered because the entries are ≪ 1000).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------

    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

        if report_every and (it % report_every == 0):
            # occasional cheap check to avoid dead‑time
            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            total_steps: int = 30000) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with exponential cooling."""
    a = seq.copy()
    n = a.shape[0]
    current_score = _objective(a)
    best_seq = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (very fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)          # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - current_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the time budget.
    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) explore several canvas sizes (powers of two are FFT‑friendly)
    # ------------------------------------------------------------------
    canvas_sizes = [2048, 4096, 8192, 16384]
    # a modest grid of exponents – the optimum is not a simple power law,
    # but sweeping gives a good starting point.
    exponents = np.linspace(0.3, 2.0, 12)

    best_seq = None
    best_score = float("inf")
    for n in canvas_sizes:
        # stop early if we run out of time
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:.2f}s] Trying canvas n={n}")

        x = np.arange(n, dtype=float) / (n - 1)                     # points in [0,1]
        for p in exponents:
            w = np.minimum(x, 1.0 - x) ** p
            if w.sum() == 0:
                continue
            a = w / w.sum()                                        # normalise to sum=1
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"   exponent={p:.3f}  ->  score={score:.6f}")

        # also try a completely random Dirichlet start (may escape symmetry)
        if time.time() < deadline:
            rnd = np.random.dirichlet(np.ones(n))
            score = _objective(rnd)
            if score < best_score:
                best_score = score
                best_seq = rnd.copy()
                print(f"   random start -> score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) local refinement with projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never happens, but keep it safe)
        n = 1024
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 10.0:
        # allocate ~80 % of the remaining budget to gradient descent
        sub_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final cheap annealing polish (optional, runs only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation (protect against accidental clipping)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # fallback uniform if something went wrong
        best_seq = np.full(best_seq.shape[0], 1.0 / best_seq.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective {best_score:.6f}")
    return [float(x) for x in best_seq]
