    best_score = _score_from_conv(_conv_self(a, L), n)
    best_a = a.copy()
    t = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: iter={t}, score={cur_score:.6f}")
            # smooth max weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:

            w_rev = w[::-1]
            # gradient
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
            # Adam moments
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = eta0 / math.sqrt(t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # project onto simplex
            a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The optimiser works in three stages:

1️⃣ **Adam on a smooth max** – we replace the hard “max” in
   2·n·max(conv(a,a)) by a log‑sum‑exp with a gradually increasing
   temperature (parameter ``alpha``).  Using Adam (first‑ and second‑
   moment adaptation) together with a projection onto the probability
   simplex yields a fast, stable descent from the starting sequence
   (the previous best or a uniform vector).

2️⃣ **Deterministic peak‑reduction** – repeatedly locate the index *k*
   where the self‑convolution is maximal.  From the pair *(i,j)*
   contributing most to this peak we move a small fraction of their
   mass to a destination that adds the least to the same peak.
   The total mass is preserved, therefore only a simplex projection
   (which is cheap) is required.  The move is accepted only if the
   true score improves, which gives a steady flattening of the
   convolution.

3️⃣ **Targeted pair‑annealing** – a short simulated‑annealing phase
   (the same routine used in the reference implementation) is run
   to escape any remaining shallow local minima.

Random perturbations and a tiny restart loop are added at the end to
exploit any remaining budget.  All helper functions are top‑level and
avoid closures or lambdas, as required by the execution environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The vector is assumed normalised (Σa=1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 0 – Adam on a smooth approximation of the maximum
# ----------------------------------------------------------------------
def _run_exp_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alphas,
    eta0: float = 0.2,
    iters_per_alpha: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(conv).
    Returns the best iterate (lowest true score) encountered.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)

    m = np.zeros_like(a)  # 1st moment
    v = np.zeros_like(a)  # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, L), n)

    t = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1

            # current score for bookkeeping
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam improve: "
                    f"iter={t:,}, score={cur_score:.6f}"
                )

            # --- smooth‑max weights (log‑sum‑exp) ---
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # --- gradient of the smooth max (2·corr) ---
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[
                : len(a) + len(w_rev) - 1
            ]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # optional clipping – prevents occasional overflow
            grad = np.clip(grad, -50.0, 50.0)

            # --- Adam update (additive) ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = eta0 / math.sqrt(t)  # simple 1/√t decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – Deterministic peak‑reduction (hill climbing)
# ----------------------------------------------------------------------
def _run_peak_reduction(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    In each step the pair (i,j) that contributes most to the current
    peak is weakened by moving a small fraction of its mass to an
    index that adds the least to the same peak.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)

    best_a = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    no_improve = 0
    max_no_improve = 4000  # give up if stagnating

    while time.time() < deadline and no_improve < max_no_improve:
        it += 1

        conv = _conv_self(a, L)
        cur_max = conv.max()
        k = int(conv.argmax())  # index of the peak

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]

        # source pair with largest product
        i_src = idx[np.argmax(contrib)]
        j_src = k - i_src
        src = [i_src] if i_src == j_src else [i_src, j_src]

        # values of a that would be added to the peak when we place mass at d
        partner = k - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        add_val = np.zeros(n)
        add_val[mask] = a[partner[mask]]

        # choose a destination that adds the *least* to the peak
        forbidden = np.zeros(n, dtype=bool)
        forbidden[src] = True
        candidates = np.where(~forbidden)[0]
        if candidates.size == 0:
            break
        d_dest = candidates[np.argmin(add_val[candidates])]

        # move a modest fraction of each source towards the destination
        frac = 0.07  # 7 % of the source mass
        a_new = a.copy()
        for s in src:
            eps = min(frac * a_new[s], a_new[s])
            a_new[s] -= eps
            a_new[d_dest] += eps

        # re‑normalise (numerical drift)
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total > 0.0:
            a_new /= total
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        # true score after the move
        conv_new = _conv_self(a_new, L)
        new_score = _score_from_conv(conv_new, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            print(
                f"[{time.time() - start_time:.2f}s] peak‑reduce improve: "
                f"iter={it:,}, score={new_score:.6f}"
            )
        else:
            no_improve += 1
            # we keep the current a (the move was rejected)
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Targeted pair‑annealing (slightly tweaked)
# ----------------------------------------------------------------------
def _run_pair_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-9,
    steps: int = 180_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly weakens the two entries that
    dominate the current maximal convolution coefficient.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20  # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a, conv, score = a_prop, conv_new, new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] anneal improve: "
                f"iter={it:,}, score={score:.6f}"
            )
    return best_seq, best_score

            starter,
        )
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    # keep the current best
    best_seq = a0.copy()
    best_score = _score_from_conv(_conv_self(a0, _next_pow2(2 * final_n - 1)), final_n)

    # ------------------------------------------------------------------
    # Phase 0 – Adam on the smooth max (≈15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase0_deadline = time.time() + max(5.0, 0.15 * remaining)

        a_adam, score_adam = _run_exp_adam(
            best_seq,
            final_n,
            phase0_deadline,
            start_time,
            alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.2,
            iters_per_alpha=12000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 0 improve: score={score_adam:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 1 – Deterministic peak reduction (≈45 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + max(5.0, 0.45 * remaining)

        a_peak, score_peak = _run_peak_reduction(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            best_score,
            best_seq,
        )
        if score_peak < best_score:
            best_score = score_peak
            best_seq = a_peak.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_peak:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – Targeted pair annealing (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_pair_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-9,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}"
            )

    # ------------------------------------------------------------------
    # Light random‑restart loop (if any time remains)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        block = min(12.0, time_left / 2.0)
        restart_deadline = time.time() + block

        pert = best_seq + 0.02 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        pert_seq, pert_score = _run_peak_reduction(
            pert, final_n, restart_deadline, start_time, best_score, pert
        )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] final score = {best_score:.6f}")

    return [float(x) for x in best_seq]
