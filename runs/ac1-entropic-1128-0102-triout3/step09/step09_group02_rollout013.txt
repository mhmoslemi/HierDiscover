        maxb = conv.max()
        return 2.0 * n * maxb

    best_a = None
    best_score = float('inf')

    # Phase 0: Random subset search
    while time.time() < deadline - 10.0:   # keep time for gradient phase
        # Choose size m adaptively based on random
        # choose m from a list to try
        m = int(np.random.choice([200,250,300,350,400,450,500]))

            best_score = s

    # If no best found, fallback uniform
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_score = score_of_a(best_a)
        print(f"fallback uniform -> score {best_score}")

    # Phase 1: Exponentiated gradient descent (to refine)
    # allocate remaining time (maybe 30% of total)
    remaining = deadline - time.time()
    if remaining > 2.0:
        # set subdeadline for gradient (maybe half of remaining)
        grad_deadline = time.time() + remaining * 0.7
        a_exp, score_exp = _run_exp_gradient(best_a, n, grad_deadline, start, eta0=0.6)

        a_greedy, score_greedy = _run_greedy_reduction(best_a, n, greedy_deadline, start,

The algorithm first draws many random subsets of the indices (size 0.5–0.75·n) and
assigns a uniform probability on the chosen indices. The subset with the
smallest max‑convolution is kept.  Afterwards a short projected‑gradient
refinement (exponentiated‑gradient on the true max) is applied.
The whole routine respects the 1000 s budget and finally returns the best
sequence as a plain list of floats.
"""

import time, math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _score_from_a(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(self‑convolution) for a (assumed sum(a)=1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) descent on the true max convolution.
    Keeps the vector on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_a(a, n, _next_pow2(2 * n - 1))
    L = _next_pow2(2 * n - 1)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # gradient only from a single index that attains the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    Returns a list of non‑negative heights that (empirically) achieve a low

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L = _next_pow2(2 * n - 1)

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – random subset sampling.
    # ------------------------------------------------------------------
    # Choose a few different target densities.
    densities = [0.40, 0.45, 0.50, 0.55, 0.60]   # fraction of n
    # Keep sampling until we run out of time needed for later phases.
    while time.time() < deadline - 5.0:
        frac = np.random.choice(densities)
        m = int(frac * n)
        # ensure at least 2 elements
        if m < 2:
            m = 2
        idx = np.random.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / m
        cur_score = _score_from_a(a, n, L)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] random subset m={m} improve: score={cur_score:.6f}")

    # Fallback – uniform distribution if nothing was found.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = _score_from_a(best_a, n, L)

    # ------------------------------------------------------------------

        if score_exp < best_score:
            best_score = score_exp
            best_a = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad final improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0.0:

    else:

        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence value: {final_val:.6f}")
    except Exception:
        pass

"""
Search for a step‑function with a low self‑convolution maximum.

The algorithm consists of two stages.

Stage 0 – Random‑subset search.
    * Choose a density ρ∈{0.40,…,0.60} and draw a random subset S⊂{0,…,n‑1}
      of size m=⌈ρ·n⌉.
    * Put a uniform probability on the chosen indices:
          a_i = 1/m  if i∈S,   a_i = 0 otherwise.
    * Using an FFT the value 2·n·max(a∗a) is computed.
    * The best subset seen so far is kept.

Stage 1 – Exponentiated‑gradient refinement.
    * Starting from the best vector found in stage 0 we run a few
      iterations of multiplicative mirror‑descent on the true
      max‑convolution (the same “exp‑grad” step used in the reference
      implementation).  The iterate is projected onto the probability
      simplex after each update.

progress messages and finally returns the best sequence as a plain Python

    The convolution is computed by FFT (linear, not circular).

    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

    """
