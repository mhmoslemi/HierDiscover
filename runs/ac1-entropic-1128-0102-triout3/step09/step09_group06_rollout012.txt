"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the historic best sequence (if available) and normalise it
   to a probability vector of length *n* (default 2000).  Falling back to a
   uniform distribution if no historic data is present.

2. Iterative improvement via a tiny linear programme:
   * With the current vector ``a`` compute M = max(a*a) (the current upper bound).
   * Solve   maximise 1ᵀb   subject to   a∗b ≤ M  and   b ≥ 0.
     The constraints are linear because (a∗b)[k] = Σ_i a_i b_{k‑i}.
   * Normalise the LP solution to unit sum (b̂ = b / Σb).  If the self‑convolution
     of b̂ is already ≤ M we replace ``a`` by ``b̂`` (big step).  Otherwise we
     mix ``a`` and ``b̂`` with a small step size *t* (convex combination) which
     is guaranteed not to increase the bound.

3. The loop runs while time permits (≤ budget_s seconds).  The best vector seen
   is stored and finally returned after a last normalisation and a single call
   to ``evaluate_sequence`` for the official score.

The core optimisation is a linear programme with at most 2 n − 1 constraints and
n variables – easily solved by SciPy’s HiGHS interior‑point solver for the
chosen problem size.  No external files are accessed; all helpers are defined
at top level and the function respects the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import math

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2·n‑1, n)) such that
        A @ b = a ∗ b   (full linear convolution).
    Each column i contains the vector a shifted by i positions.
    """
    n = a.shape[0]
    # Every column contributes a[i] on a diagonal with offset i.
    data = [np.full(n, val, dtype=np.float64) for val in a]
    offsets = np.arange(n, dtype=int)
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised starting vector (sum = 1)."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # interpolate linearly on a regular grid
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # fallback – uniform distribution
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 2000

    # ------------------------------------------------------------------
    # 1) initialise
    # ------------------------------------------------------------------
    a = _initial_candidate(n, rng)          # sum = 1
    best_a = a.copy()
    conv_a = _conv_self(a)
    M = conv_a.max()
    best_score = 2.0 * n * M
    print(f"[{time.time() - start_time:6.2f}s] start – n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based improvement loop
    # ------------------------------------------------------------------
    iteration = 0
    # start with a modest mixing factor; it will be reduced if necessary
    t_init = 0.2
    while time.time() < deadline:
        iteration += 1
        # recompute current bound
        conv_a = _conv_self(a)
        M = conv_a.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter {iteration:4d}: "
                  f"improved score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2a) solve the LP  maximise Σ b_i   s.t.   a∗b ≤ M , b ≥ 0
        # ------------------------------------------------------------------
        A = _conv_matrix(a)                     # shape (2n‑1, n)
        ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)       # minimise –Σb  → maximise Σb
        bounds = [(0.0, None)] * n

        # linear programme – use HiGHS (fast interior‑point)
        lp_res = opt.linprog(c,
                             A_ub=A,
                             b_ub=ub,
                             bounds=bounds,
                             method='highs',
                             options={'presolve': True})
        if not lp_res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed at iteration {iteration}: "
                  f"{lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print(f"[{time.time() - start_time:6.2f}s] LP returned zero mass – stop.")
            break

        b_norm = b / sum_b                     # unit‑sum candidate

        # ------------------------------------------------------------------
        # 2b) try a direct replacement if it already respects the bound
        # ------------------------------------------------------------------
        conv_b = _conv_self(b_norm)
        M_b = conv_b.max()
        if M_b < M - 1e-15:                     # strict improvement possible
            a = b_norm
            M = M_b
            print(f"[{time.time() - start_time:6.2f}s] iter {iteration:4d}: "
                  f"direct replace – new score={2.0 * n * M:.6f}")
            continue

        # ------------------------------------------------------------------
        # 2c) otherwise mix a little with the LP solution
        # ------------------------------------------------------------------
        t = t_init
        mixed = False
        while t > 1e-5:
            a_new = (1.0 - t) * a + t * b_norm
            conv_new = _conv_self(a_new)
            M_new = conv_new.max()
            if M_new < M - 1e-15:               # accept only strict improvements
                a = a_new
                M = M_new
                mixed = True
                print(f"[{time.time() - start_time:6.2f}s] iter {iteration:4d}: "
                      f"mix t={t:.4f} – new score={2.0 * n * M:.6f}")
                break
            t *= 0.5
        if not mixed:
            # no improvement – shrink the step size for future iterations
            t_init = max(t_init * 0.7, 0.02)
            print(f"[{time.time() - start_time:6.2f}s] iter {iteration:4d}: "
                  f"no improvement, reducing future t to {t_init:.3f}")

        # ------------------------------------------------------------------
        # termination guard – stop if we are barely moving any more
        # ------------------------------------------------------------------
        if time.time() - start_time > deadline - 5.0:
            # give a few seconds for the final evaluation and exit loop
            break

    # ------------------------------------------------------------------
    # 3) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
