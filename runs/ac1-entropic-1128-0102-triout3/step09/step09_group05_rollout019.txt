Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence.  The routine builds several candidate starts
(uniform, beta‑shaped, exponential, triangular), then refines the best
candidate with three phases:

1️⃣  Exponentiated‑gradient descent on a smooth log‑sum‑exp approximation
    of the max‑convolution.  The update a←a·exp(‑η∇) keeps a≥0 and the
    Euclidean simplex projection enforces Σa=1.

2️⃣  Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)).  A simple momentum scheme and a 1/√t step
    size give fast convergence.

3️⃣  Targeted pair‑annealing – the two indices that contribute most to the
    current peak of conv are weakened by moving a fraction of their
    mass to random locations.  An exponential cooling schedule and the
    Metropolis acceptance rule explore the neighbourhood.

All phases are timed; the remaining budget is spent on short random
restarts that perturb the current best solution.  The final sequence is
renormalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (full length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    # Numerical noise may produce tiny negative values – clip them.
    return np.maximum(conv, 0.0)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) (a is already normalised to sum = 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric seeds
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on a uniform grid, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    # cell centre to avoid zeros at the boundaries
    x = (i + 0.5) / n
    # work in log‑space to avoid overflow for α<1
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    a = np.exp(log_a)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _exp_sequence(beta: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(‑β·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * i)
    a /= a.sum()
    return a


def _triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangular distribution (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(i - center) / (center + 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on the smooth max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.4,
                      iters_per_alpha: int = 5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self_fft(a, L)
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # --- smooth‑max weights (log‑sum‑exp) ---
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # --- gradient of the smooth max (2·correlation) ---
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ---- exponentiated update (allow both signs) ----
            eta = eta0 / math.sqrt(it)
            # clip to avoid overflow
            grad = np.clip(grad, -10.0, 10.0)
            a = a * np.exp(-eta * grad)

            # project back onto the simplex
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 2e-2,
                        end_temp: float = 1e-8,
                        steps: int = 140_000):
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    score = 2.0 * n * conv.max()

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        contrib = a[idx_range] * a[k - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct random destinations
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction of the source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate proposed state (fast FFT version)
        conv_new = _conv_self_fft(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score
