def random_search_optimize(a_initial, conv_initial, deadline, start_time):
    a = a_initial.copy()
    conv = conv_initial.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2*n*conv.max()
    best_conv_max = conv.max()
    # parameters
    T0 = best_score * 0.01  # initial temperature maybe 1% of score
    decay_rate = 0.9995  # per iteration
    t = 0
    while time.time() < deadline:
        t += 1
        # random source and dest
        i = rng.integers(0, n)
        # ensure a[i] > 0 else can't move
        if a[i] <= 0:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1  # ensure j != i
        # choose epsilon
        # choose fraction between 0 and min(0.5, maybe)
        max_move = min(a[i], 0.2)  # limit to 0.2 mass maybe too high relative to sum=1
        if max_move <= 0:
            continue
        eps = max_move * rng.random()
        # compute new conv
        conv_new = conv.copy()
        # update slices
        conv_new[i:i+n] -= 2.0*eps*a
        conv_new[j:j+n] += 2.0*eps*a
        # self terms
        conv_new[2*i] += eps*eps
        conv_new[2*j] += eps*eps
        conv_new[i+j] -= 2.0*eps*eps
        # compute new max
        new_max = conv_new.max()
        new_score = 2.0*n*new_max
        # compute temperature
        T = max(T0 * (decay_rate**t), 1e-12)
        # accept if improvement or with probability
        if new_score < best_score or rng.random() < math.exp(-(new_score - best_score) / T):
            # Accept move update
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            # update best if better
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # print progress
                if t % 1000 == 0:
                    print(f"[{time.time() - start_time:6.2f}s] Random search improve: iter={t:,}, score={best_score:.6f}")
        # else reject, continue
    return best_a, best_score

"""
Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Diverse starts** – uniform, exponential, Gaussian, Dirichlet and the
   previously best known sequence (if present).  Keep the best raw start.
2. **Exponentiated‑gradient polish** – mirror‑descent (multiplicative
   updates) on the true max‑convolution objective for the bulk of the
   time budget.
3. **Monte‑Carlo local search** – repeatedly move a small amount of mass
   from one index to another and accept the move if it improves the
   objective or with a temperature‑dependent probability (simulated
   annealing).  Convolution updates are performed incrementally, so each
   trial costs O(n) instead of an FFT.
4. **Greedy peak reduction** – a cheap deterministic sweep that moves a
   fraction of the coefficient that contributes most to the current peak
   towards the smallest coefficient.
5. **Final normalisation** – clip to ``>=0``, renormalise to sum 1 and
   report the final score.

The method stays within the 1000 s budget, prints progress, and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)-1)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _exponential_decay(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    v = np.array([r ** i for i in range(n)], dtype=float)
    v /= v.sum()
    return v


def _gaussian_bell(n: int) -> np.ndarray:
    """Discrete Gaussian, σ≈n/6, normalised."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    g /= g.sum()
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (mirror descent)
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on f(a)=max_k (a*a)[k] (no smoothing).
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    cur_a = a.copy()
    cur_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        # exact convolution
        conv = _conv_self_fft(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient of the active shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        g = np.zeros_like(cur_a)
        g[i_idx] = 2.0 * cur_a[j_idx]

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # exponentiated update and re‑normalise
        cur_a = cur_a * np.exp(-eta * g)
        s = cur_a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a = cur_a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Monte‑Carlo local search with incremental convolution updates
# ----------------------------------------------------------------------


def _random_local_search(a_start: np.ndarray,
                         conv_start: np.ndarray,
                         deadline: float,
                         start_time: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random moves:
    move a small amount of mass from a source index to a destination
    index and accept (or sometimes uphill‑accept) the move.
    Convolution is updated incrementally in O(n) per trial.
    """
    a = a_start.copy()
    conv = conv_start.copy()
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (linear decay)
    total_time = deadline - start_time
    max_temp = cur_score * 0.015  # ~1.5 % of current score

    # how often we recompute the full convolution to avoid drift
    recompute_interval = max(5000, n)  # every few thousand moves

    moves = 0
    while time.time() < deadline:
        moves += 1

        # choose source with some mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # choose distinct destination
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # step size (larger early, smaller later)
        remaining = deadline - time.time()
        frac = remaining / total_time
        max_move = min(a[i], 0.12 * frac)  # at most 12 % of source mass
        if max_move <= 0.0:
            continue
        eps = max_move * rng.random()  # uniform in (0, max_move)

        # ------------------------------------------------------------------
        # Incremental update of the convolution (copy, modify, evaluate)
        # ------------------------------------------------------------------
        conv_new = conv.copy()
        # shift_i: a placed starting at index i
        conv_new[i:i + n] -= 2.0 * eps * a
        # shift_j: a placed starting at index j
        conv_new[j:j + n] += 2.0 * eps * a
        # self‑interaction terms from the moved mass
        conv_new[2 * i] += eps * eps
        conv_new[2 * j] += eps * eps
        conv_new[i + j] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # temperature for this iteration
        temp = max_temp * frac

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if temp > 0 and rng.random() < math.exp(-(new_score - cur_score) / temp):
                accept = True

        if accept:
            # commit the move
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            cur_score = new_score
            cur_max = new_max

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if moves % 1000 == 0:
                    print(f"[{time.time() - start_time:6.2f}s] Random search improve: iter={moves:,}, score={best_score:.6f}")

        # occasional full recomputation to keep numerical error small
        if moves % recompute_interval == 0:
            conv = _conv_self_fft(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic improvement (same as previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self_fft(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self_fft(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score
