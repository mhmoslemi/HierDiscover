"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.
The search uses several complementary heuristics:

1.  A few hand‑crafted “shape” seeds (uniform, arcsine‑like,
    power‑law tails).  Each seed is quickly refined by a short
    exponentiated‑gradient run and a cheap greedy peak‑reduction.
2.  The best seed is then optimised more thoroughly:
    * Adam optimisation on a smooth (log‑sum‑exp) surrogate of the
      maximum self‑convolution (many iterations, decreasing step‑size).
    * Exponentiated‑gradient descent on the true max‑convolution.
    * Targeted simulated‑annealing with a long cooling schedule.
    * Deterministic greedy reduction that moves mass from the
      dominant contributor of the current peak to a low‑impact index.
3.  Finally a series of random‑restart cycles (perturb → short
    gradient → short anneal) uses any remaining time.

All updates keep the vector on the probability simplex, so the
objective reduces to ``2·n·max(conv)`` (because ``sum(a)=1``).
The routine respects the 1000 s budget, prints progress and returns
the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution of ``a`` using FFT of length ``L``."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth (log‑sum‑exp) surrogate of the max
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index (sub‑gradient)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (exactly sum‑to‑1)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current max convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and dst in the current max
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination with the smallest partner value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Handy shape generators (used for diverse initialisations)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + 1e-12)
    w = np.maximum(w, 0.0)
    return w / w.sum()

def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    w = ((i + 0.5) * (n - i - 0.5) + 1e-12) ** (-p)
    w = np.maximum(w, 0.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns the best sequence found as a Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size. 4096 works well experimentally; larger sizes
    # do not change the objective much because the score is scale‑invariant.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Build a small pool of diverse seeds.
    # If a historic construction is available we also include it.
    # ------------------------------------------------------------------
    seeds = []

    try:
        # ``height_sequence_1`` is provided by the surrounding environment.
        hist = np.asarray(height_sequence_1, dtype=float)
        if hist.ndim == 1 and hist.size > 0:
            hist = np.maximum(hist, 0.0)
            hist = hist / hist.sum()
            # interpolate to the target length
            hist_interp = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, hist.size),
                hist
            )
            hist_interp = np.maximum(hist_interp, 0.0)
            hist_interp /= hist_interp.sum()
            seeds.append(hist_interp)
    except Exception:
        pass

    # hand‑crafted shapes
    seeds.append(_shape_uniform(n))
    seeds.append(_shape_arcsine(n))
    for p in [0.5, 1.0, 1.5, 2.0]:
        seeds.append(_shape_powerlaw(n, p))

    # a couple of random starts
    seeds.append(np.random.rand(n))
    seeds.append(np.random.rand(n) * np.linspace(1.0, 2.0, n))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Quick refinement of each seed (short gradient + greedy pass)
    # ------------------------------------------------------------------
    for si, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        # short exponentiated‑gradient phase (few seconds)
        exp_deadline = time.time() + min(5.0, (deadline - time.time()) * 0.05)
        a, _ = _run_exp_gradient(a0, n, exp_deadline, start_time, eta0=0.6)

        # short greedy reduction (a few thousand iterations)
        gre_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.10)
        a, _ = _run_greedy_reduction(a, n, gre_deadline, start_time,
                                     cur_best_score=best_score if best_seq is not None else float("inf"),
                                     cur_best_seq=a)

        # evaluate via the external function (more reliable than the internal proxy)
        seq = [float(x) for x in a]
        obj = evaluate_sequence(seq)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] seed {si} improve: score={obj:.6f}")

    # ------------------------------------------------------------------
    # Deep optimisation from the best seed found so far
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _shape_uniform(n)

    # ---- Phase 1 : Adam on smooth max ---------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + max(20.0, 0.15 * (deadline - time.time()))
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            n,
            phase1_deadline,
            start_time,
            alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
            lr0=0.8,
            iters_per_alpha=50000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] deep adam improve: score={score_adam:.6f}")

    # ---- Phase 2 : Exponentiated gradient -----------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            eta0=0.5,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] deep exp‑grad improve: score={score_exp:.6f}")

    # ---- Phase 3 : Targeted annealing ---------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
        a_ann, score_ann = _run_targeted_annealing(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=400_000,
        )
        if score_ann < best_score:
            best_score = score_ann
            best_seq = a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] deep anneal improve: score={score_ann:.6f}")

    # ---- Phase 4 : Deterministic greedy reduction ---------------------
    if time.time() < deadline:
        phase4_deadline = deadline
        a_gre, score_gre = _run_greedy_reduction(
            best_seq,
            n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=2_000_000,
        )
        if score_gre < best_score:
            best_score = score_gre
            best_seq = a_gre.copy()
            print(f"[{time.time() - start_time:.2f}s] deep greedy improve: score={score_gre:.6f}")

    # ------------------------------------------------------------------
    # Random‑restart polishing while there is time left
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        block = min(20.0, (deadline - time.time()) / 2.0)
        restart_deadline = time.time() + block

        # perturb current best moderately
        pert = best_seq + 0.05 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short exp‑gradient run
        pert_seq, _ = _run_exp_gradient(
            pert,
            n,
            restart_deadline,
            start_time,
            eta0=0.5,
        )
        # brief annealing if time permits
        if time.time() < deadline:
            pert_seq, _ = _run_targeted_annealing(
                pert_seq,
                n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=80_000,
            )
        seq = [float(x) for x in pert_seq]
        obj = evaluate_sequence(seq)
        if obj < best_score:
            best_score = obj
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={obj:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
