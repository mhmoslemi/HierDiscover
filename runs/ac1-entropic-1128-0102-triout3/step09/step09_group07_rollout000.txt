"""
propose_candidate
-----------------

Find a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence`` under a 1000 s wall‑clock budget.

Approach
~~~~~~~~
1. **Large support** – we use a longer sequence (≈ 9600 entries).  The
   evaluation ``2·n·max(conv)`` scales roughly like 1/n, so a larger
   n gives a lower theoretical bound and gives the optimizer more
   freedom.

2. **Three optimisation phases**
   * **Phase 1 – Exponentiated‑gradient warm‑up** on a smoothed
     approximation of the max (the same scheme as the reference code,
     but with a shorter schedule).
   * **Phase 2 – Projected sub‑gradient descent** on the exact ℓ∞‑norm
     of the self‑convolution.  A simple Adam‑style momentum is used
     and the loop runs until the allotted time is exhausted.
   * **Phase 3 – Greedy mass‑shifts** (optimal incremental refinement).
     In each step we move a small amount of mass from a “source’’ index
     that contributes heavily to the current maximal convolution bin to
     a “destination’’ index that contributes little.  The maximal admissible
     step size ε is computed analytically, giving large guaranteed
     improvements per iteration.

3. **Random restarts** – after the first full pass we perform a few
   cheap random perturbations of the best sequence and re‑run phases 2
   and 3.  This helps to escape possible local minima.

All phases respect the global deadline (``budget_s‑0.5`` seconds) and
the best sequence found so far is always kept.  The final sequence is
normalised to sum to 1 (scale‑invariant for the original objective) and
returned as a plain Python list of floats.

# Basic convolution / linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) self‑convolution a*a using an FFT of length
    a power of two.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` w.r.t. ``a`` using all entries whose
    value lies within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    best_score = float("inf")

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (softmax over the convolution)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)  # numerical stability
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected (Adam‑style) sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8):
    """
    Adam‑style projected descent on the exact ℓ∞‑norm of the self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={t:,}, score={cur_score:.6f}")

        # exact sub‑gradient of the max
        g = _subgradient_full(a, conv)

        # Adam update rules
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / math.sqrt(t)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (sum to 1, non‑negative)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Greedy optimal mass‑shift refinement
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.06,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    recompute_every: int = 8000):
    """
    Perform many cheap mass‑shifts. For a source index ``src`` and a
    destination ``dst`` the effect on the whole convolution is linear in ε.
    We compute the largest ε that still reduces the current maximal entry
    (solution of a 1‑D linear programme) and apply the move.
    """
    a = a0.copy()
    conv = _conv_self(a)                       # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # index of the current maximal convolution bin
        k_max = int(np.argmax(conv))

        # ----- source: pick an index that contributes to the max bin -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fallback – choose proportionally to a
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # ----- destination: prefer indices that *do not* appear in the max bin -----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- linear effect of moving ε from src to dst -----
        # effect on convolution entry t:  s[t] = -2·a[t‑src] + 2·a[t‑dst]
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # moving mass would not lower the current max – try another pair
            continue

        # maximal ε before some other entry overtakes the present max
        mask = s > s_max                                 # steeper slopes
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]        # ≥0
            delta_s = s[mask] - s_max                    # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # apply move analytically (O(n))
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - rel_improve:
            # genuine improvement
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid drift
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Helper – small random perturbation of a probability vector
# ----------------------------------------------------------------------
def _perturb(a: np.ndarray, magnitude: float = 0.02) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise of given *magnitude* (relative to the
    l2‑norm of the vector) and re‑project onto the simplex.
    """
    rng = np.random.default_rng()
    noise = rng.normal(scale=magnitude, size=a.shape)
    b = a + noise
    b = np.maximum(b, 0.0)
    if b.sum() > 0:
        b /= b.sum()
    else:
        b[:] = 1.0 / a.shape[0]
    return b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.

    The routine follows the three‑phase optimisation pipeline described
    above, works with a relatively large support (≈9600 entries) and
    performs a few cheap random restarts to improve robustness.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a safety margin so that we always return before the watchdog
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 0. Initialisation – upsample an existing good construction if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Choose a relatively large support; powers of two are convenient for FFT.
    final_n = 9600
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – very short exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    stage1_time = min(8.0, 0.06 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 12.0, 30.0],
        eta0=0.5,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam‑style sub‑gradient descent (major part)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly 70 % of the remaining time to this phase
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(30.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.30,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy optimal incremental refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_optimal_incremental_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.06,
            eps_end=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Optional cheap random restarts (if we still have some time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep ~5 s buffer for finalisation
        # perturb current best a bit and re‑run a short refinement
        a_pert = _perturb(best_seq, magnitude=0.01)
        sub_deadline = time.time() + 5.0   # 5 seconds per mini‑run
        a_tmp, _ = _run_subgradient_descent(
            a_pert,
            final_n,
            sub_deadline,
            start_time,
            step0=0.20,
        )
        a_tmp = _run_optimal_incremental_refine(
            a_tmp,
            final_n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.001,
        )
        conv_tmp = _conv_self(a_tmp)
        score_tmp = 2.0 * final_n * conv_tmp.max()
        if score_tmp < best_score:
            best_score = score_tmp
            best_seq = a_tmp.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={score_tmp:.6f}")

    # ------------------------------------------------------------------
    # Normalise for safety and return as a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Convert to Python floats
    return [float(x) for x in best_seq]
