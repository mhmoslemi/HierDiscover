def _simulated_annealing(a0, deadline, start_time, cur_best_score):
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = cur_best_score
    # set parameters
    max_moves = 500000
    move_factor = 0.1   # fraction of source mass to move
    T0 = 0.001   # initial temperature on scale of max (peak)
    while time.time() < deadline:
        # determine time fraction to gradually reduce T
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T0 * (1 - elapsed/total)
        # Ensure T > some small epsilon
        if T < 1e-6:
            T = 1e-6

        # Determine peak index
        k_max = conv.argmax()
        # compute indices i that contribute
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        # source index is one with largest contribution; break ties randomly
        src = i_idx[contrib.argmax()]

        # destination: choose index with smallest a (avoid src)
        if a[src] <= 1e-16:
            # source has essentially zero mass, pick random other index
            dst = (src + 1) % n
        else:
            # pick among low values
            # gather indices sorted by a, choose one of the lowest 5% randomly
            low_count = max(1, n // 20)
            low_candidates = np.argpartition(a, low_count)[:low_count]
            low_candidates = low_candidates[low_candidates != src]
            if low_candidates.size == 0:
                dst = (src + 1) % n
            else:
                dst = rng.choice(low_candidates)

        eps = move_factor * a[src]
        if eps < 1e-12:
            # reduce move_factor
            move_factor *= 0.5
            continue
        # compute new conv and a via incremental update
        conv_new = conv.copy()
        # linear shift
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        # quadratic
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        delta = new_max - conv.max()

        # acceptance: if improvement accept; else accept with probability exp(-delta/T)
        accept = False
        if delta < -1e-15:
            accept = True
        else:
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = 2.0 * n * new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: score={cur_score:.6f}")

        # optionally adapt move_factor
        # e.g., decay over time
        move_factor = max(1e-4, move_factor * 0.99999)
    return best_a, best_score

def _adam_optimize_schedule(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3*n - 2)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]
    beta1,beta2 = 0.9,0.999
    eps_adam = 1e-8
    lr0 = 0.10
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    for α in alpha_schedule:
        # allocate time for α: compute time fraction maybe (deadline - start_time) * (1/len(alpha_schedule))
        # but we also need to enforce not exceed deadline.
        # We'll run while time left > some minimal and for limited steps.
        steps = 4000  # can adjust.
        for _ in range(steps):
            if time.time() >= deadline:
                break
            conv = _conv_self(a)
            # compute smooth max and gradient
            scaled = α * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # gradient via cross-correlation
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
            # scale gradient by outer factor 2*n (same as evaluate)
            grad_a *= 2.0 * n

            t += 1
            m = beta1 * m + (1 - beta1) * grad_a
            v = beta2 * v + (1 - beta2) * (grad_a ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # evaluate occasionally
            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam (α={α}) improve score={sc:.6f}")

        # after each α, optionally re-evaluate best a

            print(f"[{time.time() - start_time:6.2f}s] Adam after α={α} best score={sc:.6f}")

        if time.time() >= deadline:
            break

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm consists of four phases:

1. **Warm‑start** – use the historic best sequence (the global
   ``height_sequence_1`` variable) and interpolate it to the working
   size.  If this is unavailable we fall back to a uniform distribution.

2. **Projected Adam on a smooth approximation of the max‑convolution**.
   The smoothing parameter α is increased in stages; for each α a
   fixed number of Adam steps is performed.  The iterate is projected
   onto the probability simplex after each step.

3. **Simulated‑annealing / stochastic local search** – repeatedly move a
   small fraction of mass from the index that contributes most to the
   current convolution peak to a low‑mass index.  Moves that do not
   improve the peak are accepted with a temperature‑dependent probability,
   allowing the search to escape shallow local minima.

4. **Fast greedy refinement** – an O(n) incremental update of the
   convolution array performs many cheap moves that strictly lower the
   peak.

All phases respect the 1000 s time budget.  The best sequence found is
re‑normalised to sum = 1 before the final evaluation is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on a smoothed max
# ----------------------------------------------------------------------


def _adam_optimize_schedule(a0: np.ndarray,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the smoothed maximum
    f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k]).
    The smoothing parameter α follows a schedule; for each α a fixed
    number of Adam steps is taken.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α approximates the true max more closely
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for α in alpha_schedule:
        # a modest but fixed number of Adam steps for each α
        steps = 4000
        for _ in range(steps):
            if time.time() >= deadline:
                break

            # convolution of the current iterate
            conv = _conv_self(a)

            # ----- smooth max value (only needed for monitoring) -----
            scaled = α * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / α                     # smooth max
            # gradient of the smooth max w.r.t. a
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]   # ∂/∂a (smooth max)

            # outer factor 2·n coming from the true objective
            grad_a = grad * (2.0 * n)

            # Adam update on the raw vector
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam (α={α}) improve "
                        f"score={sc:.6f}"
                    )

        # one final check after each α
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam after α={α} best "
                f"score={sc:.6f}"
            )
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Simulated‑annealing style stochastic local search
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that moves a small fraction of the mass from
    an index that contributes most to the current peak towards a low‑mass
    index.  Moves that worsen the peak are accepted with a temperature‑
    dependent probability, enabling occasional uphill steps.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)

    best_a = a.copy()
    best_score = cur_best_score

    # initial temperature – a small fraction of the current peak value
    init_T = 0.001 * conv.max()
    move_factor = 0.10                     # fraction of source mass to move

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling
        T = max(1e-8, init_T * (1.0 - elapsed / total))

        # -------- pick source (largest contribution to the current peak) -------
        k_max = conv.argmax()
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # -------- pick destination (a low‑mass index) ---------------------------
        # choose among the 5 % smallest coefficients (excluding src)
        low_cnt = max(1, n // 20)
        low_candidates = np.argpartition(a, low_cnt)[:low_cnt]
        low_candidates = low_candidates[low_candidates != src]
        if low_candidates.size == 0:          # fallback: any other index
            dst = (src + 1) % n
        else:
            dst = rng.choice(low_candidates)

        eps = move_factor * a[src]
        if eps < 1e-14:
            # nothing left to move from src – shrink the step size
            move_factor *= 0.5
            continue

        # ------- incremental convolution update (O(n) vectorised) --------------
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        delta = new_max - conv.max()

        accept = False
        if delta < -1e-15:                    # strict improvement
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = 2.0 * n * new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] SA improve: "
                    f"score={cur_score:.6f}"
                )

        # slowly damp the move size so that the algorithm becomes more
        # fine‑grained towards the end of the run
        move_factor = max(1e-4, move_factor * 0.99999)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – short sub‑gradient polishing (optional)
# ----------------------------------------------------------------------


def _subgrad_refine(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    A few thousand iterations of classic projected sub‑gradient descent
    on the true objective max_k (a*a)[k].  This cleans up remaining tiny
    irregularities after the stochastic phase.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] sub‑grad refine iter={it:,} "
                f"score={cur_score:.6f}"
            )
        # active set of (almost) maximal indices
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # diagonal contributions appear twice – harmless

        step = 0.02 / math.sqrt(it)   # decreasing step size
        a = _project_simplex(a - step * g)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – fast greedy peak reduction (identical to the reference code)
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a modest fraction of the coefficient that most
    contributes to the current convolution peak toward the smallest
    coefficient, accepting only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.05 * a[src]                     # 5 % of the source value
        if eps < 1e-14:
            break

        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept only strict improvements
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: "
                f"iter={it:,}, max={new_max:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: build an initial candidate (uses the historic best if possible)
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector of length n (sum = 1)."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation preserving the total mass
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main driver – the function that will be called by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny safety margin to avoid overshooting the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – a few thousand entries keep FFT cheap and give us enough
    # degrees of freedom to lower the peak.
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # 1) Warm start
    # ------------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smoothed max (≈ 55 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = min(deadline, start_time + 0.55 * budget_s)
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    best_a, best_score = _adam_optimize_schedule(a0, adam_deadline, start_time)
    print(
        f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing stochastic refinement (≈ 25 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = min(deadline, start_time + 0.80 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start stochastic annealing")
        best_a, best_score = _simulated_annealing(
            best_a, sa_deadline, start_time, rng, best_score
        )
        print(
            f"[{time.time() - start_time:6.2f}s] annealing phase finished, best score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4) Short sub‑gradient polishing (≈ 10 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = min(deadline, start_time + 0.90 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, best_score = _subgrad_refine(best_a, subgrad_deadline, start_time)
        print(
            f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished, best score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 5) Fast greedy refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start fast greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline, start_time)
        print(
            f"[{time.time() - start_time:6.2f}s] greedy phase finished, best score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 6) Final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}"
    )
    return [float(x) for x in best_a.tolist()]
