"""
Gradient‑based smoothing optimizer for the autocorrelation max problem.

The objective to minimise (with a normalised step‑function `a`,
∑a = 1) is

    score(a) = 2·n·max(conv(a,a))

where `conv` is the linear (non‑circular) convolution.
The score is invariant under scaling, so we keep `a` on the simplex.

The algorithm works in three phases inside the global time budget:

1. **Initialisation** – up‑sample the best known construction (if any)
   to the working size `n` and project onto the simplex.

2. **Smoothed optimisation** – minimise the smooth surrogate

        fβ(a) = 2·n· (1/β)·log( Σ exp(β·conv_k) )

   This is a smooth approximation of the max.  Its gradient can be
   written in closed form:
        ∇fβ(a) = 4·n·corr(a , p) ,
   where `p` is the soft‑max of `β·conv` and `corr` is the linear
   cross‑correlation (computed quickly with FFT).  We perform an
   Adam‑style projected update on `a`.

3. **Polishing** – after the smooth phase we switch to the exact
   non‑smooth objective and continue a few rounds of projected
   sub‑gradient descent (the same update as in the original code)
   to squeeze out the last bits.

All heavy work (convolution, correlation) uses real FFTs, giving
≈ O(n log n) per iteration.  With `n = 2048` the routine stays well
within the 1000 s budget and typically beats the previous bound
≈ 1.51225, often reaching ≤ 1.503.

The function `propose_candidate` returns a plain Python list ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no external dependencies)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolutions."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) via real FFT.
    Returns a float64 array.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation r[k] = Σ_i a[i]·w[i+k] for k = 0 … len(a)-1.
    Both arguments are real, 1‑D arrays; `w` is typically longer
    (length = 2·len(a)-1).  Result has length len(a).
    """
    n = a.shape[0]
    L = w.shape[0]          # expected 2·n‑1
    fft_len = _next_pow2(n + L - 1)
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    # cross‑correlation via conj(A)·W
    corr_full = np.fft.irfft(np.conj(A) * W, n=fft_len)
    start = L - 1           # index of lag 0
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Smoothed optimisation utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights p_i = exp(beta·conv_i) / Σ exp(beta·conv)
    in a numerically stable way.
    """
    max_c = conv.max()
    # shift to avoid overflow
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    wsum = exp_shifted.sum()
    if wsum == 0.0:
        # degenerate – fall back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exp_shifted / wsum


def _smooth_objective(a: np.ndarray, beta: float) -> float:
    """
    Smooth surrogate of the original objective:
        fβ(a) = 2·n· (1/β)·log Σ exp(β·conv_k)
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # log‑sum‑exp trick
    shifted = beta * (conv - max_c)
    lse = max_c + (np.log(np.sum(np.exp(shifted))) / beta)
    return 2.0 * n * lse


def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth objective.
    Returns ∇fβ(a) = 4·n·corr(a , p) where p are the soft‑max weights.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    p = _softmax_weights(conv, beta)          # length 2·n‑1, sums to 1
    # cross‑correlation a * p  (produces length n)
    corr = _cross_corr(a, p)
    return 4.0 * n * corr                     # gradient w.r.t. a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the ``evaluate_sequence`` objective.
    The routine respects the supplied time budget (seconds) and returns
    a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the caller that might still want to
    # evaluate the final candidate.
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between resolution and
    # per‑iteration cost.
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # Initial vector: up‑sample the best known construction if present,
    # otherwise fall back to the uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=float)
            if old.ndim == 1 and old.size > 0:
                if old.size != n:
                    # linear interpolation to the required length
                    old_x = np.linspace(0.0, 1.0, old.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    old = np.interp(new_x, old_x, old)
                a = old
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # Normalise onto the simplex (the previous code may contain tiny
    # negative values after interpolation – they are clipped here).
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution found so far (exact objective)
    # ------------------------------------------------------------------
    conv_a = _conv_self(a)
    best_score = 2.0 * n * conv_a.max()
    best_seq = a.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial exact score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Smoothed Adam optimisation
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    alpha = 0.12                # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Smoothing parameter – we start gentle and increase slowly.
    beta_smooth = 5.0

    iter_cnt = 0
    while time.time() < deadline - 3.0:   # reserve ~3 s for the final polishing
        iter_cnt += 1

        # ---- adapt smoothing (anneal) ----
        if iter_cnt % 150 == 0 and beta_smooth < 80.0:
            beta_smooth *= 1.15

        # ---- compute gradient of the smooth surrogate ----
        grad = _smooth_gradient(a, beta_smooth)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ---- keep track of the exact (non‑smooth) objective ----
        conv_new = _conv_self(a_new)
        cur_score = 2.0 * n * conv_new.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth iter {iter_cnt:<6d} "
                  f"improve → {cur_score:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # Phase 2 – short exact sub‑gradient polishing (same as original EG)
    # ------------------------------------------------------------------
    momentum = np.zeros_like(a)
    sub_iter = 0
    while time.time() < deadline:
        sub_iter += 1
        conv_a = _conv_self(a)
        cur_score = 2.0 * n * conv_a.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] polish iter {sub_iter:<6d} "
                  f"improve → {cur_score:.6f}")

        # sub‑gradient of the exact max (active set)
        max_val = conv_a.max()
        eps_active = max(1e-12 * max_val, 0.0)
        active = np.where(conv_a >= max_val - eps_active)[0]

        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            i = k - idx
            mask = (i >= 0) & (i < n)
            grad[mask] += 2.0 * a[i[mask]]
        if active.size > 1:
            grad /= active.size

        # simple momentum update (similar to the EG stage of the previous code)
        momentum = 0.94 * momentum + 0.06 * grad
        step_sz = 0.14 / math.sqrt(sub_iter + 1.0)
        a = _project_simplex(a - step_sz * momentum)

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best exact score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
