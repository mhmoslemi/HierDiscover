"""
A three‑stage “gradient‑plus‑refinement’’ optimizer for the
max‑self‑convolution problem.

Stage 1 – exponentiated‑gradient on a smoothed max (log‑sum‑exp)
          to quickly obtain a good shape.

Stage 2 – Adam‑type adaptive gradient descent on the same smoothed
          objective with a very large smoothing parameter, driving the
          solution close to the true ℓ∞ optimum.

Stage 3 – deterministic incremental mass‑shifts.  For a chosen
          source index (which contributes heavily to the current
          maximal convolution bin) and a destination index (which
          contributes little) the effect on every convolution entry
          is linear in the transferred mass ε.  The algorithm solves
          the 1‑D linear programme that yields the largest admissible
          ε that still strictly lowers the current maximum, then
          applies the move analytically.  The ε fraction is allowed
          to become arbitrarily small (≈10⁻⁶ of the source mass) so
          that the refinement continues until the time budget is
          exhausted.

All steps keep the vector on the probability simplex, guaranteeing
valid inputs for ``evaluate_sequence``.  The routine returns the best
sequence found before the allotted wall‑clock time expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative entries that sum to one).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------


def _run_exponential_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_schedule,
    eta0: float = 0.4,
    steps_per_alpha: int = 8000,
):
    """
    Multiplicative (exponentiated) gradient descent on the
    log‑sum‑exp approximation of the maximum convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)  # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)  # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[
            : n + len(w_rev) - 1
        ]  # length 3n‑2
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _run_adam_softmax(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha: float = 200.0,
    lr0: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adaptive‑momentum (Adam) minimisation of the log‑sum‑exp
    approximation of max(conv).  The iterate is always projected onto
    the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] adam‑softmax improve: "
                f"iter={t:,}, score={cur_score:.6f}"
            )

        # softmax weights for the current smoothing parameter
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of L = (1/alpha) logsumexp(alpha*conv)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[
            : n + len(w_rev) - 1
        ]  # length 3n‑2
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # bias‑corrected learning rate
        lr = lr0 * (math.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – optimal incremental refinement (large but exact moves)
# ----------------------------------------------------------------------


def _run_optimal_incremental_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    eps_start: float = 0.04,
    eps_end: float = 1e-6,
    rel_improve: float = 1e-14,
    recompute_every: int = 2000,
):
    """
    Repeatedly transfer a fraction of the mass from a “source’’ index
    (which contributes heavily to the current maximal convolution bin)
    to a “destination’’ index (which contributes little).  The transferred
    amount ε is chosen as the largest value that still lowers the present
    maximum – the solution of a simple 1‑D linear programme – guaranteeing
    maximal progress per iteration while keeping the per‑iteration cost O(n).
    """
    a = a0.copy()
    conv = _conv_self(a)  # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    start_refine = time.time()
    total_time = deadline - start_refine
    accepted = 0
    attempts = 0

    while time.time() < deadline:
        elapsed = time.time() - start_refine
        # Linear decay of the fraction of the source mass we are allowed to move
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # ----- pick a source index (one of the contributors to that entry) -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            attempts += 1
            continue

        # ----- pick a destination index (prefer indices that appear little in the max bin) -----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0  # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- linear effect of moving ε from src to dst on the whole convolution -----
        # s[t] = 2·(a[t‑dst] – a[t‑src])   (out‑of‑range treated as 0)
        s = np.zeros(2 * n - 1, dtype=float)
        s[src : src + n] -= 2.0 * a
        s[dst : dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            attempts += 1
            continue

        # ----- maximal admissible ε before another entry catches up -----
        mask = s > s_max  # strictly larger slope would raise another entry above the current max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]  # ≥0
            delta_s = s[mask] - s_max             # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from src
        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            attempts += 1
            continue

        # ----- apply the move analytically -----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        # genuine improvement (guard against numerical noise)
        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] opt‑refine improve: "
                    f"iter={accepted:,}, max={best_max:.9f}"
                )

        # occasional full recomputation to keep floating‑point drift under control
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

        attempts += 1

    return best_a


# ----------------------------------------------------------------------
# Optional Stage 2.5 – short projected sub‑gradient pass (fine‑tuning)
# ----------------------------------------------------------------------


def _run_subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.12,
    beta: float = 0.94,
    max_iter: int = 10 ** 7,
):
    """
    Classical projected (sub)‑gradient descent on the exact ℓ∞‑norm.
    Used as a short fine‑tuning pass after the Adam stage.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # exact sub‑gradient (all entries within relative tolerance of the max)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given wall‑clock budget.

    The algorithm follows the three‑stage scheme described in the module
    docstring, with a final optional short sub‑gradient polishing pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid hard termination
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – upsample a known construction if available,
    # otherwise start from a random simplex point.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 6000  # a compromise between freedom and speed
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – exponential‑gradient warm‑up on a smooth max
    # ------------------------------------------------------------------
    stage1_time = min(15.0, 0.07 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(
            f"[{time.time() - start_time:.2f}s] after stage 1: score={score_warm:.6f}"
        )

    # ------------------------------------------------------------------
    # Stage 2 – Adam on the smoothed max (log‑sum‑exp)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve at least 30 s for the final refinement
        stage2_deadline = time.time() + max(30.0, 0.75 * remaining)

        a_adam, score_adam = _run_adam_softmax(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            alpha=200.0,
            lr0=0.25,
            beta1=0.9,
            beta2=0.999,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(
                f"[{time.time() - start_time:.2f}s] after stage 2: score={score_adam:.6f}"
            )

    # ------------------------------------------------------------------
    # Stage 2.5 – short projected sub‑gradient polishing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a small slice (≈20 s) for this cheap pass
        if remaining > 20.0:
            sub_deadline = time.time() + 20.0
            a_sub, score_sub = _run_subgradient_descent(
                best_seq,
                final_n,
                sub_deadline,
                start_time,
                step0=0.12,
                beta=0.94,
            )
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] after sub‑grad: score={score_sub:.6f}"
                )

    # ------------------------------------------------------------------
    # Stage 3 – deterministic optimal incremental refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_optimal_incremental_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=1e-6,
            rel_improve=1e-14,
            recompute_every=2000,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
