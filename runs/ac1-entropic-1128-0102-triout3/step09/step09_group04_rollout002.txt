Propose a non‑negative step‑function that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three tightly timed phases:

1. **Warm‑start** –  initialise the vector on the simplex.
   If a previous construction is available it is interpolated,
   otherwise an “arcsine” shape a_i ∝ 1/√((i+½)(n‑i‑½)) is used, which
   is known to flatten the self‑convolution.

2. **Smooth‑max gradient descent** –  we minimise a smooth surrogate of
   the max‑convolution using the log‑sum‑exp (soft‑max) with a
   temperature ε that anneals.  The gradient is obtained by a single
   FFT‑based correlation (O(n log n)).  An Adam‑style update together
   with Euclidean projection onto the probability simplex drives the
   optimisation.

3. **Local stochastic polishing** –  when the time budget is almost
   exhausted we perform cheap random mass‑transfer moves (two‑coordinate
   swaps) accepted by a simulated‑annealing rule.  This helps to jump
   out of shallow local minima discovered by the gradient phase.

All phases stop automatically before the supplied time budget expires.
The returned value is a plain list of floats ready for ``evaluate_sequence``.

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based, no external dependencies)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using a real FFT.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_from_weights(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution w.r.t. a.
    p – soft‑max weights of length 2·n‑1 (must sum to 1).
    Returns 2·corr(p, a) where corr is the valid‑mode correlation.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    L = _next_pow2(len(p) + len(rev_a) - 1)   # ≥ 3·n‑2
    P = np.fft.rfft(p, n=L)
    AR = np.fft.rfft(rev_a, n=L)
    conv = np.fft.irfft(P * AR, n=L)          # length L
    start = n - 1
    grad = 2.0 * conv[start:start + n]        # valid correlation of length n
    return grad


def _objective(a: np.ndarray) -> float:

    Exact objective used by ``evaluate_sequence`` for a simplex‑normalised a.
    sum(a) = 1 is assumed (the function rescales internally if not).
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())

            a_new = a_cur.copy()
            a_new[i] += delta
            a_new[j] -= delta

            new_score = _objective(a_new)

            # Metropolis acceptance
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a_cur = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_cur.copy()
                    print(f"[{time.time() - start_time:6.2f}s] polish iter {it_polish:<6d} "
                          f"improve → {best_score:.6f}")

    # -------------------------------------------------
    # Final normalisation and return a plain list.
    # -------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
