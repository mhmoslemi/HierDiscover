The score that we have to minimise is  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

The objective is *scale‑invariant* – multiplying the whole vector by a constant does not change the score – therefore we can (and will) keep the vector normalised to a probability distribution  
\(\sum_i a_i = 1\).

The previous state‑of‑the‑art code performed a long exponentiated‑gradient (EG) descent that only used the **single** shift that attains the current maximum of the convolution.  
When the maximum is flat (many shifts have almost the same value) this update quickly gets stuck.

My new search routine tackles this by

1. **Diverse initialisations** – uniform, exponential, Gaussian, Dirichlet and a few sparse “spike” vectors.  
2. **A short Adam warm‑up** on a log‑sum‑exp smoothed version of the max‑convolution (exactly as before, but with a slightly larger budget).  
3. **Multi‑active EG polishing** – at each iteration the gradient is built from the *K* largest convolution entries (instead of only the absolute maximum).  This moves mass away from several competing peaks simultaneously and can break the plateaux that stopped the previous method.  
4. **Deterministic greedy peak‑reduction** (identical to the old code) as a final polishing step.  

All steps respect the time budget (default 1000 s).  The routine returns the best normalised vector found.

```python
"""
Propose a non‑negative step function (probability vector) that
minimises ``evaluate_sequence``.

Algorithm
---------
* build a diverse pool of starts (uniform, exponential, Gaussian,
  Dirichlet and several sparse spike vectors);
* a short Adam optimisation on a log‑sum‑exp surrogate of the
  max‑convolution;
* an extensive “multi‑active” exponentiated‑gradient phase that
  uses the K largest convolution entries to compute a sub‑gradient;
* a final greedy peak‑reduction that moves mass from the most
  contributing coefficient to the smallest one;
* everything is performed inside the supplied time budget (default
  1000 s).  The returned vector is normalised to sum to 1 and clipped
  to the domain required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam iterations on the smoothed max‑convolution objective.
    Returns the best vector (probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoother → sharper schedule
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    iters_per_alpha = 3000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max (valid probability vector)
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Multi‑active exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _expgrad_active_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len: int,
                             eta0: float = 0.6,
                             top_k: int = 5,
                             rel_tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) where the sub‑gradient
    is built from the ``top_k`` largest convolution entries (instead of
    only the absolute maximum).  This helps to move mass away from
    several competing peaks at once.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] MultiEG improve: iter={t:,}, score={cur_score:.6f}"
            )

        # --- active set -------------------------------------------------
        # indices of the K largest values
        k = min(top_k, conv.size)
        idx_top = np.argpartition(-conv, k - 1)[:k]

        # also include any values that are within a relative tolerance
        # of the current maximum (helps when the top is flat)
        thresh = cur_max - max(rel_tol, rel_tol * cur_max)
        idx_close = np.where(conv >= thresh)[0]
        active = np.unique(np.concatenate([idx_top, idx_close]))

        # --- compute sub‑gradient for the active set -------------------
        g = np.zeros_like(a)
        for shift in active:
            lo = max(0, shift - (n - 1))
            hi = min(n - 1, shift)
            i = np.arange(lo, hi + 1, dtype=int)
            j = shift - i
            g[i] += 2.0 * a[j]          # 2 * a_{shift-i}

        # --- exponentiated update ---------------------------------------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation (guard against drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local improvement (same as the previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Shift a fraction of the coefficient that most contributes to the current peak
    towards the smallest coefficient, accepting only strictly decreasing moves."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _sparse_start(n: int, rng: np.random.Generator,
                  min_sz: int = 30, max_sz: int = 150) -> np.ndarray:
    """Very sparse vector: a few equally‑weighted spikes."""
    m = rng.integers(min_sz, max_sz + 1)
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n)
    vec[idx] = 1.0
    return vec / vec.sum()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (normalised to sum = 1)
    that minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse set of starting points
    # ------------------------------------------------------------------
    starts = []

    # 1) previously discovered best (if available)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted seeds
    starts.append(np.full(n, 1.0 / n))               # uniform
    starts.append(_exponential_start(n, r=0.999))    # gentle decay
    starts.append(_exponential_start(n, r=0.985))    # stronger decay
    starts.append(_gaussian_start(n))                # bell curve
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 3) a few sparse spike vectors
    for _ in range(5):
        starts.append(_sparse_start(n, rng))

    # evaluate raw starts, keep the best
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – short Adam warm‑up (≈8 % of the total budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + budget_s * 0.08
    if time.time() < adam_deadline:
        cand_seq, cand_score = _adam_optimize(
            best_seq, adam_deadline, start_time, fft_len
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – multi‑active exponentiated‑gradient polishing
    #            (≈84 % of the budget)
    # ------------------------------------------------------------------
    grad_deadline = start_time + budget_s * 0.92
    if time.time() < grad_deadline:
        polished_seq, polished_score = _expgrad_active_optimize(
            best_seq, grad_deadline, start_time, fft_len,
            eta0=0.6, top_k=5, rel_tol=1e-9
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after MultiEG score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return a plain Python list (as required by the judge)
    return [float(x) for x in best_seq.tolist()]
```
