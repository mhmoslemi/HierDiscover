def _run_exp_gradient_phase(a, n, deadline, start_time):
    L = _next_pow2(2*n-1)
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a), n)
    it = 0
    alpha = 80.0
    alpha_min = 1.0
    decay = 0.99995  # per iteration
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] EG iter={it}, score={cur_score:.6f}")
        # compute softmax weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]
        # gradient of smoothed max
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[:len(a) + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
        # exponentiated update
        eta = 0.4 / math.sqrt(it)
        # clip grad to avoid overflow
        grad_clip = np.clip(grad, 0.0, 20.0)
        a = a * np.exp(-eta * grad_clip)
        # renormalise onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
        # decay alpha
        alpha = max(alpha * decay, alpha_min)
    return best_a, best_score

def _run_adam_phase(a, n, deadline, start_time):
    eps = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    L = _next_pow2(2*n-1)
    # init parameters u from a
    u = np.log(a + eps)  # small additive
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a), n)
    alpha = 20.0
    alpha_min = 0.5
    decay = 0.9999
    while time.time() < deadline:
        t += 1
        # compute a
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)  # stability
        a = exp_u / np.sum(exp_u)
        # compute conv
        conv = _conv_self(a)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam iter={t}, score={cur_score:.6f}")
        # smooth max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        w /= w_sum
        w_rev = w[::-1]
        # gradient w.r.t a
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[:len(a)+len(w_rev)-1]
        grad_a = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
        # chain rule to u
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        u = u - 0.5 * m_hat / (np.sqrt(v_hat) + eps)  # learning rate 0.5
        # decay alpha
        alpha = max(alpha * decay, alpha_min)
    return best_a, best_score
```

def _run_improved_pair_annealing(a, n, deadline, start_time, cur_best_score, cur_best_seq):
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    L = _next_pow2(2*n-1)
    conv = _conv_self(a)
    cur_score = _score_from_conv(conv, n)
    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-8
    max_steps = int(200_000)  # allocate many steps
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        # cooling schedule
        T = start_temp * (end_temp / start_temp) ** (step / max_steps)
        # recompute conv only if needed? we have conv from previous iteration; use it.
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]
        # choose top m indices (by contribution) as sources
        m_src = min(2, len(idx))
        top_indices = idx[np.argpartition(contrib, -m_src)[-m_src:]]
        # Build src list (unique)
        src_set = set()
        for i in top_indices:
            j = k - i
            src_set.add(i)
            src_set.add(j)
        src = list(src_set)
        # select low-mass destinations
        low_num = max(2, n // 200)  # 0.5% lowest
        low_idx = np.argpartition(a, low_num)[:low_num]
        # ensure destinations not in src
        dest_candidates = low_idx[~np.isin(low_idx, src)]
        dst = []
        needed = len(src)
        # if insufficient low candidates, fill random
        while len(dst) < needed:
            if dest_candidates.size > 0:
                d = rng.choice(dest_candidates)
                # remove chosen to avoid repeats
                dest_candidates = dest_candidates[dest_candidates != d]
            else:
                d = rng.integers(n)
                while d in src or d in dst:
                    d = rng.integers(n)
            dst.append(int(d))
        # propose move
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            # epsilon fraction proportional to temperature
            max_eps = 0.25 * (T / start_temp)  # up to 25% early, decreasing
            eps_frac = rng.random() * max_eps
            eps = a_prop[s] * eps_frac
            a_prop[s] -= eps
            a_prop[d] += eps
        # evaluate new conv and score
        conv_new = _conv_self(a_prop)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - cur_score
        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            conv = conv_new
            cur_score = new_score
        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] anneal iter={step}, best_score={best_score:.6f}")
    return best_a, best_score

def _run_subgradient_refine(a, n, deadline, start_time, cur_best_score):
    best_a = a.copy()
    best_score = cur_best_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.2
    L = _next_pow2(2*n-1)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subrefine iter={it}, score={cur_score:.6f}")
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # project onto simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

"""
Enhanced optimizer for the step‑function self‑convolution problem.

The algorithm consists of four stages:

* **Stage 1 – Exponentiated‑gradient on a smooth max**  
  Uses a log‑sum‑exp surrogate of max(conv) with a decaying smoothing
  parameter α.  The update a←a·exp(−η∇) stays on the simplex.

* **Stage 2 – Adam on the same surrogate**  
  Works in the log‑probability (softmax) parametrisation, allowing an
  adaptive per‑coordinate step‑size while always satisfying the simplex
  constraint.

* **Stage 3 – Improved pair‑annealing**  
  Repeatedly attacks the two entries that contribute most to the current
  maximal convolution coefficient.  Mass is moved from those sources to
  the currently smallest entries (instead of uniformly random destinations)
  and the temperature schedule follows an exponential cooling law.

* **Stage 4 – Sub‑gradient refinement**  
  A short projected sub‑gradient descent (with momentum) cleans up any
  remaining spikes.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain list of non‑negative floats that
sum to 1 (the scale is irrelevant for the objective ``evaluate_sequence``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective = 2·n·max(conv) (a is already normalised)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated gradient on a smooth max
# ----------------------------------------------------------------------
def _run_exp_gradient_phase(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a), n)

    L = _next_pow2(2 * n - 1)
    it = 0
    alpha = 80.0          # initial smoothing
    alpha_min = 1.0
    decay = 0.99995       # exponential decay per iteration

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}, score={cur_score:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]

        # gradient of the smooth max (2·corr)
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # exponentiated update
        eta = 0.4 / math.sqrt(it)
        grad = np.clip(grad, 0.0, 20.0)          # prevent overflow
        a = a * np.exp(-eta * grad)

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # decay the smoothing parameter
        alpha = max(alpha * decay, alpha_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Adam optimiser on the same smooth max
# ----------------------------------------------------------------------
def _run_adam_phase(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    eps = 1e-12
    beta1 = 0.9
    beta2 = 0.999
    L = _next_pow2(2 * n - 1)

    # initialise log‑probability parameters (softmax)
    u = np.log(a0 + eps)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0

    best_a = a0.copy()
    best_score = _score_from_conv(_conv_self(a0), n)

    alpha = 20.0          # start smoothing
    alpha_min = 0.5
    decay = 0.9999

    while time.time() < deadline:
        t += 1

        # softmax to obtain a
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / exp_u.sum()

        # current objective
        conv = _conv_self(a)
        cur_score = _score_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,}, score={cur_score:.6f}")

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient w.r.t. a
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
        grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # chain rule to the log‑probability parameters
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        u = u - 0.5 * m_hat / (np.sqrt(v_hat) + eps)   # learning rate 0.5

        # decay smoothing
        alpha = max(alpha * decay, alpha_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Improved targeted pair‑annealing
# ----------------------------------------------------------------------
def _run_improved_pair_annealing(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float,
                                 cur_best_score: float,
                                 cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    conv = _conv_self(a)
    cur_score = _score_from_conv(conv, n)

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-8
    max_steps = 200_000
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (step / max_steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]

        # take the two (or three) indices with largest contributions
        m_src = min(2, len(idx))
        top_idx = idx[np.argpartition(contrib, -m_src)[-m_src:]]
        src_set = set()
        for i in top_idx:
            src_set.add(i)
            src_set.add(k - i)
        src = list(src_set)

        # choose destinations among the currently smallest entries
        low_num = max(2, n // 200)                     # ~0.5 % of the indices
        low_idx = np.argpartition(a, low_num)[:low_num]
        low_idx = low_idx[~np.isin(low_idx, src)]

        dst = []
        needed = len(src)
        while len(dst) < needed:
            if low_idx.size > 0:
                d = rng.choice(low_idx)
            else:
                d = rng.integers(n)
                while d in src or d in dst:
                    d = rng.integers(n)
            dst.append(int(d))

        # propose a move: shift a fraction of each source to its destination
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            max_eps = 0.25 * (T / start_temp)          # up to 25 % early, vanishes later
            eps_frac = rng.random() * max_eps
            eps = a_prop[s] * eps_frac
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            conv = conv_new
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal iter={step:,}, best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Projected sub‑gradient refinement (momentum)
# ----------------------------------------------------------------------
def _run_subgradient_refine(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_best_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.2
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑refine iter={it:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score
