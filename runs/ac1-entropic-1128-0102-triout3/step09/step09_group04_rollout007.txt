"""
Propose a non‑negative step‑function (vector on the simplex) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The routine works completely within the allotted time budget.
It consists of three inexpensive stages

  1.  Smooth gradient descent on a soft‑max approximation of the max‑norm.
      The temperature (softness) is increased during the run, turning the
      objective gradually into the true max‑norm.  Gradients are obtained
      analytically by convolving the current vector with the soft‑max
      weights (FFT‑based, O(n log n)).  After each step the iterate is projected
      onto the probability simplex.

  2.  Exact sub‑gradient descent on the true max‑norm (the same routine used
      in the reference solution) but with a much smaller step‑size and a
      momentum term.  This polishes the solution obtained by stage 1.

  3.  Stochastic local search: tiny mass transfers between two random
      coordinates are accepted only if they improve the objective.  This
      helps to escape the remaining flat regions and typically yields a few
      further 10⁻⁴‑level improvements.

The algorithm never uses an LP solver (so it works for any vector length)
and runs fast enough for n ≈ 2048–4096.  The best vector found is returned
as a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small power‑of‑two helper (used for FFT lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real inputs)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_max(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth log‑sum‑exp approximation of max(conv(a,a)).
    Returns (grad, max_value) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    max_val = conv.max()
    # stabilise exponentials
    diff = conv - max_val          # ≤ 0
    w = np.exp(alpha * diff)       # max entry becomes 1
    Z = w.sum()
    if Z == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / Z                  # soft‑max weights, sum to 1

    # convolution a * rev(p)   (size 3n‑2)
    L2 = _next_pow2(3 * n - 2)
    a_fft = np.fft.rfft(a, n=L2)
    p_rev_fft = np.fft.rfft(p[::-1], n=L2)
    full = np.fft.irfft(a_fft * p_rev_fft, n=L2)[: 3 * n - 2]

    # gradient_j = 2·α·full[2n‑2‑j]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * alpha * full[idx]
    return grad, max_val


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (as in the reference solution)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).
    All indices attaining the maximum are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad
