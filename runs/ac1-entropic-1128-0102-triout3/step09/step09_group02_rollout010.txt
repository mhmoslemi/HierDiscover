def lp_refinement(a, total, L, row_ind, col_ind, deadline, max_iterations=50):
    n = len(a)
    best_a = a.copy()
    best_score = float('inf')
    best_M = None
    for it in range(max_iterations):
        if time.time() >= deadline:
            break
        # Compute conv and M
        conv = _conv_self(a, L)
        M = conv.max()
        # Build constraint matrix A
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n))
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        A_eq = np.ones((1, n))
        b_eq = np.array([total])
        bounds = [(0, None)] * n

        # Solve with time limit
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # SciPy's time limit is seconds, integer? We'll set options.
        # Use hiGHS method
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                    bounds=bounds, method='highs',
                                    options={'dual_feasibility_tolerance': 1e-9,
                                             'presolve': True,
                                             'time_limit': max(0, remaining)})
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        # update via convex combination
        # try different t values
        t_vals = [0.5, 0.2, 0.1, 0.05, 0.02]
        improved = False
        for t in t_vals:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum <= 0:
                continue
            a_new = a_new / a_new_sum * total
            conv_new = _conv_self(a_new, L)
            M_new = conv_new.max()
            score_new = (2 * n * M_new) / (total**2)  # total sum fixed
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                best_M = M_new
                a = a_new  # accept this as current a for next iteration
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP iter {it}, t={t:.3f}, new score {score_new:.6f}")
                break   # accept first t that improves
        if not improved:
            # can't improve further; break
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} no improvement, stop")
            break
    return best_a, best_score

    max‑convolution.  This quickly produces a non‑uniform shape with a
    noticeably lower peak than the uniform distribution.

2.  An iterative linear‑programming refinement (the “LP‑step” of the
    reference).  For the current sequence f we solve
        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0,
                    Σ g_i = Σ f_i .
    Here M = max(f * f).  The solution g is then mixed with f
    (convex combination) – a small step of this mixing is guaranteed not
    to increase M, and in practice it reduces it.  Repeating the LP +
    mixing step drives the objective down steadily.  The constraint matrix
    is built once (sparse Toeplitz) and reused, making each LP cheap.

3.  A final greedy “peak‑reduction” pass that moves a tiny amount of
    mass from the index contributing most to the current peak to an
    index that contributes least.  The move size is chosen by an
    exact line‑search that respects *all* convolution entries, so each
    iteration strictly lowers the peak.

All three stages respect the time budget (default 1000 s).  Progress
messages are printed, the best sequence found is stored, and the final
sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.  Returns the full convolution
    of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (exactly sum‑to‑1)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Linear‑programming refinement (LP‑step)
# ----------------------------------------------------------------------
def _conv_matrix_structure(n: int):
    """Pre‑compute the CSR index arrays for the convolution matrix."""
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    row = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
    col = np.repeat(i, n)                            # shape (n*n,)
    return row, col


def _lp_refine(a: np.ndarray,
               total: float,
               L: int,
               row_ind: np.ndarray,
               col_ind: np.ndarray,
               deadline: float,
               start_time: float,
               max_iters: int = 20) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP
        max Σ g_i  s.t.  (a * g)[k] ≤ M,  Σ g_i = total,  g ≥ 0
    and mix the solution back into a with a small convex step.
    Returns the best sequence found during the refinement.
    """
    n = a.shape[0]
    best_a = a.copy()
    conv = _conv_self(a, L)
    M = conv.max()
    best_score = 2.0 * n * M / (total ** 2)

    # mixing factor schedule – start relatively aggressive and decay
    t_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Re‑compute M for the current a
        conv = _conv_self(a, L)
        M = conv.max()

        # Build the (sparse) constraint matrix A where A @ g = a * g
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))

        # Linear program: maximise Σ g_i  → minimise -Σ g_i
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        A_eq = np.ones((1, n))
        b_eq = np.array([total])
        bounds = [(0.0, None)] * n

        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # HiGHS respects the 'time_limit' option
        opts = {"presolve": True, "time_limit": max(0.0, remaining)}
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=opts,
        )

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed on iter {it}: {res.message}")
            break

        g = res.x

        # Try a few mixing factors; accept the first that improves the score.
        improved = False
        for t in t_schedule:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum <= 0.0:
                continue
            a_new = a_new / a_new_sum * total

            conv_new = _conv_self(a_new, L)
            M_new = conv_new.max()
            score_new = 2.0 * n * M_new / (total ** 2)

            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP iter {it}, mix {t:.3f}, "
                      f"score={score_new:.6f}")
                break

        if not improved:
            # No mixing factor yielded improvement → stop the refinement.
            print(f"[{time.time() - start_time:.2f}s] LP iter {it} no improvement, stop.")
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Same routine as in the reference code, with a tighter line‑search."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # Build a_k = a[max_idx - i] on the relevant support
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination index with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # ----- exact line‑search for eps -----
        # target is to bring the current peak down to (just below) the second max
        target = second_max

        k_all = np.arange(2 * n - 1)
        # contributions of src and dst to every convolution index
        idx_src = k_all - src
        src_mask = (idx_src >= 0) & (idx_src < n)
        a_src_vec = np.zeros_like(conv)
        a_src_vec[src_mask] = a[idx_src[src_mask]]

        idx_dst = k_all - dst
        dst_mask = (idx_dst >= 0) & (idx_dst < n)
        a_dst_vec = np.zeros_like(conv)
        a_dst_vec[dst_mask] = a[idx_dst[dst_mask]]

        delta = 2.0 * (a_dst_vec - a_src_vec)          # change in conv per unit eps
        # bounds from constraints where delta > 0
        pos = delta > 1e-12
        feasible = True
        if np.any(pos):
            bound_vals = (target - conv[pos]) / delta[pos]
            # feasible eps must be ≤ these bounds and also ≤ a[src]
            eps = min(a[src], bound_vals.min())
            if eps <= 0.0:
                feasible = False
        else:
            eps = a[src] * 0.9   # no upper bound from other entries

        if not feasible or eps <= 0.0:
            # fallback – move a tiny amount
            eps = min(a[src], 0.02 * a[src])

        # ----- apply the move with incremental convolution update -----
        a_old = a.copy()
        # update convolution values in place (O(n))
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – start from the previously discovered construction if
    # available, otherwise from a random simplex point.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
        else:
            starter = None
    except Exception:
        starter = None

    # Work at a moderate size (2048) – large enough to be expressive but
    # small enough for the LP stage to be fast.
    base_n = 2048
    if starter is not None:
        a0 = np.interp(
            np.linspace(0, 1, base_n),
            np.linspace(0, 1, starter.size),
            starter,
        )
    else:
        a0 = np.random.rand(base_n)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(base_n, 1.0 / base_n)
    else:
        a0 = a0 / a0.sum()   # normalise to a probability simplex

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Exponentiated‑gradient descent (quick global shaping)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + max(10.0, 0.15 * (deadline - time.time()))
    a_exp, score_exp = _run_exp_gradient(
        best_seq,
        base_n,
        phase1_deadline,
        start_time,
        eta0=0.6,
    )
    if score_exp < best_score:
        best_score = score_exp
        best_seq = a_exp.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(30.0, 0.40 * (deadline - time.time()))
        # Pre‑compute the sparse matrix pattern once
        row_ind, col_ind = _conv_matrix_structure(base_n)
        L_fft = _next_pow2(2 * base_n - 1)
        a_lp, score_lp = _lp_refine(
            best_seq,
            total=1.0,
            L=L_fft,
            row_ind=row_ind,
            col_ind=col_ind,
            deadline=phase2_deadline,
            start_time=start_time,
            max_iters=25,
        )
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve (LP): score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic greedy reduction (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = deadline
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            base_n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=800_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve (greedy): score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the original length (4096) – the objective is
    # scale‑invariant, so interpolation does not hurt the score much.
    # ------------------------------------------------------------------
    final_n = 4096
    if final_n != base_n:
        best_seq = np.interp(
            np.linspace(0, 1, final_n),
            np.linspace(0, 1, base_n),
            best_seq,
        )
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # ------------------------------------------------------------------
    # Final evaluation (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final score {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
