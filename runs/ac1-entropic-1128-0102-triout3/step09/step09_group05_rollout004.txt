def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine length
    final_n = 8192

    # Initialize from previous sequence if provided, else uniform
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim ==1 and starter.size>0:
            # upsample via interpolation
            a0 = np.interp(np.linspace(0,1,final_n), np.linspace(0,1,starter.size), starter)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.rand(final_n)
    # Clip negative and normalize
    a0 = np.maximum(a0,0)
    a0 = a0 / a0.sum()

    best_seq = a0.copy()
    best_score = 2*final_n * np.convolve(best_seq, best_seq).max() / (best_seq.sum()**2)  # but sum=1

    # Phase 1: exponentiated gradient
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(10.0, 0.12*remaining)
    a1, s1 = _run_exp_gradient(best_seq, final_n, phase1_deadline, start,
                               alphas=[0.5,1,2,5,10,20,40,80,160],
                               eta0=0.5,
                               iters_per_alpha=12000)
    if s1 < best_score:
        best_seq, best_score = a1.copy(), s1
        print(f"[{time.time()-start:.2f}s] phase1 improve: {best_score:.6f}")

    # Phase 2: subgradient descent (main)
    if time.time()<deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(30.0, 0.70*remaining)
        a2, s2 = _run_subgradient_descent(best_seq, final_n, phase2_deadline, start, best_score)
        if s2 < best_score:
            best_seq, best_score = a2.copy(), s2
            print(f"[{time.time()-start:.2f}s] phase2 improve: {best_score:.6f}")

    # Phase 3: long simulated annealing
    if time.time()<deadline:
        steps = 500_000
        a3, s3 = _run_pair_annealing(best_seq, final_n, deadline, start,
                                    best_score, best_seq,
                                    start_temp=1e-1, end_temp=1e-9, steps=steps)
        if s3 < best_score:
            best_seq, best_score = a3.copy(), s3
            print(f"[{time.time()-start:.2f}s] phase3 improve: {best_score:.6f}")

    # Optional quick random restarts if time left
    while time.time()<deadline-5.0:
        # simple perturb and short subgradient+anneal
        pert = best_seq + 0.01 * np.random.randn(final_n)
        pert = np.clip(pert,0, None)
        pert /= pert.sum()
        # short subgradient
        sub_dead = time.time() + 5.0

"""
A three‑stage optimiser for the self‑convolution bound.

The optimiser works on a longer sequence (default 8192) than the
reference implementation.  It consists of

1️⃣  exponentiated‑gradient descent on a log‑sum‑exp surrogate of the
    maximal convolution entry (smooth, always stays on the simplex);

2️⃣  projected sub‑gradient descent on the exact objective
    (fast, makes rapid progress on the true max‑convolution);

3️⃣  a long simulated‑annealing phase that repeatedly weakens the two
    entries that contribute most to the current peak and redistributes
    a random fraction of their mass elsewhere.  The temperature schedule
    is very slow (≈10⁻¹ → 10⁻⁹ over 5·10⁵ steps) so that the algorithm can
    escape shallow local minima.

If any time remains, a few cheap random restarts are performed.
The final sequence is normalised to sum to one and returned as a plain
list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact evaluation score (sum of a is assumed to be 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on smooth max
# ----------------------------------------------------------------------


def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.5,
                      iters_per_alpha: int = 15000):
    """Exponentiated‑gradient descent on a log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    best_score = _score_from_conv(conv, n)

    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # current convolution and exact score
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth‑max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smooth max (2·corr)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- multiplicative update -----
            eta = eta0 / math.sqrt(it)
            grad = np.clip(grad, 0.0, 10.0)          # avoid overflow
            a = a * np.exp(-eta * grad)

            # project back onto the simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient coming from the maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # contributions from the peak

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Aggressive pair‑annealing
# ----------------------------------------------------------------------


def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 1e-1,
                        end_temp: float = 1e-9,
                        steps: int = 500_000):
    """
    Simulated annealing that always attacks the two entries that
    contribute most to the current maximal convolution coefficient.
    The temperature schedule is very slow, allowing the algorithm to
    leave shallow basins.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # indices that take part in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i to the peak (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # choose the pair with the largest product
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a *random* fraction (≤30 %) of each source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.30
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate the proposal (full convolution – still cheap)
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        # keep the best ever seen
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score
