"""
Propose a step‐function (non‑negative vector) with a very small
evaluate_sequence() value.

The strategy consists of three phases

1️⃣  Analytic families – generate many simple shapes (power‑tents,
    sin‑powers, geometric decay, etc.) and keep the best one.

2️⃣  Projected sub‑gradient descent – start from the best analytic shape,
    run a fast momentum sub‑gradient method on the exact objective.
    The iterate is always projected onto the probability simplex
    (∑a = 1, a_i ≥ 0).

3️⃣  Greedy “mass‑transfer” refinement – use an O(n) exact update of the
    self‑convolution when a tiny amount of mass is moved from one
    index to another.  Only downhill moves are kept (no temperature
    schedule is needed) and the current convolution is periodically
    recomputed from scratch to avoid numerical drift.

All phases respect the global time budget (default 1000 s).  The final
vector is normalised to sum 1 and returned as plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for the exact objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30          # a slightly larger initial step than before
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.1f}s] sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer refinement (exact conv update in O(n))
# ----------------------------------------------------------------------
def _greedy_refine(a: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a huge number of tiny mass‑transfer moves.
    Only downhill moves are kept.
    The convolution is updated exactly in O(n) per move.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # exact start
    cur_score = 2.0 * n * conv.max()         # sum(a) == 1
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_iters = 30_000_000                 # huge loop, will stop when time runs out
    # cooling is not used – we accept only improvements
    max_transfer = 0.30                      # move at most 30 % of the source weight

    # periodic re‑computation to avoid drift
    recompute_every = 200_000

    for it in range(1, total_iters + 1):
        if time.time() >= deadline:
            break

        # pick a source proportionally to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * max_transfer   # ≤ max_transfer·a[src]
        if eps <= 0.0:
            continue

        # ----- apply tentative move (using the *old* a values) -----
        # store values that will be needed for possible revert
        a_src_before = a[src]
        a_dst_before = a[dst]

        # update convolution exactly (see analysis)
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a

        ii = src + src
        if ii < conv.size:
            conv[ii] += eps * eps
        jj = dst + dst
        if jj < conv.size:
            conv[jj] += eps * eps
        kk = src + dst
        if kk < conv.size:
            conv[kk] -= 2.0 * eps * eps

        # update the vector itself
        a[src] -= eps
        a[dst] += eps

        # ---- evaluate ----
        new_score = 2.0 * n * conv.max()

        if new_score < cur_score - 1e-12:          # accept only strict improvements
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"greedy iter={it:,} best={new_score:.6f}")
        else:
            # ---- revert ----
            # reverse the convolution update (use the *post‑move* a values)
            # recover the original a values for the two positions
            a_src = a[src]          # this is a_src_before - eps
            a_dst = a[dst]          # this is a_dst_before + eps

            # undo slice updates (note: we need the original a values)
            conv[src:src + n] += 2.0 * eps * (a_src_before)      # add back what we subtracted
            conv[dst:dst + n] -= 2.0 * eps * (a_dst_before)      # remove what we added

            # undo epsilon‑squared corrections
            if ii < conv.size:
                conv[ii] -= eps * eps
            if jj < conv.size:
                conv[jj] -= eps * eps
            if kk < conv.size:
                conv[kk] += 2.0 * eps * eps

            # restore a
            a[src] = a_src_before
            a[dst] = a_dst_before

        # periodic full recomputation to keep numerical errors in check
        if it % recompute_every == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (sum is normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – we follow the historical setting (4096 points)
    # ------------------------------------------------------------------
    n = 4096
    xs = np.arange(n, dtype=float) / (n - 1)          # points in [0,1]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Sweep a large collection of simple analytic families
    # ------------------------------------------------------------------
    # (a) Power‑tent shapes  w = min(x,1‑x)^p
    for p in np.linspace(0.2, 2.5, 27):
        w = np.minimum(xs, 1.0 - xs) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.1f}s] power‑tent p={p:.3f} "
                  f"score={sc:.6f}")

    # (b) Sine‑power shapes  w = sin(π·x)^p
    sinx = np.sin(np.pi * xs)
    for p in np.linspace(0.5, 2.5, 21):
        w = sinx ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.1f}s] sin‑power p={p:.3f} "
                  f"score={sc:.6f}")

    # (c) Geometric decay  w_i = q^i
    for q in np.linspace(0.90, 0.995, 20):
        w = q ** np.arange(n, dtype=float)
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.1f}s] geometric q={q:.3f} "
                  f"score={sc:.6f}")

    # (d) Simple random Dirichlet draws (limited count)
    for _ in range(12):
        a = np.random.rand(n)
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.1f}s] random draw score={sc:.6f}")

    # (e) Global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.1f}s] global starter score={sc:.6f}")
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent (moderate time slice)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                               # use at most ~30 % of the rest
        sub_deadline = time.time() + remaining * 0.30
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.1f}s] sub‑grad final "
                  f"score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Greedy massive mass‑transfer refinement (the heavy part)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()
        a, final_score = _greedy_refine(a, deadline, start_time)
        if final_score < best_score:
            best_score, best_seq = final_score, a.copy()
            print(f"[{time.time() - start_time:.1f}s] greedy final "
                  f"score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation (protect against accidental numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list as required by the evaluation harness
    return [float(v) for v in best_seq]
