def mass_transfer_step(a, conv, src, dst, delta):
    if src == dst or delta <= 0:
        return False
    n = a.shape[0]

    # Clip potential negatives due to rounding
    if a[src] < 0:
        a[src] = 0.0
    if a[dst] < 0:
        a[dst] = 0.0
    return True

def _subgradient_descent_multi(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_seq = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    # Additional parameters:
    tol_rel = 1e-12  # for active set
    no_improve_counter = 0
    last_improve_iter = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(3*n-2))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve_counter = 0
            last_improve_iter = it
            print(f"[{time.time() - start_time:.2f}s] Sub-grad it={it:,} score={cur_score:.6f}")
        else:
            no_improve_counter += 1

        # Determine active set
        eps_active = cur_max * 1e-9  # tolerance
        active_idxs = np.where(conv >= cur_max - eps_active)[0]
        grad = np.zeros_like(a)
        # sum gradients for each active index
        for k in active_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        # Optionally average
        grad /= max(1, len(active_idxs))

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

under a 1000 s wall‑clock budget (n = 4096).

Algorithm
---------
1️⃣  **Fast parametric sweep** – many analytic families (geometric,
    power‑law, Beta‑type and symmetric double‑exponential) are generated,
    normalised to unit sum and evaluated with an FFT‑based convolution.
    The best candidate of this cheap sweep seeds the next stage.

2️⃣  **Projected sub‑gradient descent** – starting from that seed we run a
    sub‑gradient method on the true objective.  The active set of
    convolution indices attaining the maximum is taken (all within a tiny
    tolerance) and the (averaged) sub‑gradient is used.  Momentum,
    a 1/√t step‑size and occasional random perturbations keep the search
    from stagnating.

3️⃣  **Intensive annealing polish** – a mass‑transfer annealer moves a
    random fraction (up to 40 %) of the mass from a source index to a
    destination index.  The convolution is updated **incrementally**
    (O(n) operations, no FFT) and the Metropolis rule with an exponential
    temperature schedule accepts moves.  The best sequence seen is kept.

All helpers are top‑level; no closures or lambdas are used.
The routine respects the 1000 s budget and returns a plain Python list
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    return np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a non‑negative unit‑sum vector."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Parametric sweep helpers
# ----------------------------------------------------------------------
def _sweep_geometric(best_seq, best_score, n, L_fft, rng):
    idx = np.arange(n)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -0.5, 300)               # finer near 0
    for eps in eps_grid:
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 13):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
    return best_seq, best_score


def _sweep_power(best_seq, best_score, n, L_fft):
    idx = np.arange(n)
    for p in np.linspace(-2.0, 3.0, 151):
        seq = np.power(idx + 1, p, dtype=np.float64)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    return best_seq, best_score


def _sweep_beta(best_seq, best_score, n, L_fft):
    idx = np.arange(n)
    beta_vals = np.linspace(0.0, 2.5, 21)
    for p in beta_vals:
        for q in beta_vals:
            seq = (np.power(idx + 1, p) *
                   np.power(n - idx, q))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
    return best_seq, best_score


def _sweep_double_exponential(best_seq, best_score, n, L_fft):
    """Symmetric sum of two exponentials."""
    idx = np.arange(n)
    uniform = np.full(n, 1.0 / n)
    alphas = np.logspace(-5, -0.5, 200)    # decay rate
    for a in alphas:
        exp_shape = np.exp(-a * idx) + np.exp(-a * (n - 1 - idx))
        s = exp_shape.sum()
        if s == 0:
            continue
        exp_shape /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * exp_shape
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
    return best_seq, best_score

# Sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with an active‑set (all near‑max indices)
    and occasional random perturbations to escape plateaus.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35
    it = 0
    no_improve = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            no_improve += 1

        # ------- active set (all indices within 1e‑12 relative of the max) ----------
        eps = cur_max * 1e-12
        act_idx = np.where(conv >= cur_max - eps)[0]

        grad = np.zeros_like(a)
        for k in act_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(act_idx))

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # occasional random perturbation if stuck too long
        if no_improve > 300_000:
            noise = np.random.default_rng().normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental mass‑transfer annealer (no FFT per move)
# ----------------------------------------------------------------------
def _mass_transfer_step(a: np.ndarray,
                       conv: np.ndarray,
                       src: int,
                       dst: int,
                       delta: float) -> None:
    """
    Perform a mass transfer of size ``delta`` from ``src`` to ``dst``.
    The convolution ``conv`` (a*a) is updated in‑place using the exact
    linear formula derived from the quadratic form.
    """
    n = a.shape[0]
    # conv update must use the *old* a values, therefore we update conv first
    conv[src:src + n] -= 2.0 * delta * a
    conv[dst:dst + n] += 2.0 * delta * a
    conv[2 * src] += delta * delta
    conv[2 * dst] += delta * delta
    conv[src + dst] -= 2.0 * delta * delta

    # now modify the vector
    a[src] -= delta
    a[dst] += delta
    # guard against tiny negatives because of rounding
    if a[src] < 0.0:
        a[src] = 0.0
    if a[dst] < 0.0:
        a[dst] = 0.0


def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 1e-1,
               end_temp: float = 1e-7,
               move_frac: float = 0.40,
               steps: int = 600_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing using incremental convolution updates."""
    a = a0.copy()
    conv = _conv_self(a)                   # exact initial convolution
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * rng.random() * move_frac
        if delta <= 0.0:
            continue

        # build candidate copies
        a_cand = a.copy()
        conv_cand = conv.copy()
        _mass_transfer_step(a_cand, conv_cand, src, dst, delta)

        new_score = 2.0 * n * conv_cand.max() / (a_cand.sum() ** 2)
        delta_obj = new_score - cur_score

        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / T):
            a = a_cand
            conv = conv_cand
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional user‑provided starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (extended families)
    # ------------------------------------------------------------------
    best_seq, best_score = _sweep_geometric(best_seq, best_score, n,
                                            L_fft, rng)
    best_seq, best_score = _sweep_power(best_seq, best_score, n, L_fft)
    best_seq, best_score = _sweep_beta(best_seq, best_score, n, L_fft)
    best_seq, best_score = _sweep_double_exponential(best_seq,
                                                      best_score, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] After sweep best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient refinement (majority of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 15.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                              sub_deadline, start_time,
                                              best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n,
                                       deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-1,
                                       end_temp=1e-7,
                                       move_frac=0.40,
                                       steps=800_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
