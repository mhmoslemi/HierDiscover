assistantfinal**Solution Overview**

The objective in `evaluate_sequence` (after fixing the sum of the coefficients to 1) is  

\[
\text{score}=2\,n\; \max_{k}(a*a)[k],
\]

i.e. we have to *flatten* the self‑convolution of a non‑negative vector `a`.  
The classical “exponentiated‑gradient” method that uses only the *single* shift
attaining the maximum often gets stuck because the sub‑gradient of a `max`
operator is any convex combination of the gradients of all *active* shifts.

The new algorithm therefore

1. **generates many diverse starts** (uniform, exponential, Gaussian,
   Dirichlet with several concentrations and the historic best `height_sequence_1`);
2. **runs a short Adam warm‑up** on a log‑sum‑exp (soft‑max) surrogate
   (identical to the previous code);
3. **performs a long “active‑set exponentiated‑gradient” phase** – at every
   iteration the `top_k` largest convolution entries are taken, their gradients
   are averaged and an exponential update is applied;
4. **finishes with a deterministic greedy mass‑transfer phase** that moves a
   fraction of the coefficient that contributes most to the current peak
   towards the coefficient that contributes least (using an exact fast update);
5. **normalises and returns** the best sequence found before the time budget
   expires.

All heavy work (convolution, gradients) is carried out with FFTs, which makes
the whole search comfortably fit into the 1000 s limit.

```python
#!/usr/bin/env python
"""
Non‑negative step‑function optimiser.

The algorithm consists of:
1. a diverse pool of starts (uniform, exponential, Gaussian, Dirichlet,
   historic best);
2. a brief Adam optimisation of a log‑sum‑exp surrogate of the max‑convolution;
3. a long “active‑set exponentiated‑gradient” phase – the gradient is formed
   from the *top_k* convolution peaks (instead of a single peak);
4. a final greedy mass‑transfer polishing that uses a fast O(n) update of the
   convolution after each move.

The routine respects the supplied time budget (default 1000 s) and returns the
best sequence found.  All numbers are non‑negative and are automatically
normalised to sum to 1 before being handed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of max(conv). Returns value and the raw conv."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """Short Adam run on the smoothed max‑convolution objective."""
    a = a0.copy()
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule – gradually concentrate around the true max
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max → probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _active_expgra(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  fft_len: int,
                  top_k: int = 5,
                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a gradient built from the `top_k` largest
    convolution entries (instead of a single arg‑max shift).
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ActiveExpGrad iter={t:,} score={cur_score:.6f}"
            )

        # ---------- active set ----------
        if top_k >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            active_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        # ---------- gradient from the active shifts ----------
        g = np.zeros_like(a)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]          # sub‑gradient of (a*a)[k]

        # ---------- multiplicative update ----------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation (protect against drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] ActiveExpGrad exact score={sc:.6f}"
                )
    return best_a, best_score

# Greedy local improvement (fast O(n) update of the convolution)
# ----------------------------------------------------------------------


def _conv_fast(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper around the FFT‑based convolution."""
    fft_len = _next_pow2(2 * a.shape[0] - 1)
    return _conv_self(a, fft_len)


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of the coefficient that contributes most to the
    current peak towards the coefficient that contributes least.
    The convolution update after each move is performed in O(n) time.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute the full convolution once
    conv = _conv_fast(a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # recompute the convolution only when needed
        conv = _conv_fast(a)
        cur_max = conv.max()
        k_max = int(conv.argmax())

        # contributions of each index to the current peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        # destination = index with smallest *gradient* (i.e. smallest a[j])
        # we simply pick the global minimum entry (cheap and effective)
        dst = int(np.argmin(a))

        if src == dst:
            break

        # move a modest fraction (25 % of the source mass)
        eps = 0.25 * a[src]
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # fast update of the convolution (linear time)
        # Δconv = 2·eps·(a shifted by dst – a shifted by src)
        shift_src = np.zeros_like(a)
        shift_dst = np.zeros_like(a)

        # contributions where the moved mass appears as the *first* factor
        lo_s = max(0, src)
        hi_s = min(n - 1, src + n - 1)
        idx = np.arange(lo_s, hi_s + 1) - src
        shift_src[idx] = a[idx]

        lo_d = max(0, dst)
        hi_d = min(n - 1, dst + n - 1)
        idx = np.arange(lo_d, hi_d + 1) - dst
        shift_dst[idx] = a[idx]

        delta = 2.0 * eps * (shift_dst - shift_src)
        new_conv = conv + delta
        new_max = new_conv.max()

        if new_max < cur_max:
            a = a_prop
            conv = new_conv
            cur_max = new_max
            best_score = 2.0 * n * cur_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The pipeline:
      * generate a diverse pool of starts,
      * a short Adam warm‑up on a smoothed surrogate,
      * a long active‑set exponentiated‑gradient phase,
      * a final greedy peak‑reduction.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build the start pool
    # ------------------------------------------------------------------
    starts = []

    # historic best (if available)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # handcrafted candidates
    starts.append(np.full(n, 1.0 / n))               # uniform
    starts.append(_exponential_start(n, r=0.9995))   # very gentle decay
    starts.append(_exponential_start(n, r=0.985))    # stronger decay
    starts.append(_gaussian_start(n))                # bell
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # a few random Dirichlet draws with varying concentration
    for conc in (0.3, 0.8, 1.5, 3.0):
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # Pick the best raw start
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – brief Adam warm‑up (≈10 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.1
    cand_seq, cand_score = _adam_optimize(
        best_seq, explore_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – active‑set exponentiated‑gradient (≈80 % of the time)
    # ------------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.9
    # start with a single active shift and gradually increase to top_k=7
    active_seq, active_score = _active_expgra(
        best_seq,
        polish_deadline,
        start_time,
        fft_len,
        top_k=7,
        eta0=0.5,
    )
    if active_score < best_score:
        best_score = active_score
        best_seq = active_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after active‑expgrad score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=8000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )
    return [float(x) for x in best_seq.tolist()]
```
