"""
Propose a low‑max self‑convolution step‑function.

The search is divided into three phases and runs under a 1000 s wall‑clock
budget (a half‑second safety margin is kept).

1️⃣  **Exponentiated‑gradient warm‑up** – smooth approximation of the ℓ∞
    norm (soft‑max) is minimised by multiplicative updates; this quickly
    moves a random start into a promising region.

2️⃣  **Projected sub‑gradient descent** – the exact max‑convolution
    sub‑gradient is computed (only the active constraints) and projected
    onto the probability simplex.  Momentum and a diminishing step‑size
    give a stable descent.

3️⃣  **Simulated‑annealing refinement** – many cheap mass‑transfer moves
    are performed.  A move removes a fraction ε of the mass from a
    “source’’ index and adds it to a “destination’’ index; the whole
    convolution is updated in O(n) using the linear effect of the move.
    An annealing schedule allows occasional temporary increases of the
    maximum, letting the algorithm escape shallow local minima.
    The best sequence seen during this long phase is kept.

The algorithm works on a normalized vector (∑a_i = 1), therefore the
objective returned by ``evaluate_sequence`` equals ``2·n·max(conv)``.
The implementation is fully self‑contained and uses only NumPy.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` to length `new_n` while preserving mass.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (soft‑max)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: simulated‑annealing refinement (mass‑shifts)
# ----------------------------------------------------------------------
def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             init_temp_factor: float = 0.02,
                             final_temp_factor: float = 1e-5,
                             epsilon_factor: float = 0.05,
                             recompute_interval: int = 2000):
    """
    Perform a large number of cheap mass‑transfer moves.
    A move shifts ε·a[src] of probability mass from `src` to `dst`.
    The whole convolution is updated in O(n) using the linear effect of the move.
    An annealing temperature gradually cools, allowing occasional uphill steps.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    counter = 0      # number of accepted moves since last full recompute

    while time.time() < deadline:
        # ----- temperature (linear interpolation between start and end) -----
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total
        temp = (1.0 - frac) * init_temp_factor * max_val + frac * final_temp_factor * max_val

        # ----- pick source (prefer indices that contribute heavily to the max bin) -----
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ----- pick destination (prefer indices with little mass) -----
        inv = 1.0 / (a + 1e-12)
        inv[src] = 0.0
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        # ----- choose step size ε -----
        epsilon = a[src] * epsilon_factor * rng.random()
        if epsilon <= 0.0:
            continue

        # ----- compute new convolution cheaply -----
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * epsilon * a
        conv_new[dst:dst + n] += 2.0 * epsilon * a
        new_max = conv_new.max()
        delta = new_max - max_val

        # ----- Metropolis acceptance -----
        accept = False
        if delta <= 0:
            accept = True
        else:
            # probability to accept a worsening move
            if rng.random() < math.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            counter += 1

            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={counter:,}, max={best_max:.8f}")

        # ----- periodic full recomputation to avoid drift ----------
        if counter >= recompute_interval:
            conv = _conv_self(a)
            max_val = conv.max()
            counter = 0

    return best_a, best_max

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose the length of the sequence (larger gives more freedom)
    # ------------------------------------------------------------------
    final_n = 6000                      # can be tuned; 6000 works comfortably

    # ------------------------------------------------------------------
    # Initialise – prefer the known construction if it exists
    # ------------------------------------------------------------------
    a0 = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = _upsample(starter, final_n)
    except Exception:
        a0 = None

    if a0 is None:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    # keep the best solution seen so far
    best_seq = a0.copy()
    conv0 = _conv_self(best_seq)
    best_score = 2.0 * final_n * conv0.max()
    print(f"[0.00s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – warm‑up (exponential gradient)
    # ------------------------------------------------------------------
    stage1_end = min(deadline, start_time + 30.0)   # 30 s warm‑up
    if time.time() < stage1_end:
        a_warm, score_warm = _run_exponential_gradient(
            a0,
            final_n,
            stage1_end,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.5,
            steps_per_alpha=5000,
        )
        if score_warm < best_score:
            best_score = score_warm
            best_seq = a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # dedicate roughly 30 % of the remaining time to this phase
        phase2_time = min(200.0, 0.3 * remaining)
        phase2_end = time.time() + phase2_time
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_end,
            start_time,
            step0=0.3,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, best_max = _run_simulated_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_temp_factor=0.02,
            final_temp_factor=1e-5,
            epsilon_factor=0.05,
            recompute_interval=2000,
        )
        score_ref = 2.0 * final_n * best_max
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] final SA improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] final objective: {best_score:.6f}")
    return [float(x) for x in best_seq]
