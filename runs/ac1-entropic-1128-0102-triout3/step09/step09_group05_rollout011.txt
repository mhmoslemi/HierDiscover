    lo = max(0, K - (n - 1))
    hi = min(n - 1, K)

    # contributions product
    contrib = a[idx_range] * a[K - idx_range]

    It starts from the previously discovered best sequence (if available)
    or from a random start, computes the exact self‑convolution once and
    then repeatedly **moves a small fraction of mass from the indices
    that contribute most to the current maximal convolution entry to
    random destination indices**.

    The convolution is updated *incrementally* (no FFT in the inner loop)
    using the identity

        (a‑δ·e_i+δ·e_j) * (a‑δ·e_i+δ·e_j)
        = a*a − 2δ·shift_i(a) + 2δ·shift_j(a) + δ²·e_{2i}+δ²·e_{2j}−2δ²·e_{i+j}.

    Moves are accepted with a Metropolis rule based on an exponential
    cooling schedule.  Every few ten‑thousands of iterations a full
    recomputation of the convolution guards against numerical drift,
    and occasional global perturbations help escape deep local minima.

    The returned list is a non‑negative sequence with unit sum; the
    evaluation function `evaluate_sequence` is free to re‑normalise it.
    """
    import numpy as np, time, math

    # ------------------------------------------------------------------
    # small helpers
    # ------------------------------------------------------------------
    def next_pow2(x: int) -> int:
        """Smallest power of two ≥ x."""
        return 1 << ((x - 1).bit_length())

    # ------------------------------------------------------------------
    # initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # try to reuse the best sequence produced by the previous run
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = 4096
            a = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            raise ValueError
    except Exception:
        # fallback: random start
        n = 4096
        a = rng.random(n)
        a /= a.sum()

    # exact convolution (only once at the beginning)
    conv = np.convolve(a, a)                     # length 2*n‑1
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score

    print(f"[0.00s] start (n={n}) – score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # annealing parameters
    # ------------------------------------------------------------------
    T0 = 2e-2                # start temperature
    T_end = 1e-10            # final temperature
    max_frac_start = 0.20    # up‑front fraction of source mass that may move
    max_frac_end = 0.02      # final fraction (fine‑grained tweaking)

    it = 0
    last_full_recalc = 0

    # ------------------------------------------------------------------
    # main loop – incremental Monte‑Carlo moves
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # time‑dependent temperature and maximal transfer fraction
        elapsed = time.time() - start_time
        frac_time = elapsed / (deadline - start_time)
        T = T0 * (T_end / T0) ** frac_time
        max_frac = max_frac_start * (1.0 - frac_time) + max_frac_end * frac_time

        # --------------------------------------------------------------
        # 1. locate the peak of the current self‑convolution
        # --------------------------------------------------------------
        K = int(conv.argmax())
        lo = max(0, K - (n - 1))
        hi = min(n - 1, K)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[K - idx_range]
        i_max = idx_range[np.argmax(contrib)]
        j_max = K - i_max

        src = [i_max] if i_max == j_max else [i_max, j_max]

        # --------------------------------------------------------------
        # 2. pick distinct destination indices
        # --------------------------------------------------------------
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # --------------------------------------------------------------
        # 3. decide how much mass to move from each source
        # --------------------------------------------------------------
        src_vals = [a[s] for s in src]
        deltas = []
        for val in src_vals:
            if val <= 0.0:
                deltas.append(0.0)
                continue
            eps = rng.random() * max_frac              # up to max_frac of the current mass
            delta = eps * val
            deltas.append(delta)

        # --------------------------------------------------------------
        # 4. apply the proposal (using the *original* a for the updates)
        # --------------------------------------------------------------
        old_score = cur_score
        for s, d, delta, a_s in zip(src, dst, deltas, src_vals):
            if delta == 0.0:
                continue
            # linear part of the convolution update
            conv[s:s + n] -= 2.0 * delta * a
            conv[d:d + n] += 2.0 * delta * a

            # quadratic correction (δ² terms)
            idx2s = s * 2
            idx2d = d * 2
            idxsd = s + d
            if idx2s < conv.shape[0]:
                conv[idx2s] += delta * delta
            if idx2d < conv.shape[0]:
                conv[idx2d] += delta * delta
            if idxsd < conv.shape[0]:
                conv[idxsd] -= 2.0 * delta * delta

        # mutate the sequence itself
        for s, d, delta in zip(src, dst, deltas):
            if delta == 0.0:
                continue
            a[s] -= delta
            a[d] += delta

        a = np.maximum(a, 0.0)        # tiny numerical clean‑up
        cur_score = 2.0 * n * conv.max()
        delta_score = cur_score - old_score

        # --------------------------------------------------------------
        # 5. acceptance test (Metropolis)
        # --------------------------------------------------------------
        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = math.exp(-delta_score / max(T, 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            it += 1
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] improve – iter={it:,}, score={cur_score:.6f}")
        else:
            # ------------------------------------------------------------------
            # reject – roll back both a and conv (use the restored a for conv)
            # ------------------------------------------------------------------
            for s, d, delta, a_s in zip(src, dst, deltas, src_vals):
                if delta == 0.0:
                    continue
                a[s] += delta
                a[d] -= delta

            # after the a‑vector is back to its original state we can undo conv
            for s, d, delta, a_s in zip(src, dst, deltas, src_vals):
                if delta == 0.0:
                    continue
                conv[s:s + n] += 2.0 * delta * a
                conv[d:d + n] -= 2.0 * delta * a

                idx2s = s * 2
                idx2d = d * 2
                idxsd = s + d
                if idx2s < conv.shape[0]:
                    conv[idx2s] -= delta * delta
                if idx2d < conv.shape[0]:
                    conv[idx2d] -= delta * delta
                if idxsd < conv.shape[0]:
                    conv[idxsd] += 2.0 * delta * delta

            cur_score = old_score

        # ------------------------------------------------------------------
        # occasional full recomputation to keep numerical errors in check
        # ------------------------------------------------------------------
        it += 1
        if it - last_full_recalc >= 20000:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            last_full_recalc = it

        # ------------------------------------------------------------------
        # occasional global perturbation (restart from a noisy copy of the best)
        # ------------------------------------------------------------------
        if it % 250000 == 0 and (deadline - time.time()) > 30.0:
            pert = best_seq + 0.01 * rng.standard_normal(n) * best_seq
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s == 0.0:
                pert = np.full(n, 1.0 / n)
            else:
                pert /= s
            a = pert
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            print(f"[{time.time() - start_time:.2f}s] restart – score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # final normalisation and output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
