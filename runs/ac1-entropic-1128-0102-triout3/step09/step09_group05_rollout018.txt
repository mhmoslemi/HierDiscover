def propose_candidate(seed=0,budget_s=1000,**kwargs):

   np.random.seed(seed)
   start_time=time.time()
   deadline=start_time+budget_s-0.5

   # Determine length
   n=4096

   # Get initial sequence from height_sequence_1 if exists else uniform
   try:
       init_seq=np.asarray(height_sequence_1,dtype=float)
       if init_seq.ndim==1 and init_seq.size>0:
          init_seq = init_seq/np.sum(init_seq)
   except Exception:
       init_seq = np.full(n,1.0/n)

   # If init_seq size != n, interpolate
   if init_seq.size != n:
      init_seq=np.interp(np.linspace(0,1,n), np.linspace(0,1,init_seq.size), init_seq)
   init_seq = np.maximum(init_seq,0.0); init_seq/=init_seq.sum()

   best_seq = init_seq.copy()
   L=_next_pow2(2*n-1)
   conv_best = _conv_self(best_seq, L)
   best_score = _score_from_conv(conv_best, n)

   # Phase 1: quick subgradient for some time (10% budget)
   remaining = deadline - time.time()
   phase1_deadline = time.time() + max(5.0, 0.10 * remaining)
   best_seq, best_score = _run_subgradient_descent(best_seq, n, phase1_deadline, start_time, best_score, max_iters=None)

   # Phase 2: LP mixing iterative improvement
   def lp_improve(f_seq):
        nonlocal best_seq,best_score
        f = f_seq.copy()
        while time.time() < deadline-5.0:
            # compute conv and M
            conv = _conv_self(f, L)
            M = conv.max()
            score = _score_from_conv(conv, n)
            if score < best_score:
                best_score = score
                best_seq = f.copy()
                print(f"[{time.time()-start_time:.2f}s] LP improve: score={score:.6f}")
            # Solve LP
            g = solve_lp(f, M)
            if g is None:
                break
            # Mix
            t = 0.2  # mixing parameter
            f = (1.0 - t) * f + t * g
            f = np.maximum(f,0.0); s=f.sum()
            if s>0: f /= s
            else: f=np.full(n,1.0/n)
        return f

   # apply few LP updates if time permits
   if time.time() < deadline - 10.0:
        best_seq = lp_improve(best_seq)

   # Phase 3: final subgradient and annealing for remaining time
   if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        final_sub_deadline = time.time() + min(30.0, remaining/2.0)
        best_seq, best_score = _run_subgradient_descent(best_seq, n, final_sub_deadline, start_time, best_score, max_iters=None)

   if time.time() < deadline - 5.0:
        # final annealing
        remaining = deadline - time.time()
        steps = int(200_000 * (remaining/ (budget_s - 5)))  # approximate steps
        if steps < 10_000: steps = 10_000

"""
Propose a step‑function sequence that improves the current best bound.
The method combines three ideas:

1️⃣  Start from the best known sequence (or uniform) and run a short
    projected sub‑gradient descent to get a reasonable baseline.

2️⃣  Iteratively solve a linear programme:
        maximise Σg_j  subject to   f * g ≤ M   (component‑wise)
    where ``f`` is the current sequence, ``M`` is its maximal self‑convolution
    value and ``*`` denotes the discrete convolution.
    The LP is linear because ``f`` is fixed; the solution ``g`` tries to
    push mass into positions that do not increase the peak.  We then mix
    ``g`` back into ``f`` (``f←(1‑t)f+t·g``) and repeat while time permits.
    This “LP‑mixing’’ step is the algorithm described in the statement.

3️⃣  Finally polish the result with a short projected sub‑gradient descent
    and a targeted pair‑annealing (the same annealer used previously).

All heavy work (FFT convolution, simplex projection, LP solving) is done
with NumPy / CVXPY and runs well inside the 1000 s budget.
"""

import time
import math
import numpy as np

# optional – only needed for the LP step
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv)  (sum of a is assumed to be 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as the previous code)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the (one) maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + Euclidean projection onto the simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Pair‑annealing (same as before)
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 150_000):
    """Targeted simulated annealing that attacks the two indices
       that contribute most to the current maximal convolution entry."""
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i to the peak (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct destination indices (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction of the source masses
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear programme: maximise Σg  subject to f * g ≤ M
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """Solve  max Σ g_j   s.t. f⊛g ≤ M , g ≥ 0.
       Returns a feasible (and normalised) g, or None on failure."""
    if cp is None:               # pragma: no cover
        return None

    n = f.shape[0]
    g = cp.Variable(n)
    f_param = cp.Parameter(n)
    f_param.value = f

    # linear inequality for each convolution coordinate
    constraints = [g >= 0, cp.conv(f_param, g) <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000, verbose=False)
    except Exception:           # pragma: no cover
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_val = np.asarray(g.value).flatten()
    g_val = np.maximum(g_val, 0.0)
    total = g_val.sum()
    if total > 0.0:
        g_val /= total
    else:
        g_val = np.full_like(g_val, 1.0 / n)
    return g_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence within the allotted budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the classic 4096 entries (good trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the best known construction if available,
    # otherwise fall back to the uniform distribution.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / np.sum(starter)
    except Exception:
        starter = None

    if starter is not None and starter.size != n:
        # interpolate to the required length
        starter = np.interp(np.linspace(0, 1, n),
                            np.linspace(0, 1, starter.size),
                            starter)
    if starter is None:
        starter = np.full(n, 1.0 / n)

    best_seq = starter.copy()
    conv = _conv_self(best_seq, L)
    best_score = _score_from_conv(conv, n)

    # ------------------------------------------------------------------
    # Phase 1 – quick sub‑gradient descent (≈10 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.10 * remaining)

    best_seq, best_score = _run_subgradient_descent(
        best_seq,
        n,
        phase1_deadline,
        start_time,
        best_score,
        max_iters=None,
    )

    # ------------------------------------------------------------------
    # Phase 2 – LP‑mixing iterations.
    # Each iteration solves a modest linear programme, blends the
    # solution back into the current distribution and recomputes the
    # bound.  We keep iterating while there is time left.
    # ------------------------------------------------------------------
    mix_t = 0.2            # mixing coefficient (can be tuned)

    while time.time() < deadline - 5.0:
        conv = _conv_self(best_seq, L)
        M = conv.max()
        cur_score = _score_from_conv(conv, n)

        if cur_score < best_score:
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] LP baseline improve: "
                  f"score={cur_score:.6f}")

        g = _solve_lp_max_sum(best_seq, M)
        if g is None:
            break

        # blend and re‑normalise
        new_seq = (1.0 - mix_t) * best_seq + mix_t * g
        new_seq = np.maximum(new_seq, 0.0)
        s = new_seq.sum()
        if s > 0.0:
            new_seq /= s
        else:
            new_seq = np.full_like(new_seq, 1.0 / n)

        best_seq = new_seq
        # (the loop will recompute the score on the next pass)

    # ------------------------------------------------------------------
    # Phase 3 – a final round of fine‑grained sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        phase3_deadline = time.time() + min(30.0, remaining / 2.0)
        best_seq, best_score = _run_subgradient_descent(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            max_iters=None,
        )

    # ------------------------------------------------------------------
    # Phase 4 – targeted pair‑annealing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        steps = int(180_000 * (deadline - time.time()) / budget_s)
        steps = max(30_000, min(200_000, steps))
        best_seq, best_score = _run_pair_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=steps,
        )

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
