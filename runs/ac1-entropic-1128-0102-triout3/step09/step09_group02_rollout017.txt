def _run_lp_improvement(a0, n, deadline, start_time, t_init=0.2, max_iters=200):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2 * n * np.convolve(a, a).max()  # using full conv but compute only max

    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        # get active indices (peak and perhaps some near peaks)
        # find indices where conv >= M - 1e-6
        active_idxs = np.where(conv >= M - 1e-12)[0]  # small tolerance
        # limit number of constraints to at most, say, 5 largest contributions
        if active_idxs.size > 5:
            # pick top 5 by conv values
            top = np.argpartition(conv, -5)[-5:]
            active_idxs = top
        # Build A matrix
        rows = []
        for k in active_idxs:
            row = np.zeros(n)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            row[idx] = a[k - idx]
            rows.append(row)
        A_ub = np.vstack(rows) if rows else None
        # solve LP
        if A_ub is not None and A_ub.shape[0] > 0:
            # objective: maximize sum(g) -> minimize -sum(g)
            c = -np.ones(n)
            bounds = [(0, None)] * n
            # RHS can be set to M (the current bound)
            b_ub = np.full(A_ub.shape[0], M)
            # Use scipy.optimize.linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                g = res.x
                sum_g = g.sum()
                if sum_g > 0:
                    g = g / sum_g
                    # mixing step
                    a_candidate = (1 - t) * a + t * g
                    # project onto simplex
                    a_candidate = _project_simplex(a_candidate)
                    # evaluate score
                    conv_new = np.convolve(a_candidate, a_candidate)
                    score_new = 2 * n * conv_new.max()
                    if score_new < best_score:
                        best_score = score_new
                        best_a = a_candidate.copy()
                        # improve a
                        a = a_candidate
                        print(f"[{time.time() - start_time:.2f}s] lp improve: iter={it}, score={score_new:.6f}")
                        # keep t maybe increase?
                        continue
        # If not improved or LP fails, reduce t
        t *= 0.5
        if t < 1e-4:
            break
    return best_a, best_score

"""
search for a step‑function that minimises
    2·n·max(self‑convolution) / (sum²)

The optimiser consists of four stages:

1.  Adam optimisation on a smoothed (log‑sum‑exp) proxy of the max‑convolution.
    A modest number of iterations per smoothing parameter is enough to shape a
    good initial distribution.

2.  Exponentiated‑gradient (mirror descent) on the true max‑convolution.
    This rapidly drives the mass away from indices that dominate the peak.

3.  Targeted simulated annealing – a classic “move epsilon from the worst
    contributor to the best destination” move, with a long cooling schedule.

4.  Deterministic greedy reduction – a fast O(n) per‑move routine that repeatedly
    transfers a small amount of probability from the index that contributes most
    to the current peak towards a location that hurts the peak the least.
    It runs for the bulk of the time budget and usually pushes the objective
    below the target 1.503.

All phases respect the global time budget (default 1000 s).  The final
sequence is projected onto the probability simplex, clipped to non‑negative
values and returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # used only for a tiny optional LP step


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length L.
    Returns a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 40000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max via log‑sum‑exp -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (exactly sum‑to‑1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current peak index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2_000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build vector a_k[i] = a[max_idx - i] (zero outside the range)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination index with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        # numerical clean‑up
        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score
