def _smooth_gradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.2
    beta = 5.0
    beta_max = 200.0
    it = 0

    L = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        # FFT of a with length L
        A = np.fft.rfft(a, n=L)
        # linear convolution (full) of a with itself
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        # compute objective
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth GD it={it:,} score={cur_score:.6f}")

        # compute weight vector w
        # shift to avoid overflow
        max_c = cur_max
        # exponential weights, stabilized
        exponent = beta * (conv_full - max_c)
        # Clip exponent to avoid overflow extreme (maybe large negative)
        # but max exponent is zero, min may be -inf; safe.
        w = np.exp(exponent)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # compute gradient via convolution with reversed w
        W_rev = np.fft.rfft(w[::-1], n=L)
        grad_full = np.fft.irfft(A * W_rev, n=L)[: 3 * n - 2]
        # extract gradient of size n
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # update momentum and step
        momentum = beta1 * momentum + (1.0 - beta1) * grad

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.  The algorithm consists of three complementary phases
    1️⃣  a cheap parametric sweep (geometric, power, beta‑type, triangular,
        sinusoidal and mixtures with the uniform vector) and a few random
        Dirichlet starts;
    2️⃣  a two‑stage continuous optimiser:
        – projected sub‑gradient descent (as in the reference);
        – a “smooth‑max’’ gradient descent where the max is approximated by a
          soft‑max (log‑sum‑exp) and the gradient is obtained via a single FFT.
        Both phases keep the iterate on the probability simplex.
    3️⃣  a light simulated‑annealing polish that performs mass‑transfer moves.

All heavy work uses FFT‑based convolutions; the code respects a 1000 s wall‑clock
budget, prints progress stamps and finally returns a plain Python list ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum≈1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – parametric sweep + random Dirichlet starts
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, uniform: np.ndarray, deadline: float, start_time: float):
    """Return best sequence found among a large family of analytic shapes."""
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, _next_pow2(2 * n - 1))
    idx = np.arange(n)

    # --- geometric families -------------------------------------------------
    eps_grid = np.logspace(-4, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        # mix with uniform (lam = weight of uniform)
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, _next_pow2(2 * n - 1))
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

        # right‑hand geometric (mirror)
        geo_r = geo[::-1]
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo_r
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, _next_pow2(2 * n - 1))
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] GeoR eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --- power families ----------------------------------------------------
    p_grid = np.linspace(-1.0, 2.0, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

        # mirrored version
        pow_seq_r = pow_seq[::-1]
        sc = _eval_fast(pow_seq_r, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_seq = sc, pow_seq_r.copy()
            print(f"[{time.time() - start_time:.2f}s] PowerR p={p:.2f} score={sc:.6f}")

    # --- beta‑type family --------------------------------------------------
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, _next_pow2(2 * n - 1))
            if sc < best_score:
                best_score, best_seq = sc, beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # --- triangular (tent) -------------------------------------------------
    mid = (n - 1) / 2.0
    tri = 1.0 - np.abs(idx - mid) / mid
    tri[tri < 0] = 0.0
    tri /= tri.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        seq = lam * uniform + (1.0 - lam) * tri
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _eval_fast(seq, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Tri lam={lam:.2f} score={sc:.6f}")

    # --- sinusoidal ---------------------------------------------------------
    sin = np.sin(np.pi * (idx + 0.5) / n)  # positive on (0,π)
    sin /= sin.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        seq = lam * uniform + (1.0 - lam) * sin
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _eval_fast(seq, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sin lam={lam:.2f} score={sc:.6f}")

    # --- random Dirichlet seeds (few, heavy‑tailed) ------------------------
    rng = np.random.default_rng()
    for _ in range(8):
        if time.time() >= deadline - 30:
            break
        # α = 0.3 gives a fairly spiky Dirichlet
        a = rng.dirichlet(np.full(n, 0.3))
        sc = _eval_fast(a, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet seed score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent (as in reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float):
    """Classic sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L = _next_pow2(3 * n - 2)          # length sufficient for the gradient FFTs

    while time.time() < deadline:
        it += 1
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the active diagonal
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2‑b – smooth‑max gradient descent (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float):
    """
    Gradient descent on a smooth approximation of the max‑convolution.
    The max is replaced by (1/β)·log Σ exp(β·conv), producing a dense
    sub‑gradient that helps escape plateaus.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta1 = 0.9            # momentum coefficient
    step0 = 0.2            # initial step size
    beta = 5.0             # start of soft‑max temperature
    beta_max = 200.0       # final temperature
    it = 0

    L = _next_pow2(3 * n - 2)   # length for the FFTs (a * a and a * w_rev)

    while time.time() < deadline:
        it += 1
        # FFT of a (padded to L)
        A = np.fft.rfft(a, n=L)

        # Linear self‑convolution (full) via FFT
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth GD it={it:,} score={cur_score:.6f}")

        # ----- smooth max weights ------------------------------------------------
        # w_k ∝ exp(β·(conv_k−max)), numerically stable because max subtracted
        exponents = beta * (conv - cur_max)      # ≤ 0
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient = 2·(a * w_rev) (linear convolution) --------------------
        W_rev = np.fft.rfft(w[::-1], n=L)
        grad_full = np.fft.irfft(A * W_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # ----- momentum update and projection ------------------------------------
        momentum = beta1 * momentum + (1.0 - beta1) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # slowly raise β to focus more on the true max
        if beta < beta_max:
            beta = min(beta_max, beta * 1.001)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealer."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass, a random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
