"""
Propose a non‑negative step function (length 4096) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a combination of:
1. Fast parametric sweep (as before) to get a reasonable start.
2. Smoothed gradient descent – the max is approximated by a soft‑max.
   The gradient is obtained with two FFTs per iteration and the iterate
   is projected onto the simplex.
3. A final projected sub‑gradient descent (same scheme as the reference)
   for fine tuning.
4. A short deterministic hill‑climbing polish that moves tiny amounts of
   mass between coordinates only when the true objective improves.

All helpers are top‑level functions; no closures or lambdas are used.
The routine stops well before the 1000 s budget expires and returns a list
compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective for a (sum≈1) sequence
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (identical to the reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (or a small active set)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed objective:
        f(a) = 2 * n * softmax_tau(conv(a,a))
    The soft‑max temperature τ is exponentially decayed.
    The iterate stays on the simplex via Euclidean projection.
    Returns the best sequence found during the run.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    # initial exact score (sum≈1)
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.10          # initial temperature for the soft‑max
    tau_min = 1e-8
    tau_decay = 0.9997   # multiplicative decay per iteration
    step0 = 0.25         # initial step length

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max weights ---------------------------------
        conv = _conv_self(a, L=L_fft)          # length 2n‑1
        maxc = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # sum(w)=1

        # --- gradient via FFT (grad_j = 2 * Σ_i a_i w_{i+j}) ------------------
        w_rev = w[::-1]                         # reversed weights
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # --- projected step ----------------------------------------------------
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # --- exact objective evaluation ----------------------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # --- temperature decay -------------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing polish (mass‑transfer moves, accept only improvements)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny fraction of mass from a source index
    to a destination index, keep the move only if the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # pick proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        # move up to 10 % of the source mass
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score

    L_grad = _next_pow2(3 * n)      # for gradient convolution (a * w_rev)

    # ------------------------------------------------------------------

    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (identical to the reference)
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # geometric family
    eps_grid = np.logspace(-4, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family
    p_grid = np.linspace(-1.0, 2.0, 61)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # Beta‑type family
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # generate a random perturbation of the current best (or uniform)
        base = best_seq if np.random.rand() < 0.7 else uniform
        # relative Gaussian noise, clipped to the simplex
        noise = 0.05 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)  # ≤12 s per restart
        cand_seq, cand_score = _smooth_descent(start, sub_deadline,
                                               start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after smooth restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (bulk of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Deterministic hill‑climb polish (only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
