assistantfinal**Solution Overview**

The search consists of three stages

1. **Warm‑up** – an exponentiated‑gradient on a smoothed version of  
   `max(conv)` (log‑sum‑exp).  This quickly builds a reasonable shape.

2. **Projected sub‑gradient descent** – a long run of exact sub‑gradient
   steps on `max(conv)`.  The step size decays as `1/√t` and we keep the
   best solution seen.

3. **Deterministic greedy refinement** – the bottleneck of the previous
   implementation was the random choice of destination indices.  
   Here we
   * compute the exact sub‑gradient,
   * pick `src = argmax(g)` (the coordinate that hurts the objective the most),
   * try a few `dst = argmin(g)` candidates,
   * analytically determine the largest admissible step `ε` that still
     decreases the current maximal convolution entry,
   * apply the best move.  

   The move is applied by updating the convolution *linearly* (no FFT
   needed) and the full convolution is recomputed only occasionally
   (to avoid drift).  This gives far larger improvements per iteration
   than the tiny random steps of the previous code while remaining
   **O(n)** per iteration.

The algorithm respects the 1000 s wall‑clock budget: the warm‑up uses a
few seconds, the sub‑gradient phase consumes most of the time and the
greedy phase uses all remaining time.  The best sequence found at any
moment is stored and finally returned (normalized to sum = 1).

"""
Improved search for a low‑max self‑convolution step‑function.

The algorithm consists of three stages:

1️⃣  **Exponentiated‑gradient warm‑up** on a smooth (log‑sum‑exp) surrogate
    of the max.  This quickly creates a decent distribution.

2️⃣  **Projected sub‑gradient descent** on the exact ℓ∞‑norm of the
    self‑convolution.  A simple 1/√t schedule is used and the best
    iterate is kept.

3️⃣  **Deterministic greedy refinement** – each iteration picks the
    coordinate with the largest sub‑gradient as a source and several
    low‑gradient coordinates as destinations.  For each (src,dst) pair
    the maximal admissible step ε (the solution of a 1‑D LP) is
    computed analytically, the best pair is applied and the whole
    convolution is updated in O(n).  The move is accepted only if it
    truly lowers the current maximum, guaranteeing monotone improvement.

All phases respect the 1000 s wall‑clock budget and the function returns
the best sequence (a list of non‑negative heights) found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up (smoothed max)
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 3000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10 ** 7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: deterministic greedy refinement (large guaranteed moves)
# ----------------------------------------------------------------------


def _run_greedy_refinement(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.04,
                           eps_end: float = 0.001,
                           recompute_every: int = 5000,
                           k_dest: int = 5):
    """
    Greedy refinement that moves mass from the coordinate with the
    largest sub‑gradient (source) to a few low‑gradient coordinates
    (destinations).  For each (src,dst) pair the maximal admissible step
    ε is obtained analytically; the pair giving the biggest reduction of
    the current maximal convolution entry is applied.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    accepted = 0
    last_recomp = 0

    while time.time() < deadline:
        # linearly decay the fraction of mass we are allowed to move
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        max_idx = int(np.argmax(conv))

        # sub‑gradient of the current max (exact)
        subgrad = _subgradient_full(a, conv)
        src = int(np.argmax(subgrad))

        # candidate destinations – indices with smallest sub‑gradient
        dest_candidates = np.argsort(subgrad)[: k_dest + 5]
        moves = []

        # try each candidate destination
        for dst in dest_candidates:
            if dst == src:
                continue

            # ----- linearised effect on the max bin -----
            s_max = 0.0
            i_src = max_idx - src
            if 0 <= i_src < n:
                s_max -= 2.0 * a[i_src]
            i_dst = max_idx - dst
            if 0 <= i_dst < n:
                s_max += 2.0 * a[i_dst]

            if s_max >= 0.0:
                continue   # would not decrease the current max

            # ----- full slope vector (needed for ε⋆) -----
            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a

            # ----- maximal ε before another entry overtakes the max -----
            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[max_idx] - conv[mask]          # ≥0
                delta_s = s[mask] - s_max                       # >0
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf

            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)
            if epsilon <= 0.0:
                continue

            new_max_est = conv[max_idx] + epsilon * s_max
            moves.append((new_max_est, src, dst, epsilon, s, s_max))

        # If no deterministic move succeeded, fall back to a random pair
        if not moves:
            # random source among indices with non‑negligible mass
            src_candidates = np.where(a > 1e-15)[0]
            if src_candidates.size == 0:
                break
            src = int(rng.choice(src_candidates))
            dst = int(rng.integers(n))
            if dst == src:
                continue

            i_src = max_idx - src
            s_max = -2.0 * a[i_src] if 0 <= i_src < n else 0.0
            i_dst = max_idx - dst
            if 0 <= i_dst < n:
                s_max += 2.0 * a[i_dst]
            if s_max >= 0.0:
                continue

            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a

            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[max_idx] - conv[mask]
                delta_s = s[mask] - s_max
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf

            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)
            if epsilon <= 0.0:
                continue

            new_max_est = conv[max_idx] + epsilon * s_max
            moves.append((new_max_est, src, dst, epsilon, s, s_max))

        # Pick the best move (lowest new max)
        if not moves:
            # nothing could be improved – exit
            break

        moves.sort(key=lambda x: x[0])
        new_max, src, dst, epsilon, s, s_max = moves[0]

        # Apply the move (exact linear update)
        conv += epsilon * s
        a[src] -= epsilon
        a[dst] += epsilon
        accepted += 1

        # Occasionally renormalise to counter floating‑point drift
        if accepted % (recompute_every // 10) == 0:
            total = a.sum()
            if total > 0:
                a /= total

        max_val = conv.max()
        if max_val < best_max - 1e-12:
            best_max = max_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy refine improve: "
                  f"iter={accepted:,}, max={best_max:.8f}")

        # Full recomputation to avoid error accumulation
        if accepted - last_recomp >= recompute_every:
            conv = _conv_self(a)
            max_val = conv.max()
            last_recomp = accepted

    return best_a
