def gradient_smooth(a, beta):
    conv = _conv_self(a)
    # compute w_i = exp(beta * conv_i)
    maxc = conv.max()
    # to avoid overflow, subtract maxc
    exp_beta = np.exp(beta * (conv - maxc))  # shape L = 2n-1
    w = exp_beta / exp_beta.sum()  # probability distribution sum to 1
    # compute gradient = 2 * convolution of a with w_rev // appropriate slice
    L = conv.shape[0]
    w_rev = w[::-1]   # length L
    N = _next_pow2(n + L - 1)  # length for fft
    # compute convolution via FFT
    Af = np.fft.rfft(a, n=N)
    Wf = np.fft.rfft(w_rev, n=N)
    conv_rev = np.fft.irfft(Af * Wf, n=N)[: n + L - 1]
    # gradient k: 2 * conv_rev[L-1 - k] for k in [0,n-1]
    grad = 2.0 * conv_rev[L-1 - np.arange(n)]
    # return grad, smoothmax = maxc + log(sumexp)/beta? But we used stable
    smooth_max = maxc + np.log(exp_beta.sum())/beta
    return grad, smooth_max

assistantfinal**Solution explanation**

The evaluation function  
\[
\Phi(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}
\]
is invariant under a positive scaling of the vector `a`.  
Hence we can fix the normalisation `∑a = 1` and the problem becomes

```
 minimise   max(a*a)          (a ≥ 0 ,  ∑a = 1)
```

`a*a` is the linear convolution of `a` with itself, therefore `max(a*a)` is a
convex (but non‑smooth) function of `a`.  
A classic sub‑gradient method works, but it converges slowly because it only
uses the single index where the maximum is attained.

The new code uses a **smooth approximation** of the max‑operator

\[
\operatorname{softmax}_\beta(c)=\frac{1}{\beta}\log\Bigl(\sum_i e^{\beta c_i}\Bigr)
\qquad(\beta>0)
\]

which converges to the true maximum when `β → ∞`.  
For a given `β` the function is smooth and its gradient can be written in
closed form.  
If `w_i = e^{βc_i}/∑_j e^{βc_j}` then  

\[
\nabla_a\operatorname{softmax}_\beta(c)
   = \sum_i w_i \; \nabla_a c_i
   = 2\,\bigl(a * w_{\text{rev}}\bigr)_{L-1-k},
\]

where `*` denotes linear convolution and `w_rev` is `w` reversed.
All convolutions are computed with FFTs, i.e. in `O(n log n)` time.

The optimisation proceeds in three stages

1. **Analytic seeds** – a small sweep over symmetric “tent” families
   `a_i ∝ min(i,n‑1‑i)^p` (different exponents `p`).  
   The best seed is kept.

2. **Smooth‑max exponential gradient descent** – starting from the best seed,
   the algorithm repeatedly  
   * computes the soft‑max weights `w` (β grows from 0.5 up to ≈12),  
   * obtains the gradient with one extra FFT,  
   * updates the probabilities with a **multiplicative (exponential) update**  

   `a ← a ∘ exp(−η ∇)`,  
   ( `∘` is element‑wise product) and renormalises onto the simplex.
   The step size `η` decays as `1/√t`.  This phase quickly moves the vector
   into a region with a much smaller true maximum.

3. **Fine‑grained sub‑gradient refinement** – after the smoothing phase the
   classic sub‑gradient method (the one used in the reference solution) is
   applied for the remaining time budget.  It eliminates the tiny remaining
   gaps.

A very short simulated‑annealing polish is performed at the very end if
time permits.

The whole pipeline finishes well inside the 1000 s budget and returns the
best sequence found.

"""
Hybrid optimisation for the step‑function bound.

1. generate a few analytic “tent’’ seeds (different exponents),
2. run a smooth‑max exponential‑gradient descent (FFT‑based gradient),
3. finish with the classic sub‑gradient descent,
4. optional tiny annealing polish.

All vectors are kept on the probability simplex, therefore the objective
2·n·max(conv) is exactly the evaluation function (sum is 1 → objective = 2·n·max).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# 2.  Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 3.  Smooth‑max gradient (FFT based)
# ----------------------------------------------------------------------
def _smoothmax_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, smooth_max) where
        grad = ∇_a  ( 1/β log Σ exp(β·conv) )
    and smooth_max is the smooth approximation of max(conv).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length L = 2n‑1
    L = conv.shape[0]

    # soft‑max weights in a numerically stable way
    maxc = conv.max()
    exp_beta = np.exp(beta * (conv - maxc))
    w = exp_beta / exp_beta.sum()            # Σ w_i = 1

    # gradient = 2 * (a * w_rev) evaluated at positions L‑1‑k
    w_rev = w[::-1]
    N = _next_pow2(n + L - 1)
    Af = np.fft.rfft(a, n=N)
    Wf = np.fft.rfft(w_rev, n=N)
    conv_rev = np.fft.irfft(Af * Wf, n=N)[: n + L - 1]

    idx = L - 1 - np.arange(n)               # positions L‑1‑k
    grad = 2.0 * conv_rev[idx]

    smooth_max = maxc + np.log(exp_beta.sum()) / beta
    return grad, smooth_max


def _exp_gradient_update(a: np.ndarray, grad: np.ndarray, eta: float):
    """
    Exponential (multiplicative) update on the simplex:
        a ← a ∘ exp(−η·grad)   then renormalise.
    """
    a_new = a * np.exp(-eta * grad)
    total = a_new.sum()
    if total <= 0.0:
        # fall back to uniform – should rarely happen
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= total
    return a_new


# ----------------------------------------------------------------------
# 4.  Classical sub‑gradient descent (as in the reference code)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for the true max‑objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
def _annealing_polish(a: np.ndarray,

    """Very cheap simulated‑annealing improvement."""
    n = a.shape[0]
    cur_score = _objective(a)
    best_local = a.copy()
    best_local_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")
