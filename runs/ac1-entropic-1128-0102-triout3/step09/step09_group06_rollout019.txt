The evaluation function

```
score(a) = 2·n·max ( a * a )
```

( `*` = linear convolution, `∑ a = 1` after normalisation )
is driven only by the highest entry of the self‑convolution.  
A good strategy is therefore:

1. **Start from a decent known construction** – the variable
   `height_sequence_1` already contains the best sequence found so far.

2. **Optimise a smooth approximation of the `max`** (log‑sum‑exp) with a
   long Adam run.  The approximation is exact when the smoothing
   parameter `α` becomes large, so we increase `α` gradually.

3. **Finetune with a true sub‑gradient step** – the active set of
   indices that attain the maximal convolution value provides a cheap
   sub‑gradient.

4. **Post‑process:** any convex combination with the uniform vector
   (`u_i = 1/n`) can only decrease the worst convolution entry, because
   the uniform distribution spreads the mass evenly.  A tiny line search
   over the mixing weight gives a final boost.

All operations are performed with FFT based convolutions (`O(n log n)`)
which makes thousands of iterations possible within the 1000 s budget.

The complete, self‑contained implementation follows.  It respects the
required signature `propose_candidate(seed: int = 0,
budget_s: int = 1000) -> list[float]` and prints progress messages
so that the runner can see what happens.

```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The algorithm consists of

1. warm start from the previously best construction,
2. long Adam optimisation on a smooth log‑sum‑exp approximation of
   the max‑convolution,
3. a short projected sub‑gradient polishing,
4. a cheap line‑search mixing the result with the uniform vector
   (which almost always reduces the maximal convolution entry).

All heavy work is done with FFT‑based convolutions, which are fast
even for n = 4800.  The routine respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_of(a: np.ndarray, n: int) -> float:
    """Exact score 2·n·max(conv) (sequence is assumed to sum to 1)."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k]).
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_of(a, n)

    # FFT length for the gradient (needs a bit of padding for the correlation)
    fft_len = _next_pow2(3 * n - 2)

    # schedule of α – from very smooth to almost the true max
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0, 32768.0]

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for alpha in alphas:
        # run Adam until the global deadline is hit
        while time.time() < deadline:
            # --- convolution and smooth‑max value --------------------
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # log‑sum‑exp (stable)
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                # smoothed max
            # gradient of f_α :  (2·n) appears later in the score
            # soft‑max weights over the convolution entries
            w = np.exp(scaled - max_scaled)
            w /= w.sum()

            # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            # centre part corresponds to indices 0 … n‑1
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # --- Adam update with projection -----------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)           # simple decay

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 400 == 0:
                cur_score = _score_of(a, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve "
                          f"α={alpha:.0f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient polishing (exact max, not smoothed)
# ----------------------------------------------------------------------


def _subgradient_polish(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       beta: float = 0.94,
                       step0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true objective
    max_k (a*a)[k] (with simple momentum).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_of(a, n)
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve "
                  f"iter={it:,} score={cur_score:.6f}")

        # --- sub‑gradient (active set) ----------------------------
        # all indices that are within 1e‑12 relative of the maximum
        thresh = cur_max * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair (i,j) contributes a_j to ∂/∂a_i
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector (often improves the max)
# ----------------------------------------------------------------------


def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      budget_s: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Search a mixing weight λ∈[0,0.5] for the convex combination
        a' = (1‑λ)·a + λ·u  ,  u = uniform.
    The best mixture (in terms of the true score) is returned.
    """
    u = np.full(n, 1.0 / n)
    best_a = a
    best_score = _score_of(a, n)

    # simple grid search – cheap because each evaluation is O(n log n)
    for lam in np.linspace(0.0, 0.5, 51):
        a_mix = (1.0 - lam) * a + lam * u
        # projection is not needed – the mixture already sums to 1 and stays non‑negative
        cur_score = _score_of(a_mix, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_mix.copy()
            print(f"[{time.time() - start_time:6.2f}s] mix λ={lam:.3f} "
                  f"improved score={cur_score:.6f}")
        # stop early if the time budget is exhausted
        if time.time() - start_time > budget_s * 0.95:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial candidate (warm start)
# ----------------------------------------------------------------------


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Use the historic best sequence if it exists,
    otherwise fall back to the uniform distribution.
    """
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            # Resize by linear interpolation if necessary
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                seq /= s
                return seq
    except Exception:
        pass
    # uniform fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget.
    It returns the best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – the original challenge used 4800 entries
    n = 4800

    # --------------------------------------------------------------
    # 1) warm start
    # --------------------------------------------------------------
    a0 = _initial_candidate(n, rng)

    # --------------------------------------------------------------
    # 2) long Adam optimisation on the smooth max
    # --------------------------------------------------------------
    adam_deadline = start_time + 0.55 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    best_a, best_score = _adam_smooth_max(a0, n, adam_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, "
          f"best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = min(deadline, time.time() + 0.35 * budget_s)
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, best_score = _subgradient_polish(best_a, n,
                                                 subgrad_deadline,
                                                 start_time,
                                                 beta=0.94,
                                                 step0=0.04)
        print(f"[{time.time() - start_time:6.2f}s] sub‑gradient finished, "
              f"best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) final mixing with the uniform vector (very cheap)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start uniform mixing")
        best_a, best_score = _mix_with_uniform(best_a, n,
                                               deadline - time.time(),
                                               start_time)
        print(f"[{time.time() - start_time:6.2f}s] mixing finished, "
              f"best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) final normalisation & evaluation
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    tot = best_a.sum()
    if tot > 0:
        best_a /= tot
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = "
          f"{final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
```
