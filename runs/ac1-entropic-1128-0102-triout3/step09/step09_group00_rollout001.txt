"""
Propose a step‑function (non‑negative heights) that improves the
known bound ≈ 1.5088.

Main ideas
----------
* Use a larger canvas (n = 8192) – the target objective
  2·n·max(conv)/(∑a)² tends to the continuous optimum 1.5 as n grows,
  so a bigger n gives a finer discretisation of the optimal “tent”
  shape.
* Initialise the sequence from a small family of symmetric tent‑type
  shapes a_i ∝ min(i,n‑1‑i)^p (p around 1).  The best exponent is
  selected by a cheap sweep.
* Refine the chosen start with a projected sub‑gradient descent on the
  exact (non‑smooth) objective.  Momentum and a 1/√t step‑size are
  used; after each step the vector is projected onto the probability
  simplex.
* If time permits, a light simulated‑annealing polish is performed
  (tiny random mass transfers).
* The final vector is normalised to sum 1 (the objective is
  scale‑invariant) and returned as a plain Python list.

The whole search respects the 1000 s time budget.  All helper
functions are defined at top‑level so they can be pickled if needed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real to real)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The function is identical to ``evaluate_sequence`` but works on a
    NumPy array and avoids the defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:            # guard against divide‑by‑zero
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto { w ≥ 0, Σ w = 1 }.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent for the non‑smooth objective.
    Returns the best iterate seen before ``deadline`` together with its
    objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum parameters
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5                 # a slightly larger initial step than before
    it = 0

    # main loop
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry (break ties arbitrarily)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional, runs only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets; returns best found."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # a fixed (small) budget of steps – we stop early if time runs out
    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast cooling)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # pick a source index proportional to its mass (more likely to move mass)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass to dst
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 2) Optionally include the global starter if it exists
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Local refinement with projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never triggered in practice)
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:.2f}s] Fallback uniform score={best_score:.6f}")

    remaining = deadline - time.time()
    if remaining > 20.0:                     # reserve a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Very light annealing polish (only if we still have time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return Python list as required by the harness
    return [float(v) for v in best_seq]
