"""
Hierarchical optimisation of a non‑negative step function.

The optimiser works in five stages:

1.  A coarse optimisation on a tiny vector (n≈256) using Adam on a
    smooth log‑sum‑exp approximation of the max‑convolution.
2.  The coarse solution is interpolated to a larger size and refined.
    The same pipeline (Adam → exponentiated‑gradient → targeted
    annealing → greedy peak reduction) is applied at each resolution.
3.  The final resolution (n=4096) receives the bulk of the 1000 s
    budget; its greedy stage is allowed many more iterations than in
    the reference code.
4.  If any time remains, short random‑restart cycles (perturb‑,
    exp‑gradient, anneal) are performed.
5.  The best vector found is normalised to sum to one and returned as
    a plain Python list.  All numbers are non‑negative, so
    ``evaluate_sequence`` will accept the result.

The algorithm is deliberately lightweight (pure NumPy) and respects the
single‑CPU restriction.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length L (≥ 2·len(a)-1).
    Returns the full convolution of length 2·len(a)-1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                               # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp approximation of max(conv).
    Returns the best vector seen and the corresponding objective
    value (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve (α={alpha:.2g}): "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) directly on max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a “source’’ index that contributes
    most to the current peak to a random “destination’’ index.
    Acceptance follows a standard exponential cooling schedule.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current peak information
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]

        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # propose a move
        eps = a[src] * rng.random() * 0.20      # up to 20 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfers a small amount of probability mass from the index
    that contributes most to the current maximal convolution value to an
    index that contributes the least, updating the convolution incrementally.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)          # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 200          # more frequent than the reference code

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually contribute to the current maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build the partner‑vector a_k[i] = a[max_idx‑i] (zero outside the range)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # destination candidate: smallest partner value (but not the source)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # compute a sensible step size
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimisation pipeline for a given size
# ----------------------------------------------------------------------
def _optimise_at_scale(a0: np.ndarray,
                       n: int,
                       stage_deadline: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """
    Run the four phases (Adam → exp‑grad → anneal → greedy) on a vector of
    length ``n`` within ``stage_deadline`` seconds.
    Returns the best vector seen together with its objective
    (2·n·max(conv)).
    """
    # ---- Phase 1 – Adam (≈15 % of the time) ----
    now = time.time()
    phase1_deadline = now + max(4.0, 0.15 * (stage_deadline - now))
    a_adam, score_adam = _run_adam_smoothing(
        a0,
        n,
        phase1_deadline,
        start_time,
        alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0],
        lr0=0.6,
        iters_per_alpha=25000,
    )
    best_seq = a_adam
    best_score = score_adam

    # ---- Phase 2 – Exponentiated gradient (≈25 % of the time) ----
    if time.time() < stage_deadline:
        now = time.time()
        phase2_deadline = now + max(6.0, 0.25 * (stage_deadline - now))
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            eta0=0.6,
        )
        if score_exp < best_score:
            best_seq, best_score = a_exp, score_exp

    # ---- Phase 3 – Annealing (≈30 % of the time) ----
    if time.time() < stage_deadline:
        now = time.time()
        phase3_deadline = now + max(6.0, 0.30 * (stage_deadline - now))
        # Choose number of annealing steps proportionally to allotted time
        steps = int(max(150_000,
                        min(800_000, (phase3_deadline - now) * 2500)))
        a_ann, score_ann = _run_targeted_annealing(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=steps,
        )
        if score_ann < best_score:
            best_seq, best_score = a_ann, score_ann

    # ---- Phase 4 – Deterministic greedy reduction (≈30 % of the time) ----
    if time.time() < stage_deadline:
        now = time.time()
        phase4_deadline = stage_deadline
        # allocate as many iterations as the remaining time permits
        est_iters = int(max(200_000,
                            min(5_000_000, (phase4_deadline - now) * 2500)))
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=est_iters,
        )
        if score_greedy < best_score:
            best_seq, best_score = a_greedy, score_greedy

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Hierarchical optimiser for the step‑function problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total wall‑clock budget in seconds.

    Returns
    -------
    list[float]
        The best sequence of non‑negative heights found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a safety margin

    # ------------------------------------------------------------------
    # We progressively increase the problem size.
    # Early stages are cheap and give a good global shape.
    # ------------------------------------------------------------------
    sizes = [256, 512, 1024, 2048, 4096]
    best_seq = None
    best_score = float("inf")
    prev_seq = None

    for i, n in enumerate(sizes):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # allocate a modest slice for the early sizes, reserving most time for the final one
        if i == len(sizes) - 1:
            alloc = remaining * 0.60
        else:
            alloc = min(30.0, remaining * 0.08)

        stage_deadline = time.time() + alloc

        if prev_seq is None:
            a0 = np.random.rand(n)
        else:
            # linear interpolation to the next resolution
            a0 = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, prev_seq.size),
                           prev_seq)

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        a_opt, score_opt = _optimise_at_scale(a0, n, stage_deadline, start_time)

        # compute the true objective (the same expression used inside the phases)
        conv = _conv_self(a_opt, _next_pow2(2 * n - 1))
        cur_score = 2.0 * n * conv.max()
        print(f"[{time.time() - start_time:.2f}s] Size {n:<5d} – best score {cur_score:.6f}")

        if cur_score < best_score:
            best_score, best_seq = cur_score, a_opt.copy()
        prev_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Final fine‑tuning on the full size (if we have spare time)
    # ------------------------------------------------------------------
    if best_seq is not None and (time.time() < deadline - 5.0):
        a_refined, score_refined = _optimise_at_scale(best_seq,
                                                      best_seq.size,
                                                      deadline,
                                                      start_time)
        conv = _conv_self(a_refined, _next_pow2(2 * best_seq.size - 1))
        cur_score = 2.0 * best_seq.size * conv.max()
        if cur_score < best_score:
            best_seq, best_score = a_refined.copy(), cur_score
            print(f"[{time.time() - start_time:.2f}s] Final refinement – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
