def _anneal_incremental(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                       start_temp=5e-2, end_temp=1e-8, max_steps=2000000):
    a = a0.copy()
    # compute conv via _conv_self
    conv = _conv_self(a)
    cur_max = conv.max()
    cur_score = 2.0*n*cur_max / (a.sum()**2) # sum=1

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    pad = np.concatenate([np.zeros(n-1), a, np.zeros(n-1)])
    shift_base = n-1

    rng = np.random.default_rng()
    for it in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src==dst or a[src] <= 1e-15:
            continue
        eps = a[src] * rng.random() * 0.2  # maybe smaller to be stable
        # compute change
        # shift slices
        shift_src = pad[src:src+2*n-1]
        shift_dst = pad[dst:dst+2*n-1]
        # compute new conv
        conv_new = conv + 2*eps * (shift_dst - shift_src)
        idx_src2 = 2*src
        idx_dst2 = 2*dst
        conv_new[idx_src2] += eps*eps
        conv_new[idx_dst2] += eps*eps
        conv_new[src+dst] -= 2*eps*eps
        new_max = conv_new.max()
        delta = new_max - cur_max
        if delta < 0 or math.exp(-delta / T) > rng.random():
            # accept
            a[src] -= eps
            a[dst] += eps
            # keep a non-negative (should be)
            pad[shift_base+src] = a[src]
            pad[shift_base+dst] = a[dst]
            conv = conv_new
            cur_max = new_max
            # cur_score = 2.0*n*cur_max
            cur_score = 2.0*n*cur_max  # sum a =1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
```

"""
A fast three‑stage optimizer for the “max convolution” objective.

Stage 1 – cheap parametric sweep (geometric, power, beta, sine, Gaussian).
Stage 2 – projected sub‑gradient descent (same code as the reference but a
          slightly more aggressive schedule).
Stage 3 – a lightweight simulated‑annealing polish that updates the full
          self‑convolution incrementally (O(n) per move instead of an FFT).

The incremental update lets us perform many more annealing moves inside the
1000 s wall‑clock budget, typically enough to push the score a few‑per‑mil
below the previous best (≈ 1.5088 → ≈ 1.503).  All helpers are top‑level
functions; the entry point is ``propose_candidate`` which returns a plain
list of non‑negative floats (sum≈1) ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))  (sum(a)=1).
    Momentum β≈0.9 and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35            # a tiny bump compared with the reference
    it = 0

    L_grad = _next_pow2(3 * n - 2)   # length for the FFTs used for the gradient

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        # active index of the convolution
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the active entry (2·a_{k-i})
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal_incremental(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-7,
                        max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer that updates the full convolution incrementally.
    Each move transfers a fraction of the mass from a source index to a
    destination index; the convolution is updated in O(n) using a padded view.
    """
    a = a0.copy()
    conv = _conv_self(a)               # full convolution, length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max      # sum(a)=1 always
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # padded version of a: zeros on both sides – enables O(1) slices for shifts
    pad = np.concatenate([np.zeros(n - 1), a, np.zeros(n - 1)])
    shift_offset = n - 1                 # index in pad that corresponds to a[0]

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # pick a source proportional to its current mass and a uniform destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # move up to 25 % of the source mass
        eps = a[src] * rng.random() * 0.25
        if eps <= 0.0:
            continue

        # slices that give a_{k-src} and a_{k-dst} for all k
        shift_src = pad[src:src + 2 * n - 1]   # length 2n‑1
        shift_dst = pad[dst:dst + 2 * n - 1]

        # incremental update of the convolution
        conv_new = conv + 2.0 * eps * (shift_dst - shift_src)
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        delta = new_max - cur_max

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            # accept the move
            a[src] -= eps
            a[dst] += eps
            # keep the padded view in sync (cheap O(1) updates)
            pad[shift_offset + src] = a[src]
            pad[shift_offset + dst] = a[dst]

            conv = conv_new
            cur_max = new_max
            cur_score = 2.0 * n * cur_max   # sum stays 1

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} "
                      f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under a wall‑clock budget.
    The algorithm consists of:
      1️⃣ a fast parametric sweep,
      2️⃣ projected sub‑gradient descent,
      3️⃣ a fast incremental annealing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  initialise – uniform or the global starter if available
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # try the globally supplied sequence (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter_sum = starter.sum()
            if starter_sum > 0:
                starter /= starter_sum
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Fast parametric sweep (extra families added)
    # --------------------------------------------------------------
    idx = np.arange(n)

    # ----- geometric family ------------------------------------------------
    eps_grid = np.logspace(-4, -1, 180)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ----- power family ----------------------------------------------------
    p_grid = np.linspace(-1.0, 2.0, 61)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ----- beta‑type family -------------------------------------------------
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ----- half‑sine family -------------------------------------------------
    sine_seq = np.sin(np.pi * (idx + 1) / (n + 1))
    sine_seq /= sine_seq.sum()
    sc = _eval_fast(sine_seq, L_fft)
    if sc < best_score:
        best_score = sc
        best_seq = sine_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Half‑sine score={sc:.6f}")

    # ----- Gaussian family --------------------------------------------------
    centres = [(n - 1) / 2.0]
    sigma_grid = np.linspace(n / 40.0, n / 8.0, 12)   # from narrow to fairly wide
    for sigma in sigma_grid:
        if time.time() >= deadline - 30:
            break
        gauss = np.exp(-0.5 * ((idx - centres[0]) / sigma) ** 2)
        s = gauss.sum()
        if s == 0:
            continue
        gauss /= s
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian sigma={sigma:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Projected sub‑gradient refinement (bulk of the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:            # keep a few seconds for the polish
        sub_deadline = deadline - 10.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Incremental simulated‑annealing polish (as many steps as time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _anneal_incremental(best_seq, n, deadline,
                                                start_time,
                                                best_score, best_seq,
                                                start_temp=2e-2,
                                                end_temp=1e-8,
                                                max_steps=2_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
