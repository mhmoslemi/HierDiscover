def _adam_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    base_lr = 0.05

    L_conv = _next_pow2(3*n-2)
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _conv_self(a, L=L_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum()**2) # sum is 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={cur_score:.6f}")

        # get active index k
        k = int(conv.argmax())
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k-idx]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size schedule: decaying over time
        lr = base_lr / np.sqrt(t) # optional
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)

"""
Propose a step function using a multi‑stage search:

1️⃣  Fast parametric families (geometric, power, beta) and a
    double‑exponential mixture explored by random sampling.
2️⃣  Projected Adam (momentum‑based) descent on the exact
    nonsmooth objective, always projecting onto the probability simplex.
3️⃣  Light mass‑transfer simulated‑annealing polish.

All stages respect a 1000 s wall‑clock budget and return a plain Python
list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution  # optional, may be unused

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum sequences → 2·n·max(conv) )
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Return the exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _geom_mix(eps: float, lam: float, n: int, idx: np.ndarray) -> np.ndarray:
    """lam·uniform + (1‑lam)·geometric with ratio r=1‑eps."""
    r = 1.0 - eps
    geo = np.power(r, idx, dtype=np.float64)
    geo_sum = geo.sum()
    if geo_sum == 0.0:
        return None
    geo /= geo_sum
    uniform = np.full(n, 1.0 / n)
    seq = lam * uniform + (1.0 - lam) * geo
    seq = np.maximum(seq, 0.0)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        return None
    seq /= seq_sum
    return seq


def _power_seq(p: float, n: int, idx: np.ndarray) -> np.ndarray:
    """a_i ∝ (i+1)^p."""
    seq = np.power(idx + 1, p, dtype=np.float64)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _beta_seq(p: float, q: float, n: int, idx: np.ndarray) -> np.ndarray:
    """Beta‑type analogue: (i+1)^p·(n‑i)^q."""
    seq = (np.power(idx + 1, p) *
           np.power(n - idx, q))
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


def _double_exp_seq(alpha: float, beta: float, w_fwd: float,
                    w_rev: float, n: int, idx: np.ndarray) -> np.ndarray:
    """
    a = w_fwd·exp(-α·i) + w_rev·exp(-β·(n‑1‑i)) + (1‑w_fwd‑w_rev)·uniform.
    Returned array is normalised to sum‑1.
    """
    if w_fwd + w_rev > 1.0:
        return None
    w_uni = 1.0 - w_fwd - w_rev
    fwd = np.exp(-alpha * idx, dtype=np.float64)
    rev = np.exp(-beta * (n - 1 - idx), dtype=np.float64)
    seq = w_fwd * fwd + w_rev * rev + w_uni
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        return None
    seq /= s
    return seq


# ----------------------------------------------------------------------
# Adam‑style projected descent on the exact objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on f(a)=2·n·max(conv(a,a)).
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.15          # a moderate initial learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    Lg = _next_pow2(3 * n - 2)   # length for gradient‑FFT

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=Lg)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # active lag
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a step size that slowly decays (∝1/√t)
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 400_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)                     # biased by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.45            # up‑to 45 % transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
