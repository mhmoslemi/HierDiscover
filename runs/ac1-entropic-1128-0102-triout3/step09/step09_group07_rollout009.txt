"""
Candidate generator for minimizing the evaluate_sequence objective.

The search proceeds in three phases, each respecting the global time
budget (≈ 1000 s):

1️⃣  A very short exponentiated‑gradient warm‑up on a smooth approximation
    of the max‑convolution.  This quickly moves the distribution away from
    pathological corners.

2️⃣  Projected sub‑gradient descent with momentum on the exact ℓ∞‑norm of
    the self‑convolution.  The iterate is always projected onto the
    probability simplex (non‑negative, sum = 1).

3️⃣  Greedy “mass‑shift’’ refinement.  At every step we:
    – compute the exact sub‑gradient of the current maximum,
    – pick a source index with a large gradient (contributing heavily to the
      maximal bin) and a destination with a small gradient,
    – compute analytically the largest admissible ε that keeps the ℓ∞‑norm
      non‑increasing (a 1‑D linear programme),
    – apply the move in O(n) time.
    When no improving move can be found for a while we inject a tiny random
    jitter to escape plateaus.  The refinement runs until the time budget
    expires and returns the best distribution seen.

All numbers are non‑negative and the final sequence is normalised to sum = 1.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up (very short)
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    """A lightweight EG warm‑up; runs only a few seconds."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG improve: iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)          # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10 ** 7):
    """
    Projected sub‑gradient descent on the exact ℓ∞‑norm of the
    self‑convolution.  Momentum (`beta`) speeds up convergence.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: greedy large‑step refinement
# ----------------------------------------------------------------------


def _run_greedy_refine(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       max_idle: int = 250,
                       jitter_frac: float = 0.02):
    """
    Greedy refinement that repeatedly shifts mass from a high‑gradient
    index to a low‑gradient index.  The step size ε is the largest value
    that keeps the ℓ∞‑norm non‑increasing (solving a 1‑D LP).  When no
    improving move can be found for ``max_idle`` iterations we add a tiny
    random jitter to escape a plateau.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    idle = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # compute sub‑gradient to guide source/destination choice
        g = _subgradient_full(a, conv)
        # source ≈ largest gradient, destination ≈ smallest
        src_candidates = np.argpartition(-g, range(5))[:5]    # 5 largest
        dst_candidates = np.argpartition(g, range(5))[:5]    # 5 smallest

        k_max = int(np.argmax(conv))

        best_reduction = 0.0
        best_move = None          # (src, dst, epsilon, s)

        for src in src_candidates:
            if a[src] < 1e-15:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue

                # s_max for the current maximal bin
                a_ks = a[k_max - src] if (0 <= k_max - src < n) else 0.0
                a_kd = a[k_max - dst] if (0 <= k_max - dst < n) else 0.0
                s_max = -2.0 * a_ks + 2.0 * a_kd
                if s_max >= 0.0:
                    # this particular move would not reduce the current max
                    continue

                # build the full slope vector (O(n))
                s = np.zeros(2 * n - 1, dtype=float)
                s[src:src + n] -= 2.0 * a
                s[dst:dst + n] += 2.0 * a

                # maximal admissible ε before another entry overtakes the max
                mask = s > s_max + 1e-18
                if mask.any():
                    delta_conv = conv[k_max] - conv[mask]
                    delta_s = s[mask] - s_max
                    eps_candidates = delta_conv / delta_s
                    eps_candidates = eps_candidates[eps_candidates > 1e-15]
                    eps_star = eps_candidates.min() if eps_candidates.size else np.inf
                else:
                    eps_star = np.inf

                # we also cannot move more mass than is present at src
                eps_limit = a[src] * 0.15        # move up to 15 % of src mass
                epsilon = min(eps_star, eps_limit)
                if epsilon <= 0.0:
                    continue

                reduction = -epsilon * s_max      # guaranteed improvement in the max bin
                if reduction > best_reduction + 1e-14:
                    best_reduction = reduction
                    best_move = (src, dst, epsilon, s)

        if best_move is not None:
            src, dst, epsilon, s = best_move
            a[src] -= epsilon
            a[dst] += epsilon
            conv = conv + epsilon * s
            max_val = conv.max()
            idle = 0
            if max_val < best_max - 1e-14:
                best_max = max_val
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: iter={it:,}, max={best_max:.8f}")
        else:
            # no improving pair found – this is an idle step
            idle += 1
            if idle >= max_idle:
                # inject a tiny random perturbation to escape the plateau
                src = rng.integers(0, n)
                dst = rng.integers(0, n - 1)
                if src == dst or a[src] < 1e-12:
                    continue
                eps = a[src] * jitter_frac * rng.random()
                if eps <= 0.0:
                    continue
                s = np.zeros(2 * n - 1, dtype=float)
                s[src:src + n] -= 2.0 * a
                s[dst:dst + n] += 2.0 * a
                a[src] -= eps
                a[dst] += eps
                conv = conv + eps * s
                max_val = conv.max()
                idle = 0
                # occasional full recomputation to keep numerical drift in check
                if it % 1500 == 0:
                    conv = _conv_self(a)
                    max_val = conv.max()

        # keep a numerically clean copy every few thousand moves
        if it % 3000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee a return before the external timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – use the previously known construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # a slightly larger length gives the optimisation more degrees of freedom
    final_n = 8000
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – short exponentiated‑gradient warm‑up
    # ------------------------------------------------------------------
    stage1_deadline = time.time() + min(5.0, 0.05 * (deadline - time.time()))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (major chunk of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve at least 30 s for the final refinement
        stage2_deadline = time.time() + max(30.0, 0.70 * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.30,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – greedy large‑step refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_greedy_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_idle=300,
            jitter_frac=0.02,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
