def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096

    # define candidate init functions:
    def uniform():
        x=np.ones(n)
        return x/x.sum()

    def arcsine():
        # avoid zeros: i from 0..n-1, shift by 0.5
        i = np.arange(n, dtype=float)
        w = 1.0/np.sqrt((i+0.5)*(n-0.5-i))
        w[np.isnan(w)] = 0.0
        w = np.maximum(w,0)
        return w / w.sum()

    def sin_shape():
        i = np.arange(n)
        w = np.sin(np.pi * (i+0.5) / n)   # zero at edges, peak centre
        w = np.maximum(w,0)
        return w / w.sum()

    def triangular():
        i = np.arange(n)
        w = np.minimum(i+1, n-i)
        return w / w.sum()
    # maybe exponential
    def exp_decay():
        i = np.arange(n)
        lam = 5.0/n
        w = np.exp(-lam*i)
        return w / w.sum()

    candidates = [uniform(), arcsine(), sin_shape(), triangular(), exp_decay()]
    # also random
    candidates.append(np.random.rand(n))
    candidates[-1] /= candidates[-1].sum()

    best_seq = None
    best_score = float('inf')
    for a in candidates:
        a = np.maximum(a,0.0)
        a /= a.sum()
        score = _objective(a, n)  # compute 2*n*max(conv)/sum^2
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"init candidate score={score:.6f}")

    # Phase 1: Adam with smoothing
    # We'll set alphas list decreasing: start high to approximate max, gradually lower.
    alphas = [80, 40, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02]
    phase1_deadline = min(deadline, time.time() + max(30.0, 0.12*(deadline - time.time())))
    best_seq, best_score = _run_adam_smoothing(best_seq, n, phase1_deadline, start, alphas,
                                                lr0=0.6, iters_per_alpha=40000)
    # Phase2: Exponential gradient
    phase2_deadline = min(deadline, time.time() + max(30.0, 0.25*(deadline - time.time())))
    best_seq, score2 = _run_exp_gradient(best_seq, n, phase2_deadline, start,
                                         eta0=0.6)
    if score2 < best_score:
        best_score = score2
        best_seq = best_seq.copy()

    # Phase3: Targeted annealing
    if time.time() < deadline:
        phase3_deadline = min(deadline, time.time() + max(30.0, 0.30*(deadline - time.time())))
        best_seq, score3 = _run_targeted_annealing(best_seq, n, phase3_deadline, start,
                                                   best_score, best_seq,
                                                   start_temp=5e-2, end_temp=1e-9,
                                                   steps=500_000)
        if score3 < best_score:
            best_score = score3
            best_seq = best_seq.copy()
    # Phase4: greedy
    if time.time() < deadline:
        phase4_deadline = deadline
        best_seq, score4 = _run_greedy_reduction(best_seq, n, phase4_deadline, start,
                                                 best_score, best_seq,
                                                 max_iters=2_000_000)
        if score4 < best_score:
            best_score = score4
            best_seq = best_seq.copy()

    # Random restarts final: small perturbations
    while time.time() < deadline - 5.0:
        # small random perturbation based on best_seq
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.maximum(pert, 0)
        pert /= pert.sum()
        # short exp gradient
        sub_deadline = time.time() + 10.0
        pert, score pert = _run_exp_gradient(pert, n, sub_deadline, start, eta0=0.6)
        if score pert < best_score:
            best_score = score pert
            best_seq = pert.copy()
            print(f"restart improvement score={best_score:.6f}")

    # final normalisation and output list
    best_seq = np.maximum(best_seq,0.0)
    total = best_seq.sum()
    if total>0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)

    return [float(x) for x in best_seq]

"""
propose_candidate – fast multi‑phase optimisation for the max‑self‑convolution
================================================================================
We want a non‑negative vector `a` (length n) that minimises

        score(a) = 2·n·max(conv(a,a)) / (sum(a)²)

(the same value returned by ``evaluate_sequence``).  The objective is
homogeneous, so we keep the vector on the probability simplex
(`sum(a)=1`) and the score simplifies to `2·n·max(conv)`.

The optimiser works in five stages:

1. **Initial seeds** – several analytically‑motivated shapes (uniform,
   arcsine‑like, sinusoidal, triangular, exponential, Gaussian) and a
   random seed are generated and the best one is kept.

2. **Adam on a smooth log‑sum‑exp surrogate** – for a list of smoothing
   parameters the (differentiable) soft‑max of the convolution is
   minimised with Adam; this quickly discovers a good global shape.

3. **Exponentiated‑gradient descent on the true max** – a mirror‑descent
   step using the sub‑gradient at the current maximal convolution index,
   projected onto the simplex.

4. **Targeted simulated annealing** – a Metropolis walk that moves
   mass from the index contributing most to the current peak to a
   randomly‑chosen destination; the temperature is exponentially
   cooled.  The convolution is updated incrementally for speed.

5. **Deterministic greedy reduction** – each iteration reduces the
   current maximal entry by moving a tiny amount of mass from the
   “worst” source index towards a “best” destination index; the full
   convolution is recomputed only periodically.

If time remains, short random‑restart cycles (perturb → exp‑grad →
anneal) are performed.  The routine respects the 1000 s budget,
printing progress and finally returning the best sequence as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length L.
    Returns a vector of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 40000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max via log‑sum‑exp -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smoothed max (see derivation) -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + w_rev.size - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient at the active index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # mirror‑descent multiplicative update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # project back onto simplex (exact normalisation)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing (incremental updates)
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current peak
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]

        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # propose move – a fraction of src’s mass
        eps = a[src] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, Lfft)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)  # full convolution (length 2·n‑1)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually contribute to the current peak
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and dst in the current peak
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # destination: index whose partner in this peak is minimal
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-30)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        # each shift gets +2·eps·a_old[partner] from the moved mass
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation (removes accumulated drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Helper – generate a handful of analytically motivated seeds
# ----------------------------------------------------------------------
def _seed_candidates(n: int) -> list[np.ndarray]:
    """Return a list of initial probability vectors."""
    i = np.arange(n, dtype=float)

    # 1. uniform
    uniform = np.ones(n)

    # 2. arcsine‑like (large weight near the borders)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = np.maximum(arcsine, 0.0)

    # 3. half‑sin (zero at the borders, peak in the middle)
    sin_shape = np.sin(np.pi * (i + 0.5) / n)

    # 4. triangular (ramp up then down)
    triangular = np.minimum(i + 1, n - i)

    # 5. exponential decay from the left
    lam = 5.0 / n
    exp_decay = np.exp(-lam * i)

    # 6. Gaussian centred, several bandwidths
    gauss_candidates = []
    for sigma_rel in (0.03, 0.06, 0.12):
        sigma = sigma_rel * n
        centre = (n - 1) / 2.0
        g = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        gauss_candidates.append(g)

    # 7. random start (re‑normalised)
    rng = np.random.default_rng()
    random_vec = rng.random(n)

    candidates = [uniform, arcsine, sin_shape,
                  triangular, exp_decay] + gauss_candidates + [random_vec]

    # Normalise all to the simplex
    out = []
    for v in candidates:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v = v / s
        out.append(v)
    return out

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found within the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096                     # length used in the reference run

    # ------------------------------------------------------------------
    # 1. Choose a good starting point from a set of analytical seeds
    # ------------------------------------------------------------------
    candidates = _seed_candidates(n)
    best_seq = None
    best_score = float("inf")
    for idx, a in enumerate(candidates):
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] seed {idx} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Phase‑1 – Adam on a smoothed max (≈12 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline,
                          time.time() + max(12.0,
                                            0.12 * (deadline - time.time())))
    alphas = [80, 40, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05]
    best_seq, score1 = _run_adam_smoothing(
        best_seq,
        n,
        phase1_deadline,
        start_time,
        alphas,
        lr0=0.6,
        iters_per_alpha=40000,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = best_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
              f"score={score1:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase‑2 – Exponentiated‑gradient descent (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = min(deadline,
                              time.time() + max(30.0,
                                                0.30 * (deadline - time.time())))
        best_seq, score2 = _run_exp_gradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            eta0=0.6,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = best_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score2:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase‑3 – Targeted annealing (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = min(deadline,
                              time.time() + max(30.0,
                                                0.30 * (deadline - time.time())))
        best_seq, score3 = _run_targeted_annealing(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=500_000,
        )
        if score3 < best_score:
            best_score = score3
            best_seq = best_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: "
                  f"score={score3:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase‑4 – Deterministic greedy reduction (≈20 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase4_deadline = deadline
        best_seq, score4 = _run_greedy_reduction(
            best_seq,
            n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=2_000_000,
        )
        if score4 < best_score:
            best_score = score4
            best_seq = best_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 4 improve: "
                  f"score={score4:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional short random‑restart cycles with the remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # a brief perturbation of the current best solution
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # a quick exp‑gradient refinement (≈8 s)
        sub_deadline = time.time() + 8.0
        pert, pert_score = _run_exp_gradient(
            pert,
            n,
            sub_deadline,
            start_time,
            eta0=0.6,
        )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: "
                  f"score={pert_score:.6f}")

        # a very short annealing to polish the perturbation
        if time.time() < deadline:
            sub_deadline = min(deadline,
                               time.time() + 12.0)
            pert, pert_score = _run_targeted_annealing(
                pert,
                n,
                sub_deadline,
                start_time,
                best_score,
                pert,
                start_temp=2e-2,
                end_temp=1e-9,
                steps=80_000,
            )
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal‑restart improve: "
                      f"score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (ensure the simplex) and return as a list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The competition harness will call ``evaluate_sequence`` on the list.
    return [float(x) for x in best_seq]
