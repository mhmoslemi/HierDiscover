"""
Propose a step‑function (non‑negative vector on the simplex) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The routine works under a 1000 s budget and consists of three stages:

1. **Optional LP balancing** – if ``cvxpy`` is available we perform a few cheap
   linear‑programming updates of the form
   maximise Σb  subject to  b≥0  and  a*b ≤ M, where M is the current peak.
   This quickly reduces the objective from a naive start.

2. **Smoothed‑max gradient descent** – we minimise a smooth approximation of
   the peak using a soft‑max (log‑sum‑exp) with a gradually increasing
   smoothing parameter λ.  The gradient is computed via FFT‑based
   convolutions and the iterate is projected onto the probability simplex.
   This stage drives the solution close to the optimum.

3. **Random local search** – a greedy hill‑climbing phase that repeatedly
   moves a small amount of mass between two random coordinates and keeps
   the change only if the objective improves.  The convolution is updated
   with a full FFT each trial (still cheap for n≈2000).

The best sequence found is returned as a plain ``list[float]``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    Xf = np.fft.rfft(x, n=fft_len)
    Yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Xf * Yf, n=fft_len)[: n + m - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns the soft‑max weights w_k = exp(lam*conv_k) / Σ exp(lam*conv).
    The calculation is shifted by max(conv) for numerical stability.
    """
    max_c = conv.max()
    # shift to avoid overflow; values are tiny anyway, but we keep it robust
    shifted = lam * (conv - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _grad_softmax(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum (log‑sum‑exp) w.r.t. a.
    It is 2·(a * w) truncated to the first |a| entries, where w are the
    soft‑max weights of conv.
    """
    w = _softmax_weights(conv, lam)                 # length 2·n‑1
    # convolution a * w (full) → take first n entries for the gradient
    full = _conv_fft(a, w)                          # length 3·n‑2
    grad = 2.0 * full[: a.shape[0]]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a*a) for a already normalised (sum=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows three stages
    (optional LP, smoothed‑max gradient descent, random greedy search)
    and respects the supplied time budget (seconds).  Returns a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the caller
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. Choose problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=float).ravel()
        if init.size == 0:
            init = np.full(2048, 1.0 / 2048)
    else:
        init = np.full(2048, 1.0 / 2048)

    n = init.size
    a = init.copy()
    # Normalise onto the simplex (sum = 1)
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start_time:6.2f}s] initial n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Optional LP balancing (quick few iterations, if cvxpy present)
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp

        # we will spend at most ~10 % of the remaining time on this stage
        lp_deadline = time.time() + 0.1 * (deadline - time.time())
        lp_iter = 0
        while time.time() < lp_deadline and lp_iter < 20:
            lp_iter += 1
            conv_a = _conv_self(a)
            M = conv_a.max()

            # Build convolution matrix T (shape (2n‑1, n))
            T = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                T[i:i + n, i] = a

            # Linear programme: maximise Σb  s.t. b≥0, T·b ≤ M
            b = cp.Variable(n, nonneg=True)
            constraints = [T @ b <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200)

            if b.value is None:
                break
            b_vec = np.maximum(b.value, 0.0)
            sum_b = b_vec.sum()
            if sum_b <= 0:
                break
            g = b_vec / sum_b

            # try a few convex combinations and keep the best
            best_local = a
            best_local_score = best_score
            for t in (0.2, 0.5, 0.8, 1.0):
                cand = (1.0 - t) * a + t * g
                cand = _project_simplex(cand)
                sc = _objective(cand)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = cand
            a = best_local

            if best_local_score < best_score - 1e-9:
                best_score = best_local_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP iter {lp_iter:02d} "
                      f"improve → {best_score:.6f}")

    except Exception:  # pragma: no cover
        # cvxpy missing or failed – just skip the LP stage
        pass

    # ------------------------------------------------------------------
    # 3. Smoothed‑max gradient descent (main optimisation engine)
    # ------------------------------------------------------------------
    grad_deadline = deadline - 15.0          # keep a few seconds for the final stage
    it = 0
    lam = 5.0                                 # start with a modest smoothing
    while time.time() < grad_deadline:
        it += 1
        conv = _conv_self(a)
        # soft‑max gradient
        grad = _grad_softmax(a, conv, lam)

        # step‑size schedule (decreasing)
        step = 0.12 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # occasionally tighten λ (make the smooth approximation closer to the true max)
        if it % 500 == 0:
            lam = min(lam * 1.15, 200.0)

        # evaluate
        cur_score = _objective(a)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad iter {it:<6d} "
                  f"λ={lam:5.2f} improve → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Random greedy local search (tiny perturbations)
    # ------------------------------------------------------------------
    local_deadline = deadline
    it_local = 0
    while time.time() < local_deadline:
        it_local += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to move (up to 30 % of the source entry)
        max_move = a[i] * 0.3
        if max_move <= 1e-12:
            continue
        d = np.random.rand() * max_move
        if d <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= d
        cand[j] += d
        # ensure numeric safety
        if cand[i] < 0:
            continue
        cand = _project_simplex(cand)       # re‑project to avoid drift

        sc = _objective(cand)
        if sc < best_score - 1e-9:
            best_score = sc
            best_seq = cand.copy()
            a = cand                         # continue from the improved point
            print(f"[{time.time() - start_time:6.2f}s] local move {it_local:<6d} "
                  f"improve → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final normalisation & return
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
