"""
step‑function optimiser
=======================

A three‑stage optimiser for the ``evaluate_sequence`` objective.

1. **Exploratory shape scan** – generate a variety of parametric
   probability vectors (power‑law, symmetric Gaussian, exponential,
   two‑edge blocks).  Each candidate is normalised and evaluated;
   the best becomes the starting point for the next stage.

2. **Projected Adam on a smooth approximation of the max‑convolution**.
   We minimise the smooth function
       f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
   while keeping ``a`` on the probability simplex.  The smoothing
   parameter α is increased geometrically; Adam’s adaptive steps are
   projected onto the simplex after each update.

3. **Fine‑grained local refinement** – a fast greedy routine that moves
   a small fraction of the coefficient that contributes most to the
   current convolution peak to the smallest coefficient.  The move is
   accepted only when it strictly reduces the peak.  After a first
   pass with a 5 % transfer fraction a second pass with 1 % and a final
   pass with 0.2 % are performed.

The whole search respects the 1000 s time limit, prints intermediate
scores and finally returns the best sequence found as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                      # weights sum to 1

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre of correlation corresponds to indices 0..n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

# Stage 0 – parametric shape scan
# ----------------------------------------------------------------------


def _shape_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}  (edges become larger for p>0)."""
    i = np.arange(n, dtype=float) + 0.5
    den = i * (n + 0.5 - i)
    # add a tiny epsilon to avoid division by zero at the borders
    eps = 1e-12
    a = (den + eps) ** (-p)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _shape_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Symmetric discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _shape_edge_block(n: int, edge_frac: float) -> np.ndarray:
    """Mass concentrated on the two outer edge blocks."""
    k = max(1, int(edge_frac * n))
    a = np.zeros(n, dtype=float)
    a[:k] = 1.0
    a[-k:] = 1.0
    # keep a tiny uniform background to avoid a zero‑sum interior
    a += 1e-6
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a warm start (historic best) or a uniform vector."""
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s > 0:
                return seq / s
    except Exception:
        pass
    return np.full(n, 1.0 / n)

def _adam_optimize(a0: np.ndarray, deadline: float, start_time: float,
                   n: int) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth approximation of max(conv).
    Returns the best vector found and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α gives a tighter approximation
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0,
                      2048.0, 8192.0, 16384.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for alpha in alpha_schedule:
        while time.time() < deadline:
            # compute current full convolution (FFT version)
            conv = _conv_self(a)

            # smooth‑max value (for logging only)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            smooth_max = lse / alpha

            # gradient of the *original* objective 2·n·max(conv)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = float(evaluate_sequence(a.tolist()))
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha:.0f}) "
                          f"score={sc:.6f}")

            # stop early if we are running out of time for this α
            if time.time() > deadline:
                break
    return best_a, best_score

# Stage 2 – Sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray, deadline: float, start_time: float,
                      n: int,
                      beta: float = 0.94,
                      step0: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,} "
                  f"score={cur_score:.6f}")

        # active set (indices where conv is close to the max)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the max (sum of all active contributions)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair contributes a[j] to ∂/∂a[i] and a[i] to ∂/∂a[j]
            g[i] += a[j]
            g[j] += a[i]        # diagonal counted twice – inconsequential

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Stage 3 – Fast greedy peak reduction (multi‑pass)

def _fast_greedy_improve(a0: np.ndarray, deadline: float,
                         start_time: float,
                         n: int,
                         max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the convolution peak.
    Performs three passes with decreasing transfer fractions.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    # three passes – coarse → fine
    eps_factors = [0.05, 0.01, 0.002]

    for eps_factor in eps_factors:
        while time.time() < deadline and it < max_iters:
            it += 1
            k_max = conv.argmax()
            cur_max = conv[k_max]

            # indices that take part in the peak
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = k_max - i_idx
            contrib = a[i_idx] * a[j_idx]
            src = i_idx[contrib.argmax()]          # most responsible index

            # destination = smallest coefficient (different from src)
            dst_candidates = np.delete(np.arange(n), src)
            dst = dst_candidates[a[dst_candidates].argmin()]

            eps = eps_factor * a[src]              # move a fraction of src
            if eps < 1e-14:
                break

            # incremental update of the full convolution (O(n))
            conv_tmp = conv.copy()
            # linear part: -2·eps·a shifted by src  + 2·eps·a shifted by dst
            conv_tmp[src:src + n] -= 2.0 * eps * a
            conv_tmp[dst:dst + n] += 2.0 * eps * a
            # quadratic corrections
            conv_tmp[2 * src] += eps * eps
            conv_tmp[2 * dst] += eps * eps
            conv_tmp[src + dst] -= 2.0 * eps * eps

            new_max = conv_tmp.max()
            if new_max < cur_max:          # strict improvement required
                conv = conv_tmp
                a[src] -= eps
                a[dst] += eps
                best_score = 2.0 * n * new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] greedy (eps={eps_factor:.3f}) "
                      f"iter={it:,}, max={new_max:.6f}")

        # after each pass re‑evaluate remaining time
        if time.time() >= deadline:
            break

    return best_a, best_score

        best_a, best_score = _adam_optimize(best_a, adam_deadline,
                                            start_time, n)
        print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.30 * (deadline - time.time())
        print(f"[{time.time() - start_time:6.2f}s] start sub‑gradient polishing")
        best_a, best_score = _subgrad_optimize(best_a, subgrad_deadline,
                                               start_time, n,
                                               beta=0.94, step0=0.05)
        print(f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fast greedy refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline,
                                                  start_time, n)
        print(f"[{time.time() - start_time:6.2f}s] greedy phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = float(evaluate_sequence(best_a.tolist()))
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return as plain Python list
    return [float(x) for x in best_a.tolist()]
