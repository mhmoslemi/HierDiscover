"""
Propose a step‑function by alternating linear‑programming and greedy
post‑processing.

* Phase 1 – LP refinement *
    Starting from a probability vector `a` (sum = 1) we compute
    M = max(a*a).  Keeping `a` fixed, the constraints
        (a * b)_k ≤ M   (k = 0…2n‑2)
    are linear in `b`.  We solve the LP that maximises Σ b_j.
    The optimum `b` is then mixed with the current `a`
        a ← (1‑t)·a + t·b   (t≈0.2)
    and renormalised.  Repeating this step strictly reduces the peak
    of the self‑convolution.

* Phase 2 – Greedy peak reduction *
    When the LP loop runs out of time we switch to a fast deterministic
    move: locate the index of the current maximal convolution entry,
    take a little mass from the most contributing source index and give
    it to a destination that contributes the least, updating the
    convolution incrementally.  This often shaves the last few 10⁻⁴
    from the objective.

The whole routine respects the 1000 s budget, prints occasional
progress messages and finally returns the best sequence found (as a plain
list of floats).  The sequence is non‑negative, its sum is 1, and the
objective `evaluate_sequence` is therefore `2·n·max(a*a)`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_max(a: np.ndarray, L: int) -> float:
    """Return max entry of the (full) self‑convolution a*a using FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    # conv length is L, but only first 2*len(a)-1 entries are valid
    conv = conv[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix `A` such that (A @ b)[k] = (a * b)[k].
    The k‑th row corresponds to the k‑th entry of the full convolution.
    """
    n = a.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    # each column j contributes the vector a shifted by j positions
    for j in range(n):
        A[j:j + n, j] = a
    return A


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm consists of an LP‑based refinement phase followed by a
    greedy deterministic phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 1024                         # length of the vector (adjustable)
    t_mix = 0.20                     # step size for the convex combination
    lp_max_iters = 200               # safety limit for LP iterations
    greedy_max_iters = 2_000_000     # upper bound for greedy moves

    # ------------------------------------------------------------------
    # Initialise a random probability vector
    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a /= a.sum()                     # enforce sum = 1

    # cache the FFT length for fast convolutions
    Lfft = _next_pow2(2 * n - 1)

    # keep the best sequence seen so far
    max_conv = _conv_self_max(a, Lfft)
    best_score = 2.0 * n * max_conv
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] start: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Alternating linear programme
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 30.0 and lp_iter < lp_max_iters:
        lp_iter += 1

        # current peak of the self‑convolution
        M = _conv_self_max(a, Lfft)

        # Build the convolution matrix A (A @ b <= M)
        A = _build_conv_matrix(a)

        # Linear programme: maximise Σ b_j  ⇔ minimise −Σ b_j
        c = -np.ones(n, dtype=np.float64)          # objective vector
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # right‑hand side
        bounds = [(0.0, None)] * n

        # Solve with the HiGHS interior‑point method (fast for moderate size)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True})
        if not res.success:
            # If the LP fails for any reason we break out of the LP phase.
            print(f"[{time.time() - start_time:.2f}s] LP failed at iteration {lp_iter}")
            break

        b = res.x
        # Normalise b to keep it a probability vector
        s = b.sum()
        if s > 0:
            b /= s
        else:
            b = np.full(n, 1.0 / n)

        # Convex combination and renormalisation
        a = (1.0 - t_mix) * a + t_mix * b
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # Evaluate current score (exact, because sum(a)=1)
        cur_max = _conv_self_max(a, Lfft)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter:4d} improve: score={cur_score:.6f}")

        # Optional: slowly decrease the mixing parameter to fine‑tune
        if lp_iter % 20 == 0:
            t_mix = max(0.02, t_mix * 0.95)

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic greedy reduction of the peak
    # ------------------------------------------------------------------
    # Pre‑compute the full convolution once; we will keep it up‑to‑date
    conv = np.fft.irfft(np.fft.rfft(a, n=Lfft) * np.fft.rfft(a, n=Lfft), n=Lfft)
    conv = conv[: 2 * n - 1]               # keep only the valid part
    greedy_iter = 0
    recompute_interval = 1000

    while time.time() < deadline - 5.0 and greedy_iter < greedy_max_iters:
        greedy_iter += 1

        # Locate the current maximum of the convolution
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # Find the indices that actually contribute to this maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # Determine a destination index that has the smallest partner in the peak
        # (i.e. the smallest a[max_idx - j] among all j)
        a_partner = np.zeros(n, dtype=np.float64)
        a_partner[idx] = a[max_idx - idx]
        # pick a destination different from src with minimal partner value
        dst_candidates = np.where((a_partner == a_partner.min()) & (np.arange(n) != src))[0]
        if dst_candidates.size == 0:
            dst = (src + n // 2) % n   # fallback to a far index
        else:
            dst = np.random.choice(dst_candidates)

        # How much mass can we move?  Use a fraction of the source mass.
        eps = min(a[src], 0.02 * a[src])   # modest step size

        if eps <= 0:
            continue

        # ----- Incremental update of the convolution (O(n)) -----
        # contribution of moving eps from src to dst:
        #   (a_new * a_new) = (a - eps*e_src + eps*e_dst) * (same)
        # the change can be written as:
        #   Δconv = -2*eps * (a shifted by src) + 2*eps * (a shifted by dst)
        # We reuse the *current* a (before the move) for the shift vectors.
        a_old = a.copy()
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[dst:dst + n] += 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        # Keep everything non‑negative (numerical safety)
        a[a < 0] = 0.0

        # Periodic exact recomputation to avoid drift
        if greedy_iter % recompute_interval == 0:
            conv = np.fft.irfft(np.fft.rfft(a, n=Lfft) *
                                np.fft.rfft(a, n=Lfft), n=Lfft)
            conv = conv[: 2 * n - 1]

        # Evaluate the new score
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter {greedy_iter:7d} improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (ensuring a proper probability vector)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished: best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
