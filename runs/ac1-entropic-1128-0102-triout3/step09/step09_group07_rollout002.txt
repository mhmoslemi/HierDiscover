"""
Multi‑stage optimisation for a non‑negative step‑function.
1️⃣  Fast exponential‑gradient warm‑up on a smooth approximation of the max‑norm.
2️⃣  Deterministic incremental refinement:
    – each iteration moves mass from a “source’’ index that strongly contributes
      to the current maximal convolution bin to a “destination’’ index that
      contributes little.
    – the linear effect of the move on the whole self‑convolution is computed
      analytically, and the largest step‑size ε that still reduces the current
      maximum (solving a 1‑D LP) is taken.
    – this yields orders‑of‑magnitude larger improvements per iteration while
      keeping the cost O(n) per step.
The algorithm respects a 1000 s wall‑clock budget, keeps the best sequence
seen and returns it as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up on smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient via log‑sum‑exp
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: deterministic incremental refinement (large guaranteed moves)
# ----------------------------------------------------------------------
def _run_deterministic_refine(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              recompute_every: int = 5000,
                              verbose_every: int = 1000):
    """
    Repeatedly transfer mass from a source index that contributes heavily to the
    current maximal convolution bin to a destination index that contributes little.
    The step size ε is chosen analytically – the largest value that still reduces
    the current maximum (solving a simple 1‑D linear programme).  The update of
    the convolution is performed analytically, yielding O(n) work per iteration.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    it = 0
    accepted = 0
    last_print = time.time()

    while time.time() < deadline:
        # --- locate the currently maximal bin ---------------------------------
        k_max = int(np.argmax(conv))

        # --- source: index with largest counterpart a[k_max‑i] among those that still have mass ---
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        counterpart = a[k_max - i_range]                # values that appear in the max bin
        have_mass = a[i_range] > 1e-15
        if not np.any(have_mass):
            # fallback: pick any index with some mass
            idx_nonzero = np.where(a > 1e-15)[0]
            if idx_nonzero.size == 0:
                break
            src = rng.choice(idx_nonzero)
        else:
            src_candidates = i_range[have_mass]
            cnt_candidates = counterpart[have_mass]
            src = src_candidates[np.argmax(cnt_candidates)]

        # --- destination: index with smallest counterpart (zero is best) ----------
        idx = np.arange(n)
        mask_valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_counter = np.zeros(n)
        dest_counter[mask_valid] = a[k_max - idx[mask_valid]]
        # exclude the source itself
        dest_counter[src] = np.inf
        zero_mask = dest_counter < 1e-12
        if np.any(zero_mask):
            dst = rng.choice(np.where(zero_mask)[0])
        else:
            dst = np.argmin(dest_counter)

        if src == dst:
            continue

        # --- linear effect of moving ε from src to dst on the whole convolution ---
        s = np.zeros(2 * n - 1, dtype=float)
        # subtract contribution of src
        s[src:src + n] -= 2.0 * a
        # add contribution of dst
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # this pair cannot lower the current maximum – skip
            continue

        # --- maximal admissible ε before another entry overtakes the maximum ---
        mask = s > s_max                      # slopes steeper than at the max entry
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥ 0
            delta_s = s[mask] - s_max                     # > 0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        # limit ε by the amount of mass we can take from the source
        eps_limit = a[src] * 0.9                      # keep a tiny residue
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 1e-12:
            continue

        # --- apply the move ----------------------------------------------------
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        # accept only if we really improve (guard against tiny numeric noise)
        if new_max < max_val - 1e-14:
            a[src] -= epsilon
            a[dst] += epsilon
            conv = conv_new
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] deterministic improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasionally recompute the convolution from scratch to avoid drift
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

        # print a lightweight progress line every few seconds
        if time.time() - last_print > 10.0:
            print(f"[{time.time() - start_time:.2f}s] deterministic status: "
                  f"iter={it:,}, max={max_val:.8f}, accepted={accepted}")
            last_print = time.time()

    return best_a

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine uses a fast exponential‑gradient warm‑up followed by a
    deterministic refinement that analytically computes the largest feasible
    mass‑transfer step at each iteration.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to guarantee a return before the external timeout
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise – use the previously discovered construction if available,
    # otherwise start from a discretised arcsine distribution (known to give a
    # fairly flat self‑convolution).
    # --------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter / starter.sum(), final_n)
        else:
            raise ValueError
    except Exception:
        # arcsine‑like start: a[i] ∝ 1 / sqrt(x_i (1‑x_i)),  x_i = (i+0.5)/n
        x = (np.arange(final_n) + 0.5) / final_n
        a0 = 1.0 / np.sqrt(x * (1.0 - x))
        a0 = np.nan_to_num(a0, nan=0.0, posinf=0.0, neginf=0.0)
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Stage 1 – exponential‑gradient warm‑up (short, to give a decent start)
    # --------------------------------------------------------------
    if time.time() < deadline:
        stage1_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.1)
        a_warm, score_warm = _run_exponential_gradient(
            a0,
            final_n,
            stage1_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.4,
            steps_per_alpha=5000,
        )
        if score_warm < best_score:
            best_score = score_warm
            best_seq = a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deterministic incremental refinement (the heavy work)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref = _run_deterministic_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            recompute_every=5000,
        )
        conv_ref = _conv_self(a_ref)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] final deterministic improve: score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # optional sanity check (does not affect the return value)
    try:
        final_val = evaluate_sequence([float(x) for x in best_seq])
        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
