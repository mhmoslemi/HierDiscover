"""
search for a non‑negative step‑function (vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The code uses three ideas:

1. **Parameterized warm‑starts** – a small collection of analytic families
   (uniform, exponential, symmetric exponential, power‑law, triangular,
   cosine) is sampled and the best one is kept.

2. **Smoothed‑max gradient descent** – the nonsmooth max is approximated by
   a log‑sum‑exp (temperature α).  The gradient of the smoothed objective
   is a weighted Toeplitz‑multiply which can be computed by a single FFT
   (convolution of the weight vector with the reversed current vector).
   Momentum (Nesterov–like) and a decreasing step‑size are used.
   The iterate is always projected onto the probability simplex
   (non‑negative, sum = 1).

3. **Final exponentiated‑gradient polishing** – a short EG loop
   (multiplicative update) on the true sub‑gradient of the max improves
   the bound a little more.

The routine respects a global time budget (default 1000 s) and returns a
plain ``list[float]`` that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (no external dependencies)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using a real FFT.
    """
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)[:L]
    return conv


def _weighted_corr_grad(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k p_k·(a*a)_k   with respect to a (a ≥ 0).
    It equals 2·(p * a_rev)[n-1 : n-1+n] where * denotes a linear convolution
    and a_rev is the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = p.shape[0]               # = 2·n-1
    N = _next_pow2(L + n - 1)    # size needed for linear convolution
    P = np.fft.rfft(p, n=N)
    A = np.fft.rfft(a_rev, n=N)
    conv = np.fft.irfft(P * A, n=N)
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).  Used only in the final EG phase.
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max(1e-12 * max_val, 0.0)
    idx_max = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in idx_max:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if idx_max.size > 1:
        grad /= idx_max.size
    return grad


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (core optimiser)
# ----------------------------------------------------------------------
def _gd_optimize(start_a: np.ndarray,
                 n: int,
                 deadline: float,
                 lr0: float = 0.25,
                 beta: float = 0.9,
                 alpha0: float = 1.0,
                 alpha_factor: float = 1.03,
                 alpha_max: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Perform a momentum gradient descent on the smoothed max objective.
    Returns the best vector found and its score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    momentum = np.zeros_like(a)
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}s] GD iter {it:5d}  improvement → {cur_score:.6f}"
            )

        # ----- smoothed max (log‑sum‑exp) -----
        max_c = conv.max()
        # stable exponentiation
        w = np.exp(alpha * (conv - max_c))
        p = w / w.sum()

        # ----- gradient of the smoothed objective -----
        grad = _weighted_corr_grad(a, p)

        # ----- momentum step + projection onto simplex -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = lr0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # increase temperature slowly, capped at alpha_max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_a, best_score

def _family_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _family_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    return a / a.sum()


def _family_sym_exponential(n: int, lam: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(np.arange(n) - centre))
    return a / a.sum()


def _family_powerlaw(n: int, beta: float) -> np.ndarray:
    a = 1.0 / (np.arange(1, n + 1, dtype=float) ** beta)
    return a / a.sum()


def _family_triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(np.arange(n) - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _family_cosine(n: int) -> np.ndarray:
    # non‑negative cosine on [0,π] shifted up
    a = np.cos(np.pi * (np.arange(n) / (n - 1) - 0.5))
    a = a - a.min() + 1e-12            # make strictly positive
    return a / a.sum()

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation objective.
    The routine respects the supplied time budget (seconds) and returns a
    plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # Problem size – 2048 gives a good trade‑off between resolution and speed.
    # --------------------------------------------------------------
    n = 2048

    # --------------------------------------------------------------
    # Initialise with a collection of analytic families (including the previous best)
    # --------------------------------------------------------------
    candidates = [_family_uniform(n)]

    # a few sampled exponential / power‑law families
    for lam in np.logspace(-3, 0, 6):
        candidates.append(_family_exponential(n, lam))
        candidates.append(_family_sym_exponential(n, lam))
    for beta in np.linspace(0.2, 2.0, 7):
        candidates.append(_family_powerlaw(n, beta))

    candidates.append(_family_triangular(n))
    candidates.append(_family_cosine(n))

    # if the notebook supplied a previous construction, use it as a warm‑start
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != n:
                    # linear interpolation to required length
                    old_x = np.linspace(0.0, 1.0, prev.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    prev = np.interp(new_x, old_x, prev)
                prev = np.maximum(prev, 0.0)
                prev_sum = prev.sum()
                if prev_sum > 0:
                    prev = prev / prev_sum
                else:
                    prev = np.full(n, 1.0 / n)
                candidates.append(prev)
        except Exception:
            pass

    best_seq = np.full(n, 1.0 / n)
    best_score = 2.0 * n * _conv_self(best_seq).max()

    for idx, a in enumerate(candidates):
        cur = 2.0 * n * _conv_self(a).max()
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] family {idx:2d}  score = {cur:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – many short random restarts with gradient descent
    # --------------------------------------------------------------
    # allocate roughly 55 % of the remaining time to the random‑restart phase
    restart_deadline = time.time() + 0.55 * (deadline - time.time())

    while time.time() < restart_deadline:
        # draw a Dirichlet start with a random concentration parameter
        conc = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, conc))

        # run a quick GD (few thousand iterations, bounded by time)
        sub_deadline = time.time() + 2.0   # at most two seconds per restart
        a_opt, sc = _gd_optimize(a0, n, sub_deadline,
                                 lr0=0.30, beta=0.85,
                                 alpha0=1.0, alpha_factor=1.04,
                                 alpha_max=180.0)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improved → {sc:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long fine‑tuning on the current best vector
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_opt, sc = _gd_optimize(best_seq, n, deadline,
                                 lr0=0.25, beta=0.90,
                                 alpha0=2.0, alpha_factor=1.02,
                                 alpha_max=250.0)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] fine‑tune improved → {sc:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – short exponentiated‑gradient polishing (sub‑gradient of true max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        momentum = np.zeros_like(a)
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] EG iter {it:5d}  improve → {cur_score:.6f}"
                )
            # sub‑gradient of the max‑norm
            grad = _subgradient_max(a, conv)
            eta = 0.12 / math.sqrt(it)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    # final normalisation (just in case numerical drift introduced a tiny sum‑error)
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
