def _run_greedy_incremental_refine(a0, n, deadline, start_time, eps_start=0.04, eps_end=0.0002, rel_improve=1e-12, candidate_src=5, candidate_dst=5):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it=0
    while time.time() < deadline:
        t_elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - (t_elapsed) / total_time))
        k_max = int(np.argmax(conv))

        # Determine src candidates
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        i_indices = np.arange(i_start, i_end+1, dtype=int)
        j_indices = k_max - i_indices
        contrib = a[i_indices] * a[j_indices]
        # Get top src candidates based on contribution or a[i]? Use contribution.
        # If total contributions zero, fall back to top a values
        if contrib.sum() > 0:
            # sort descending contributions
            top_order = np.argsort(-contrib)  # indices into i_indices
            src_candidates = i_indices[top_order[:candidate_src]]
        else:
            src_candidates = np.argsort(-a)[:candidate_src]

        # Destination candidates: lowest a values not in src (or not close)
        dst_order = np.argsort(a)
        dst_candidates = []
        for idx in dst_order:
            if idx not in src_candidates:
                dst_candidates.append(idx)
            if len(dst_candidates) >= candidate_dst:
                break
        dst_candidates = np.array(dst_candidates, dtype=int)

        # Also consider some random dst for diversity
        if len(dst_candidates) < candidate_dst:
            extra = rng.choice(n, size=candidate_dst - len(dst_candidates), replace=False)
            dst_candidates = np.unique(np.concatenate([dst_candidates, extra]))

        best_pair = None
        best_new_max = max_val
        best_eps = 0.0

        # Evaluate each pair
        for src in src_candidates:
            if a[src] < 1e-15:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue
                # s vector
                s = np.zeros(2*n-1, dtype=float)
                s[src:src+n] -= 2.0 * a
                s[dst:dst+n] += 2.0 * a
                s_max = s[k_max]
                if s_max >= 0:
                    continue
                # compute eps_star
                mask = s > s_max
                if np.any(mask):
                    delta_conv = conv[k_max] - conv[mask]
                    delta_s = s[mask] - s_max
                    eps_candidates = delta_conv / delta_s
                    eps_candidates = eps_candidates[eps_candidates > 0.0]
                    eps_star = eps_candidates.min() if eps_candidates.size else np.inf
                else:
                    eps_star = np.inf
                eps_limit = a[src] * eps_frac
                epsilon = min(eps_star, eps_limit)
                if epsilon <= 0.0:
                    continue
                conv_new = conv + epsilon * s
                new_max = conv_new.max()
                if new_max < best_new_max - rel_improve:
                    best_new_max = new_max
                    best_pair = (src, dst)
                    best_eps = epsilon
                    best_s = s
                    # can break early maybe? Continue scanning for better.

        if best_pair is not None:
            src, dst = best_pair
            # apply update
            conv = conv + best_eps * best_s
            a[src] -= best_eps
            a[dst] += best_eps
            max_val = best_new_max
            it += 1
            if best_new_max < best_max - rel_improve:
                best_max = best_new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] greedy refine improve: iter={it:,}, max={best_max:.8f}")
        else:
            # fallback to random move similar to previous algorithm
            # (optional) break if no improvement possible
            # We could try random move with small epsilon to see if any improvement.
            # We'll implement a small random move attempt
            src = rng.choice(np.where(a > 1e-15)[0])
            dst = rng.choice(np.where(np.arange(n)!=src)[0])
            # compute s and epsilon as before
            s = np.zeros(2*n-1, dtype=float)
            s[src:src+n] -= 2.0 * a
            s[dst:dst+n] += 2.0 * a
            s_max = s[k_max]
            if s_max >= 0:
                continue
            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[k_max] - conv[mask]
                delta_s = s[mask] - s_max
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf
            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)
            if epsilon <= 0.0:
                continue
            conv_new = conv + epsilon * s
            new_max = conv_new.max()
            if new_max < max_val - rel_improve:
                conv = conv_new
                a[src] -= epsilon
                a[dst] += epsilon
                max_val = new_max
                it += 1
                # update best
                if new_max < best_max - rel_improve:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] greedy refine random improve: iter={it:,}, max={best_max:.8f}")

            conv = _conv_self(a)
            max_val = conv.max()
    return best_a

    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    # For epsilon schedule, compute fraction based on elapsed time relative to total_time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linearly decay epsilon fraction to eps_end
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))
        k_max = int(np.argmax(conv))
        # Determine src candidates
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_indices = np.arange(i_start, i_end + 1, dtype=int)
        j_indices = k_max - i_indices
        contrib = a[i_indices] * a[j_indices]
        if contrib.sum() > 0:
            top_order = np.argsort(-contrib)  # descending
            src_candidates = i_indices[top_order[:candidate_src]]
        else:
            src_candidates = np.argsort(-a)[:candidate_src]
        # Determine dst candidates: smallest a not in src_candidates
        dst_order = np.argsort(a)
        dst_candidates = []
        for idx in dst_order:
            if idx not in src_candidates:

            dst_candidates = np.unique(np.concatenate([dst_candidates, extra]))
        # Find best move
        best_pair = None
        best_new_max = max_val
        best_eps = 0.0
        best_s = None
        for src in src_candidates:
            if a[src] < 1e-15:

            for dst in dst_candidates:

"""
Optimise a non‑negative step‑function to minimise the value returned by
``evaluate_sequence`` (i.e. 2·n·max(self‑convolution) after the sum of the
coefficients is normalised to 1).

The optimisation proceeds in three phases within the given time budget:

1️⃣  Exponentiated‑gradient warm‑up on a smooth log‑sum‑exp approximation
    of the max‑convolution.  This quickly obtains a decent distribution.

2️⃣  Projected sub‑gradient descent with momentum on the exact ℓ∞‑norm of the
    self‑convolution.  The bulk of the time budget is spent here.

3️⃣  Greedy incremental refinement.  In each iteration a set of candidate
    source indices (that contribute heavily to the current maximal
    convolution entry) and destination indices (that contribute little) is
    examined.  For every candidate pair the largest admissible mass transfer
    ε is computed analytically (a tiny 1‑D linear programme).  The pair that
    yields the greatest reduction of the current maximum is applied.  The
    admissible fraction of the source mass decreases linearly from ``eps_start``
    to ``eps_end`` so the final stage can make very fine adjustments.

The best sequence seen throughout the run is returned as a plain Python list.
All numbers are non‑negative and the sequence is normalised to sum 1 before
returning, satisfying the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient warm‑up on smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: greedy incremental refinement
# ----------------------------------------------------------------------
def _run_greedy_incremental_refine(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   eps_start: float = 0.05,
                                   eps_end: float = 0.0001,
                                   rel_improve: float = 1e-12,
                                   candidate_src: int = 8,
                                   candidate_dst: int = 8,
                                   recompute_every: int = 2000):
    """
    Greedy refinement that repeatedly transfers a fraction of mass from a source
    index (strongly contributing to the current maximal convolution bin) to a
    destination index (contributing little).  For every candidate pair the
    maximal admissible step size ε is computed analytically; the pair giving the
    largest reduction of the current maximum is applied.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear decay of the fraction of source mass we are allowed to move
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        k_max = int(np.argmax(conv))

        # ---- source candidates (indices that contribute to the max bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.sum() > 0:
            order = np.argsort(-contrib)[:candidate_src]
            src_candidates = i_range[order]
        else:
            src_candidates = np.argsort(-a)[:candidate_src]

        # ---- destination candidates (indices with little mass) ----
        dst_order = np.argsort(a)
        dst_candidates = []
        for idx in dst_order:
            if idx not in src_candidates:
                dst_candidates.append(idx)
            if len(dst_candidates) >= candidate_dst:
                break
        dst_candidates = np.array(dst_candidates, dtype=int)
        # If still missing, fill with random indices
        if dst_candidates.size < candidate_dst:
            extra = rng.choice(n, size=candidate_dst - dst_candidates.size, replace=False)
            dst_candidates = np.unique(np.concatenate([dst_candidates, extra]))

        # ---- search the best move among the candidates ----
        best_pair = None
        best_new_max = max_val
        best_eps = 0.0
        best_s = None

        for src in src_candidates:
            if a[src] < 1e-15:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue

                # effect vector s (only two sliced updates)
                s = np.zeros(2 * n - 1, dtype=float)
                s[src:src + n] -= 2.0 * a
                s[dst:dst + n] += 2.0 * a

                s_max = s[k_max]
                if s_max >= 0.0:
                    continue   # would not lower the current maximum

                # maximal admissible ε (1‑D LP)
                mask = s > s_max
                if np.any(mask):
                    delta_conv = conv[k_max] - conv[mask]
                    delta_s = s[mask] - s_max
                    eps_candidates = delta_conv / delta_s
                    eps_candidates = eps_candidates[eps_candidates > 0.0]
                    eps_star = eps_candidates.min() if eps_candidates.size else np.inf
                else:
                    eps_star = np.inf

                eps_limit = a[src] * eps_frac
                epsilon = min(eps_star, eps_limit)
                if epsilon <= 0.0:
                    continue

                # evaluate new maximum
                conv_tmp = conv.copy()
                conv_tmp[src:src + n] -= 2.0 * epsilon * a
                conv_tmp[dst:dst + n] += 2.0 * epsilon * a
                new_max = conv_tmp.max()

                if new_max < best_new_max - rel_improve:
                    best_new_max = new_max
                    best_pair = (src, dst)
                    best_eps = epsilon
                    best_s = s

        # ---- apply the best move if it improves the objective ----
        if best_pair is not None:
            src, dst = best_pair
            # update convolution in‑place
            conv[src:src + n] -= 2.0 * best_eps * a
            conv[dst:dst + n] += 2.0 * best_eps * a
            # transfer mass
            a[src] -= best_eps
            a[dst] += best_eps
            max_val = best_new_max
            it += 1
            accepted += 1

            if best_new_max < best_max - rel_improve:
                best_max = best_new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] greedy refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        else:
            # ---- fallback: a single random admissible move ----
            src_choices = np.where(a > 1e-15)[0]
            if src_choices.size == 0:
                break
            src = rng.choice(src_choices)
            dst_candidates = np.delete(np.arange(n), src)
            dst = rng.choice(dst_candidates)

            s = np.zeros(2 * n - 1, dtype=float)
            s[src:src + n] -= 2.0 * a
            s[dst:dst + n] += 2.0 * a
            s_max = s[k_max]
            if s_max >= 0.0:
                continue

            mask = s > s_max
            if np.any(mask):
                delta_conv = conv[k_max] - conv[mask]
                delta_s = s[mask] - s_max
                eps_candidates = delta_conv / delta_s
                eps_candidates = eps_candidates[eps_candidates > 0.0]
                eps_star = eps_candidates.min() if eps_candidates.size else np.inf
            else:
                eps_star = np.inf

            eps_limit = a[src] * eps_frac
            epsilon = min(eps_star, eps_limit)
            if epsilon <= 0.0:
                continue

            conv_tmp = conv.copy()
            conv_tmp[src:src + n] -= 2.0 * epsilon * a
            conv_tmp[dst:dst + n] += 2.0 * epsilon * a
            new_max = conv_tmp.max()
            if new_max < max_val - rel_improve:
                conv = conv_tmp
                a[src] -= epsilon
                a[dst] += epsilon
                max_val = new_max
                it += 1
                accepted += 1
                if new_max < best_max - rel_improve:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] greedy refine random improve: "
                          f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to avoid drift
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of half a second to guarantee a return before timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – use the previously known construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                     # length of the step‑function
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – warm‑up (exponential gradient)
    # ------------------------------------------------------------------
    stage1_time = min(12.0, 0.07 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=6000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (major part of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep at least 30 s for the final greedy refinement
        stage2_time = max(30.0, 0.4 * remaining)
        stage2_deadline = time.time() + stage2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – greedy incremental refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_greedy_incremental_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.05,
            eps_end=0.0001,
            candidate_src=8,
            candidate_dst=8,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
