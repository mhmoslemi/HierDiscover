    return w / w_sum

"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Warm‑start selection** – generate a few simple candidate
   probability vectors (uniform, several symmetric Beta‑shaped
   distributions) and keep the one with the smallest exact score.

2. **Smooth‑max Adam** – minimise a log‑sum‑exp (smooth) proxy for the
   max‑convolution using projected Adam.  A schedule of increasing
   smoothness parameters drives the iterate ever closer to the true
   objective.

3. **Fast FFT‑based sub‑gradient descent** – the exact sub‑gradient of
   the max‑convolution can be written as a cross‑correlation, which we
   evaluate with a single FFT per iteration.  This makes it possible
   to run many more iterations than the reference implementation,
   achieving a finer reduction of the peak.

4. **(Optional) tiny greedy polishing** – a few cheap moves that
   transfer a small fraction of mass from the most responsible index
   to the smallest one, accepted only if they strictly improve the
   objective.

The routine respects the 1000 s time limit, prints progress messages
and finally returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max function
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Warm‑start generation
# ----------------------------------------------------------------------


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on a discrete grid of size n,
    then normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n          # points in (0,1)
    # avoid overflow for extreme α; clamp to a minimal positive value
    with np.errstate(divide='ignore'):
        w = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # for α<1 the density blows up at the ends – replace infinities
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return w / total


def _candidate_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest list of plausible starting vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # a few symmetric Beta‑shapes (U‑shaped for α<1, flat for α≈1)
    for a in [0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.1]:
        cand.append(_beta_distribution(n, a))

    # a random Dirichlet draw (moderately sparse)
    cand.append(rng.dirichlet(np.ones(n) * 0.6))

    # if a historic best exists, reuse it (as a probability vector)
    try:
        hist = np.asarray(height_sequence_1, dtype=float)
        if hist.ndim == 1 and hist.size > 0:
            if hist.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, hist.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                hist = np.interp(new_x, old_x, hist)
            hist = np.maximum(hist, 0.0)
            s = hist.sum()
            if s > 0:
                cand.append(hist / s)
    except Exception:
        pass

    return cand


# ----------------------------------------------------------------------
# Phase 1 – Projected Adam on the smooth max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max.  Returns the best iterate
    (as a probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # A reasonably long schedule of smoothing parameters.
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        while time.time() < deadline:
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                     # smoothed max
            # gradient (including outer factor 2·n)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha}) "
                          f"score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Fast FFT‑based sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgrad_gradient_fft(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max_k (a*a)[k] given the binary mask of
    active (maximal) positions.  The gradient equals
        2 * (a ⋆ mask)
    where ⋆ denotes cross‑correlation.  Implemented via a single FFT.
    """
    n = a.shape[0]
    # reverse mask for correlation via convolution
    mask_rev = mask[::-1]
    fft_len = _next_pow2(n + mask.shape[0] - 1)      # = 3·n‑2 (rounded up)
    Af = np.fft.rfft(a, n=fft_len)
    Mf = np.fft.rfft(mask_rev, n=fft_len)
    corr = np.fft.irfft(Af * Mf, n=fft_len)[: n + mask.shape[0] - 1]

    offset = mask.shape[0] - 1
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad


def _subgrad_optimize_fast(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           step0: float = 0.20,
                           tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the FFT‑based exact gradient.
    Returns the best vector encountered and its exact evaluation.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: it={it:,} "
                  f"score={cur_score:.6f}")

        # build binary mask of active (maximal) convolution entries
        mask = np.zeros_like(conv, dtype=float)
        mask[conv >= cur_max - tol] = 1.0

        # exact sub‑gradient via FFT
        grad = _subgrad_gradient_fft(a, mask)

        # step size decreasing like 1/√t
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * grad)

        # occasional exact re‑evaluation (cheap enough)
        if it % 20000 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] sub‑grad exact eval: it={it:,} "
                      f"score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – tiny greedy polish (optional)
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very lightweight greedy refinement: move a small fraction of the
    mass from the index that contributes most to the current peak
    towards the smallest coefficient.  The convolution is updated
    incrementally (O(n) per move).  This phase is optional and only
    runs on the remaining time.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        # destination = smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.02 * a[src]                     # 2 % transfer (more fine‑grained)
        if eps < 1e-14:
            break

        # incremental update of the convolution
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # strict improvement required
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: it={it:,} "
                  f"max={new_max:.6f}")

    return best_a, best_score

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries keep FFT cheap and still
    # give the optimiser enough degrees of freedom.
    n = 4800

    # ------------------------------------------------------------------
    # 1) warm‑start pool – pick the best out of a handful of simple shapes
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] generating warm‑start candidates")
    candidates = _candidate_pool(n, rng)
    best_a = None
    best_score = float('inf')
    for idx, cand in enumerate(candidates):
        sc = evaluate_sequence(cand.tolist())
        print(f"  candidate {idx+1}/{len(candidates)} – score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[{time.time() - start_time:6.2f}s] warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smoothed max (≈40 % of budget)
    # ------------------------------------------------------------------
    adam_deadline = start_time + 0.40 * budget_s
    print(f"[{time.time() - start_time:6.2f}s] start Adam optimisation")
    best_a, best_score = _adam_optimize_long(best_a, adam_deadline, start_time)
    print(f"[{time.time() - start_time:6.2f}s] Adam phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) FFT‑based sub‑gradient descent (≈55 % of budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, start_time + 0.95 * budget_s)
    print(f"[{time.time() - start_time:6.2f}s] start fast sub‑gradient descent")
    best_a, best_score = _subgrad_optimize_fast(best_a, subgrad_deadline, start_time,
                                                step0=0.20)
    print(f"[{time.time() - start_time:6.2f}s] sub‑gradient phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional greedy polishing using the remaining seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] start greedy refinement")
        best_a, best_score = _fast_greedy_improve(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:6.2f}s] greedy phase finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
