def _adam_descent(a0, n, deadline, start_time, best_score):
    a = a0.copy()
    best_a = a.copy()
    best = best_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    L = _next_pow2(3*n-2) # maybe use larger for gradient

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L=L)
        max_idx = int(conv.argmax())
        cur_max = conv[max_idx]
        cur_score = 2.0*n*cur_max/(a.sum()**2)
        if cur_score < best:
            best = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam it={t} score={cur_score:.6f}")

        # gradient: derivative of max conv w.r.t a: g_i = 2*a_{k-i} if i contributes
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[max_idx - idx]  # as previously

        # If multiple indices have same max value (rare), we can average their gradients.

        # Update m,v
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.01 / (np.sqrt(v_hat) + eps)  # 0.01 maybe adapt
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best
```

def _update_conv(conv, a, src, dst, eps):
    # conv: np.ndarray length 2n-1, mutable
    # a: current a before the move (so includes src mass still)
    # compute effect of moving eps from src to dst.
    # b = a copy after modification? Not needed.
    # Remove src mass:
    conv_src_start = src
    conv_src_end = src + n   # exclusive
    conv[conv_src_start:conv_src_end] -= 2 * eps * a[:]
    # Add dst mass:
    conv_dst_start = dst
    conv_dst_end = dst + n
    conv[conv_dst_start:conv_dst_end] += 2 * eps * a[:]
    # Δ*Δ contributions:
    conv[2*src] += eps*eps
    conv[2*dst] += eps*eps
    conv[src + dst] -= 2 * eps * eps
    # Also the src removal changed a[src] by -eps, and a[dst] by +eps for future updates.

def _incremental_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq):
    a = a0.copy()
    conv = _conv_self(a)
    current_score = 2.0 * n * conv.max() / (a.sum()**2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    steps = 0
    total_time = deadline - start_time
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        t = elapsed / total_time
        T = 0.05 * (1 - t) + 1e-6 * t   # start 0.05 gradually to near 0.
        src = rng.choice(n, p=a)  # choose source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        # random fraction up to 0.5
        eps = a[src] * rng.random() * 0.5
        # copy arrays for evaluation
        a_prop = a.copy()
        conv_prop = conv.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # apply incremental update to conv_prop using a (original)
        conv_prop[src:src+n] -= 2 * eps * a
        conv_prop[dst:dst+n] += 2 * eps * a
        conv_prop[2*src] += eps*eps
        conv_prop[2*dst] += eps*eps
        conv_prop[src+dst] -= 2*eps*eps
        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max / (a_prop.sum()**2)  # sum stays 1
        delta = new_score - current_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] IncAnneal step={steps} best={best_score:.6f}")
        steps += 1
        # occasional full recompute of conv for numerical stability:
        if steps % 200000 == 0:
            conv = _conv_self(a)  # recompute
    return best_seq, best_score
```

"""
Propose a step‑function (non‑negative list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 within a 1000 s wall‑clock budget.

Algorithm
---------

1️⃣  **Rich parametric sweep** – generate many analytical families
    (geometric, power, beta, triangular) and also convex mixtures of each
    with the uniform vector.  All candidates are normalised to sum = 1 and
    evaluated with a fast FFT‑based convolution.  The best candidate of this
    cheap sweep is kept.

2️⃣  **Projected sub‑gradient descent** – starting from the best candidate,
    a momentum sub‑gradient method is applied to the exact objective.
    Iterates are projected onto the probability simplex.  This phase uses
    only a few hundred seconds, leaving ample time for polishing.

3️⃣  **Fast incremental annealing** – a mass‑transfer simulated‑annealing
    optimiser that updates the convolution incrementally (O(n) per move
    instead of O(n log n)).  Many more annealing steps can be performed,
    allowing the solution to escape the local minimum reached by the
    sub‑gradient phase.

The final sequence is normalised once more and returned as a plain Python
list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4          # a bit larger than the reference
    it = 0
    L_grad = _next_pow2(3 * n - 2)   # length for the gradient FFTs

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # index attaining the maximum (may be multiple – take the first)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the active term

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental simulated‑annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _incremental_annealing(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           cur_best_score: float,
                           cur_best_seq: np.ndarray,
                           start_temp: float = 5e-2,
                           end_temp: float = 1e-8,
                           max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Fast annealer that updates the convolution incrementally.
    Moves a fraction of the mass from a source index to a destination index.
    """
    a = a0.copy()
    conv = _conv_self(a)                     # exact convolution for the start
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    total_time = deadline - start_time
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1
        # linear temperature decay
        elapsed = time.time() - start_time
        t = elapsed / total_time
        T = start_temp * (1.0 - t) + end_temp * t

        # pick source proportionally to its mass and a uniform destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 30 % of the source mass (random fraction)
        eps = a[src] * rng.random() * 0.30

        # copy current state for the proposal
        a_prop = a.copy()
        conv_prop = conv.copy()

        a_prop[src] -= eps
        a_prop[dst] += eps

        # incremental update of the convolution (uses the *original* a)
        # removal from src
        conv_prop[src:src + n] -= 2.0 * eps * a
        # addition to dst
        conv_prop[dst:dst + n] += 2.0 * eps * a
        # Δ*Δ terms
        conv_prop[2 * src] += eps * eps
        conv_prop[2 * dst] += eps * eps
        conv_prop[src + dst] -= 2.0 * eps * eps

        new_score = 2.0 * n * conv_prop.max() / (a_prop.sum() ** 2)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal step={step:,} best={best_score:.6f}")

        # occasional full recompute to avoid drift
        if step % 200_000 == 0:
            conv = _conv_self(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1️⃣  Start with a uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    #   Global starter (if provided by the harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Rich parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # ---- geometric family (mix with uniform) -------------------------
    eps_grid = np.logspace(-8, -1, 300)            # epsilon = 1‑r, very fine
    lam_grid = np.linspace(0.0, 1.0, 21)           # mixing weight
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        geo /= geo_sum
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power family (a_i ∝ (i+1)^p) -------------------------------
    p_grid = np.linspace(-2.0, 4.0, 301)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p, dtype=np.float64)
        pow_sum = pow_seq.sum()
        if pow_sum == 0.0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")
        # mix with uniform
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * pow_seq
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Power+uni p={p:.2f} lam={lam:.2f} score={sc:.6f}")

    # ---- beta‑type families (a_i ∝ (i+1)^p (n‑i)^q) -----------------
    beta_params = np.arange(0.0, 4.5, 0.5)          # 0,0.5,…,4
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1.0, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0.0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")
            # mix with uniform as well
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * beta_seq
                seq = np.maximum(seq, 0.0)
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Beta+uni p={p:.1f} q={q:.1f} lam={lam:.2f} score={sc:.6f}")

    # ---- simple triangular shape (symmetric peak) --------------------
    tri = np.minimum(idx + 1.0, n - idx)          # a_i ∝ min(i+1, n-i)
    tri /= tri.sum()
    sc = _eval_fast(tri, L_fft)
    if sc < best_score:
        best_score = sc
        best_seq = tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular shape score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient refinement (use a substantial part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve ~30 s for the final annealing polish
    if remaining > 40.0:
        sub_deadline = deadline - 30.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline,
                                             start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Incremental annealing polish (use all time that is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _incremental_annealing(best_seq, n,
                                                   deadline,
                                                   start_time,
                                                   best_score,
                                                   best_seq,
                                                   start_temp=5e-2,
                                                   end_temp=1e-8,
                                                   max_steps=1_200_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final improve score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
