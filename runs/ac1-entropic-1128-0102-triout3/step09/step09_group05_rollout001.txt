"""
Hybrid optimiser – extended & diversified.

The optimiser works in several stages:

1️⃣  Exponentiated‑gradient on a smooth log‑sum‑exp surrogate of the
    maximum self‑convolution entry (much longer than the reference
    implementation).

2️⃣  Projected sub‑gradient descent on the true objective, first with the
    classic single‑peak gradient and then with a *multi‑peak* gradient
    (average of the top‑k convolution peaks).  The update uses Adam‑style
    per‑coordinate step‑sizes.

3️⃣  Targeted pair‑annealing (same idea as the reference code) but with
    a longer schedule and a tiny diffusion step every 1000 iterations
    to spread mass locally.

4️⃣  Final diffusive smoothing passes and a short Adam‑sub‑gradient run
    to polish the solution.

If time permits, the algorithm performs cheap random restarts (small
perturbations of the current best) and repeats steps 2‑4.

All helper functions are defined at module level – no closures or
lambdas – and the total running time is kept inside the supplied
budget (default 1000 s).  The returned list is normalised to a sum of 1,
so ``evaluate_sequence`` will compute ``2·n·max(conv)`` directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The normalization Σa=1 is assumed."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on the smooth max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.4,
                      iters_per_alpha: int = 12000):
    """Exponentiated‑gradient descent on a log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, _next_pow2(2 * n - 1)), n)

    L = _next_pow2(2 * n - 1)
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ---- smooth‑max weights (log‑sum‑exp) ----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ---- gradient of the smooth max (2·corr) ----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ---- exponentiated update ----
            eta = eta0 / math.sqrt(it)
            grad = np.clip(grad, 0.0, 10.0)          # avoid overflow
            a = a * np.exp(-eta * grad)

            # renormalise onto the simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Classical sub‑gradient descent (single peak)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None):
    """Projected sub‑gradient descent on f(a)=2·n·max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the unique maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2‑b – Multi‑peak Adam‑style sub‑gradient
# ----------------------------------------------------------------------
def _run_multi_peak_subgradient(a0: np.ndarray,
                                n: int,
                                deadline: float,
                                start_time: float,
                                best_score_initial: float,
                                top_m: int = 3,
                                step0: float = 0.2,
                                beta1: float = 0.9,
                                beta2: float = 0.999,
                                eps: float = 1e-8,
                                max_iters: int | None = None):
    """
    Adam‑style projected gradient using the average of the `top_m`
    largest convolution entries as the surrogate gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] multi‑sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ---- build weight vector having mass on the top‑m peaks ----
        if top_m == 1:
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0
        else:
            idx_peaks = np.argpartition(-conv, top_m)[:top_m]
            w = np.zeros_like(conv)
            w[idx_peaks] = 1.0 / top_m          # equal weight

        w_rev = w[::-1]

        # ---- gradient = 2 * correlation(a, w_rev)  (via FFT) ----
        Af = np.fft.rfft(a, n=L)
        Wr = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wr, n=L)[: len(a) + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)   # per‑coordinate step size
        a = a - step * m_hat

        # ---- enforce simplex ----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted pair‑annealing with occasional diffusion
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 250_000,
                        diffusion_interval: int = 1000,
                        diffusion_kernel: np.ndarray = np.array([0.25, 0.5, 0.25])):
    """
    Simulated annealing that repeatedly weakens the two indices that
    contribute most to the current maximal convolution entry.
    Every `diffusion_interval` iterations a tiny local diffusion
    (convolution with a small kernel) is applied to smooth the distribution.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # range of i that can pair with j = k‑i
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i (i paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # pick the pair (i,j) with largest product
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # choose distinct destinations (avoid the sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction of the source masses
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20   # ≤20 % of source mass
            a_prop[s] -= eps
            a_prop[d] += eps

        # evaluate the proposal
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

        # occasional tiny diffusion to spread mass locally
        if diffusion_interval and (it % diffusion_interval == 0):
            a = np.convolve(a, diffusion_kernel, mode='same')
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _conv_self(a, L)
            score = _score_from_conv(conv, n)

    return best_seq, best_score

# Small diffusion utility (used after the main phases)

def _diffuse(a: np.ndarray, passes: int = 6,
             kernel: np.ndarray = np.array([0.25, 0.5, 0.25])) -> np.ndarray:
    """Repeatedly smooth `a` with a tiny kernel, re‑normalising each time."""
    for _ in range(passes):
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a

            end_temp=1e-8,
            steps=250_000,
        )
        if score_ann < best_score:
            best_score = score_ann
            best_seq = a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: score={score_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Light diffusion and a short Adam‑sub‑gradient polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # diffusion passes
        best_seq = _diffuse(best_seq, passes=8)

        # one final short Adam run (focus on the single global peak)
        final_deadline = min(deadline, time.time() + 12.0)
        a_final, score_final = _run_subgradient_descent(
            best_seq,
            final_n,
            final_deadline,
            start_time,
            best_score,
            max_iters=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final sub‑grad improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Optional cheap random restarts (if any time left)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # small perturbation of the current best
        pert = best_seq + 0.015 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # short sub‑gradient run
        rest_deadline = time.time() + 20.0
        a_rst, score_rst = _run_subgradient_descent(
            pert,
            final_n,
            rest_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        if time.time() < deadline:
            # brief annealing on the perturbed solution
            a_rst, score_rst = _run_pair_annealing(
                a_rst,
                final_n,
                deadline,
                start_time,
                best_score,
                a_rst,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if score_rst < best_score:
            best_score = score_rst
            best_seq = a_rst.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={score_rst:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # (optional) sanity check – uncomment the following line if you wish
    # print("Final score (internal):",
    #       _score_from_conv(_conv_self(best_seq, _next_pow2(2 * final_n - 1)), final_n))

    return [float(x) for x in best_seq]
