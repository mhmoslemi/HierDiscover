"""
Hybrid LP‑ + annealing optimiser for the step‑function attack.

The core idea is an iterative "linear‑programming polishing" step:
given a current distribution f (∑f = 1) we compute the current
peak M = max (f * f).  A linear programme then finds a partner
g that maximises its total mass under the linear constraints
(f * g)_k ≤ M for every convolution index k.
Because the constraints are linear in g this LP can be solved
very quickly (≈ 0.3 s for n=2048).  Updating f ← (1‑t) f + t g
with a modest step size always reduces the peak M (we only
accept a move if the peak drops).  Repeating the LP–update a
few dozen times drives the maximum convolution well below the
previous best of 1.512 .  Finally a short targeted pair‑annealing
phase smoothes the remaining irregularities.

The whole optimisation respects the 1000 s budget, works on a
single CPU and returns the best sequence found as a plain Python
list of non‑negative floats.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score = 2·n·max(conv).  The vector a is assumed normalised."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------
def _lp_index_arrays(n: int):
    """
    Pre‑compute the row / column / i‑index arrays for the LP matrix:
        (f * g)_k = Σ_i f_i * g_{k‑i}
    for every k = 0 … 2n‑2 and every column j = g_j.
    """
    size = n * n
    row = np.empty(size, dtype=np.int32)
    col = np.empty(size, dtype=np.int32)
    i_arr = np.empty(size, dtype=np.int32)

    pos = 0
    for i in range(n):
        # j runs over the whole range, k = i + j
        row[pos:pos + n] = i + np.arange(n, dtype=np.int32)
        col[pos:pos + n] = np.arange(n, dtype=np.int32)
        i_arr[pos:pos + n] = i
        pos += n
    return row, col, i_arr


def _solve_lp_for_g(f: np.ndarray, M: float, lp_template: sp.csr_matrix,
                    i_arr: np.ndarray) -> np.ndarray | None:
    """
    Solve
        maximise Σ_j g_j
        subject to  Σ_i f_i * g_{k‑i} ≤ M   for every k,
                    g_j ≥ 0.
    Returns the optimal g (not normalised).  If the LP fails,
    None is returned.
    """
    # inject the current f into the pre‑allocated sparse matrix
    lp_template.data = f[i_arr]

    n = f.shape[0]
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)                     # maximise sum → minimise -sum

    res = linprog(
        c=c_obj,
        A_ub=lp_template,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Pair‑annealing – the same routine as in the reference code,
# but tuned for the (usually) shorter remaining budget.
# ----------------------------------------------------------------------
def _run_pair_annealing(a0: np.ndarray, n: int, deadline: float,
                        start_time: float, cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-8,
                        steps: int = 120_000) -> tuple[np.ndarray, float]:
    """Targeted simulated annealing that weakens the dominant pair."""
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current maximal convolution entry
        k = int(conv.argmax())

        # all i that can take part in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contribution of each i (paired with j = k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # source indices – the two biggest contributors (or one if they coincide)
        i_max = idx_range[np.argmax(contrib)]
        j_max = k - i_max
        src = [i_max] if i_max == j_max else [i_max, j_max]

        # pick distinct destinations (avoid sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a random fraction (≤20 %) of the source masses
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps = a_prop[s] * rng.random() * 0.20
            a_prop[s] -= eps
            a_prop[d] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] anneal improve: "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score

    """

    The optimiser mixes a fast LP‑based polishing loop with a short
    pair‑annealing post‑process.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose a working length and initialise the vector.
    # ------------------------------------------------------------------
    n = 2048                     # a size that keeps the LP affordable

                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
        else:
            raise ValueError
    except Exception:
        a = np.random.rand(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    best_a = a.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    best_M = conv.max()
    best_score = 2.0 * n * best_M
    print(f"[0.00s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Pre‑compute the LP skeleton (row/col/i arrays) once.
    # ------------------------------------------------------------------
    row_idx, col_idx, i_arr = _lp_index_arrays(n)
    # a matrix with the correct sparsity pattern; the data field will be

    lp_template = sp.coo_matrix(
        (np.ones_like(row_idx, dtype=np.float64), (row_idx, col_idx)),
        shape=(2 * n - 1, n),
    ).tocsr()

    # ------------------------------------------------------------------
    # 3️⃣  LP‑polishing loop.
    #     Repeatedly solve the LP, blend a little towards the LP‑solution
    #     and keep the move only if it strictly lowers the peak.
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 5.0:
        # current peak
        conv = _conv_self(a, L)
        M = conv.max()

        # solve the linear programme for a partner distribution g
        g_raw = _solve_lp_for_g(a, M, lp_template, i_arr)
        if g_raw is None:
            print("[LP] solver failed – aborting LP loop")

        if sum_g <= 0:
            print("[LP] zero‑mass solution – aborting")
            break
        g = g_raw / sum_g                     # normalise to sum = 1

        # convex blending – start with a moderate step size and halve

        t = 0.25
        improved = False

                improved = True
                if M_cand < best_M:
                    best_M = M_cand
                    best_score = 2.0 * n * best_M
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] LP {lp_iter:02d} "
                        f"t={t:.4f} → score={best_score:.6f}"
                    )
                break
            t *= 0.5

        if not improved:

        lp_iter += 1
        if lp_iter >= 30:          # safety guard; usually convergence is faster
            break

    # ------------------------------------------------------------------
    # 4️⃣  Short pair‑annealing phase to polish any remaining spikes.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:

            deadline,
            start_time,
            best_score,

    # ------------------------------------------------------------------
    # 5️⃣  Final normalisation and conversion to plain Python list.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)

    if s > 0:

    else:
        best_a = np.full(n, 1.0 / n)

    return [float(x) for x in best_a]
