"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a, a)) / (sum(a)²)

The algorithm mixes three ideas:

1. **Sidon‑set initialisation** – we build a set of indices whose pairwise
   sums are all distinct (a Sidon/B₂ set).  Giving each of those positions
   equal weight already reduces the peak of the self‑convolution because
   each convolution index receives at most one pair of non‑zero entries.

2. **Linear‑programming refinement with line‑search** – if ``cvxpy`` is
   available we solve the same LP as in the reference code, but instead of
   trying a few fixed mixing coefficients we perform a cheap 1‑D line search
   on the convex combination of the current vector and the LP‑solution.
   This usually yields a larger drop in the peak.

3. **Smooth Adam‑descent** – we replace the sub‑gradient steps by a
   smoothed approximation of the ``max`` (soft‑max with a temperature
   τ).  The gradient of the smooth surrogate is computed by a weighted
   convolution and an Adam optimiser updates the vector, after which we
   project back onto the probability simplex.  τ is slowly annealed,
   allowing the optimiser to first move quickly on a smooth landscape and
   later focus on the true max‑objective.

All stages respect the global time budget (default 1000 s).  The best
vector seen so far is returned as a plain list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (no external dependencies)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility used by the FFT based convolution."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)[: 2 * n - 1]
    return conv


def _smooth_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Soft‑max weights for a smooth approximation of max(conv).
    Returns a vector w of the same length as conv that sums to 1.
    """
    if tau <= 0:
        # avoid division by zero – fall back to a hard max
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        return w
    # shift for numerical stability
    shifted = conv - conv.max()
    expv = np.exp(shifted / tau)
    sumexp = expv.sum()
    if sumexp == 0:
        return np.full_like(conv, 1.0 / conv.size)
    return expv / sumexp


def _gradient_smooth(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth max surrogate w.r.t. a.
    It is 2 * (w *reversed* a) where w are the soft‑max weights.
    """
    n = a.shape[0]
    w = _smooth_weights(conv, tau)               # length 2n‑1
    a_rev = a[::-1]                              # reversed a
    fft_len = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=fft_len)
    A = np.fft.rfft(a_rev, n=fft_len)
    c = np.fft.irfft(W * A, n=fft_len)[: len(w) + n - 1]
    # slice that corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * c[n - 1 : n - 1 + n]
    # numerical safety
    grad = np.real_if_close(grad, tol=1e-12)
    return grad


def _generate_sidon_set(n: int, target_k: int, rng: random.Random) -> np.ndarray:
    """
    Greedy random construction of a Sidon (B₂) set: all pairwise sums are
    distinct.  Returns a binary mask of length n with ``target_k`` ones
    (or as many as could be placed after a reasonable number of trials).
    """
    best_mask = None
    best_size = 0
    max_trials = 200
    for _ in range(max_trials):
        order = list(range(n))
        rng.shuffle(order)
        used_sums = set()
        mask = np.zeros(n, dtype=bool)
        for pos in order:
            # compute all sums with already selected positions (including self)
            conflict = False
            for p in np.where(mask)[0]:
                if (pos + p) in used_sums:
                    conflict = True
                    break
                if (2 * pos) in used_sums:   # self‑sum
                    conflict = True
                    break
            if conflict:
                continue
            # no conflict: add the element
            mask[pos] = True
            for p in np.where(mask)[0]:
                used_sums.add(pos + p)
            if mask.sum() >= target_k:
                break
        if mask.sum() > best_size:
            best_size = mask.sum()
            best_mask = mask.copy()
        if best_size >= target_k:
            break
    return best_mask if best_mask is not None else np.zeros(n, dtype=bool)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine follows the three‑stage
    strategy described in the module docstring and respects the supplied
    time budget (in seconds).  The returned value is a plain list of
    floats.
    """
    np.random.seed(seed)
    rng = random.Random(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # -------------------------------------------------
    # Problem size – 2048 gives a good trade‑off.
    # -------------------------------------------------
    n = 2048

    # -------------------------------------------------
    # Initial vector
    # -------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                if tmp.size != n:
                    # linear interpolation to the required length
                    old_x = np.linspace(0.0, 1.0, tmp.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    tmp = np.interp(new_x, old_x, tmp)
                a = tmp / tmp.sum()
            else:
                a = np.full(n, 1.0 / n)
        except Exception:
            a = np.full(n, 1.0 / n)
    else:
        # Sidon‑set initialisation (target ~55 marks)
        mask = _generate_sidon_set(n, target_k=55, rng=rng)
        if mask.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = np.zeros(n, dtype=float)
            a[mask] = 1.0 / mask.sum()

    best_seq = a.copy()
    conv_a = _conv_self(a)
    best_score = float(2.0 * n * conv_a.max())
    print(f"[{time.time() - start_time:6.2f}s] initial score {best_score:.6f}")

    # -------------------------------------------------
    # Helper to build the convolution matrix needed by the LP.
    # -------------------------------------------------
    def _build_T(vec: np.ndarray) -> np.ndarray:
        T = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            T[i:i + n, i] = vec
        return T

    # -------------------------------------------------
    # Stage 1 – LP balancing with refined line search (if cvxpy available)
    # -------------------------------------------------
    try:
        import cvxpy as cp
        # reserve about half of the remaining budget for the LP stage
        lp_time = (deadline - time.time()) * 0.5
        lp_deadline = time.time() + lp_time
        iteration = 0
        while time.time() < lp_deadline:
            iteration += 1
            conv_a = _conv_self(a)
            M = conv_a.max()
            T = _build_T(a)

            b = cp.Variable(n, nonneg=True)
            constraints = [T @ b <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=300, verbose=False)
            except Exception:
                print("[LP] Solver failure – exiting LP stage.")
                break

            if b.value is None:
                print("[LP] No solution returned – exiting LP stage.")
                break

            b_vec = np.maximum(b.value, 0.0)
            sum_b = b_vec.sum()
            if sum_b <= 0.0:
                break
            g = b_vec / sum_b               # normalised partner

            # -------------------------------------------------
            # Find optimal convex combination using a cheap line search
            # -------------------------------------------------
            conv_g = _conv_self(g)
            conv_ag = _conv_two(a, g)

            t_candidates = np.linspace(0.0, 1.0, 21)   # 0,0.05,…,1
            best_local_a = a
            best_local_score = 2.0 * n * conv_a.max()

            for t in t_candidates:
                # (1‑t)a + t g  expanded in convolution space
                mixed_conv = ((1.0 - t) ** 2) * conv_a + \
                             2.0 * t * (1.0 - t) * conv_ag + \
                             (t ** 2) * conv_g
                cand_score = 2.0 * n * mixed_conv.max()
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local_a = (1.0 - t) * a + t * g
                    best_local_a = _project_simplex(best_local_a)

            a = best_local_a
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP iter {iteration:<3d} "
                      f"improve → {cur_score:.6f}")

            if iteration >= 30:   # enough for a good start
                break
    except Exception:
        # cvxpy not available – simply skip the LP stage.
        print("[LP] cvxpy not available – skipping LP balancing.")

    # -------------------------------------------------
    # Stage 2 – Adam optimiser on the smooth max surrogate
    # -------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.5                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the soft‑max (starts relatively high)
    tau = max(_conv_self(a).max(), 1e-6) * 0.5
    tau_decay = 0.9995            # multiplicative decay per iteration

    iter_adam = 0
    sub_deadline = deadline - 2.0      # keep a bit of time for a final polishing
    while time.time() < sub_deadline:
        iter_adam += 1
        conv_a = _conv_self(a)
        cur_score = 2.0 * n * conv_a.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter {iter_adam:<6d} "
                  f"improve → {cur_score:.6f}")

        # gradient of the smooth surrogate
        grad = _gradient_smooth(a, conv_a, tau)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iter_adam)
        v_hat = v / (1.0 - beta2 ** iter_adam)

        step = lr / (np.sqrt(v_hat) + eps_adam)
        a = a - step * m_hat
        a = _project_simplex(a)

        # anneal temperature
        tau = max(tau * tau_decay, 1e-9)

    # -------------------------------------------------
    # Stage 3 – short polishing with exponentiated gradient (as in the reference)
    # -------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        it = 0
        while time.time() < deadline:
            it += 1
            conv_a = _conv_self(a)
            cur_score = 2.0 * n * conv_a.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG iter {it:<6d} "
                      f"improve → {cur_score:.6f}")

            # simple sub‑gradient of the true max
            max_val = conv_a.max()
            max_inds = np.where(conv_a >= max_val - 1e-12)[0]
            grad = np.zeros_like(a)
            idx = np.arange(n)
            for k in max_inds:
                i = k - idx
                mask = (i >= 0) & (i < n)
                grad[mask] += 2.0 * a[i[mask]]
            if max_inds.size > 1:
                grad /= max_inds.size

            eta = 0.20 / math.sqrt(it)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

    # -----------------------------------------------------------------
    # Final normalisation and return a plain Python list.
    # -----------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_obj = float(2.0 * n * _conv_self(best_seq).max())
    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {final_obj:.6f}")
    return [float(x) for x in best_seq]
