```python
import numpy as np
import time
import math

def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def conv_full(a):
    n = a.shape[0]
    L = next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

def project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    # Use algorithm O(n log n)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
```

```python
def adam_smoothing(a, n, deadline, alphas, lr0=0.6, iters_per_alpha=10000):
    # a is probability vector sum 1.
    best_a = a.copy()
    best_score = 2 * n * conv_full(a).max()  # since sum=1
    L = next_pow2(2*n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1
            # compute convolution
            conv = conv_full(a)
            cur_score = 2 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Adam improve: iter={it}, score={cur_score:.6f}")
            # smooth max via log-sum-exp
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # convolution of a with reversed w
            w_rev = w[::-1]
            # compute gradient: derivative of smooth max wrt a
            # using convolution property: grad = 2 * (a * w_rev) (convolution)
            # Actually we need correlation: sum_j w_j * a_{k-j}
            # Use FFT: grad_i = 2 * sum_{j} w_j * a_{i + offset - j}? The original code uses correlation.
            # We'll adopt same as original.
            Af = np.fft.rfft(a, n=L)
            W = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * W, n=L)[:n + len(w_rev) - 1]
            # The gradient for each a_i is 2 * corr[ len(w_rev)-1 - i ] as in original.
            grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1-beta1_pow)
            v_hat = v/(1-beta2_pow)
            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # ensure nonnegative and sum to 1 via simplex projection
            a = project_simplex(a)
    return best_a, best_score
```

```python
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine n

    if time.time() < deadline:

            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time()-start_time:.2f}s] phase2 improve: score={score_exp:.6f}")

    # Phase 3: Greedy peak reduction
    if time.time() < deadline:

        phase3_deadline = min(deadline, time.time() + remaining * 0.6)

            best_seq = a_greedy.copy()

    while time.time() < deadline - 5.0:

            best_seq = pert_seq.copy()

    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:

    print(f"[{time.time()-start_time:.2f}s] final score approx {best_score:.6f}")
    return [float(x) for x in best_seq]
```

"""
Propose a step‑function with a very low autocorrelation peak.

The algorithm runs inside the 1000 s budget and consists of
four stages, each respecting the remaining time:

1. **Adam on a smoothed max‑convolution** – a log‑sum‑exp proxy is
   optimized with Adam, keeping the vector on the probability simplex.
   This quickly shapes a good global profile.

2. **Exponentiated‑gradient descent** – a true max‑convolution is used.
   The update is performed on the probability simplex by multiplicative
   mirror descent.

3. **Targeted greedy peak‑shaving** – repeatedly locate the index of
   the maximal convolution value, pick the component that contributes
   most to this peak and move a small amount of mass to a component that
   contributes the least.  The convolution is updated incrementally
   (O(n) per move) and the peak is reduced until the time expires.

4. **Random‑restart polishing** – if time is left, the current best
   sequence is perturbed and a short round of stages 2 + 3 is applied.
   Improvements are kept.

All vectors are kept normalised (sum = 1); the evaluation metric then
reduces to `2·n·max(conv)`.  The code reports intermediate scores and
finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 15000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_full(a).max()

    L = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_full(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ---- smooth max (log‑sum‑exp) ----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ---- gradient of the smoothed max ----
            Af = np.fft.rfft(a, n=L)
            W = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * W, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ---- Adam update ----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_full(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # d/d a_i of a_i*a_{k-i}

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the probability simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score

# Phase 3 – Deterministic greedy reduction of the convolution peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_full(a)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000          # keep exactness in check

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore current maximum)
        tmp = conv[max_idx]
        conv[max_idx] = -np.inf
        second_max = conv.max()
        conv[max_idx] = tmp

        # indices that actually contribute to this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        partner = max_idx - idx
        contrib = a[idx] * a[partner]
        src = idx[np.argmax(contrib)]

        # choose a destination where the partner value is smallest
        a_partner = a[partner]                    # values paired with max_idx
        min_val = a_partner.min()
        dst_candidates = idx[(a_partner == min_val) & (idx != src)]
        if dst_candidates.size == 0:
            # fall back to a random different index
            dst = src
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partners for src and dst in the current peak
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        a_old = a.copy()

        # ----- remove epsilon from src (incremental convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[2 * src] += eps * eps

        # update the vector
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # ----- add epsilon at dst -----
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * dst] -= eps * eps

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_full(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score
