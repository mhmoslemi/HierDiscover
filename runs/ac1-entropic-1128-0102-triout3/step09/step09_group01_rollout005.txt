    n = a.shape[0]
    a_rev = a[::-1]  # reversed
    # Convolution via FFT:

def _adam_optimize(a0, n, deadline, start_time, best_score):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)  # length 2n-1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # compute smooth max weight
        # schedule alpha
        alpha = 5.0 + (500.0 - 5.0) * ((time.time() - start_time) / (deadline - start_time))

        # compute w
        w = np.exp(alpha * conv)

        # adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)

def _adam_optimize(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05

    L_conv = _next_pow2(2*n - 1)
    L_grad = _next_pow2(3*n - 2)

    it = 0
    while time.time() < deadline:
        it += 1

        # compute conv
        conv = _conv_self(a, L=L_conv)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # alpha schedule increase
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = 5.0 + 495.0 * progress  # from 5 to 500

        # compute smooth weights
        w = np.exp(alpha * conv)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum

        # gradient
        grad = _gradient_from_weights(a, w, L_grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 * (1 - progress)  # decaying lr
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score

    start_time = time.time()

    n = 4096
    L_fft = _next_pow2(2*n - 1)

    # baseline uniform
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _eval_score(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # try global starter if defined
    try:
        starter_arr = np.asarray(height_sequence_1, dtype=float)

            starter_arr = starter_arr / starter_arr.sum()
            cur = _eval_score(starter_arr, L_fft)
            if cur < best_score:
                best_score = cur

        pass

    # Phase: Adam optimizer

"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 (the size used by the reference solution).

The algorithm consists of three stages

1️⃣  Initialise from the previously discovered best sequence (``height_sequence_1``)
    or from the uniform distribution if that variable is unavailable.

2️⃣  A projected Adam optimiser on a *smoothed* version of the objective.
    The smooth max is approximated by a soft‑max with a temperature α
    that is increased from 5 to 500 during the optimisation (continuation).
    Gradients are obtained analytically via a cross‑correlation
    (implemented with FFTs) and the iterate is projected onto the
    probability simplex after every step.

3️⃣  A short mass‑transfer simulated‑annealing polishing phase
    (tiny random moves between two coordinates) to escape any remaining
    shallow local minima.

All heavy work is performed with FFT‑based convolutions (O(n log n)),
the whole routine respects a 1000 s wall‑clock budget and returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full convolution (no truncation).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum(a)=1 → score = 2·n·max(conv))
# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray, L_conv: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_conv)
    max_conv = conv.max()
    n = a.shape[0]
    if max_conv < 0:
        return float("inf")
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha*conv_i) / sum_j exp(alpha*conv_j)."""
    # conv is non‑negative, alpha is modest → no overflow worries
    w = np.exp(alpha * conv)
    s = w.sum()
    if s == 0.0:
        # fall back to uniform – should never happen
        return np.full_like(conv, 1.0 / conv.size)
    return w / s


def _grad_from_weights(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of f(a)=∑_i w_i (a∗a)_i.
    For non‑negative a we have ∇f = 2 * cross‑correlation(a, w).
    Implemented via FFT as 2 * (rev(a) ∗ w) and slicing the central part.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # FFT of the two operands (zero‑padded to length L_grad)
    A_rev = np.fft.rfft(a_rev, n=L_grad)
    W = np.fft.rfft(w, n=L_grad)

    # full linear convolution of a_rev and w
    conv_rev_full = np.fft.irfft(A_rev * W, n=L_grad)
    needed = 3 * n - 2
    if conv_rev_full.shape[0] < needed:
        conv_rev_full = np.concatenate(
            [conv_rev_full, np.zeros(needed - conv_rev_full.shape[0])]
        )
    else:
        conv_rev_full = conv_rev_full[:needed]

    start = n - 1                      # centre of the full convolution
    grad = 2.0 * conv_rev_full[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with simplex projection.
    The objective is the smooth max of the self‑convolution, controlled
    by a temperature parameter α that is increased linearly from 5 to 500.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    L_conv = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- exact objective (for bookkeeping) -----
        conv = _conv_self(a, L_conv)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ----- continuation schedule for α -----
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = 5.0 + 495.0 * progress          # linear from 5 → 500

        # ----- smooth weights & gradient -----
        w = _smooth_weights(conv, alpha)
        grad = _grad_from_weights(a, w, L_grad)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr_start * (1.0 - progress)       # slowly decay learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)                # stay on the probability simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-7,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealer: move a small fraction of the mass from a
    random source index to a random destination, accepting moves that
    improve the exact objective or with a Metropolis probability.
    """
    a = a0.copy()
    L_conv = _next_pow2(2 * n - 1)

    # current exact score
    conv = _conv_self(a, L_conv)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the proposal
        conv_prop = _conv_self(a_prop, L_conv)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
