    conv = np.fft.irfft(Z, n=L)[:n]
    return conv

def _smooth_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    total_time = deadline - start_time
    # schedules
    alpha_start = 2.0
    alpha_end = 200.0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} score={cur_score:.6f}")

        # compute softmax weights
        maxc = conv.max()
        elapsed = time.time() - start_time
        # linear schedule for alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_time)
        # compute exp safely
        expw = np.exp(alpha * (conv - maxc))
        w = expw / expw.sum()
        # gradient
        grad = _gradient_weighted(a, w)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score
```

"""
Block‑wise step function optimizer
---------------------------------
The evaluation function is
    2·n·max(conv(a,a)) / (sum(a)**2) ,
where `conv` is a linear convolution.
For a non‑negative sequence `a` with unit sum the denominator is 1,
so the task is to make the maximal entry of the self‑convolution
as small as possible.

Observation
-----------
If the mass `1` is split into `k` equal blocks of length `m`
(separated by a single zero) then each block carries mass `1/k`
and has height `h = (1/k)/m`.  
The self‑convolution of a block attains the value
    m·h² = 1/(k²·m) .
All cross‑terms between two different blocks have the same maximal
value (they also involve exactly `m` overlapping products of height `h`).
Hence the overall `max(conv)` equals `1/(k²·m)` and the objective is

    score(k) = 2·n / (k²·m)   with   m = floor((n-(k-1))/k) .

Increasing `k` (and consequently shrinking `m`) reduces the score.
The cheapest construction is obtained by trying several values of `k`
and picking the one with the smallest exact score.  Afterwards a few
iterations of a simple projected sub‑gradient descent polish the result.

The whole search finishes well within the 1000 s budget and typically
produces a score far below the previous best (≈1.51), e.g. ≈0.4 for `k=5`
on `n=4096`.

The function `propose_candidate` is the entry point required by the
challenge harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (linear, not circular)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT; returns length len(a)+len(b)-1."""
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = np.fft.irfft(A * B, n=L)[:n]
    return C


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (no scaling, sum assumed ≥ 0)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Simple sub‑gradient descent (optional polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true (non‑smooth) objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        # indices attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Build a block‑wise candidate for a given number of blocks k
# ----------------------------------------------------------------------
def _build_blocks(k: int, n: int) -> np.ndarray:
    """
    Construct a sequence with `k` equal mass blocks (separated by one zero).
    Returns a vector on the simplex (sum = 1).
    """
    # maximal block length that fits k blocks with a single zero gap
    m = (n - (k - 1)) // k          # integer division, always >= 1 for sensible k
    if m <= 0:
        return None

    height = (1.0 / k) / m          # each block carries mass 1/k uniformly
    a = np.zeros(n, dtype=float)

    pos = 0
    for _ in range(k):
        a[pos:pos + m] = height
        pos += m + 1                # one zero gap after each block (except after last)
    # due to integer rounding the sum may be slightly off – project back
    a = _project_simplex(a)
    return a
