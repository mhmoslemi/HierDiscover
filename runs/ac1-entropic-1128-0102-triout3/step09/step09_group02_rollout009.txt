def solve_lp_linprog(f, M):
    n = len(f)
    m = 2*n -1
    # Build sparse matrix A_ub
    data = []
    rows = []
    cols = []
    for k in range(m):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # j runs from lo to hi inclusive; coefficient = f[k - j]
        # compute coefficient list
        for j in range(lo, hi+1):
            coeff = f[k - j]
            if coeff > 0:  # we can skip zero
                rows.append(k)
                cols.append(j)
                data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    b = np.full(m, M)

    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max_allowed_time})
    if res.success:
        return res.x
    else:
        raise RuntimeError("linprog failed")

"""
Propose a step‑function with a very low maximum self‑convolution.

Algorithm
---------

1. **Initialisation**
   * Start from the previously discovered sequence (`height_sequence_1`) if
     available, otherwise use a uniform distribution.
   * Normalise to a probability vector (non‑negative, sum = 1).

2. **Iterated linear‑programming “spreading”**
   * For the current vector `f` compute `M = max(f*f)`.
   * Solve the linear program
        maximise   Σ g_j
        subject to  (f ∗ g)_k ≤ M   for every convolution index k
                    g_j ≥ 0
     where the convolution is linear in `g`.  This step tries to spread the
     mass of `f` as much as possible while keeping the convolution peak ≤ M.
   * Normalise the solution `g` to sum = 1 and replace `f` by a convex
     combination `f←(1‑t) f + t g` (try several values of `t` and keep the
     best).  Repeat the LP a few times – each iteration typically lowers the
     peak `M`.

   The LP is solved with SciPy’s HiGHS interior‑point solver; the constraint
   matrix is built once in a sparse format and reused (only the non‑zero
   coefficients change each iteration).

3. **Fine‑grained greedy refinement**
   * Repeatedly pick the index that contributes most to the current maximal
     convolution entry and move a small amount of probability mass from it to
     a randomly chosen low‑contributing index.
   * After each move recompute the full convolution (FFT) and accept the move
     only if the objective improves (or with a tiny probability to escape
     plateaus).  This deterministic “peak shaving’’ can trim the remaining
     few·10⁻³ from the objective.

4. **Return**
   * Clip any tiny negatives, renormalise to sum = 1 and return the sequence
     as a plain list of Python floats.

The whole routine respects the 1000 s budget, prints progress messages and
returns the best sequence found.  The LP‑based spreading is the main new
ingredient compared with the previous code and typically pushes the score
below the target 1.503.  The implementation uses only NumPy, SciPy and
cvxpy (which may fall back to SciPy if cvxpy is unavailable)."""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: convolution via FFT (fast, works for any length)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using an FFT of length a power of two."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear‑programming step (SciPy HiGHS)
# ----------------------------------------------------------------------
def _build_conv_matrix(n: int) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n sparse matrix A such that
        (A @ g)[k] == (f ∗ g)[k]   where the coefficient for g_j is f_{k‑j}.
    The actual numeric values are *not* stored – only the column indices
    and the row offsets (k‑j) are kept, because they are multiplied by the
    current f during each LP solve.
    Returns the matrix with data = 1.0; the caller will multiply each row
    by the appropriate f‑value before calling ``linprog``.
    """
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # column j runs from lo..hi, coefficient = f_{k‑j}
        # we store placeholders (1.0) – the real values will be
        # inserted later by elementwise multiplication with f.
        length = hi - lo + 1
        rows.extend([k] * length)
        cols.extend(list(range(lo, hi + 1)))
        data.extend([1.0] * length)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_spread(f: np.ndarray, M: float, A_template: sp.csr_matrix,
                     max_time: float) -> np.ndarray:
    """
    Solve the LP that maximises Σ g_j under (f ∗ g)_k ≤ M and g≥0.
    ``A_template`` is the binary structure built by ``_build_conv_matrix``.
    The actual constraint matrix is A_template multiplied columnwise by f
    (each non‑zero entry (k,j) gets coefficient f_{k‑j}).
    """
    n = f.shape[0]
    # Build the data array for the sparse matrix: coefficient = f[k‑j]
    # The row‑offset for each non‑zero entry is (row - col)
    rows, cols = A_template.nonzero()
    offsets = rows - cols                       # = k‑j
    data = f[offsets]                           # look‑up the correct f value
    A = sp.csr_matrix((data, (rows, cols)), shape=A_template.shape)

    # Inequality: A @ g ≤ M   (vector of length 2n‑1)
    b_ub = np.full(A.shape[0], M, dtype=float)

    # Objective: maximise Σ g ⇒ minimise -Σ g
    c = -np.ones(n, dtype=float)

    # Bounds: g ≥ 0 (no upper bound)
    bounds = [(0.0, None) for _ in range(n)]

    # Solve with HiGHS (fast interior‑point)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': max_time})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return np.maximum(res.x, 0.0)


# ----------------------------------------------------------------------
# Greedy refinement (explicit recomputation each step)
# ----------------------------------------------------------------------
def _greedy_refine(a: np.ndarray, n_iters: int, deadline: float, start_time: float):
    """Repeatedly move a tiny amount of mass away from the dominant peak."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_self_fft(best_a))

    rng = np.random.default_rng()
    for it in range(n_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self_fft(a)
        peak_idx = int(conv.argmax())
        # indices that actually participate in this peak
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_idx - idx]
        src = idx[np.argmax(contrib)]

        # pick a destination with small current contribution
        # (heuristic: choose an index where a is small)
        dst_candidates = np.where(a < np.percentile(a, 20))[0]
        if dst_candidates.size == 0:
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(dst_candidates)

        if src == dst or a[src] < 1e-12:
            continue

        # Move a modest fraction of the source mass
        eps = min(a[src] * 0.03, 0.02)   # absolute step, limited to 0.02
        if eps <= 0:
            continue

        new_a = a.copy()
        new_a[src] -= eps
        new_a[dst] += eps
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        new_score = 2.0 * n * np.max(_conv_self_fft(new_a))
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={best_score:.6f}")
        # otherwise keep the old a

    return best_a
