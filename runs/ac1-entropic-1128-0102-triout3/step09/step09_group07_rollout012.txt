def compute_obj_and_grad(x, alpha, n):
    # x is array length n
    # compute a
    a = np.exp(x - np.max(x))  # to avoid overflow
    a_sum = a.sum()
    a = a / a_sum
    # conv
    Lc = _next_pow2(2*n - 1)
    A = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(A * A, n=Lc)[:2*n-1]
    # compute w via softmax of alpha*conv
    alpha_conv = alpha * conv
    max_alpha_conv = np.max(alpha_conv)
    w = np.exp(alpha_conv - max_alpha_conv)
    w_sum = w.sum()
    w = w / w_sum
    # objective
    obj = (max_alpha_conv + np.log(w_sum)) / alpha  # log-sum-exp/alpha
    # gradient wrt a: 2*conv(w, a_rev) (full conv length 3n-2); need a_rev
    a_rev = a[::-1]
    Lg = _next_pow2(3*n - 2)
    W = np.fft.rfft(w, n=Lg)
    A_rev = np.fft.rfft(a_rev, n=Lg)
    grad_full = np.fft.irfft(W * A_rev, n=Lg)[:3*n-2]  # length 3n-2
    # extract middle n entries
    start = n-1
    grad_a = 2.0 * grad_full[start:start+n]
    # gradient wrt x (softmax)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    # Return objective (scalar) and gradient (vector)
    return obj, grad_x

"""
Propose a step‑function that minimises the evaluation metric
    score = 2·n·max(a∗a)   (with Σa = 1).

The algorithm consists of three parts:

1️⃣  **Smooth‑max L‑BFGS optimisation** – we minimise a smooth
    surrogate of the max (log‑sum‑exp) using a soft‑max parameterisation
    a_i = exp(x_i)/Σexp(x).  The surrogate is gradually sharpened by
    increasing the temperature α (0.5 → 200).  Gradients are obtained
    analytically via FFTs (O(n log n)) and fed to SciPy’s L‑BFGS‑B.
    Several random restarts are performed – the best result is kept.

2️⃣  **Optional up‑sampling** – the best vector is linearly interpolated
    to a larger length (if time permits) and refined again with the
    same L‑BFGS schedule.  A larger n gives extra degrees of freedom
    for lowering the peak of the self‑convolution.

3️⃣  **Fine‑grained mass‑shifts** – an O(n) greedy refinement moves mass
    from “source’’ indices that heavily contribute to the current
    maximal convolution bin to “destination’’ indices that contribute
    little.  The maximal admissible step size ε is computed analytically
    by solving a 1‑D linear programme, guaranteeing a genuine decrease
    of the peak in each accepted move.

The whole procedure respects a 1000 s wall‑clock budget and returns the
best sequence (list of non‑negative heights) found.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective and its gradient
# ----------------------------------------------------------------------
def _obj_and_grad(x: np.ndarray, alpha: float, n: int):
    """
    Returns (value, gradient) of the smooth surrogate:
        f(x) = (1/α)·log Σ_k exp(α·(a*a)_k)    with a = softmax(x)
    Gradient is with respect to x (the logits of the soft‑max).
    """
    # ----- a = softmax(x) -----
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()

    # ----- self‑convolution a*a -----
    Lc = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(A * A, n=Lc)[: 2 * n - 1]

    # ----- soft‑max over the convolution entries -----
    alpha_conv = alpha * conv
    max_ac = np.max(alpha_conv)
    w = np.exp(alpha_conv - max_ac)
    w_sum = w.sum()
    w = w / w_sum

    # ----- value of the surrogate (log‑sum‑exp / α) -----
    val = (max_ac + np.log(w_sum)) / alpha

    # ----- gradient w.r.t. a -----
    # grad_a[i] = 2 * Σ_k w_k a_{k-i}
    a_rev = a[::-1]
    Lg = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, n=Lg)
    A_rev = np.fft.rfft(a_rev, n=Lg)
    grad_full = np.fft.irfft(W * A_rev, n=Lg)[: 3 * n - 2]
    start = n - 1
    grad_a = 2.0 * grad_full[start:start + n]

    # ----- chain rule to get grad w.r.t. x (soft‑max Jacobian) -----
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return float(val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS optimisation for a given start vector
# ----------------------------------------------------------------------
def _run_lbfgs(a_start: np.ndarray,
               alphas,
               n,
               deadline: float,
               start_time: float):
    """
    Optimises the smooth surrogate for the sequence a_start.
    Returns the best a found and its true score (2·n·max conv).
    """
    # initialise logits
    x = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = 2.0 * n * _conv_self(best_a).max()

    for alpha in alphas:
        if time.time() >= deadline:
            break

        # allocate a modest number of L‑BFGS iterations based on remaining time
        remain = deadline - time.time()
        max_iter = max(20, int(min(200, remain * 150)))   # ~150 it/s on typical hardware

        def fun(logits):
            return _obj_and_grad(logits, alpha, n)

        res = scipy.optimize.minimize(fun,
                                      x,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': max_iter,
                                               'disp': False,
                                               'gtol': 1e-12})

        x = res.x
        # reconstruct a
        a = np.exp(x - np.max(x))
        a /= a.sum()

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS α={alpha:.1f} improve: score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy optimal incremental refinement (Phase 3 of the original code)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine(a0: np.ndarray,
                                    n: int,
                                    deadline: float,
                                    start_time: float,
                                    eps_start: float = 0.04,
                                    eps_end: float = 0.001,
                                    rel_improve: float = 1e-12,
                                    recompute_every: int = 5000):
    """
    Performs many tiny mass‑shifts.  For a selected source and destination
    the effect on the whole convolution is linear in ε, allowing us to
    compute the largest feasible ε that still lowers the current maximum.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0
    accepted = 0

    while time.time() < deadline:
        # linearly decay the fraction of mass we are allowed to move
        elapsed = time.time() - start_time
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal index
        k_max = int(np.argmax(conv))

        # ---- source: pick an index that contributes to the maximal bin ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0.0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- destination: prefer indices that appear rarely in the max bin ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                     # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- linear effect of moving ε mass from src to dst on the convolution ----
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a               # loss of src
        s[dst:dst + n] += 2.0 * a               # gain at dst

        s_max = s[k_max]
        if s_max >= 0.0:
            continue          # would not lower the current max

        # ---- admissible ε before another entry overtakes the current max ----
        mask = s > s_max                     # slopes larger than s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        eps_limit = a[src] * eps_frac
        epsilon = min(eps_star, eps_limit)

        if epsilon <= 0.0:
            continue

        # ---- apply the move analytically (O(n)) ----
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        if new_max < max_val - rel_improve:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1

            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] incr‑refine improve: iter={it:,}, max={best_max:.8f}")

        if accepted % recompute_every == 0 and accepted > 0:
            # re‑compute from scratch to avoid drift
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a
