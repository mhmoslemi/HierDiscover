"""
Improved minimiser for the self‑convolution bound.

The optimiser works in three stages:

1️⃣  A dense exponentiated‑gradient phase on a log‑sum‑exp (smooth‑max)
    surrogate of the convolution peak.  The temperature α is increased
    gradually (0.5 → 160) while a decaying learning‑rate η≈η₀/√t is used.
    This quickly drives the sequence towards a low‑peak region while
    preserving non‑negativity via the exponential update and a simplex
    projection.

2️⃣  A fast Adam‑style projected sub‑gradient method on the true
    objective  f(a)=2·n·max(conv(a,a)).  Adam’s adaptive moments give a
    much larger effective step‑size than the vanilla sub‑gradient used
    previously, allowing more progress with the same compute budget.

3️⃣  A global pair‑annealing stage.  Instead of moving only a single
    pair, we move the mass of the *k* strongest contributors to the
    current maximal convolution entry to random locations.  The temperature
    schedule is exponential and the Metropolis rule is applied.

If time permits the routine performs cheap random‑restart perturbations,
each followed by a short Adam run, to escape possible local minima.

The whole search respects the 1000 s budget; after the deadline the best
sequence found is returned as a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value (sum of a is assumed to be 1)."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient on a smooth approximation of max
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      eta0: float = 0.4,
                      iters_per_alpha: int = 15000):
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate of max(conv).
    Returns the best iterate seen (lowest true score).
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score_from_conv(_conv_self(a, L), n)

    it = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG improve: iter={it:,}, "
                      f"score={cur_score:.6f}")

            # smooth‑max weights (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smooth max (2·corr)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: len(a) + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # exponentiated update (positive, multiplicative)
            eta = eta0 / math.sqrt(it)
            grad = np.clip(grad, 0.0, 10.0)          # avoid overflow
            a = a * np.exp(-eta * grad)

            # projection onto the simplex (preserves Σa = 1)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Adam‑style projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,

                          init_step: float = 0.2):

    Keeps the iterate in the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = _score_from_conv(conv, n)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={t:,}, "
                  f"score={cur_score:.6f}")

        # index of the current maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max at position k
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = init_step / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Global pair‑annealing (multi‑pair moves)
# ----------------------------------------------------------------------
def _run_multi_pair_annealing(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              cur_best_score: float,
                              cur_best_seq: np.ndarray,
                              start_temp: float = 5e-2,
                              end_temp: float = 1e-8,
                              steps: int = 300_000,
                              top_k: int = 3):
    """
    Simulated annealing that simultaneously weakens the *top_k*
    strongest contributors to the current maximal convolution entry.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = _score_from_conv(conv, n)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate a peak index
        k = int(conv.argmax())

        # all indices that can take part in this sum
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)

        # contributions of each pair (i, k‑i)
        contrib = a[idx_range] * a[k - idx_range]

        # indices of the top_k contributions
        top_idxs = idx_range[np.argpartition(contrib, -top_k)[-top_k:]]

        # source positions (unique – if i==j we keep it once)
        src = np.unique(np.concatenate([top_idxs, k - top_idxs]))

        # destination indices (avoid sources)
        dst = []
        while len(dst) < len(src):
            d = rng.integers(n)
            if d not in src and d not in dst:
                dst.append(d)

        # propose moving a fraction (≤15 %) of each source mass
        a_prop = a.copy()
        for s, d in zip(src, dst):
            if a_prop[s] < 1e-14:
                continue
            eps_move = a_prop[s] * rng.random() * 0.15
            a_prop[s] -= eps_move
            a_prop[d] += eps_move

        # evaluate
        conv_new = _conv_self(a_prop, L)
        new_score = _score_from_conv(conv_new, n)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: simple parametric families (used for a quick initial scan)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n)


def _shape_linear_ramp(n: int) -> np.ndarray:
    """Increasing linear ramp, mirrored to be symmetric."""
    half = np.arange(1, n // 2 + 1)
    ramp = np.concatenate([half, half[::-1] if n % 2 == 0 else half[:-1][::-1]])
    return ramp.astype(float)


def _shape_exponential(n: int, c: float) -> np.ndarray:
    """exp(‑c·|i‑centre|) shape."""
    centre = (n - 1) / 2.0
    d = np.abs(np.arange(n) - centre)
    return np.exp(-c * d)


def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n‑i)^p, symmetrised."""
    i = np.arange(n, dtype=float) + 1.0
    a = i ** p + (n - i + 1) ** p
    return a


def _scan_parametric_shapes(n: int, deadline: float, start_time: float):
    """
    Explore a few low‑dimensional families and keep the best.
    Returns (seq, score).
    """
    L = _next_pow2(2 * n - 1)
    best_seq = None
    best_score = float('inf')

    families = [
        ("uniform", _shape_uniform, [()]),
        ("ramp", _shape_linear_ramp, [()]),
        ("exp", _shape_exponential, [(0.001,), (0.003,), (0.01,), (0.03,), (0.1,)]),
        ("power", _shape_power, [(-1.0,), (-0.5,), (0.0,), (0.5,), (1.0,), (2.0,)]),
    ]

    for name, maker, params in families:
        for par in params:
            if time.time() >= deadline:
                return best_seq, best_score
            a = maker(n, *par) if par else maker(n)
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                continue
            a /= a.sum()
            conv = _conv_self(a, L)
            sc = _score_from_conv(conv, n)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] param scan ({name}{par}) "
                      f"score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Choose an initial vector (parametric scan → best) or fallback.
    # --------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)

    # try to reuse the global construction if it exists
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        pass

    # a quick parametric scan (takes only a few seconds)
    scan_deadline = time.time() + min(12.0, 0.02 * (deadline - start_time))
    seq_scan, score_scan = _scan_parametric_shapes(n, scan_deadline, start_time)

    if starter is not None:
        # Upsample the previous best to the target length
        a0 = np.interp(
            np.linspace(0, 1, n),
            np.linspace(0, 1, starter.size),
            starter,
        )
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # pick the better of the two candidates
    if seq_scan is not None and score_scan < _score_from_conv(_conv_self(a0, L), n):
        a0 = seq_scan
        print(f"[{time.time() - start_time:.2f}s] initial from parametric scan: "
              f"score={score_scan:.6f}")

    best_seq = a0.copy()
    best_score = _score_from_conv(_conv_self(best_seq, L), n)

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient on smooth max
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:  # we need at least a few seconds
        phase1_end = time.time() + max(8.0, 0.12 * remaining)
        alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            n,
            phase1_end,
            start_time,
            alphas=alphas,
            eta0=0.45,
            iters_per_alpha=20000,
        )
        if score_exp < best_score:
            best_seq, best_score = a_exp, score_exp
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_exp:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style sub‑gradient descent (major chunk of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_end = time.time() + max(20.0, 0.70 * remaining)
        a_adam, score_adam = _run_adam_subgradient(
            best_seq,
            n,
            phase2_end,
            start_time,
            init_step=0.25,
        )
        if score_adam < best_score:
            best_seq, best_score = a_adam, score_adam
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Global multi‑pair annealing (use the rest of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_anneal, score_anneal = _run_multi_pair_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-9,
            steps=400_000,
            top_k=4,
        )
        if score_anneal < best_score:
            best_seq, best_score = a_anneal, score_anneal
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_anneal:.6f}")

    # --------------------------------------------------------------
    # Optional random restarts (if any time left)
    # --------------------------------------------------------------
    while time.time() < deadline - 6.0:
        time_left = deadline - time.time()
        block = min(25.0, time_left / 2.0)
        restart_deadline = time.time() + block

        # perturb current best a little
        pert = best_seq + 0.015 * np.random.randn(n)
        pert = np.maximum(pert, 0.0)
        if pert.sum() == 0.0:
            pert = np.random.rand(n)
        pert /= pert.sum()

        # short Adam run
        pert_seq, pert_score = _run_adam_subgradient(
            pert,
            n,
            restart_deadline,
            start_time,
            init_step=0.20,
        )
        if pert_score < best_score:
            best_seq, best_score = pert_seq, pert_score
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return as plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python floats for the external evaluator
    return [float(v) for v in best_seq]
