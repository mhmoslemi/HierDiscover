        fft_a = np.fft.rfft(a, n=L)

        else:
            w = w / w_sum
        # compute gradient via correlation
        fft_w = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(fft_w) * fft_a, n=L)[: (2*n - 1) + n - 1]  # length 3n-2

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)

        theta = cssv[rho] / (rho + 1.0)

The algorithm works in two phases:

1.  A smooth Adam‑style optimisation on a log‑sum‑exp surrogate of the
    max‑convolution.  The smoothing parameter α is slowly increased,
    turning the surrogate from a soft average into a hard max.  After
    each gradient step the iterate is symmetrised and projected onto the
    probability simplex.

2.  A classic projected sub‑gradient descent (the method used in the
    reference solution) is applied to the best vector from phase 1 for
    the remaining time budget, providing a fine‑grained polish.

All vectors are kept non‑negative, sum to 1 and are symmetric, which is
known to be optimal for this problem.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection of v onto {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    Projected sub‑gradient descent for the true objective.
    Returns the best iterate seen before `deadline`.
    """

    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

        cur_max = conv.max()

        lo = max(0, k - (n - 1))

# Phase‑1: smooth Adam optimisation on a log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_adam_opt(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smoothed approximation of `max(conv)`.
    Returns the best vector seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # FFT length sufficient for correlation (len(w)+len(a)-1)
    L = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    # smoothing schedule
    alpha = 0.5                # start with a fairly soft max
    alpha_max = 200.0
    max_est_iters = 200_000
    alpha_inc = (alpha_max - alpha) / max_est_iters

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # FFT of the current iterate
        fft_a = np.fft.rfft(a, n=L)

        # self‑convolution to obtain the true maximum
        conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # smoothed max via log‑sum‑exp (weights sum to 1)
        exp_shift = conv - max_conv                 # ≤ 0
        w = np.exp(alpha * exp_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective:
        #   g_i = 2 * Σ_j w_{i+j} a_j   (valid correlation)
        fft_w = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(fft_w) * fft_a, n=L)[: 3 * n - 2]
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                     # gentle decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # enforce symmetry – the optimum is symmetric
        a_new = (a_new + a_new[::-1]) * 0.5

        # back onto the probability simplex
        a = _project_simplex(a_new)

        # evaluate the true (non‑smoothed) objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} score={cur_score:.6f}")

        # increase smoothing strength
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # leave a few seconds for the final sub‑gradient polish
        if time.time() > deadline - 12.0:
            break

    return best_a, best_score

# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence.

    Phase 1 (≈60 % of the time budget) performs a smooth Adam optimisation
    on a soft approximation of the max‑convolution.  Phase 2 (the remaining
    time) refines the result with the classic projected sub‑gradient descent
    used in the reference solution.

    The returned sequence is non‑negative, sums to 1 and is given as a plain
    Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # a small safety margin – the harness aborts after the deadline
    deadline_total = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – use the same canvas as the reference runner.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 0) Initialise from a known good candidate if it exists,
    #    otherwise start from a symmetric triangular (tent) shape.
    # --------------------------------------------------------------
    init = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                init = starter / s
                print(f"[{time.time() - start:.2f}s] Loaded global starter")
    except Exception:
        pass

    if init is None:
        x = np.arange(n, dtype=float)
        tent = np.minimum(x, n - 1 - x)          # linear tent, peak in the centre
        init = tent / tent.sum()
        print(f"[{time.time() - start:.2f}s] Using triangular start")

    # --------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation (≈60 % of the total budget)
    # --------------------------------------------------------------
    phase1_deadline = start + budget_s * 0.60
    if time.time() < phase1_deadline:
        best_seq, best_score = _smooth_adam_opt(init, phase1_deadline, start)
        print(f"[{time.time() - start:.2f}s] End of smooth phase, best={best_score:.6f}")
    else:
        best_seq = init
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – classic projected sub‑gradient descent for polishing
    # --------------------------------------------------------------
    if time.time() < deadline_total - 3.0:      # keep a couple of seconds for final normalisation
        best_seq, best_score = _subgradient_descent(
            best_seq, deadline_total, start
        )
        print(f"[{time.time() - start:.2f}s] Sub‑grad phase finished, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (the evaluator also projects, but we keep the
    # vector clean for readability).
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = _objective(best_seq)
    print(f"[{time.time() - start:.2f}s] Finished – final objective = {final_obj:.6f}")

    return [float(v) for v in best_seq]
