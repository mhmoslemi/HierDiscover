def compute_max_conv(a):
    n = len(a)
    fft_len = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The routine follows a three‑stage heuristic:

1. **Entropic (exponentiated‑gradient) descent** – many cheap
   multiplicative updates on the probability simplex.  The step size is
   deliberately small (∝1/√t) so we can approach the optimum much more
   closely than the reference implementation.

2. **Gradient‑guided mass transfer** – using the sub‑gradient of the
   current maximal convolution entry we move a fraction of mass from
   indices with largest gradient to those with smallest gradient.
   This directly attacks the active constraints that define the current
   peak.

3. **Simulated‑annealing polishing** – random transfers of a tiny
   amount of probability mass are accepted either if they improve the
   score or with a temperature‑controlled probability.  The temperature
   decays exponentially, allowing a final fine‑grained exploration of
   the neighbourhood of the best candidate found.

All stages respect the global time budget (default 1000 s) and keep the
best sequence seen so far.  The final answer is returned as a plain
list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / optimisation helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two (used for zero‑padding in FFT)."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (returns length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective value for a (assumes sum(a)=1)."""
    n = a.shape[0]
    maxb = _conv_self(a).max()
    return 2.0 * n * maxb


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (a ≥ 0).

    Returns a vector g where g_i = 2·∑_{k∈K} a_{k-i},
    K = set of indices where conv attains its maximum (within tolerance).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _eg_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """One entropic‑gradient step (multiplicative update + renormalisation)."""
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    return a_new


def _random_transfer(a: np.ndarray, max_step: float) -> np.ndarray:
    """
    Transfer a random amount of probability mass (≤ max_step) from one
    index to another, keeping the vector on the simplex.
    """
    n = a.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    transferable = min(a[i], max_step)
    if transferable <= 0.0:
        return a  # nothing to move
    delta = np.random.uniform(0.0, transferable)

    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # due to rounding we may get tiny negatives – clip and renormalise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a, 1.0 / n)
    return a_new


def _jitter_restart(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Produce a perturbed copy of ``a`` by adding Gaussian noise of
    magnitude ``eps`` and projecting back onto the simplex.
    """
    n = a.shape[0]
    noise = np.random.randn(n) * eps
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new

    ``evaluate_sequence`` objective.  The algorithm consists of:

    1. Entropic gradient descent (many cheap multiplicative updates).
    2. Gradient‑guided deterministic mass transfers.
    3. A short simulated‑annealing polishing phase.

    The best vector encountered is returned as a plain list of floats.
    """

    # Basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = 2048

    # Initialise a (use the previously discovered construction if it exists)
    # ------------------------------------------------------------------

                    tmp = np.interp(new_x, old_x, tmp)
                a = tmp / tmp.sum()
            else:

    # ------------------------------------------------------------------
    transfer_deadline = start_time + 0.80 * budget_s
    trans_it = 0
    while time.time() < transfer_deadline:
        trans_it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        # Sub‑gradient gives us a direction: move mass from high to low gradient
        grad = _subgradient_max(a, conv)
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # Transfer a modest fraction of the source mass
        delta = min(a[i_high] * 0.5, 0.02)   # at most 2 % of total probability
        if delta <= 0.0:
            break

        a_candidate = a.copy()
        a_candidate[i_high] -= delta
        a_candidate[i_low] += delta

        cand_score = _score(a_candidate)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] transfer iter {trans_it:<4d} improve → {cand_score:.6f}")
            continue

        # If no improvement, try a small random jitter and a few EG steps
        a_jitter = _jitter_restart(a, eps=0.01)
        # a short EG “re‑warm” – 200 iterations max
        warm_it = 0
        while warm_it < 200 and time.time() < transfer_deadline:
            warm_it += 1
            conv_j = _conv_self(a_jitter)
            grad_j = _subgradient_max(a_jitter, conv_j)
            eta_j = 0.02 / math.sqrt(warm_it + 1.0)
            a_jitter = _eg_update(a_jitter, grad_j, eta_j)
            cand_score_j = _score(a_jitter)
            if cand_score_j < best_score:
                best_score = cand_score_j
                best_seq = a_jitter.copy()
                a = a_jitter
                print(f"[{time.time() - start_time:6.2f}s] jitter‑EG improve → {cand_score_j:.6f}")
                break

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing polishing
    # ------------------------------------------------------------------
    sa_deadline = deadline
    cur_score = _score(a)
    temperature = 0.005               # initial temperature (in objective units)
    decay = 0.99995                   # exponential decay per iteration
    sa_it = 0
    max_step = 0.02                   # maximal amount of mass moved per trial

    while time.time() < sa_deadline:
        sa_it += 1
        a_candidate = _random_transfer(a, max_step)
        cand_score = _score(a_candidate)

        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA iter {sa_it:<6d} improve → {cand_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = a_candidate
                cur_score = cand_score

        # Cool down
        temperature *= decay
        # Reduce step size slowly together with temperature (helps final precision)
        max_step = max(1e-5, max_step * decay)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
