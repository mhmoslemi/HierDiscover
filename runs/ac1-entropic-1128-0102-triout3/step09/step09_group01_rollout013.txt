assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a\bigr)^{2}}
\]

is homogeneous – we may normalise the step‑function to a probability distribution
(`∑a = 1`).  
The problem is therefore to **flatten the self‑convolution** as much as possible.
Instead of the classic sub‑gradient that only looks at a single maximising index,
the code below uses a *smooth* approximation of the maximum:

\[
\operatorname{smooth\_max}_\beta (c)=\frac{1}{\beta}
\log\!\Bigl(\sum_{k} e^{\beta\,c_k}\Bigr)
\xrightarrow[\beta\to\infty]{}\max_k c_k .
\]

For a fixed temperature `β` the objective becomes differentiable and its
gradient can be written in closed form.  
We work with the *log‑it* variables `w` and keep the probabilities
`a = softmax(w)`.  This guarantees non‑negativity and `∑a = 1` without any
extra projection.

The optimisation consists of three phases

1. **Warm‑up (β small)** – the smooth‐max is very flat, the gradient moves the
   distribution towards a globally flat convolution.
2. **Continuation** – β is increased geometrically (`5 → 20 → 80 → …`);
   each stage refines the solution while staying in the basin of the previous
   one.
3. **Final polish** – a cheap mass‑transfer simulated‑annealing run eliminates
   the last tiny bumps.

All heavy work (convolution, gradient) is performed with FFTs (`O(n log n)`);
the whole routine finishes well inside the 1000 s wall‑clock limit.

```python
"""
Proposes a step‑function (non‑negative list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 (default).  The algorithm is a three‑stage optimiser:

  1️⃣  Smooth‑max gradient descent on the softmax‑parametrised probabilities.
      β (temperature) is increased geometrically – a classic continuation /
      homotopy method.  Gradients are obtained analytically using two FFTs
      per iteration and the update is performed with Adam.

  2️⃣  The best vector found is fed to a light simulated‑annealing polish
      (mass‑transfer moves) that can escape the remaining local traps.

The routine respects the 1000 s wall‑clock budget, prints progress
information and finally returns a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    a must be a 1‑D real array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of real vectors x (len m) and y (len p) using FFT.
    Returns the full linear convolution of length m+p‑1 (truncated to L).
    """
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)
    return conv


# ----------------------------------------------------------------------
# Fast objective (sum == 1) – exact max version, used for logging
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray, L: int) -> float:
    """Exact objective for a probability vector (sum≈1)."""
    conv = _conv_self(a, L=L)
    max_c = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_c)   # denominator is 1 because sum(a)=1


# ----------------------------------------------------------------------
# Smooth‑max based gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float,
                         L_conv: int, L_grad: int) -> tuple[float, np.ndarray]:
    """
    Returns
        smooth_max = (1/beta) * log( sum_k exp(beta * conv_k) )
        grad_a    = ∂/∂a ( 2 n * smooth_max )
    where conv = a*a (linear convolution, length 2n‑1).
    Both quantities are needed for the Adam updates.
    """
    n = a.shape[0]

    # 1️⃣  linear convolution a*a
    conv = _conv_self(a, L=L_conv)                     # shape (2n‑1,)

    # 2️⃣  soft‑max weights (avoid overflow)
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()                     # w sums to 1

    # 3️⃣  smooth objective value (actually we only need the exact max later)
    #    smooth_max = (max_c + np.log(w.sum())) / beta  # not used further

    # 4️⃣  gradient of 2 n * smooth_max w.r.t a
    #     ∂J/∂a_i = 4 n * sum_k w_k * a_{k-i}
    #   This is a convolution of w with a reversed.
    a_rev = a[::-1]                                   # length n
    conv2 = _conv_linear(w, a_rev, L=L_grad)          # length ≥ 3n‑2
    segment = conv2[n - 1: n - 1 + n]                 # the needed n entries
    grad_a = 4.0 * n * segment

    # Return the *exact* max (for logging) together with the gradient.
    # The smooth part is only used inside the optimiser.
    return max_c, grad_a


# ----------------------------------------------------------------------
# Simulated annealing polish (identical to the reference, but a bit faster)
# ----------------------------------------------------------------------
def _anneal_polish(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealer – runs until the deadline."""
    a = a0.copy()
    n = a.shape[0]
    score = _exact_score(a, L_conv=_next_pow2(2 * n - 1))
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    # temperature schedule
    start_temp = 5e-2
    end_temp = 1e-7
    total_steps = 300_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # pick source proportional to its mass, random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _exact_score(a_prop, L_conv=_next_pow2(2 * n - 1))
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
```
