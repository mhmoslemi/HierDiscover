def upscale(seq, n_target):
    x_old = np.linspace(0,1,len(seq))
    x_new = np.linspace(0,1,n_target)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new,0.0)
    seq_new /= seq_new.sum()
    return seq_new
```

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Coarse parametric exploration** – generate a few hundred candidate
   shapes (exponential, power‑law, product‑power, sine‑window, etc.) on a
   modest size (n≈2000).  Keep the best candidate according to the exact
   objective ``2·n·max(conv)`` (sum is normalised to 1).

2. **Upscale** – linearly interpolate the best small‑scale shape to the
   target length (n=4800) and renormalise.

3. **Long Adam optimisation on a smooth approximation of the max** –
   minimise the soft‑max of the auto‑convolution with a schedule of
   increasing smoothing parameters.  After every few hundred steps the
   exact objective is evaluated and the best iterate is stored.

4. **Projected sub‑gradient polishing** – from the best Adam point run a
   classic sub‑gradient descent on the true max‑convolution (with a
   decaying step size and momentum).

5. **Fast greedy peak reduction** – repeatedly move a small fraction of
   the coefficient that contributes most to the current peak towards the
   smallest coefficient; the convolution is updated incrementally so that
   thousands of moves fit into the remaining time budget.

The routine respects the 1000 s time limit, prints progress and finally
returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth‑max
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Objective helpers (fast version for a fixed length)
# ----------------------------------------------------------------------


def _objective_fast(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) assuming sum(a)=1."""
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – coarse parametric exploration on a small problem
# ----------------------------------------------------------------------


def _explore_small(n_small: int, deadline: float, start_time: float):
    """Return a promising vector of length n_small."""
    best_a = np.full(n_small, 1.0 / n_small)
    best_score = _objective_fast(best_a, n_small)

    # helper to evaluate a candidate and possibly improve best_a
    def try_candidate(a):
        nonlocal best_a, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a /= s
        score = _objective_fast(a, n_small)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"explore improve: score={score:.6f}")

    # 1) uniform (already the initial best)
    try_candidate(best_a)

    # 2) exponential families  a_i = r**i
    for r in np.linspace(0.92, 0.9995, 18):
        a = r ** np.arange(n_small, dtype=float)
        try_candidate(a)

    # 3) product‑power families  a_i ∝ ((i+1)*(n‑i))**p
    idx = np.arange(1, n_small + 1, dtype=float)
    prod = idx * (n_small - idx + 1.0)
    for p in np.linspace(-4.0, 4.0, 41):
        a = np.power(prod, p, where=prod > 0, out=np.zeros_like(prod))
        try_candidate(a)

    # 4) simple power‑law  a_i ∝ (i+1)**p
    for p in np.linspace(-2.5, 2.5, 41):
        a = np.power(idx, p, where=idx > 0, out=np.zeros_like(idx))
        try_candidate(a)

    # 5) sine window (non‑negative)
    a = np.sin(math.pi * (np.arange(n_small) + 0.5) / n_small)
    try_candidate(a)

    # 6) symmetric triangular bump (peak in the centre)
    centre = (n_small - 1) / 2.0
    tri = 1.0 - np.abs(np.arange(n_small) - centre) / (centre + 1.0)
    try_candidate(tri)

    # 7) a few random Dirichlet draws (α = 0.5 … 3)
    for alpha in (0.5, 1.0, 2.0, 3.0):
        a = np.random.default_rng().gamma(alpha, size=n_small)
        try_candidate(a)

    # 8) local random perturbations around the current best
    rng = np.random.default_rng()
    for _ in range(30):
        eps = rng.normal(scale=0.02, size=n_small)
        a = best_a + eps
        try_candidate(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – long Adam optimisation on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize_long(a0: np.ndarray,
                        deadline: float,
                        start_time: float):
    """Projected Adam on the smoothed max with an increasing α schedule."""
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0, 8192.0]

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        # run Adam steps with the current smoothing parameter
        while time.time() < deadline:
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha          # smoothed max
            # outer factor 2·n, exactly the objective returned by evaluate_sequence
            obj = 2.0 * n * f_val

            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 / np.sqrt(t)

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_simplex(a_new)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve (α={alpha:.0f}) "
                          f"score={sc:.6f}")

            # safety break for the outer α loop
            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – classic sub‑gradient polishing
# ----------------------------------------------------------------------


def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      beta: float = 0.94,
                      step0: float = 0.05):
    """Projected sub‑gradient descent on max_k (a*a)[k] with momentum."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # active set (indices whose contribution is within 1e-12 of the peak)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient for the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless

        # momentum + step‑size schedule
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – fast greedy peak reduction
# ----------------------------------------------------------------------


def _fast_greedy_improve(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 500_000):
    """Move a small fraction from the coefficient that most harms the peak
    towards the smallest coefficient; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = conv.argmax()
        cur_max = conv[k_max]

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # most responsible index

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.05 * a[src]                     # transfer 5 % of the source mass
        if eps < 1e-14:
            break

        # ---- incremental convolution update ----
        conv_tmp = conv.copy()
        conv_tmp[src:src + n] -= 2.0 * eps * a
        conv_tmp[dst:dst + n] += 2.0 * eps * a
        conv_tmp[2 * src] += eps * eps
        conv_tmp[2 * dst] += eps * eps
        conv_tmp[src + dst] -= 2.0 * eps * eps

        new_max = conv_tmp.max()
        if new_max < cur_max:                      # accept only strict improvements
            conv = conv_tmp
            a[src] -= eps
            a[dst] += eps
            best_score = 2.0 * n * new_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] fast‑greedy improve: "
                  f"iter={it:,}, max={new_max:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – upscale a small sequence to a larger length
# ----------------------------------------------------------------------


def _upscale(seq_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation from len(seq_small) → n_target and renormalise."""
    old_x = np.linspace(0.0, 1.0, seq_small.shape[0])
    new_x = np.linspace(0.0, 1.0, n_target)
    seq = np.interp(new_x, old_x, seq_small)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s > 0:
        seq /= s
    else:
        seq = np.full_like(seq, 1.0 / n_target)
    return seq
