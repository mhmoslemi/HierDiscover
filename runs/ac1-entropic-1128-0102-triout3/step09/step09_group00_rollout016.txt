        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
        return conv

    def objective_from_max_conv(max_conv):
        # sum fixed to target_sum => O = max_conv
        return float(max_conv)

    # Initialize f from previous best or uniform

        init_seq = np.maximum(init_seq, 0.0)

            raise ValueError

        print(f"[{time.time() - start_time:.2f}s] Loaded height_sequence_1")

    """Euclidean projection onto simplex {w >=0, sum w = 1}."""
    n = v.shape[0]

The algorithm proceeds in three phases

                         target_sum: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for max‑convolution with fixed sum."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _conv_self(a).max()

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_max:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the scaled simplex
        a = _project_simplex_sum(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (the “active‑set” step from the paper)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               target_sum: float,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """Perform a few LP‑based improvement steps."""
    n = f.shape[0]

    # pre‑compute the index pattern of the Toeplitz matrix (independent of f)
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel().astype(np.int32)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)

    # current feasible point
    cur_f = f.copy()
    cur_max = _conv_self(cur_f).max()
    best_f = cur_f.copy()
    best_score = cur_max
    print(f"[{time.time() - start_time:.2f}s] LP start: max_conv={cur_max:.6f}")

    # mixing coefficient (decreases if a step is not successful)
    t = 0.25

    while time.time() < deadline:
        # build the constraint matrix for the present f
        data = np.repeat(cur_f, n)                     # length n*n
        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)),
                                    shape=(2 * n - 1, n))

        # LP: maximise Σ g_j   s.t. 0 ≤ g,  (f ∗ g)_k ≤ cur_max
        c = -np.ones(n)                                 # maximize → minimise -Σ g_j
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, cur_max, dtype=float)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if not res.success:                            # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 <= 0:                                    # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP gave zero mass – abort")
            break

        # rescale g to the canonical total mass
        g = g0 * (target_sum / s0)

        # convex update
        cand_f = (1.0 - t) * cur_f + t * g
        cand_f = np.maximum(cand_f, 0.0)
        cand_f = cand_f * (target_sum / cand_f.sum())

        cand_max = _conv_self(cand_f).max()
        print(f"[{time.time() - start_time:.2f}s] LP step t={t:.3f} → max_conv={cand_max:.6f}")

        # accept only if we make progress
        if cand_max < cur_max - 1e-9:
            cur_f, cur_max = cand_f, cand_max
            if cand_max < best_score:
                best_score = cand_max
                best_f = cand_f.copy()
                print(f"[{time.time() - start_time:.2f}s] New best LP score={best_score:.6f}")
        else:
            # step was not useful – shrink the mixing factor
            t = max(0.01, t * 0.7)

        # a modest limit on the number of LP steps keeps the total time low
        if time.time() > deadline:
            break

    return best_f, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of heights for a step function whose evaluation score
    is as low as possible within the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    n = 4096
    target_sum = math.sqrt(2.0 * n)            # canonical normalisation

    # --------------------------------------------------------------
    # 0) start from the previously best known construction if present
    # --------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)   # noqa: F821
        seq0 = np.maximum(seq0, 0.0)
        if seq0.size == 0 or seq0.sum() == 0:
            raise ValueError
        f = seq0 * (target_sum / seq0.sum())
        print(f"[{time.time() - start:.2f}s] Loaded global starter.")
    except Exception:                                        # pragma: no cover
        f = np.full(n, target_sum / n)                        # uniform start
        print(f"[{time.time() - start:.2f}s] Using uniform start.")

    # avoid exact zeros (the LP would become unbounded)
    eps = 1e-12
    f = np.maximum(f, eps)
    f = f * (target_sum / f.sum())

    # --------------------------------------------------------------
    # 1) LP refinement (if SciPy is available)
    # --------------------------------------------------------------
    if SCIPY_AVAILABLE:
        remaining = deadline - time.time()
        lp_deadline = time.time() + max(5.0, remaining * 0.5)
        f, lp_score = _lp_refine(f, target_sum, lp_deadline, start)
        print(f"[{time.time() - start:.2f}s] LP stage finished, score={lp_score:.6f}")
    else:
        print(f"[{time.time() - start:.2f}s] SciPy unavailable – skipping LP refinement.")

    # --------------------------------------------------------------
    # 2) Short sub‑gradient polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:                                 # leave time for final steps
        sub_deadline = time.time() + remaining * 0.85
        f, sub_score = _subgradient_descent(f, target_sum,
                                            sub_deadline, start)
        print(f"[{time.time() - start:.2f}s] Sub‑gradient polishing finished, score={sub_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final sanity checks and conversion to plain list
    # --------------------------------------------------------------
    # Clip to the limits imposed by evaluate_sequence (0 … 1000)
    f = np.clip(f, 0.0, 1000.0)

    # Ensure the canonical total mass – evaluation is scale‑invariant,
    # but keeping the same normalisation makes the result reproducible.
    if f.sum() == 0:                                     # pragma: no cover
        f = np.full(n, target_sum / n)
    else:
        f = f * (target_sum / f.sum())

    return [float(x) for x in f]
