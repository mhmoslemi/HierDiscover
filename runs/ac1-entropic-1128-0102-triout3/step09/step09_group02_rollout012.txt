    import numpy as np, time, math, scipy.optimize as opt
    # Setup
    # Get initial sequence from height_sequence_1 if available, else uniform.
    # function to compute exact score
    def exact_score(a):
        conv = np.fft.ifft(np.fft.fft(a, n=2*n-1) * np.fft.fft(a, n=2*n-1)).real
        max_conv = conv.max()
        return float(2 * n * max_conv)
    # ... but we can compute conv using np.convolve for simplicity.

    # Smoothed objective
    def smooth_obj_and_grad(y, alpha):
        # y is vector length n
        # compute a as softmax
        expy = np.exp(y - np.max(y))  # subtract max for stability
        a = expy / expy.sum()
        # conv via FFT
        L = 1 << ((2*n-2).bit_length())
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
        # compute softmax weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        w /= w_sum
        # weighted sum of conv (approx max)
        weighted_sum = np.dot(w, conv)
        obj = 2 * n * weighted_sum   # smooth objective
        # gradient wrt a
        # compute w_rev for correlation
        w_rev = w[::-1]
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[:n + len(w_rev) - 1]

```
"""
Algorithm:
1. Initialise from previous best or uniform.
2. Perform a few L‑BFGS optimisations on a smooth soft‑max
   approximation of the max‑self‑convolution.  The distribution is
   parametrised by unconstrained logits (softmax) so the simplex
   constraint is satisfied automatically.
3. Refine the resulting vector with a deterministic greedy
   reduction of the current peak (same idea as the reference code,
   but driven by the exact convolution and an adaptive step size).
4. Return the best sequence found.
"""

import time, math, numpy as np
try:
    import scipy.optimize as _opt
    _has_scipy = True
except Exception:
    _has_scipy = False

def _next_pow2(x):
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _full_conv(a):
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _exact_score(a):
    """Exact evaluation of the objective (sum(a) is assumed =1)."""
    conv = _full_conv(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)

def _smooth_obj_and_grad(y, alpha, n, L):
    """
    Smoothed objective and its gradient w.r.t. the logits y.
    The distribution a = softmax(y) is a point on the probability simplex.
    The smooth objective is 2·n·∑_k w_k·(a*a)_k
    where w is the soft‑max of α·(a*a).
    """
    # soft‑max to obtain a
    max_y = np.max(y)
    expy = np.exp(y - max_y)
    a = expy / expy.sum()

    # self‑convolution of a
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

    # soft‑max weights on the convolution entries
    scaled = alpha * conv
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = w.sum()
    w /= w_sum

    # weighted sum (smooth approximation of the max)
    weighted_sum = np.dot(w, conv)
    obj = 2.0 * n * weighted_sum

    # gradient w.r.t. a :   ∂/∂a (∑_k w_k conv_k) = 2·(w ⊛ a_rev)
    w_rev = w[::-1]
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
    len_w = len(w_rev)          # = 2*n-1
    grad_a = 4.0 * n * corr[len_w - 1 - np.arange(n)]

    # chain rule through soft‑max
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    return obj, grad_y

def _lbfgs_optimize(a0, alphas, max_iter_per_alpha, deadline, start_time):
    """
    Run a short L‑BFGS optimisation for each value of alpha.
    Returns the best vector (as a probability distribution) found.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    # initialise logits from the current distribution
    eps = 1e-12
    y = np.log(a0 + eps)

    best_a = a0.copy()
    best_score = _exact_score(best_a)

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # set a reasonable limit for the internal optimiser
        remain = deadline - time.time()
        # allow at most max_iter_per_alpha, but also bound by a rough
        # estimate of remaining seconds (≈ 100 iters per second)
        max_iter = min(max_iter_per_alpha, int(remain * 80) + 1)

        if max_iter <= 0:
            break

        if not _has_scipy:
            # SciPy not available – break early
            break

        # minimise the smooth objective
        res = _opt.minimize(
            fun=_smooth_obj_and_grad,
            x0=y,
            args=(alpha, n, L),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'disp': False}
        )
        # re‑extract the probability vector
        y = res.x
        expy = np.exp(y - np.max(y))
        a = expy / expy.sum()

        sc = _exact_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] L‑BFGS α={alpha:.1f} score={sc:.6f}")

    return best_a, best_score

def _run_greedy_reduction(a0, cur_best_score, cur_best_seq, deadline, start_time,
                          max_iters=2_000_000):
    """
    Deterministic greedy reduction of the current peak.
    Very similar to the reference implementation, but with a
    slightly larger iteration budget and a tighter epsilon rule.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    conv = _full_conv(a)                     # full convolution (length 2·n‑1)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    # pre‑allocate a temporary array for the “partner” values used in the update
    a_old = np.empty_like(a)

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # locate the second‑largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build the mirror of a around the max index (a_k[i] = a[max_idx−i])
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination where a_k is minimal (but not the source)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = (src + 1) % n
        else:
            dst = dst_candidates[np.random.randint(dst_candidates.size)]

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # target epsilon that would bring the peak down to (just below) second_max
        if delta_a > 0:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
        else:
            eps = min(a[src], 0.02 * a[src])

        if eps <= 0:
            continue

        # ----- incremental update of the full convolution -----
        a_old[:] = a
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        if it % recompute_interval == 0:
            conv = _full_conv(a)   # re‑compute to suppress floating‑point drift

        cur_score = _exact_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine consists of:
      * a short L‑BFGS optimisation on a smooth surrogate of the max‑convolution,
      * a deterministic greedy reduction of the current peak,
      * optional light random perturbations if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Initial distribution – use the previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = np.maximum(starter, 0.0)
            a0 /= a0.sum()
        else:
            raise ValueError
    except Exception:
        # fallback: uniform distribution of length 4096
        final_n = 4096
        a0 = np.full(final_n, 1.0 / final_n)

    n = a0.shape[0]

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth approximation of the max‑convolution.
    # ------------------------------------------------------------------
    # A sequence of increasing smoothing parameters.
    alphas = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 3200.0]
    # Rough per‑alpha iteration budget (the optimiser itself decides).
    max_iter_per_alpha = 200
    a_lbfgs, score_lbfgs = _lbfgs_optimize(
        a0, alphas, max_iter_per_alpha, deadline, start_time
    )

    best_seq = a_lbfgs.copy()
    best_score = score_lbfgs

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic greedy reduction.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            best_score,
            best_seq,
            deadline,
            start_time,
            max_iters=2_000_000
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 greedy best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Light random perturbations (simple hill‑climbing) if time remains.
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # take a tiny random move and keep it if it improves the exact score
        src = np.random.randint(0, n)
        dst = np.random.randint(0, n)
        if src == dst or best_seq[src] < 1e-14:
            continue
        eps = best_seq[src] * np.random.rand() * 0.05   # up to 5 % of src mass
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        # keep numeric safety
        cand[cand < 0] = 0.0
        cand /= cand.sum()
        cand_score = _exact_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[{time.time() - start_time:.2f}s] random‑hill improve: score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # Return the final probability vector as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]

"""
Algorithm:
1. Initialise a probability vector from the previously discovered best
   sequence (if available) or from a uniform distribution.
2. Run a few L‑BFGS optimisations on a smooth soft‑max surrogate of the
   max‑self‑convolution.  The distribution is parametrised by unconstrained
   logits (softmax) so the simplex constraint is satisfied automatically.
3. Refine the obtained vector with a deterministic greedy reduction of the
   current peak of the exact convolution (the same idea as the reference
   code, but with an adaptive step size and a larger iteration budget).
4. If time permits, perform very cheap random hill‑climbing steps.
5. Return the best sequence found (plain Python list of floats).

The implementation respects the 1000 s time budget, prints progress
messages and works without any file I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy for L‑BFGS
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:      # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) using an FFT of the
    next power‑of‑two size.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _exact_score(a: np.ndarray) -> float:
    """Exact objective value (sum(a) is kept equal to 1)."""
    conv = _full_conv(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Smoothed objective (soft‑max) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, alpha: float,
                         n: int, L: int):
    """
    Returns (value, gradient) of the smooth surrogate of the max‑convolution.
    The probability vector is a = softmax(y).
    The surrogate is   2·n·∑_k w_k·(a*a)_k,
    where w_k = softmax(alpha·(a*a)_k).
    """
    # ----- a = softmax(y) -----
    max_y = np.max(y)
    expy = np.exp(y - max_y)
    a = expy / expy.sum()

    # ----- self‑convolution a*a -----
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

    # ----- soft‑max weights on the convolution entries -----
    scaled = alpha * conv
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = w.sum()
    w /= w_sum

    # ----- smooth objective (≈ max) -----
    weighted_sum = np.dot(w, conv)
    obj = 2.0 * n * weighted_sum

    # ----- gradient w.r.t. a -----
    w_rev = w[::-1]                       # reverse for correlation
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
    len_w = len(w_rev)                    # = 2·n‑1
    grad_a = 4.0 * n * corr[len_w - 1 - np.arange(n)]

    # ----- chain rule through the soft‑max parametrisation -----
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    return obj, grad_y


# ----------------------------------------------------------------------
# Phase 1 – short L‑BFGS runs for several smoothing parameters
# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray, alphas, max_iter_per_alpha,
                    deadline: float, start_time: float):
    """
    Runs a small L‑BFGS optimisation for each α in *alphas*.
    Returns the best probability vector discovered.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)

    # initialise logits from the supplied distribution
    eps = 1e-12
    y = np.log(a0 + eps)

    best_a = a0.copy()
    best_score = _exact_score(best_a)

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # estimate a sensible upper bound for the internal iteration count
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_alpha, int(remaining * 80) + 1)
        if max_iter <= 0 or not _HAS_SCIPY:
            break

        res = _opt.minimize(fun=_smooth_obj_and_grad,
                            x0=y,
                            args=(alpha, n, L),
                            method='L-BFGS-B',
                            jac=True,
                            options={'maxiter': max_iter, 'disp': False})
        y = res.x                                 # start next run from here
        expy = np.exp(y - np.max(y))
        a = expy / expy.sum()

        sc = _exact_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS α={alpha:.1f} score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 2_000_000):
    """
    Repeatedly moves a small amount of mass from the index that
    contributes most to the current peak to an index that contributes least,
    updating the full convolution incrementally.
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = _full_conv(a)                     # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500                # every few thousand steps recompute exactly

    a_old = np.empty_like(a)

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually appear in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build a mirrored copy of a around the peak
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination where the mirrored value is minimal
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = (src + 1) % n
        else:
            dst = dst_candidates[np.random.randint(dst_candidates.size)]

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # adaptive epsilon aiming at lowering the peak to just below the second best
        if delta_a > 0:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
        else:
            eps = min(a[src], 0.02 * a[src])

        if eps <= 0:
            continue

        # ----- incremental update of the convolution (O(n)) -----
        a_old[:] = a
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        if it % recompute_interval == 0:
            conv = _full_conv(a)   # clean‑up floating‑point drift

        cur_score = _exact_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} score={cur_score:.6f}")

    return best_a, best_score

        cand[cand < 0] = 0.0

        sc = _exact_score(cand)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
