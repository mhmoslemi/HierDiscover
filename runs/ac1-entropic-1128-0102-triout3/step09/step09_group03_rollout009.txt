"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Diverse starts** – uniform, a few Dirichlet draws, a Gaussian
   window and the previously known best (if available).  All starts are
   normalised to sum = 1.

2. **Adam on a smooth surrogate** – we minimise a log‑sum‑exp
   approximation of the max‑convolution.  The smoothing parameter
   ``alpha`` is increased slowly (0.5 → 1024) and Adam is run for many
   cheap iterations.  The best vector seen is retained.

3. **k‑active exponentiated‑gradient** – a mirror‑descent on the *true*
   objective, but the sub‑gradient is taken from the *K* largest
   convolution entries (instead of only the single maximum).  This
   balances several peaks simultaneously and usually pushes the
   objective a bit lower than the original single‑peak EG.

4. **Random perturbations** – every few thousand iterations a tiny
   random jitter (∼10⁻⁴ of the current mass) is added and the vector
   is re‑normalised.  This helps to escape flat plateaus.

5. **Greedy peak reduction** – a deterministic local move that moves a
   fraction of the coefficient that most contributes to the current
   peak towards the currently smallest coefficient.  Only strict
   improvements are kept.

The whole routine respects the supplied ``budget_s`` (default 1000 s)
and prints intermediate scores.  The final sequence is clipped to
``[0,∞)`` and re‑normalised before being returned.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Fast exact objective (identical to evaluate_sequence for non‑negative a)
# ----------------------------------------------------------------------


def _exact_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv) / (sum(a)**2).  a is assumed non‑negative."""
    n = a.shape[0]
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:                # guard – should never happen
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Adam optimisation of the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(a: np.ndarray, alpha: float, fft_len: int):
    """
    Returns (smooth objective value, gradient w.r.t. a) for the
    log‑sum‑exp surrogate of max(conv) with smoothing factor ``alpha``.
    """
    n = a.shape[0]
    conv = _conv_self(a, fft_len)                     # a*a
    scaled = alpha * conv
    max_scaled = scaled.max()
    # soft‑max
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    w /= w_sum

    # value = (1/α) * log sum exp(α·conv)
    obj = (max_scaled + np.log(w_sum)) / alpha

    # gradient of the surrogate:
    #   ∂obj/∂a_i = 2 * Σ_k w_k a_{k-i}
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wr = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wr, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return obj, grad


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   fft_len: int,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Adam minimisation of the smooth surrogate.
    Returns the best vector seen and its *exact* score.
    """
    a = a0.copy()
    n = a.shape[0]

    # work in the log‑space (soft‑max will re‑normalise)
    y = np.log(np.maximum(a, 1e-16))

    # schedule for the smoothing parameter
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 1024.0, 2048.0]
    iters_per_alpha = max_iters // len(alphas)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = _exact_score(a, fft_len)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # a = softmax(y)
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / expy.sum()

            # smooth objective & its gradient
            _, grad = _smooth_obj_and_grad(a, alpha, fft_len)

            # back‑prop through the soft‑max
            grad_y = a * (grad - np.dot(grad, a))

            # Adam update on the logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y -= lr_t * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if t % 500 == 0:
                sc = _exact_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time():.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )

    # final exact score for the last a
    final_sc = _exact_score(a, fft_len)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# k‑active exponentiated‑gradient on the true objective
# ----------------------------------------------------------------------


def _k_active_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len: int,
    eta0: float = 0.5,
    K: int = 3,
    perturb_every: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) using the *K* largest
    convolution entries to build the sub‑gradient.  Random jitter is
    injected every ``perturb_every`` iterations.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _exact_score(a, fft_len)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}s] k‑EG improve: iter={t:,}, score={cur_score:.6f}"
            )

        # indices of the K largest convolution entries
        top_k = np.argpartition(conv, -K)[-K:]

        # accumulate the sub‑gradient from those shifts
        g = np.zeros_like(a)
        for k in top_k:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += 2.0 * a[j]

        # average over the K shifts
        g /= K

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # exponentiated update
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional random jitter to escape plateaux
        if perturb_every and (t % perturb_every == 0):
            rng = np.random.default_rng(t + 12345)
            jitter = rng.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

    # final exact score
    final_sc = _exact_score(a, fft_len)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local peak reduction (same as before)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          max_iters: int = 5_000) -> tuple[np.ndarray, float]:
    """Move a fraction of the coefficient that contributes most to the peak
    towards the smallest coefficient, accepting only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    best_score = 2.0 * n * _conv_self(a, fft_len).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time():.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.99) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _uniform_start(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of:
      * building several diverse starts,
      * a long Adam‑optimisation of a smooth surrogate,
      * an intensive k‑active exponentiated‑gradient phase,
      * a final greedy peak‑reduction.
    The best vector ever seen (scaled to sum = 1) is returned as a Python
    list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem size – we keep the original length (4800) because the
    # convolution factor 2·n is baked into the objective.
    # --------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Assemble a pool of starts
    # --------------------------------------------------------------
    starts = []

    # previous best, if the global variable is present
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.shape[0] != n:
                prev = _upsample(prev, n)
            starts.append(prev / prev.sum())
    except Exception:
        pass

    starts.append(_uniform_start(n))
    starts.append(_exponential_start(n, r=0.985))
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_gaussian_start(n))
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # a few extra Dirichlet draws for extra diversity
    for _ in range(3):
        starts.append(_dirichlet_start(n, conc=rng.uniform(0.3, 3.0), rng=rng))

    # evaluate raw starts, keep the best
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _exact_score(a0, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1} score = {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate (≈15 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    seq_adam, score_adam = _adam_optimize(best_seq, phase1_deadline, fft_len)
    if score_adam < best_score:
        best_score = score_adam
        best_seq = seq_adam.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – k‑active exponentiated‑gradient (≈70 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.85
    seq_eg, score_eg = _k_active_expgrad_optimize(
        best_seq, phase2_deadline, fft_len, eta0=0.5, K=3, perturb_every=5_000
    )
    if score_eg < best_score:
        best_score = score_eg
        best_seq = seq_eg.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after k‑EG score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy local improvement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq_final, score_final = _greedy_local_improve(best_seq, deadline, max_iters=10_000)
        if score_final < best_score:
            best_score = score_final
            best_seq = seq_final.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Final normalisation / safety checks
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final verification using the supplied function (slow but safe)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
