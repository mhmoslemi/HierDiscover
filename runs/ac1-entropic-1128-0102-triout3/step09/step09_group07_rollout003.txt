    max_val = conv.max()

        i_start = max(0, k_max - (n - 1))

            src = rng.choice(i_range, p=prob_src)

            src = rng.choice(n, p=prob_src)

"""
Improved three‑stage search for a low‑max self‑convolution step function.

Stage 1 – exponentiated‑gradient warm‑up on a smoothed max (log‑sum‑exp).
Stage 2 – projected sub‑gradient descent on the exact ∞‑norm of the
          self‑convolution (momentum + Euclidean simplex projection).
Stage 3 – aggressive optimal incremental refinement: move a *full* amount of
          mass from a source index that contributes heavily to the current
          maximal convolution bin to a destination index that is (almost)
          empty, computing the largest admissible step size ε analytically.
          The removal of the tiny “fraction” limitation from the previous
          implementation yields far larger per‑iteration reductions, while the
          per‑iteration complexity stays O(n).  The loop runs until the time
          budget expires, keeping the best sequence seen.

The algorithm works for any reasonable length (chosen here as 8000) and
adapts its internal deadlines to the supplied wall‑clock budget.  The best
sequence found is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All indices whose value lies within rel_tol of the maximum are treated
    as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each pair (i,j) contributes a[j] to gradient of a[i] and vice‑versa
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max (log‑sum‑exp) gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent with KL divergence)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – aggressive optimal incremental refinement (full‑mass moves)
# ----------------------------------------------------------------------
def _run_optimal_incremental_refine_v2(a0: np.ndarray,
                                      n: int,
                                      deadline: float,
                                      start_time: float,
                                      recompute_every: int = 2000):
    """
    Repeatedly transfer as much mass as possible from a source index that
    contributes heavily to the current maximal convolution bin to a destination
    index that is (almost) empty.  The admissible step size ε is computed
    analytically (solution of a 1‑D linear programme), thus each accepted move
    reduces the current maximum by a noticeable amount.
    """
    a = a0.copy()
    conv = _conv_self(a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    it = 0
    accepted = 0
    stagnant = 0      # counts iterations without improvement

    while time.time() < deadline:
        # current maximal convolution index
        k_max = int(np.argmax(conv))

        # --- source selection: probability proportional to contribution to max bin ---
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            # fall back to pure mass‑proportional sampling
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # --- destination selection: inverse‑mass weighting (avoid src) ---
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[src] = 0.0
        prob_dst = inv_weights / inv_weights.sum()
        dst = rng.choice(n, p=prob_dst)

        # --- effect vector s (linear in ε) on the whole convolution ---
        s = np.zeros(2 * n - 1, dtype=float)
        s[src:src + n] -= 2.0 * a
        s[dst:dst + n] += 2.0 * a

        s_max = s[k_max]
        if s_max >= 0.0:
            # this pair would not lower the current maximal entry
            stagnant += 1
            if stagnant > 10000:
                # try a different random pair next iteration
                stagnant = 0
            continue

        # --- maximal admissible ε before another entry overtakes the current max ---
        mask = s > s_max
        if np.any(mask):
            delta_conv = conv[k_max] - conv[mask]          # ≥0
            delta_s = s[mask] - s_max                     # >0
            eps_candidates = delta_conv / delta_s
            eps_candidates = eps_candidates[eps_candidates > 0.0]
            eps_star = eps_candidates.min() if eps_candidates.size else np.inf
        else:
            eps_star = np.inf

        epsilon = min(a[src], eps_star)   # move as much as possible, bounded by source mass

        if epsilon <= 0.0:
            continue

        # --- apply the move analytically ---
        conv_new = conv + epsilon * s
        new_max = conv_new.max()

        # require a genuine improvement (avoid floating‑point noise)
        if new_max < max_val - 1e-12:
            conv = conv_new
            a[src] -= epsilon
            a[dst] += epsilon
            max_val = new_max
            accepted += 1
            it += 1
            stagnant = 0

            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] opt‑refine‑v2 improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional full recomputation to keep numerical drift in check
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing
    ``evaluate_sequence`` under the supplied wall‑clock budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total wall‑clock budget in seconds (default 1000).

    Returns
    -------
    List[float]
        The best sequence found (already normalised to sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initialise – try to reuse a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # use a relatively fine discretisation; larger n gives a richer search space
    final_n = 8000
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – exponential‑gradient warm‑up (smooth max)
    # ------------------------------------------------------------------
    stage1_deadline = min(deadline, start_time + 0.07 * (deadline - start_time))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve at least 25 s for the final refinement
        stage2_deadline = time.time() + max(25.0, 0.78 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.3,          # slightly larger initial step than before
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive optimal incremental refinement (full‑mass moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_optimal_incremental_refine_v2(
            best_seq,
            final_n,
            deadline,
            start_time,
            recompute_every=1500,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = float(2.0 * final_n * conv_ref.max())
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
