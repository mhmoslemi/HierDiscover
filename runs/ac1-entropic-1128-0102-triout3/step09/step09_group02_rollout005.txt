assistantfinal**Solution Overview**

The attack aims at minimizing  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

The term \((a*a)_k\) is the self‑convolution of the step‑function.  
If the mass of the vector is spread over many positions but the pair‑sums
\(i+j\) are **uniformly** distributed, the maximal convolution coefficient
approaches the average value \(1/(2n)\) and the theoretical lower bound
\(F(a)=1\) is approached.

The program therefore builds **sparse** candidates, improves them with a
small linear‑programming (LP) step (the same idea used in the reference
algorithm) and finishes with a cheap simulated‑annealing / exponential‑
gradient polishing phase.  The whole search runs for the allotted 1000 s
and always returns the best vector found.

Key ideas
* **Sparse start** – only a fraction of the positions carry mass.
  This reduces the number of pair‑sums that collide and gives a much
  smaller peak of the convolution.
* **LP improvement** – for a fixed vector `a` we solve  

  \[
  \max_{b\ge 0}\;\sum_j b_j\qquad
  \text{s.t.}\;\;(a*b)_k\le M\;(k=0,\dots,2n-2),
  \]

  where \(M=\max_k (a*a)_k\).  
  The optimal `b` is a vector that can be mixed with `a` to lower the
  maximal convolution.
* **Mirror‑descent polishing** – an exponentiated‑gradient step on the
  true (non‑smoothed) max‑convolution quickly removes the remaining
  excess.
* **Random restarts** – a new sparse start is generated whenever time
  permits; the best of all runs is kept.

The implementation uses only NumPy and CVXPy (fallback to a pure‑numeric
heuristic if CVXPy is unavailable) and respects the 1000 s time budget.

```python
"""
Propose a step‑function with a very low maximal self‑convolution.

The algorithm works as follows (all phases respect the global time budget):

1.  Initialise a sparse vector (few non‑zero entries, normalised to sum‑1).
2.  Repeatedly improve it by solving a linear programme that finds a
    vector `b` with maximal ℓ₁‑norm under the constraint
    (a * b)ₖ ≤ M, where M = max(a*a).  The new candidate is a convex
    combination a←(1‑t)a + t·b.
3.  Apply a few exponentiated‑gradient steps on the true max‑convolution
    (multiplicative mirror descent) – fast and keeps the vector on the
    simplex.
4.  If time allows, perform short simulated‑annealing moves to escape
    possible plateaus.
5.  Keep the best sequence seen across all random restarts and return it
    as a plain Python list.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv) (sum(a) is assumed = 1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)          # dense version is fine for n ≤ 2000
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Linear‑programming step (reference “find g₀”)
# --------------------------------------------------------------

try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _CVXPY_AVAILABLE = False


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀb   subject to   b ≥ 0   and   (a*b)_k ≤ M.
    Returns the optimal b as a NumPy array, or None if the solver fails.
    """
    if not _CVXPY_AVAILABLE:          # pragma: no cover
        return None

    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)

    constraints = []
    # (a*b)_k = Σ_i a_i b_{k-i}
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices of b that appear in this sum
        idx = np.arange(lo, hi + 1, dtype=int)
        # coefficients a_{k-idx}
        coeff = a[k - idx]
        expr = cp.sum(cp.multiply(coeff, b[idx]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    except Exception:                # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.array(b.value).flatten()


# --------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution
# --------------------------------------------------------------

def _exp_gradient_step(a: np.ndarray, lr: float) -> np.ndarray:
    """
    One mirror‑descent step on the true max‑convolution.
    Keeps the vector on the probability simplex.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(conv.argmax())                     # index of the current peak
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]               # ∂ (a*a)_k / ∂ a_i
    # multiplicative update
    a_new = a * np.exp(-lr * grad)
    total = a_new.sum()
    if total <= 0.0 or np.isnan(total):
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new = a_new / total
    return a_new


# --------------------------------------------------------------
# Simple simulated‑annealing (tiny perturbations)
# --------------------------------------------------------------

def _anneal_one_step(a: np.ndarray, conv: np.ndarray, temp: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray]:
    """
    Propose a move that transfers a random fraction of mass from a
    “source” index (participating in the current peak) to a random
    “destination”.  Accept with Metropolis probability.
    Returns the possibly modified (a, conv).
    """
    n = a.shape[0]
    # locate the current maximum of the convolution
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    contrib = a[idx] * a[k - idx]
    if contrib.sum() > 0.0:
        src = rng.choice(idx, p=contrib / contrib.sum())
    else:
        src = rng.choice(n)

    dst = rng.integers(n)
    if src == dst or a[src] < 1e-12:
        return a, conv

    eps = a[src] * rng.random() * 0.15          # move at most 15 % of src
    a_prop = a.copy()
    a_prop[src] -= eps
    a_prop[dst] += eps
    conv_new = np.convolve(a_prop, a_prop)
    delta = conv_new.max() - conv.max()
    if delta < 0.0 or math.exp(-delta / temp) > rng.random():
        return a_prop, conv_new
    return a, conv


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget and returns the best sequence
    discovered as a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned for 1000 s on a single core)
    # ------------------------------------------------------------------
    final_n = 1024                # moderate size – fast FFT + LP
    density = 0.25                # fraction of non‑zero entries in a start
    max_lp_iters = 20             # LP updates per inner optimisation
    exp_steps = 200               # exponentiated‑gradient steps after LP
    anneal_total_steps = 80_000   # moves in the annealing phase
    anneal_start_temp = 4e-2
    anneal_end_temp   = 1e-9

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate the current vector (sum is forced to 1)
    # ------------------------------------------------------------------
    def _evaluate(a_vec: np.ndarray) -> float:
        return _score(a_vec)

    # ------------------------------------------------------------------
    # Random‑restart loop (runs as long as time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # ----- 1. sparse random initialisation -----------------------
        m = max(2, int(density * final_n))
        idx = rng.choice(final_n, size=m, replace=False)
        a = np.zeros(final_n, dtype=float)
        a[idx] = rng.random(m) + 1e-6
        a /= a.sum()                         # normalise to sum = 1

        cur_score = _evaluate(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] restart init improve: "
                  f"score={cur_score:.6f}")

        # ----- 2. LP‑based refinement --------------------------------
        if _CVXPY_AVAILABLE:
            for lp_it in range(max_lp_iters):
                if time.time() >= deadline:
                    break
                M = np.convolve(a, a).max()
                b = _solve_lp_max_sum(a, M)
                if b is None:
                    break
                # scale to the simplex
                b_sum = b.sum()
                if b_sum == 0.0:
                    continue
                b = b / b_sum
                # convex combination (t = 0.2 works well empirically)
                t = 0.2
                a = (1.0 - t) * a + t * b
                a = np.maximum(a, 0.0)
                a /= a.sum()
                cur_score = _evaluate(a)
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] LP improve: "
                          f"iter={lp_it+1}, score={cur_score:.6f}")
        else:
            # If CVXPy is not available we simply skip the LP phase.
            pass

        # ----- 3. Exponentiated‑gradient polishing ------------------
        for eg in range(exp_steps):
            if time.time() >= deadline:
                break
            lr = 0.6 / math.sqrt(eg + 1)
            a = _exp_gradient_step(a, lr)
            cur_score = _evaluate(a)
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                      f"step={eg+1}, score={cur_score:.6f}")

        # ----- 4. Targeted simulated annealing (light) -------------
        if time.time() < deadline:
            conv = np.convolve(a, a)
            steps = anneal_total_steps
            for it in range(1, steps + 1):
                if time.time() >= deadline:
                    break
                # exponential cooling
                T = anneal_start_temp * (anneal_end_temp / anneal_start_temp) ** (it / steps)
                a, conv = _anneal_one_step(a, conv, T, rng)
                cur_score = _score(a)           # fast: max(conv) already available
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                          f"it={it:,}, score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # End of one restart – keep the best sequence seen so far
        # ------------------------------------------------------------------
        # Small pause to avoid extremely tight loops if we are close to the deadline
        if time.time() > deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # Final normalisation (the evaluator caps and rescales anyway)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (never happens in practice)
        best_seq = np.full(final_n, 1.0 / final_n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / final_n)

    # Return a plain Python list as required by the outer harness
    return [float(x) for x in best_seq]
```
