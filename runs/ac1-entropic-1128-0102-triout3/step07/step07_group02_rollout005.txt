"""
Optimise a non‑negative step function (probability simplex) to minimise
``evaluate_sequence``.  The method builds on the previous approach but
introduces two new optimisation phases:

1. **Exponentiated sub‑gradient descent** – a mirror‑descent update on the
   exact max‑convolution using a KL‑divergence projection onto the simplex.
   The sub‑gradient is the (averaged) gradient of all active convolution
   entries (those attaining the maximum).  This phase quickly drives the
   objective down from a warm‑start.

2. **Targeted simulated‑annealing (peak shaving)** – repeatedly moves a
   fraction of probability mass from indices that contribute most to the
   current maximal convolution entry to indices that contribute little.
   The Metropolis acceptance rule with a slowly cooling temperature allows
   occasional uphill moves and helps escaping shallow local minima.

The algorithm respects the 1000 s budget, automatically splitting time
between the two phases, and falls back to a uniform distribution if no
valid start is available.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: current objective (sum a is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_max_convolution(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns the sub‑gradient of max_k conv[k] w.r.t. a.
    For each active index k (where conv equals the maximum) the gradient is
    2 * J_k a, where J_k is the Toeplitz matrix with ones on the anti‑diagonal
    i + j = k.  The returned gradient is the average over all active k.
    """
    n = a.shape[0]
    M = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    max_inds = np.where(conv >= M - 1e-14)[0]
    grad = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i = 0 … k
            grad[: k + 1] += 2.0 * a[k :: -1]          # a[k],a[k‑1],…,a[0]
        else:
            # i = k‑n+1 … n‑1
            start = k - n + 1
            grad[start:] += 2.0 * a[start:][::-1]      # reversed slice

    grad /= max(len(max_inds), 1)
    return grad


# ----------------------------------------------------------------------
# Phase A – exponentiated (mirror) sub‑gradient descent
# ----------------------------------------------------------------------
def _run_exp_subgrad(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.5,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the KL‑divergence (exponentiated gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eps = 1e-30                         # avoid log(0)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Exp‑subgrad improve: "
                  f"n={a.shape[0]}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the active convolution entry
        grad = _grad_max_convolution(a, conv)

        # diminishing learning rate (η_t = step0 / sqrt(t))
        eta = step0 / math.sqrt(it)

        # exponentiated update (mirror descent onto simplex)
        log_a = np.log(a + eps) - eta * grad
        a = np.exp(log_a)
        a /= a.sum()                     # project onto simplex (exact)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – targeted simulated‑annealing peak‑shaving
# ----------------------------------------------------------------------
def _run_peak_shaving(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑contribution to low‑contribution indices."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    T_start = 5e-2
    T_end = 1e-6

    it = 0
    while time.time() < deadline:
        it += 1

        # recompute convolution and identify the (one) most active shift
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))

        # contributions of each coefficient to that shift
        contrib = np.zeros(n)
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – weighted by contribution (more mass moved away from big contributors)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – pick from low‑contribution tail
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-16:
            continue

        # step size (fraction of donor mass) shrinks with iteration count
        eps_frac = 0.3 * (0.5 ** (it / 20000.0))   # start 0.3, decay slowly
        eps_move = a[i_donor] * np.random.rand() * eps_frac

        a_prop = a.copy()
        a_prop[i_donor] -= eps_move
        a_prop[i_rec] += eps_move

        # evaluate
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature based on overall progress
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Utility: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Analytic starter shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_cosine(n: int) -> np.ndarray:
    """Half‑cosine window w(x)=sin(pi*x) on [0,1]"""
    x = np.linspace(0.0, 1.0, n)
    w = np.sin(np.pi * x)
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _make_cosine2(n: int) -> np.ndarray:
    """Raised cosine (Hann) window"""
    x = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    w /= w.sum()
    return w


def _make_parabola(n: int) -> np.ndarray:
    """Quadratic shape rising then falling (parabolic)"""
    x = np.linspace(-1, 1, n)
    w = 1 - x ** 2
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _make_semicircle(n: int) -> np.ndarray:
    """Discrete semicircle distribution on [0,1]"""
    x = np.linspace(-1, 1, n)
    w = np.sqrt(1 - x ** 2)
    w[w < 0] = 0.0
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of:
      1. Warm‑starts from a previous construction if available.
      2. Tries a handful of analytic starter shapes.
      3. Runs an exponentiated‑gradient sub‑gradient phase.
      4. Refines with a targeted simulated‑annealing “peak‑shaving’’ phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Determine a reasonably large target length.
    # --------------------------------------------------------------
    TARGET_N = 12000                     # chosen to keep each FFT cheap
    # --------------------------------------------------------------
    # 2. Prepare warm‑start candidates.
    # --------------------------------------------------------------
    starters = []

    # a) previous best (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            starters.append(_upsample(prev, TARGET_N))
    except Exception:
        pass

    # b) analytic shapes
    starters.extend([
        _make_uniform(TARGET_N),
        _make_cosine(TARGET_N),
        _make_cosine2(TARGET_N),
        _make_parabola(TARGET_N),
        _make_semicircle(TARGET_N),
    ])

    # ensure all starters are normalised (tiny rounding errors)
    starters = [s / s.sum() for s in starters if s.sum() > 0]

    # --------------------------------------------------------------
    # 3. Split the remaining time between the two optimisation phases.
    # --------------------------------------------------------------
    total_time = deadline - start_time
    # allocate 35 % to the exponentiated‑gradient phase, 55 % to SA,
    # keep a 10 % safety margin.
    exp_deadline = start_time + total_time * 0.35
    sa_deadline  = deadline - total_time * 0.10

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase A: exponentiated sub‑gradient descent on each starter.
    # --------------------------------------------------------------
    for idx, a0 in enumerate(starters):
        if time.time() >= exp_deadline:
            break
        # give each starter a fair slice of the remaining budget
        remaining = exp_deadline - time.time()
        per_starter_deadline = time.time() + remaining / max(1, len(starters) - idx)

        a_opt, score_opt = _run_exp_subgrad(
            a0,
            per_starter_deadline,
            start_time,
            step0=0.7,               # a bit more aggressive than the previous code
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Starter {idx} improve: "
                  f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Phase B: targeted simulated‑annealing (peak shaving).
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < sa_deadline:
        a_opt, score_opt = _run_peak_shaving(
            best_seq,
            sa_deadline,
            start_time
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final SA improve: "
                  f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Fallback: uniform distribution if everything else failed.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _make_uniform(TARGET_N)

    # Final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
