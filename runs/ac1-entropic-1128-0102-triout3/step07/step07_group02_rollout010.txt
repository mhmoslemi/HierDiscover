def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    # base lengths: [512, 1024, 2048, 4096, 8192, 16384]
    lengths = [512, 1024, 2048, 4096, 8192, 16384]    # adjust to time
    best_seq = None
    best_score = float('inf')
    # warm start from previous if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate time slice - maybe 15% of total each stage, adjust by n
        # compute remaining_time
        remaining = deadline - time.time()
        # allocate proportionally
        stage_time = min(remaining * 0.2, 120)  # limited to <=120 sec per stage
        stage_deadline = time.time() + stage_time
        # generate initial a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # arcsine shape
            idx = np.arange(n)
            a0 = 1.0/np.sqrt((idx+1)*(n-idx))
            a0 = np.maximum(a0, 0)
            a0 /= a0.sum()
        # run Adam optimizer
        a_opt, score_opt = _run_adam_opt(a0, n, stage_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam stage n={n} improve: {score_opt:.6f}")
    # after loops, optional local search
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a_final, score_final = _run_targeted_local_search(best_seq, n, deadline, start_time)

def _run_adam_opt(a0, n, deadline, start_time,
                  lr0=0.1, beta1=0.9, beta2=0.999, eps=1e-8,
                  T_start=0.02, T_end=5e-4, schedule_iters=500000):
    v = np.log(a0 + 1e-12)    # raw
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_a = a0.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        a = np.exp(v)
        a_sum = a.sum()
        if a_sum == 0:
            break
        a = a / a_sum
        # compute conv
        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam iter={it} n={n} score={cur_score:.6f}")
        # temperature schedule
        T = T_start * (T_end / T_start) ** (it / 20000.0)   # like earlier
        # compute gradient via softmax weight
        # w = softmax(conv/T)
        scaled = (conv - maxc) / T
        exps = np.exp(scaled)
        w = exps / exps.sum()
        rev_a = a[::-1]
        # grad_a = 2 * central part of conv of rev_a and w
        grad_full = _conv_rev_weight(rev_a, w)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # gradient w.r.t v via softmax (a = softmax(v))
        # ∂L/∂v_i = a_i * (grad_a_i - sum_j a_j grad_a_j)
        avg = np.dot(a, grad_a)
        grad_v = a * (grad_a - avg)
        # Adam updates
        lr = lr0 / math.sqrt(it)  # optional decay
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v2_hat = v2 / (1 - beta2 ** it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    # compute final a
    final_a = np.exp(v)
    final_a = final_a / final_a.sum()
    return final_a, best_score

"""
Propose a step‑function that (hopefully) beats the previous 1.51256 bound.

Algorithm
---------
1. Warm‑start from the previous best construction (global ``height_sequence_1``)
   if it exists, otherwise from a smooth “arcsine‑shaped’’ distribution.
2. Progressive refinement – repeatedly up‑sample the current best sequence
   to a larger length (512 → 1 024 → 2 048 → 4 096 → 8 192 → 16 384).
3. At each length run a *soft‑max Adam* optimiser:
   * a is represented as a soft‑max of unconstrained logits (guarantees
     non‑negativity and simplex normalisation);
   * the smooth objective is  T·log Σ exp(conv/T)  (T is an annealed
     temperature).  Its gradient is computed analytically via FFT‑based
     convolution;
   * Adam with a decaying learning‑rate (≈1/√t) is used.
4. After the final length finishes, run a targeted “peak‑shaving’’ local
   search (the same idea as the previous code) to polish the result.
5. Return the best sequence seen (normalised to sum = 1).

The whole routine respects the 1000 s budget; tiny safety margins are left
for the harness.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence ``a_rev`` (length n) with a weight
    vector ``w`` (length m).  Result has length n+m-1.
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution (soft‑max temperature)
# ----------------------------------------------------------------------
def _run_adam_opt(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lr0: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    T_start: float = 2.0e-2,
    T_end: float = 5.0e-4,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation to the max‑convolution.
    Returns the best sequence encountered and its exact score
    (2·n·max(conv) because the sum of the sequence stays 1).
    """
    # initialise logits for a soft‑max parametrisation
    logits = np.log(a0 + 1e-15)          # avoid log(0)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # current probabilities
        a = np.exp(logits)
        a_sum = a.sum()
        if a_sum == 0:
            break
        a /= a_sum                     # soft‑max normalisation

        # convolution and exact score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                f"iter={it}, score={cur_score:.6f}"
            )

        # temperature schedule (exponential decay)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # soft‑max weights for the smooth max
        scaled = (conv - cur_max) / T       # stabilise by subtracting max
        exps = np.exp(scaled)
        w = exps / exps.sum()               # shape (2n‑1,)

        # gradient wrt a (see derivation in the description)
        rev_a = a[::-1]
        full = _conv_rev_weight(rev_a, w)   # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]   # central block of length n

        # chain rule through the soft‑max parametrisation
        avg = np.dot(a, grad_a)             # ⟨a, grad_a⟩
        grad_logits = a * (grad_a - avg)    # ∂L/∂logits

        # Adam update (learning‑rate decays like 1/√t)
        lr = lr0 / math.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        # bias‑correction (cheap, safe for the limited number of iterations)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # final normalisation (just in case numerical drift occurred)
    final_a = np.exp(logits)
    final_a /= final_a.sum()
    return final_a, best_score


# ----------------------------------------------------------------------
# Targeted local search – same idea as the previous implementation
# ----------------------------------------------------------------------
def _run_targeted_local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Move a small amount of probability mass away from indices that
    contribute most to the current maximal convolution entry.
    Accept occasional uphill moves (simulated‑annealing style).
    """
    a = a0.copy()
    cur_conv = _conv_self(a)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)

        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor chosen proportionally to its contribution
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # receiver chosen among low‑contribution indices
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # annealing schedule for the Metropolis acceptance rule
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] local improve: "
                    f"n={n}, iter={it}, score={new_score:.6f}"
                )

    return best_a, best_score
