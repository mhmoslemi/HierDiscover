"""
Optimiser for ``evaluate_sequence`` (minimise 2·n·max(conv(a,a))/sum(a)^2).

The algorithm works in three stages and deliberately uses a *larger* vector
length (default 8192).  A longer vector gives a tighter theoretical lower
bound (≈1) and provides more degrees of freedom for flattening the
self‑convolution.  The stages are:

1.  Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
    The surrogate parameter α is increased gradually (continuation), so the
    optimiser first sees a very smooth landscape and then focusses on the
    true max.

2.  Projected sub‑gradient descent where the sub‑gradient is taken as the
    average of all *active* constraints (all convolution indices whose
    value is within a tiny tolerance of the current maximum).  This avoids
    the zig‑zag behaviour of using a single max index.

3.  A light simulated‑annealing polish that moves mass between coordinates
    (single‑source moves and three‑site re‑weight moves).  Temperature is
    exponential; Metropolis acceptance guarantees that the best solution
    found is never lost.

The optimiser respects the supplied time budget (≈1000 s).  All helpers are
top‑level functions; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (full length 2·len(a)−1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam on the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best iterate and its (unsmoothed) score 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length that works for all alphas in this phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # current (unsmoothed) score – just for logging
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- soft‑max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Multi‑active sub‑gradient descent
# ----------------------------------------------------------------------
def _run_multi_subgradient(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Gradient is the average of the gradients of *all* nearly‑maximal
    convolution entries (active set).  This reduces oscillations and yields
    a smoother descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94            # momentum decay
    step0 = 0.3            # initial step size factor
    it = 0
    eps_active = 1e-14    # tolerance for the active set

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ----- active set (all k within eps of the maximum) -----
        active = np.where(conv >= max_val - eps_active)[0]

        # ----- aggregate gradient over the active set -----
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]          # contribution of this k
        if len(active) > 1:
            grad /= len(active)                    # any convex combination works

        # ----- momentum + projected step -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Light mass‑transfer annealing.  Two move types are used:

    * single‑source → destination transfer (up to 30 % of source mass)
    * redistribution of the total mass on three randomly chosen sites

    Temperature decays exponentially; Metropolis acceptance is used.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if rng.random() < 0.7:               # single‑source move
            src = rng.choice(n, p=a)         # source weighted by current mass
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-18:
                continue
            eps = a[src] * rng.random() * 0.30   # up to 30 % of src mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:                                 # three‑site redistribution
            # choose three *distinct* sites (biased by mass)
            idxs = rng.choice(n, size=3, replace=False, p=a)
            total = a[idxs].sum()
            if total < 1e-18:
                continue
            # random Dirichlet(1,1,1) split of the total mass
            r = np.sort(rng.random(2))
            new_weights = np.array([r[0], r[1] - r[0], 1.0 - r[1]]) * total
            a_prop = a.copy()
            a_prop[idxs] = new_weights

        # evaluate
        new_max = _conv_self(a_prop).max()
        new_score = 2.0 * n * new_max
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score
