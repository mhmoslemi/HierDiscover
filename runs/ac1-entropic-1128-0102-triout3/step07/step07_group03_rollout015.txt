def _solve_lp_max_sum(f, M):
    n = len(f)
    # rows: 2*n-1
    # compute rows, cols.
    i = np.arange(n).reshape(-1,1)
    j = np.arange(n).reshape(1,-1)
    rows = (i + j).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n).astype(np.float64)
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a step‑function with a very low self‑convolution peak.

Algorithm
---------
1. Initialise a modest‑size (≈2000) probability vector.
   * If a previous best (`height_sequence_1`) is available, down‑sample it,
     otherwise start from the uniform distribution.
2. Perform a tiny parametric sweep: mix the uniform vector with an
   “edge‑heavy’’ shape a_i ∝ ((i+1)*(n‑i))^{-γ}.  The best mixture is kept.
3. Run a fast targeted local‑search (mass‑shaving) on the modest size
   vector for a few seconds.
4. Optional LP‑refinement (if SciPy is present):
   – solve max‑sum(g) s.t. f∗g ≤ M (current max‑convolution)  
   – mix the obtained g back into f with a small step size, repeat.
5. Upsample the best short vector to the final length (9600) preserving total
   mass.
6. Run a long targeted local‑search on the full‑size vector until the 1000 s
   budget is exhausted.
7. Return the final normalised sequence (list of floats).

The whole routine respects the time budget, prints progress and the current
score (= 2·n·max(conv)), and falls back to a uniform distribution if anything
fails.
"""
# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# SciPy is optional – we only need it for the LP refinement.
try:
    import scipy.optimize
    import scipy.sparse
    _HAS_SCIPY = True
except Exception:                       # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)-1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution of seq."""
    # For modest sizes np.convolve is fine and a bit faster than FFT.
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Evaluation score = 2·n·max(conv) (sum(seq) is assumed = 1)."""
    n = len(seq)
    return 2.0 * n * _max_conv(seq)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n, preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑shifting walk that reduces the maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: proportional to contribution (focus on hot spots)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: from low‑contribution region
        low_thr = np.percentile(contrib, 20)            # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3      # up to 30 % of donor
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, requires SciPy)
# ----------------------------------------------------------------------
def _precompute_lp_indices(n: int):
    """Return row / column index arrays for the convolution constraints."""
    i = np.arange(n).reshape(-1, 1)
    j = np.arange(n).reshape(1, -1)
    rows = (i + j).ravel()
    cols = np.tile(np.arange(n), n)
    return rows, cols


def _lp_refine(a: np.ndarray,
               max_iters: int,
               stage_deadline: float,
               start_time: float) -> np.ndarray:
    """Iteratively improve a by solving the LP max‑sum(g) s.t. a∗g ≤ M."""
    if not _HAS_SCIPY:          # pragma: no cover
        return a
    n = a.shape[0]
    rows, cols = _precompute_lp_indices(n)
    best_a = a.copy()
    best_M = _max_conv(best_a)

    for it in range(max_iters):
        if time.time() >= stage_deadline:
            break
        M = best_M

        # Build the (2n‑1)×n constraint matrix A_ub where
        # (A_ub @ g)[k] = sum_i a_i * g_{k-i}
        data = np.repeat(best_a, n)
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)),
                                        shape=(2 * n - 1, n))

        c = -np.ones(n)                     # maximise sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                     bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() <= 0:
            break
        g /= g.sum()                         # enforce sum = 1

        # Try several mixing rates; keep the first that actually reduces M.
        improved = False
        for t in (0.5, 0.2, 0.1, 0.05, 0.02):
            a_new = (1.0 - t) * best_a + t * g
            M_new = _max_conv(a_new)
            if M_new < best_M:
                best_M = M_new
                best_a = a_new
                improved = True
                print(f"[{time.time() - start_time:6.2f}s] LP improve: "
                      f"iter={it}, t={t:.2f}, score={2 * n * M_new:.6f}")
                break
        if not improved:
            # No improvement with any mixing factor – stop early.
            break

    return best_a


# ----------------------------------------------------------------------
# Parameter sweep (edge‑heavy mixtures)
# ----------------------------------------------------------------------
def _parameter_sweep(base_seq: np.ndarray, n: int) -> np.ndarray:
    """Search a few simple parametric families for a low‑peak start."""
    # Uniform vector (already normalised).
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score(best_seq)

    # include the provided base_seq as a candidate
    base_score = _score(base_seq)
    if base_score < best_score:
        best_seq = base_seq.copy()
        best_score = base_score

    # edge‑heavy family a_i ∝ ((i+1)*(n-i))^{-γ}
    i = np.arange(n) + 1
    denom = i * (n - i + 1)               # (i)*(n-i+1) > 0 for all i
    gammas = np.linspace(0.0, 0.5, 11)    # 0 → uniform, 0.5 → fairly edge‑biased
    alphas = [0.0, 0.3, 0.6, 1.0]         # mixing weight of edge‐shape

    for gamma in gammas:
        edge = denom.astype(float) ** (-gamma)
        edge /= edge.sum()
        for alpha in alphas:
            seq = (1.0 - alpha) * uniform + alpha * edge
            cur_score = _score(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[param sweep] gamma={gamma:.3f}, alpha={alpha:.1f}, "
                      f"score={cur_score:.6f}")

    return best_seq

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a tiny parametric sweep, fast targeted local search,
    an optional linear‑programming refinement and a final long‑run local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we never exceed the allotted budget
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a modest working size (fast LP / local search)
    # ------------------------------------------------------------------
    n_base = 2000
    if starter is not None:
        a_base = _upsample(starter, n_base)
    else:
        a_base = np.full(n_base, 1.0 / n_base)

    # ------------------------------------------------------------------
    # 1️⃣  Very cheap parametric sweep for a good start
    # ------------------------------------------------------------------
    a_best = _parameter_sweep(a_base, n_base)

    # ------------------------------------------------------------------
    # 2️⃣  Short local‑search refinement on the modest size vector
    # ------------------------------------------------------------------
    time_left = deadline - time.time()

    short_deadline = time.time() + min(30.0, time_left * 0.10)
    a_best, _ = _run_targeted_local_search(a_best, n_base, short_deadline, start_time)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Return the plain Python list expected by the harness
    # ------------------------------------------------------------------
    return [float(x) for x in a_big]
