"""
A fast multi‑phase optimiser for the max‑convolution objective.

The algorithm works in three stages (all time bounded by the supplied
budget):

1. **Parametric seeding** – generate a family of symmetric “U‑shaped”
   distributions a_i ∝ distance(i,nearest‑edge)^{-β}.  A small grid of
   β values is scanned and the best seed (lowest 2·n·max(conv)) is kept.

2. **Smooth + exact optimisation** – starting from the best seed (and,
   optionally, a few random restarts) we run two projected gradient
   procedures:
   * a temperature‑softened soft‑max surrogate (Nesterov momentum)
   * sub‑gradient descent on the true max‑convolution using an
     active‑set of nearly‑maximal indices.

   Both phases keep the vector on the probability simplex.

3. **Fine‑grained simulated annealing** – a Metropolis walk that moves
   tiny amounts of probability mass between random positions and occasionally
   smooths the whole sequence.  The temperature is decreased geometrically
   to a near‑zero final value.

The code respects the 1000 s wall‑clock budget, prints progress,
and finally returns a plain Python list of non‑negative heights that
minimise the supplied ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# 3. Parametric “U‑shape” seed generator
# ----------------------------------------------------------------------
def _beta_shape(n: int, beta: float) -> np.ndarray:
    """
    Symmetric shape a_i ∝ distance(i, nearest edge)^{-beta}.
    The distance is measured from the centre of each cell (i+0.5).
    """
    idx = np.arange(n, dtype=float)
    # distance to the closer edge (in cell‑units)
    d = np.minimum(idx + 0.5, n - idx - 0.5)
    # avoid div‑by‑zero – the smallest distance is 0.5
    a = d ** (-beta)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# 4. Objective (sum = 1, so score = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) assuming sum(a) = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 5. Soft‑max surrogate gradient (temperature‑smoothed max)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of T·log‑sum‑exp(conv/T) w.r.t. a."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # stable soft‑max weights
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    # the gradient of 2·∑ a_i a_{k-i} is 2·(rev(a) * w) restricted to length n
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# 6. Smooth (soft‑max) projected gradient descent
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov‑style GD on the smooth max‑convolution surrogate."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # temperature schedule (log‑linear)
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve "
                  f"iter={it:,} score={cur_score:.6f}")

        # geometric temperature decay
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# 7. Exact sub‑gradient descent (active‑set)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve "
                  f"iter={it:,} score={cur_score:.6f}")

        # active set: indices within rel_eps of the current maximum
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# 8. Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves tiny mass amounts and applies occasional smoothing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:               # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                eps = a[i] * np.random.rand() * 0.25  # up to 25 % of mass at i
                a_prop = a.copy()
                a_prop[i] -= eps
                a_prop[j] += eps
            else:
                continue
        else:                                      # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score

# 9. Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, explores a parametric family,
    refines with gradient and sub‑gradient phases and finishes with a
    simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n_final = 9600                     # final resolution (used by the baseline)
    total_length = n_final

    # Split the budget: 70 % for deterministic optimisation, 30 % for annealing
    grad_budget = (deadline - start_time) * 0.70
    anneal_budget = (deadline - start_time) - grad_budget

    # ------------------------------------------------------------------
    # 1. Parametric seeding (beta‑shape scan)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    betas = np.linspace(0.2, 2.5, 13)   # coarse grid, cheap to evaluate

    for beta in betas:
        if time.time() >= deadline:
            break
        a = _beta_shape(n_final, beta)
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] beta={beta:.2f} init score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient + sub‑gradient refinement (single long run)
    # ------------------------------------------------------------------

        # Sub‑gradient GD (only if time left)

        if sc_opt < best_score:

    # ------------------------------------------------------------------
