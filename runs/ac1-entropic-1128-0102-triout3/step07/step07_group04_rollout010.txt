def subgradient_max(a, conv):
    max_val = conv.max()
    # tolerance: 0 for exact max
    inds = np.where(conv >= max_val - 1e-12)[0] # maybe restrict.
    n = a.shape[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if inds.size > 1:
        grad /= inds.size
    return grad

def anneal_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    conv = conv_self(a)
    cur_score = 2 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    steps = int(5e5)  # maybe adjust depending on time left
    while time.time() < deadline and it < steps:
        it += 1
        # temperature schedule:
        T = 0.01 * (0.001 / 0.01)**(it/steps) # linear in exponent?
        src = rng.choice(n, p=a)  # a sum =1
        dst = rng.integers(0, n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.5  # transfer up to half mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # no need to renorm if we keep sum unchanged
        conv_new = conv_self(a_prop)
        new_score = 2 * n * conv_new.max()
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Search for a low‑max‑self‑convolution step function.

Algorithm
---------
1. Build a diverse portfolio of parametric vectors (uniform, exponential,
   power‑law, shifted Gaussian) of length `n = 8192`.  Each vector is
   normalised to sum‑to‑one.
2. Evaluate every candidate with the exact objective
      F(a) = 2·n·max(conv(a))                (sum(a)=1 ⇒ the denominator is 1)
   and keep the best few.
3. Run projected sub‑gradient descent on each kept candidate.
   The sub‑gradient of `max(conv(a))` is computed analytically; after each
   update the vector is projected onto the probability simplex.
   The runs share the remaining time budget.
4. Finish with a light simulated‑annealing (mass‑transfer) phase that
   explores nearby solutions and accepts worsening moves with a
   Metropolis rule.
5. Return the best vector found as a plain Python list.

All steps respect the overall time budget (default 1000 s).  The code
uses only NumPy and the supplied `evaluate_sequence` for score checks.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two not less than `x`."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(self‑convolution)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of g(a)=max(conv(a)) w.r.t. a
    (a is non‑negative, sum(a)=1).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Object‑function wrapper (exact objective for a normalised vector)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a)) assuming sum(a)=1.
    The function returns ``float('inf')`` if the sequence would be rejected
    by ``evaluate_sequence`` (e.g. sum too small).
    """
    s = a.sum()
    if s < 1e-9:
        return float("inf")
    a = a / s
    conv = _conv_self(a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.15,
                     beta: float = 0.94,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (step / max_steps)

        src = rng.choice(n, p=a)               # source proportional to its mass
        dst = rng.integers(0, n)                # destination uniform
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30     # transfer up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation) – not needed here but kept for completeness
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain list of floats (automatically
    normalised to sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Global problem size – we work directly at the largest size
    # --------------------------------------------------------------
    n = 8192

    # --------------------------------------------------------------
    # Optional starter (previous best) – normalise if present
    # --------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # --------------------------------------------------------------
    # 1️⃣  Build a portfolio of simple parametric candidates
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # previous best, if any (already normalised)
    if starter is not None:
        if starter.shape[0] != n:
            starter = _upsample(starter, n)
        candidates.append(starter.copy())

    # exponential decay a_i ∝ r^i
    for r in np.linspace(0.90, 0.9999, 30):
        seq = np.exp(np.arange(n, dtype=float) * np.log(r))
        seq = seq / seq.sum()
        candidates.append(seq)

    # power‑law decreasing a_i ∝ (n‑i)^q  (q≥0)
    for q in np.linspace(0.0, 6.0, 31):
        seq = (np.arange(n, 0, -1, dtype=float)) ** q
        seq = seq / seq.sum()
        candidates.append(seq)

    # shifted Gaussian (mass concentrated near the left side)
    x = np.arange(n, dtype=float)
    for sigma in (0.05, 0.10, 0.15, 0.20):
        mu = n * 0.25
        seq = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        seq = seq / seq.sum()
        candidates.append(seq)

    # --------------------------------------------------------------
    # 2️⃣  Quick evaluation and keep the best few (≤5)
    # --------------------------------------------------------------
    scored = []
    for seq in candidates:
        if time.time() >= deadline:
            break
        score = _objective(seq)
        scored.append((seq, score))
        print(f"[{time.time() - start_time:.2f}s] init candidate score={score:.6f}")

    # sort by score, keep the top 5 distinct solutions
    scored.sort(key=lambda kv: kv[1])
    top_seqs = [kv[0] for kv in scored[:5]]
    best_seq = top_seqs[0].copy()
    best_score = scored[0][1]

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent on each top candidate (share time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seqs:
        # allocate ~70 % of remaining time to sub‑gradient work
        sub_time_total = remaining * 0.70
        per_seq_time = sub_time_total / len(top_seqs)

        for seq in top_seqs:
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seq_time
            seq_opt, score_opt = _run_subgradient(
                seq, sg_deadline, start_time,
                step0=0.18, beta=0.94
            )
            if score_opt < best_score:
                best_score = score_opt
                best_seq = seq_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] subgrad improve (global) score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing refinement (use what is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _run_annealing(
            best_seq, deadline, start_time,
            start_temp=2e-2, end_temp=1e-7, max_steps=300_000
        )
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
