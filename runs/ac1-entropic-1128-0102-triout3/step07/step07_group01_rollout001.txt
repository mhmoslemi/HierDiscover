"""propose_candidate – multi‑phase simplex optimisation for a step‑function
=======================================================================

The algorithm searches for a non‑negative sequence `a` (interpreted as a
probability distribution, Σa = 1) that minimises

    2·n·max_k (a * a)_k ,

i.e. the `evaluate_sequence` score.  Three stages are used:

1. **Warm‑up** – exponentiated‑gradient descent on a smooth approximation of
   the max‑convolution, with an occasional symmetrisation step.
2. **Adaptive projected sub‑gradient descent** – a simple line‑search style
   method that accepts only moves that strictly lower the maximal
   self‑convolution entry.  After each step the vector is projected onto the
   probability simplex and symmetrised (a_i ← (a_i+a_{n‑1‑i})/2) because the
   optimal solution is expected to be symmetric.
3. **Targeted local refinement** – tiny random transfers of mass away from
   indices that contribute most to the current maximal convolution entry.
   Only improving transfers are kept.

The three stages share the time budget (≈1000 s).  The best sequence found
throughout the run is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) with respect to a, using all indices
    whose value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n‑1‑i} and renormalise to sum = 1."""
    b = (a + a[::-1]) / 2.0
    s = b.sum()
    if s > 0:
        b /= s
    return b


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated gradient (smooth‑max) with occasional symmetrisation
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative (exponentiated) update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a /= a.sum()

        # occasional symmetrisation to keep the search in the symmetric sub‑space
        if it % 2000 == 0:
            a = _symmetrize(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : adaptive projected sub‑gradient descent with line‑search style steps
# ----------------------------------------------------------------------
def _run_adaptive_subgradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              step0: float = 0.5,
                              step_inc: float = 1.02,
                              step_dec: float = 0.5,
                              min_step: float = 1e-9):
    """
    Simple adaptive sub‑gradient method that only accepts strictly decreasing
    max‑convolution values.  After every trial the vector is projected onto the
    simplex and symmetrised.
    """
    a = a0.copy()
    a /= a.sum()
    conv = _conv_self(a)
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2.0 * n * best_max
    step = step0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        grad = _subgradient_full(a, conv)

        # trial move
        a_trial = _project_simplex(a - step * grad)
        a_trial = _symmetrize(a_trial)

        conv_trial = _conv_self(a_trial)
        trial_max = conv_trial.max()

        if trial_max < cur_max - 1e-14:   # accept only strict improvement
            a = a_trial
            cur_max = trial_max
            step = min(step * step_inc, 1.0)   # slowly enlarge step
            if trial_max < best_max:
                best_max = trial_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                print(f"[{time.time() - start_time:.2f}s] adaptive improve: "
                      f"iter={it:,}, max={best_max:.8f}")
        else:
            # back‑track
            step = max(step * step_dec, min_step)
            # occasional random nudge to shake out of tiny plateaus
            if step <= min_step:
                idx = rng.integers(n)
                delta = a[idx] * 0.01
                a[idx] = max(a[idx] - delta, 0.0)
                a += delta / (n - 1)
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                step = step0

        # safety break
        if it % 20000 == 0 and time.time() >= deadline:
            break

    return best_a, best_max


# ----------------------------------------------------------------------
# Phase‑4 : targeted local refinement (tiny moves focused on the max)
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         eps_frac: float = 0.018):
    """
    Moves a small fraction of mass away from indices that contribute most
    to the current maximal convolution entry.  Only moves that lower the
    maximum are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current maximal index
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to that entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)
        if a[src] < 1e-15:
            continue

        # random destination (different from src)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional early exit check
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a

    a final targeted refinement, always keeping the best candidate found.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    final_n = 4800        # a good trade‑off between expressiveness and speed

    else:

        a0 /= a0.sum()

    a0 = _symmetrize(a0)

    # Record the best solution seen so far
    best_seq = a0.copy()
    best_score = 2.0 * final_n * _conv_self(a0).max()
    print(f"[{time.time() - start_time:.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – short exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    stage1_deadline = min(deadline, start_time + 25.0)
    a_warm, score_warm = _run_exponential_gradient(

        )
        score_sub = 2.0 * final_n * max_sub
        if score_sub < best_score:
            best_score = score_sub

            print(f"[{time.time() - start_time:.2f}s] after adaptive subgrad: "

    # ------------------------------------------------------------------

        )

        if score_ref < best_score:

            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine: score={best_score:.6f}")

    # ------------------------------------------------------------------

    best_seq = np.maximum(best_seq, 0.0)

    if total > 0:

    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
