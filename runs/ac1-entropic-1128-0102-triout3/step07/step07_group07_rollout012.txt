from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _lp_step(a):
    n = len(a)
    conv = _conv_self(a)
    M = conv.max()
    # choose active indices: take top K values
    K = min(10, len(conv))
    idx_sorted = np.argpartition(-conv, K-1)[:K]  # unsorted top K
    active = np.sort(idx_sorted) # optional sort

    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        rows.extend([row_idx]*len(i))
        cols.extend(j.tolist())
        data.extend(a[i].tolist())
    A = csr_matrix((data, (rows, cols)), shape=(len(active), n))
    b_ub = np.full(len(active), M)
    # maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0,None)]*n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a non‑negative step function that (almost) minimizes ``evaluate_sequence``.

Algorithm
---------
* Build a small collection of diverse initial vectors (uniform, exponential,
  triangular, a random smooth vector and the optional previous best).
* For each start run a fast projected sub‑gradient descent on the exact
  max‑convolution objective (the objective is homogeneous, therefore the
  sequence is kept on the probability simplex).
* Afterwards perform a few Frank‑Wolfe‑style LP “mass‑expansion’’ steps:
  we maximise the total mass of a new vector b subject to the current
  convolution bound ‖a∗b‖∞≤M (M is the current maximum of a∗a).  The obtained
  b is mixed with a (convex combination) and projected back onto the simplex.
  This operation often reduces the maximum convolution further.
* Finally a short simulated‑annealing phase refines the solution by random
  mass‑transfer moves.
* The whole routine respects the total time budget (default 1000 s) and
  returns the best sequence found.

The code uses only NumPy, SciPy (for a tiny linear programme) and the
provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# ----- Helper functions (FFT convolution, simplex projection, etc.) -----
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # Soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # Cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserves sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# ----- Projected sub‑gradient descent on the true (max) objective -----
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) gradient descent on
        f(a) = max_k (a*a)[k]   with a on the simplex.
    Returns the best vector seen and its objective value 2·n·max_conv.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient of the max (active set) --------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# ----- LP “mass‑expansion’’ step (Frank‑Wolfe style) --------------------
# ----------------------------------------------------------------------


def _solve_lp_for_active(a: np.ndarray,
                         M: float,
                         active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve
        max   sum b_j
        s.t.  sum_i a_i b_{k-i} <= M   for each k in active_idx
              b_j >= 0
    Returns the optimal b (or None on failure).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        rows.extend([row] * len(i))
        cols.extend(j.tolist())
        data.extend(a[i].tolist())

    if not rows:
        return None

    A = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M, dtype=float)
    c = -np.ones(n, dtype=float)          # maximise sum -> minimise -sum
    bounds = [(0, None)] * n

    # use HiGHS interior‑point solver (handles sparse A)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _run_lp_alternation(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_steps: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the result with the
    current vector.  The mixing factor t is adapted dynamically.
    Returns the best vector and its exact evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    # ensure normalisation on the simplex
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a.sum()

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    t = 0.5                     # initial mixing weight

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # choose a modest active set: the top‑k entries (k≈20)
        top_k = min(20, conv.size)
        idx_top = np.argpartition(-conv, top_k - 1)[:top_k]

        b = _solve_lp_for_active(a, M, idx_top)
        if b is None:
            break

        # mix and project back onto the simplex
        g = b / (b.sum() + 1e-16)
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = _project_simplex(a_candidate)

        sc = evaluate_sequence(a_candidate.tolist())
        if sc < best_score - 1e-9:
            best_score = sc
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:6.2f}s] LP improve step={step+1}, score={sc:.6f}")
            # if improvement, be a little bolder
            t = min(0.9, t * 1.1)
        else:
            # otherwise become more conservative
            t = max(0.1, t * 0.7)

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx + 1}")

        # initialise on the simplex
        a = np.maximum(a0, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a.sum()

        # ------------------------------------------------------------------
        # 1. Projected sub‑gradient descent (the heavy‑weight polishing phase)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            # devote roughly half of the remaining budget to this phase
            gd_deadline = time.time() + 0.5 * remaining
            a_gd, score_gd = _run_gradient_descent(
                a,
                gd_deadline,
                start_time,
                beta=0.94,
                step0=0.25,
            )
            # score_gd is already 2·n·max_conv; with sum=1 this equals evaluate()
            if score_gd < best_score:
                best_score = score_gd
                best_seq = a_gd.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (grad) *** "
                      f"score={best_score:.6f}")
            a = a_gd  # continue from the polished vector

        # ------------------------------------------------------------------
        # 2. Frank‑Wolfe‑style LP mass‑expansion steps
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            lp_deadline = time.time() + 0.2 * remaining
            a_lp, score_lp = _run_lp_alternation(a, lp_deadline, start_time)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (LP) *** "
                      f"score={best_score:.6f}")
            a = a_lp

        # ------------------------------------------------------------------
        # 3. Final annealing (use whatever time is left)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _run_annealing(
                a,
                deadline,
                start_time,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=300_000,
            )
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST (anneal) *** "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time (guard against numerical drift) and
    # return the result as a plain list of Python floats.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
