def _run_subgrad_adam(a0, n, deadline, start_time, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = np.inf
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad ADAM improve: n={n}, iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))
        # gradient as per convex subgradient
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

def _run_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # compute contributions for shift idx_max
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        # donor weighted by contribution
        prob = contrib / contrib.sum() if contrib.sum() > 0 else None
        if prob is not None and prob.sum() > 0:
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: low contribution indices
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.5  # up to half mass
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

"""
A three‑phase search for a non‑negative step function that minimises
``evaluate_sequence``.

Phase A – “soft‑max” Adam optimisation.
    The true objective 2·n·max(conv) is approximated by the smooth
    log‑sum‑exp of the self‑convolution.  Adam adapts the step size per
    coordinate and a Euclidean projection onto the probability simplex
    (non‑negative entries that sum to 1) keeps the iterates feasible.

Phase B – exact sub‑gradient Adam on the true max‑convolution.
    The sub‑gradient of the active (max) entry is cheap to compute.
    Adam again provides a stable, adaptive descent.

Phase C – focused “peak‑shaving’’ local search.
    A small amount of probability mass is moved from indices that
    contribute most to the current maximal convolution entry to indices
    that contribute little.  A Metropolis acceptance rule with an
    annealed temperature permits occasional uphill moves.

The routine works on a single, relatively large sequence (≈20 k terms),
uses only NumPy, and respects the 1000‑second budget.
"""
# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple helper: keep a vector on the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a soft‑max of the convolution (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of  Σ_k w_k(conv_k)·conv_k  where
    w_k = soft‑max(conv_k / temperature).
    The factor 2 comes from ∂/∂a_i (aᵀ J_k a) for the Toeplitz matrix J_k.
    """
    # stabilised soft‑max
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    rev_a = a[::-1]                                     # reversed sequence
    full = _conv_rev_weight(rev_a, w)                  # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]                  # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the smooth max‑convolution (Phase A)
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr0: float = 0.8,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      temp_start: float = 2e-2,
                      temp_end: float = 5e-4,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth (log‑sum‑exp) surrogate of max(conv).
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()                 # true objective (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max ADAM improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (exponential decay)
        T = temp_start * (temp_end / temp_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                         # scalar learning‑rate decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser for the exact max‑convolution (Phase B)
# ----------------------------------------------------------------------
def _run_subgrad_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam on the true (nonsmooth) objective 2·n·max(conv).
    The sub‑gradient is taken from the currently active convolution entry.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad ADAM improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient for the active entry
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving) – Phase C
# ----------------------------------------------------------------------
def _run_local_search(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass away from indices that
    contribute most to the current maximal convolution entry.
    A simple Metropolis rule with a cooling temperature allows occasional
    uphill moves to escape shallow local minima.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the active shift
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        # pick a donor proportional to its contribution
        if contrib.sum() > 0.0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # pick a recipient from the low‑contribution tail
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # move a fraction of the donor's mass
        eps = a[i_donor] * np.random.rand() * 0.5   # up to 50 % of donor
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # annealed temperature
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta < 0.0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine consists of three phases (soft‑max Adam → exact sub‑gradient
    Adam → targeted local search) on a single large sequence (≈20 k terms).
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so the harness never times‑out while we are still working
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from the previous best construction if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a sufficiently large length – larger n gives a tighter
    # theoretical lower bound (≈2·n/(2·n‑1)).
    # ------------------------------------------------------------------
    n_target = 20000                               # fits comfortably in memory
    if starter is not None:
        # linear up‑sampling while preserving total mass
        old_x = np.linspace(0.0, 1.0, starter.size)
        new_x = np.linspace(0.0, 1.0, n_target)
        a0 = np.interp(new_x, old_x, starter)
    else:
        a0 = np.random.rand(n_target)

    a0 = _project_simplex(a0)                     # guarantee feasibility

    # ------------------------------------------------------------------
    # Time allocation
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    soft_time   = total_time * 0.45
    subgrad_time = total_time * 0.35
    # the remainder (plus a tiny safety margin) goes to the local search
    local_deadline = deadline - 0.1

    soft_deadline   = start_time + soft_time
    subgrad_deadline = soft_deadline + subgrad_time

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase A – smooth Adam optimisation
    # ------------------------------------------------------------------
    a_soft, score_soft = _run_softmax_adam(
        a0, n_target, soft_deadline, start_time,
        lr0=0.8, beta1=0.9, beta2=0.999,
        temp_start=2e-2, temp_end=5e-4
    )
    if score_soft < best_score:
        best_score = score_soft
        best_seq = a_soft.copy()
        print(f"[{time.time() - start_time:6.2f}s] softmax final improve: "
              f"score={score_soft:.6f}")

    # ------------------------------------------------------------------
    # Phase B – exact sub‑gradient Adam
    # ------------------------------------------------------------------
    if time.time() < subgrad_deadline:
        a_sub, score_sub = _run_subgrad_adam(
            best_seq, n_target, subgrad_deadline, start_time,
            lr0=0.5, beta1=0.9, beta2=0.999
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad final improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase C – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if time.time() < local_deadline:
        a_loc, score_loc = _run_local_search(
            best_seq, n_target, local_deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final sanity normalisation (accumulated rounding may have drifted)
    best_seq = np.asarray(best_seq, dtype=float)
    best_seq = _project_simplex(best_seq)

    # return a plain Python list – the harness will call evaluate_sequence on it
    return [float(x) for x in best_seq]
