    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2*n - 1
    it = 0
    while time.time() < deadline:
        it += 1
        # compute elapsed fraction for eps_frac decay
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1 - frac) + min_eps_frac * frac

        k_max = int(conv.argmax())
        # contributions to max
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        # destination candidate weights
        # compute dest contribution weight = a_{k_max - dst} if valid else 0.
        dst_idx = np.arange(n)
        j_dst = k_max - dst_idx
        valid_dst = (j_dst >= 0) & (j_dst < n)
        dest_weight = np.zeros(n)
        dest_weight[valid_dst] = a[j_dst[valid_dst]]
        # Inverse weighting: lower dest_weight better
        inv = 1.0 / (dest_weight + 1e-12)
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst:
            continue
        # if a[src] extremely low skip
        if a[src] < 1e-16:
            continue
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0:
            continue

        # compute new conv via O(N) update
        delta = np.zeros(conv_len)
        # add contributions of moving eps to dst
        delta[dst:dst+n] += 2 * eps * a
        delta[src:src+n] -= 2 * eps * a
        # eps^2 corrections
        delta[2*src] += eps * eps
        delta[2*dst] += eps * eps
        delta[src + dst] -= 2 * eps * eps

        new_conv = conv + delta
        new_max = new_conv.max()

        if new_max < max_val - 1e-14:
            # accept
            a[src] -= eps
            a[dst] += eps

                best_a = a.copy()

        if it % 200000 == 0:
            # check progress
            if time.time() >= deadline:
                break
    return best_a

"""
A multi‑phase optimizer for the self‑convolution bound.

Phase 1 – exponential‑gradient descent on a log‑sum‑exp approximation of the
maximum of the self‑convolution.

Phase 2 – projected sub‑gradient descent on the exact (non‑smoothed) max.

Phase 3 – an O(n) “targeted‑refine” that moves a tiny amount of probability
mass away from indices that contribute most to the current maximal
convolution entry.  The move is accepted only if the maximal value strictly
decreases; the step‑size decays during the run and the destination is
chosen among the indices with the smallest contribution to the offending
sum, which yields a much higher improvement rate than a uniform random
choice.

All stages respect the overall 1000 s budget and the best sequence found
throughout the run is returned (normalised to sum = 1).  The code works
with any existing global `height_sequence_1` (a previously found construction)
as a warm start, but also works from a fresh random initialisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 10000):
    """Multiplicative mirror‑descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """Projected sub‑gradient descent on the true maximal convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of max
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : O(n) targeted local refinement (fast, fine‑grained)
# ----------------------------------------------------------------------
def _run_targeted_refine_optimized(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   init_eps_frac: float = 0.05,
                                   min_eps_frac: float = 0.001):
    """
    Very fast O(n) refinement.
    Moves a small fraction of mass away from indices that contribute most to the
    current maximal convolution entry.  Destination indices with the smallest
    contribution to that entry are preferred.  The step‑size decays linearly
    from `init_eps_frac` to `min_eps_frac`.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    conv_len = 2 * n - 1
    it = 0

    while time.time() < deadline:
        it += 1

        # linearly decay the move‑size
        elapsed = time.time() - start_time
        total = deadline - start_time
        decay = min(1.0, elapsed / total)
        eps_frac = init_eps_frac * (1.0 - decay) + min_eps_frac * decay

        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # choose source index – weighted by its contribution to the max
        # --------------------------------------------------------------
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # --------------------------------------------------------------
        # choose destination – favour indices that *do not* contribute to k_max
        # --------------------------------------------------------------
        dst_idx = np.arange(n)
        jdst = k_max - dst_idx
        valid_dst = (jdst >= 0) & (jdst < n)
        dest_weight = np.zeros(n, dtype=float)
        dest_weight[valid_dst] = a[jdst[valid_dst]]
        # indices with zero contribution are the best
        zero_mask = dest_weight < 1e-14
        if np.any(zero_mask):
            candidates = np.where(zero_mask)[0]
        else:
            # otherwise pick the smallest contribution
            min_val = dest_weight.min()
            candidates = np.where(dest_weight == min_val)[0]
        dst = rng.choice(candidates)

        if src == dst:
            continue

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        eps = a[src] * eps_frac * rng.random()
        if eps <= 0.0:
            continue

        # fast O(n) update of the convolution (exact up to round‑off)
        delta = np.zeros(conv_len, dtype=float)

        # linear part 2*eps*(a_{k-dst} - a_{k-src})
        delta[dst:dst + n] += 2.0 * eps * a
        delta[src:src + n] -= 2.0 * eps * a

        # quadratic eps² correction (only three entries are affected)
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        conv_prop = conv + delta
        new_max = conv_prop.max()

        # accept only strict improvement
        if new_max < max_val - 1e-14:
            # commit the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine_opt improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodically recompute the convolution from scratch to avoid drift
        if it % 50000 == 0:
            conv = _conv_self(a)
            max_val = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the time budget.  The routine follows the
    three‑phase scheme described above and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – warm‑up with exponentiated gradient
    # ------------------------------------------------------------------
    # Allocate modest time (≈5 % of the budget) for a quick warm start.
    phase1_time = min(30.0, 0.05 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.5,
        steps_per_alpha=10000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Keep around 70 % of the remaining budget for this phase.
        phase2_time = max(30.0, 0.70 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fast targeted refinement (fine‑grained)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine_optimized(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps_frac=0.05,
            min_eps_frac=0.001,
        )
        # final objective (pure objective, no clamping)
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
