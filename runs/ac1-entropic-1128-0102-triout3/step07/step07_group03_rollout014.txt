"""
Simple stochastic hill‑climbing optimizer for the step‑function problem.

The evaluator used by the competition is:
    2 * n * max(conv(a,a)) / (sum(a)**2)

All of the heuristics below work under the (usual) normalisation
sum(a)=1, because the ratio is homogeneous – scaling the whole vector
does not change the score.  The algorithm therefore keeps the total
mass exactly constant while it repeatedly moves a small amount of mass
from a “donor’’ index to a “receiver’’ index.  Moves are accepted if they
decrease the current objective, otherwise they are accepted with a
Metropolis probability that slowly cools down.  The convolution is
computed by FFT (full linear convolution) which is O(n log n) and fast
enough to allow many thousands of proposals within the 1000 s budget.

If a previously discovered good sequence is available in the global
variable ``height_sequence_1`` it is used as a warm‑start, otherwise a
uniform distribution is used.

The final sequence is projected back onto the probability simplex
(non‑negative, sum=1) before being returned.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: project onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast (full) linear self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length = 2*len(a)-1)."""
    n = a.shape[0]
    # next power of two for zero‑padding (FFT length)
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (identical to the official evaluator for sum(a)=1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2·n·max(conv) – the value returned by evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def _stochastic_hill_climb(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           eps_initial: float = 0.05,
                           eps_decay: float = 0.9999) -> np.ndarray:
    """
    Perform a long‑run Metropolis‑style hill‑climb.
    a0 must already be normalised to sum = 1.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _score(best)

    n = a.shape[0]
    cur_score = best_score
    eps = eps_initial
    it = 0

    while time.time() < deadline:
        it += 1

        # pick donor (must have some mass) and recipient
        donor = np.random.randint(n)
        if a[donor] < 1e-12:
            continue
        recip = np.random.randint(n)
        if recip == donor:
            continue

        # amount of mass to move (a fraction of the donor's current mass)
        delta = a[donor] * eps * np.random.rand()
        a_prop = a.copy()
        a_prop[donor] -= delta
        a_prop[recip] += delta

        # keep the vector feasible (no negative entries)
        if a_prop[donor] < 0:
            continue

        cand_score = _score(a_prop)

        # temperature schedule – linearly decaying in log‑space
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = max(1e-8, (1.0 - elapsed) * 0.1)   # simple linear schedule

        accept = cand_score < cur_score or math.exp((cur_score - cand_score) / T) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best score: {best_score:.6f}")

        # slowly shrink the step size – this helps to fine‑tune at the end
        eps *= eps_decay
        if eps < 1e-6:
            eps = eps_initial * 0.1   # occasional reset to keep moves possible

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    official score.  The routine works as follows:

        1. Warm‑start from ``height_sequence_1`` if it exists,
           otherwise start from a uniform distribution of length 9600.
        2. Run a long stochastic hill‑climb that preserves the total
           mass (sum = 1) while shuffling it between indices.
        3. Project the final vector onto the simplex and return it as a
           plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Warm start
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim == 1 and init.size > 0:
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0:
                init = init / s
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback: uniform distribution, length 9600 (same as the previous best)
        n = 9600
        init = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Stochastic hill‑climb
    # ------------------------------------------------------------------
    best_seq = _stochastic_hill_climb(init, deadline, start_time)

    # ------------------------------------------------------------------
    # Final sanity projection (the optimizer always keeps sum=1,
    # but rounding errors can accumulate)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        # extremely unlikely – fall back to uniform
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
