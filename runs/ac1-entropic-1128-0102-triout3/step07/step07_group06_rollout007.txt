def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = np.inf

    # Stage schedule lengths
    lengths = [512, 2048, 4096, 8192]
    # allocate times: 60% for gradient phases (Adam+subgrad), 40% for annealing
    total_time = deadline - start
    grad_budget = total_time * 0.6
    anneal_budget = total_time - grad_budget

    # Run gradient optimization across lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate slice of grad_budget proportionally
        proportion = n / sum(lengths)
        stage_deadline = time.time() + grad_budget * proportion

        # Initialize
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)
            a0 /= a0.sum()
        
        # Adam phase
        a_opt, score = _adam_optimize(a0, n, stage_deadline)
        # Subgradient phase
        if time.time() < stage_deadline:
            a_opt2, score2 = _subgrad_optimize(a_opt, n, stage_deadline)
            if score2 < score:
                a_opt, score = a_opt2, score2

        # Evaluate true objective
        cur_score = 2 * n * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] length {n} improved: {cur_score:.6f}")

        # Update remaining grad budget
        elapsed = time.time() - start
        grad_budget = max(0.0, deadline - start - anneal_budget - elapsed)

    # Final annealing on best_seq
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a_ref, score_ref = _annealing_optimize(best_seq, n, deadline)
        # compute true objective
        cur_score = 2 * n * _conv_self(a_ref).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start:.2f}s] final annealing improve: {cur_score:.6f}")

    # Ensure nonnegatives and normalize
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]

def _gradient_softmax(a, w):
    rev_a = a[::-1]
    n = a.shape[0]
    m_len = w.shape[0]
    fft_len = 1 << ((n + m_len - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
    grad = 2.0 * full[n-1:n-1+n]
    return grad
```

def _annealing_optimize(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    cur_score = best_score

    # set parameters
    start_temp = 0.5
    end_temp = 1e-7

    total_steps = int(3e6)  # allocate many steps
    i = 0
    while time.time() < deadline and i < total_steps:
        i += 1
        # compute temperature geometrically:
        fraction = i / total_steps
        T = start_temp * (end_temp / start_temp) ** fraction

        # propose move
        if rng.random() < 0.85:
            i_idx, j_idx = rng.integers(0, n, size=2)
            if i_idx != j_idx and a[i_idx] > 0:
                delta = a[i_idx] * rng.random() * 0.3  # up to 30% transfer
                a_prop = a.copy()
                a_prop[i_idx] -= delta
                a_prop[j_idx] += delta

            k = rng.choice([3,5,7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2 * n * _conv_self(a_prop).max()
        delta_score = new_score - cur_score

        if delta_score < 0 or np.exp(-delta_score / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            #print progress

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The optimiser works in three phases within the 1000 s wall‑clock budget:

1. **Adam on a temperature‑smoothed max‑convolution** – a fast first‑order
   method that approximates the true ∞‑norm by a soft‑max.  The learning
   rate is adapted with Adam and the iterates are projected onto the
   probability simplex.

2. **Active–set sub‑gradient descent** – refines the solution by using the
   true max‑convolution and an active set of nearly‑binding indices.

3. **Extended simulated‑annealing** – mass‑transfer and local smoothing
   moves are accepted with a geometric temperature schedule; this phase
   runs for the remaining time and is the main source of final improvement.

The algorithm constantly tracks the best sequence (in terms of the exact
objective 2·n·max(conv) – sum is kept equal to 1) and up‑samples it to a
larger resolution when time permits.  Multiple restarts and a simple
sinusoidal warm‑start are used to escape poor local minima.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Compute the soft‑max weights of conv at a given temperature."""
    maxc = conv.max()
    # shift for numerical stability
    scaled = (conv - maxc) / temperature
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    w = exps / exps.sum()
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution.
    a : current sequence (length n)
    w : soft‑max weights (length 2·n‑1)
    Returns grad (length n).
    """
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]                # = 2*n-1
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    # central slice yields the gradient vector
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   T_start: float = 5e-2,
                   T_end: float = 5e-4,
                   max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Adam on the temperature‑smoothed max‑convolution (projected onto simplex)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n} it={it} score={cur_score:.6f}")

        # temperature schedule (geometric)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        w = _softmax_weights(conv, T)          # soft‑max of convolution
        grad = _softmax_gradient(a, w)         # gradient of the smooth max

        # Adam update (note the minus sign for descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.15,
                      beta: float = 0.9,
                      rel_eps: float = 0.01,
                      max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve: n={n} it={it} score={cur_score:.6f}")

        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m_len = w.shape[0]
        fft_len = 1 << ((n + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _annealing_optimize(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        start_temp: float = 0.5,
                        end_temp: float = 1e-7,
                        max_steps: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk with tiny mass‑transfer moves and occasional smoothing."""
    rng = np.random.default_rng()
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a).max()
    best_seq = a.copy()
    best_score = cur_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        if rng.random() < 0.85:                     # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * rng.random() * 0.30       # up to 30 % of source mass
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                        # local smoothing move
            k = rng.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n} it={it} score={cur_score:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, runs Adam+sub‑gradient phases on
    increasing resolutions and finishes with a long simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously exported construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage schedule – progressively finer resolutions.
    # ------------------------------------------------------------------
    stage_lengths = [512, 2048, 4096, 8192]          # final length will be 8192 (or up‑sampled later)
    total_n = sum(stage_lengths)

    total_time = deadline - start_time
    grad_time_budget = total_time * 0.65            # ~65 % for gradient phases
    anneal_time_budget = total_time - grad_time_budget

    # ------------------------------------------------------------------
    # Gradient phases on each resolution.
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time proportional to resolution size
        stage_time = grad_time_budget * (n / total_n)
        stage_deadline = time.time() + stage_time

        # initialise a vector for this resolution
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # sinusoidal warm‑start + small random perturbation
            a0 = np.sin(np.linspace(0.0, math.pi, n))
            a0 += 0.01 * np.random.rand(n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # ---- Phase 1: Adam on smooth surrogate ----
        a_opt, score_opt = _adam_optimize(
            a0, n, stage_deadline, start_time,
            lr=0.6, beta1=0.9, beta2=0.999,
            T_start=5e-2, T_end=5e-4, max_iter=10**7
        )

        # ---- Phase 2: exact sub‑gradient descent (if time permits) ----
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _subgrad_optimize(
                a_opt, n, stage_deadline, start_time,
                step0=0.12, beta=0.9, rel_eps=0.01, max_iter=10**7
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # keep global best
        cur_score = 2.0 * n * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Stage n={n} improve: score={cur_score:.6f}")

        # recompute remaining gradient budget
        elapsed = time.time() - start_time
        grad_time_budget = max(0.0,
                               deadline - start_time - anneal_time_budget - elapsed)

    # ------------------------------------------------------------------
    # Optional up‑sampling before the final annealing (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # double the resolution if we still have at least a few seconds left
        if remaining > 30.0 and best_seq.shape[0] < 15000:
            new_n = min(15000, best_seq.shape[0] * 2)
            best_seq = _upsample(best_seq, new_n)

    # ------------------------------------------------------------------
    # Final annealing phase – long run on the final resolution.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _annealing_optimize(
            best_seq, n_final,
            deadline, start_time,
            start_temp=0.5, end_temp=1e-7,
            max_steps=8_000_000   # a large number; actual steps limited by time
        )
        cur_score = 2.0 * n_final * _conv_self(a_ref).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final anneal improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()

        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list
    return [float(x) for x in best_seq]
