def generate_shape(kind, param, n):

"""
Propose a step‑function that (approximately) flattens its self‑convolution.

Algorithm
---------
1. **Parameterised warm‑starts** – several analytically‑motivated shapes
   (Beta‑U‑shapes, uniform, random) are generated on a small scale.
2. **Progressive up‑sampling** – the current best shape is linearly
   interpolated to a larger length and refined again.  Lengths are
   doubled successively up to ~3·10⁴ terms.
3. **Smooth max‑convolution optimisation** – for each scale we minimise a
   soft‑max surrogate of the max‑convolution using Adam‑style
   projected gradient descent on the probability simplex.
4. **Intensive peak‑shaving** – the final candidate is polished by a
   long simulated‑annealing local search that repeatedly moves a small
   amount of mass from indices that contribute most to the current
   maximal convolution entry to low‑contribution indices.
5. The best sequence seen is normalised to sum 1 and returned.

All computation respects the 1000 s budget; the routine keeps a safety
margin for the harness.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: centre of simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper for the soft‑max surrogate
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    Returns ∂/∂a Σ_k w_k conv_k where w_k = softmax(conv_k / temperature).
    The factor 2 comes from the symmetry of the Toeplitz matrix.
    """
    # compute soft‑max weights
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # avoid overflow
    w = exps / exps.sum()

    # convolution of reversed a with the weight vector w
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1: n - 1 + n]           # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.6,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation of the soft‑max surrogate."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size
        alpha = step0 / (math.sqrt(it) + 1.0)

        a = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (optional, kept cheap)
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.4,
                    beta: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution entry."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient of the active entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / (math.sqrt(it) + 1.0)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (mass‑preserving)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric shapes (Beta‑U family, uniform, random)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1} (U‑shaped when α<1).
    a_i ∝ (i+0.5)^{α-1} * (n-i-0.5)^{α-1}
    """
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) * (n - i - 0.5)
    # raise to (α-1); protect against zero
    vals = np.power(vals, alpha - 1.0, where=vals > 0, out=np.zeros_like(vals))
    vals[vals < 0] = 0.0
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals = np.full(n, 1.0 / n)
    return vals


def _make_uniform_shape(n: int) -> np.ndarray:
    """Trivial uniform distribution."""
    return np.full(n, 1.0 / n)


def _make_random_shape(n: int) -> np.ndarray:
    """Random point on the simplex."""
    rng = np.random.rand(n)
    rng /= rng.sum()
    return rng


# ----------------------------------------------------------------------
# Targeted “peak‑shaving’’ local search (simulated‑annealing style)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly relocate a small fraction of probability mass from indices
    strongly contributing to the current maximal convolution entry to
    low‑contribution indices.  A simple Metropolis rule with a cooling
    schedule accepts occasional uphill moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # move a modest fraction of donor mass
        eps = a[i_donor] * np.random.rand() * 0.25
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature schedule (linear in elapsed fraction)
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine builds several analytically‑motivated starting points,
    progressively refines them with projected Adam on a soft‑max surrogate,
    and finishes with an intensive peak‑shaving local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Progressive refinement schedule (doubling lengths)
    # ------------------------------------------------------------------
    target_lengths = [1024, 2048, 4096, 8192, 16384, 32768]   # up to ~3·10⁴
    total_len = sum(target_lengths)
    total_time = deadline - start_time
    # about 60 % of the budget for gradient phases,
    # the rest for the final local search
    grad_budget = total_time * 0.60
    local_budget = total_time - grad_budget

    best_seq = None
    best_score = float("inf")
    elapsed = 0.0

    # ------------------------------------------------------------------
    # Gradient phases (soft‑max + optional sub‑gradient)
    # ------------------------------------------------------------------
    for n in target_lengths:
        if time.time() >= deadline:
            break

        # allocate time proportionally to length
        stage_frac = n / total_len
        stage_time = max(1.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise candidate
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # try a few analytically motivated shapes and keep the best
            candidates = [
                _make_beta_shape(n, 0.3),   # very U‑shaped
                _make_beta_shape(n, 0.5),   # arcsine‑like
                _make_beta_shape(n, 0.7),   # milder U‑shape
                _make_uniform_shape(n),
                _make_random_shape(n)
            ]
            # quick evaluation of the surrogate (max convolution) to pick init
            scores = []
            for cand in candidates:
                conv = _conv_self(cand)
                scores.append(2.0 * n * conv.max())
            a0 = candidates[int(np.argmin(scores))]

        # ---- Adam optimisation on the soft‑max surrogate ----
        a_opt, score_opt = _run_softmax_adam(
            a0, n, stage_deadline, start_time,
            step0=0.6, beta1=0.9, beta2=0.999
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # ---- optional short sub‑gradient refinement (cheap) ----
        if time.time() < deadline:
            sub_deadline = min(deadline, time.time() + 0.15 * stage_time)
            a_opt2, score_opt2 = _run_subgrad_gd(
                best_seq, n, sub_deadline, start_time,
                step0=0.4, beta=0.9
            )
            if score_opt2 < best_score:
                best_score = score_opt2
                best_seq = a_opt2.copy()
                print(f"[{time.time() - start_time:6.2f}s] Subgrad refinement n={n} "
                      f"improve: score={score_opt2:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # ------------------------------------------------------------------
    # Final intensive local search (peak‑shaving)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (required by the harness)
    return [float(x) for x in best_seq]
