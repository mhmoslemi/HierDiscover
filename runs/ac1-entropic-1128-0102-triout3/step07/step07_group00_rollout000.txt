def conv_self_fft(a, L):
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

def gradient_logsumexp(a, conv, alpha, L):
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    w /= w_sum
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[:n + len(w_rev) - 1]
    # compute gradient shape length n
    # The gradient for each a_i is 2 * sum over w of shift contributions.
    # In earlier code they used: grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    # That's the same.
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

            # compute w weights for log-sum-exp
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            w /= w_sum
            # reversed
            w_rev = w[::-1]
            # compute correlation via FFT
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            # gradient
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)

            lr = 0.5 / math.sqrt(iteration)  # schedule
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project onto simplex
            a = _project_simplex(a)

            # optionally break if iteration > some max, but time will control.

    return best, best_score

def _subgradient_optimize(a0, n, deadline, start_time, init_score):
    a = a0.copy()
    best_seq = a.copy()
    best_score = init_score
    L = _next_pow2(2*n-1)
    momentum = np.zeros_like(a)
    beta = 0.94
    iteration = 0
    step0 = 0.5

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self_fft(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad (n={n}) improve: iter={iteration}, score={cur_score:.6f}")

        # find max index(s)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(iteration)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_seq, best_score

def _annealing_optimize(a0, n, deadline, start_time, best_seq, best_score):
    a = a0.copy()
    L = _next_pow2(2*n-1)
    score = 2.0 * n * _conv_self_fft(a, L).max()

    rng = np.random.default_rng()
    # schedule steps based on time remaining
    steps = 200_000
    start_temp = 5e-2
    end_temp = 1e-6

    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        # compute temperature exponentially
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        # choose source weighted by mass (p = a)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)

        a_prop[src] -= eps
        a_prop[dst] += eps
        # compute new score
        conv = _conv_self_fft(a_prop, L)
        new_score = 2.0 * n * conv.max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score_local:
            best_score_local = score
            best_seq_local = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal (n={n}) improve: iter={it}, score={score:.6f}")

"""
Propose a near‑optimal non‑negative step‑function (probability vector)
that minimises ``evaluate_sequence``.

Algorithm
---------
* Work on several candidate lengths (powers of two).  Longer vectors give
  more degrees of freedom while the objective 2·n·max(conv) stays roughly
  constant, so a larger n can improve the bound.
* For each length we perform a few independent restarts.
* Each restart runs three optimisation phases
    1. **Adam on a smooth approximation** of the max‑convolution.
       The smooth surrogate is the log‑sum‑exp (soft‑max) with a schedule
       of increasing sharpness parameters ``alpha``.
    2. **Projected sub‑gradient descent** on the true objective
       (using the gradient at the currently maximal convolution index)
       with a simple momentum term.
    3. **Very light simulated‑annealing polish** that transfers a small
       amount of mass between two coordinates; the temperature is
       exponentially decayed.
* All iterates are kept on the probability simplex by Euclidean projection.
* The best sequence found over all lengths and restarts is returned.
* The whole routine respects the supplied time budget (≈1000 s).

The code is deliberately written with only top‑level functions (no closures,
no lambdas) and uses only ``numpy`` and the standard library.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a via FFT.
    Returns the linear convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Phase‑1: Adam on a smooth (log‑sum‑exp) surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alphas=(0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0),
                   lr0: float = 0.5,
                   iters_per_alpha: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best iterate and its objective value (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    # FFT length large enough for both the convolution (2n‑1) and the
    # correlation needed for the gradient (n+2n‑2 = 3n‑2).
    L = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    best_score = float("inf")

    for alpha in alphas:
        # limit the inner loop by both iteration count and the deadline
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # ----- forward convolution (linear) -----
            Af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam (n={n}) improve: "
                      f"alpha={alpha:.1f}, iter={it:,}, score={cur_score:.6f}")

            # ----- smooth‑max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)     # stability trick
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smooth surrogate -----
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          init_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Only the gradient at the current maximal index is used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.5
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad (n={n}) improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # index of the current maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : Tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing_optimize(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_seq: np.ndarray,
                        cur_best_score: float,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-6,
                        steps: int = 180_000) -> tuple[np.ndarray, float]:
    """
    Light mass‑transfer simulated annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    score = 2.0 * n * _conv_self_fft(a, L).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponentially decaying temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)               # source weighted by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * rng.random() * 0.30    # move up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop, L).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal (n={n}) improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

            # Phase 1 – Adam on smoothed max (≈15 % of the per‑length budget)
            # --------------------------------------------------------------
            phase1_deadline = time.time() + max(5.0, 0.15 * (phase_deadline - time.time()))
            a_adam, score_adam = _adam_optimize(
                a0,
                n,
                phase1_deadline,
                start_time,
                alphas=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
                lr0=0.5,
                iters_per_alpha=6000,
            )
            cur_seq, cur_score = a_adam, score_adam

            # --------------------------------------------------------------
            # Phase 2 – exact‑max projected sub‑gradient descent
            # --------------------------------------------------------------
            if time.time() < phase_deadline:
                phase2_deadline = phase_deadline - 5.0     # keep a few seconds for phase 3
                a_sub, score_sub = _subgradient_optimize(
                    cur_seq,
                    n,
                    phase2_deadline,
                    start_time,
                    cur_score,
                )
                cur_seq, cur_score = a_sub, score_sub

            # --------------------------------------------------------------
            # Phase 3 – final annealing polish (use any leftover time)
            # --------------------------------------------------------------
            if time.time() < phase_deadline:
                a_final, score_final = _annealing_optimize(
                    cur_seq,
                    n,
                    phase_deadline,
                    start_time,
                    cur_seq,
                    cur_score,
                    start_temp=5e-2,
                    end_temp=1e-6,
                    steps=200_000,
                )
                cur_seq, cur_score = a_final, score_final

            # --------------------------------------------------------------
            # Keep the best sequence among all restarts / lengths
            # --------------------------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] New global best: n={n}, "
                      f"score={cur_score:.6f}")

        # If we are close to the overall deadline, break early
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback)
        best_score = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2

    # Normalize one final time to satisfy the evaluate_sequence checks
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # Convert to a plain Python list (the harness expects this)
    return [float(x) for x in best_seq]
