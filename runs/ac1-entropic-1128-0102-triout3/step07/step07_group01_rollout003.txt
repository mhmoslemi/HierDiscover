"""
Propose a step‑function with a lower upper bound.

The algorithm is a multi‑phase optimizer for the objective  

    F(a) = 2·n·max(conv(a,a))          (n = len(a), a_i ≥ 0, Σ a_i = 1)

It works on a fixed length `n` (here 7200) and consists of:

1. **Exponential‑gradient warm‑up** on a smoothed max (log‑sum‑exp).
2. **Projected sub‑gradient descent** with momentum on the true max.
3. **Linear‑program improvement** – maximise the ℓ₁‑norm of a vector `b`
   while keeping the convolution `a * b` ≤ current maximum.  The LP
   is built only for the most‑active convolution indices, making it fast.
4. **Targeted local refinement** that moves a tiny amount of probability
   mass away from coordinates that contribute most to the current maximal
   convolution entry.  Only moves that strictly decrease the maximum are
   accepted.

The phases are executed sequentially respecting a 1000 s budget.
The best sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : linear‑program improvement (active‑set only)
# ----------------------------------------------------------------------
def _lp_improve(active_a: np.ndarray,
                M: float,
                active_k: int = 30) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   b ≥ 0 and  (a * b)[k] ≤ M
    only for the `active_k` largest convolution entries.
    Returns a normalized vector `g` (sum = 1) or None if LP fails.
    """
    n = active_a.shape[0]
    conv = _conv_self(active_a)

    # pick the indices with the largest convolution values
    idx = np.argsort(conv)[-active_k:]

    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        rows.extend([row_idx] * len(j))
        cols.extend(j.tolist())
        data.extend(active_a[i].tolist())

    A = sparse.csr_matrix((data, (rows, cols)), shape=(active_k, n))
    b_vec = np.full(active_k, M)

    # linprog minimises; minimise -Σb  ⇔ maximise Σb
    res = linprog(c=-np.ones(n),
                  A_ub=A,
                  b_ub=b_vec,
                  bounds=(0, None),
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    return b / sum_b  # normalised to unit sum


# ----------------------------------------------------------------------
# Phase‑4 : targeted local refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         eps_frac: float = 0.025) -> np.ndarray:
    """
    Move a small fraction of mass away from the coordinates that most
    contribute to the current maximal convolution entry.  Only strictly
    improving moves are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        k_max = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)
        if a[src] < 1e-14:
            continue

        # destination – a *different* index, preferring low‑mass locations
        low_mask = a < np.percentile(a, 20)
        low_idx = np.where(low_mask)[0]
        if low_idx.size > 0:
            dst = rng.choice(low_idx)
        else:
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:      # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional early exit
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a

    ``evaluate_sequence`` under the given time budget.
    The method combines exponentiated‑gradient warm‑up, sub‑gradient descent,
    a small linear‑program improvement loop and a final targeted refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – a larger n gives a tighter asymptotic bound.
    # 7200 is a compromise between quality and speed.
    # ------------------------------------------------------------------
    final_n = 7200

    # ------------------------------------------------------------------
    # Initialise – use the historic construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = _upsample(starter, final_n)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick exponential‑gradient warm‑up
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 15.0)
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.5,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score, best_seq = score_warm, a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (the heavy lifting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep ~60 % of the remaining budget for this phase
        phase2_deadline = time.time() + max(30.0, 0.6 * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.3,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based improvement (active‑set only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Reserve a modest chunk for this phase (≈15 % of total time)
        while time.time() < deadline - 30.0:
            conv_cur = _conv_self(best_seq)
            M_cur = conv_cur.max()
            g = _lp_improve(best_seq, M_cur, active_k=30)
            if g is None:
                break
            # small mixing keeps the iteration stable
            t_mix = 0.15
            new_seq = (1 - t_mix) * best_seq + t_mix * g
            new_seq = np.maximum(new_seq, 0.0)
            new_seq /= new_seq.sum()
            new_score = 2.0 * final_n * _conv_self(new_seq).max()
            if new_score < best_score:
                best_score, best_seq = new_score, new_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] LP improve: score={new_score:.6f}")
            else:
                # if no improvement, reduce mixing to be less aggressive
                t_mix *= 0.7
                if t_mix < 1e-3:
                    break

    # ------------------------------------------------------------------
    # Phase 4 – targeted local refinement (tiny mass re‑allocation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10**7,
            eps_frac=0.03,
        )
        final_score = 2.0 * final_n * _conv_self(a_ref).max()
        if final_score < best_score:
            best_score, best_seq = final_score, a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
