def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]

    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Objective and sub‑gradient
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (non‑negative entries).
    For every index k attaining the maximum we add the contribution
    of a to the derivative of conv_k.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam on a soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Minimise   Oτ(a)=2·n·τ·log∑exp(conv(a)/τ)
    using Adam with projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # exact (non‑smoothed) score for bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ----- soft‑max weights -----
        conv = _conv_self(a)
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # length 2·n‑1, sums to 1

        # gradient of Oτ = 4·n·∑ a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)     # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]   # the required sums
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.95,
                             step0: float = 0.18,
                             reset_interval: int = 50000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   F(a)=2·n·max(conv(a)).
    Uses exponential moving‑average momentum and occasional momentum reset.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    last_reset = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

        # occasional momentum reset to avoid stagnation
        if it - last_reset >= reset_interval and np.random.rand() < 0.2:
            momentum.fill(0.0)
            last_reset = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The method combines several Adam phases,
    a long sub‑gradient phase and a final simulated‑annealing refinement.
    The returned list is normalised so that its sum equals 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optional starter (previous run) – will be normalised automatically
    # --------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # --------------------------------------------------------------
    # Phase 0 – many short Adam runs on a modest size (n≈2048)
    # --------------------------------------------------------------
    n0 = 2048
    phase0_deadline = start_time + (deadline - start_time) * 0.15
    restarts = 12
    for r in range(restarts):
        if time.time() >= deadline:
            break

        # pick a diverse start
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        elif r % 3 == 1:
            # Dirichlet with concentration 0.3  (more sparse)
            a0 = np.random.dirichlet(alpha=np.full(n0, 0.3))
        elif r % 3 == 2:
            # linear ramp (increasing) then normalise
            ramp = np.linspace(0.0, 1.0, n0)
            a0 = ramp / ramp.sum()
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # initialise τ from the current max‑convolution
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 10.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0,
            n=n0,
            deadline=phase0_deadline,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-12,
            tau_decay=0.9995,
            step_size=0.25,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Phase 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback – uniform vector (should never be needed)
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # Phase 1 – up‑sample to 4096 and continue Adam
    # --------------------------------------------------------------
    if time.time() >= deadline:
        best_seq = best_seq / best_seq.sum()
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)
    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 10.0, 1e-10)

    phase1_deadline = time.time() + (deadline - time.time()) * 0.25
    a1_opt, score1 = _run_adam_softmax(
        a1,
        n=n1,
        deadline=phase1_deadline,
        start_time=start_time,
        tau0=tau0_1,
        tau_end=1e-13,
        tau_decay=0.9994,
        step_size=0.12,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] Phase 1 improve: score={score1:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – up‑sample to 8192 and continue Adam
    # --------------------------------------------------------------
    if time.time() >= deadline:
        best_seq = best_seq / best_seq.sum()
        return [float(x) for x in best_seq]

    n2 = 8192
    a2 = _upsample(best_seq, n2)
    conv2 = _conv_self(a2)
    tau0_2 = max(conv2.max() * 10.0, 1e-10)

    phase2_deadline = time.time() + (deadline - time.time()) * 0.30
    a2_opt, score2 = _run_adam_softmax(
        a2,
        n=n2,
        deadline=phase2_deadline,
        start_time=start_time,
        tau0=tau0_2,
        tau_end=1e-14,
        tau_decay=0.9993,
        step_size=0.09,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] Phase 2 improve: score={score2:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – up‑sample to 16384 (if time permits) and Adam again
    # --------------------------------------------------------------
    if time.time() < deadline:
        n3 = 16384
        a3 = _upsample(best_seq, n3)
        conv3 = _conv_self(a3)
        tau0_3 = max(conv3.max() * 10.0, 1e-10)

        phase3_deadline = time.time() + (deadline - time.time()) * 0.15
        a3_opt, score3 = _run_adam_softmax(
            a3,
            n=n3,
            deadline=phase3_deadline,
            start_time=start_time,
            tau0=tau0_3,
            tau_end=1e-15,
            tau_decay=0.9992,
            step_size=0.07,
        )
        if score3 < best_score:
            best_score = score3
            best_seq = a3_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Phase 3 improve: score={score3:.6f}")
        n_final = n3
    else:
        n_final = best_seq.shape[0]

    # --------------------------------------------------------------
    # Phase 4 – long projected sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase4_deadline = deadline - 5.0          # leave a few seconds for SA
        a4_opt, score4 = _run_subgradient_descent(
            best_seq,
            n=n_final,
            deadline=phase4_deadline,
            start_time=start_time,
            beta=0.96,
            step0=0.22,
            reset_interval=40000,
        )
        if score4 < best_score:
            best_score = score4
            best_seq = a4_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Phase 4 improve: score={score4:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            n=n_final,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=400_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final anneal improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
