"""
Heuristic optimisation for the “max self‑convolution” problem.

The pipeline (all steps respect the global 1000 s budget)

1.  Several short Adam runs on a modest size (n≈2048) on a soft‑max
    surrogate of the max‑convolution.  This gives a good initial shape.
2.  Upsample to n=8192 and continue a longer Adam run (still on the
    surrogate).  The sequence stays on the probability simplex.
3.  Projected sub‑gradient descent on the true objective
    F(a)=2·n·max(conv(a)) (with simple diminishing step size).
4.  Targeted simulated‑annealing:
    * pick the index k where the current convolution attains its maximum,
    * move a random fraction of mass from an index that contributes heavily
      to this maximum to a random destination,
    * accept moves with a standard Metropolis rule.
   Because source indices are chosen with probability proportional to their
   contribution a_i·a_{k−i} the annealer concentrates on the “bottleneck”
   of the current solution and is able to escape shallow local minima.
5.  A final short sub‑gradient polishing pass.

All vectors are projected onto the simplex after every update,
ensuring non‑negativity and unit sum (the evaluator then returns 2·n·max(conv)).
The algorithm is deliberately lightweight and runs without any external
dependencies beyond NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise the smoothed objective Oτ using Adam and simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # numerical stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # length 2·n‑1, sums to 1

        # gradient of Oτ:  grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # the needed slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling schedule for τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(a.shape[0])

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < a.shape[0])
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing (mass‑transfer) – main new ingredient
# ----------------------------------------------------------------------


def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 2e-2,
                            end_temp: float = 1e-7,
                            max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Moves mass from indices that contribute most to the current maximum
    convolution value to random destinations.
    Accepts worsening moves with a temperature schedule.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # index where the convolution attains its maximum
        k_max = int(np.argmax(conv))

        # contributions a_i * a_{k_max-i} for all feasible i
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        jdx = k_max - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            src = rng.choice(idx, p=probs)
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer a random fraction of src's mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] targeted anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optional starter from a previous run
    # --------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # --------------------------------------------------------------
    # Stage 0 – several short Adam runs on a modest size (n≈2048)
    # --------------------------------------------------------------
    n0 = 2048
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 10.0, 1e-8)

        slice_time = max(30.0, 0.12 * (deadline - time.time()))
        slice_deadline = time.time() + slice_time

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-12,
            tau_decay=0.9995,
            step_size=0.25,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # --------------------------------------------------------------
    # Stage 1 – up‑sample to a larger size (≈8192) and continue Adam
    # --------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 10.0, 1e-10)

    # leave a modest reserve for the later annealing & final polish
    time_left = deadline - time.time()
    reserve = 40.0
    adam_time = max(5.0, time_left - reserve)
    adam_deadline = time.time() + adam_time

    a1_opt, score1 = _run_adam_softmax(
        a1, n1, adam_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-13,
        tau_decay=0.9994,
        step_size=0.12,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = best_seq.shape[0]
    time_left = deadline - time.time()
    # keep 30 s for the targeted annealer and 10 s for a final polish
    anneal_reserve = 30.0
    final_reserve = 10.0
    subgrad_time = max(0.0, time_left - anneal_reserve - final_reserve)
    subgrad_deadline = time.time() + subgrad_time

    a2_opt, score2 = _run_subgradient_descent(
        best_seq, n2, subgrad_deadline, start_time,
        beta=0.94,
        step0=0.13,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – targeted simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_targeted_annealing(
            best_seq,
            n2,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            max_steps=800_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] targeted anneal improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final short sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline - 1.0
        a_final, score_final = _run_subgradient_descent(
            best_seq,
            n2,
            final_deadline,
            start_time,
            beta=0.94,
            step0=0.13,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improve: "
                  f"score={score_final:.6f}")

    # --------------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
