"""
Propose a low‑objective step‑function.

The algorithm builds on the three‑phase method that achieved a score
≈1.5125, but adds several refinements:

*   A second, “gradient‑aware” local refinement that moves mass away from
    indices that contribute most to the (near‑maximal) convolution entries.
    The move is symmetrised and the step size slowly shrinks.

*   Optional Gaussian smoothing moves – a tiny amount of the current
    distribution is replaced by a locally smoothed version; the move is
    kept only if it lowers the maximal autocorrelation.

*   A short simulated‑annealing phase that occasionally accepts
    non‑improving moves, helping the search escape shallow plateaus.

*   Multi‑start: after a long optimisation from the provided starter
    (``height_sequence_1`` if present) the remaining budget is spent on
    several short runs started from random initialisations; the best
    sequence over all runs is returned.

All operations respect the simplex constraint (non‑negative, sum = 1)
by Euclidean projection.  Convolutions are performed with an FFT‑based
method for speed.  The code respects the 1000 s wall‑clock limit and
periodically prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _smooth_gaussian(a: np.ndarray, sigma: float = 1.5) -> np.ndarray:
    """Return a Gaussian‑smoothed copy of a (mass is re‑normalised)."""
    # Kernel size chosen as 6σ rounded up to an odd integer
    radius = int(math.ceil(3 * sigma))
    x = np.arange(-radius, radius + 1)
    kern = np.exp(-0.5 * (x / sigma) ** 2)
    kern /= kern.sum()
    smooth = np.convolve(a, kern, mode='same')
    smooth = np.maximum(smooth, 0.0)
    total = smooth.sum()
    if total > 0:
        smooth /= total
    else:
        smooth = np.full_like(smooth, 1.0 / a.shape[0])
    return smooth

# Phase‑1 : exponentiated‑gradient on a smoothed max (unchanged)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase‑3 : targeted local refinement (enhanced)
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         init_eps_frac: float = 0.025,
                         smooth_prob: float = 0.001,
                         rel_tol: float = 1e-8):
    """
    Gradient‑aware refinements that move a tiny fraction of probability mass
    away from indices that dominate the current maximal convolution entries.
    The distribution is symmetrised after each accepted move and an occasional
    Gaussian smoothing move is tried.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()

    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    it = 0
    # linearly decreasing epsilon fraction helps fine‑tuning later
    while time.time() < deadline and it < max_iters:
        it += 1
        # adapt eps fraction slowly
        eps_frac = init_eps_frac * (1.0 - it / max_iters) ** 0.5

        # probability of picking a source index – use sub‑gradient of (near) max
        grad = _subgradient_full(a, conv, rel_tol=rel_tol)
        total_grad = grad.sum()
        if total_grad <= 0:
            prob_src = a / a.sum()
        else:
            prob_src = grad / total_grad

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-16:
            continue

        # choose a destination uniformly (different from src)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # Symmetrise and re‑normalise
        a_prop = (a_prop + a_prop[::-1]) / 2.0
        s = a_prop.sum()
        if s > 0:
            a_prop /= s
        else:
            a_prop = np.full_like(a_prop, 1.0 / n)

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # Occasionally try a Gaussian smoothing move
        if rng.random() < smooth_prob:
            a_smooth = _smooth_gaussian(a, sigma=1.3)
            # enforce symmetry (already symmetric) and normalise
            a_smooth = (a_smooth + a_smooth[::-1]) / 2.0
            conv_smooth = _conv_self(a_smooth)
            max_smooth = conv_smooth.max()
            if max_smooth < max_val - 1e-12:
                a = a_smooth
                conv = conv_smooth
                max_val = max_smooth
                if max_smooth < best_max - 1e-12:
                    best_max = max_smooth
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth improve: "
                          f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 : short simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   init_temp: float = 4e-5,
                   final_temp: float = 5e-9,
                   max_iters: int = 10**7,
                   eps_frac: float = 0.018):
    """
    Very low‑temperature annealing that sometimes accepts non‑improving
    mass transfers.  Designed only as a finishing touch – it runs for a
    short period (a few tens of seconds) and can shave off a tiny bit of
    the maximal convolution value.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # linear temperature schedule
        t_frac = (deadline - time.time()) / (deadline - start_time)
        temp = final_temp + (init_temp - final_temp) * t_frac

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = (a_prop + a_prop[::-1]) / 2.0
        a_prop /= a_prop.sum()

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # Metropolis criterion (very small temperature)
            prob = math.exp(-(new_max - cur_max) / temp) if temp > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 100000 == 0 and time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Helper: run the full optimisation pipeline from a given start vector
# ----------------------------------------------------------------------
def _optimise_from(a0: np.ndarray,
                   n: int,
                   budget_s: float,
                   start_time: float,
                   stage1_frac: float = 0.07,
                   stage2_frac: float = 0.78) -> tuple[np.ndarray, float]:
    """
    Run exponential‑gradient warm‑up, sub‑gradient descent and the
    three refinement phases within the allocated sub‑budget.
    Returns the best sequence found and its raw score 2·n·max(conv).
    """
    deadline = start_time + budget_s

    # ----- Stage 1 : exponential‑gradient (quick warm‑up) -----
    stage1_time = min(12.0, stage1_frac * (deadline - time.time()))
    a_warm, score_warm = _run_exponential_gradient(
        a0,
        n,
        time.time() + stage1_time,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=6000,
    )
    best_a = a_warm
    best_score = score_warm

    # ----- Stage 2 : sub‑gradient descent (the heavy part) -----
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_time = max(30.0, stage2_frac * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_a,
            n,
            time.time() + stage2_time,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_a = a_sub

    # ----- Stage 3 : targeted local refinement -----
    if time.time() < deadline:
        a_ref = _run_targeted_refine(
            best_a,
            n,
            deadline,
            start_time,
            max_iters=10**7,
            init_eps_frac=0.025,
            smooth_prob=0.001,
            rel_tol=1e-8,
        )
        conv_ref = _conv_self(a_ref)
        score_ref = 2.0 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_ref

    # ----- Stage 4 : brief annealing fine‑tuning (if time permits) -----
    if time.time() < deadline - 1.0:
        a_anneal = _run_annealing(
            best_a,
            n,
            deadline,
            start_time,
            init_temp=4e-5,
            final_temp=5e-9,
            max_iters=10**7,
            eps_frac=0.018,
        )
        conv_ann = _conv_self(a_anneal)
        score_ann = 2.0 * n * conv_ann.max()
        if score_ann < best_score:
            best_score = score_ann
            best_a = a_anneal

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    global_deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine the base length – 4800 works well, but we keep the
    # possibility to increase it if enough time remains.
    # ------------------------------------------------------------------
    base_n = 4800

    # ------------------------------------------------------------------
    # Load starter if available, otherwise initialise randomly.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a_initial = _upsample(starter, base_n)
    else:
        a_initial = np.random.rand(base_n)
        a_initial /= a_initial.sum()

    # ------------------------------------------------------------------
    # Run a long optimisation from the starter (most of the budget).
    # ------------------------------------------------------------------
    remaining_budget = global_deadline - time.time()
    if remaining_budget > 30.0:
        # allocate ~80 % of the remaining time to this main run
        main_budget = 0.80 * remaining_budget
        best_seq, best_score = _optimise_from(
            a_initial,
            base_n,
            main_budget,
            time.time(),
            stage1_frac=0.07,
            stage2_frac=0.78,
        )
    else:
        # not enough time – just keep the starter
        best_seq = a_initial
        best_score = 2.0 * base_n * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Use the rest of the budget for several short random restarts.
    # ------------------------------------------------------------------
    while time.time() < global_deadline - 5.0:   # keep a few seconds for finalisation
        time_left = global_deadline - time.time()
        # each restart gets a modest 8–12 % slice of the remaining time
        restart_budget = max(5.0, 0.10 * time_left)

        a_rand = np.random.rand(base_n)
        a_rand /= a_rand.sum()
        seq, score = _optimise_from(a_rand, base_n, restart_budget, time.time(),
                                    stage1_frac=0.10, stage2_frac=0.70)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] NEW GLOBAL BEST: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(base_n, 1.0 / base_n)

    # convert to list of floats as required by the grader
    return [float(x) for x in best_seq]
