def _lp_refinement(a, n, deadline, start_time, max_iter=5000, tol_rel=0.001, max_active=5):
   while time.time()<deadline and iter<max_iter:
       conv = _conv_self(a)
       max_val = conv.max()
       # Identify active indices close to max
       # Use relative tolerance: conv >= max_val * (1 - tol_rel)
       active_idx = np.where(conv >= max_val * (1 - tol_rel))[0]
       if active_idx.size > max_active:
           # keep those with largest conv values
           top = np.argsort(conv[active_idx])[-max_active:]
           active_idx = active_idx[top]
       # Compute gradient vectors for each active index
       # For each shift s, gradient g_i = 2*a_{s-i} if index in range else 0
       # We'll construct matrix G of shape (m, n)
       m = active_idx.size
       G = np.zeros((m, n))
       for row, s in enumerate(active_idx):
           i_arr = np.arange(n)
           j_arr = s - i_arr
           mask = (j_arr >= 0) & (j_arr < n)
           G[row, i_arr[mask]] = 2.0 * a[j_arr[mask]]
           # Note: all entries for i where j=s-i not in range remain 0
       # Solve LP: maximize d subject to G @ delta + d <= 0, sum(delta)=0, -a <= delta <= 1? (no upper bound)
       # Variables: delta_0,...,delta_{n-1}, d
       c = np.zeros(n+1)
       c[-1] = -1.0  # maximize d => minimize -d
       # Inequality constraints G @ delta + d <= 0 => A_ub = [G, 1s], b_ub = 0
       A_ub = np.hstack((G, np.ones((m, 1))))
       b_ub = np.zeros(m)
       # Equality sum(delta) = 0
       A_eq = np.hstack((np.ones((1, n)), np.zeros((1, 1))))
       b_eq = np.array([0.0])
       # Bounds:
       bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]  # d>=0
       # Solve
       res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
       if not res.success:
           # Could not find direction; break
           break
       delta = np.array(res.x[:n])
       d_val = res.x[-1]  # reduction
       if d_val <= 1e-12:
           break
       # Perform line search scaling to ensure actual max reduction
       step_factor = 1.0
       while step_factor > 1e-4:
           a_new = a + step_factor * delta
           # ensure nonneg
           if np.any(a_new < -1e-12):
               step_factor *= 0.5
               continue
           # enforce sum=1 (should hold)
           # compute new max
           conv_new = _conv_self(a_new)
           max_new = conv_new.max()
           if max_new <= max_val - step_factor * d_val * 0.9:  # need actual reduction, tolerance 10%
               a = a_new
               # maybe project onto simplex for numerical stability
               a = a / a.sum()
               print(f"[{time.time() - start_time:6.2f}s] LP refine: iter={iter}, step={step_factor:.3f}, d={d_val:.6e}, max={max_new:.6e}")
               break
           else:
               step_factor *= 0.5
       else:
           # cannot find acceptable step, break
           break

       iter+=1

   # Return improved a and score (2*n*max)
   conv_final = _conv_self(a)
   score = 2.0 * n * conv_final.max()
   return a, score

def _lp_refine(a, n, deadline, start_time):
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # early break if no improvement over best
        if max_val * 2 * n >= best_score - 1e-12:
            # we might still try to improve but if not better maybe break if stagnation
            pass
        # get active indices
        # Use tolerance = max_val * 0.001 maybe
        # also ensure at least one index
        tol = max_val * 0.001
        active = np.where(conv >= max_val - tol)[0]
        # limit number
        if active.size > 10:
            # sort by conv descending and pick top 10
            top_idxs = np.argsort(conv[active])[-10:]
            active = active[top_idxs]
        # if no active, break
        if active.size == 0:
            break
        # Build G
        G = np.zeros((active.size, n))
        for row, s in enumerate(active):
            # contributions: grad_i = 2*a_{s-i} for i where j = s-i in [0,n)
            # Equivalent to: for i in range(max(0, s-n+1), min(s+1, n)):
            # j = s-i
            i_start = max(0, s - (n - 1))
            i_end = min(s, n - 1)
            # we can fill range vectorized:
            i_arr = np.arange(i_start, i_end + 1)
            j_arr = s - i_arr
            G[row, i_arr] = 2.0 * a[j_arr]
        # constraints for delta and reduction d
        m = active.size
        c = np.zeros(n+1)
        c[-1] = -1.0
        A_ub = np.hstack((G, np.ones((m, 1))))
        b_ub = np.zeros(m)
        A_eq = np.hstack((np.ones((1, n)), np.zeros((1, 1))))
        b_eq = np.array([0.0])
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]
        # time limit per LP solve: allocate some small time
        lp_time_limit = max(0.0, deadline - time.time() - 0.01)  # leave little margin
        # we may skip if time left < small
        if lp_time_limit < 0.01:
            break
        # call linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': lp_time_limit})
        if not res.success:
            # no direction found or time out
            break
        delta = res.x[:n]
        d_val = res.x[n]  # predicted reduction in max conv
        if d_val <= 1e-12:
            break
        # line search scaling
        step_factor = 1.0
        improved = False
        while step_factor > 1e-4:
            a_candidate = a + step_factor * delta
            # enforce nonnegativity: delta lower bound ensures nonnegative when step_factor=1, but with scaling step may still produce negative? Actually delta lower bound ensures a+delta >= 0; scaling less will also be >0.
            # compute conv and max
            conv_candidate = _conv_self(a_candidate)
            max_candidate = conv_candidate.max()
            if max_candidate <= max_val - step_factor * d_val * 0.9:
                # accept
                a = a_candidate
                # maintain normalization
                a_sum = a.sum()
                if a_sum != 0:
                    a = a / a_sum
                cur_score = 2.0 * n * max_candidate
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP refine iter={it}, step={step_factor:.3f}, d={d_val:.6e}, score={cur_score:.6f}")
                improved = True
                break
            else:
                step_factor *= 0.5
        if not improved:
            # cannot improve, break
            break
        # continue loop
    return best_a, best_score

    remaining_time = deadline - start_time
    # allocate time half of remaining for subgradient (maybe 0.35 total)
    stage2_time = total_time * 0.35
    stage2_deadline = start_time + stage1_time + stage2_time

        a_sub, score_sub = _run_subgradient_gd(best_a, n, stage2_deadline, start_time,

            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad best score {score_sub:.6f}")

    # Stage 3: LP refinement for remaining time
    if time.time() < deadline - 0.1:
        a_lp, score_lp = _lp_refine(best_a, n, deadline, start_time)

    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()

    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    # convert to list
    return [float(x) for x in best_a.tolist()]

"""
Hybrid optimizer for the self‑convolution minimisation problem.

The routine mixes three complementary ideas:
  1. Warm‑start from a previously discovered solution (``height_sequence_1``) or a
     random simplex vector.
  2. Multi‑target projected sub‑gradient descent – at each iteration the
     gradient of several of the largest convolution entries (the active set)
     is summed, giving a descent direction that reduces *all* near‑maximal
     peaks simultaneously.
  3. Targeted stochastic “peak‑shaving’’ local search that moves a small
     amount of mass from indices that contribute most to the current maximum
     to low‑contribution indices.  The step size decays with a simulated‑annealing
     temperature schedule, allowing very fine adjustments in the final phase.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1).  The optimiser runs for the whole 1000‑second budget, printing
intermediate scores.  The best sequence found is returned as a plain Python
list, ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# Attempt to import the LP solver – it is optional but may give a tiny extra boost.
try:
    from scipy.optimize import linprog  # type: ignore
    _HAS_LP = True
except Exception:  # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution (full, non‑circular) via FFT – fast for the sizes we use.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Multi‑target projected sub‑gradient descent.
# ----------------------------------------------------------------------
def _run_multi_target_subgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 1.0,
    beta: float = 0.5,
    max_iter: int = 10 ** 7,
    active_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that reduces several of the largest
    convolution entries at once (active set of size ``active_k``).

    Returns the best point seen and its (un‑normalised) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] multi‑subgrad improve: "
                f"iter={it}, score={cur_score:.6f}"
            )

        # ---- active set (largest convolution entries) ----
        # ``np.argpartition`` is O(n) and gives the indices of the k largest values.
        active_idx = np.argpartition(conv, -active_k)[-active_k:]

        # ---- gradient = sum of gradients of the selected entries ----
        grad = np.zeros_like(a)
        i_arr = np.arange(n)
        for s in active_idx:
            j_arr = s - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            grad[i_arr[mask]] += 2.0 * a[j_arr[mask]]

        # ---- Nesterov‑type momentum and diminishing step size ----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted stochastic local search (peak shaving).
# ----------------------------------------------------------------------
def _run_targeted_local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    max_eps_frac: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk that tries to lower the current maximal convolution
    entry.  The temperature (hence the allowed uphill moves) decays linearly in
    log‑time, while the transferred fraction of donor mass also shrinks with the
    temperature, enabling very fine adjustments at the end.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: pick proportionally to contribution (focus on hot spots)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: pick from low‑contribution region
        low_thr = np.percentile(contrib, 20)  # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # temperature schedule (linear in log‑time)
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed

        # shrink the transferred amount as we cool down
        eps_frac = max_eps_frac * T / start_temp
        eps = a[i_donor] * np.random.rand() * eps_frac

        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] local improve: "
                    f"iter={it}, score={new_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional LP‑based fine‑tuning (active constraints only).
# ----------------------------------------------------------------------
def _run_lp_refinement(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    rel_tol: float = 1e-3,
    max_active: int = 8,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming step that tries to reduce the maximal convolution value.
    Only the most critical constraints (the ``max_active`` largest convolution
    entries) are linearised.  The LP seeks a direction ``delta`` that reduces all
    active entries by at least a slack ``d``.  The direction is applied with a
    back‑tracking line search to stay feasible.
    """
    if not _HAS_LP:                               # pragma: no cover
        # LP solver unavailable – fall back to no‑op.
        return a0, 2.0 * n * _conv_self(a0).max()

    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active indices – either those within rel_tol of the maximum
        # or the top ``max_active`` values (whichever yields fewer constraints).
        near = np.where(conv >= max_val * (1 - rel_tol))[0]
        if near.size > max_active:
            # keep only the truly largest entries
            near = np.argpartition(conv, -max_active)[-max_active:]

        m = near.size
        if m == 0:
            break

        # Build gradient matrix G (m × n) where G_s,i = 2·a_{s‑i} (if defined)
        G = np.zeros((m, n))
        i_arr = np.arange(n)
        for row, s in enumerate(near):
            j_arr = s - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            G[row, i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # LP variables: delta_0 … delta_{n‑1}, d (predicted reduction)
        # maximise d  ⇔ minimise -d
        c = np.zeros(n + 1)
        c[-1] = -1.0

        # Inequalities: G·delta + d <= 0   (i.e. each active entry tries to decrease)
        A_ub = np.hstack((G, np.ones((m, 1))))
        b_ub = np.zeros(m)

        # Equality: total mass unchanged
        A_eq = np.hstack((np.ones((1, n)), np.zeros((1, 1))))
        b_eq = np.array([0.0])

        # Bounds: delta_i >= -a_i   (keeps entries non‑negative)
        bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]  # d >= 0

        # Small time budget for each LP solve
        lp_time = max(0.0, deadline - time.time() - 0.02)
        if lp_time <= 0.0:
            break

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": lp_time},
        )
        if not res.success:
            break

        delta = np.asarray(res.x[:n])
        d_val = float(res.x[-1])
        if d_val <= 1e-12:
            break

        # line search – try the full step first; if it does not improve,
        # halve repeatedly.
        step = 1.0
        improved = False
        while step > 1e-4:
            a_try = a + step * delta
            # non‑negativity is already enforced by the bounds, but rounding can
            # introduce tiny negatives – clip safely.
            a_try = np.maximum(a_try, 0.0)
            conv_try = _conv_self(a_try)
            max_try = conv_try.max()
            if max_try <= max_val - step * d_val * 0.9:
                a = a_try
                cur_score = 2.0 * n * max_try
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] LP refine iter={it}, "
                        f"step={step:.3f}, d={d_val:.6e}, score={cur_score:.6f}"
                    )
                improved = True
                break
            step *= 0.5

        if not improved:          # pragma: no cover
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a multi‑target sub‑gradient descent, a final
    stochastic peak‑shaving walk and a tiny LP‑based refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose problem size – use the length of the warm‑start if possible,
    # otherwise fall back to 9600 (the size that gave the previous best).
    # ------------------------------------------------------------------
    if starter is not None:
        n = starter.shape[0]
        a0 = starter.copy()
    else:
        n = 9600
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Multi‑target projected sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + (deadline - start_time) * 0.55
    a_sub, score_sub = _run_multi_target_subgrad(
        a0,
        n,
        subgrad_deadline,
        start_time,
        step0=1.0,
        beta=0.6,
        max_iter=10 ** 7,
        active_k=7,
    )
    best_a = a_sub.copy()
    best_score = score_sub
    print(f"[{time.time() - start_time:6.2f}s] after sub‑grad: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stochastic peak‑shaving local search – consumes the remaining bulk of
    # the budget.
    # ------------------------------------------------------------------
    local_deadline = deadline - (deadline - start_time) * 0.12  # reserve a little for LP
    a_loc, score_loc = _run_targeted_local_search(
        best_a, n, local_deadline, start_time, start_temp=5e-2, end_temp=1e-6
    )
    if score_loc < best_score:
        best_score = score_loc
        best_a = a_loc.copy()
        print(f"[{time.time() - start_time:6.2f}s] after local search: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Tiny LP‑based refinement (optional – only if SciPy is available)
    # ------------------------------------------------------------------
    if time.time() < deadline and _HAS_LP:
        a_lp, score_lp = _run_lp_refinement(
            best_a, n, deadline, start_time, rel_tol=5e-4, max_active=8
        )
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] after LP refine: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (project onto simplex once more to wipe out any
    # tiny drift caused by numerical error).
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    # Return a plain Python list – the evaluation harness expects a list.
    return [float(x) for x in best_a.tolist()]
