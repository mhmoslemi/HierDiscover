"""
Improved step‑function search for the ``evaluate_sequence`` objective.

The algorithm consists of three stages:

1. **Exploratory multi‑start (small size)** – many random initial
   distributions of length ``n_small`` are refined by a few thousand
   Adam‑style updates on a smooth (soft‑max) surrogate of the max‑convolution.
   The best candidate is kept.

2. **High‑resolution refinement** – the selected candidate is up‑sampled
   to a large length (default 9600).  Two passes are performed:
   * soft‑max Adam optimisation,
   * exact sub‑gradient Adam optimisation on the active set of tight
     convolution constraints.

   A small variance‑penalty term (gradient of the squared deviation of the
   convolution from its mean) is added to the gradient; this encourages a
   flatter convolution and therefore a lower maximum.

3. **Extended simulated annealing** – a Metropolis walk with tiny
   mass‑transfer moves and occasional local smoothing finalises the solution.

All iterates are projected onto the probability simplex (non‑negative,
unit‑sum) after each update.  The routine respects the supplied wall‑clock
budget (default 1000 s) and returns a plain Python list ready for the
evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of a weighted convolution (used for soft‑max, active set and variance).
# ----------------------------------------------------------------------
def _weighted_conv_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    For a fixed weight vector w (same length as full convolution a*a),
    returns 2 * (rev(a) * w) limited to the original dimension of a.
    """
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a[::-1], n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp (soft‑max) of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # stable soft‑max
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    return _weighted_conv_gradient(a, w)


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex) for a generic gradient.
# ----------------------------------------------------------------------
def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    grad_func,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Generic Adam optimiser with projection onto the simplex.
    ``grad_func`` must accept the current a and return a gradient array.
    Returns the best iterate (lowest 2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    while time.time() < deadline:
        t += 1
        grad = grad_func(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}"
            )
    return best_a, best_score

def _run_softmax_adam(

    deadline: float,
    start_time: float,
    T_start: float = 5e-2,
    T_end: float = 5e-4,
    lr: float = 0.08,
    lambda_var: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Adam on a mixture of soft‑max (max‑conv) and variance‑penalty gradients."""
    # pre‑compute constant average of a convolution (depends only on n via unit sum)
    avg_const = 1.0 / (2 * n - 1)

    def grad_func(a):
        conv = _conv_self(a)
        # soft‑max part
        # temperature decays logarithmically with the iteration count
        it = grad_func.it
        T = T_start * (T_end / T_start) ** (it / 20000.0)
        g_max = _softmax_gradient(a, conv, T)

        # variance‑penalty part: encourage flat convolution
        diff = conv - avg_const
        g_var = _weighted_conv_gradient(a, diff)

        # linear blend
        return g_max + lambda_var * g_var

    # iterator counter stored as an attribute of the function object
    grad_func.it = 0

    def wrapped_grad(a):
        grad_func.it += 1
        return grad_func(a)

    return _run_adam(a0, n, deadline, start_time, wrapped_grad, lr=lr)


# ----------------------------------------------------------------------
# Exact sub‑gradient Adam phase (active‑set).
# ----------------------------------------------------------------------
def _run_subgradient_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    rel_eps: float = 0.01,
    lr: float = 0.15,
) -> tuple[np.ndarray, float]:
    """Adam on the exact sub‑gradient defined by the active set of constraints."""

    def grad_func(a):
        conv = _conv_self(a)
        max_val = conv.max()
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        return _weighted_conv_gradient(a, w)

    return _run_adam(a0, n, deadline, start_time, grad_func, lr=lr)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer + occasional smoothing).
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 0.3,
    end_temp: float = 1e-7,
    steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass and applies local smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:  # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25  # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:  # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode="same")
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, performs an exploratory multi‑start
    on a modest size, up‑samples the best candidate, refines it with Adam‑based
    soft‑max and sub‑gradient phases, and finishes with an extended annealing.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously known construction.
    # ------------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Phase‑1: cheap multi‑start on a small problem size.
    # ------------------------------------------------------------------
    n_small = 2000
    phase1_deadline = start_time + (deadline - start_time) * 0.20
    while time.time() < phase1_deadline:
        # three different families of random starts
        choice = np.random.randint(0, 3)
        if choice == 0:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()
        elif choice == 1:
            # sine‑squared shape (smooth, non‑uniform)
            x = np.linspace(0, np.pi, n_small)
            a0 = np.sin(x) ** 2
            a0 /= a0.sum()
        else:
            # sparse spikes (exponential weights)
            a0 = np.random.exponential(scale=1.0, size=n_small)
            a0 /= a0.sum()

        # a short soft‑max Adam run
        a_opt, score_opt = _run_softmax_adam(
            a0, n_small, phase1_deadline, start_time, lr=0.07, lambda_var=0.5
        )
        # a short sub‑gradient Adam run
        a_opt, score_opt = _run_subgradient_adam(
            a_opt, n_small, phase1_deadline, start_time, rel_eps=0.015, lr=0.12
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] multi‑start improve: n={n_small}, score={score_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase‑2: up‑sample and perform a thorough refinement on the large size.
    # ------------------------------------------------------------------
    n_large = 9600
    if best_seq is None:
        # fall back to a simple random start if everything failed
        a0 = np.random.rand(n_large)
        a0 /= a0.sum()
        best_seq = a0
        best_score = 2.0 * n_large * _conv_self(a0).max()

    a_large = _upsample(best_seq, n_large)

    # allocate remaining time between gradient work and annealing
    remaining = deadline - time.time()
    grad_budget = remaining * 0.65
    anneal_budget = remaining - grad_budget
    grad_deadline = time.time() + grad_budget

    # Soft‑max Adam refinement
    a_opt, score_opt = _run_softmax_adam(
        a_large,
        n_large,
        grad_deadline,
        start_time,
        lr=0.06,
        lambda_var=0.35,
    )
    if score_opt < best_score:
        best_score = score_opt
        best_seq = a_opt.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] softmax‑large improve: n={n_large}, score={score_opt:.6f}"
        )

    # Exact sub‑gradient Adam refinement (if time permits)
    if time.time() < grad_deadline:
        a_opt, score_opt = _run_subgradient_adam(
            a_opt,
            n_large,
            grad_deadline,
            start_time,
            rel_eps=0.008,
            lr=0.10,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad‑large improve: n={n_large}, score={score_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase‑3: final annealing on the best large‑scale candidate.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            n_large,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=0.3,
            end_temp=1e-7,
            steps=6_000_000,
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Final sanity checks and return a plain Python list.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
