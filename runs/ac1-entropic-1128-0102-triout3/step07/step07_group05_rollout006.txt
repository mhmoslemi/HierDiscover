def _grad_smooth_max(a, conv, alpha, fft_len):
    # compute softmax weights for each conv element
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= np.sum(w)

    # cross-correlation a with w (reversed)
    w_rev = w[::-1]

    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]  # length = n+len(w_rev)-1 = 3n-2
    # extract grad of length n (center part): indices (len(w_rev)-1 - i) for i in 0..n-1
    offset = len(w_rev) - 1
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad

def _project_simplex(v):
    # based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
```

def _adam_phase(a_init, n, deadline, best_score, best_seq):
    a = a_init.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # softmax to get a
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / np.sum(expy)

            # conv via FFT
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = np.max(scaled)
            # compute lse (log-sum-exp)
            lse = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
            f_val = lse / alpha

            # gradient of smoothed max wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient wrt logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # Evaluate occasionally
            if i % 400 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam improve α={alpha} iter={i} score={score:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    a = expy / np.sum(expy)
    return a, best_score, best_seq

"""
Hybrid optimizer for the step‑function attack (improved).

The algorithm follows three stages (all executed within the given time budget):
1. Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
   The smoothing parameter α is increased gradually; a softmax parametrisation
   keeps the vector on the probability simplex.
2. Exact sub‑gradient projected descent on the true max‑convolution.
   Momentum (Nesterov‑style) and a decreasing step‑size are used; the iterate
   is projected onto the simplex after each step.
3. Fast mass‑transfer simulated annealing.
   Moves transfer a small ε‑fraction of mass from a source index (chosen
   proportionally to its current weight) to a destination index.
   ε shrinks with the temperature so that the final phase performs fine
   adjustments.  The convolution is updated incrementally in O(n) time per
   move, allowing millions of moves.

The best sequence observed during the whole run is returned (normalised to
sum = 1, all entries non‑negative).  The code works for any existing
`height_sequence_1` and respects the 1000 s time limit.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns a vector of shape (n,).
    """
    n = a.shape[0]

    # soft‑max over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – Adam on smoothed max‑convolution
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """Run Adam on the log‑sum‑exp smooth approximation of the max‑convolution."""
    a = a0.copy()
    # work in log‑soft‑max (logits) space
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # α schedule – gradually tighten the smooth approximation
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
                      160.0, 320.0, 640.0]
    iters_per_alpha = 12000

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max → a
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / expy.sum()

            # convolution
            conv = _conv_self(a)

            # smoothed objective value (for diagnostics only)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            _ = 2.0 * n * f_val  # not used directly

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t. logits y (chain rule through soft‑max)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    a = expy / expy.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Phase 2 – Exact projected sub‑gradient descent
# ----------------------------------------------------------------------


def _gradient_descent_phase(a0: np.ndarray, n: int, deadline: float,
                            start_time: float, best_score: float,
                            best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Polish the solution with exact sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_sc = best_score

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_sc:
            best_sc = cur_score
            best_a = a.copy()
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Gradient improve iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # Nesterov‑style momentum
        momentum = 0.94 * momentum + 0.06 * g

        # diminishing step‑size
        step = 0.25 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 3 – Fast incremental simulated annealing
# ----------------------------------------------------------------------


def _annealing_phase(a0: np.ndarray, n: int, deadline: float,
                     start_time: float, best_score: float,
                     best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing with temperature‑dependent step size.
    The convolution is kept incrementally, enabling tens of millions of moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq_local = best_seq.copy()
    best_sc = best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    # heuristic total number of moves (will be cut short by the timeout)
    total_steps = 2_500_000
    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # source chosen proportionally to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # epsilon shrinks with temperature (fine‑grained at the end)
        eps_factor = 0.30 * (T / start_temp)
        eps_factor = max(eps_factor, 0.004)  # keep a tiny step size
        eps = a[src] * rng.random() * eps_factor
        if eps <= 0.0:
            continue

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            score = new_score
            if score < best_sc:
                best_sc = score
                best_seq_local = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal improve iter={it:,}, score={score:.6f}")
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

    return best_seq_local, best_sc
