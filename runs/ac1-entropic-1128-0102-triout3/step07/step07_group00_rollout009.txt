    # compute a = softmax(z)
    a = np.exp(z - np.max(z))
    a /= a.sum()
    conv = _conv_self(a)
    # compute w = softmax(alpha * conv)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    # gradient of smoothed max wrt a
    # w_rev = w[::-1]
    w_rev = w[::-1]
    L_global = FFT_LEN as computed
    Af = np.fft.rfft(a, n=L_global)
    Wrev = np.fft.rfft(w_rev, n=L_global)
    corr = np.fft.irfft(Af * Wrev, n=L_global)[: n + len(w_rev) - 1]
    grad_a = 2.0 * corr[len(w_rev)-1 - np.arange(n)]   # shape (n,)

    dot = np.dot(grad_a, a)

    f_val = 2.0 * n * smax
    return f_val, grad_z

"""
A three‑stage optimiser for ``evaluate_sequence``.

Stage 1 – L‑BFGS on a smooth log‑sum‑exp surrogate of the maximal
convolution using a softmax re‑parameterisation (a = softmax(z)).
Gradients are obtained analytically via the convolution theorem, the
method gives a fast, quasi‑Newton descent and typically lowers the
bound well below the previous best.

Stage 2 – Projected sub‑gradient descent on the true objective.
The sub‑gradient is taken at the currently maximal convolution index;
a simple momentum scheme with a decreasing step‑size is used and the
iterate is projected onto the probability simplex after each update.

Stage 3 – Light simulated‑annealing polish that moves a small amount
of mass between random coordinates.  The temperature decays
exponentially.

The routine respects the supplied time budget (≈1000 s).  If a global
``height_sequence_1`` is present it is up‑sampled and used as a starter;
otherwise a random Dirichlet start is generated.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy for L‑BFGS
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize
    _HAS_SCIPY = True
except Exception:                         # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value: 2·n·max(conv(a,a)).  a is assumed to sum to 1."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Phase‑1: L‑BFGS on smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_obj_grad(z: np.ndarray, n: int, alpha: float,
                     fft_len: int):
    """
    Smooth surrogate of the max‑convolution with its gradient.
    The variables are the logits z (a = softmax(z)).
    Returns (value, gradient_wrt_z).
    """
    # a = softmax(z)
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / exp_z.sum()

    # convolution a*a
    conv = _conv_self(a, fft_len)                     # length 2n‑1

    # soft‑max weights for the smoothed max
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    w /= w_sum                                        # weights sum to 1

    # smooth value = (log‑sum‑exp)/α
    log_sum = np.log(w_sum) + max_scaled
    smax = log_sum / alpha
    value = 2.0 * n * smax

    # gradient wrt a:  Σ_i w_i ∇_a conv_i  =  a * (w_rev ⋆ a) * 2  (see derivation)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    # chain‑rule: a = softmax(z) → ∂a/∂z = diag(a) – a aᵀ
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return float(value), grad_z.astype(np.float64)


def _phase_l_bfgs(a0: np.ndarray, n: int,
                  deadline: float, start_time: float,
                  fft_len: int):
    """Run a short continuation of L‑BFGS on increasing α."""
    best_a = a0.copy()
    best_val = _objective(best_a, n, fft_len)

    # initialise logits safely (add tiny constant to avoid log(0))
    eps = 1e-14
    z = np.log(best_a + eps)

    # schedule of smoothness parameters
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0]

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # minimise the smooth surrogate
        def fun(z_vec):
            val, grad = _smooth_obj_grad(z_vec, n, alpha, fft_len)
            return val, grad
        # a modest number of L‑BFGS iterations is enough
        opts = dict(maxiter=30, disp=False)
        res = minimize(fun, z, method='L-BFGS-B', jac=True, options=opts)
        z = res.x
        # recover the probability vector
        a = np.exp(z - np.max(z))
        a /= a.sum()
        cur_val = _objective(a, n, fft_len)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS α={alpha:.1f} improve: "
                  f"score={cur_val:.6f}")
    return best_a, best_val

# ----------------------------------------------------------------------
# Phase‑2: Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _phase_subgradient(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       fft_len: int, best_score_initial: float):
    """Simple momentum sub‑gradient descent with a √t step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94                # momentum decay
    step0 = 0.30               # initial step magnitude
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient at the maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a_new = a - step * momentum
        a_new = _project_simplex(a_new)

        a = a_new
    return best_a, best_score

# ----------------------------------------------------------------------
# Phase‑3: Tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _phase_annealing(a0: np.ndarray, n: int,
                     deadline: float, start_time: float,
                     cur_best_score: float, cur_best_seq: np.ndarray,
                     fft_len: int,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-6,
                     steps: int = 200_000):
    """Mass‑transfer annealing with an exponential temperature schedule."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a, fft_len).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # draw a source weighted by its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of src's mass to dst
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, fft_len).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    deadline = start_time + budget_s - 0.3          # small safety margin

    # --------------------------------------------------------------
    # Initialise – use existing global if available, else random.
    # --------------------------------------------------------------
    final_n = 8192                                 # power of two → FFT friendly

        if starter.ndim == 1 and starter.size > 0:

            src_x = np.linspace(0.0, 1.0, starter.size)
            dst_x = np.linspace(0.0, 1.0, final_n)

            starter /= starter.sum()
    except Exception:                              # pragma: no cover

    if starter is not None:

    else:

        a0 /= a0.sum()

    # pre‑compute the FFT length we'll reuse throughout the run
    fft_len = _next_pow2(2 * final_n - 1)

    best_seq = a0.copy()
    best_score = _objective(best_seq, final_n, fft_len)

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth surrogate of max(conv)
    # --------------------------------------------------------------
    if _HAS_SCIPY and time.time() < deadline:
        phase1_deadline = time.time() + max(5.0, 0.12 * (deadline - time.time()))
        a_lbfgs, score_lbfgs = _phase_l_bfgs(best_seq, final_n,
                                             phase1_deadline, start_time,
                                             fft_len)
        if score_lbfgs < best_score:
            best_seq, best_score = a_lbfgs, score_lbfgs
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={score_lbfgs:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact‑max projected sub‑gradient descent (major chunk)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep ~70 % of the remaining budget for this phase
        phase2_deadline = time.time() + max(30.0, 0.70 * remaining)
        a_sub, score_sub = _phase_subgradient(best_seq, final_n,
                                              phase2_deadline, start_time,
                                              fft_len, best_score)
        if score_sub < best_score:
            best_seq, best_score = a_sub, score_sub
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final annealing polish (use any leftover time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _phase_annealing(best_seq, final_n,
                                                deadline, start_time,
                                                best_score, best_seq,
                                                fft_len,
                                                start_temp=5e-2,
                                                end_temp=1e-6,
                                                steps=250_000)
        if score_final < best_score:
            best_seq, best_score = a_final, score_final
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ----------------------------------------------------------------------
    # Normalise one last time and return as a plain Python list.
    # ----------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
