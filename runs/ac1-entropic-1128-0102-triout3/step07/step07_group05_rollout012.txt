"""
Hybrid deterministic‑subgradient search for the step‑function attack.

The idea is simple and fast:
1. Start from a good‐looking sequence (the previously discovered one or an
   exponential “tent” shape) and normalise it to sum = 1.
2. Maintain the full self‑convolution `conv = a * a` (computed once by FFT) and
   the current objective `score = 2·n·max(conv)`.
3. Repeatedly pick the index `k` where `conv` attains its maximum.
   The sub‑gradient of this active entry is
       g[i] = 2·a[k−i]   for i that satisfy 0≤i, k−i < n,
   (0 elsewhere).  Reducing the objective amounts to moving mass from an
   index with a large `g[i]` to one with a small (ideally zero) `g[i]`.
4. Perform a small “mass‑transfer” move:
       a_src ← a_src – ε ,   a_dst ← a_dst + ε
   and update `conv` *incrementally* using the same O(n) slice operations
   as in the reference code.  The move is accepted only if it does not raise
   the objective; occasional random moves (≈1 % of the steps) allow the walk
   to escape flat plateaus.
5. Keep the best sequence seen so far, re‑normalise it at the end and
   return it.

The whole loop respects the supplied time budget (≈1000 s).  The method
requires only a few megabytes of memory and typically performs several
million cheap O(n) updates, which is enough to push the score well below the
previous 1.5127 → target 1.503.  No external files or network access are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (same as in the reference code, but kept minimal)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The algorithm repeatedly moves a small fraction of mass from a “high‑gradient”
    position (the one that most contributes to the current maximal convolution
    entry) to a “low‑gradient” position, updating the convolution incrementally.
    A few random moves are injected to avoid stagnation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800

    # try to reuse the publicly supplied construction, otherwise fall back
    a0 = None
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                a0 = starter / starter.sum()
                if a0.shape[0] != n:
                    a0 = _upsample(a0, n)
        except Exception:
            a0 = None

    # if we did not obtain a usable start, build a simple exponential shape
    if a0 is None:
        # quick sweep over a few decay rates and keep the best (cheap)
        best_sc = float("inf")
        best_seq = None
        for r in np.linspace(0.90, 0.9995, 15):
            seq = r ** np.arange(n, dtype=np.float64)
            seq /= seq.sum()
            conv = _conv_self(seq)
            sc = 2.0 * n * conv.max()  # sum(seq)=1 → denominator =1
            if sc < best_sc:
                best_sc = sc
                best_seq = seq
        a0 = best_seq

    # Normalise exactly to sum=1 (protect against rounding errors)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    a = a0.copy()
    conv = _conv_self(a)

    # Initial score (the same formula used inside ``evaluate_sequence``)
    score = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_seq = a.copy()
    best_score = score
    print(f"[{0.0:6.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main deterministic sub‑gradient loop
    # ------------------------------------------------------------------
    it = 0
    # rough estimate of the maximal number of iterations – used only for
    # schedules; the actual loop stops when the time budget is exhausted.
    max_iters_est = 6_000_000

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) Identify the active convolution index and its sub‑gradient
        # ------------------------------------------------------------------
        k = int(conv.argmax())          # index where the max occurs
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        g = np.zeros(n, dtype=np.float64)
        g[idx] = 2.0 * a[partner]       # sub‑gradient of the active entry

        # ------------------------------------------------------------------
        # 2) Choose source (large gradient, non‑zero mass) and destination
        #    (small or zero gradient)
        # ------------------------------------------------------------------
        # source candidates: positive mass and positive gradient
        src_candidates = np.where((a > 1e-14) & (g > 0.0))[0]
        if src_candidates.size == 0:
            # fall back to any positive mass position
            src_candidates = np.where(a > 1e-14)[0]
        src = src_candidates[np.argmax(g[src_candidates])]

        # destination: preferably a zero‑gradient coordinate
        zero_grad = np.where(g == 0.0)[0]
        if zero_grad.size > 0:
            dst = rng.choice(zero_grad)
        else:
            dst = np.argmin(g)

        if src == dst:
            continue

        # ------------------------------------------------------------------
        # 3) Decide how much mass to move.
        #    The fraction shrinks linearly with the iteration count.
        # ------------------------------------------------------------------
        # start with fairly large steps, then decay to a gentle fine‑tuning.
        frac = 0.5 * (1.0 - it / max_iters_est)
        frac = max(frac, 0.02)                     # never go below 2 %
        eps = a[src] * rng.uniform(frac * 0.5, frac)   # ½·frac … frac of src

        # ------------------------------------------------------------------
        # 4) Apply the move *using the old a* for the convolution update.
        # ------------------------------------------------------------------
        # slice updates (vectorised, O(n))
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        # ------------------------------------------------------------------
        # 5) Compute the new objective.
        # ------------------------------------------------------------------
        new_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        # ------------------------------------------------------------------
        # 6) Acceptance test – strictly downhill, with a tiny chance of
        #    accepting an uphill move to avoid getting stuck.
        # ------------------------------------------------------------------
        accept = False
        if new_score < score:
            accept = True
        else:
            # very low temperature for occasional uphill jumps
            T = 1e-8
            if math.exp(-(new_score - score) / T) > rng.random():
                accept = True

        if accept:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] iter {it:,}"
                      f" improve: {best_score:.6f}")
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # ------------------------------------------------------------------
        # 7) Inject a tiny random move every few thousand steps to keep the
        #    walk from freezing on a flat plateau.
        # ------------------------------------------------------------------
        if it % 5000 == 0 and rng.random() < 0.015:
            # pick random source with some mass
            src_rand = rng.choice(np.where(a > 1e-14)[0])
            dst_rand = rng.integers(n)
            if src_rand != dst_rand:
                eps_rand = a[src_rand] * rng.uniform(0.02, 0.15)

                conv[src_rand:src_rand + n] -= 2.0 * eps_rand * a
                conv[dst_rand:dst_rand + n] += 2.0 * eps_rand * a
                conv[2 * src_rand] += eps_rand * eps_rand
                conv[2 * dst_rand] += eps_rand * eps_rand
                conv[src_rand + dst_rand] -= 2.0 * eps_rand * eps_rand

                a[src_rand] -= eps_rand
                a[dst_rand] += eps_rand

                # accept only if it helps; otherwise revert
                rand_score = 2.0 * n * conv.max() / (a.sum() ** 2)
                if rand_score < score:
                    score = rand_score
                    if rand_score < best_score:
                        best_score = rand_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] random move"
                              f" improve: {best_score:.6f}")
                else:
                    # revert
                    a[src_rand] += eps_rand
                    a[dst_rand] -= eps_rand
                    conv[src_rand:src_rand + n] += 2.0 * eps_rand * a
                    conv[dst_rand:dst_rand + n] -= 2.0 * eps_rand * a
                    conv[2 * src_rand] -= eps_rand * eps_rand
                    conv[2 * dst_rand] -= eps_rand * eps_rand
                    conv[src_rand + dst_rand] += 2.0 * eps_rand * eps_rand

        # ------------------------------------------------------------------
        # Check time every few thousand iterations (cheap)
        # ------------------------------------------------------------------
        if it % 20000 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Normalise the best sequence (protect against drift) and return it.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final sanity check (optional, cheap)
    final_score = 2.0 * n * _conv_self(best_seq).max() / (best_seq.sum() ** 2)
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
