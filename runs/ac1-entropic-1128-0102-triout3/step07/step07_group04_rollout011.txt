def propose_candidate(seed=0, budget_s=1000, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # define lengths
    n_small = 2048

    best_params = []  # list of (score, shape_vector)
    k_best = 5

        # maintain top k

"""
Propose a step‑function minimising the “max self‑convolution” objective.

The algorithm consists of three phases:

1️⃣  Parameter‑search (≈20 % of the time budget).  
    Randomly generate simple symmetric families (exponential, power, beta,
    piecewise‑constant) on a modest size (n≈2048) and keep the best shape.

2️⃣  Refinement on the full size (n≈8192).  
    The best shape is up‑sampled and polished by:
        – Adam optimisation of a smooth log‑sum‑exp surrogate,
        – Projected sub‑gradient descent on the true max‑convolution,
        – A light simulated‑annealing mass‑transfer.

3️⃣  Final normalisation and return the vector as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two vectors."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise O_τ(a)=2·n·τ·log∑exp(conv(a)/τ) with Adam."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp

        # gradient of O_τ
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)          # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA moving a random ε‑fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape generators for the cheap parameter search
# ----------------------------------------------------------------------


def _make_exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric shape a_i ∝ exp(c·|i‑centre|)."""
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    d = np.abs(idx - centre)
    w = np.exp(c * d)
    w /= w.sum()
    return w


def _make_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric shape a_i ∝ (|i‑centre|+1)^p."""
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    d = np.abs(idx - centre) + 1.0
    w = d ** p
    w /= w.sum()
    return w


def _make_beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric beta‑U shape with parameter a (0<a<1)."""
    x = np.linspace(0, 1, n)
    eps = 1e-12
    # pdf ∝ x^{a-1} (1‑x)^{a-1}; for a<1 this is U‑shaped
    w = (np.clip(x, eps, 1 - eps) ** (a - 1)) * (np.clip(1 - x, eps, 1 - eps) ** (a - 1))
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _make_step_shape(n: int, K: int) -> np.ndarray:
    """
    Piecewise‑constant shape with K blocks.
    Mass for each block is drawn from a Dirichlet distribution.
    """
    probs = np.random.dirichlet(np.ones(K))
    block_len = n // K
    extra = n % K
    shape = np.empty(n, dtype=float)
    start = 0
    for i in range(K):
        length = block_len + (1 if i < extra else 0)
        shape[start:start + length] = probs[i] / length
        start += length
    shape /= shape.sum()
    return shape

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """

    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase –0: cheap symmetric‑shape search on a modest size
    # ------------------------------------------------------------------
    n_small = 2048
    best_seq_small = None
    best_score_small = float("inf")
    # allocate ~15 % of total budget for the cheap search
    search_deadline = start_time + min(180.0, (deadline - start_time) * 0.15)

    shape_types = ["exp", "pow", "beta", "step"]
    while time.time() < search_deadline:
        typ = np.random.choice(shape_types)
        if typ == "exp":
            c = np.random.uniform(0.0, 0.004)          # controls edge‑heaviness
            shape = _make_exponential_shape(n_small, c)
        elif typ == "pow":
            p = np.random.uniform(0.0, 3.0)            # power‑law edge boost
            shape = _make_power_shape(n_small, p)
        elif typ == "beta":
            a = np.random.uniform(0.2, 0.9)            # beta‑U shape
            shape = _make_beta_shape(n_small, a)
        else:  # step
            K = np.random.choice([4, 8, 12, 16])
            shape = _make_step_shape(n_small, K)

        # enforce symmetry and normalise
        shape = (shape + shape[::-1]) / 2.0
        s = shape.sum()
        if s > 0:
            shape /= s
        else:
            shape = np.full_like(shape, 1.0 / shape.size)

        conv = _conv_self(shape)
        score = 2.0 * n_small * conv.max()
        if score < best_score_small:
            best_score_small = score
            best_seq_small = shape.copy()
            print(f"[{time.time() - start_time:6.2f}s] param‑search new best: "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Fall back to the previously‑found global best if our cheap search failed
    # ------------------------------------------------------------------
    if best_seq_small is None:
        if "height_sequence_1" in globals():
            try:
                tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
                if tmp.ndim == 1 and tmp.size > 0:

            except Exception:
                pass
        if best_seq_small is None:
            best_seq_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Phase 1: upscale to the target size and perform a thorough refinement
    # ------------------------------------------------------------------
    n_big = 8192
    seq_big = _upsample(best_seq_small, n_big)
    seq_big = (seq_big + seq_big[::-1]) / 2.0          # keep symmetry
    seq_big /= seq_big.sum()

    # remaining budget after cheap search
    remaining = deadline - time.time()
    if remaining <= 5.0:
        # not enough time – return what we have
        return [float(x) for x in seq_big]

    # --------------------------------------------------------------
    # Stage 0 – Adam on the smooth surrogate (moderate time slice)
    # --------------------------------------------------------------
    adam_time = min(200.0, remaining * 0.35)
    adam_deadline = time.time() + adam_time
    conv0 = _conv_self(seq_big)
    tau0 = max(conv0.max() * 10.0, 1e-8)

    seq_adam, score_adam = _run_adam_softmax(
        seq_big, n_big, adam_deadline, start_time,
        tau0=tau0,
        tau_end=1e-13,
        tau_decay=0.9994,
        step_size=0.12,
    )
    best_seq = seq_adam
    best_score = score_adam

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        subgrad_deadline = time.time() + remaining * 0.55
        seq_sub, score_sub = _run_subgradient_descent(
            best_seq, n_big, subgrad_deadline, start_time,
            beta=0.94,
            step0=0.13,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = seq_sub.copy()

    # --------------------------------------------------------------
    # Stage 3 – very light simulated‑annealing refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq_ref, score_ref = _run_annealing(
            best_seq,
            n_big,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        if score_ref < best_score:
            best_seq = seq_ref.copy()
            best_score = score_ref

    # ------------------------------------------------------------------
    # Final symmetry enforcement and normalisation before return
    # ------------------------------------------------------------------
    best_seq = (best_seq + best_seq[::-1]) / 2.0
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to pure Python list
    return [float(x) for x in best_seq]
