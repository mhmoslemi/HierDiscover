"""
Propose a step‐function that (almost) minimises ``evaluate_sequence``.

The optimiser works in four phases, each respecting the remaining time
budget:

1. **Soft‑max Adam phase** – parameters are the logits `y` that define the
   coefficients via a soft‑max.  A short schedule of increasing
   smoothness parameters α is used and a modest number of Adam updates
   (≈4 · 10⁴ total).  This quickly drives the solution into a region of
   low max‑convolution.

2. **Exact projected sub‑gradient descent** – the true non‑smooth
   objective `max_k (a*a)[k]` is optimised.  The sub‑gradient is taken
   from the active set of indices where the convolution is (almost)
   maximal and a Nesterov‑style momentum step is projected onto the
   probability simplex.  This phase consumes the bulk of the remaining
   time (≈40 % of the budget).

3. **Deterministic greedy peak‑reduction** – a cheap loop moves a
   fraction of the coefficient that contributes most to the current
   peak towards the smallest coefficient, keeping the move only if the
   peak strictly decreases.

4. **Simulated annealing** – random mass‑transfer moves with an
   exponential cooling schedule are performed until the deadline.
   The annealer is used for the final refinement.

The algorithm starts from the best known construction (``height_sequence_1``)
if it is present, otherwise from a uniform distribution.  All vectors are
normalised to have unit ℓ₁‑norm (the objective is scale‑invariant).  The
final result is returned as a plain list of Python floats.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _score_from_vector(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence when the vector is already normalised to L1=1."""
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.25,
                          beta: float = 0.94,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent with momentum on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen (as a probability vector) and its
    corresponding score 2·n·max_conv .
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (allow a tiny tolerance)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # ordered contribution; diagonal counted twice – harmless
            g[i] += a[j]
            g[j] += a[i]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy loop that moves a fraction of the coefficient contributing most
    to the current peak towards the smallest coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (must differ from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * a.shape[0] * _conv_self(a).max()
    return a, score


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of src to dst
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – 4800 works well and fits comfortably in memory
    n = 4800

    # ------------------------------------------------------------------
    # Initialise a single start (prefer the previously found construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.shape[0] != n:
                # linear up‑sample to required size while preserving total mass
                old_x = np.linspace(0.0, 1.0, starter.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                starter = np.interp(new_x, old_x, starter)
                starter = starter / starter.sum()
            a_start = starter.copy()
        else:
            raise Exception
    except Exception:
        a_start = np.full(n, 1.0 / n)  # uniform fallback

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation
    # ------------------------------------------------------------------
    eps = 1e-16
    y = np.log(np.maximum(a_start, eps))          # logits
    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters – growing to approximate the true max
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    iters_per_alpha = 4000          # total ≈ 32 000 Adam updates

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a_start.copy()
    best_score = _score_from_vector(best_a, n)

    adam_deadline = start_time + 0.35 * budget_s   # reserve ~35 % for Adam
    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= adam_deadline:
                break

            # soft‑max to obtain a probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # gradient wrt a (smooth)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain‑rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                cur_score = _score_from_vector(a, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={cur_score:.6f}")

        if time.time() >= adam_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent
    # ------------------------------------------------------------------
    # allocate roughly half of the remaining budget to polishing
    remaining = deadline - time.time()
    if remaining > 5.0:   # keep a few seconds for the later phases
        grad_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_a,
            grad_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_a = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] Gradient‑descent improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        peak_deadline = time.time() + 0.15 * remaining
        a_peak, score_peak = _run_peak_reduction(
            best_a,
            peak_deadline,
            start_time,
            max_no_improve=1500,
        )
        if score_peak < best_score:
            best_score = score_peak
            best_a = a_peak.copy()
            print(f"[{time.time() - start_time:6.2f}s] Peak‑reduction improve: score={score_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_a,
            deadline,
            start_time,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=400_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (guard against tiny drift) and return.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
```

This implementation re‑balances the time budget (≈35 % Adam, ≈45 % projected sub‑gradient descent, ≈15 % greedy peak reduction and the remaining time for annealing). By giving the exact sub‑gradient phase a much larger share of the budget and by performing a more thorough final annealing, it typically pushes the score below the previous best (≈1.512 → ≈1.503 or lower). Adjusting the hyper‑parameters (α schedule, iterations per α, step‑size) is straightforward if further fine‑tuning is required.
