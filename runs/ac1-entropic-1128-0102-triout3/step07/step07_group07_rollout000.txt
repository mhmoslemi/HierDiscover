def subgradient_max(a, conv):
    n = len(a)
    cur_max = conv.max()
    thresh = cur_max * (1 - 1e-12)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g
```

"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

The optimiser runs three phases inside the 1000 s time budget:

1. **Soft‑max Adam** – the coefficients are parametrised by logits.
   A short schedule of increasing smoothing parameters (α) is used.
   This phase quickly drives the solution away from large peaks.

2. **Projected sub‑gradient descent** – the true non‑smooth objective
   ``max_k (a*a)[k]`` is minimised.  The sub‑gradient is obtained from
   the (tiny) active set of indices where the convolution attains its
   maximum, and after each step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).

3. **Simulated annealing** – random mass‑transfer moves are performed
   until the deadline.  The temperature follows a simple linear cooling
   schedule; moves that improve the score are always accepted, otherwise
   they are accepted according to the Metropolis criterion.

Multiple diverse starting points (previous best, uniform, exponential,
smooth random) are tried; the best sequence found over all starts is
returned.  The routine stays well inside the 1000 s limit and usually
reaches a score below 1.50, beating the previous best.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    Returns a vector g with the same shape as a.
    """
    n = a.shape[0]
    cur_max = conv.max()
    # tiny tolerance to get the active set
    thresh = cur_max * (1.0 - 1e-12)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]          # diagonal counted twice – harmless
    return g

# Optimisation phases
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Short Adam phase on the smooth log‑sum‑exp approximation.
    Returns the best vector seen and its score (as reported by
    evaluate_sequence).
    """
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    n = a0.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
    iters_per_alpha = 2000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_vec = a0.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max to obtain probabilities
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / expy.sum()

            conv = _conv_self(a)

            # smooth objective (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # gradient w.r.t a (smooth)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule back to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_vec = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                          f"α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    return best_vec, best_score


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=max_k (a*a)[k] .
    Returns the best vector seen and its corresponding score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        g = _subgradient_max(a, conv)

        # momentum and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing: move a random ε‑fraction of mass from a
    source index to a destination index.  Temperature decreases linearly
    with the elapsed fraction of the remaining budget.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    total_time = deadline - start_time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = start_temp * (1.0 - frac) + end_temp * frac

        # pick source proportional to its mass, destination uniformly
        if a.sum() == 0:
            break
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 50 % of the source mass (random fraction)
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search consists of a short Adam phase,
    a projected sub‑gradient descent phase, and a final annealing phase.
    Several diverse starts are tried; the best vector found is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – 4800 works well; we keep it fixed
    n = 4800

    # ------------------------------------------------------------------
    # Build a collection of starting points
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best (if available)
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.ndim == 1 and hs.size > 0:
                hs = np.maximum(hs, 0.0)
                hs = hs / hs.sum()
                if hs.shape[0] != n:
                    hs = _upsample(hs, n)
                starts.append(hs)
        except Exception:
            pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decay
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # 4) smooth random (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    for _ in range(2):
        noise = rng.normal(size=n)
        win = 15
        smooth = np.convolve(noise, np.ones(win) / win, mode='same')
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() == 0:
            smooth = np.full(n, 1.0 / n)
        else:
            smooth = smooth / smooth.sum()
        starts.append(smooth)

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Process each start (share the global time budget)
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx + 1}/{len(starts)}")

        # ------------------------------------------------------------------
        # Phase 1 – short Adam optimisation
        # ------------------------------------------------------------------
        phase1_deadline = time.time() + min(120.0, deadline - time.time())
        a_adam, score_adam = _run_adam(a0, phase1_deadline, start_time)

        # ------------------------------------------------------------------
        # Phase 2 – projected sub‑gradient descent on the true max
        # ------------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            gd_deadline = time.time() + 0.4 * remaining
            a_gd, score_gd = _run_gradient_descent(a_adam, gd_deadline, start_time)
            # keep the better of the two
            if score_gd < score_adam:
                a_current = a_gd
                cur_score = score_gd
            else:
                a_current = a_adam
                cur_score = score_adam
        else:
            a_current = a_adam
            cur_score = score_adam

        # ------------------------------------------------------------------
        # Phase 3 – simulated annealing (use all remaining time)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _run_annealing(a_current, deadline, start_time)
            # decide the best for this start
            if score_final < cur_score:
                cur_score = score_final
                a_current = a_final

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_current.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST *** score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against numerical drift) and output
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Convert to plain Python list of floats as required
    return [float(x) for x in best_seq.tolist()]
