# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers (real input)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)   (used for Adam‑type optimisation)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the true max‑convolution (using sub‑gradient)
# ----------------------------------------------------------------------
def _run_adam_subgrad(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.4,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation where the gradient is the sub‑gradient of max(conv).
    Returns the best vector seen and its true score (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam‑subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfer)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑interpolation up‑sampling + re‑normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Upsample (linear interpolation) and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, seq)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.full_like(out, 1.0 / new_n)
    return out


# ----------------------------------------------------------------------
# Adam on the soft‑max surrogate (used only for a quick warm‑start)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Oτ(a)=2·n·τ·log∑exp(conv(a)/τ)
    using Adam and projection onto the simplex.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam‑softmax iter={it:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")

        # --- soft‑max weights ---
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # --- gradient of Oτ ---
        a_rev = a[::-1]
        conv_rev = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(p), n=_next_pow2(3 * n - 2))[: 3 * n - 2]
        inner = conv_rev[n - 1:n - 1 + n]
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method combines a quick warm‑start (soft‑max Adam), a long Adam run on the
    exact max‑convolution sub‑gradient, and a final simulated‑annealing polish.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 0) Obtain a reasonable starting vector (use previous if present)
    # --------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # Target length – a power of two that still runs comfortably under the budget
    n_target = 16384

    if starter is not None:
        a_current = _upsample(starter, n_target)
    else:
        # No good starter – launch a few quick Adam‑softmax runs on a modest size
        n_small = 2048
        best_small = None
        best_small_score = float("inf")
        # allocate a short slice of the budget for these warm‑starts
        slice_time = max(20.0, 0.08 * (deadline - time.time()))
        slice_deadline = time.time() + slice_time

        for r in range(6):
            if time.time() >= slice_deadline:
                break
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()
            conv0 = _conv_self(a0)
            tau0 = max(conv0.max() * 10.0, 1e-8)
            a_opt, score = _run_adam_softmax(
                a0, n_small, slice_deadline, start_time,
                tau0=tau0, tau_end=1e-12, tau_decay=0.9995, step_size=0.25
            )
            if score < best_small_score:
                best_small_score = score
                best_small = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] warm‑start {r} improve: score={score:.6f}")

        if best_small is None:
            a_current = np.random.rand(n_target)
            a_current /= a_current.sum()
        else:
            a_current = _upsample(best_small, n_target)

    # --------------------------------------------------------------
    # 1) Adam on the exact sub‑gradient of max‑convolution
    # --------------------------------------------------------------
    conv = _conv_self(a_current)
    best_score = 2.0 * n_target * conv.max()
    best_seq = a_current.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial score: {best_score:.6f}")

    # Reserve the bulk of the remaining budget for this stage
    remaining = deadline - time.time()
    subgrad_time = remaining * 0.85
    subgrad_deadline = time.time() + subgrad_time

    a_adam, score_adam = _run_adam_subgrad(
        best_seq, n_target,
        subgrad_deadline, start_time,
        step0=0.4, beta1=0.9, beta2=0.999, eps=1e-8
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:6.2f}s] after adam‑subgrad: score={score_adam:.6f}")

    # --------------------------------------------------------------
    # 2) Very light simulated‑annealing polish (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq, n_target,
            deadline, start_time,
            best_score, best_seq,
            start_temp=2e-2, end_temp=1e-7,
            steps=500_000
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improvement: score={score_final:.6f}")

    # --------------------------------------------------------------
    # 3) Normalise to the simplex and return a plain Python list
    # --------------------------------------------------------------
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
