"""
Search for a non‑negative step function that minimizes

    2·n·max(a*a) / (sum a)^2

using a three‑stage strategy:

1.  Fast parametric sweep (U‑shaped families) on the full length.
2.  Sub‑gradient descent on the true max‑convolution (projected onto the simplex).
3.  Targeted “peak‑shaving” local search with a Metropolis acceptance rule.

The code respects a 1000 s budget, prints progress, and finally returns a plain
Python list suitable for the evaluation harness.
"""
# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (fast for large n)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Score (exactly the same formula as evaluate_sequence)
# ----------------------------------------------------------------------
def _score(sequence: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum^2).  Assumes sequence is non‑negative."""
    s = sequence.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(sequence)
    return 2.0 * sequence.shape[0] * conv.max() / (s * s)

# ----------------------------------------------------------------------
# Parametric families (U‑shaped) -------------------------------------------------
def _shape_exp(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam*i) + exp(-lam*(n-1-i))   (lam ≥ 0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return a

def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i = ((i+1)*(n-i))**p   (p ≤ 0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = i * (n - i + 1.0)
    a = np.power(w, p, where=w > 0, out=np.zeros_like(w))
    return a

def _shape_cos(n: int) -> np.ndarray:
    """a_i = 1 + cos(pi*i/(n-1)) – peaks at the two ends."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(np.pi * i / (n - 1))
    return a

def _shape_mix(n: int, lam: float, p: float) -> np.ndarray:
    """Mixture of exponential and power‑law shapes."""
    return _shape_exp(n, lam) * _shape_power(n, p)

# ----------------------------------------------------------------------
# Randomised parametric search (stage 1)
# ----------------------------------------------------------------------
def _param_search(n: int, deadline: float, start_time: float):
    """Explore a few hundred random parameter settings and keep the best."""
    best_seq = None
    best_score = float("inf")
    families = ["exp", "power", "cos", "mix"]
    while time.time() < deadline:
        family = random.choice(families)
        if family == "exp":
            lam = random.uniform(5e-4, 5e-3)   # modest decay
            seq = _shape_exp(n, lam)
        elif family == "power":
            p = random.uniform(-1.5, -0.1)    # negative exponent
            seq = _shape_power(n, p)
        elif family == "cos":
            seq = _shape_cos(n)
        else:  # mix
            lam = random.uniform(5e-4, 5e-3)
            p = random.uniform(-1.5, -0.1)
            seq = _shape_mix(n, lam, p)

        # small random jitter to break symmetry (helps later refinement)
        if random.random() < 0.3:
            jitter = np.random.rand(n) * 1e-4
            seq = seq + jitter

        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total <= 0.0:
            continue
        seq = seq / total
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] param {family:4s} "
                  f"score={sc:.6f}")
    return best_seq, best_score

# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (stage 2)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 1.0,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 7):
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max] / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter={it} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the active shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # momentum + diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Targeted local search (stage 3)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float):
    """Mass‑shifting walk that reduces the maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max] / (a.sum() ** 2)

        # contribution of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_idx)

        if donor == recipient or a[donor] <= 1e-15:
            continue

        eps = a[donor] * np.random.rand() * 0.3  # move up to 30 % of donor mass
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max() / (a_prop.sum() ** 2)
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta < 0.0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local iter={it} "
                      f"score={new_score:.6f}")

    return best_a, best_score

# Main entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (probability vector) that
    attempts to minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # tiny safety margin to avoid overrunning the budget
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine final length (use the length we have experience with)
    # ------------------------------------------------------------------
    final_n = 9600

    # ------------------------------------------------------------------
    # Warm‑start from previous best if available
    # ------------------------------------------------------------------
    warm_seq = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            warm_seq = tmp / tmp.sum()
            if warm_seq.shape[0] != final_n:
                # up‑sample / down‑sample to required size
                old_n = warm_seq.shape[0]
                old_x = np.linspace(0.0, 1.0, old_n)
                new_x = np.linspace(0.0, 1.0, final_n)
                warm_seq = np.interp(new_x, old_x, warm_seq)
                warm_seq = warm_seq / warm_seq.sum()
    except Exception:
        warm_seq = None

    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep (≈20 % of the time budget)
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    stage1_deadline = time.time() + 0.20 * time_remaining
    print(f"Stage 1: parametric sweep (up to {stage1_deadline - time.time():.1f}s)")

    best_seq = None
    best_score = float("inf")

    if warm_seq is not None:
        best_seq = warm_seq.copy()
        best_score = _score(best_seq)
        print(f"  Warm‑start score={best_score:.6f}")

    seq_ps, score_ps = _param_search(final_n, stage1_deadline, start_time)
    if seq_ps is not None and score_ps < best_score:
        best_seq = seq_ps
        best_score = score_ps

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient refinement (≈60 % of the remaining time)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline and best_seq is not None:
        remaining = deadline - now
        stage2_deadline = now + 0.60 * remaining
        print(f"Stage 2: sub‑gradient descent (up to {stage2_deadline - now:.1f}s)")
        refined_seq, refined_score = _run_subgradient_gd(
            best_seq, final_n, stage2_deadline, start_time,
            step0=1.0, beta=0.5, max_iter=10 ** 7
        )
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"  Sub‑gradient improvement: {refined_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted local search (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        print("Stage 3: targeted local search")
        loc_seq, loc_score = _run_targeted_local_search(
            best_seq, final_n, deadline, start_time
        )
        if loc_score < best_score:
            best_seq = loc_seq
            best_score = loc_score
            print(f"  Local‑search improvement: {loc_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drifts may appear)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"Final best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
