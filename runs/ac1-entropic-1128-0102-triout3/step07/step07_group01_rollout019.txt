def _spectral_flatten(a, target_mag=None, steps=100):
    For step in range(steps):
        Af = np.fft.rfft(a, n=L) # length L
        if target_mag is None:
            # compute mean magnitude squared of non-DC components
            mag_sq = np.abs(Af)**2
            # Exclude DC (index 0) from average
            avg = np.mean(mag_sq[1:])  # maybe except DC?
            target_mag = np.sqrt(avg)
        # Build new Af2:
        Af2 = Af.copy()
        Af2[0] = Af[0]  # preserve DC amplitude
        # set magnitude of other components to target_mag, keep phase:
        phases = Af[1:] / np.maximum(np.abs(Af[1:]), 1e-12)
        Af2[1:] = target_mag * phases
        # Inverse transform to get new a:
        a_new = np.fft.irfft(Af2, n=L)[:n]  # only first n values.
        # Clip negatives, renorm:
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.full_like(a_new, 1.0 / n)
        a = a_new
    return a

def _run_spectral_flatten(a0, n, deadline, start_time, iters=2000):
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    for i in range(iters):
        if time.time() >= deadline:
            break
        Af = np.fft.rfft(a, n=L)
        # compute target magnitude as sqrt(mean(|Af[1:]|^2))
        mag_sq = np.abs(Af[1:])**2
        target = np.sqrt(np.mean(mag_sq))
        # preserve DC amplitude
        phases = Af[1:] / np.maximum(np.abs(Af[1:]), 1e-12)
        Af_new = Af.copy()
        Af_new[1:] = target * phases
        # inverse
        a_new = np.fft.irfft(Af_new, n=L)[:n]

        s = a_new.sum()
        if s > 0:

        else:

        a = a_new
    return a

def _run_targeted_refine_adaptive(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # adapt eps_frac: start bigger, decay over iterations.
        # we can compute time fraction.
        elapsed = time.time() - start_time
        total = deadline - start_time
        # eps_frac = max(0.001, 0.02 * (1 - elapsed/total) )
        eps_frac = max(0.001, 0.018 * (1 - elapsed / total))
        # choose src with probability proportional to contribution to max conv.
        k_max = int(np.argmax(conv))
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

"""
Improved step‑function search for the “evaluate_sequence” objective.

The pipeline consists of four stages:

1. **Spectral flattening** – repeatedly force the Fourier magnitude of the
   sequence to be (approximately) constant while keeping the vector
   non‑negative and normalised.  This spreads the mass in a way that tends
   to produce a flat autocorrelation.

2. **Warm‑up with exponentiated‑gradient** – a smoothed max (log‑sum‑exp)
   is minimised, giving a fast reduction of the worst convolution entry.

3. **Exact projected sub‑gradient descent** – the true ℓ∞‑norm of the
   self‑convolution is tackled directly.  A larger initial step and a
   slightly smaller momentum decay accelerate convergence.

4. **Adaptive targeted refinement** – tiny amounts of probability mass are
   moved away from the indices that contribute most to the current maximal
   convolution entry.  The fraction of moved mass decays over time, allowing
   coarse‑grained improvements early on and fine tuning later.

Multiple restarts are not needed: the whole budget (≈1000 s) is split
between the stages, and each stage stops automatically when the time
budget is exhausted.  The returned sequence is non‑negative, normalised
to sum 1 and can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    The result is scaled by 1/L where L is the FFT length.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_true_max(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Return the true (un‑scaled) maximal convolution value."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    if conv is None:
        conv = _conv_self(a)
    # conv is scaled by 1/L, so multiply back
    return conv.max() * L


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 0 – spectral flattening
# ----------------------------------------------------------------------
def _run_spectral_flatten(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         iters: int = 3000) -> np.ndarray:
    """
    Enforce a flat Fourier magnitude (except the DC term) while keeping
    the vector non‑negative and normalised.  This usually yields a
    distribution whose self‑convolution is already fairly level.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)

    for it in range(iters):
        if time.time() >= deadline:
            break

        Af = np.fft.rfft(a, n=L)

        # target magnitude = RMS of non‑DC components
        mag_sq = np.abs(Af[1:]) ** 2
        rms = np.sqrt(np.mean(mag_sq))

        # preserve the phase of each coefficient
        phase = Af[1:] / np.maximum(np.abs(Af[1:]), 1e-15)
        Af_new = Af.copy()
        Af_new[1:] = rms * phase

        a_new = np.fft.irfft(Af_new, n=L)[:n]
        a_new = np.maximum(a_new, 0.0)

        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        a = a_new

        if it % 500 == 0:
            # report progress
            conv = _conv_self(a)
            score = 2 * n * _next_pow2(2 * n - 1) * conv.max()
            print(f"[{time.time() - start_time:.2f}s] spectral it={it:,} score≈{score:.6f}")

    return a


# ----------------------------------------------------------------------
# Stage 1 – exponentiated gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * _next_pow2(2 * n - 1) * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5,
                             beta: float = 0.90,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * _next_pow2(2 * n - 1) * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – adaptive targeted refinement
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         eps_start: float = 0.018,
                         eps_end: float = 0.001):
    """
    Tiny mass moves are performed repeatedly; the fraction of moved mass
    decays linearly from ``eps_start`` to ``eps_end`` as the allotted
    time runs out.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1

        # linear decay of step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        eps_frac = eps_end + (eps_start - eps_end) * max(0.0, 1.0 - elapsed / total)

        # location of the current maximum
        k_max = int(np.argmax(conv))

        # contributions of each index to this maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)

        if a[src] <= 1e-15:
            continue

        # destination – uniform different index
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-14:          # enforce strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.9f}")

        # occasional early exit if stagnating for a long period
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a
