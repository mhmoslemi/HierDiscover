def _run_primal_dual(a0, n, deadline, start_time, step_a0=0.5, step_w0=0.5, max_iter=10**7):
    a = a0.copy()
    w = np.full(2*n-1, 1.0/(2*n-1))
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # convolution
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] primal_dual improve: n={n}, iter={it}, score={cur_score:.6f}")

        # adaptive step sizes
        alpha = step_a0 / np.sqrt(it)
        beta = step_w0 / np.sqrt(it)

        # gradient for a: 2 * rev(a) * w (conv)
        rev_a = a[::-1]
        # conv rev_a with w
        fft_len = 1 << ((n + (2*n-1) - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A*B, n=fft_len)[: n + (2*n-1) - 1]
        grad_a = 2.0 * full[n-1:n-1 + n]

        # update a via projected gradient
        a = _project_simplex(a - alpha * grad_a)

        # gradient for w: c = conv
        # update w via exponentiated gradient ascent
        # exponentiate with relative scaling to avoid overflow
        # Use beta as step size in exponent
        # w = w * np.exp(beta * conv)
        # Normalize to simplex
        # To improve stability, subtract max from conv
        maxc = conv.max()
        exp_ = np.exp(beta * (conv - maxc))
        w = w * exp_
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

    return best_a, best_score

```python
"""
Primal–dual projected gradient for min‑max auto‑correlation.

The algorithm works in stages of increasing resolution.
At each stage a primal variable a (the step heights) lives on the
probability simplex and a dual variable w (a probability distribution
over the convolution indices) lives on the simplex as well.
We perform simultaneous projected gradient descent on a and
exponentiated‑gradient ascent on w.
The objective 2·n·max((a*a)) is evaluated after each iteration and the
best sequence seen is kept.  Finally a simple simulated‑annealing
refinement is applied.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _conv_rev_w(rev_a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution rev_a (length n) with w (length 2n-1). Returns slice needed for grad."""
    n = rev_a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    # extract indices n-1 ... n-1+n-1 (total n entries)
    return full[n - 1:n - 1 + n]

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Primal‑dual projected gradient / exponentiated‑gradient ascent.
# ----------------------------------------------------------------------
def _run_primal_dual(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     step_a0: float = 0.8,
                     step_w0: float = 0.5,
                     beta: float = 0.9,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Simultaneous primal (projected gradient) – dual (exponentiated‑gradient) loop.
    Returns the best sequence a found and its objective value 2·n·max(conv).
    """
    a = a0.copy()
    # initialise dual weights uniformly over the 2n‑1 convolution indices
    w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    momentum = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # --- compute convolution and current objective ---
        conv = _conv_self(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:   # strict improvement
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] primal‑dual improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # --- step sizes (diminishing) ---
        alpha = step_a0 / math.sqrt(it)        # primal step
        gamma = step_w0 / math.sqrt(it)        # dual (exponentiated‑gradient) step

        # --- gradient w.r.t. a : 2 * rev(a) * w ---
        rev_a = a[::-1]
        grad_a = 2.0 * _conv_rev_w(rev_a, w)

        # --- momentum update (Nesterov‑style) ---
        momentum = beta * momentum + (1.0 - beta) * grad_a
        a_next = a - alpha * momentum
        a = _project_simplex(a_next)

        # --- dual ascent (exponentiated gradient) ---
        # (dual objective is max_c = max_k conv_k, its sub‑gradient w.r.t. w is conv)
        # we perform a multiplicative update on the simplex.
        max_c_val = max_c
        # shift by max to avoid overflow
        exp_factor = np.exp(gamma * (conv - max_c_val))
        w = w * exp_factor
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

    return best_a, best_score

# Simulated annealing refinement (same as baseline, but a bit more aggressive)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass and occasionally smooths."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25        # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

"""
Primal–dual projected gradient for min‑max auto‑correlation.

The search proceeds through increasing resolutions.
At each resolution we maintain a primal variable `a` (the step heights)
on the probability simplex and a dual variable `w` (a distribution over
convolution indices) also on the simplex.  A projected gradient step
updates `a` while an exponentiated‑gradient ascent updates `w`.  The
objective 2·n·max((a*a)) is evaluated after each iteration and the best
sequence seen is kept.  After all stages a simulated‑annealing refinement
finishes the search.

The method differs from the reference implementation by using a
primal‑dual loop instead of a separate soft‑max / sub‑gradient phase.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_rev_w(rev_a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of rev_a (length n) with w (length 2n‑1).
    Returns the slice needed for the gradient (n entries).
    """
    n = rev_a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return full[n - 1:n - 1 + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Primal‑dual optimisation (projected gradient + exponentiated‑gradient)
# ----------------------------------------------------------------------
def _run_primal_dual(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     step_a0: float = 0.8,
                     step_w0: float = 0.5,
                     beta: float = 0.9,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Simultaneous primal (projected gradient) – dual (exponentiated‑gradient) loop.
    Returns the best sequence a found and its objective value 2·n·max(conv).
    """
    a = a0.copy()
    w = np.full(2 * n - 1, 1.0 / (2 * n - 1))  # uniform dual distribution
    momentum = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] primal‑dual improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # step sizes decaying as 1/sqrt(it)
        alpha = step_a0 / math.sqrt(it)      # primal step
        gamma = step_w0 / math.sqrt(it)      # dual step

        # gradient w.r.t. a : 2 * rev(a) * w
        rev_a = a[::-1]
        grad_a = 2.0 * _conv_rev_w(rev_a, w)

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad_a
        a = _project_simplex(a - alpha * momentum)

        # dual ascent (exponentiated gradient)
        # maximise sum_k w_k * conv_k  -> gradient = conv
        max_c_val = max_c
        exp_factor = np.exp(gamma * (conv - max_c_val))
        w = w * exp_factor
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.shape[0])

    return best_a, best_score

        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # optional: report the value as seen by the host evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final evaluated score: {final_val:.6f}")
    except Exception:
        pass

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
