"""
Hybrid optimiser for the step‑function anti‑concentration problem.

The new version adds a short L‑BFGS‑B phase on a smooth log‑sum‑exp
approximation of the maximal self‑convolution.  This provides a very fast
refinement of the distribution before the (expensive) exact projected‑
gradient polishing and the final incremental simulated‑annealing.
A higher starting temperature and finer move size are used in the annealing
stage.  The whole pipeline respects the 1000 s budget and returns the best
sequence seen.
"""

import time
import math
import numpy as np

# optional SciPy for L‑BFGS‑B
try:
    import scipy.optimize
    _has_scipy = True
except Exception:          # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS callbacks (avoid closures)
# ----------------------------------------------------------------------
_global_alpha = None   # smoothing parameter for the soft‑max objective
_global_n = None       # length of the sequence
_global_fft_len = None


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    maxy = np.max(y)
    expy = np.exp(y - maxy)
    return expy / expy.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# L‑BFGS helpers (global‑state based callbacks)
# ----------------------------------------------------------------------
def _lbfgs_obj(y: np.ndarray) -> float:
    """
    Objective for L‑BFGS: smoothed 2·n·max(conv) with current _global_alpha.
    """
    global _global_alpha, _global_n, _global_fft_len
    a = _softmax(y)
    conv = _conv_self(a)
    # log‑sum‑exp smoothing
    scaled = _global_alpha * conv
    max_scaled = scaled.max()
    lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
    f_val = lse / _global_alpha
    return float(2.0 * _global_n * f_val)


def _lbfgs_grad(y: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. logits y.
    """
    global _global_alpha, _global_n, _global_fft_len
    a = _softmax(y)
    conv = _conv_self(a)
    grad_a = _grad_smooth_max(a, conv, _global_alpha, _global_fft_len)
    grad_a *= 2.0 * _global_n
    # chain rule through soft‑max
    grad_y = a * (grad_a - np.dot(grad_a, a))
    return grad_y.astype(float)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (same as previous version)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """
    Simple projected gradient descent with momentum on the exact max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing (tuned parameters)
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 2_000_000):
    """
    Mass‑transfer annealing with a higher starting temperature and smaller moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-1           # higher temperature than before
    end_temp = 1e-5

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # source proportional to current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 10 % of the source mass (smaller steps)
        eps = a[src] * rng.random() * 0.10

        # ----- apply incremental convolution update (using old a) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis criterion
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # reject – revert
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Pipeline:
      1. L‑BFGS‑B on a smoothed max‑convolution (multiple α values).
      2. Exact projected sub‑gradient polishing.
      3. High‑temperature incremental annealing.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                a0 = _upsample(a0, n)
        else:
            raise Exception
    except Exception:                     # pragma: no cover
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{0.0:6.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS‑B on a smoothed objective (if SciPy is available)
    # ------------------------------------------------------------------
    if _has_scipy and time.time() < deadline:
        # initialise logits from the current a0
        y = np.log(np.maximum(a0, 1e-16))
        fft_len = _next_pow2(3 * n - 2)

        # increasing smoothing parameter schedule
        alpha_schedule = [5, 10, 20, 40, 80, 160, 320, 640]
        for alpha in alpha_schedule:
            if time.time() >= deadline:
                break
            _global_alpha = float(alpha)
            _global_n = n
            _global_fft_len = fft_len

            # L‑BFGS‑B call – only a few iterations are needed at each α
            res = scipy.optimize.minimize(_lbfgs_obj,
                                          y,
                                          method='L-BFGS-B',
                                          jac=_lbfgs_grad,
                                          options={'maxiter': 30, 'disp': False})
            y = res.x
            # obtain the distribution from the updated logits
            a = _softmax(y)

            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] L‑BFGS improve: α={alpha}, score={sc:.6f}")

        # final a from the latest logits
        a = _softmax(y)
    else:
        # fall‑back to the original Adam phase (kept for completeness)
        a = a0.copy()
        # (the Adam implementation from the previous version could be inserted here
        #  if desired, but the L‑BFGS phase is the main improvement)

    # ------------------------------------------------------------------
    # Phase 2 – Exact projected gradient polishing (intense but time‑aware)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly half of the remaining time to polishing
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            a,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Fast incremental annealing (most of the leftover budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_incremental_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            total_steps=2_000_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
