"""
Hybrid optimiser for the step‑function attack (improved).

The search consists of four stages, each respecting a global time budget
(≈1000 s) and always keeping the vector on the probability simplex.

1. **Adam on a smooth log‑sum‑exp approximation of the max‑convolution**.
   The smoothing factor α is increased up to 1024, giving a very tight
   gradient for the true max‑operator.

2. **Exact sub‑gradient descent (momentum‑projected)** – a long polishing
   phase that minimises the true maximal convolution.

3. **Targeted simulated annealing** – instead of a blind source→dest
   transfer we deliberately pick a source index that participates in the
   currently worst convolution entry (the shift with maximal mass).  This
   focuses the search where it matters most.  The temperature schedule
   starts at 1.0 and cools geometrically to 1e‑7, with many millions of
   cheap O(n) moves and periodic exact recomputation of the full
   convolution to avoid drift.

4. **Final short gradient polish** – cleans up the result after annealing.

All phases keep the best ever sequence (and its score) and return the
normalised vector at the end.  The algorithm works for any initial
vector; if the global variable ``height_sequence_1`` exists it is used as a
warm start, otherwise a fresh random Dirichlet vector is created.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1)."""
    # np.convolve is O(n²) but n=4800 – cheap enough for occasional full recompute.
    return np.convolve(a, a)


def _grad_smooth_max(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     fft_len: int) -> np.ndarray:
    """Gradient of  (1/α)·log Σ exp(α·conv)  w.r.t. a (uses FFT)."""
    n = a.shape[0]

    # soft‑max over convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Phase 1 – Adam on smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimisation of the log‑sum‑exp smoothed objective.
    Returns the final vector, the best score seen and the best sequence.
    """
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # α schedule – up to 1024 for a very tight approximation
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0,
                      320.0, 640.0, 1024.0]
    iters_per_alpha = 9000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max reconstruction of a
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective (for diagnostics)
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # actual objective is 2·n·f_val, but we only need gradients

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule to logits y (soft‑max)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y -= lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.0f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Phase 2 – Exact projected sub‑gradient descent (polish)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """
    Momentum projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Grad improve: iter={it:,}, score={cur_score:.6f}")

        # identify active shifts (within 1e-12 of max)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient: sum of contributions from each active shift
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair (i,j) contributes a_j to gradient of a_i and vice‑versa
            g[i] += a[j]
            g[j] += a[i]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted incremental simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_seq: np.ndarray,
                            cur_best_score: float,
                            total_steps: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that always picks a source index involved in the
    current maximal convolution entry.  Moves are mass‑transfer updates
    performed in O(n) using the same slice trick as before.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_T = 1.0
    end_T = 1e-7
    recompute_every = max(1, total_steps // 200)   # full recompute ~0.5 % of steps

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule (geometric)
        T = start_T * (end_T / start_T) ** (it / total_steps)

        # location of the worst (largest) convolution entry
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        prod = a[idx] * a[k_max - idx]
        total_prod = prod.sum()
        if total_prod > 0:
            src = rng.choice(idx, p=prod / total_prod)
        else:
            src = rng.choice(idx)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        if eps <= 0.0:
            continue

        # ----- apply move (using the old a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move (a now holds the old values) -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Targeted anneal improve: iter={it:,}, score={score:.6f}")

        # occasional exact recomputation to avoid drift
        if it % recompute_every == 0:
            a_sum = a.sum()
            if a_sum != 0:
                a /= a_sum
            conv = _conv_self(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Re‑eval improve (recompute): iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method mixes an Adam‑phase, long sub‑gradient polishing and a targeted
    simulated‑annealing, all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.6

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter.copy()
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)

    # protect against negatives and normalise onto the simplex
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{0.0:6.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed objective
    # ------------------------------------------------------------------
    a_adam, best_score, best_seq = _adam_phase(
        best_seq, n, deadline, start_time, best_score, best_seq
    )

    # ------------------------------------------------------------------
    # Phase 2 – Exact projected gradient polishing (intense)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly 30 % of the remaining budget to gradient polish
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.30 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # give the remaining time to annealing
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_seq,
            best_score,
            total_steps=6_000_000,        # as many moves as time permits
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final short gradient polish (cleanup)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        fin_deadline = time.time() + remaining
        a_clean, score_clean = _run_gradient_descent(
            best_seq,
            n,
            fin_deadline,
            start_time,
            beta=0.94,
            step0=0.15,
            max_iter=2_000_000,
        )
        if score_clean < best_score:
            best_score = score_clean
            best_seq = a_clean.copy()
            print(f"[{time.time() - start_time:6.2f}s] final polish improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    # Return a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
