"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1.  Choose a target length N (≈ 1.5·10⁴).  Initialise a (sum = 1) either from the
    previous best construction or from a sparse Dirichlet draw.
2.  Iteratively improve the distribution by **mirror‑descent** on a sub‑gradient
    of the (exact) max‑convolution:
        * compute the self‑convolution a∗a (FFT, O(N log N)).
        * locate the K largest convolution entries (K≈3) – these are the active
          constraints.
        * build a weight vector w that has 1 on those K indices, 0 elsewhere.
        * the sub‑gradient of Σ_k w_k·(a∗a)_k is 2·rev(a)∗w, which is obtained
          fast with another FFT.
        * update a multiplicatively: a ← a·exp(−η·g) (entropy mirror map),
          then renormalise onto the simplex.
    η (step size) follows a 1/√t schedule; a momentum term smooths the direction.
3.  Every few hundred iterations a tiny Gaussian jitter is added (and re‑normalised)
    to escape shallow plateaus.
4.  The best (lowest‑objective) sequence seen during the run is kept.
5.  When the time budget (≈ 1000 s) expires the best sequence is returned as a
    plain Python list.

The mirror‑descent update respects the simplex constraints without an
explicit Euclidean projection, often converging faster than the projected
gradient used previously.  By handling several (near‑maximal) convolution
entries simultaneously the algorithm reduces the peak of the autocorrelation
more aggressively, aiming for an objective below the previous 1.512 ⋯ target
≈ 1.503.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of reversed a with a weight vector w.
#          Returns the full linear convolution (length = n + len(w) - 1).
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent sub‑gradient optimiser (handles several top peaks)
# ----------------------------------------------------------------------
def _mirror_descent_subgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    eta0: float = 0.7,
    beta: float = 0.9,
    K: int = 3,
    jitter_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence and its (unnormalised) objective value.
    """
    a = a0.copy()
    # ensure positivity
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    # momentum accumulator for the gradient (exponential moving average)
    g_avg = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- evaluate current objective -----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] MD iter={it:5d} "
                f"score={cur_score:.6f}"
            )

        # ----- pick the K largest convolution entries -----
        # (if fewer than K entries exist, use all of them)
        if K >= conv.size:
            peak_idx = np.arange(conv.size)
        else:
            # argpartition gives the K largest (unsorted)
            peak_idx = np.argpartition(conv, -K)[-K:]

        # ----- build weight vector for those peaks -----
        w = np.zeros_like(conv)
        w[peak_idx] = 1.0  # uniform weight over the selected peaks

        # ----- sub‑gradient of Σ w_k (a∗a)_k = 2·rev(a)∗w -----
        rev_a = a[::-1]
        full = _conv_rev_weight(rev_a, w)          # length = 3n‑2
        g = 2.0 * full[n - 1 : n - 1 + n]          # centre block of length n

        # ----- momentum smoothing -----
        g_avg = beta * g_avg + (1.0 - beta) * g

        # ----- step‑size (1/√t) schedule -----
        eta = eta0 / math.sqrt(it)

        # ----- mirror‑descent (entropy) update -----
        a = a * np.exp(-eta * g_avg)

        # ----- keep positivity and re‑normalise onto the simplex -----
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- occasional jitter to avoid stagnation -----
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 1e-4
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

    return best_a, best_score
