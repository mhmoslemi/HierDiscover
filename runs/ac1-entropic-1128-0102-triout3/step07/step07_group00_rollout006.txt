"""
Optimiser for the “max‑self‑convolution” evaluation.

The overall strategy is a multi‑restart local‑search pipeline:

*   A fast “Adam” stage on a log‑sum‑exp (smooth‑max) surrogate quickly shapes
    the distribution.
*   A projected sub‑gradient descent on the true objective refines the shape.
*   A light simulated‑annealing polish removes residual peaks.
*   The whole pipeline is run several times on a modest size (≈1024) – cheap
    enough to explore many starts – and the best run is up‑sampled to the
    target size (4096) where a final short refinement is performed.

The code respects the 1000 s budget, falls back to the global
``height_sequence_1`` if it exists, and returns a plain Python list of
non‑negative numbers that sum to one (the evaluator rescales anyway).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    max_b = _conv_self(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Phase 1 – Adam on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _run_adam_smoothing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alphas,
    lr0: float = 0.5,
    iters_per_alpha: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest exact objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    L = _next_pow2(3 * n - 2)  # FFT length used for the gradient

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # current (unsmoothed) objective – only for logging
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam improve: "
                    f"iter={it:,}, score={cur_score:.6f}"
                )

            # ---------- smooth max ----------
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)   # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ---------- gradient of smooth max ----------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ---------- Adam update ----------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected (sub‑)gradient descent on the exact max
# ----------------------------------------------------------------------


def _run_subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_score: float,
    max_iters: int = None,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    Uses the (unique) maximal entry to compute a sub‑gradient.
    Stops either at *deadline* or after *max_iters* iterations.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while (deadline is None or time.time() < deadline) and (
        max_iters is None or it < max_iters
    ):
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Sub‑grad improve: "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # sub‑gradient from the current maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Light simulated‑annealing polish
# ----------------------------------------------------------------------


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        src = rng.choice(n, p=a)           # source weighted by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move ≤30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Anneal improve: "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Full optimisation pipeline for a *single* length
# ----------------------------------------------------------------------


def _optimise_one(
    a0: np.ndarray,
    n: int,
    budget: float,
    start_time: float,
    global_best_seq: np.ndarray = None,
    global_best_score: float = None,
) -> tuple[np.ndarray, float]:
    """
    Runs the three phases on a given start vector a0 (sum≈1) for *budget*
    seconds and returns the best sequence found.  The returned score is the
    exact objective (2·n·max(conv)).
    """
    deadline = start_time + budget

    # ---------- Phase 1 – Adam smoothing ----------
    phase1_deadline = time.time() + max(5.0, 0.12 * (deadline - time.time()))
    a_adam, score_adam = _run_adam_smoothing(
        a0,
        n,
        phase1_deadline,
        start_time,
        alphas=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        lr0=0.5,
        iters_per_alpha=8000,
    )
    best_seq = a_adam
    best_score = score_adam

    # ---------- Phase 2 – Sub‑gradient descent ----------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(20.0, 0.70 * (deadline - time.time()))
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub

    # ---------- Phase 3 – Annealing ----------
    if time.time() < deadline:
        a_fin, score_fin = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            max_steps=300_000,
        )
        if score_fin < best_score:
            best_score = score_fin
            best_seq = a_fin

    # Return the best found in this run
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Multi‑restart search for a low‑max‑self‑convolution step function.
    The function respects the ~1000 s time budget and returns a list of
    non‑negative heights (a probability vector).  It first searches on a
    modest size (≈1024) to explore many starts cheaply, then up‑samples the
    best candidate to the target size (4096) and performs a final refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Target length and a cheap “small” length for the bulk of the search
    # ------------------------------------------------------------------
    N_target = 4096
    N_small = 1024

    # ------------------------------------------------------------------
    # Keep the globally provided start if it exists.
    # ------------------------------------------------------------------
    global_start = None
    try:
        global_start = np.asarray(height_sequence_1, dtype=float)
        if global_start.ndim != 1 or global_start.size == 0:
            global_start = None
    except Exception:
        global_start = None

    # Helper to normalise a vector to the simplex
    def _normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0:
            v = np.full_like(v, 1.0 / v.size)
        else:
            v = v / s
        return v

    # ------------------------------------------------------------------
    # Baseline: evaluate the supplied global start (if any)
    # ------------------------------------------------------------------
    overall_best_seq = None
    overall_best_score = float("inf")
    if global_start is not None:
        # If the global start is not length N_target, interpolate it.
        if global_start.size != N_target:
            x_old = np.linspace(0.0, 1.0, global_start.size)
            x_new = np.linspace(0.0, 1.0, N_target)
            global_start = np.interp(x_new, x_old, global_start)
        global_start = _normalise(global_start)
        score = _objective(global_start)
        overall_best_seq = global_start
        overall_best_score = score
        print(f"[{time.time() - start_time:.2f}s] Global start score: {score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑restart loop on the cheap size N_small
    # ------------------------------------------------------------------
    # Decide how many restarts we can afford.  We keep at least 30 s for the
    # final up‑sampled refinement.
    min_restart_budget = 110.0
    restarts = 0
    while True:
        time_left = deadline - time.time()
        if time_left < min_restart_budget + 30.0:
            break
        restarts += 1
        # Allocate a fresh slice of the remaining budget (~40 % of the leftovers)
        # but never less than the minimum.
        restart_budget = min(
            max(min_restart_budget, 0.40 * time_left), time_left - 30.0
        )
        print(
            f"[{time.time() - start_time:.2f}s] Restart #{restarts} – "
            f"budget {restart_budget:.1f}s"
        )
        # Random start for the small problem
        rng = np.random.default_rng(seed + restarts * 12345)
        a0_small = rng.random(N_small)
        a0_small = _normalise(a0_small)

        # Run the three‑phase optimiser on the small problem
        best_small, score_small = _optimise_one(
            a0_small, N_small, restart_budget, start_time
        )
        print(
            f"[{time.time() - start_time:.2f}s] Restart #{restarts} small "
            f"score: {score_small:.6f}"
        )

        # Upsample to the target length and perform a short final polish
        x_small = np.linspace(0.0, 1.0, N_small)
        x_target = np.linspace(0.0, 1.0, N_target)
        upsampled = np.interp(x_target, x_small, best_small)
        upsampled = _normalise(upsampled)

        # Remaining time after the upsampled polish (give it a modest slice)
        time_for_final = min(30.0, deadline - time.time())
        if time_for_final > 5.0:  # need at least a few seconds
            final_seq, final_score = _optimise_one(
                upsampled, N_target, time_for_final, start_time
            )
        else:
            final_seq, final_score = upsampled, _objective(upsampled)

        print(
            f"[{time.time() - start_time:.2f}s] Restart #{restarts} final "
            f"score: {final_score:.6f}"
        )

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] *** New overall best: "
                f"{overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # If we still have time left, run a last refinement starting from the
    # current overall best (or from a fresh random start if none yet).
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        print(
            f"[{time.time() - start_time:.2f}s] Final refinement, "
            f"{time_left:.1f}s remaining"
        )
        if overall_best_seq is None:
            a0 = np.random.rand(N_target)
            a0 = _normalise(a0)
        else:
            a0 = overall_best_seq.copy()
        final_seq, final_score = _optimise_one(
            a0, N_target, time_left, start_time
        )
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] *** Final best improved: "
                f"{overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise one last time and return as a plain list.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback: uniform distribution
        overall_best_seq = np.full(N_target, 1.0 / N_target)

    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    total = overall_best_seq.sum()
    if total > 0:
        overall_best_seq /= total
    else:
        overall_best_seq = np.full(N_target, 1.0 / N_target)

    print(
        f"[{time.time() - start_time:.2f}s] Completed – final objective "
        f"{_objective(overall_best_seq):.6f}"
    )
    return [float(x) for x in overall_best_seq]
