def build_constraint_matrix(f, indices):
    n = len(f)
    m = len(indices)
    rows = []
    cols = []
    data = []

        lo = max(0, k - (n - 1))

        i = np.arange(lo, hi + 1)
        j = k - i

        rows.extend([row_idx] * i.size)
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    # Build sparse CSR matrix
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    return A

        return f, M, False

Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.
The optimiser combines three ideas:

1. **Linear‑programming “mix‑step’’** – for the current distribution *f* we
   solve a small LP that finds a distribution *g* with maximal total mass

   The parameters are represented in the log‑domain and projected onto the
   probability simplex after every Adam step.

3. **Exact projected sub‑gradient + greedy peak‑reduction + annealing** – a
   final polishing stage runs a true sub‑gradient descent on the max‑convolution,
   a cheap deterministic mass‑transfer step and a short simulated‑annealing
   routine.  The whole pipeline respects the 1000 s budget and returns the
   best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _score(a: np.ndarray) -> float:
    """
    Fast internal score: 2·n·max(a*a) / (sum(a)**2).
    Identical to ``evaluate_sequence`` up to the clipping performed there.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Linear‑programming mix step
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray, ks: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A (|ks| × n) so that A @ g = (f * g)[ks].
    Row r corresponds to convolution index k = ks[r]; the entry
    A[r, j] = f[k‑j] when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for r, k in enumerate(ks):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        rows.extend([r] * i.size)
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))


def _lp_mix_step(f: np.ndarray, t_mix: float, M: float,
                 active_idx: np.ndarray) -> np.ndarray:
    """
    One LP‑based mixing step.
    Solve   max Σ g_j   s.t. (f * g)[k] ≤ M  for k∈active_idx , g ≥ 0.
    Return the convex combination (1‑t)·f + t·g_norm (still normalised to sum 1).
    """
    # build constraints
    A = _build_constraint_matrix(f, active_idx)

    # linear program (maximise Σ g ⇒ minimise –Σ g)
    n = f.shape[0]
    c = -np.ones(n)                     # minimise –sum(g)
    b = np.full(A.shape[0], M)          # RHS = current peak
    bounds = [(0, None)] * n

    # solve – HiGHS is fast even for a few hundred constraints
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return f  # fallback: keep the old vector

    g = res.x
    if g.sum() == 0:
        return f

    # normalise g to a probability vector (sum = 1)
    g = g / g.sum()

    # convex mixture
    new_f = (1.0 - t_mix) * f + t_mix * g
    new_f = np.maximum(new_f, 0.0)
    s = new_f.sum()
    if s > 0:
        new_f /= s
    else:
        new_f = np.full_like(new_f, 1.0 / n)

    return new_f


def _run_lp_phase(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Repeated LP mixing until the peak cannot be reduced any more
    or time runs out.  The mixing factor t is slowly decreased.
    """
    a = a0.copy()
    t_mix = 0.25
    min_t = 1e-4
    while time.time() < deadline and t_mix > min_t:
        conv = _conv_self(a)
        M = conv.max()
        # active set – indices whose value is within 1e‑6 relative of the peak
        active = np.where(conv >= M * (1.0 - 1e-6))[0]

        # keep the set reasonably small (≤200 rows) for speed
        if active.size > 200:
            # take the 200 largest entries
            active = np.argsort(-conv)[:200]

        a_new = _lp_mix_step(a, t_mix, M, active)

        if _score(a_new) < _score(a) - 1e-12:
            a = a_new
            print(f"[{time.time() - start_time:6.2f}s] LP mix improve: M={M:.6e}, "
                  f"score={_score(a):.6f}")
        else:
            # no improvement – shrink mixing factor
            t_mix *= 0.5

    return a


# ----------------------------------------------------------------------
# Smooth Adam optimisation on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _run_adam_phase(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """Adam on a smoothed max‑convolution objective (log‑sum‑exp)."""
    n = a0.shape[0]
    y = np.log(np.maximum(a0, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # α‑schedule – from very smooth to almost exact max
    alphas = np.geomspace(0.5, 2000.0, num=12)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t_iter = 0

    best_a = a0.copy()
    best_sc = _score(best_a)

    for alpha in alphas:
        # a few thousand steps per α (adjust if time is low)
        steps = 4000
        for _ in range(steps):
            if time.time() >= deadline:
                return best_a

            # soft‑max probabilities
            maxy = np.max(y)
            expy = np.exp(y - maxy)
            a = expy / expy.sum()

            conv = _conv_self(a)

            # smooth approximation of max
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule to logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t_iter) / (1.0 - beta1 ** t_iter))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if _score(a) < best_sc:
                best_sc = _score(a)
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                      f"score={best_sc:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true (max) objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (momentum) sub‑gradient descent on max (a*a)."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: it={it}, "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the max (active set)
        thresh = cur_max * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """Greedy transfer of mass from the coefficient that contributes most to the peak."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (other than src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    final_score = _score(a)
    return a, final_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Classic annealing: random ε‑fraction moves between two indices."""
    a = a0.copy()
    n = a.shape[0]
    score = _score(a)
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: it={it:,}, "
                  f"score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The algorithm runs several starts, each consisting of:

      1. an LP‑mixing phase,
      2. a smooth Adam refinement,
      3. exact projected sub‑gradient descent,
      4. deterministic peak reduction,
      5. final simulated annealing.

    The best sequence over all starts is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4800 works well and matches the earlier runs
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors (all normalised to sum = 1)
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best, if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.shape[0] != n:
                starter = _upsample(starter, n)
            starter = starter / starter.sum()
            starts.append(starter)
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decay
    r = 0.995
    exp_vec = r ** np.arange(n, dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # 4) triangular
    tri = np.concatenate([np.arange(1, n // 2 + 1),
                          np.arange(n // 2, 0, -1)])
    starts.append(tri.astype(float) / tri.sum())

    # 5) random Dirichlet (γ(1,1))
    rng = np.random.default_rng()
    dir_vec = rng.gamma(1.0, 1.0, size=n)
    starts.append(dir_vec / dir_vec.sum())

    # 6) Beta‑shaped smooth vector
    a_param = 0.5 + 4.5 * rng.random()
    b_param = 0.5 + 4.5 * rng.random()
    x = np.linspace(0, 1, n)
    beta_vec = np.where((x > 0) & (x < 1),
                        x ** (a_param - 1) * (1 - x) ** (b_param - 1),
                        0.0)
    if beta_vec.sum() > 0:
        starts.append(beta_vec / beta_vec.sum())

    # 7) Gaussian‑filtered noise
    noise = rng.normal(size=n)
    smooth = np.convolve(noise, np.ones(15) / 15, mode='same')
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() > 0:
        starts.append(smooth / smooth.sum())

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Iterate over the starts (time‑sharing)
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] START #{idx + 1}/{len(starts)}")

        # ensure a proper probability vector
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 > 0:
            a = a0 / s0
        else:
            a = np.full_like(a0, 1.0 / n)

        # --------------------------------------------------------------
        # 1) LP mixing phase (≈15 % of the remaining budget)
        # --------------------------------------------------------------
        lp_deadline = time.time() + min(180.0, (deadline - time.time()) * 0.25)
        a = _run_lp_phase(a, lp_deadline, start_time)

        # --------------------------------------------------------------
        # 2) Smooth Adam phase (≈35 % of the remaining budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            adam_deadline = time.time() + min(300.0, (deadline - time.time()) * 0.35)
            a = _run_adam_phase(a, adam_deadline, start_time)

        # --------------------------------------------------------------
        # 3) Exact projected sub‑gradient descent (≈20 % of the budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            gd_deadline = time.time() + (deadline - time.time()) * 0.2
            a, _ = _run_gradient_descent(a, gd_deadline, start_time)

        # --------------------------------------------------------------
        # 4) Deterministic peak‑reduction (quick cheap pass)
        # --------------------------------------------------------------
        if time.time() < deadline:
            pr_deadline = time.time() + (deadline - time.time()) * 0.1
            a, _ = _run_peak_reduction(a, pr_deadline, start_time, max_no_improve=1200)

        # --------------------------------------------------------------
        # 5) Final simulated‑annealing (use whatever time is left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a, _ = _run_annealing(a, deadline, start_time,
                                  start_temp=1e-1, end_temp=1e-7,
                                  steps=400_000)

        # --------------------------------------------------------------
        # Evaluate candidate and update global best
        # --------------------------------------------------------------
        cand_score = _score(a)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final normalisation (guard against tiny numerical drift) and
    # conversion to plain Python floats.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # The official score is obtained by the provided evaluate_sequence()
    final_evaluation = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] FINAL reported score = {final_evaluation:.6f}")

    return [float(x) for x in best_seq.tolist()]
