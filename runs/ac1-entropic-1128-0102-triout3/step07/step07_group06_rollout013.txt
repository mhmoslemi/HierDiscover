"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.
The search consists of three stages.

1. **Parametric warm‑start** – try several simple families (uniform,
   linear, quadratic and exponential decay).  The best among them is taken
   as a seed.

2. **Soft‑max + sub‑gradient descent** – a projected Nesterov‐GD on a
   temperature‑smoothed version of the max‑convolution, followed by an
   exact sub‑gradient phase that works on the true max‑convolution.
   Both phases keep the vector on the probability simplex.

3. **Simulated‑annealing refinement** – cheap mass‑transfer moves and
   occasional local smoothing are accepted with a geometric temperature
   schedule.  The annealer runs until the time budget is exhausted.

All heavy work (convolution, gradient) uses FFTs (O(n log n)).  The code
adapts the number of iterations to the remaining wall‑clock time, so the
total run never exceeds the supplied budget (default 1000 s).  The final
vector is normalised to sum to one and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x≥0, sum x = 1}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (full length = 2·n‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) surrogate of the max‑convolution and its gradient.
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """∇_a  (temperature‑soft‑max of convolution)."""
    maxc = conv.max()
    # stabilised exponentials
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()                     # weights over the full conv vector
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    # the gradient has exactly n components
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected Nesterov‑style gradient descent on the soft‑max surrogate.
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule – from 5e‑2 down to 5e‑4
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improvement "
                  f"(n={n}, it={it}, score={cur_score:.6f})")
        # geometric temperature decay
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.12,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improvement "
                  f"(n={n}, it={it}, score={cur_score:.6f})")

        # ---- active set (values within rel_eps of the maximum) ----
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing with tiny mass‑transfer moves and occasional
# local smoothing.  The routine stops when the deadline is reached.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25        # up to 25 % of the mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # occasional smoothing
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improvement "
                  f"(n={n}, it={it}, score={score:.6f})")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum == 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    The routine respects the wall‑clock budget, builds a parametric warm‑start,
    refines it with smooth and exact sub‑gradient descent, and finishes with a
    relatively long simulated‑annealing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Small safety margin – we stop a bit before the official deadline.
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a final resolution.  12000 gives a good trade‑off between
    # expressive power and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 12_000

    # ------------------------------------------------------------------
    # Stage 0 – parametric warm‑start (few deterministic families).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # family parameters (p exponent for a_i ~ (1 - i/(n-1))^p)
    for p in np.linspace(0.2, 5.0, 20):
        idx = np.arange(n, dtype=float)
        a = ((n - 1 - idx) / (n - 1)) ** p
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] parametric init p={p:.2f} score={sc:.6f}")

    # a few random seeds as additional starting points
    for _ in range(3):
        a = rng.random(n)
        a /= a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] random init score={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – soft‑max gradient descent (Nesterov + diminishing step).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _run_softmax_gd(
            best_seq, n, deadline, start_time,
            step0=0.6, beta=0.9, max_iter=500_000
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] after softmax best={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent on the true max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _run_subgradient_gd(
            best_seq, n, deadline, start_time,
            step0=0.12, beta=0.9, max_iter=300_000, rel_eps=0.008
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] after sub‑grad best={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (dominant remaining budget).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, sc_ref = _run_annealing(
            best_seq, n,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.5, end_temp=1e-7,
            max_steps=2_000_000

        if sc_ref < best_score:

    # ------------------------------------------------------------------
    # Normalise finally (tiny numerical drift may have occurred).
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:

    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # Return a plain Python list – the harness will call ``evaluate_sequence``.
    return [float(x) for x in best_seq]
