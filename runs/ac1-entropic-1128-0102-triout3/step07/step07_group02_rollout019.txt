"""
Propose a step‑function (non‑negative vector on the simplex) that minimizes
the provided ``evaluate_sequence`` function.

Algorithm – three phases
------------------------
1. **Exploratory warm‑start** – several short runs of a smooth soft‑max
   gradient descent on a modest size (≈2k).  The best among the random
   Dirichlet starts is kept and up‑sampled to the final size.
2. **Heavy sub‑gradient optimisation** – on the full length (≈2·10⁴) a
   projected sub‑gradient method is run for the bulk of the budget.
   Instead of using the single maximal convolution entry we use the
   average sub‑gradient of the *top‑K* entries (K≈5) which gives a stronger
   descent direction.
3. **Peak‑shaving local search** – a fast Monte‑Carlo move that shifts a
   small amount of probability mass from indices that contribute most to
   the current maximal convolution entry to indices that contribute very
   little.  The Metropolis rule with a slowly decreasing temperature
   allows occasional uphill steps.

All operations are performed with NumPy and FFT‑based convolutions; the
simplex projection is exact and O(n log n) per iteration.  The routine
adapts its time allocation to the 1000 s budget, keeps the best sequence
seen and returns it as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (full) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    # size of FFT – next power of two for speed
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Convolution of reversed vector with a weight vector (used in soft‑max)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """∇_a  Σ_k softmax(conv_k / T) * conv_k ."""
    # stability shift
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                # soft‑max weights

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)    # length = n + (2n‑1) – 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]   # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Phase‑A: smooth soft‑max projected gradient descent
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 1.0,
                    beta: float = 0.8,
                    T_start: float = 5e-2,
                    T_end: float = 5e-5,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n} "
                  f"it={it} score={cur_score:.6f}")

        # temperature diminishes exponentially
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it + 1)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑B: sub‑gradient descent on the *exact* max‑convolution.
# Several top entries are used to obtain a richer descent direction.
# ----------------------------------------------------------------------
def _run_subgrad_multi_gd(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.6,
                         beta: float = 0.95,
                         top_k: int = 5,
                         max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the average gradient of the top‑K
    convolution entries (exact max, not soft‑max)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n} "
                  f"it={it} score={cur_score:.6f}")

        # indices of the top‑k convolution values
        if conv.size <= top_k:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        grad = np.zeros_like(a)
        # accumulate gradients of the selected entries
        for idx in idxs:
            i_arr = np.arange(n)
            j_arr = idx - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            grad[mask] += 2.0 * a[j_arr[mask]]
        grad /= float(len(idxs))

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n`` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Targeted local search – “peak shaving”
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Move a small amount of probability mass from high‑contributing
    indices (w.r.t. the current maximal convolution entry) to low‑contributing
    ones.  A simple Metropolis rule with a cooling temperature is used."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor proportional to its contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # choose a low‑contribution recipient
        low_thr = np.percentile(contrib, 10)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.35
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature schedule
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n} "
                      f"it={it} score={new_score:.6f}")

    return best_a, best_score

    except Exception:
        pass

    # random restarts – each run gets at most ~4 s (enough for a few thousand iters)
    while time.time() < small_deadline:
        if best_small is None:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()
        else:
            a0 = best_small.copy()
            # inject a small random perturbation to avoid being stuck
            noise = np.random.randn(n_small) * 0.02
            a0 = _project_simplex(a0 + noise)

        # give each run a tiny personal deadline (≈4 s)
        run_deadline = time.time() + 4.0
        a_opt, score_opt = _run_softmax_gd(
            a0, n_small, run_deadline, start_time,
            step0=1.0, beta=0.8,
            T_start=5e-2, T_end=5e-5
        )
        if score_opt < best_small_score:
            best_small_score = score_opt
            best_small = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] new best small n={n_small} "
                  f"score={score_opt:.6f}")

    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Upsample to the final size
    # ------------------------------------------------------------------
    a_big = _upsample(best_small, n_target)

    # ------------------------------------------------------------------
    # Phase‑B – intensive sub‑gradient descent on the full size.
    # Allocate the bulk of the remaining budget here.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # leave ~30 s for the final local search
    subgrad_deadline = time.time() + max(0.0, remaining - 30.0)

    a_opt, score_opt = _run_subgrad_multi_gd(
        a_big, n_target, subgrad_deadline, start_time,
        step0=0.6, beta=0.95, top_k=5
    )
    best_seq = a_opt.copy()
    best_score = score_opt
    print(f"[{time.time() - start_time:6.2f}s] subgrad final n={n_target} "
          f"score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase‑C – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_target, deadline, start_time
        )
        if score_loc < best_score:
            best_seq = a_loc.copy()
            best_score = score_loc
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may appear because of rounding)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list – the harness will call evaluate_sequence
    return [float(x) for x in best_seq]
