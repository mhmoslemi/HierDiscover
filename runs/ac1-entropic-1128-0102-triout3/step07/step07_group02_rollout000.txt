```python
def _gradient_pnorm(a, p):
    conv = _conv_self(a)
    # compute S = sum(conv**p)
    S = np.sum(conv**p)
    if S == 0:
         return np.zeros_like(a)
    w = (conv**(p-1)) * (S**(1.0/p - 1.0))
    rev_a = a[::-1]
    # conv length 2n-1
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n-1:n-1+n]   # centre part length n
    return grad
```

   experiments is up‑sampled to a longer length (≈19000) and the same
   Lp‑gradient descent is run with a higher final ``p`` and tighter
   step‑size schedule.

4. **Targeted local search** – a simulated‑annealing “peak shaving”
   routine moves a small amount of mass from indices that contribute most
   to the current maximal convolution entry to low‑contribution indices.
   Accepts occasional uphill moves to escape shallow minima.

The whole routine respects the 1000 s budget; the time is split roughly
into 20 % random initial searches, 60 % large‑scale refinement and
20 % final local polishing.  The best sequence seen (normalised to sum = 1)
is returned as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: linear up‑sampling preserving total mass (probability simplex)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the Lp‑norm of the convolution (smooth max approximation)
# ----------------------------------------------------------------------
def _gradient_pnorm(a: np.ndarray, p: float) -> np.ndarray:
    """
    Compute ∇_a ( (∑_k (a*a)[k]^p)^{1/p} ).
    Returned gradient has the same shape as `a`.
    """
    conv = _conv_self(a)                     # length 2n‑1
    # compute S = Σ conv_k^p
    # guard against under‑flow / overflow
    conv_p = conv ** p
    S = np.sum(conv_p)
    if S == 0.0:
        return np.zeros_like(a)

    # weight for each convolution entry
    weight = (conv ** (p - 1.0)) * (S ** (1.0 / p - 1.0))

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, weight)   # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]        # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected gradient descent on the Lp‑norm of the convolution
# ----------------------------------------------------------------------
def _run_pnorm_gd(a0: np.ndarray,
                  n: int,
                  p_start: float,
                  p_end: float,
                  deadline: float,
                  start_time: float,
                  step0: float = 0.6,
                  beta: float = 0.9,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov‑style) gradient descent on the Lp‑norm smoothed
    objective, with p increasing from `p_start` to `p_end` as time
    progresses.
    Returns the best sequence seen and its true evaluate_sequence score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    # true objective (since sum(a)=1): 2·n·max(conv)
    best_score = 2.0 * n * np.max(_conv_self(a))

    it = 0
    stage_start = time.time()

    while time.time() < deadline and it < max_iter:
        it += 1

        # schedule p
        elapsed_frac = (time.time() - stage_start) / (deadline - stage_start + 1e-12)
        p = p_start * (p_end / p_start) ** elapsed_frac

        conv = _conv_self(a)
        cur_score = 2.0 * n * np.max(conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] pnorm improve: n={n}, it={it}, "
                  f"p={p:.2f}, score={cur_score:.6f}")

        # gradient of the smoothed max
        grad = _gradient_pnorm(a, p)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak shaving) – same as the previous implementation
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of probability mass away from indices that
    contribute most to the current maximal convolution entry.
    Accepts occasional uphill moves (simulated‑annealing style).
    Returns the best sequence seen and its true score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # pick donor proportionally to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # pick receiver among low contributors
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation helpers – generate a variety of starting distributions
# ----------------------------------------------------------------------
def _init_random(n: int) -> np.ndarray:
    v = np.random.rand(n) + 1e-6
    return v / v.sum()


def _init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _init_triangular(n: int) -> np.ndarray:
    # symmetric triangle: increase then decrease
    i = np.arange(n)
    a = np.minimum(i + 1, n - i)
    return a / a.sum()


def _init_cosine(n: int) -> np.ndarray:
    # cosine‑squared window, zero at the ends
    i = np.arange(n)
    a = np.cos(np.pi * (i - (n - 1) / 2) / (n - 1)) ** 2
    a = np.maximum(a, 0)
    return a / a.sum()


def _init_gaussian(n: int) -> np.ndarray:
    i = np.arange(n)
    sigma = 0.15   # controls spread
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
    return a / a.sum()


def _init_edge_heavy(n: int) -> np.ndarray:
    # 1 / sqrt(i * (n-i)) shape – more mass near the boundaries
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i) + 1e-12)
    return a / a.sum()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Combines diverse initialisations, Lp‑norm gradient descent, up‑sampling,
    and a final targeted local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start (previous best construction, if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Phase 0 – small‑scale exploratory searches
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # choose a modest length for rapid experiments
    small_n = 8000
    # allocate ~20 % of the total budget for this phase
    phase0_deadline = start_time + (budget_s * 0.20)

    init_funcs = [_init_random, _init_uniform, _init_triangular,
                  _init_cosine, _init_gaussian, _init_edge_heavy]

    # If a warm start exists, also add a blended initialisation
    def _init_blended(n):
        if starter is None:
            return _init_random(n)
        up = _upsample(starter, n)
        rand = _init_random(n)
        blend = 0.5 * up + 0.5 * rand
        return blend / blend.sum()
    init_funcs.append(_init_blended)

    for init_f in init_funcs:
        if time.time() >= phase0_deadline:
            break
        a0 = init_f(small_n)
        # Run a short Lp‑norm descent (p: 4 → 12)
        a_opt, score_opt = _run_pnorm_gd(
            a0,
            small_n,
            p_start=4.0,
            p_end=12.0,
            deadline=phase0_deadline,
            start_time=start_time,
            step0=0.8,
            beta=0.9,
            max_iter=10 ** 6,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase0 improve (init={init_f.__name__}): "
                  f"score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – large‑scale refinement on a longer grid
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        large_n = 19200  # near the size used by the reference solution
        a0 = _upsample(best_seq, large_n)

        # give ~60 % of the remaining time to this phase
        remaining = deadline - time.time()
        phase1_deadline = time.time() + remaining * 0.60

        a_opt, score_opt = _run_pnorm_gd(
            a0,
            large_n,
            p_start=8.0,
            p_end=20.0,
            deadline=phase1_deadline,
            start_time=start_time,
            step0=0.5,
            beta=0.9,
            max_iter=10 ** 7,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase1 improve: n={large_n}, score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_loc, score_loc = _run_targeted_local_search(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (protect against drift)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list (the harness expects this)
    return [float(x) for x in best_seq]
