import time, math, numpy as np
from typing import List, Tuple

try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _subgrad_max(a: np.ndarray, active_mask: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] using the active mask (binary vector of length 2n‑1)."""
    n = a.shape[0]
    K = active_mask.shape[0]
    w_rev = active_mask[::-1]
    L = _next_pow2(n + K - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wf, n=L)[: n + K - 1]
    start = K - 1
    g = 2.0 * corr[start:start + n]
    return g

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up = up / total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _compute_score(a: np.ndarray) -> float:
    """Score = 2*n*max(conv). Assumes sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()

def _run_adam(a0: np.ndarray, deadline: float, start_time: float,
              alpha_schedule=None, iters_per_alpha=5000) -> Tuple[np.ndarray, float]:
    """Smooth Adam optimisation on soft‑max parametrisation."""
    if alpha_schedule is None:
        alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    n = a0.shape[0]
    # work in log‑domain
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))
    fft_len = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t_iter = 0

    best_a = a0.copy()
    best_score = _compute_score(best_a)

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            # Soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # Smoothed objective using log‑sum‑exp (for logging only)
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            # f_val = lse / alpha
            # obj = 2.0 * n * f_val

            # Gradient of smoothed max
            # soft‑max weights over convolution entries
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # gradient wrt a: 2 * (a ⋆ w)
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wf = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wf, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t_iter) / (1.0 - beta1 ** t_iter))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 200 == 0:
                sc = _compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")
    return best_a, best_score

def _run_proj_grad(a0: np.ndarray, deadline: float, start_time: float,
                   beta: float = 0.94, step0: float = 0.25,
                   max_iter: int = 10**7) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (allow small tolerance)
        tol = 1e-12
        active = (conv >= cur_max * (1.0 - tol)).astype(float)  # binary mask

        # sub‑gradient
        g = _subgrad_max(a, active)

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)
    return best_a, best_score

def _run_annealing(a0: np.ndarray, deadline: float, start_time: float,
                   start_temp: float = 5e-2, end_temp: float = 1e-6,
                   max_steps: int = 400_000) -> Tuple[np.ndarray, float]:
    """Randomised mass‑transfer annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # linear cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # source chosen proportionally to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")
    return best_seq, best_score

def _lp_refine(a: np.ndarray, t: float = 0.5) -> np.ndarray:
    """One LP refinement step (if SciPy available). Returns new a or original if failed."""
    if not SCIPY_AVAILABLE:
        return a
    n = a.shape[0]
    K = 2 * n - 1
    M = _conv_self(a).max()
    # build dense matrix A of shape (K, n) where A[k, j] = a[k - j] if valid else 0
    A = np.zeros((K, n), dtype=float)
    for j in range(n):
        A[j:j+n, j] = a
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(K)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print("LP solve failed:", e)
        return a
    if not res.success:
        # LP didn't converge; return unchanged
        return a
    b = res.x
    b_sum = b.sum()
    if b_sum <= 0:
        return a
    b_norm = b / b_sum
    a_new = (1.0 - t) * a + t * b_norm
    # ensure still feasible after convex combination (optional projection)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The driver runs a multi‑stage optimisation:
        1. Adam on a soft‑max parametrisation (small problem size).
        2. Projected sub‑gradient descent on the exact max‑convolution objective.
        3. Optional linear‑programming refinement (one or two LP solves).
        4. Upscaling to the final length and final annealing.
    Time is split automatically between the stages.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Small problem size for the heavy stages
    # --------------------------------------------------------------
    n_small = 2000
    n_big = 4800   # final length (same as reference)

    # initialise a small vector – a smooth random start
    a_small = np.random.rand(n_small)
    a_small = np.maximum(a_small, 0.0)
    a_small /= a_small.sum()

    # --------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate
    # --------------------------------------------------------------
    # allocate roughly 40 % of the budget
    phase1_deadline = time.time() + 0.40 * (deadline - time.time())
    a_small, score_small = _run_adam(a_small, phase1_deadline, start_time,
                                     alpha_schedule=[0.5,1.0,2.0,5.0,10.0,20.0,40.0,80.0],
                                     iters_per_alpha=6000)

    # --------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.30 * (deadline - time.time())
        a_small, score_small = _run_proj_grad(a_small, phase2_deadline, start_time,
                                              beta=0.94, step0=0.25,
                                              max_iter=10**6)

    # --------------------------------------------------------------
    # Phase 3 – optional LP refinement (a few iterations)
    # --------------------------------------------------------------
    if SCIPY_AVAILABLE and time.time() < deadline:
        # we attempt up to three LP refinements, each using a fraction of remaining time
        for i in range(3):
            remaining = deadline - time.time()
            if remaining < 2.0:  # not enough time for another LP
                break
            a_small = _lp_refine(a_small, t=0.5)
            # after refinement we recompute score for logging
            score_small = _compute_score(a_small)
            print(f"[{time.time() - start_time:6.2f}s] LP refinement {i+1}, score={score_small:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Upscale to final length
    # --------------------------------------------------------------
    a_big = _upsample(a_small, n_big)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    # --------------------------------------------------------------
    # Phase 5 – final polishing on the big problem (short PGD)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase5_deadline = time.time() + 0.15 * (deadline - time.time())
        a_big, score_big = _run_proj_grad(a_big, phase5_deadline, start_time,
                                          beta=0.94, step0=0.2,
                                          max_iter=5 * 10**5)

    # --------------------------------------------------------------
    # Phase 6 – simulated annealing on the final vector
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_big, score_big = _run_annealing(a_big, deadline, start_time,
                                          start_temp=5e-2, end_temp=1e-6,
                                          max_steps=500_000)

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total > 0:
        a_big /= total
    else:
        a_big = np.full_like(a_big, 1.0 / a_big.shape[0])

    final_score = _compute_score(a_big)
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in a_big.tolist()]

"""
Optimisation driver for minimise evaluate_sequence.

The search proceeds in three stages:
1. A small‑scale Adam optimiser on a soft‑max parametrisation.
2. Exact projected sub‑gradient descent on the max‑convolution objective.
3. (Optional) a linear‑programming refinement that expands the total mass
   while keeping the current maximum convolution bounded.

The resulting vector is up‑sampled to the final length and a short
final projected‑gradient polishing is performed.  Time is split
automatically between the stages so the total runtime never exceeds
the supplied budget.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# SciPy is optional – the LP refinement is used only if it is available.
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgrad_max(a: np.ndarray, active_mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] using the active mask.
    `active_mask` is a binary vector of length 2·n‑1 indicating the
    indices where the convolution attains its maximal value (within tolerance).
    """
    n = a.shape[0]
    K = active_mask.shape[0]
    w_rev = active_mask[::-1]
    L = _next_pow2(n + K - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wf, n=L)[: n + K - 1]
    start = K - 1
    g = 2.0 * corr[start:start + n]
    return g


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve total mass)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _compute_score(a: np.ndarray) -> float:
    """
    Compute the objective value 2·n·max(conv(a,a)) assuming sum(a)=1.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              deadline: float,
              start_time: float,
              alpha_schedule: List[float] | None = None,
              iters_per_alpha: int = 6000) -> Tuple[np.ndarray, float]:
    """
    Smooth Adam optimisation on a soft‑max parametrisation.
    The objective is a smoothed max of the self‑convolution.
    """
    if alpha_schedule is None:
        alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0,
                          20.0, 40.0, 80.0, 160.0]

    n = a0.shape[0]
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))                # logits
    fft_len = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t_iter = 0

    best_a = a0.copy()
    best_score = _compute_score(best_a)

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution of the current distribution
            conv = _conv_self(a)

            # smooth max via log‑sum‑exp
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()                     # soft‑max weights over convolution entries

            # gradient of the smoothed max (2·(a ⋆ w_rev))
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wf = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wf, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # chain rule to the logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t_iter) / (1.0 - beta1 ** t_iter))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 200 == 0:
                sc = _compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                          f"α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

    return best_a, best_score


def _run_proj_grad(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   beta: float = 0.94,
                   step0: float = 0.25,
                   max_iter: int = 10 ** 6) -> Tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _compute_score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # active set – indices where the convolution is (almost) maximal
        tol = 1e-12
        active_mask = (conv >= cur_max * (1.0 - tol)).astype(float)

        # sub‑gradient via FFT (much faster than the double loop)
        g = _subgrad_max(a, active_mask)

        # Nesterov‑style momentum and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   max_steps: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Classical simulated annealing: move a random ε‑fraction of mass
    between two indices.  The schedule is exponential.
    """
    a = a0.copy()
    n = a.shape[0]
    score = _compute_score(a)
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _compute_score(a_prop)
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


def _lp_refine(a: np.ndarray, t: float = 0.5) -> np.ndarray:
    """
    One linear‑programming refinement step.
    Maximise Σ b_j subject to a * b ≤ M (component‑wise) and b ≥ 0.
    The solution b is then blended with the current a.
    """
    if not _SCIPY_AVAILABLE:
        return a

    n = a.shape[0]
    K = 2 * n - 1
    M = _conv_self(a).max()

    # Build the constraint matrix A (size K × n) where A[k, j] = a[k‑j] if valid.
    A = np.zeros((K, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a

    c = -np.ones(n)                     # maximise Σ b_j  →  minimise –Σ b_j
    bounds = [(0, None)] * n
    b_ub = M * np.ones(K)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:  # pragma: no cover
        return a

    if not res.success:
        return a

    b = res.x
    b_sum = b.sum()
    if b_sum <= 0:
        return a

    b_norm = b / b_sum
    a_new = (1.0 - t) * a + t * b_norm
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The driver runs a multi‑stage optimisation:
        1. Adam on a smooth surrogate (small problem size).
        2. Projected sub‑gradient descent on the exact max‑convolution.
        3. Optional LP refinement.
        4. Upscaling to the final length and a short final polishing.
    Time is split automatically between the stages.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Small problem size for the heavy stages
    # --------------------------------------------------------------
    n_small = 2000          # small enough for LP, large enough for a good shape
    n_big = 4800            # final length (the reference size)

    # Start from a smooth random distribution
    a_small = np.random.rand(n_small)
    a_small = np.maximum(a_small, 0.0)
    a_small /= a_small.sum()

    # --------------------------------------------------------------
    # Phase 1 – Adam on the smoothed max‑convolution
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.45 * (deadline - time.time())
    a_small, _ = _run_adam(a_small, phase1_deadline, start_time,
                           alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0,
                                           20.0, 40.0, 80.0, 160.0],
                           iters_per_alpha=6000)

    # --------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + 0.30 * (deadline - time.time())
        a_small, _ = _run_proj_grad(a_small, phase2_deadline, start_time,
                                    beta=0.94, step0=0.25, max_iter=10 ** 6)

    # --------------------------------------------------------------
    # Phase 3 – optional LP refinement (a few iterations)
    # --------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline:
        for it in range(3):
            remaining = deadline - time.time()
            if remaining < 2.0:
                break
            a_small = _lp_refine(a_small, t=0.5)
            cur_score = _compute_score(a_small)
            print(f"[{time.time() - start_time:6.2f}s] LP refinement {it + 1}: "
                  f"score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – up‑sample to the final length
    # --------------------------------------------------------------
    a_big = _upsample(a_small, n_big)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    # --------------------------------------------------------------
    # Phase 5 – short projected‑gradient polishing on the big vector
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase5_deadline = time.time() + 0.15 * (deadline - time.time())
        a_big, _ = _run_proj_grad(a_big, phase5_deadline, start_time,
                                  beta=0.94, step0=0.20,
                                  max_iter=5 * 10 ** 5)

    # --------------------------------------------------------------
    # Phase 6 – final simulated annealing (lightweight)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_big, _ = _run_annealing(a_big, deadline, start_time,
                                  start_temp=5e-2, end_temp=1e-6,
                                  max_steps=200_000)

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total > 0:
        a_big /= total
    else:
        a_big = np.full_like(a_big, 1.0 / a_big.shape[0])

    final_score = _compute_score(a_big)
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Convert to plain Python floats for the evaluator.
    return [float(x) for x in a_big.tolist()]
