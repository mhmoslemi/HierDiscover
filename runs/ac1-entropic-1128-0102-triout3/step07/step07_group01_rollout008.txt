def _beta_shape(n, alpha, beta):
    # generate symmetric Beta distribution on [0,1] discretized at n points.
    # Use positions = (i+0.5)/n.
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # In case alpha or beta < 1 we may have inf at edges; replace inf by large finite.
    # but if alpha,beta <= 0, the PDF diverges; we cap.
    a[np.isinf(a)] = np.max(a[~np.isinf(a)]) if np.any(~np.isinf(a)) else 0.0
    total = a.sum()
    if total == 0 or np.isnan(total):
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a

def _run_simulated_annealing(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    it = 0
    # initial temperature set as a fraction of current max
    T0 = cur_max * 0.1 + 1e-9
    while time.time() < deadline:
        it += 1
        # choose source index weighted by a or by contributions to current max (like earlier)
        src = np.random.choice(n, p=a/a.sum())
        # avoid picking index with zero mass
        if a[src] < 1e-15:
            continue
        # destination randomly from others
        dst = np.random.randint(n-1)
        if dst >= src:
            dst += 1
        # move fraction eps
        eps = a[src] * np.random.random() * 0.02  # up to 2%
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        delta = new_max - cur_max
        # temperature linearly decays
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / (deadline - start_time))
        accept = delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12))
        if accept:

"""
Hybrid optimisation for the “minimise max self‑convolution” problem.

The search proceeds in three stages:

1️⃣  A cheap parametric sweep (U‑shaped power law, exponential,
    uniform and a few random Dirichlet draws) gives a good “shape”.
2️⃣  From the best shape we run a projected sub‑gradient descent
    on the exact max‑convolution (the same routine used in the
    reference implementation).  The loop stops automatically when
    the time budget is exhausted.
3️⃣  A simulated‑annealing refinement moves a tiny amount of probability
    mass away from the indices that contribute most to the current
    maximal convolution entry.  Up‑hill moves are occasionally accepted
    (temperature decays linearly) which helps escape shallow local minima.

All operations respect the simplex constraints (non‑negative,
unit‑sum) by Euclidean projection.  Helper functions are written at
top‑level, no closures or lambdas are used and the total run‑time is
kept below the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for the cheap sweep
# ----------------------------------------------------------------------
def _sym_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution a_i ∝ ((i+½)(n‑½‑i))^p (p can be negative)."""
    i = np.arange(n, dtype=float)
    prod = (i + 0.5) * (n - 0.5 - i)               # >0 for all i
    a = np.power(prod, p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0 or np.isnan(s):
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _exp_shape(n: int, q: float) -> np.ndarray:
    """Exponential family a_i ∝ exp(q·(i‑centre))."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(q * (i - centre))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0 or np.isnan(s):
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up (optional)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    """Multiplicative mirror‑descent on a smoothed log‑sum‑exp of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient via log‑sum‑exp
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """Projected sub‑gradient descent on max(self‑convolution)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             eps_frac: float = 0.02) -> np.ndarray:
    """
    Tiny mass moves guided by the current maximal convolution entry.
    Moves that improve the max are always kept; uphill moves are accepted
    with a linearly decaying temperature.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    it = 0

    # initial temperature – 10 % of the current max (empirical)
    T0 = max(cur_max * 0.1, 1e-12)

    while time.time() < deadline:
        it += 1

        # probability of picking a source – proportional to its contribution
        # to the currently maximal entry.
        k_max = int(np.argmax(conv))

        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)
        if a[src] < 1e-14:
            continue

        # destination – any other index, chosen uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        delta = new_max - cur_max

        # linearly decreasing temperature
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / (deadline - start_time))

        accept = (delta < 0) or (rng.random() < math.exp(-delta / (T + 1e-20)))
        if accept:
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={it:,}, max={best_max:.8f}")

        # occasional early termination check
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Fast score (identical to evaluate_sequence for normalised vectors)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(conv)/sum(a)^2 – fast version using FFT."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    The routine respects the given time budget (≈1000 s) and returns the best
    sequence found as a plain Python list.

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # spare a little margin

    # ------------------------------------------------------------------
    # Choose a working length.  If a previous construction is present we
    # reuse its length, otherwise we fall back to 4800 (the length that
    # yielded the published best bound).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = int(starter.size)
        else:
            n = 4800
    except Exception:
        n = 4800

    # ------------------------------------------------------------------
    # --------------------------------------------------- cheap parametric sweep
    # ------------------------------------------------------------------
    def _make_candidate(arr):
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s == 0 or np.isnan(s):
            arr = np.full(n, 1.0 / n)
        else:
            arr = arr / s
        return arr

    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # symmetric power‑law family (U‑shapes)

    # exponential family (peaked / dipped)
    for q in np.linspace(-4.0, 4.0, 30):
        candidates.append(_exp_shape(n, q))

    # a few random Dirichlet draws (broad exploration)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(10):
        candidates.append(rng.dirichlet(np.ones(n)))

    except Exception:
        pass

    # evaluate all candidates (stop early if time is low)
    best_seq = None
    best_score = float("inf")
    for a in candidates:
        if time.time() > deadline - 120:      # reserve ~2 min for the heavy phases
            break
        a = _make_candidate(a)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] parametric new best: score={best_score:.6f}")

    # ------------------------------------------------------------------

            best_score = score_warm

    # ------------------------------------------------------------------
