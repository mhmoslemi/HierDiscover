"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Parameterisation** – the n‑vector is stored in the log‑domain
   (soft‑max).  This guarantees non‑negativity and unit ℓ₁‑norm
   at every iteration.

2. **Smooth Adam optimisation** – we minimise a smooth approximation of
   the max‑convolution:
        fα(a) = (1/α)·log Σ_k exp(α·(a*a)[k]).
   A schedule of increasing smoothing parameters α is used, ending at a
   large value (α≈640) so that fα≈max(a*a).  Adam updates (with bias
   correction) drive the solution quickly into a low‑peak region.
   The schedule, the per‑α iteration count and the learning‑rate are
   chosen so that the whole Adam phase occupies roughly 70 % of the
   time budget.

3. **Exact projected sub‑gradient descent** – after the Adam phase we
   polish the solution on the true objective
        g(a)=max_k (a*a)[k] .
   The sub‑gradient is obtained from the set of “active’’ indices where
   the convolution is (almost) maximal.  A simple Euclidean projection
   onto the probability simplex keeps the iterate feasible.

4. **Greedy peak reduction** – a cheap deterministic loop moves a
   fraction of the coefficient that contributes most to the current
   convolution peak towards the smallest coefficient, keeping only
   improving moves.

5. **Final simulated‑annealing refinement** – a short annealing routine
   performs random mass‑transfer moves (with an exponential cooling
   schedule) to escape possible plateaus.

Multiple random restarts are tried (uniform, exponential, triangular,
random Dirichlet and the previous best construction if available); the
best sequence found overall is returned.  All steps respect the shared
time budget (default 1000 s).  The code uses only NumPy and SciPy (via
``scipy.optimize.linprog`` is *not* required) and runs well within the
single‑CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true (max) objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for   f(a)=max_k (a*a)[k] .
    Returns the best vector seen (as a probability vector) and its
    corresponding score 2·n·max_conv .
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s]  proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient (active set)
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # ordered pairs (diagonal counted twice – harmless)

        # momentum + step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy loop that moves a fraction of the coefficient that contributes
    most to the current convolution peak towards the smallest coefficient.
    The move is kept only if it strictly lowers the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx

        contrib = a[i_idx] * a[j_idx]                 # ordered contribution
        src = i_idx[contrib.argmax()]                # source of most mass

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s]  peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Classic annealing: move a random ε‑fraction of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s]  anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that
    minimises ``evaluate_sequence``.  Uses a mix of smooth Adam
    optimisation, exact sub‑gradient polishing, deterministic peak
    reduction and a final annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressive power and speed
    n = 3000

    # ------------------------------------------------------------------
    # Build a collection of diverse starting points.
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best construction (if available)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.shape[0] != n:
                starter = _upsample(starter, n)
            starts.append(starter)
    except Exception:
        pass

    # 2) uniform
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decay
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # 4) triangular
    if n % 2 == 0:
        half = n // 2
        tri = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
    else:
        half = n // 2
        tri = np.concatenate([np.arange(1, half + 2), np.arange(half + 1, 0, -1)])
    starts.append(tri / tri.sum())

    # 5) random Dirichlet (moderately concentrated)
    rng = np.random.default_rng()
    dirichlet = rng.gamma(shape=0.5, scale=1.0, size=n)
    starts.append(dirichlet / dirichlet.sum())

    # ------------------------------------------------------------------
    # Keep track of the overall best candidate.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Process each start independently, sharing the global time budget.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s]  start #{idx + 1}")

        # --------------------------------------------------------------
        # Stage 1 – smooth Adam optimisation (log‑soft‑max parametrisation)
        # --------------------------------------------------------------
        eps = 1e-16
        y = np.log(np.maximum(a0, eps))

        fft_len = _next_pow2(3 * n - 2)

        # smoothing schedule – we finish with a very sharp α
        alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                         40.0, 80.0, 160.0, 320.0, 640.0]
        # allocate roughly 70 % of the remaining budget to this phase
        adam_deadline = start_time + 0.70 * (deadline - start_time)

        # Adam hyper‑parameters
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr0 = 0.12
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0

        # per‑α iteration budget – will be trimmed by the time check
        iters_per_alpha = 12_000

        # keep the best vector seen during Adam
        local_best_seq = a0.copy()
        local_best_score = float(evaluate_sequence(local_best_seq.tolist()))

        for alpha in alpha_schedule:
            for i in range(iters_per_alpha):
                if time.time() >= adam_deadline:
                    break

                # soft‑max → probability vector
                maxy = np.max(y)
                exp_y = np.exp(y - maxy)
                a = exp_y / exp_y.sum()

                # smoothed objective (log‑sum‑exp over the convolution)
                conv = _conv_self(a)
                scaled = alpha * conv
                max_scaled = scaled.max()
                lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
                f_val = lse / alpha
                # (the factor 2·n is only needed for reporting)
                obj = 2.0 * n * f_val

                # gradient w.r.t. a
                grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

                # chain rule to logits
                grad_y = a * (grad_a - np.dot(grad_a, a))

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_y
                v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                lr_t = lr0 * (math.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
                y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

                # occasional exact evaluation

                    sc = evaluate_sequence(a.tolist())
                    if sc < local_best_score:
                        local_best_score = sc
                        local_best_seq = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s]  Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

            if time.time() >= adam_deadline:

        # --------------------------------------------------------------
        # Stage 2 – exact projected sub‑gradient descent on true max‑conv
        # --------------------------------------------------------------
        if time.time() < deadline:

            polish_deadline = time.time() + 0.20 * remaining   # 20 % of what is left
            a_polish, score_polish = _run_gradient_descent(
                local_best_seq,
                polish_deadline,
                start_time,
                beta=0.94,

            )
            if score_polish < local_best_score:

                local_best_seq = a_polish.copy()
                print(f"[{time.time() - start_time:6.2f}s]  polishing improve: score={score_polish:.6f}")

        # --------------------------------------------------------------
        # Stage 3 – deterministic greedy peak‑reduction
        # --------------------------------------------------------------
        if time.time() < deadline:
            peak_deadline = time.time() + 0.07 * (deadline - time.time())
            a_peak, score_peak = _run_peak_reduction(

            )
            if score_peak < local_best_score:
                local_best_score = score_peak
                local_best_seq = a_peak.copy()
                print(f"[{time.time() - start_time:6.2f}s]  peak‑reduce improve: score={score_peak:.6f}")

        # --------------------------------------------------------------
        # Stage 4 – final simulated annealing
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _run_annealing(
                local_best_seq,
                deadline,
                start_time,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=300_000,
            )
            if score_final < local_best_score:
                local_best_score = score_final
                local_best_seq = a_final.copy()
                print(f"[{time.time() - start_time:6.2f}s]  final anneal improve: score={score_final:.6f}")

        # --------------------------------------------------------------
        # Update the global best if this start beats it.
        # --------------------------------------------------------------
        if local_best_score < best_score:

            best_seq = local_best_seq.copy()

    return [float(x) for x in best_seq.tolist()]
