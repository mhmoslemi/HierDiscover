"""
Improved step‑function optimisation.

The optimiser works in three phases (with optional restarts):
1.  A short exponentiated‑gradient warm‑up on a smooth approximation of the
    max‑convolution.
2.  A long projected Adam‑style sub‑gradient descent on the exact max‑convolution.
3.  An aggressive hill‑climbing refinement that repeatedly moves a tiny amount
    of mass away from indices that contribute most to the current maximal
    autocorrelation entry.  The convolution is updated incrementally (O(n))
    so millions of cheap moves can be performed.

If time permits a secondary random restart is launched to escape possible
local minima.  The routine respects a 1000 s budget and returns the best
non‑negative sequence (normalised to unit sum) it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Uses all entries whose value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # length sufficient for correlation
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror‑descent on simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Phase‑2 : Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.5,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    """
    Adam‑like update on the exact sub‑gradient of max(conv).
    Each iteration projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step (we *subtract* because we minimise)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score

# Phase‑4 : fast targeted refinement (incremental convolution update)

def _update_conv_move(conv: np.ndarray, a: np.ndarray,
                     src: int, dst: int, eps: float) -> np.ndarray:
    """
    Return a new convolution vector after moving `eps` mass from `src` to `dst`.
    Updates are performed in O(n) using slice arithmetic.
    """
    n = a.shape[0]
    e_src = -eps
    e_dst = eps
    conv_new = conv.copy()

    # linear part: 2 * (e_src * a) shifted by src, similarly for dst
    conv_new[src:src + n] += 2.0 * e_src * a
    conv_new[dst:dst + n] += 2.0 * e_dst * a

    # quadratic part from the moved mass itself
    max_idx = 2 * n - 1
    if 2 * src < max_idx:
        conv_new[2 * src] += e_src * e_src
    if src + dst < max_idx:
        conv_new[src + dst] += 2.0 * e_src * e_dst
    if 2 * dst < max_idx:
        conv_new[2 * dst] += e_dst * e_dst

    return conv_new


def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10 ** 7,
                         eps_frac: float = 0.025) -> np.ndarray:
    """
    Hill‑climbing that repeatedly moves a tiny fraction of probability mass
    away from indices that dominate the current maximal autocorrelation entry.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # contributions of each index to this maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()  # fallback

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # destination – any other index (uniform)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # amount of mass to move (tiny fraction of src)
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # compute the tentative new convolution efficiently
        conv_prop = _update_conv_move(conv, a, src, dst, eps)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:          # enforce strict improvement
            # accept move
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional early exit when time is exhausted
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a

        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2: Adam‑style projected sub‑gradient descent (majority of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep about 70 % of the remaining budget for this phase
        stage2_time = max(30.0, 0.70 * remaining)
        stage2_deadline = time.time() + stage2_time

        a_adam, score_adam = _run_adam_subgradient(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.45,
            beta1=0.9,
            beta2=0.999,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3: targeted refinement (aggressive hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10 ** 7,
            eps_frac=0.018,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Optional second random restart (if still time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # start from a fresh random point, keep most of the remaining budget
        a_restart = np.random.rand(final_n)
        a_restart /= a_restart.sum()
        # a short warm‑up
        stage1_deadline = time.time() + 6.0
        a_warm2, _ = _run_exponential_gradient(
            a_restart,
            final_n,
            stage1_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0],
            eta0=0.3,
            steps_per_alpha=3000,
        )
        # long Adam descent
        stage2_deadline = deadline
        a_adam2, _ = _run_adam_subgradient(
            a_warm2,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.35,
        )
        # refinement
        a_ref2 = _run_targeted_refine(
            a_adam2,
            final_n,
            deadline,
            start_time,
            max_iters=5 * 10 ** 6,
            eps_frac=0.015,
        )
        conv_r = _conv_self(a_ref2)
        score_r = 2.0 * final_n * conv_r.max()
        if score_r < best_score:
            best_score = score_r
            best_seq = a_ref2.copy()
            print(f"[{time.time() - start_time:.2f}s] second‑restart improve: score={score_r:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Convert to plain Python list (as required by evaluate_sequence)
    return [float(x) for x in best_seq]
