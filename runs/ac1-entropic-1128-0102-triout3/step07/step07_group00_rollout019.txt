"""
Search for a simple power‑law step function that gives a very low
evaluate_sequence value.

The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  For a positive
vector we can normalise the sum to 1 without changing the score.
If we choose a decreasing power law

    a_i ∝ (n‑i)ᵖ  ,   i = 0 … n‑1

the convolution can be analysed analytically and the maximal entry is
obtained either in the centre (where the two tails overlap) or very
near the origin (where a₀·a₁ dominates).  By scanning a few integer – and
optionally fractional – exponents p we can locate a shape whose maximal
convolution entry is dramatically smaller than the uniform vector.
The search is tiny (a handful of evaluations) and safely finishes inside
the 1000 s budget.

The function tries a few lengths (powers of two) and a grid of exponents,
keeps the best sequence seen, and finally returns it as a plain Python
list.
"""

import time
import numpy as np

def _make_powerlaw_seq(n: int, p: float) -> np.ndarray:
    """
    Return a non‑negative vector of length n with components
    a[i] ∝ (n-i)^p, normalised so that sum(a)=1.
    """
    # rev = n-1-i, rev+1 = n-i  (always ≥1)
    rev = np.arange(n - 1, -1, -1, dtype=float) + 1.0   # shape (n,)
    # power – works for integer and fractional p
    raw = rev ** p
    # normalise to a probability vector
    total = raw.sum()
    if total == 0.0:
        # fallback to uniform if something goes wrong
        return np.full(n, 1.0 / n, dtype=float)
    return raw / total


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method builds a family of decreasing
    power‑law vectors a_i ∝ (n-i)^p, evaluates a few choices of length n
    and exponent p, and returns the best one found before the time budget
    expires.
    """
    # The harness provides evaluate_sequence – we just call it.
    # -----------------------------------------------------------------
    # time handling
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # -----------------------------------------------------------------
    # Search configuration
    # -----------------------------------------------------------------
    # lengths to try – powers of two are FFT‑friendly (though
    # evaluate_sequence uses np.convolve, the sizes are still modest)
    lengths = [2048, 4096, 8192]

    # coarse grid of exponents (integer) and a finer grid around the best
    coarse_ps = list(range(0, 7))          # 0 … 6
    fine_step = 0.2                       # step size for the refinement pass
    fine_range = 1.0                      # how far on each side of the coarse optimum

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Main loop over lengths and exponents
    # -----------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- coarse search -------------------------------------------------
        for p in coarse_ps:
            if time.time() >= deadline:
                break

            seq = _make_powerlaw_seq(n, float(p))
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:               # safety – treat as infinity
                score = float("inf")

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n}, p={p:.1f} → {score:.6f}")

        # ----- refinement around the best exponent for this length ---------
        if best_seq is None:
            continue
        # centre the refinement around the best p seen for this n
        # (store the best p from the coarse pass)
        # Find the p that gave best_score for this n
        # (re‑run the coarse loop but keep the best p)
        best_p_for_n = None
        best_score_n = float("inf")
        for p in coarse_ps:
            seq = _make_powerlaw_seq(n, float(p))
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                score = float("inf")
            if score < best_score_n:
                best_score_n = score
                best_p_for_n = float(p)

        if best_p_for_n is None:
            continue

        # fine sweep
        p_start = max(0.0, best_p_for_n - fine_range)
        p_end = best_p_for_n + fine_range
        p_vals = np.arange(p_start, p_end + fine_step, fine_step)

        for p in p_vals:
            if time.time() >= deadline:
                break
            seq = _make_powerlaw_seq(n, p)
            try:
                score = evaluate_sequence(seq.tolist())
            except Exception:
                score = float("inf")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] fine n={n}, p={p:.2f} → {score:.6f}")

    # -----------------------------------------------------------------

    if best_seq is None:
