def _incremental_anneal(a, deadline, start_time):
    n = a.shape[0]; conv = np.convolve(a,a); score=2*n*conv.max()
    best_a = a.copy(); best_score = score
    rng = np.random.default_rng()
    start_temp = 0.05
    end_temp = 1e-5
    steps = 0
    while time.time() < deadline:
        steps += 1
        # schedule temperature based on steps
        T = start_temp * (end_temp/start_temp)**(steps / 1e6) # scaling lines
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12: continue
        eps = a[src] * rng.random() * 0.3
        # update convolution exactly
        # conv_new = conv
        # convolution of a with itself:
        # conv_k = sum_i a_i a_{k-i}
        # When moving epsilon from src to dst:
        # a[src] -> a[src]-eps, a[dst] -> a[dst]+eps
        # The change in conv can be computed:
        # For each i, contributions involving src or dst change.
        # For pair (src, i): conv_{src+i} changes by -2*eps*a[i] (since two ordered contributions)
        # For pair (dst, i): conv_{dst+i} changes by +2*eps*a[i]
        # For self terms: conv_{2*src} changes by -eps*eps, conv_{2*dst} changes by +eps*eps
        # conv_{src+dst} changes by -2*eps*eps (since both src and dst contributions change? Actually both pairs (src,dst) and (dst,src) contributions changed.)
        # We'll implement same as previous code.
        conv[src:src+n] -= 2.0 * eps * a
        conv[dst:dst+n] += 2.0 * eps * a
        conv[2*src] += eps*eps
        conv[2*dst] += eps*eps
        conv[src+dst] -= 2.0 * eps * eps
        a[src] -= eps
        a[dst] += eps
        new_score = 2.0 * n * conv.max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            # reject
            a[src] += eps
            a[dst] -= eps
            conv[src:src+n] += 2.0 * eps * a
            conv[dst:dst+n] -= 2.0 * eps * a
            conv[2*src] -= eps*eps
            conv[2*dst] -= eps*eps
            conv[src+dst] += 2.0 * eps*eps

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

The algorithm consists of three independent stages that respect a 1000 s
time budget:

1. **Linear‑programming “mass‑inflation”** –  
   Given the current distribution a (∑a = 1) we compute M = max(a∗a).
   We then solve the LP  

        maximise   1ᵀ g
        subject to (a ∗ g)_k ≤ M   for all k = 0 … 2n‑2
                   g ≥ 0

   The matrix (a∗g) is a Toeplitz matrix and can be built with a single
   call to ``scipy.sparse.diags``.  The LP pushes as much mass as
   possible into a new vector g without increasing the worst convolution
   entry.  After normalising g (∑g = 1) we replace a by a convex
   combination  a←(1−t)·a + t·g, slowly decreasing the step size t.
   Repeating this step quickly drives the maximum convolution down.

2. **Upsampling** – once the LP phase has stabilised we linearly
   interpolate the best small‑size vector (n≈1200) to the target length
   n=4800.  Interpolation preserves the ℓ₁‑norm.

3. **Fast incremental simulated‑annealing** – the upsampled vector is
   refined by a specialised annealing loop.  In each move a small
   amount ε of mass is transferred from a source index to a destination
   index; the convolution is updated in O(n) by adding the exact changes
   to the affected entries.  Uphill moves are accepted with probability
   ``exp(−Δ/T)`` using a geometric cooling schedule.  The best sequence
   seen throughout the whole run is returned.

All numbers are non‑negative, the sum is normalised to 1 before the
final call to ``evaluate_sequence``.  The method stays well inside the
1000 s wall‑time limit on a single CPU.
"""

import time
import math
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving its sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise sum(g)
        subject to (a * g)_k ≤ M   (k = 0 … 2n‑2)
                   g ≥ 0

    Returns the optimal vector g (not normalised).
    """
    n = a.shape[0]

    # build the Toeplitz constraint matrix:
    # entry (k, j) = a_{k-j}  if 0 ≤ k‑j < n else 0
    offsets = np.arange(n)                     # diagonal offsets 0,…,n‑1
    # each diagonal consists of the same constant a[i]
    data = np.tile(a[:, np.newaxis], (1, n))   # shape (n, n)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    # maximize sum(g) → minimise –sum(g)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # solve with the HiGHS interior‑point solver (fast for moderate sizes)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback: return the current a (no improvement possible)
        return a.copy()
    return res.x


def _incremental_anneal(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        total_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Fast simulated‑annealing that transfers a small ε‑fraction of mass
    between two positions while updating the convolution incrementally.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # choose source proportional to its current mass (helps focus on heavy entries)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score
