def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine final n target: 16384 maybe, but ensure base memory ok.
    n_target = 16384  # may be large but okay.

    # Load previous best if exists
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        if base_seq.ndim == 1 and base_seq.size > 0:
            # ensure sum not zero
            base_seq = base_seq / base_seq.sum()
    else:
        base_seq = None

    # define stages sizes list
    stages = [2048, 4096, 8192, n_target]
    best_seq = None
    best_score = float('inf')
    # iterate through stages
    for n in stages:
        if time.time() >= deadline:
            break

        # upsample from previous best (or random)
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif base_seq is not None and base_seq.size == n:
            a = base_seq.copy()
        elif base_seq is not None:
            a = _upsample(base_seq, n)
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # Adam stage
        conv = _conv_self(a)
        tau0 = max(conv.max() * 10.0, 1e-8)
        # allocate time slice proportionally: maybe longer for later stages
        remaining = deadline - time.time()
        # allocate about 0.4 of remaining for Adam in each stage
        slice_alloc = min(remaining * 0.4, max(30.0, 0.12 * remaining))
        slice_deadline = time.time() + slice_alloc

        a, score_adam = _run_adam_softmax(a, n, slice_deadline, start_time,
                                          tau0=tau0,
                                          tau_end=1e-13,
                                          tau_decay=0.9995,
                                          step_size=0.3)

        # evaluate true objective
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage {n} Adam improve: score={cur_score:.6f}")

        # Subgradient stage: allocate remaining time for this stage (maybe ~0.4)
        if time.time() < deadline:
            sub_deadline = deadline - 2.0  # keep 2 secs for final fine.
            a_opt, score_sub = _run_subgradient_descent(best_seq, n, sub_deadline, start_time,
                                                        beta=0.95,
                                                        step0=0.2)
            cur_score = 2.0 * n * _conv_self(a_opt).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage {n} subgrad improve: score={cur_score:.6f}")

            # Annealing: allocate some time based on remaining
            if time.time() < deadline:
                a_ref, score_ref = _run_annealing(best_seq, n, deadline,
                                                  start_time,
                                                  best_score,
                                                  best_seq,
                                                  start_temp=2e-2,
                                                  end_temp=1e-7,
                                                  steps=500_000)
                cur_score = 2.0 * n * _conv_self(a_ref).max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_ref.copy()
                    print(f"[{time.time() - start_time:6.2f}s] stage {n} anneal improve: score={cur_score:.6f}")

        # proceed to next stage

    # after all stages, ensure sum normalized.
    best_seq = np.asarray(best_seq, dtype=float)
    sumval = best_seq.sum()
    if sumval > 0:
        best_seq /= sumval
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]

"""
Multi‑scale optimiser for the max‑self‑convolution objective.

The optimiser works in several phases, all of them respecting the global
time budget (default 1000 s).

1.  Initialise from the best known construction (`height_sequence_1`) if it
    exists, otherwise from a random vector.
2.  Upsample the vector to a series of increasing sizes (2048 → 4096 →
    8192 → 16384).  At each size a short Adam run on a smooth
    log‑sum‑exp surrogate is performed – this gives a good “shape”.
3.  On the final size a long projected sub‑gradient descent (Euclidean
    projection onto the simplex) minimises the true objective
    `2·n·max(conv)`.
4.  A second descent phase uses exponentiated‑gradient updates
    (mirror‑descent with the entropy Bregman divergence).  The two
    descent schemes complement each other and usually improve the bound
    beyond what the original code achieved.
5.  A very light simulated‑annealing mass‑transfer is run for the last
    seconds to escape tiny traps.

All vectors are kept on the probability simplex (non‑negative,
sum = 1), which makes the evaluation function scale‑invariant.
The function returns a plain Python list of floats that can be fed to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT based linear convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for true objective)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑sum‑exp) surrogate of max‑conv
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-13,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Oτ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam and projection onto the simplex.
    Returns the best vector (according to the true objective) and its
    true score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for book‑keeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f} τ={tau:.2e}")

        # ---- soft‑max weights ----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ---- gradient of Oτ ----
        # grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # required slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (a lies on the simplex).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the true objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               step0: float = 0.20,
                               max_iter: int = 5 * 10 ** 6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the entropy Bregman divergence.
    Update rule:
        a ← a ⊙ exp(−η·grad)   (⊙ element‑wise product)
    followed by renormalisation onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve (n={n}) "
                  f"iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)               # non‑negative
        eta = step0 / math.sqrt(it)                    # diminishing step
        # multiplicative update; clip to avoid under‑flow
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) "
                  f"iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    # ------------------------------------------------------------------
    size_schedule = [2048, 4096, 8192, 16384]

    best_seq = None
    best_score = float("inf")

    for n in size_schedule:
        if time.time() >= deadline:
            break

        # up‑sample from the best vector seen so far (or from starter)
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # --------------------------------------------------------------
        # Phase 1 – Adam on the smooth surrogate.
        # --------------------------------------------------------------
        conv0 = _conv_self(a)
        tau0 = max(conv0.max() * 10.0, 1e-8)

        # allocate a modest slice of the remaining time (≈30 % of what is left)
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.30, max(20.0, 0.12 * remaining))
        slice_deadline = time.time() + slice_time

        a, _ = _run_adam_softmax(
            a0=a,
            n=n,
            deadline=slice_deadline,
            start_time=start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.30,
        )
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] size {n} Adam improve: "
                  f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Euclidean projected sub‑gradient descent (heavy lifting).
        # --------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = deadline - 2.0      # reserve a couple seconds for later phases
            a, _ = _run_subgradient_descent(
                a0=best_seq,
                n=n,
                deadline=sub_deadline,
                start_time=start_time,
                beta=0.95,
                step0=0.30,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} subgrad improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Exponentiated‑gradient mirror descent.
        # --------------------------------------------------------------
        if time.time() < deadline:
            eg_deadline = deadline - 2.0
            a, _ = _run_exponentiated_gradient(
                a0=best_seq,
                n=n,
                deadline=eg_deadline,
                start_time=start_time,
                step0=0.20,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} EG improve: "
                      f"score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Light simulated annealing for the last few seconds.
        # --------------------------------------------------------------
        if time.time() < deadline:
            a, _ = _run_annealing(
                a0=best_seq,
                n=n,
                deadline=deadline,
                start_time=start_time,
                cur_best_score=best_score,
                cur_best_seq=best_seq,
                start_temp=2e-2,
                end_temp=1e-7,
                steps=300_000,
            )
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] size {n} anneal final improve: "
                      f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
