"""
Propose a step‑function (non‑negative vector) minimizing
``evaluate_sequence``.  The algorithm follows a multi‑stage strategy:

1. **Progressive up‑sampling** – start with a short vector (≈800 entries)
   and repeatedly double its length while staying on the probability simplex.
2. **Smooth (soft‑max) gradient descent** – fast Nesterov GD on a
   temperature‑smoothed version of the max‑convolution.
3. **Exact sub‑gradient descent** – true max‑convolution with an active set.
4. **Linearised‑LP refinement** – at every stage we linearise the quadratic
   constraints on the active convolution entries and solve a tiny LP
   (highs) that directly reduces the current maximum.  The LP is built only
   from the most critical indices, so it is cheap even for a few‑thousand‑
   element vectors.
5. **Extended simulated annealing** – final long annealing walk that moves
   small fractions of mass and occasionally smooths the vector.

All steps respect the global 1000 s wall‑clock budget, constantly checking
the deadline.  The best vector found (normalised to sum = 1) is returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max surrogate and its gradient (smooth approximation of the max)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # numerical stability
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov‑style GD on the soft‑max surrogate."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv) using an active set."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (linearising active quadratic constraints)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               max_iters: int = 15,
               active_frac: float = 0.02,
               max_active: int = 200) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of a linearised‑LP refinement.
    Each iteration:
        • pick the most critical convolution indices (largest values)
        • build a linear upper‑bound constraint using the current vector
        • solve a tiny LP (highs) that minimises the new upper‑bound t
        • accept the solution if the true objective improves
    Returns the best vector obtained in the allotted time.
    """
    best_a = a0.copy()
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(best_a)
        max_val = conv.max()
        # select active indices (largest values)
        cutoff = max_val * (1.0 - active_frac)
        active = np.where(conv >= cutoff)[0]
        if active.size > max_active:
            # keep only the very top values
            active = np.argsort(conv)[-max_active:]

        # Build LP matrices
        m = active.size
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        a_curr = best_a  # alias for readability

        for row, k in enumerate(active):
            # i that contribute to (a_curr * a)[k] are i in [max(0,k-(n-1)), min(n-1,k)]
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=int)
            # coefficient = a_curr[k - i]  (linearisation)
            A_ub[row, i_range] = a_curr[k - i_range]
            # coefficient for t
            A_ub[row, -1] = -0.5
            # RHS = -0.5 * f_k   where f_k = conv[k]
            b_ub[row] = -0.5 * conv[k]

        # Equality: sum a_i = 1, t is free (but we bound it >=0)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: a_i >= 0, t >= 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # Objective: minimise t
        c = np.zeros(n + 1)
        c[-1] = 1.0

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
        except Exception as e:
            print(f"[{time.time() - start_time:6.2f}s] LP failed: {e}")
            break

        if not res.success:
            # LP infeasible – break out of refinement
            print(f"[{time.time() - start_time:6.2f}s] LP iteration {it} infeasible")
            break

        a_candidate = res.x[:-1]
        # Numerical safety – re‑project (should already sum to 1)
        a_candidate = _project_simplex(a_candidate)

        # Evaluate true objective
        conv_cand = _conv_self(a_candidate)
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_a = a_candidate
            print(f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, iter={it}, score={cand_score:.6f}")
        else:
            # No improvement – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer + occasional smoothing)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass and applies local smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25        # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

            )
            if score_opt3 < score_opt:
                a_opt, score_opt = a_opt3, score_opt3

        # Keep the globally best sequence found so far.
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # recompute remaining gradient budget after this stage
        elapsed = time.time() - start_time
        grad_time_budget = max(0.0,
                               deadline - start_time - anneal_time_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – operates on the final resolution we ended up with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-7,
            steps=8_000_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be triggered).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness.
    return [float(x) for x in best_seq]
