def _run_targeted_local_search_improved(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # compute gradient contribution for max index
        # g_i = 2 * a_j where j = idx_max - i (if in bounds)
        g = np.zeros(n)
        # vectorized:
        # compute positions j = idx_max - np.arange(n)
        j_arr = idx_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        g[mask] = 2.0 * a[j_arr[mask]]

        # donor: highest a_i * g_i (or just highest g_i if a_i > 0)
        # choose index with large a_i and large g_i
        donor_scores = a * g
        i_donor = int(np.argmax(donor_scores))
        # receiver: lowest g_i (prefer indices with small a to keep distribution)
        i_receiver = int(np.argmin(g + 1e-12 * a))  # break ties with small a

        if i_donor == i_receiver or a[i_donor] <= 1e-15:
            # fallback to random move
            i_donor = np.random.choice(n, p=a) if a.sum()>0 else np.random.randint(n)
            i_receiver = np.random.randint(n)
            if i_donor == i_receiver:
                continue

        # propose delta: move a fraction of donor mass
        eps = a[i_donor] * np.random.rand() * 0.5  # up to 50% of donor
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_receiver] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed_frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed_frac
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve2: n={n}, iter={it}, score={new_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the previous best construction if available,
   otherwise from a uniform simplex point.
2. Progressive up‑sampling to a series of lengths (≈2 k → 38 k).
3. At each length:
   * Phase A – exponentiated (mirror) gradient descent on a
     soft‑max approximation of the maximum self‑convolution.
   * Phase B – exact sub‑gradient descent on the true maximum entry.
4. Final polishing – deterministic “peak‑shaving” local search that
   moves mass from indices that contribute most to the current maximal
   convolution entry to low‑contribution indices, with a simulated‑annealing
   acceptance rule.
5. Optional short perturbation‑restart loops if time remains.
6. Return the best sequence (normalised to a probability simplex).

All operations stay on the probability simplex, i.e. the sum of the
sequence is 1, which makes the evaluation function equal to
``2·n·max(conv)`` and independent of overall scaling.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of a soft‑max of the convolution (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k) * conv_k where w_k = soft‑max(conv/temperature).
    """
    maxc = conv.max()
    # stability: shift before exponentiating
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: exponentiated (mirror) gradient descent on soft‑max max
# ----------------------------------------------------------------------
def _exp_grad_descent(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.2,
                      T_start: float = 2.0e-2,
                      T_end: float = 5.0e-4,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using a multiplicative update.
    The smooth objective is  T·log Σ exp(conv/T) .
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (slow exponential decay)
        T = T_start * (T_end / T_start) ** (it / 25000.0)

        # gradient of the smooth max (soft‑max weighted)
        grad = _softmax_gradient(a, conv, T)

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update (mirror descent w.r.t. KL)
        # clip gradient to avoid overflow in exp
        grad_clipped = np.clip(grad, -100.0, 100.0)
        a = a * np.exp(-eta * grad_clipped)

        # keep positivity and renormalise to the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser: projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.5,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution entry."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient of the active convolution entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        # momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase C – deterministic “peak‑shaving” local search
# ----------------------------------------------------------------------
def _run_targeted_local_search_improved(a0: np.ndarray,
                                        n: int,
                                        deadline: float,
                                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Move probability mass from indices that contribute most to the current
    maximal convolution entry towards low‑contribution indices.
    Acceptance follows a simulated‑annealing schedule.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # gradient of the active max entry: g_i = 2 * a_{idx_max-i} if in range
        g = np.zeros(n)
        j_arr = idx_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        g[mask] = 2.0 * a[j_arr[mask]]

        # donor = index with large a_i * g_i (big impact if reduced)
        donor_scores = a * g
        i_donor = int(np.argmax(donor_scores))

        # receiver = index with smallest g_i (least impact if increased)
        i_receiver = int(np.argmin(g + 1e-12 * a))

        # safeguard against degenerate choices
        if i_donor == i_receiver or a[i_donor] <= 1e-15:
            # fallback to a small random move
            probs = a / a.sum()
            i_donor = int(np.random.choice(n, p=probs))
            i_receiver = np.random.randint(n)
            if i_donor == i_receiver:
                continue

        # propose moving a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_receiver] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # simulated‑annealing acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local‑improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine combines mirror‑descent on a smooth max, exact sub‑gradient
    descent, and a deterministic peak‑shaving local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start (previous best construction, if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule (progressive up‑sampling)
    # ------------------------------------------------------------------
    length_schedule = [2400, 4800, 9600, 19200, 38400]   # final length ≈ 4·10⁴
    total_time = deadline - start_time
    grad_budget = total_time * 0.60   # 60 % for gradient phases
    local_budget = total_time - grad_budget

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Gradient phases on each length
    # ------------------------------------------------------------------
    for idx, n in enumerate(length_schedule):
        if time.time() >= deadline:
            break

        # remaining time for gradient work (excluding the local‑search chunk)
        elapsed = time.time() - start_time
        remaining_grad = max(0.0, deadline - start_time - local_budget - elapsed)
        if remaining_grad <= 1.0:
            break

        # share of remaining_grad proportional to the current length
        rem_lengths = sum(length_schedule[idx:])
        stage_time = remaining_grad * n / rem_lengths
        stage_deadline = time.time() + stage_time

        # initialise a0
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Phase A – exponentiated gradient descent (smooth max)
        a_exp, sc_exp = _exp_grad_descent(a0, n, stage_deadline, start_time)
        if sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad stage n={n} improve: score={sc_exp:.6f}")

        # Phase B – exact sub‑gradient descent (tighten further)
        # Use any remaining time in this stage for sub‑gradient work
        if time.time() < stage_deadline - 0.5:
            sub_deadline = stage_deadline
            a_sub, sc_sub = _run_subgrad_gd(_upsample(best_seq, n), n,
                                            sub_deadline, start_time)
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad stage n={n} improve: score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase C – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        a_loc, sc_loc = _run_targeted_local_search_improved(best_seq, n,
                                                             deadline, start_time)
        if sc_loc < best_score:
            best_score = sc_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: score={sc_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional perturbation‑restart loops if a little time is left
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        # add a small Gaussian jitter and re‑normalise
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = np.abs(pert)
        s = pert.sum()
        if s > 0:
            pert /= s
        else:
            pert = np.full(n, 1.0 / n)

        # short local‑search run (max 30 s or remaining time)
        tmp_deadline = time.time() + min(30.0, deadline - time.time())
        a_tmp, sc_tmp = _run_targeted_local_search_improved(pert, n,
                                                             tmp_deadline, start_time)
        if sc_tmp < best_score:
            best_score = sc_tmp
            best_seq = a_tmp.copy()
            print(f"[{time.time() - start_time:6.2f}s] perturbed improve: score={sc_tmp:.6f}")

        if deadline - time.time() < 5.0:
            break

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may happen because of rounding)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
