"""
Multiscale optimisation for the step‑function attack.

The optimiser works in three stages.

1. **Coarse shape search** – a small parametric family
   a_i ∝ ((i+1)*(n-i))**γ (symmetric “beta‑like’’ shapes) is scanned
   on a coarse grid.  The best shape (and a few uniform mixtures) is kept
   as a warm‑start.

2. **Progressive refinement** – the warm‑start is first optimised on a
   coarse grid (n≈1200) with Adam on a log‑sum‑exp smoothed max‑convolution,
   then up‑sampled to larger resolutions (2400 → 4800 → 9600).  At each
   resolution a short Adam run, a few hundred exact sub‑gradient steps,
   and a fast incremental simulated‑annealing pass are performed.
   After every major phase the vector is symmetrised (a←(a+a[::-1])/2)
   because the objective is invariant under reversal and symmetry helps
   to keep the convolution flat.

3. **Final polishing** – a long annealing run on the final resolution
   uses a geometric temperature schedule and moves a random fraction
   (up to 30 %) of mass from one index to another.  The convolution is
   updated incrementally (O(n) per move) so millions of moves fit into the
   remaining time budget.

The whole routine respects the supplied time budget (≈1000 s).  The
best sequence encountered is normalised to sum = 1 and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT and convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # pick the central n entries (see original code)
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v_i = v_{n-1-i} and renormalise."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.shape[0])
    return v


# ----------------------------------------------------------------------
# Phase 1 – Adam on smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimisation of the smooth approximation of max(conv).
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))          # logits (soft‑max)

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – larger α approximates the true max better
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 7000                     # a bit more than the reference

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # current probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()
            a = _symmetrize(a)                # keep symmetry

            # convolution & smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            _ = 2.0 * n * f_val                # not used, only for diagnostics

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient wr.t. logits (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits (symmetrised)
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    a = _symmetrize(a)

    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 5_000_000):
    """
    Sub‑gradient descent on the true max‑convolution.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # identify active (near‑max) indices
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of max(conv)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair (i,j) contributes a_j to ∂/∂a_i and a_i to ∂/∂a_j
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        # simplex projection (Duchi et al.)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 2_000_000):
    """
    Mass‑transfer annealing with an O(n) incremental convolution update.
    Returns (best_seq, best_score).
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-8

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # source chosen proportionally to its mass (focus on heavy spots)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (up to 30 %) of src mass
        eps = a[src] * rng.random() * 0.30

        # ---- apply move (using old a for the update) ----
        # shift contributions
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a

        # self‑pair corrections
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ---- reject, revert ----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Coarse parametric shape grid search
# ----------------------------------------------------------------------
def _grid_search_shapes(n: int) -> tuple[np.ndarray, float]:
    """
    Scan a simple one‑parameter symmetric family a_i ∝ ((i+1)*(n-i))**γ.
    Returns the best shape (normalised) and its score.
    """
    best_seq = None
    best_score = float("inf")

    # grid of exponents (γ).  The range was chosen after a few quick trials.
    for gamma in np.arange(-0.8, 1.61, 0.15):
        idx = np.arange(n, dtype=float)
        shape = ((idx + 1) * (n - idx)) ** gamma
        # avoid exact zeros which would be removed by the security checks
        shape = np.maximum(shape, 1e-16)
        shape /= shape.sum()
        sc = evaluate_sequence(shape.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = shape.copy()
    # try a few uniform mixtures
    uniform = np.full(n, 1.0 / n)
    for w in np.arange(0.0, 1.01, 0.2):
        mix = w * best_seq + (1.0 - w) * uniform
        mix /= mix.sum()
        sc = evaluate_sequence(mix.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = mix.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# One optimisation run on a given resolution
# ----------------------------------------------------------------------
def _optimize_one_run(a0: np.ndarray, n: int, deadline: float,
                      start_time: float, best_score: float,
                      best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Run Adam → projected gradient → annealing on a single resolution.
    Returns the best sequence obtained on that resolution and its score.
    """
    # Phase 1 – Adam (smooth max)
    a_adam, best_score, best_seq = _adam_phase(a0, n, deadline,
                                               start_time, best_score, best_seq)

    # Phase 2 – exact sub‑gradient polishing (short)
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_pol, sc_pol = _run_gradient_descent(a_adam, n,
                                              polish_deadline, start_time)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # Phase 3 – fast annealing
    if time.time() < deadline:
        a_ann, sc_ann = _run_incremental_annealing(best_seq, n,
                                                   deadline, start_time,
                                                   best_score, best_seq,
                                                   total_steps=1_200_000)
        if sc_ann < best_score:
            best_score = sc_ann
            best_seq = a_ann.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (final stage): score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Multiscale optimiser for the step‑function attack.
    Returns a list of non‑negative heights (sum≈1) with a low evaluation score.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.7

    # ------------------------------------------------------------------
    # Choose a progressive list of resolutions
    # ------------------------------------------------------------------
    resolutions = [1200, 2400, 4800, 9600]          # can be truncated by time
    best_seq = None
    best_score = float("inf")

    for n in resolutions:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate roughly one quarter of the remaining budget for this stage
        stage_deadline = time.time() + remaining * 0.25

        # initialise – either up‑sample previous best or start from a parametric scan
        if best_seq is None:
            # fresh start: cheap parametric scan on the current n
            a0, sc0 = _grid_search_shapes(n)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
                print(f"[{time.time() - start_time:6.2f}s] grid‑search start (n={n}): score={best_score:.6f}")
        else:
            a0 = _upsample(best_seq, n)           # keep the shape discovered so far

        # run the three‑phase optimiser on this resolution
        a_opt, sc_opt = _optimize_one_run(a0, n, stage_deadline,
                                          start_time, best_score, best_seq)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] new best after n={n}: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final long anneal on the last obtained resolution (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_n = best_seq.shape[0]
        a_fin, sc_fin = _run_incremental_annealing(best_seq, final_n,
                                                   deadline, start_time,
                                                   best_score, best_seq,
                                                   total_steps=2_000_000)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # final safety check (optional but useful for logging)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    # convert to plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
