"""
Improved search for a low‑autocorrelation step‑function.

The algorithm works in three stages:

1. **Low‑resolution Adam optimisation** – start with a short vector (≈2000
   entries) and minimise a temperature‑smoothed “soft‑max’’ of the
   self‑convolution using the Adam optimiser.  The low resolution makes
   many iterations cheap while still shaping the distribution.

2. **Multi‑scale refinement** – up‑sample the best low‑resolution vector, run
   a short sub‑gradient descent (active‑set) on the full length (≈9600) and
   finish with a long simulated‑annealing walk that includes mass‑transfer,
   local smoothing and value‑swap moves.  The walk is biased by a
   geometric temperature schedule.

3. **Permutation polishing** – randomise the ordering of the final mass
   vector several times, keeping any improvement.  Because the convolution
   depends on the ordering, this cheap step can still lower the peak.

All vectors are projected onto the probability simplex after each
update, guaranteeing non‑negative entries that sum to 1 (the exact normaliser
used by ``evaluate_sequence``).  The routine respects the supplied
time‑budget (default 1000 s) and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1} using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2*n-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n``; renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Evaluation – fast replica of evaluate_sequence (no clamping needed).
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv)/sum(a)²  (identical to evaluate_sequence for valid a)."""
    n = a.shape[0]
    max_b = np.max(_conv_self(a))
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) surrogate of the max‑convolution and its gradient.
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the temperature‑smoothed max (soft‑max) of conv."""
    maxc = conv.max()
    # shift by the max for numerical stability
    scaled = (conv - maxc) / temperature
    # log‑sum‑exp trick
    shift = np.max(scaled)
    exps = np.exp(scaled - shift)
    w = exps / exps.sum()                     # weights sum to 1
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate.
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr0: float = 0.3,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Adam on the smoothed max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    # temperature schedule: start relatively warm and decay geometrically
    T_start, T_end = 5.0e-2, 5.0e-5

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # geometric temperature decay (tuned for ~2·10⁴ iters)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution (active set).
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps_start: float = 0.05,
                        rel_eps_end: float = 0.005) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with a shrinking active‑set tolerance."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # gradually tighten the active‑set tolerance
        rel_eps = rel_eps_start * (rel_eps_end / rel_eps_start) ** (it / 20000.0)
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m_len = w.shape[0]
        fft_len = 1 << ((n + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Simulated‑annealing refinement (mass‑transfer, smoothing, swaps).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.4,

    """Metropolis walk with three move types: mass‑transfer, local smoothing,
    and value‑swap.  The walk is long (few million steps) and dominates the
    latter part of the budget."""
    a = a0.copy()
    score = _score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        move_type = np.random.rand()
        if move_type < 0.80:                     # mass‑transfer
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                eps = a[i] * np.random.rand() * 0.30   # ≤30 % of mass at i
                a_prop = a.copy()
                a_prop[i] -= eps
                a_prop[j] += eps
            else:
                continue
        elif move_type < 0.95:                   # local smoothing
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)
        else:                                    # value‑swap
            i, j = np.random.randint(0, n, size=2)
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        new_score = _score(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score
