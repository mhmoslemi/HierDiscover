"""
Propose a step‑function that minimizes `evaluate_sequence`.

Algorithm
---------
1. Initialise a moderate‑size distribution (n≈800) from the previous best
   construction (if available) by linear interpolation, otherwise from a
   random simplex point.
2. Repeatedly solve a linear programme
      maximise   Σ_j b_j
      subject to  b_j ≥ 0  and  (a * b)_k ≤ M   for every k,
   where a is the current distribution and M = max(a*a).  The LP
   tells us how much total mass can be added without breaking the
   current convolution ceiling.
3. Normalise the LP solution to unit sum to obtain a partner g.
   If g already improves the objective replace a←g,
   otherwise take a convex combination a←(1‑α)a+αg (α≈0.5) and keep the
   better of the two.
4. After a few LP steps (or when the time budget is almost used) run a
   short simulated‑annealing “peak‑shaving’’ local search to polish the
   result.
5. Return the best sequence seen, normalised to a probability simplex.

The routine works entirely on the simplex (scaling does not affect the
objective) and never exceeds the given time budget.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: linear up‑sampling / down‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` from its current length to `new_n`
    and renormalisation to sum 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper: full linear self‑convolution (naïve for small n)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Helper: build the (2n‑1)×n matrix that maps b ↦ a*b
# ----------------------------------------------------------------------
def _conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """Matrix A such that (A @ b) == np.convolve(a, b)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        # column i contributes a[i] to rows i … i+n‑1
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
# Linear programme: maximise total mass of b under the current
# convolution ceiling M = max(a*a).
# ----------------------------------------------------------------------
def _lp_max_mass(a: np.ndarray, M: float):
    """
    Solve
        maximise   Σ b_j
        subject to  b_j ≥ 0
                    (a * b)_k ≤ M    for all k
    Returns the optimal b and its total mass.  If the problem is
    infeasible or the optimiser fails, ``(None, 0)`` is returned.
    """
    n = a.shape[0]
    A = _conv_constraint_matrix(a)        # shape (2n‑1, n)
    ub = np.full(A.shape[0], M)

    # linprog minimises c·x, so we use -1 to maximise sum(b)
    c = -np.ones(n)

    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None, 0.0
    b = res.x
    total = b.sum()
    return b, total


# ----------------------------------------------------------------------
# Targeted local search (peak shaving) – same idea as the previous code
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               deadline: float,
                               start_time: float) -> np.ndarray:
    """
    Small simulated‑annealing loop that moves a tiny amount of probability
    mass from indices that heavily contribute to the current maximal
    convolution entry to indices that contribute little.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    # temperature schedule (fast cooling)
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to this maximal entry
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = np.random.randint(n)
        else:
            receiver = np.random.choice(low_candidates)

        if donor == receiver or a[donor] <= 1e-15:
            continue

        eps = a[donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` using a small‑scale LP + local‑search scheme.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a modest problem size – large enough to be expressive but
    # small enough for many LP solves.
    # ------------------------------------------------------------------
    n = 800   # can be altered; 800 gives a 1599‑length convolution matrix

    # ------------------------------------------------------------------
    # Warm‑start from the global best if it is present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    if starter is not None:
        a = _upsample(starter, n)
    else:
        a = np.random.rand(n)
        a /= a.sum()

    # keep the best sequence seen so far
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    print(f"[0.00s] initial score: n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Core loop – alternating LP improve + optional mixing
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline and lp_iter < 30:
        lp_iter += 1
        conv_a = _conv_self(a)
        M = conv_a.max()

        b, total_mass = _lp_max_mass(a, M)
        if b is None or total_mass <= 1e-12:
            print(f"[{time.time() - start_time:6.2f}s] LP failed or degenerate, stopping.")
            break

        # Normalise the LP solution to unit sum → partner g
        g = b / total_mass

        # Evaluate both candidates
        conv_g = _conv_self(g)
        M_g = conv_g.max()
        score_g = 2.0 * n * M_g

        # mixed candidate (half‑half)
        alpha = 0.5
        mix = (1.0 - alpha) * a + alpha * g
        conv_mix = _conv_self(mix)
        M_mix = conv_mix.max()
        score_mix = 2.0 * n * M_mix

        # pick the best among a, g, mix
        candidates = [(a, best_score), (g, score_g), (mix, score_mix)]
        # sort by score (ascending)
        candidates.sort(key=lambda x: x[1])
        a, cur_score = candidates[0]

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve #{lp_iter}: score={cur_score:.6f}")

        # early exit if improvement becomes tiny
        if best_score < 1.45:          # already excellent – bail out early
            break

    # ------------------------------------------------------------------
    # Final polishing with a quick simulated‑annealing local search.
    # The remaining time (if any) is handed to the local search routine.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc = _run_targeted_local_search(best_seq, deadline, start_time)
        conv_loc = _conv_self(a_loc)
        score_loc = 2.0 * n * conv_loc.max()
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Normalise the final answer (tiny drift might appear)
    # ------------------------------------------------------------------
    final = np.asarray(best_seq, dtype=float)
    s = final.sum()
    if s > 0:
        final /= s
    else:
        final = np.full_like(final, 1.0 / final.size)

    # Return a plain Python list for the harness
    return [float(x) for x in final]

    return b, total

# Targeted local search (peak shaving) – same idea as the previous code
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               deadline: float,
                               start_time: float) -> np.ndarray:
    """
    Simulated‑annealing loop that moves a tiny amount of probability mass
    from indices that heavily contribute to the current maximal convolution
    entry to indices that contribute little.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the maximal entry
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = np.random.randint(n)
        else:
            receiver = np.random.choice(low_candidates)

        if donor == receiver or a[donor] <= 1e-15:
            continue

        eps = a[donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` using a small‑scale LP + local‑search scheme.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – big enough to be expressive, small enough for many LPs
    # ------------------------------------------------------------------
    n = 800

    # ------------------------------------------------------------------
    # Warm‑start from the globally available construction if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    if starter is not None:
        a = _upsample(starter, n)
    else:
        a = np.random.rand(n)
        a /= a.sum()

    best_seq = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    print(f"[0.00s] initial score: n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Core loop – alternating LP improvement and simple mixing
    # ------------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline and lp_iter < 30:
        lp_iter += 1
        conv_a = _conv_self(a)
        M = conv_a.max()

        b, total_mass = _lp_max_mass(a, M)
        if b is None or total_mass <= 1e-12:
            print(f"[{time.time() - start_time:6.2f}s] LP failed or degenerate, stopping.")
            break

        # Normalise the LP solution to unit sum → partner g
        g = b / total_mass

        # Evaluate candidates: original, pure g, and a simple mix
        conv_g = _conv_self(g)
        score_g = 2.0 * n * conv_g.max()

        alpha = 0.5
        mix = (1.0 - alpha) * a + alpha * g
        conv_mix = _conv_self(mix)
        score_mix = 2.0 * n * conv_mix.max()

        # Choose the best of the three
        candidates = [(a, best_score), (g, score_g), (mix, score_mix)]
        candidates.sort(key=lambda x: x[1])
        a, cur_score = candidates[0]

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve #{lp_iter}: score={cur_score:.6f}")

        # Early exit if we already have a very good bound
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # Final polishing with a quick simulated‑annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_loc = _run_targeted_local_search(best_seq, deadline, start_time)
        conv_loc = _conv_self(a_loc)
        score_loc = 2.0 * n * conv_loc.max()
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Normalise the final answer (tiny drift may appear)
    # ------------------------------------------------------------------
    final = np.asarray(best_seq, dtype=float)
    s = final.sum()
    if s > 0:
        final /= s
    else:
        final = np.full_like(final, 1.0 / final.size)

    # Return a plain Python list for the harness
    return [float(x) for x in final]
