"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

The algorithm works in three stages and spends the allotted 1000 s
budget automatically:

1. **Diverse warm‑starts** – a handful of analytically‑motivated and random
   initial probability vectors (uniform, Hann window, half‑sine, random Dirichlet)
   are created for a moderate length (default 4096).  If a previous
   ``height_sequence_1`` is available it is also used.

2. **Fast first‑order optimisation** – each start is refined by a few
   thousand iterations of a Adam optimiser applied to a
   temperature‑smoothed surrogate of the max‑convolution (soft‑max).
   The surrogate temperature is decreased geometrically and every
   iterate is projected onto the probability simplex.

3. **Long simulated‑annealing refinement** – the best vector obtained
   from step 2 is then processed by a Metropolis walk that transfers tiny
   fractions of mass between random coordinates and occasionally applies
   a local smoothing kernel.  The temperature decays exponentially from
   0.5 to 1e‑7.  The walk runs until the time budget is exhausted (or a
   target score < 1.503 is reached).

All heavy linear algebra (self‑convolution) is performed with FFT
(O(n log n)).  The code is completely self‑contained, uses only
``numpy``/``math``/``time`` and respects the 1‑CPU rule.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after every update.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Very degenerate case – return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full length = 2·n‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # choose next power of two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max surrogate of the max‑convolution and its gradient.
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation
        S_T(conv) = temperature * log sum exp(conv/temperature)
    with respect to the vector a.  The formula follows from the chain rule.
    """
    maxc = conv.max()
    # stability‑wise shift by maxc before scaling
    scaled = (conv - maxc) / temperature
    # compute soft‑max weights
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    # convolution of reversed a with w gives the gradient of conv·w
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Fast Adam optimiser on the smooth surrogate.
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iters: int = 200_000,
                      lr: float = 0.15,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      T0: float = 5e-2,
                      T_end: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the temperature‑smoothed max‑convolution.
    Returns the best vector seen and its (2·n·max) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current temperature (geometric decay)
        frac = it / max_iters
        T = T0 * (T_end / T0) ** frac

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}")

        grad = _softmax_gradient(a, conv, T)

        # Adam update (note the descent direction = -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (active set) with Adam‑style adaptation.
# ----------------------------------------------------------------------
def _run_subgradient_adam(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 200_000,
                          lr: float = 0.12,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the true max‑convolution (sub‑gradient obtained from
    an active set of near‑max indices).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: n={n}, it={it}, score={cur_score:.6f}")

        # active set: indices whose value is within rel_eps of the max
        eps_thr = max_val * rel_eps
        active = np.where(conv >= max_val - eps_thr)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # build weight vector w that averages over the active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        m_len = w.shape[0]
        fft_len = 1 << ((n + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        # Adam step (descent = -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   init_score: float,
                   start_temp: float = 0.5,
                   end_temp: float = 1e-7,
                   max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that repeatedly transfers a small fraction of mass
    between two random coordinates and occasionally smooths the whole
    vector with a short uniform kernel.
    """
    a = a0.copy()
    score = init_score
    best_seq = a.copy()
    best_score = score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # decide move type
        if np.random.rand() < 0.88:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # move a random fraction (up to 20 %) of the donor's mass
            epsilon = a[i] * np.random.rand() * 0.20
            a_prop = a.copy()
            a_prop[i] -= epsilon
            a_prop[j] += epsilon
        else:                                            # local smoothing
            k = np.random.choice([3, 5, 7, 9])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        # evaluate new candidate (fast evaluation: 2·n·max(conv))
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

        # Early exit if we already beat the target (1.503)
        if best_score < 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: generate a few deterministic seeds (window functions).
# ----------------------------------------------------------------------
def _seed_vectors(n: int):
    """Yield a few nice initial probability vectors of length n."""
    # uniform
    yield np.full(n, 1.0 / n)

    # Hann (raised cosine) window
    x = np.arange(n)
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * x / (n - 1)))
    hann = hann / hann.sum()
    yield hann

    # Half‑sine window (positive sine over (0,π))
    half_sine = np.sin(np.pi * (x + 0.5) / n)
    half_sine = half_sine / half_sine.sum()
    yield half_sine

    # Triangular (linear rise‑fall)
    tri = np.concatenate([np.arange(1, n // 2 + 1),
                          np.arange(n // 2, 0, -1)])
    tri = tri[:n]                     # handle odd n
    tri = tri / tri.sum()
    yield tri

    # Random Dirichlet (drawn from exponential)
    rand = np.random.rand(n)
    rand = rand / rand.sum()
    yield rand

    # ------------------------------------------------------------------
    # Keep the globally best vector seen.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Iterate over a handful of seeds respecting the remaining time.
    # ------------------------------------------------------------------
    seeds_tried = 0
    for init_vec in _seed_vectors(n):
        if time.time() >= deadline:
            break

        # Use warm start (if any) as an extra seed
        if warm_start is not None and seeds_tried == 0:
            a0 = warm_start.copy()
        else:
            a0 = init_vec.copy()

        # ------------------------------------------------------------------
        # Phase 1 – Adam on the soft‑max surrogate (fast, smooth).
        # ------------------------------------------------------------------
        phase_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.25)
        a_opt, score_opt = _run_adam_softmax(
            a0, n, phase_deadline, start_time,
            max_iters=200_000, lr=0.12, T0=5e-2, T_end=5e-4
        )

        # ------------------------------------------------------------------
        # Phase 2 – Adam on the true max (sub‑gradient) if time permits.
        # ------------------------------------------------------------------
        if time.time() < deadline:
            phase_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.25)
            a_opt2, score_opt2 = _run_subgradient_adam(
                a_opt, n, phase_deadline, start_time,
                max_iters=200_000, lr=0.10, rel_eps=0.01
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # ------------------------------------------------------------------
        # Keep the best seen so far.
        # ------------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] seed #{seeds_tried} improve: score={score_opt:.6f}")

        seeds_tried += 1

    # ------------------------------------------------------------------
    # Final long annealing phase (dominates the remaining budget).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # compute current score (fast formula)
        cur_score = 2.0 * n * _conv_self(best_seq).max()
        best_seq, best_score = _run_annealing(
            best_seq, n, deadline, start_time,
            init_score=cur_score,
            start_temp=0.5, end_temp=1e-7,
            max_steps=5_000_000
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal result: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event of failure.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Final normalisation (to avoid tiny drift).
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
