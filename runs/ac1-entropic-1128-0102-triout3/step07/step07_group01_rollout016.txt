"""
Multi‑phase optimisation for the step‑function bound.

The pipeline consists of

1. **Warm‑up (exponentiated gradient)** – a smoothed‑max loss
   (softmax of the self‑convolution) is minimised by mirror descent
   on the probability simplex.

2. **Precise descent (projected sub‑gradient with momentum)** – the exact
   max‑convolution is tackled directly; a decaying step‑size and a
   Nesterov‑style momentum give fast convergence.

3. **Simulated‑annealing (SA) fine‑tuning** – random source→destination
   mass transfers are accepted both if they improve the current
   maximum and with a Boltzmann probability when they are slightly
   worse.  This helps escape local minima that the pure sub‑gradient
   method can get stuck in.

4. **Targeted refinement** – moves are focused on the indices that
   actively realise the current maximal convolution entry; only strict
   improvements are kept, driving the final bound down to the last
   achievable fraction of a second.

All stages respect the overall time budget (default 1000 s).  The
returned list is a non‑negative probability vector (sum = 1) that can
be fed directly into ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each ordered pair (i,j) contributes a[j] to grad[i] and a[i] to grad[j]
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative entries that sum to one).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 – exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 5000):
    """
    Mirror‑descent with KL‑divergence (exponentiated gradient) on the
    soft‑max of the self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)        # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ----- smooth‑max gradient (softmax weights) -----
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)   # safe shifting
        w /= w.sum()
        w_rev = w[::-1]

        # convolution a * w_rev (via FFT)
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative (mirror) update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        # schedule handling
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Phase‑2 – exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.3,
                             beta: float = 0.9,
                             max_iter: int = 10 ** 7):
    """
    Classic sub‑gradient descent on the true max‑convolution,
    with momentum and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             rng: np.random.Generator,
                             init_temp_frac: float = 0.4,
                             final_temp_frac: float = 1e-6,
                             max_move_frac: float = 0.12):
    """
    Random mass transfers with a temperature schedule.
    Moves are drawn preferentially from the indices that contribute
    most to the current maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature expressed as a fraction of the current max
    T0 = cur_max * init_temp_frac
    Tf = cur_max * final_temp_frac
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # linear cooling in the log‑temperature domain (exponential decay)
        elapsed = time.time() - start_time
        # proportion of the allotted SA time that has already passed
        frac = (time.time() - (deadline - total_time)) / total_time
        # temperature drops exponentially from T0 to Tf
        T = T0 * (Tf / T0) ** frac

        # ----- source selection (probability proportional to contribution) -----
        k_max = int(np.argmax(conv))
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # ----- destination (uniform, different from src) -----
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # ----- amount to move -----
        eps = a[src] * rng.random() * max_move_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        delta = new_max - cur_max

        # Metropolis acceptance
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-30)):
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 – targeted local refinement (tiny moves centred on the max)
# ----------------------------------------------------------------------


def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10 ** 7,
                         eps_frac: float = 0.018):
    """
    Moves a tiny fraction of probability mass away from the indices that
    dominate the current maximal convolution entry.  Only strictly
    improving moves are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        k_max = int(np.argmax(conv))

        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            prob_src = a / a.sum()
        else:
            prob_src = contrib / total_contrib

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:       # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # Normalisation target: we always work with a probability vector.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient warm‑up
    # --------------------------------------------------------------
    phase1_time = min(14.0, 0.08 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=5000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (major chunk of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep ~70 % of the remaining budget for this phase
        phase2_time = max(30.0, 0.70 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.3,
            beta=0.9,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly half of the rest to SA, the other half to the final refine
        phase3_time = max(10.0, 0.45 * remaining)
        phase3_deadline = time.time() + phase3_time

        a_sa = _run_simulated_annealing(
            best_seq,
            final_n,
            phase3_deadline,
            start_time,
            rng,
            init_temp_frac=0.5,
            final_temp_frac=1e-7,
            max_move_frac=0.10,
        )
        conv_sa = _conv_self(a_sa)
        score_sa = 2.0 * final_n * conv_sa.max()
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA phase improve: score={score_sa:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – targeted local refinement (tiny, focused moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10 ** 7,
            eps_frac=0.018,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and return the best sequence found
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
