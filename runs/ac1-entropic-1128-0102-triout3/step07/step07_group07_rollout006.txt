"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Fast parametric sweep** – generate many symmetric Beta‑shaped
   candidates a_i ∝ x_i^{α‑1}(1‑x_i)^{α‑1} (x_i the centre of the i‑th cell)
   for a wide range of shape parameters α∈(0.2,5).  Also test a few
   random Dirichlet draws.  The evaluation is cheap (np.convolve) and
   yields a good initial point far below the uniform baseline.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   minimise the true objective  f(a)=max_k (a*a)[k]  by repeatedly
   (a) computing the convolution,
   (b) forming a sub‑gradient from the active (maximal) indices,
   (c) taking a momentum step and projecting back onto the probability
   simplex.  This phase quickly lowers the peak of the self‑convolution.

3. **Deterministic peak reduction** – a cheap greedy loop that moves a
   fraction of the coefficient that contributes most to the current peak
   towards the smallest coefficient, accepting only improving moves.

4. **Long simulated‑annealing refinement** – random mass‑transfer moves
   with an exponential cooling schedule escape any remaining plateaus.

The routine respects a total time budget (default 1000 s) by cutting each
stage short when the deadline approaches.  The final sequence is returned
as a plain list of non‑negative floats with unit sum, ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _beta_seq(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on n points, normalised."""
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n          # cell centres in (0,1)
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) sub‑gradient descent on
        f(a) = max_k (a*a)[k] .
    Returns the best vector seen and its score 2·n·max_conv .
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient of the max (active set) --------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # ordered pairs; diagonal counted twice (harmless)

        # momentum update & step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy loop that moves a fraction of the coefficient contributing most
    to the current maximal convolution peak towards the currently smallest
    coefficient.  The move is kept only if it strictly lowers the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (different from src)
        dst_cands = np.delete(np.arange(n), src)
        dst = dst_cands[a[dst_cands].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: "
                  f"max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Classic annealing: move a random ε‑fraction of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine consists of a fast parametric sweep, projected sub‑gradient
    descent, greedy peak reduction and a long simulated‑annealing refinement,
    all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # final problem size (the benchmark uses 4800, we keep it)
    n = 4800

    # ------------------------------------------------------------------
    # 1) QUICK PARAMETRIC SWEEP (Beta‑shapes + random Dirichlet)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # Beta‑shape sweep (α from 0.2 to 5.0, denser near 1)
    alphas = np.concatenate([
        np.linspace(0.2, 5.0, 30),
        np.logspace(-0.5, 0.5, 30)   # roughly 0.316 … 3.16
    ])
    for alpha in alphas:
        if time.time() >= deadline:
            break
        seq = _beta_seq(n, alpha)
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] beta α={alpha:.4f} improve: score={sc:.6f}")

    # few random Dirichlet seeds
    rng = np.random.default_rng()
    for _ in range(15):
        if time.time() >= deadline:
            break
        seq = rng.dirichlet(np.ones(n))
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] random dirichlet improve: score={sc:.6f}")

    # fallback to uniform if nothing was generated
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2) PROJECTED SUB‑GRADIENT DESCENT (exact max objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly half of the remaining time to polishing
        remaining = deadline - time.time()
        grad_deadline = time.time() + 0.45 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            grad_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] gradient polish improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # 3) DETERMINISTIC PEAK‑REDUCTION
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        peak_deadline = time.time() + 0.15 * remaining
        a_peak, score_peak = _run_peak_reduction(
            best_seq,
            peak_deadline,
            start_time,
            max_no_improve=1500,
        )
        if score_peak < best_score:
            best_score = score_peak
            best_seq = a_peak.copy()
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: score={score_peak:.6f}")

    # ------------------------------------------------------------------
    # 4) LONG SIMULATED ANNEALING REFINEMENT
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            deadline,
            start_time,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=800_000,          # generous step budget; loop stops at deadline
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Convert to plain Python list of floats as required by the framework
    return [float(x) for x in best_seq.tolist()]
