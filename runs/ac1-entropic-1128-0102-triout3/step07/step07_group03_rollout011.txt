1. **Warm‑start / deterministic seeds** – the previously found best
   sequence (if any) and a few manually crafted shapes (exponential,
   linear, power‑law and logistic) are evaluated quickly; the best of
   them becomes the current incumbent.

2. **Random Dirichlet restarts** – many short runs on a smaller
   dimension (≈ 1 k–2 k) use a simple sub‑gradient method with Nesterov
   momentum.  The improved small‑dimensional vectors are up‑sampled to
   the final length (9600) and kept if they beat the incumbent.

3. **High‑precision sub‑gradient refinement** – the incumbent is
   refined on the full length with many more sub‑gradient steps,
   again with momentum and a diminishing step size.

4. **Deterministic peak‑shaving** – a greedy local‑search that repeatedly
   transfers a fraction of mass from the indices that contribute most
   to the current maximal convolution entry to the indices that
   contribute least.  Each move is accepted only if it lowers the
   objective, ensuring monotone progress.

All operations respect the constraints (non‑negative, sum‑to‑one) by
projecting onto the probability simplex after each update.  The routine
runs for at most ``budget_s`` seconds (default 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate seq to length new_n and renormalise to sum 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Fast objective (scale‑invariant)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a already normalised to sum 1."""
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (Nesterov style)
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray, max_iters: int, deadline: float) -> tuple[np.ndarray, float]:
    """Return the best point seen during a fixed‑iteration run."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = _fast_score(a)

    beta = 0.7          # momentum factor
    step0 = 0.5         # initial step size factor

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        k = int(conv.argmax())                     # active shift
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[i[mask]] = 2.0 * a[j[mask]]           # sub‑gradient of active entry

        # Nesterov momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak‑shaving local search (monotone improvement)
# ----------------------------------------------------------------------
def _greedy_shave(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Repeatedly move mass from high‑gradient to low‑gradient indices."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = _fast_score(a)

    while time.time() < deadline:
        conv = _conv_self(a)
        k = int(conv.argmax())

        # gradient of the active entry (same as sub‑gradient)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[i[mask]] = 2.0 * a[j[mask]]

        # donor = index with large gradient * present mass
        donor_mask = a > 1e-12
        if not np.any(donor_mask):
            break
        donor_candidates = np.where(donor_mask)[0]
        donor = donor_candidates[np.argmax(grad[donor_candidates])]

        # recipient = index with smallest gradient (often near zero)
        recipient = int(np.argmin(grad))

        if donor == recipient:
            # try a random pair as a fallback
            donor, recipient = np.random.choice(n, 2, replace=False)
            if a[donor] <= 1e-12:
                continue

        # move a fraction of the donor's mass
        eps = a[donor] * 0.25          # up to 25 % of donor
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _fast_score(a_prop)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
        else:
            # if not improving, try a smaller step
            eps_small = a[donor] * 0.05
            if eps_small < 1e-12:
                # give up on this pair
                a = a_prop
                continue
            a_prop[donor] += eps               # revert previous move
            a_prop[donor] -= eps_small
            a_prop[recipient] += eps_small
            a_prop = _project_simplex(a_prop)
            new_score = _fast_score(a_prop)
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                a = a_prop
            else:
                # no progress – exit to avoid endless loops
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    The routine combines:
      * deterministic seeds (previous best, exponential/linear/logistic shapes)
      * many short Dirichlet‑restart sub‑gradient runs on a small dimension
      * a long sub‑gradient refinement on the final length
      * a deterministic greedy peak‑shaving phase.

    All time is spent within the ``budget_s`` limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # ------------------------------------------------------------------
    # Fixed final length (the value used by prior runs)
    # ------------------------------------------------------------------
    n_final = 9600

    # ------------------------------------------------------------------
    # Helper to keep the best incumbent
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start from the previous best if it exists
    # ------------------------------------------------------------------
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            seq = tmp / tmp.sum()
            seq = _upsample(seq, n_final)
            seq = _project_simplex(seq)
            sc = _fast_score(seq)
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:6.2f}s] warm‑start score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣ Deterministic shape family seeds
    # ------------------------------------------------------------------
    def _try_candidate(a):
        nonlocal best_seq, best_score
        sc = _fast_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start_time:6.2f}s] new seed score = {sc:.6f}")

    # exponential decay a_i ∝ r**i
    for r in (0.9995, 0.997, 0.995, 0.99, 0.98):
        if time.time() >= deadline:
            break
        a = np.array([(1.0 - r) * (r ** i) for i in range(n_final)], dtype=float)
        a = _project_simplex(a)
        _try_candidate(a)

    # linear decreasing a_i ∝ (n‑i)
    for power in (1, 2):
        if time.time() >= deadline:
            break
        idx = np.arange(n_final)
        a = (n_final - idx) ** power
        a = _project_simplex(a)
        _try_candidate(a)

    # power‑law decreasing a_i ∝ (n‑i)**p with fractional p
    for p in (0.5, 1.5, 2.5):
        if time.time() >= deadline:
            break
        idx = np.arange(n_final)
        a = (n_final - idx) ** p
        a = _project_simplex(a)
        _try_candidate(a)

    # logistic (sigmoid) shape
    for slope in (0.02, 0.05, 0.1):
        if time.time() >= deadline:
            break
        idx = np.arange(n_final)
        mid = n_final / 2.0
        a = 1.0 / (1.0 + np.exp((idx - mid) / (slope * n_final)))
        a = _project_simplex(a)
        _try_candidate(a)

    # If nothing succeeded (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n_final, 1.0 / n_final)
        best_score = _fast_score(best_seq)

    # ------------------------------------------------------------------
    # 3️⃣ Random Dirichlet restarts on a smaller dimension
    # ------------------------------------------------------------------
    # Reserve about 20 % of the remaining budget for the final phases
    reserve_for_final = 200.0                     # seconds
    while time.time() < deadline - reserve_for_final:
        # choose a modest size for fast iterations
        n_small = np.random.randint(1200, 2401)   # 1.2 k – 2.4 k
        a_small = np.random.dirichlet(np.ones(n_small))

        # run a short sub‑gradient optimisation on the small dimension
        sub_deadline = time.time() + 4.0          # ~4 s per restart
        a_opt_small, _ = _subgrad_optimize(a_small, max_iters=3000, deadline=sub_deadline)

        # up‑sample to the final length and project
        a_up = _upsample(a_opt_small, n_final)
        a_up = _project_simplex(a_up)

        sc = _fast_score(a_up)
        if sc < best_score:
            best_seq, best_score = a_up.copy(), sc
            print(f"[{time.time() - start_time:6.2f}s] random restart improved score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Long sub‑gradient refinement on the full length
    # ------------------------------------------------------------------
    # Allocate the majority of the remaining time to this phase
    refine_deadline = deadline - reserve_for_final / 2.0
    a_refined, sc_refined = _subgrad_optimize(best_seq, max_iters=10 ** 7, deadline=refine_deadline)
    if sc_refined < best_score:
        best_seq, best_score = a_refined.copy(), sc_refined
        print(f"[{time.time() - start_time:6.2f}s] refinement improved score = {sc_refined:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Deterministic greedy peak‑shaving (final fine‑tuning)
    # ------------------------------------------------------------------
    shave_deadline = deadline
    a_shaved, sc_shaved = _greedy_shave(best_seq, deadline=shave_deadline)
    if sc_shaved < best_score:
        best_seq, best_score = a_shaved.copy(), sc_shaved
        print(f"[{time.time() - start_time:6.2f}s] greedy shave improved score = {sc_shaved:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation – guard against numerical drift
    # ------------------------------------------------------------------
    best_seq = _project_simplex(best_seq)
    best_seq = np.asarray(best_seq, dtype=float)
    best_seq /= best_seq.sum()                     # final safety normalisation

    # Return plain Python list (the harness expects a list of floats)
    return [float(x) for x in best_seq]
