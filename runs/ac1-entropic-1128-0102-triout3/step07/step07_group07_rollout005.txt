"""
Propose a non‑negative step function that (almost) minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Fast scoring** – a tiny helper computes the exact objective using an
   FFT‑based convolution (identical to ``evaluate_sequence`` for valid inputs).

2. **Random‑restart sub‑gradient descent** – many short runs are launched.
   Each run:
        * draws a random start (Dirichlet, Gaussian, exponential, triangular,
          or uniform),
        * performs projected sub‑gradient descent on the *true* objective
          ``max_k (a*a)[k]`` (the max of the self‑convolution) using a
          momentum scheme and an Euclidean simplex projection,
        * follows with a cheap greedy peak‑reduction,
        * finishes with a brief simulated‑annealing sweep.

   The best sequence seen across all restarts is retained.

3. **Final polishing** – the current global best is polished with a longer
   projected‑gradient phase, another peak‑reduction, and a final annealing
   run, using the remaining time budget.

All stages respect the global 1000 s deadline.  The code is pure‑NumPy,
uses only the provided ``evaluate_sequence`` for the final check and
remains well within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _fast_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a normalised (sum==1) vector."""
    conv = _conv_self(a)
    maxb = conv.max()
    # a is already normalised, so sum_a == 1
    return 2.0 * a.shape[0] * maxb


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) sub‑gradient descent for
        f(a) = max_k (a*a)[k] .

    Returns the best vector seen and the corresponding 2·n·max_conv value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:6.2f}s] pgd iter={it:,} score={cur_score:.6f}")

        # sub‑gradient: sum of gradients of all (almost) active convolutions
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of the coefficient that contributes most to
    the current peak towards the smallest coefficient, keeping the move only
    if it strictly lowers the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx

        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            # occasional report
            if time.time() - start_time > 0 and no_improve % 200 == 0:
                print(f"[{time.time() - start_time:6.2f}s] peak‑reduce max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Classic annealing: move a random ε‑fraction of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method performs many short random‑restart sub‑gradient runs,
    keeps the best candidate and finally refines it with longer polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4800 works well (the reference size)
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Fast scoring (identical to ``evaluate_sequence`` for valid inputs)
    # ------------------------------------------------------------------
    def fast_score(a: np.ndarray) -> float:
        return _fast_score(a)

    # ------------------------------------------------------------------
    # Initialise global best (maybe from the previous construction)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                if starter.shape[0] != n:
                    starter = _upsample(starter, n)
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0:
                    starter = starter / s
                else:
                    starter = np.full(n, 1.0 / n)
                sc = fast_score(starter)
                best_seq, best_score = starter, sc
                print(f"[{time.time() - start_time:6.2f}s] init from height_sequence_1 score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Random‑restart loop
    # ------------------------------------------------------------------
    restart_id = 0
    rng = np.random.default_rng()

    while True:
        now = time.time()
        if now >= deadline - 5.0:            # keep a few seconds for final polishing
            break

        restart_id += 1
        # ------- draw a random start -----------------
        method = rng.choice(['dirichlet', 'gaussian', 'exp', 'tri', 'uniform'])
        if method == 'dirichlet':
            concentration = rng.uniform(0.1, 4.0)
            a0 = rng.dirichlet(np.full(n, concentration))
        elif method == 'gaussian':
            sigma = rng.uniform(0.05, 0.3) * n
            mu = rng.uniform(0, n - 1)
            xs = np.arange(n)
            a0 = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
            a0 = a0 / a0.sum()
        elif method == 'exp':
            r = rng.uniform(0.95, 0.999)
            a0 = r ** np.arange(n, dtype=float)
            a0 = a0 / a0.sum()
        elif method == 'tri':
            half = n // 2
            tri = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)])
            a0 = tri.astype(float)
            a0 = a0 / a0.sum()
        else:  # uniform
            a0 = np.full(n, 1.0 / n)

        # ------- short sub‑gradient optimisation ---------
        # allocate most of the remaining budget to this restart (but cap it)
        time_left = deadline - now
        sub_deadline = now + min(20.0, time_left * 0.4)

        a_opt, sc_opt = _run_gradient_descent(
            a0,
            sub_deadline,
            start_time,
            beta=0.90,
            step0=0.5,
            max_iter=10**7
        )

        # ------- quick greedy peak reduction ------------
        peak_deadline = time.time() + min(1.5, (deadline - time.time()) * 0.1)
        a_opt, sc_opt = _run_peak_reduction(a_opt, peak_deadline, start_time,
                                            max_no_improve=500)

        # ------- brief annealing -------------------------
        anneal_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.2)
        a_opt, sc_opt = _run_annealing(a_opt, anneal_deadline, start_time,
                                       start_temp=1e-1, end_temp=1e-7,
                                       steps=80000)

        # ------- update global best -----------------------
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST restart {restart_id} *** score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing of the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Long projected‑gradient phase (uses remaining time)
        pgd_deadline = time.time() + max(0.0, deadline - time.time() - 5.0)
        a_ref, sc_ref = _run_gradient_descent(
            best_seq,
            pgd_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
            max_iter=10**7
        )
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final pgd improvement: score={best_score:.6f}")

        # final greedy pass
        peak_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.1)
        a_ref, sc_ref = _run_peak_reduction(best_seq, peak_deadline, start_time,
                                            max_no_improve=1000)
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final peak‑reduce improvement: score={best_score:.6f}")

        # final annealing (use whatever time remains)
        if time.time() < deadline:
            a_ref, sc_ref = _run_annealing(best_seq, deadline, start_time,
                                           start_temp=5e-2, end_temp=1e-7,
                                           steps=200_000)
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] final annealing improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return as plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
