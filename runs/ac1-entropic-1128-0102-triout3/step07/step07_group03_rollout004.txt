"""
Propose a step‑function that lowers the self‑convolution peak.
The search consists of three phases:

1. Multi‑scale warm‑start + Adam optimisation on a smooth max‑conv
   (soft‑max approximation).  Early stages use short vectors to obtain
   a good shape quickly; the final stage works on the full length
   (default 9600).  Adam’s per‑coordinate step–size adaption is more
   robust than plain Nesterov momentum.

2. Refined sub‑gradient descent on the true max‑convolution.  This
   directly reduces the active diagonal using a decreasing step size.

3. Stochastic “peak‑shaving’’ walk (simulated annealing).  At each
   iteration a small amount of mass is transferred from indices that
   contribute most to the current maximal convolution entry to indices
   that contribute least; uphill moves are accepted with a temperature
   schedule.

All iterates are projected onto the probability simplex
({x≥0, ∑x=1}), which makes the evaluation function
`evaluate_sequence` scale‑invariant.  The routine respects the 1000 s
budget and prints intermediate scores.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cs = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cs / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cs[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    # next power‑of‑two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        Σ_k softmax_k(conv/temperature)·conv_k
    w.r.t. a.
    """
    maxc = conv.max()
    # shift by max for numerical stability
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                # softmax weights over the diagonal indices
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

# Optimisation phases
# ----------------------------------------------------------------------
def _run_adam_gd(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 lr: float = 0.12,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimisation on the soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # temperature schedule (same as in the original script)
    T_start, T_end = 2.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # annealed temperature for the smooth max
        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)

        # Adam update (projected onto simplex afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.6,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 8) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent directly on the max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve: n={n}, it={it}, score={cur_score:.6f}")

        # sub‑gradient of the active diagonal
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # momentum & diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _run_local_search(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing walk that shaves the current maximal diagonal."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # annealing schedule for the Metropolis criterion
    T_start, T_end = 5e-2, 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to the active diagonal
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # pick a donor index proportional to its contribution
        if contrib.sum() > 0:
            donor = np.random.choice(n, p=contrib / contrib.sum())
        else:
            donor = np.random.randint(n)

        # pick a recipient from the low‑contributing tail
        low_thr = np.percentile(contrib, 20)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size == 0:
            rec = np.random.randint(n)
        else:
            rec = np.random.choice(low_cands)

        if donor == rec or a[donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of the donor's mass
        eps = a[donor] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature as a function of elapsed fraction
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed
        accept = (delta < 0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Local improve: n={n}, it={it}, score={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine performs:
        1. Multi‑scale Adam optimisation on a soft‑max surrogate.
        2. Sub‑gradient refinement on the true max‑convolution.
        3. A simulated‑annealing peak‑shaving walk.
    The best sequence found within the time budget is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start (previous best if available)
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Phase 1 – progressive Adam optimisation (soft‑max)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]          # final length = 9600
    total_len_weight = sum(stage_lengths)

    total_time = deadline - start_time
    adam_budget = total_time * 0.55
    subgrad_budget = total_time * 0.30
    local_budget = total_time - adam_budget - subgrad_budget

    best_seq = None
    best_score = float("inf")

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time for the current stage (more weight to later stages)
        stage_frac = n / total_len_weight
        stage_time = max(2.0, adam_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        a_opt, score_opt = _run_adam_gd(
            a0, n, stage_deadline, start_time,
            lr=0.12, beta1=0.9, beta2=0.999, eps=1e-8,
            max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Stage n={n} Adam improve: score={score_opt:.6f}")

        # update remaining Adam budget
        elapsed = time.time() - start_time
        adam_budget = max(0.0, deadline - start_time - subgrad_budget - local_budget - elapsed)

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on the final length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        sub_deadline = deadline - local_budget * 0.5   # keep half of local budget for the walk
        a_refined, score_refined = _run_subgradient_gd(
            best_seq, n_final, sub_deadline, start_time,
            step0=0.6, beta=0.5, max_iter=10 ** 8
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve: score={score_refined:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _run_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final local improve: score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may occur because of floating errors)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (the evaluation harness expects a list)
    return [float(x) for x in best_seq]
