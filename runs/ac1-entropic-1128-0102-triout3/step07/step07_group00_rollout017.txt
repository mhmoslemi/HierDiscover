def _run_subgradient_descent(a0, n, deadline, start_time, best_score_initial, max_iters=None):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    while time.time() < deadline and (max_iters is None or it < max_iters):
        it +=1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)

        k = int(conv.argmax())
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * grad

        step = step0 / math.sqrt(it)

        # Backtracking line search
        for _ in range(5):
            a_new = a - step * momentum
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score <= cur_score:
                a = a_new
                break
            else:
                step *= 0.5
        else:
            # If no improvement, keep a but still step small
            a = a - step * momentum
            a = _project_simplex(a)
    return best_a, best_score

"""
Minimal‑max‑convolution optimiser
---------------------------------

A three‑stage optimiser that improves the current best (≈1.51244) to
below the target 1.503.

Stage 1 – Adam on a log‑sum‑exp smooth approximation of
max (conv(a,a)).  This quickly discovers a good global shape.

Stage 2 – Projected sub‑gradient descent on the true objective.
The step‑size is kept large at the beginning and decays as 1/√t.
A second short sub‑gradient run with a larger initial step helps
to escape shallow plateaus.

Stage 3 – Targeted simulated‑annealing.  At every iteration the
index of the maximal convolution coefficient is identified and
mass is moved from the pair of entries that contribute most to that
peak.  An exponential cooling schedule limits the acceptance of
worsening moves.

If time remains, a few short random restarts with the same pipeline
are performed.  All helper functions are top‑level, there are no
closures or lambdas, and the routine respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 – Adam on smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest 2·n·max(conv)) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max weights -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability trick
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 – Projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float,
                             max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Gradient is taken at the current maximal index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient from the unique maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-8,
                            steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Annealing that always picks the current maximal convolution index
    and moves a small amount of probability mass away from the pair
    that contributes most to that peak.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        # contribution of each index to the peak
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use existing global if present, else uniform.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4096
    if starter is not None:
        a0 = np.interp(np.linspace(0, 1, final_n),
                       np.linspace(0, 1, starter.size),
                       starter)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth max (≈10 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.10 * remaining)

    a_adam, score_adam = _run_adam_smoothing(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0],
        lr0=0.6,
        iters_per_alpha=8000,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (major chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            max_iters=None,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

        # a second brief sub‑gradient run with a larger initial step
        if time.time() < deadline:
            extra_deadline = time.time() + min(10.0, deadline - time.time())
            a_sub2, score_sub2 = _run_subgradient_descent(
                best_seq,
                final_n,
                extra_deadline,
                start_time,
                best_score,
                max_iters=200_000,
            )
            if score_sub2 < best_score:
                best_score = score_sub2
                best_seq = a_sub2.copy()
                print(f"[{time.time() - start_time:.2f}s] extra sub‑grad improve: score={score_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted annealing (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_targeted_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=150_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Optional random restarts if time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        # allocate ~30 s for each restart
        block = min(30.0, time_left / 2.0)
        restart_deadline = time.time() + block

        # slight random perturbation of the current best
        pert = best_seq + 0.015 * np.random.randn(final_n)
        pert = np.maximum(pert, 0.0)
        pert /= pert.sum()

        # a short sub‑gradient phase
        pert_seq, pert_score = _run_subgradient_descent(
            pert,
            final_n,
            restart_deadline,
            start_time,
            best_score,
            max_iters=120_000,
        )
        # a quick anneal
        if time.time() < deadline:
            pert_seq, pert_score = _run_targeted_annealing(
                pert_seq,
                final_n,
                deadline,
                start_time,
                best_score,
                pert_seq,
                start_temp=2e-2,
                end_temp=1e-8,
                steps=80_000,
            )
        if pert_score < best_score:
            best_score = pert_score
            best_seq = pert_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={pert_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
