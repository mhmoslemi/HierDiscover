"""
Propose a candidate step‑function that improves the max‑self‑convolution bound.

Algorithm Overview
------------------
1. **Generate a diverse set of well‑known window shapes** (uniform,
   triangular, Hann, Hamming, Blackman, Kaiser, Gaussian, exponential
   etc.) as well as a possible previous best (`height_sequence_1`) and a few
   random vectors.  All are normalised to sum = 1.

2. **Pick the best initial shape** by evaluating the true objective
   (`evaluate_sequence`).  This gives a good starting point that is already
   close to the optimum.

3. **Smooth optimisation** – minimise a soft‑max surrogate of the
   max‑convolution using Adam.  The surrogate
   `Oτ(a)=2·n·τ·log Σ exp(conv(a)/τ)` is differentiable; we project after
   every update onto the probability simplex.

4. **Exact optimisation** – projected sub‑gradient descent on the true
   max‑convolution (`F(a)=2·n·max(conv(a))`).  A Nesterov‑style momentum and
   a decreasing stepsize (`~1/√t`) are used.

5. **Escape local minima** – a lightweight simulated‑annealing phase that
   moves a random fraction of mass between two cells.

6. **Optional up‑sampling** – if time permits, the current best vector is
   linearly interpolated to a larger size (≈16384) and the whole
   optimisation pipeline is run briefly again for a finer representation.

All steps respect the global time budget (default 1000 s).  The final
vector is projected onto the simplex and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the final polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam on the soft‑max (log‑sum‑exp) surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise   Oτ(a)=2·n·τ·log∑exp(conv(a)/τ)
    using Adam and projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # exact (non‑smoothed) score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ------- soft‑max weights -------
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stabilise
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ------- gradient of Oτ -------
        # grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # needed segment
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on true max‑conv
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (∑a=1, a≥0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _generate_window_candidates(n: int):

    candidates = []

    # uniform
    uni = np.full(n, 1.0 / n)
    candidates.append(("uniform", uni))

    # triangular (Bartlett)
    tri = np.bartlett(n)
    candidates.append(("triangular", tri / tri.sum()))

    # Hann (np.hanning)
    hann = np.hanning(n)
    candidates.append(("hann", hann / hann.sum()))

    # Hamming
    hamming = np.hamming(n)
    candidates.append(("hamming", hamming / hamming.sum()))

    # Blackman
    blackman = np.blackman(n)
    candidates.append(("blackman", blackman / blackman.sum()))

    # Kaiser with a range of betas
    for beta in [0.5, 2, 4, 6, 8, 10]:
        k = np.kaiser(n, beta)
        candidates.append((f"kaiser_{beta}", k / k.sum()))

    # Gaussian windows (different sigmas)
    for sigma in [0.15, 0.25, 0.35, 0.45]:
        x = np.arange(n)
        center = (n - 1) / 2.0
        g = np.exp(-0.5 * ((x - center) / (sigma * n / 2)) ** 2)
        candidates.append((f"gaussian_{sigma}", g / g.sum()))

    # Exponential decay windows
    for scale in [0.08, 0.12, 0.16, 0.20]:

        center = (n - 1) / 2.0
        e = np.exp(-np.abs(x - center) / (scale * n))
        candidates.append((f"exp_{scale}", e / e.sum()))

    return candidates

    for i in range(3):
        rand_vec = np.random.rand(n)
        rand_vec /= rand_vec.sum()
        candidates.append((f"random_{i}", rand_vec))

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best starter
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for name, seq in candidates:
        if time.time() >= deadline:
            break
        # evaluate using the provided function (handles clipping etc.)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] init {name:12s} score={score:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()
        print("Fallback to uniform vector.")

    # ------------------------------------------------------------------
    # Stage 0 – Adam on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        conv0 = _conv_self(best_seq)
        tau0 = max(conv0.max() * 10.0, 1e-8)
        # allocate up to ~30 % of the remaining budget to this stage
        remaining = deadline - time.time()
        adam_time = min(remaining * 0.3, 250.0)
        adam_deadline = time.time() + adam_time

        best_seq, score = _run_adam_softmax(
            best_seq,
            n,
            adam_deadline,
            start_time,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.18,
        )
        if score < best_score:
            best_score = score
            print(f"[{time.time() - start_time:6.2f}s] after Adam best score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(10.0, remaining * 0.5)  # use most of the rest
        best_seq, score = _run_subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            beta=0.94,
            step0=0.13,
        )
        if score < best_score:
            best_score = score
            print(f"[{time.time() - start_time:6.2f}s] after Subgrad best score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Light simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # number of moves scales with the remaining time budget
        steps = int(300_000 * max(0.4, (deadline - time.time()) / (budget_s * 0.4)))
        steps = max(100_000, min(600_000, steps))
        best_seq, score = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=5e-8,
            steps=steps,
        )
        if score < best_score:
            best_score = score
            print(f"[{time.time() - start_time:6.2f}s] after Anneal best score={score:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sample & brief refinement (if a good chunk of time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        if time_left > 120.0:  # only if we have at least 2 minutes left
            n_big = 16384
            seq_big = _upsample(best_seq, n_big)

            # quick Adam on the larger grid
            conv_big = _conv_self(seq_big)
            tau0_big = max(conv_big.max() * 10.0, 1e-8)
            adam_deadline_big = time.time() + min(30.0, time_left * 0.3)
            seq_big, score_big = _run_adam_softmax(
                seq_big,
                n_big,
                adam_deadline_big,
                start_time,
                tau0=tau0_big,
                tau_end=1e-13,
                tau_decay=0.9996,
                step_size=0.12,
            )

            # short sub‑gradient polish
            if time.time() < deadline:
                sub_deadline_big = time.time() + min(45.0, time_left * 0.3)
                seq_big, score_big = _run_subgradient_descent(
                    seq_big,
                    n_big,
                    sub_deadline_big,
                    start_time,
                    beta=0.94,
                    step0=0.12,
                )

            if score_big < best_score:
                best_score = score_big
                best_seq = seq_big.copy()
                print(f"[{time.time() - start_time:6.2f}s] upsample refinement score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
