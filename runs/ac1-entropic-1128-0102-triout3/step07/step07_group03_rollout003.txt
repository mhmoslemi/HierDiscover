        contrib = np.zeros(n)
        contrib[valid] = a[i_arr[valid]] * a[j_arr[valid]]

        donor = np.argmax(contrib)

        rec_candidates = rec_candidates[rec_candidates != donor]

def _subgrad_adam(a0, n, max_iter, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best = a.copy()
    best_score = 2*n*_conv_self(a).max()
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2*n*conv[idx_max]
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        mask = (j >= 0) & (j < n)

def _greedy_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2*n*conv.max()
    best = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1
        # recompute conv and idx_max each iteration
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2*n*conv[idx_max]

        # contributions from each i to this shift
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        donor = int(np.argmax(contrib))
        # donor amount avail
        if a[donor] <= 1e-12:
            continue
        # recipient: minimal contribution excluding donor
        # sort by contribution
        low_idx = np.where(contrib <= np.percentile(contrib, 15))[0]  # bottom 15%
        low_idx = low_idx[low_idx != donor]
        if low_idx.size == 0:
            rec = np.random.randint(n)
        else:
            rec = int(np.random.choice(low_idx))

"""
Propose a step‑function (a probability vector) that lowers
the self‑convolution peak.
The routine mixes many short Adam‑based sub‑gradient runs with a
deterministic greedy “peak‑shaving” walk.  Several random restarts
explore the non‑convex landscape; the final walk focuses on the
indices that contribute most to the current maximal convolution
entry.  The whole search respects a 1000 s budget and returns a
list of non‑negative floats that sum to one (the scaling does not
matter for ``evaluate_sequence``).
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution (linear, full) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple (fast) projection onto the probability simplex:
#   set negatives to zero and renormalise.
# ----------------------------------------------------------------------
def _renorm_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to >=0 and re‑scale to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:                                   # degenerate – fall back to uniform
        v.fill(1.0 / v.size)
    return v


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent on  max(conv)
# ----------------------------------------------------------------------
def _subgrad_adam(a0: np.ndarray,
                  n: int,
                  max_iter: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """Adam on the sub‑gradient of the active convolution shift."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max]

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}s] Adam improve: iter={t} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the active shift
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        grad[i[mask]] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.4 / (np.sqrt(v_hat) + eps)          # base step size
        a = a - step * m_hat
        a = _renorm_simplex(a)

    # final check
    final_score = 2.0 * n * _conv_self(a).max()
    if final_score < best_score:
        best_score = final_score
        best = a.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Greedy „peak‑shaving“ local search
# ----------------------------------------------------------------------
def _greedy_local_search(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Move mass from the biggest contributors of the active shift to
    the weakest contributors; accept uphill moves with a slowly cooling
    temperature."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max]

        # contributions of each index to this shift
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        donor = int(np.argmax(contrib))
        if a[donor] <= 1e-14:
            # nothing to give – pick a random donor
            donor = np.random.randint(n)

        # recipient – pick among the low‑contribution tail
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_candidates = np.where(contrib <= low_thr)[0]
        low_candidates = low_candidates[low_candidates != donor]
        if low_candidates.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = int(np.random.choice(low_candidates))

        # move a fraction of the donor's mass
        eps = a[donor] * np.random.rand() * 0.35   # ≤ 35 % of donor
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps
        a_prop = _renorm_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        # temperature schedule (geometric decay)
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T_start, T_end = 5e-2, 1e-6
        T = T_start * (T_end / T_start) ** elapsed

        accept = (delta < 0.0) or (np.exp(-delta / (T + 1e-12)) > np.random.rand())
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] greedy local improve: "
                      f"iter={it}, score={new_score:.6f}")

        # optional early stop if we already beat the target
        if best_score < 1.503:
            break

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The strategy is:
      1.  Use the previous best (global ``height_sequence_1``) if it exists.
      2.  Run many short Adam‑based sub‑gradient runs from random
          Dirichlet starts (different concentration parameters) to
          explore the landscape.
      3.  Keep the best vector found.
      4.  Refine it with a greedy peak‑shaving local walk for the
          remaining time.
    The routine respects the 1000 s budget and returns a plain Python
    list of floats (sum approximately 1, all non‑negative).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start: reuse the previously discovered sequence if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length selection – use the length of the warm‑start if present,
    # otherwise fall back to 9600 (the length that gave good results before)
    # ------------------------------------------------------------------
    if starter is not None:
        n = starter.size
    else:
        n = 9600

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Allocate time: 65 % for many Adam restarts, 35 % for final greedy walk
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    sub_budget = total_time * 0.65
    local_budget = total_time - sub_budget
    sub_start = time.time()

    restart = 0
    # each restart gets roughly 3–4 s (enough for ~2000 Adam steps)
    while time.time() - sub_start < sub_budget:
        remaining = sub_budget - (time.time() - sub_start)
        if remaining < 2.0:
            break
        per_restart_deadline = time.time() + min(4.0, remaining)

        # ------------------------------------------------------------------
        # Random initialisation – Dirichlet with a random concentration.
        # Small concentration → sparse start, large → near‑uniform.
        # ------------------------------------------------------------------
        if restart == 0 and starter is not None:
            a0 = starter.copy()
        else:
            alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])   # explore sparsity vs uniformity
            a0 = np.random.gamma(alpha, 1.0, size=n)
            a0 = _renorm_simplex(a0)

        # ------------------------------------------------------------------
        # Adam sub‑gradient run (fixed number of iterations)
        # ------------------------------------------------------------------
        a_opt, score_opt = _subgrad_adam(
            a0, n, max_iter=3000, deadline=per_restart_deadline
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart {restart} new best: "
                  f"score={score_opt:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely) fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final greedy peak‑shaving walk – uses the remaining time budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined, score_refined = _greedy_local_search(
            best_seq, n, deadline, start_time
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] final greedy improvement: "
                  f"score={score_refined:.6f}")

    # ------------------------------------------------------------------
    # Final sanity normalisation (tiny drift may have occurred)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list – the harness expects a list of floats
    return [float(x) for x in best_seq.tolist()]
