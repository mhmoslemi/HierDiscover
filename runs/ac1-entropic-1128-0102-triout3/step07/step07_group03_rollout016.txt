def _run_deterministic_shave(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    while time.time() < deadline:
        idx_max = int(np.argmax(conv))
        # compute contributions for this shift
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        # donor: index with max contribution and a[i] > small epsilon
        donors = np.where((contrib > 0) & (a > 1e-12))[0]
        if donors.size == 0:
            break
        i_donor = donors[np.argmax(contrib[donors])]
        # recipient: index with minimal contribution
        recipients = np.where(a > 0)[0]  # all indices have maybe zero but we want positive
        # we prefer low contribution among indices not same as donor
        # compute contributions for others
        low_indices = np.where(contrib <= np.percentile(contrib, 20))[0]
        # Exclude i_donor
        low_indices = low_indices[low_indices != i_donor]
        if low_indices.size == 0:
            i_rec = np.random.choice(np.arange(n))
        else:
            i_rec = low_indices[np.argmin(contrib[low_indices])]
        # epsilon: transfer up to a[i_donor] * 0.5

"""
Propose a step‑function with a low self‑convolution peak.

Algorithm
---------
1. Initialise a probability vector (sum = 1).  If a previous best
   ``height_sequence_1`` is available it is up‑sampled; otherwise a random
   Dirichlet vector is used.
2. Perform several *restart* cycles (time‑budgeted).  Each cycle perturbs the
   current best vector, then runs
      * projected sub‑gradient descent on the true max‑convolution,
        keeping the best iterate;
      * a deterministic “peak‑shaving’’ walk that repeatedly transfers mass
        from the index that contributes most to the current maximal convolution
        entry to an index that contributes least.
   The best vector seen over all cycles is retained.
3. With the remaining time a stochastic Metropolis local search
   (identical to the one used in the previous state‑of‑the‑art) is run for a
   final polishing.
4. The vector is re‑normalised onto the simplex and returned as a plain
   Python list.

All heavy work (convolutions, projections, up‑sampling) uses NumPy/FFT and
fits comfortably into the 1000 s budget.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (projected onto simplex)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.5,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_max = conv[idx_max]
        cur_score = 2.0 * n * cur_max          # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the active shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # momentum & step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic peak‑shaving walk (greedy mass transfer)
# ----------------------------------------------------------------------
def _run_deterministic_shave(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Greedy local search: move mass from the biggest contributor to
    the smallest contributor of the current maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to the maximal entry
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        # donor = index with largest contribution and non‑zero mass
        donors = np.where((contrib > 0) & (a > 1e-15))[0]
        if donors.size == 0:
            break
        i_donor = donors[np.argmax(contrib[donors])]

        # recipient = index with smallest contribution (prefer low‑mass zones)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        low_idx = low_idx[low_idx != i_donor]
        if low_idx.size == 0:
            # fallback: random different index
            i_rec = (i_donor + np.random.randint(1, n)) % n
        else:
            i_rec = low_idx[np.argmin(contrib[low_idx])]

        if a[i_donor] <= 1e-15:
            continue

        # move up to 30 % of donor mass (smaller steps help stability)
        eps = a[i_donor] * np.random.rand() * 0.3
        a[i_donor] -= eps
        a[i_rec]   += eps

        # keep on simplex (sum is preserved, only non‑negativity may be violated)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        conv = _conv_self(a)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] deterministic improve: "
                  f"n={n}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted stochastic local search (the previous implementation)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis walk that shaves peaks of the max‑convolution."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the maximal entry
        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        # donor: weighted by contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: from low‑contribution region
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] stochastic improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine performs several restart cycles, each consisting of
    sub‑gradient descent and a deterministic peak‑shaving walk, and finishes
    with a stochastic Metropolis local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin to avoid hard timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Target length – a larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n_final = 12000                     # can be increased if time permits

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if it exists.
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = np.maximum(tmp, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
            else:
                starter = None
    except Exception:
        starter = None

    if starter is not None:
        best_seq = _upsample(starter, n_final)
    else:
        # random Dirichlet start
        best_seq = np.random.rand(n_final)
        best_seq /= best_seq.sum()

    # compute its initial score
    cur_conv = _conv_self(best_seq)
    best_score = 2.0 * n_final * cur_conv.max()
    print(f"[0.00s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Restart loop – each cycle tries to beat the current best.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate ~70 % of the total budget for restart cycles,
    # keep the rest for the final stochastic walk.
    restart_budget = remaining * 0.70
    n_restarts = 4                                 # 4‑5 restarts fit well
    restart_time = restart_budget / n_restarts

    for r in range(n_restarts):
        if time.time() >= deadline - 20:     # keep a small margin for final phase
            break

        # small random perturbation of the current best
        a0 = best_seq + np.random.normal(scale=0.02, size=n_final)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0 / n_final)
        else:
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Sub‑gradient refinement (most of the restart time)
        # --------------------------------------------------------------
        sub_deadline = time.time() + restart_time * 0.60
        a_ref, score_ref = _run_subgradient_gd(
            a0, n_final, sub_deadline, start_time,
            step0=0.5, beta=0.5, max_iter=10 ** 7
        )

        # --------------------------------------------------------------
        # Deterministic peak‑shaving greedy walk (the rest of the restart)
        # --------------------------------------------------------------
        local_deadline = time.time() + restart_time * 0.40
        a_loc, score_loc = _run_deterministic_shave(
            a_ref, n_final, local_deadline, start_time
        )

        # --------------------------------------------------------------
        # Keep the best sequence seen in this restart
        # --------------------------------------------------------------
        for seq, sc in ((a_ref, score_ref), (a_loc, score_loc)):
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart {r} improve: "
                      f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final stochastic polishing with the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_fin, score_fin = _run_targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_fin < best_score:
            best_score = score_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:6.2f}s] final improve: score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (tiny drift may have occurred)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] finished – final score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
