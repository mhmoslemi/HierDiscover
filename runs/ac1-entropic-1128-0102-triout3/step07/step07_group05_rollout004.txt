```python
def _adam_smooth_max(a0, n, deadline, start_time, best_score, best_seq):
    a = a0.copy()
    # create logits y from a to keep positivity
    y = np.log(np.maximum(a, 1e-16))
    fft_len = _next_pow2(3 * n - 2)
    # schedule of alphas: geometric series
    alphas = np.geomspace(0.5, 400.0, num=12)
    iters_per_alpha = 20000  # increased
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break
            # compute a from logits
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()
            conv = _conv_self(a)
            # smoothed max objective
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradients
            grad_conv = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)
            # gradient w.r.t y
            grad_y = a * (grad_conv - np.dot(grad_conv, a))
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            # periodic evaluation
            if i % 500 == 0:
                sc = 2.0 * n * _conv_self(a).max()
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")
        if time.time() >= deadline:
            break
    # final a
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq

def _simplex_projection(v):
    # Projects onto simplex sum=1, v >=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
```

def _activity_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq, total_steps=2_000_000):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    start_temp = 0.05
    end_temp = 1e-7
    update_interval = 2000
    src_prob = a / a.sum()  # initial
    dst_prob = np.ones(n) / n
    it = 0
    while time.time() < deadline and it < total_steps:
        # recompute activity every update_interval steps
        if it % update_interval == 0:
            M = conv.max()
            tol = 1e-12
            # active constraints
            active = np.where(conv >= (1 - tol) * M)[0]
            # compute activity via gradient for max conv
            act = np.zeros(n)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                i = np.arange(lo, hi + 1, dtype=int)
                j = k - i
                act[i] += a[j]
                act[j] += a[i]
            # ensure nonzero
            src_prob = act + 1e-16
            src_prob /= src_prob.sum()
            # low activity for dst:
            max_act = act.max()
            dst_weight = max_act - act + 1e-16
            dst_prob = dst_weight / dst_weight.sum()
        it += 1
        # temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)
        # pick src,dst
        src = rng.choice(n, p=src_prob)
        dst = rng.choice(n, p=dst_prob)
        if src == dst or a[src] < 1e-16:
            continue
        eps = a[src] * rng.random() * 0.30
        # apply move with conv update
        # subtract contribution from src
        conv[src:src + n] -= 2.0 * eps * a
        # add contribution from dst
        conv[dst:dst + n] += 2.0 * eps * a
        # self terms
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps
        # update a
        a[src] -= eps
        a[dst] += eps
        new_score = 2.0 * n * conv.max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # revert
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")
    return best_seq, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser consists of three stages, all respecting a 1000 s budget.

1. **Adam on a smooth max‑convolution** –  a log‑sum‑exp surrogate of the
   max‑convolution is minimised.  The smoothing parameter α is increased
   gradually, allowing the optimiser to first move in a smooth landscape
   and then focus on the true max.  Gradients are obtained via FFTs, and the
   variables are kept on the probability simplex by a soft‑max re‑parameterisation.

2. **Projected sub‑gradient polishing** –  the exact (non‑smoothed) objective is
   polished with a sub‑gradient method that projects onto the simplex after
   each step (Euclidean projection).  This removes any remaining bias from the
   smooth surrogate.

3. **Activity‑guided simulated annealing** –  mass is transferred from indices
   that heavily contribute to the current maximal convolution (high activity)
   to indices that contribute little (low activity).  The convolution is updated
   incrementally in O(n) per move, and a geometric temperature schedule accepts
   occasional uphill moves.

The best sequence seen during the whole run is returned (normalised to sum = 1).
All numbers are non‑negative, satisfying the checks in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a) = (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – Adam optimisation on a smoothed objective
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the log‑sum‑exp surrogate of max‑convolution.
    Returns the final vector and updates the global best.
    """
    a = a0.copy()
    # work in logits to keep a on the simplex
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # geometric schedule for the smoothing parameter α
    alphas = np.geomspace(0.5, 400.0, num=12)
    iters_per_alpha = 15000                     # more iterations than before

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.15

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # recover a from logits
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            mscaled = scaled.max()
            lse = mscaled + np.log(np.exp(scaled - mscaled).sum())
            f_val = lse / alpha          # ≈ max(conv) when α is large

            # gradient w.r.t a of the smoothed max (then scale by 2·n)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t logits (soft‑max Jacobian)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation (using the true objective)
            if i % 500 == 0:
                score = 2.0 * n * _conv_self(a).max()
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, "
                          f"iter={i:,}, score={score:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Phase 2 – Exact projected (sub‑)gradient descent
# ----------------------------------------------------------------------
def _projected_subgradient(a0: np.ndarray, n: int,
                           deadline: float, start_time: float,
                           beta: float = 0.94,
                           step0: float = 0.25,
                           max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution (no smoothing).
    Returns the best vector found and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # identify active constraints (within relative tolerance)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]          # each product contributes to both factors

        # Nesterov‑style momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step‑size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _simplex_projection(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Activity‑guided simulated annealing
# ----------------------------------------------------------------------
def _activity_annealing(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        total_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing that preferentially moves probability mass from
    indices with high activity (i.e. they contribute strongly to the current
    maximal convolution) towards indices with low activity.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7
    update_interval = 2000                     # recompute activity every this many steps

    # start with uniform probability for the destination, activity‑based for source
    src_prob = np.full(n, 1.0 / n)
    dst_prob = src_prob.copy()

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # recompute activity vectors periodically
        if it % update_interval == 0:
            M = conv.max()
            tol = 1e-12
            active = np.where(conv >= (1.0 - tol) * M)[0]

            act = np.zeros(n)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                i = np.arange(lo, hi + 1, dtype=int)
                j = k - i
                act[i] += a[j]
                act[j] += a[i]

            # source: proportional to activity (high activity → higher chance)
            src_prob = act + 1e-12
            src_prob /= src_prob.sum()

            # destination: proportional to *low* activity
            max_act = act.max()
            dst_weight = (max_act - act) + 1e-12
            dst_prob = dst_weight / dst_weight.sum()

        # geometric cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # propose a move
        src = rng.choice(n, p=src_prob)
        dst = rng.choice(n, p=dst_prob)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score
