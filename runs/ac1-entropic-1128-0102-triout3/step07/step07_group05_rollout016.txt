def _cross_corr_grad(a, w, fft_len):
    # w length maybe 2n-1
    # compute a cross-correlation a ⋆ w
    # a shape (n,)
    a_fft = np.fft.rfft(a, n=fft_len)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(a_fft * w_fft, n=fft_len)
    # length of corr = fft_len, but we only need first n + len(w_rev) -1 entries
    # Actually corr as returned will have length fft_len, but beyond needed region it contains wrap-around due to using rfft without zero padding to length >= n+len(w)-1; but we did pad to fft_len >= n + len(w)-1.
    # So we need first n + len(w_rev) -1 entries:
    needed_len = n + len(w_rev) - 1
    corr = corr[:needed_len]
    grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]
    return grad

"""
search for a low‑autocorrelation (step) function.

The optimiser consists of three stages, all respecting the 1000 s budget:

1. **Adam on a smoothed maximum** – the objective
   f(a)=max(a*a) is approximated by a log‑sum‑exp.
   Parameters are kept on the probability simplex via a soft‑max of logits.
   The smoothing parameter α is increased gradually.

2. **Exact projected sub‑gradient polishing** – after the Adam phase the true
   non‑smoothed maximal convolution is minimised by sub‑gradient descent with
   Nesterov momentum on the simplex.

3. **Guided incremental simulated‑annealing** – a mass‑transfer annealer.
   In each move the source and destination indices are chosen from the
   sub‑gradient of the *current* maximal convolution (instead of uniformly at
   random).  The convolution is updated incrementally in O(n), allowing many
   cheap moves.  A geometric temperature schedule accepts occasional uphill
   moves.

All numbers are forced to be non‑negative; the final vector is normalised to
unit sum (the evaluation is scale‑invariant).  The best sequence found during
the whole run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the (exact) max convolution (binary mask)
# ----------------------------------------------------------------------
def _grad_max_subgradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  sum_{k} mask[k]·conv_k  (conv = a*a) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)        # enough room for a (len n) + mask (len 2n‑1)
    mask_rev = mask[::-1]
    Af = np.fft.rfft(a, n=L)
    Mf = np.fft.rfft(mask_rev, n=L)
    corr = np.fft.irfft(Af * Mf, n=L)[: n + len(mask_rev) - 1]
    grad = 2.0 * corr[len(mask_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Runs Adam on the smoothed log‑sum‑exp objective.
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    # initialise logits for the soft‑max
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # schedule: higher α gives a tighter approximation of max
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 7000

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max → current a
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution / smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient wrt logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Exact projected gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 5_000_000):
    """
    Sub‑gradient descent on the exact max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active indices (within tiny relative tolerance)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # compute sub‑gradient for the active set
        mask = np.zeros_like(conv)
        mask[active] = 1.0
        g = _grad_max_subgradient(a, mask)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score

                          total_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing guided by the sub‑gradient of the current maximal
    convolution entry.  The convolution is updated incrementally in O(n) per
    move, so many steps can be performed within the remaining time.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    for it in range(total_steps):
        if time.time() >= deadline:
            break

        # ----- compute sub‑gradient for the current worst offset -----
        max_val = conv.max()
        # tolerance – treat values indistinguishable from the maximum as active
        active = np.where(conv >= max_val * (1.0 - 1e-12))[0]
        mask = np.zeros_like(conv)
        mask[active] = 1.0
        grad = _grad_max_subgradient(a, mask)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if src == dst or a[src] < 1e-12:
            # fallback to a random move (still respecting the mass‑transfer rule)
            src = int(rng.choice(n, p=a))
            dst = int(rng.integers(n))
            if src == dst or a[src] < 1e-12:
                continue

        # move up to 30 % of the source mass
        eps = a[src] * 0.30
        if eps <= 0.0:
            continue

        # ----- apply move (incremental convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ---- reject – restore the previous state ----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] guided anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The method blends Adam smoothing, exact sub‑gradient polishing and a
    gradient‑guided annealing phase, all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid overrunning the budget
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                a0 = _upsample(a0, n)
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{0.0:6.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed max‑convolution
    # ------------------------------------------------------------------
    a_adam, best_score, best_seq = _adam_phase(
        best_seq, n, deadline, start_time, best_score, best_seq
    )

    # ------------------------------------------------------------------
    # Phase 2 – Exact projected gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Guided incremental annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_guided_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            total_steps=300_000,          # as many steps as the remaining budget allows
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final guided anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final safety check (optional)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    # return as plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
