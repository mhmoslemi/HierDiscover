Adam algorithm:

Initialize m = 0, v = 0, t = 0.

At each iteration compute subgradient g (maybe via _subgradient_full, which identifies active set with tolerance). Then update:

t += 1

m = beta1 * m + (1 - beta1) * g

v = beta2 * v + (1 - beta2) * (g**2)

m_hat = m / (1 - beta1**t)

v_hat = v / (1 - beta2**t)

update = -lr * m_hat / (np.sqrt(v_hat) + epsilon)

def _run_incremental_refine(a0, n, deadline, start_time, eps_start=0.05, eps_end=0.001, temp_start=0.1, temp_end=0.001):
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    total_iters = 0

    # We'll schedule epsilon and temperature decreasing linearly with time.
    while time.time() < deadline:
        total_iters += 1
        # compute elapsed fraction
        t_frac = (time.time() - start_time) / (deadline - start_time)
        eps_frac = eps_start * (1 - t_frac) + eps_end * t_frac   # linear decay
        temp = temp_start * (1 - t_frac) + temp_end * t_frac
        # locate current max shift
        k_max = int(np.argmax(conv))
        # contributions vector
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i = np.arange(lo, hi+1)
        j = k_max - i
        contrib = a[i] * a[j]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)
        if a[src] <= 0:
            continue
        # choose destination:
        # choose uniformly among indices with low a (to spread)
        dst = rng.integers(n-1)
        if dst >= src: dst += 1
        # epsilon
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0: continue

        # copy conv for update
        conv_new = conv.copy()

        # second-order updates

        conv_new[idx2src] += eps*eps
        conv_new[idx2dst] += eps*eps
        conv_new[idxsum] -= 2*eps*eps
        # compute new max
        new_max = conv_new.max()
        # accept with probability
        accept = False
        if new_max < cur_max:

        if accept:

            a[dst] += eps
            # update conv and cur_max
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:

                best_a = a.copy()

"""
Improved step‑function search.

The algorithm consists of three stages:

1️⃣  Exponentiated‑gradient (soft‑max) warm‑up – quickly pushes the
    distribution away from the uniform shape while staying on the
    simplex.

2️⃣  Projected‑Adam sub‑gradient descent – uses the exact sub‑gradient
    of the max‑convolution (active constraints) together with Adam‑style
    adaptive step sizes and a projection onto the probability simplex.

3️⃣  Incremental local refinement with simulated‑annealing – moves a tiny
    amount of probability mass from indices that contribute most to the
    current maximal convolution entry to random destinations.  The
    convolution is updated *incrementally* (O(n) per move) which allows
    millions of trials inside the remaining time budget.  A temperature
    schedule lets the method escape shallow plateaus.

The whole routine respects the 1000 s budget, returns the best sequence
found and works without any filesystem or network access.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – exponentiated gradient on a smoothed max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 8000):
    """Multiplicative (mirror‑descent) updates using a smoothed max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – projected Adam on the exact max‑subgradient
# ----------------------------------------------------------------------


def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.25,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    """Adam‑style projected updates using the exact sub‑gradient of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: "
                  f"iter={t:,}, score={cur_score:.6f}")

        # exact sub‑gradient (active set)
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / math.sqrt(t)
        a_new = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – incremental local refinement with annealing
# ----------------------------------------------------------------------


def _run_incremental_refine(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            eps_start: float = 0.04,
                            eps_end: float = 0.001,
                            temp_start: float = 0.12,
                            temp_end: float = 0.001):
    """
    Tiny mass moves from indices that heavily contribute to the current
    maximal convolution entry to random destinations.  The convolution
    is updated incrementally (O(n)), allowing millions of trials.
    Moves are accepted if they improve the maximum, otherwise with a
    probability that decreases with a temperature schedule (simulated
    annealing).
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    it = 0
    recompute_interval = 20000  # avoid drift

    while time.time() < deadline:
        it += 1

        # linear schedules for epsilon and temperature
        frac = (time.time() - start_time) / (deadline - start_time)
        eps_frac = eps_start * (1.0 - frac) + eps_end * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # location (shift) of the current maximum
        k_max = int(np.argmax(conv))

        # contributions of each index to this maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k_max - i
        contrib = a[i] * a[j]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)

        if a[src] <= 1e-15:
            continue

        # destination – uniform among the other indices
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # amount of mass to move
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- incremental update of the convolution ----
        conv_new = conv.copy()
        # first‑order term: -2*eps*a around src, +2*eps*a around dst
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        # second‑order term (tiny)
        idx2src = 2 * src
        idx2dst = 2 * dst
        idxsum = src + dst
        conv_new[idx2src] += eps * eps
        conv_new[idx2dst] += eps * eps
        conv_new[idxsum] -= 2.0 * eps * eps
        # numerical safety
        # (negative values can appear only due to rounding)
        np.maximum(conv_new, 0.0, out=conv_new)

        new_max = conv_new.max()

        # acceptance test
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            # simulated annealing move
            delta = new_max - cur_max
            if temp > 0:
                prob = math.exp(-delta / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            # apply the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # periodic full recomputation to keep things accurate
        if it % recompute_interval == 0:
            conv = _conv_self(a)
            cur_max = conv.max()

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the provided time budget.
    The routine combines three optimisation phases and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a working length.  4800 works well; we keep it fixed.
    # --------------------------------------------------------------
    n = 4800

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if available.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, n)
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient warm‑up
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    stage1_deadline = time.time() + min(80.0, 0.10 * time_left)

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected Adam sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        stage2_deadline = time.time() + min(300.0, 0.70 * time_left)

        a_adam, score_adam = _run_adam_subgradient(
            best_seq,
            n,
            stage2_deadline,
            start_time,
            step0=0.25,
            beta1=0.9,
            beta2=0.999,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_adam:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – incremental local refinement with annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_incremental_refine(
            best_seq,
            n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.001,
            temp_start=0.12,
            temp_end=0.001,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and output
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
