def run_adam_subgradient(a0, n, deadline, start):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_obj(a,n)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # base learning rate
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2*n*max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] adam subgrad improve: iter={t:,}, score={cur_score:.6f}")
        # gradient for max index
        k = int(np.argmax(conv))
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        grad = np.zeros(n)
        grad[i] += a[j]
        grad[j] += a[i]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # projection onto simplex
        a = _project_simplex(a)
        # avoid numerical issues
        a = np.maximum(a, 0.0)
    return best_a, best_score

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm consists of four stages:
1.  Exponential‑gradient warm‑up on a smoothed version of the max‐convolution.
    The smoothing parameter α is increased gradually up to large values,
    while a multiplicative (entropy) update keeps the variables on the simplex.
2.  Adam‑based projected sub‑gradient descent that directly minimises the
    true max‑convolution.  The adaptive learning‑rate greatly speeds up the
    convergence compared with a fixed schedule.
3.  Targeted local refinement that moves a tiny amount of mass away from
    coordinates that contribute most to the current maximal convolution entry.
    Only strictly improving moves are kept – this guarantees monotonic progress.
4.  Occasional random restarts: a small Gaussian perturbation of the best
    solution is performed and a short Adam run follows; this helps to escape
    shallow local minima.

All stages respect the global time budget (default 1000 s) and the best
sequence encountered is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of appropriate length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerically impossible, but safe
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """Objective used in the competition: 2·n·max(a*a).  Assumes sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 1 – Exponential‑gradient warm‑up (smooth‑max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    total_iter = 0
    # α schedule – grows up to very large values to approach the exact max
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
              200.0, 500.0, 1000.0, 2000.0]
    steps_per_alpha = 5000               # can be increased if time permits

    fft_len = _next_pow2(3 * n - 2)       # length needed for the gradient convolution

    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # current convolution and objective
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                      f"α={alpha:g}, iter={total_iter:,}, score={cur_score:.6f}")

            # smooth‑max weights (log‑sum‑exp trick)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stable exponentiation
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            w_rev = w[::-1]

            # gradient of the smooth max (via FFT)
            Af = np.fft.rfft(a, n=fft_len)
            W = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * W, n=fft_len)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # multiplicative update (entropy‑mirror descent)
            total_iter += 1
            eta = 0.4 / math.sqrt(total_iter)
            upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
            a *= upd
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a[:] = 1.0 / n

    return best_a, best_score


# ----------------------------------------------------------------------
# 2 – Adam‑based projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07                # base learning‑rate; tuned empirically
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam sub‑grad improve: "
                  f"iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient of the (exact) max – only the active index is needed
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad = np.zeros_like(a)
        grad[i] += a[j]
        grad[j] += a[i]                     # handles the i=j case automatically

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex (guarantees sum=1, a≥0)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3 – Targeted local refinement (tiny redistributions)
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10 ** 7,
                         eps_frac: float = 0.025) -> np.ndarray:
    """
    Repeatedly move a small amount of probability mass away from coordinates
    that contribute most to the current maximal convolution entry.
    Only moves that strictly lower the maximum are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # index where the convolution attains its maximum
        k_max = int(np.argmax(conv))

        # contribution of each coefficient to that maximal entry
        idx = np.arange(n)
        jdx = k_max - idx
        valid = (jdx >= 0) & (jdx < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[idx[valid]] * a[jdx[valid]]
        total_contrib = contrib.sum()

        # probability of picking a source index: proportional to its contribution
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # choose a destination uniformly (different from src)
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # tiny amount of mass to move
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:          # strict improvement required
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a

# 4 – Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5         # tiny safety margin

    # --------------------------------------------------------------
    #  Initialise: try to reuse a previously found construction if
    #  available, otherwise start from a random Dirichlet‑like vector.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 5600            # a modestly larger grid than the previous 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = rng.random(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective(best_seq)

    # --------------------------------------------------------------
    #  Stage 1 – exponential‑gradient warm‑up (smooth max)
    # --------------------------------------------------------------
    if time.time() < deadline:
        stage1_deadline = time.time() + min(30.0, 0.15 * (deadline - time.time()))
        a_warm, score_warm = _run_exponential_gradient(
            best_seq, final_n, stage1_deadline, start_time
        )
        if score_warm < best_score:
            best_score, best_seq = score_warm, a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    #  Stage 2 – Adam‑based projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate the majority of the remaining budget to this stage
        stage2_deadline = time.time() + max(30.0, 0.70 * remaining)
        a_sub, score_sub = _run_adam_subgradient(
            best_seq, final_n, stage2_deadline, start_time
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: score={score_sub:.6f}")

    # --------------------------------------------------------------
    #  Stage 3 – targeted local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10 ** 7,
            eps_frac=0.032,               # slightly larger steps for fine‑grained tweaking
        )
        score_ref = _objective(a_ref)
        if score_ref < best_score:
            best_score, best_seq = score_ref, a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 3 improve: score={score_ref:.6f}")

    # --------------------------------------------------------------
    #  Stage 4 – occasional random restarts + short Adam runs
    # --------------------------------------------------------------
    while time.time() < deadline:
        # create a perturbed copy of the current best
        perturb = best_seq + 0.02 * rng.standard_normal(final_n) * best_seq
        perturb = np.maximum(perturb, 0.0)
        s = perturb.sum()
        if s > 0:
            perturb /= s
        else:
            perturb = np.full_like(perturb, 1.0 / final_n)

        # short Adam run (a few thousand iterations)
        short_end = time.time() + min(15.0, deadline - time.time())
        a_tmp, score_tmp = _run_adam_subgradient(perturb, final_n, short_end, start_time)

        # keep improvement if any
        if score_tmp < best_score:
            best_score, best_seq = score_tmp, a_tmp.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve: score={score_tmp:.6f}")

        # one more round of targeted refinement on the new best
        if time.time() < deadline:
            a_ref2 = _run_targeted_refine(
                best_seq,
                final_n,
                deadline,
                start_time,
                max_iters=5 * 10 ** 5,
                eps_frac=0.035,
            )
            score_ref2 = _objective(a_ref2)
            if score_ref2 < best_score:
                best_score, best_seq = score_ref2, a_ref2.copy()
                print(f"[{time.time() - start_time:.2f}s] post‑restart refine improve: score={score_ref2:.6f}")

    # --------------------------------------------------------------
    #  Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] final best score: {best_score:.6f}")
    return [float(x) for x in best_seq]
