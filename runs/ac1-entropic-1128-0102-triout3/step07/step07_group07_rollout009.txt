def _cross_corr(a,w):
    n = len(a); L = len(w)
    fft_len = _next_pow2(n+L-1)
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w, n=fft_len)
    corr = np.fft.irfft(np.conj(Af) * Wf, n=fft_len)[: n+L-1]

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Our strategy is a fast three‑phase optimizer that respects the 1000 s
budget:

1. **Lightweight Adam smoothing** – a soft‑max parametrisation of the
   coefficients is optimised for a few thousand iterations only (instead
   of the many tens of thousands used previously).  This brings the
   solution quickly into a low‑max‑convolution region while leaving most
   of the time budget for the later phases.

2. **Exact sub‑gradient descent on the true max‑convolution objective**.
   The sub‑gradient of ``max_k (a*a)[k]`` is obtained efficiently via an
   FFT‑based cross‑correlation (no expensive Python loops).  With a
   momentum update and a step size decaying as 1/√t we can run many
   thousands of iterations in the remaining time.

3. **Greedy peak‑reduction + simulated annealing** – a deterministic
   mass‑transfer step removes the dominant contribution to the current
   peak; a final annealing phase performs random transfers with an
   exponential cooling schedule to escape possible plateaus.

Multiple diverse starting points (uniform, exponential, Beta‑shaped,
random Dirichlet and the previous best construction if available) are
tried; the best sequence found overall is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w = 2·(a⋆w) is the gradient of Σ w_k·conv_k
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wr = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wr, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]
    grad = 2.0 * corr[w_rev.shape[0] - 1 - np.arange(a.shape[0])]
    return grad


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_j a[j]·w[i+j]  for i = 0,…,n‑1.
    Implemented by FFT using the conjugate trick.
    """
    n = a.shape[0]
    L = w.shape[0]
    fft_len = _next_pow2(n + L - 1)
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(np.conj(Af) * Wf, n=fft_len)[: n + L - 1]
    start = L - 1
    return corr_full[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – Light Adam optimisation (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    A few thousand Adam updates on the smoothed max objective.
    Returns the best coefficient vector encountered (L1‑normalised).
    """
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))          # logits
    n = a0.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # modest schedule – enough to get close to a good region
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    iters_per_alpha = 1500

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a0.copy()
    best_score = float("inf")

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a
            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective and its gradient
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradient w.r.t. a (true objective = 2·n·f_val)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional cheap evaluation (exact score = 2·n·max(conv) because sum=1)
            if (_ + 1) % 200 == 0:
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                          f"α={alpha:.1f}, iter={_ + 1:,}, score≈{cur_score:.6f}")

    # final projection to the simplex (ensures exact L1=1)
    best_a = _project_simplex(best_a)
    return best_a


# ----------------------------------------------------------------------
# Phase 2 – Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_phase(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Sub‑gradient descent on the true objective
        f(a) = max_k (a*a)[k]   with L1‑projection after each step.
    Returns the best coefficient vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()   # because sum(a)=1

    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.5
    it = 0
    rel_tol = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve: iter={it:,}, "
                  f"score≈{cur_score:.6f}")

        # active set (indices attaining the maximum up to a tiny tolerance)
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient via FFT‑based cross‑correlation
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _cross_corr(a, w)

        # momentum and projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a


# ----------------------------------------------------------------------
# Phase 3 – Greedy peak reduction
# ----------------------------------------------------------------------


def _peak_reduction_phase(a0: np.ndarray, deadline: float,
                          start_time: float,
                          max_no_improve: int = 2000) -> np.ndarray:
    """
    Deterministic greedy loop that moves mass from the coefficient
    most responsible for the current peak towards the smallest coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # choose destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: "
                  f"max≈{new_max:.6f}")
        else:
            no_improve += 1

    return a


# ----------------------------------------------------------------------
# Phase 4 – Simulated annealing
# ----------------------------------------------------------------------


def _annealing_phase(a0: np.ndarray, deadline: float, start_time: float,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-6,
                     max_steps: int = 200_000) -> np.ndarray:
    """
    Classic annealing: move a random ε‑fraction of mass between two indices.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score≈{score:.6f}")

    return best_seq
