"""
Propose a step‑function that (hopefully) beats the previous 1.51256 bound.

Algorithm (≈1000 s total)
-------------------------
1. Initialise a probability vector on the simplex.
   • If a previous construction (height_sequence_1) exists we up‑sample it.
   • Otherwise we start from the uniform distribution.
2. Phase A – Adam‑optimised soft‑max gradient descent.
   • The smooth‑max is approximated by a soft‑max with an exponential cooling
     schedule (T_start → T_end).
   • Adam (β1=0.9, β2=0.999) gives adaptive step sizes and converges faster
     than simple momentum.
3. Phase B – Adam‑optimised exact sub‑gradient descent.
   • The gradient is taken only at the current maximal convolution entry
     (the true, non‑smoothed objective).
4. Phase C – Aggressive “peak‑shaving’’ local search.
   • Repeatedly shift a random fraction of probability mass from indices
     that contribute most to the current peak to low‑contributing indices.
   • A Metropolis acceptance rule with a slowly cooling temperature allows
     uphill moves to escape shallow traps.
5. The best sequence seen across all phases is returned (as a plain Python
   list).  All vectors are projected onto the probability simplex after
   every update.

The routine respects the 1000 s budget, keeping a small safety margin for
the evaluation harness.
"""

# ----------------------------------------------------------------------
# Imports

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, ∑x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max with temperature) of the convolution.
    The factor 2 comes from the derivative of aᵀ H_k a (Hankel matrix).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # stabilise exponentials
    scaled -= scaled.max()
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient (used in Phase B)
# ----------------------------------------------------------------------
def _exact_subgrad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max‑convolution entry.
    """
    idx_max = int(np.argmax(conv))
    n = a.shape[0]
    i_arr = np.arange(n)
    j_arr = idx_max - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j_arr[mask]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (generic)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   grad_fn,
                   n: int,
                   deadline: float,
                   start_time: float,
                   step0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation.  grad_fn(a, conv) must return a
    gradient of the *smooth* objective that we are currently minimising.
    The routine returns the best sequence seen and its (raw) score
    2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n} iter={it} score={cur_score:.6f}")

        # compute gradient
        grad = grad_fn(a, conv)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (similar to the original schedule)
        step = step0 / math.sqrt(it)

        # Adam update + projection onto simplex
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase A – soft‑max Adam descent
# ----------------------------------------------------------------------
def _run_softmax_adam(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.2) -> tuple[np.ndarray, float]:
    """Adam optimiser where the gradient is the soft‑max of the convolution."""
    T_start = 2.0e-2
    T_end = 5.0e-4
    iteration = 0

    def grad_fn(a, conv):
        nonlocal iteration
        iteration += 1
        # temperature schedule (exponential decay)
        T = T_start * (T_end / T_start) ** (iteration / 20000.0)
        return _softmax_gradient(a, conv, T)

    return _adam_optimize(a0, grad_fn, n, deadline, start_time, step0=step0)

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:

    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase C – targeted local search (peak shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of probability mass from indices that contribute most
    to the current maximal convolution entry to low‑contribution indices.
    A Metropolis rule with a cooling temperature accepts occasional uphill
    moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the *maximal* entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # receiver from the low‑contribution tail
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # amount to move (up to 30 % of donor mass)
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # linear temperature decay between start_temp and end_temp
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n} iter={it} score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of:
      * soft‑max Adam descent (Phase A)
      * exact‑max Adam descent (Phase B)
      * aggressive peak‑shaving local search (Phase C)
    The best sequence encountered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start: use the previous best construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a target length (large enough to give a good bound, but still fast)
    # ------------------------------------------------------------------
    target_n = 12000                       # a compromise between quality & speed

    # initialise a0
    if starter is not None:
        a0 = _upsample(starter, target_n)
    else:
        a0 = np.full(target_n, 1.0 / target_n)   # uniform start

    # ------------------------------------------------------------------
    # Phase A – soft‑max Adam gradient descent
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.35
    a_a, score_a = _run_softmax_adam(a0, target_n, phase_a_deadline, start_time,
                                     step0=0.25)

    # ------------------------------------------------------------------
    # Phase B – exact‑max Adam descent
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.65
    a_b, score_b = _run_subgrad_adam(a_a, target_n, phase_b_deadline, start_time,
                                     step0=0.15)

    # ------------------------------------------------------------------
    # Phase C – targeted local refinement
    # ------------------------------------------------------------------
    a_c, score_c = _run_targeted_local_search(a_b, target_n, deadline, start_time)

    # ------------------------------------------------------------------
    # Pick the best of the three phases
    # ------------------------------------------------------------------
    best_seq = a_c
    best_score = score_c
    if score_a < best_score:
        best_score = score_a
        best_seq = a_a
    if score_b < best_score:
        best_score = score_b
        best_seq = a_b

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may happen because of rounding)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] final score (2·n·max_b) = {best_score:.6f}")
    # return as plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
