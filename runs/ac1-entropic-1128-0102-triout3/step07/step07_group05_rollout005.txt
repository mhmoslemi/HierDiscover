    # standard algorithm O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)

"""
Hybrid optimisation for the step‑function attack (new version).

The algorithm works in two clear phases and fits comfortably into a
1000 s budget.

Phase 1 – Exponentiated‑gradient descent on the *exact* max‑convolution.
For the current sequence `a` we locate the (possibly several) convolution
indices attaining (or almost attaining) the maximum, build the exact
sub‑gradient of `max(a*a)` w.r.t. `a` and perform a mirror‑descent step
`a_i ← a_i·exp(-η·g_i)`.  The step size `η` shrinks slowly; every
accepted move is projected back onto the probability simplex by normalising.
The sub‑gradient is computed only for the active set, which makes a single
iteration cheap (O(n log n) for the FFT‑based convolution plus O(|active|·n)
for the gradient, the active set is tiny).

Phase 2 – Fast incremental simulated‑annealing.  A mass‑transfer move
(`ε` from a source index to a destination index) is applied and the
convolution is updated incrementally in O(n) using slice operations, as
in the previous state‑of‑the‑art code.  The temperature follows a geometric
schedule and uphill moves are accepted with the Metropolis probability.
Both phases keep the best sequence encountered so far.

All numbers are kept non‑negative and the final sequence is normalised to
unit sum, satisfying the security checks in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------
def _active_subgradient(a: np.ndarray, conv: np.ndarray, rel_tol: float = 1e-12
                        ) -> tuple[np.ndarray, float]:
    """
    Returns (g, M) where g is the sub‑gradient of max(conv) at `a` and
    M = max(conv).  Only indices with conv ≥ (1‑rel_tol)·M are used.
    """
    n = a.shape[0]
    M = conv.max()
    thresh = M * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    # If the active set is huge (rare), keep at most 100 indices
    if active.size > 100:
        order = np.argsort(-conv)[:100]
        active = order

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]

    g *= 2.0
    return g, M


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, n: int, deadline: float,
                          start_time: float,
                          best_seq: np.ndarray,
                          best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on  f(a)=max(a*a).
    Returns the best sequence and its score found before the deadline.
    """
    a = a0.copy()
    best_local_seq = best_seq.copy()
    best_local_score = best_score

    L = _next_pow2(2 * n - 1)          # FFT length (fixed)

    # step‑size schedule
    eta = 1.0
    min_eta = 1e-6
    decay = 0.9995
    iteration = 0

    while time.time() < deadline and eta > min_eta:
        iteration += 1

        # current convolution & objective
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        M = conv.max()
        score = 2.0 * n * M

        if score < best_local_score:
            best_local_score = score
            best_local_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: "
                  f"iter={iteration:,} score={score:.6f}")

        # sub‑gradient of the max‑convolution
        g, _ = _active_subgradient(a, conv)

        # try a few scaled step sizes, keep the first that improves
        improved = False
        for factor in (1.0, 0.5, 0.2):
            eta_test = eta * factor
            a_cand = a * np.exp(-eta_test * g)
            a_cand = np.maximum(a_cand, 1e-16)      # keep positivity
            a_cand /= a_cand.sum()                  # projection onto simplex

            # evaluate candidate (fast, using convolution)
            Af_cand = np.fft.rfft(a_cand, n=L)
            conv_cand = np.fft.irfft(Af_cand * Af_cand, n=L)[: 2 * n - 1]
            score_cand = 2.0 * n * conv_cand.max()

            if score_cand < score:
                a = a_cand
                improved = True
                break

        if not improved:
            eta *= 0.7          # aggressive reduction when stuck
        else:
            eta *= decay        # gentle decay when progressing

        # keep eta safely above the floor
        if eta < min_eta:
            eta = min_eta

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing (as in the previous state of the art)
# ----------------------------------------------------------------------
def _incremental_annealing(a0: np.ndarray, n: int, deadline: float,
                           start_time: float,
                           cur_best_score: float,
                           cur_best_seq: np.ndarray,
                           total_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing where the convolution is updated incrementally.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # pick a source proportionally to its mass (focus on heavy positions)
        src = rng.choice(n, p=a)
        dst = rng.integers(0, n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    start_time = time.time()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    total = best_seq.sum()
    if total > 0:

    else:

    # final safety check (optional)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")
