def _run_lp_refinement(a0: np.ndarray, deadline: float, start_time: float,
                       max_iter: int = 20, mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of linear‑program based peak‑shaving.
    Returns best sequence and its score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # Build matrix A for constraints A @ g <= M
        rows = 2 * n - 1
        # Build A as dense
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            start = i
            A[start:start + n, i] = a[i]  # fill column i with a[i]
        # Solve LP: maximize sum(g) <=> minimize -sum(g)
        c = -np.ones(n, dtype=float)
        b = np.full(rows, M, dtype=float)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g
        # Mixing
        a = (1 - mix_factor) * a + mix_factor * g
        # Project onto simplex to counter numeric drift
        a = _project_simplex(a)
        # Evaluate score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it}/{max_iter}, n={n}, score={cur_score:.6f}")
        # optional early stop if improvement small
        if it >= max_iter:
            break
    return best_a, best_score

Propose a step‑function with a low self‑convolution peak.

Algorithm
---------
1. Initialise a probability vector (warm‑started if a previous solution is
   available) at a modest length (800).
2. Perform a few iterations of a *linear‑program* based “peak‑shaving”.
   For the current f we solve
        max Σ g_j   s.t.   f * g ≤ M ,   g ≥ 0

   simplex so that the evaluation routine is scale‑invariant).

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth max‑convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """∇_a Σ_k softmax_k(conv/temperature)·conv_k."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Return the best point seen and its (un‑normalised) score = 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.5,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Project the iterates onto the simplex while descending the max‑conv."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_max = conv[idx_max]
        cur_score = 2.0 * n * cur_max          # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑shifting walk that reduces the maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (peak‑shaving)
# ----------------------------------------------------------------------
def _run_lp_refinement(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       max_iter: int = 15,
                       mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """Iteratively push mass onto indices where the current peak is low."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        M = conv.max()

        # Build the A‑matrix for constraints A @ g ≤ M  (size (2n‑1) × n)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a[i]        # column i contributes a[i] on a band of length n

        c = -np.ones(n, dtype=float)          # maximise Σ g  ⇔ minimise –Σ g
        b = np.full(rows, M, dtype=float)
        bounds = [(0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # If the LP fails we abort the refinement.
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g                         # normalise to a probability vector

        # convex combination and projection
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it}/{max_iter}, "
                  f"n={n}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a short LP‑based peak‑shaving stage, multi‑stage
    smooth‑max gradient descent, sub‑gradient refinement and a final
    Metropolis local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from the previous best, if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 0 : small length (800) and LP refinement
    # ------------------------------------------------------------------
    init_n = 800
    if starter is not None:
        a0 = _upsample(starter, init_n)
    else:
        a0 = np.random.rand(init_n)
        a0 /= a0.sum()

    # allocate roughly 10 % of the total budget to the LP stage
    lp_deadline = time.time() + (deadline - start_time) * 0.10
    a_lp, score_lp = _run_lp_refinement(a0, lp_deadline, start_time,
                                         max_iter=12, mix_factor=0.5)
    best_seq = a_lp
    best_score = score_lp

    # ------------------------------------------------------------------
    # Time budgeting for the remaining phases
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    softmax_budget = total_time * 0.60
    subgrad_budget = total_time * 0.25
    local_budget   = total_time - softmax_budget - subgrad_budget

    # ------------------------------------------------------------------
    # Progressive up‑sampling stages (soft‑max GD)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # final length 9600
    total_len_weight = sum(stage_lengths)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the soft‑max budget proportional to n
        stage_frac = n / total_len_weight
        stage_time = max(2.0, softmax_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this length (keep the best so far)
        a0_stage = _upsample(best_seq, n)

        # optional tiny LP refinement for the two smallest stages
        if n <= 2400:
            lp_deadline_stage = time.time() + min(1.0, stage_time * 0.07)
            a0_stage, _ = _run_lp_refinement(a0_stage, lp_deadline_stage,
                                             start_time, max_iter=5,
                                             mix_factor=0.5)

        # soft‑max gradient descent
        a_opt, score_opt = _run_softmax_gd(
            a0_stage, n, stage_deadline, start_time,
            step0=0.8, beta=0.9, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # update remaining soft‑max budget
        elapsed = time.time() - start_time
        softmax_budget = max(0.0,
                             deadline - start_time - subgrad_budget - local_budget - elapsed)

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the final length (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        subgrad_deadline = time.time() + subgrad_budget
        a_refined, score_refined = _run_subgradient_gd(
            best_seq, n_final, subgrad_deadline, start_time,
            step0=0.5, beta=0.5, max_iter=10 ** 7
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑gradient improve: "
                  f"score={score_refined:.6f}")

    # ------------------------------------------------------------------
    # Targeted local search (peak shaving) – uses the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (the harness expects a list)
    return [float(x) for x in best_seq]
