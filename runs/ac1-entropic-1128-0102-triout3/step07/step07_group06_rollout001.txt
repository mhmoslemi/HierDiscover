"""
Propose a step‑function (non‑negative vector on the simplex) that minimizes
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Parametric warm‑start** – a family a_i ∝ ((i+1)*(n‑i))^(-p) (U‑shaped)
   is mixed with the uniform distribution.  A small grid over (p,λ) is
   searched; the best candidate is kept.

2. **Adam‑based soft‑max gradient descent** – we minimise a smooth
   approximation of the max‑convolution using the log‑sum‑exp (soft‑max)
   surrogate.  The gradient is obtained via FFT, Nesterov‑style momentum is
   replaced by Adam’s adaptive steps, and after each update the vector is
   projected onto the probability simplex.

3. **Greedy mass‑transfer hill‑climbing** – with the remaining budget we
   repeatedly move a random fraction of mass from one coordinate to another,
   keeping the move only if it reduces the true objective
   ``2·n·max(conv)``.  This finishes the refinement.

The whole routine respects a 1000 s wall‑clock limit, keeps all entries
non‑negative, and returns a plain Python list.
"""
import time
import math

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities – linear self‑convolution via FFT (O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) of the convolution.
    w_i ∝ exp(conv_i / T) and grad = 2·rev(a) * w  (linear convolution).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))  # for numerical stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    cur_score = 2.0 * n * _conv_self(a).max()

    while time.time() < deadline:
        # pick two distinct indices, move a small fraction from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 1e-15:
            continue
        # up to 20 % of the mass at i
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill improve → score={cur_score:.6f}")

        # occasional smoothing move (helps escape tiny plateaus)
        if np.random.rand() < 0.01:
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_smooth = np.convolve(a, kernel, mode="same")
            a_smooth = _project_simplex(a_smooth)
            smooth_score = 2.0 * n * _conv_self(a_smooth).max()
            if smooth_score < cur_score:
                a = a_smooth
                cur_score = smooth_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smoothing improve → score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
