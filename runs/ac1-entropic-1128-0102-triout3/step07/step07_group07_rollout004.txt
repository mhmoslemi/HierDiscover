def propose_candidate(seed=0, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 4800 # maybe adapt later

    # param sweep
    best_seq = None
    best_score = float('inf')
    # uniform start
    uniform = np.full(n, 1.0/n)
    best_seq = uniform
    best_score = _calc_score(uniform)
    # exponential sweep
    for lam in np.logspace(-5, -1, num=30): # lam in [1e-5, 0.1]
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        sc = _calc_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a
    # power law sweep
    for p in np.linspace(0, 2.0, num=40):
        a = 1.0 / ((np.arange(1, n+1))**p)
        a /= a.sum()
        sc = _calc_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a

    # linear decreasing sweep
    for c in np.linspace(0, 1.0/(n-1), num=30):
        a = np.maximum(1 - c*np.arange(n), 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _calc_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a

    print("[{:.2f}s] param sweep best score {:.6f}".format(time.time()-start_time, best_score))

    # gradient descent refinement
    # allocate half the remaining budget to gradient descent
    remaining = deadline - time.time()
    gd_deadline = time.time() + 0.5 * remaining
    a_refined, gd_score = _run_projected_gradient_descent(best_seq.copy(), gd_deadline, start_time)

        a_final, final_score = _run_annealing(best_seq.copy(), deadline, start_time, start_temp=1e-2, end_temp=1e-6, steps=200_000)

def _run_projected_gradient_descent(a0, deadline, start_time,
                                   step0=0.5,
                                   beta=0.9,
                                   max_iter=10**7):
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = _calc_score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # compute gradient at max index only
        k = conv.argmax()
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        # add contributions
        # for each i, a[j] contribution
        g[i] += a[j]
        g[j] += a[i]

        # momentum and step
        momentum = beta * momentum + (1 - beta) * g
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

        # evaluate
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad descent improved: iter={it}, score={cur_score:.6f}")
    return best_a, best_score

"""
Optimisation of a non‑negative step function for ``evaluate_sequence``.

The search is performed in three stages:

1. **Low‑dimensional parametric sweep** – try several families of
   simple decreasing shapes (exponential, power‑law, linear).  The
   objective 2·n·max(a*a) (the sum is normalised to 1) is evaluated
   with an FFT‑based convolution, which is much faster than calling
   ``evaluate_sequence`` repeatedly.

2. **Exponentiated‑gradient refinement** – starting from the best
   candidate of the sweep, run a mirror‑descent (exponentiated‑gradient)
   on the true ``max`` objective.  The gradient is the exact sub‑gradient
   of the active convolution index, the update is performed in the
   log‑domain and the probabilities are obtained by a soft‑max, which
   automatically respects the simplex constraints.  A diminishing step
   size (η∝1/√t) and a solid time‑budget guard are used.

3. **Final simulated‑annealing polishing** – a short stochastic
   mass‑transfer annealing (the same routine as the reference code)
   exploits any remaining time to escape tiny plateaus.

All helper functions are top‑level, no closures are used, and the whole
procedure respects the 1000 s budget.  The returned list is a plain
Python list of non‑negative floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _calc_score(a: np.ndarray, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming sum(a)=1."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Parametric families (exponential, power‑law, linear)
# ----------------------------------------------------------------------


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.power(i, -p, where=i != 0, out=np.ones_like(i))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def _make_linear(n: int, slope: float) -> np.ndarray:
    """a_i = max(1 - slope·i, 0), normalised."""
    i = np.arange(n, dtype=float)
    a = np.maximum(1.0 - slope * i, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# 2. Exponentiated‑gradient refinement on the true max‑objective
# ----------------------------------------------------------------------


def _run_exponentiated_gradient(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               eta0: float = 0.5,
                               beta: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on f(a)=max_k (a*a)[k].
    Returns best seen vector and its corresponding score 2·n·max.
    """
    n = a0.shape[0]
    # initialise logits in log‑domain
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    best_a = a0.copy()
    best_score = _calc_score(best_a, n)

    it = 0
    while time.time() < deadline:
        it += 1

        # rebuild probabilities from logits
        maxy = np.max(y)
        expy = np.exp(y - maxy)
        a = expy / expy.sum()

        # convolution and identify the active index
        conv = _conv_self(a)
        k = conv.argmax()

        # sub‑gradient of the active convolution entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] += a[j]
        g[j] += a[i]          # diagonal counted twice – correct for derivative

        # gradient w.r.t. logits (mirror‑descent)
        grad_y = a * (g - np.dot(g, a))

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        y = y - eta * grad_y

        # evaluate current score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Simulated‑annealing polishing (identical to reference routine)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Random mass‑transfer annealing."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4. Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The algorithm follows the 3‑stage plan described

    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – the reference size works well; we keep it.
    # --------------------------------------------------------------
    n = 4800

    best_seq = np.full(n, 1.0 / n)          # uniform as fallback

    for lam in np.logspace(-5, -1, num=30):

        if sc < best_score:

    print(f"[{time.time() - start_time:6.2f}s] sweep after exponential best={best_score:.6f}")

    # power‑law – p ∈ [0, 2]
    for p in np.linspace(0.0, 2.0, num=40):

        if sc < best_score:

            best_seq = a
    print(f"[{time.time() - start_time:6.2f}s] sweep after power‑law best={best_score:.6f}")

    # linear decreasing – slope from 0 up to 1/(n‑1)
    for slope in np.linspace(0.0, 1.0 / (n - 1), num=30):

    print(f"[{time.time() - start_time:6.2f}s] sweep after linear best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) EXPO‑GRADIENT REFINEMENT (uses remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:                     # leave a few seconds for the final phase
        eg_deadline = time.time() + 0.6 * remaining
        a_refined, eg_score = _run_exponentiated_gradient(
            best_seq.copy(),

            start_time,
            eta0=0.7,                     # a slightly larger initial step
        )
        if eg_score < best_score:
            best_score = eg_score
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG refinement improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) FINAL ANNEALING (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, final_score = _run_annealing(

            start_temp=2e-2,
            end_temp=1e-6,
            steps=200_000,

        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise once more (guard against numeric drift) and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_eval = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_eval:.6f}")

    return [float(x) for x in best_seq.tolist()]
