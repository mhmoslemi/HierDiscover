"""
Step‑function optimizer for the ``evaluate_sequence`` objective.

The algorithm is deliberately different from the reference implementation:

* Several simple analytic initialisations (uniform, linear, exponential,
  power‑law) are generated at a modest resolution (≈2000 points).
* Each initialise is refined by a **projected sub‑gradient descent** that works
  on the exact max‑convolution (active‑set) – this phase is cheap at the
  small resolution, so we can afford many iterations.
* The best intermediate vector is linearly up‑sampled to the final resolution
  (≈9600 points) and refined again (few thousand sub‑gradient steps).
* Finally a long **simulated‑annealing** walk moves tiny fractions of mass
  between indices and occasionally smooths the vector.  The annealing loop
  runs until the time budget is exhausted, giving it the majority of the
  1000 s wall‑clock budget.

All vectors are kept non‑negative and renormalised to sum to 1, which is
without loss of generality because the evaluation function is homogeneous.
The code prints progress messages (time stamp, stage, score) so that you can
track the optimisation while it runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x ≥ 0, sum x = 1}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on 2·n·max(conv) using an active set.
# ----------------------------------------------------------------------
def _run_subgradient_gd(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.15,
    beta: float = 0.9,
    max_iter: int = 10 ** 7,
    rel_eps: float = 0.01,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (active‑set)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}"
            )

        # ----- active set -------------------------------------------------
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:  # fallback (should not happen)
            active = np.array([conv.argmax()], dtype=int)

        # weight vector w = average over active indices
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # Nesterov‑style momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer + occasional smoothing).
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 0.3,
    end_temp: float = 1e-7,
    steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Metropolis walk on the simplex that shuffles mass."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:
            # move a random fraction of mass from i to j
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25  # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:
            # occasional smoothing (kernel width 3/5/7)
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode="same")
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: quick objective for a normalized vector.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) (assumes sum(a)=1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine works inside the given time budget, builds several analytic
    starting points at a modest resolution, refines them with intensive
    sub‑gradient descent, upsamples the best candidate to the final length
    (≈9600) and finishes with a long simulated‑annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters (feel free to tune)
    # ------------------------------------------------------------------
    n_small = 2000                 # cheap resolution for the heavy sub‑gradient phase
    n_final = 9600                 # final resolution used for annealing
    total_budget = budget_s - 0.5

    # Allocate about 30 % of the time to the sub‑gradient refinements,
    # the remaining 70 % to the final annealing.
    subgrad_deadline = start_time + total_budget * 0.30
    anneal_deadline = deadline

    # ------------------------------------------------------------------
    # Build a collection of simple initial vectors (all normalised).
    # ------------------------------------------------------------------
    init_vectors = []

    # 1) uniform
    init_vectors.append(np.full(n_small, 1.0 / n_small))

    # 2) linear descending (a_i ∝ n‑i)
    lin_desc = np.arange(n_small, 0, -1, dtype=float)
    lin_desc /= lin_desc.sum()
    init_vectors.append(lin_desc)

    # 3) exponential (a_i ∝ r^i) – we try a few r values
    for r in (0.999, 0.9995, 0.9999):
        exp_vec = (r ** np.arange(n_small)).astype(float)
        exp_vec /= exp_vec.sum()
        init_vectors.append(exp_vec)

    # 4) power‑law (a_i ∝ (i+1)^{-p}) with p≈0.5, 0.8
    for p in (0.5, 0.8):
        pow_vec = (np.arange(1, n_small + 1, dtype=float) ** (-p))
        pow_vec /= pow_vec.sum()
        init_vectors.append(pow_vec)

    # 5) a few random smooth vectors (random + small Gaussian blur)
    for _ in range(3):
        rnd = np.random.rand(n_small)
        rnd = np.convolve(rnd, np.ones(5) / 5, mode="same")
        rnd /= rnd.sum()
        init_vectors.append(rnd)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Small‑resolution sub‑gradient phase.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(init_vectors):
        if time.time() >= subgrad_deadline:
            break
        print(
            f"[{time.time() - start_time:6.2f}s] start subgrad init {idx + 1}/{len(init_vectors)} (n={n_small})"
        )
        a_opt, score_opt = _run_subgradient_gd(
            a0,
            n_small,
            subgrad_deadline,
            start_time,
            step0=0.30,
            beta=0.9,
            max_iter=10 ** 7,
            rel_eps=0.015,
        )
        # keep the best small‑resolution candidate
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] new global best (small) score={score_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Upsample the best small vector to the final resolution and give it
    # a modest amount of sub‑gradient work (few × 10³ iterations).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        best_seq = np.full(n_small, 1.0 / n_small)

    a_big = _upsample(best_seq, n_final)

    # a quick refinement at the final size
    if time.time() < subgrad_deadline:
        a_big, score_big = _run_subgradient_gd(
            a_big,
            n_final,
            subgrad_deadline,
            start_time,
            step0=0.15,
            beta=0.9,
            max_iter=50_000,
            rel_eps=0.01,
        )
        if score_big < best_score:
            best_score = score_big
            best_seq = a_big.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] improved after up‑sample subgrad: score={score_big:.6f}"
            )
    else:
        score_big = _objective(a_big)

    # ------------------------------------------------------------------
    # Final annealing stage – runs until the overall deadline.
    # ------------------------------------------------------------------
    if time.time() < anneal_deadline:
        print(f"[{time.time() - start_time:6.2f}s] starting final annealing (n={n_final})")
        a_ref, score_ref = _run_annealing(
            a_big,
            n_final,
            anneal_deadline,
            start_time,
            best_score,
            a_big,
            start_temp=0.5,
            end_temp=1e-7,
            steps=10_000_000,
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improvement: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise one final time (tiny drift may have occurred) and return.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # The harness expects a plain Python list.
    return [float(x) for x in best_seq]
