"""
Hybrid optimiser with extra greedy flattening and random restarts.

The algorithm proceeds in several stages, each respecting the 1000 s budget.

1.  Optional random‑restart loop – a short optimisation is performed from a
    fresh random start (or a perturbed copy of the current best).  This helps
    to escape local minima that the deterministic Adam → gradient → anneal
    pipeline may get stuck in.

2.  Adam on a smoothed log‑sum‑exp approximation of the max‑convolution.
    The parameters are kept on the simplex via a soft‑max of log‑weights.

3.  Projected sub‑gradient descent (exact max) – a classic Nesterov‑style
    descent with a Euclidean projection onto the probability simplex.

4.  Fast incremental simulated‑annealing – mass is transferred between two
    indices using an O(n) incremental update of the convolution, allowing
    millions of moves.

5.  Greedy “flattening’’ – an inexpensive deterministic step that repeatedly
    moves a small fraction of mass from the index with the largest sub‑gradient
    (i.e. contributing most to the peak of the convolution) to the index with
    the smallest sub‑gradient.  The convolution is again updated incrementally.

The best sequence ever encountered (across all restarts) is returned after
normalisation to sum = 1.  All numbers are non‑negative, satisfying the checks
in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (fast, without the security‑checks of evaluate_sequence)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """Return the raw objective 2·n·max(a*a) / (sum(a)^2).  a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _adam_phase(a0: np.ndarray, n: int,
                deadline: float, start_time: float,
                best_score: float, best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """Run a few Adam steps on the smoothed objective."""
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)
    # a modest schedule – a few α levels are enough for a quick improvement
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    iters_per_alpha = 4000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max → a on simplex
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradient w.r.t. a (scaled by 2·n later)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = _score(a, n)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10_000_000):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a, n)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set (values close to the max)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_to_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 1_000_000):
    """Mass‑transfer annealing with O(n) incremental convolution updates."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # choose source proportionally to its mass – focuses work where there is mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of source mass (larger fractions early are handled by the temperature)
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # reject: revert everything
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps

        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

        # occasional re‑creation of the convolution from scratch (keeps round‑off low)
        if it % 100_000 == 0:
            conv = _conv_self(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy flattening – deterministic mass transfer based on sub‑gradient
# ----------------------------------------------------------------------
def _greedy_flattening(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       cur_best_score: float,
                       cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Repeatedly move a small fraction of mass from the index with largest
    sub‑gradient to the one with smallest sub‑gradient, updating the convolution
    incrementally.  Very cheap (O(n) per iteration)."""
    a = a0.copy()
    conv = _conv_self(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    while time.time() < deadline:
        it += 1
        maxB = conv.max()
        # pick a single index attaining the max (any is fine)
        k_max = int(np.argmax(conv))

        # sub‑gradient for that single active index
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] += a[j[mask]]
        g[mask] += a[i[mask]]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] < 1e-14:
            break

        improved = False
        # try a few decreasing fractions of the donor mass
        for frac in (0.30, 0.20, 0.10, 0.05, 0.02, 0.01):
            eps = a[src] * frac
            if eps < 1e-14:
                continue

            # apply move (incremental update)
            conv[src:src + n] -= 2.0 * eps * a
            conv[dst:dst + n] += 2.0 * eps * a
            conv[2 * src] += eps * eps
            conv[2 * dst] += eps * eps
            conv[src + dst] -= 2.0 * eps * eps

            a[src] -= eps
            a[dst] += eps

            new_max = conv.max()
            if new_max < maxB - 1e-15:        # improvement
                maxB = new_max
                cur_score = 2.0 * n * maxB / (a.sum() ** 2)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it:,}, score={cur_score:.6f}")
                improved = True
                break
            # reject – revert
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps

        if not improved:
            # no improving move can be found – stop this phase
            break

        # keep the convolution tidy
        if it % 10_000 == 0:
            conv = _conv_self(a)

    return best_seq, best_score

# Local optimisation pipeline (used for a single start)

def _local_optimize(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Run the full pipeline (Adam → gradient → anneal → greedy) on a given start."""
    best_seq = start_vec.copy()
    best_score = _score(best_seq, n)

    # 1. Adam (quick)
    a_adam, best_score, best_seq = _adam_phase(best_seq, n,
                                               deadline, start_time,
                                               best_score, best_seq)

    # 2. Projected sub‑gradient descent (more iterations if time permits)
    if time.time() < deadline:
        a_gd, sc_gd = _run_gradient_descent(best_seq, n,
                                            deadline, start_time,
                                            max_iter=5_000_000)
        if sc_gd < best_score:
            best_score, best_seq = sc_gd, a_gd.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad‑descent improve: score={best_score:.6f}")

    # 3. Incremental simulated annealing
    if time.time() < deadline:
        a_ann, sc_ann = _run_incremental_annealing(best_seq, n,
                                                   deadline, start_time,
                                                   best_score, best_seq,
                                                   total_steps=2_000_000)
        if sc_ann < best_score:
            best_score, best_seq = sc_ann, a_ann.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={best_score:.6f}")

    # 4. Greedy flattening
    if time.time() < deadline:

        if sc_gf < best_score:
            best_score, best_seq = sc_gf, a_gf.copy()
            print(f"[{time.time() - start_time:6.2f}s] final greedy improve: score={best_score:.6f}")

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    The routine mixes random restarts, Adam‑based smoothing, exact sub‑gradient
    descent, fast incremental simulated annealing and a final greedy
    flattening, all respecting the supplied time budget.
    """

    start_time = time.time()

    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Determine problem size and an initial vector
    # ------------------------------------------------------------------
    try:

        if starter.ndim != 1 or starter.size == 0:

        starter = None

    n = 4800                     # same size as the reference – good trade‑off
    if starter is not None:

            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)

        a0 /= a0.sum()

    # track the overall best solution
    best_seq = a0.copy()

    print(f"[0.00s] initial score (fast estimate): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – a few random restarts with limited budget
    # ------------------------------------------------------------------
    attempt = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:

        # give each attempt at most 30 s (or half of the remaining time, whichever is smaller)
        local_deadline = time.time() + min(30.0, remaining * 0.5)

        if np.random.rand() < 0.5:

            print(f"[{time.time() - start_time:6.2f}s] ** NEW GLOBAL BEST ** score={best_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
