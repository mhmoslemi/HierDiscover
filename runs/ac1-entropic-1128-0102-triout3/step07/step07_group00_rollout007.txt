"""
Propose an improved non‑negative step‑function (probability vector) that
minimises `evaluate_sequence`.  The optimiser works in three stages:

1.  Adam on a log‑sum‑exp (soft‑max) surrogate of the maximal convolution.
    The smoothing parameter α is increased progressively, giving a fast
    “global” search while keeping the vector on the probability simplex.

2.  Projected sub‑gradient descent on the true objective.
    The sub‑gradient is taken as the average of all active maximal
    convolution entries, and a momentum term with a slowly decaying step‑size
    is used.  This phase consumes the bulk of the time budget.

3.  Simulated‑annealing polish that moves a small amount of mass between
    coordinates.  The temperature schedule is exponential and the source
    of mass is biased toward the indices that belong to the current maximal
    convolution block, which yields a more focused search.

All phases respect the supplied time budget (≈ 1000 s).  The length of the
candidate vector is increased to 8192 (a power of two) – this gives a
tighter theoretical lower bound while still being fast enough to optimise.
The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on the smooth max surrogate
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 5000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp approximation of max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length that works for any α in this phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # current (unsmoothed) objective – only for logging
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # --- soft‑max weights -------------------------------------------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # stability trick
            w /= w.sum()
            w_rev = w[::-1]

            # --- gradient of the smoothed max -----------------------------------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # --- Adam update ----------------------------------------------------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # stay on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – multi‑peak sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.96          # higher momentum for smoother progress
    step0 = 0.5          # larger initial step
    it = 0
    tol = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # average gradient over all (almost) maximal indices
        max_idxs = np.where(conv >= cur_max - tol)[0]
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(len(max_idxs), 1)

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : focused simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Light‑weight annealing that preferentially moves mass away from the
    active support of the maximal convolution entry.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current maximal convolution block
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        active_idx = np.arange(lo, hi + 1, dtype=int)

        # source – bias toward active block (large contribution to the peak)
        if rng.random() < 0.7 and a[active_idx].sum() > 1e-12:
            src = rng.choice(active_idx, p=a[active_idx] / a[active_idx].sum())
        else:
            src = rng.choice(n, p=a)   # global weighted choice

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length (power of two) – larger lengths give tighter lower bound.
    # 8192 works well within the budget.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # Initialise – try to reuse the existing global sequence if present.
    # ------------------------------------------------------------------
    starter = None
    try:
        hs1 = globals().get('height_sequence_1')
        if hs1 is not None:
            starter = np.asarray(hs1, dtype=float)
    except Exception:
        starter = None

    if starter is not None and starter.ndim == 1 and starter.size > 0:
        # Upsample/interpolate to the target length
        a0 = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, starter.size),
                       starter)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        # Random Dirichlet start (all positive)
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth max (≈10 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.10 * remaining)

    a_adam, score_adam = _run_adam_smoothing(
        a0,
        n,
        phase1_deadline,
        start_time,
        alphas=[0.3, 0.8, 2.0, 5.0, 12.0, 30.0, 70.0, 150.0],
        lr0=0.5,
        iters_per_alpha=6000,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact‑max projected sub‑gradient descent (dominant chunk)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.75 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final annealing polish (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return as plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
