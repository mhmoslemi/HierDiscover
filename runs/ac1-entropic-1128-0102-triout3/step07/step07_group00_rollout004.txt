def _subgrad_topk_sym(a0,n,deadline,start_time,initial_best_score,top_k=5):
    a = a0.copy()
    best_a = a.copy()
    best_score = initial_best_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    total_iter = 0
    while time.time() < deadline:
        total_iter += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] top‑k sub‑grad improve: score={cur_score:.6f}")
        # top-k indices
        if top_k >= conv.shape[0]:
            idxs = np.arange(conv.shape[0])
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # positions i = lo..hi, contribution a_{k-i}
            # vectorized:
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(idxs)  # average
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(total_iter)
        a = a - step * momentum
        # enforce symmetry and simplex
        a = (a + a[::-1]) / 2.0
        a = _project_simplex(a)
    return best_a, best_score

def _anneal_sym(a0,n,deadline,start_time,cur_best_score,cur_best_seq,
                start_temp=5e-2,end_temp=1e-6,steps=200_000):
    a = a0.copy()
    score = 2.0*n* _conv_self(a).max()
    best_score = cur_best_score
    best_seq = cur_best_seq.copy()
    rng = np.random.default_rng()
    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp/start_temp) ** (it/steps)
        # choose src weighted by current a
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        # move portion eps
        eps = a[src] * rng.random() * 0.4  # up to 40% of src
        # apply symmetric move
        src_mir = n-1-src
        dst_mir = n-1-dst
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if src_mir != src:
            a_prop[src_mir] -= eps
            a_prop[dst_mir] += eps
        # ensure nonnegative: clip
        if np.any(a_prop < -1e-12):
            continue  # reject move that would become negative
        # compute new score
        new_score = 2.0*n* _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: score={score:.6f}")
    return best_seq, best_score

Propose a non‑negative step function (probability vector) that minimises the
provided ``evaluate_sequence``.  The optimiser differs from the previous
three‑phase Adam/sub‑gradient/annealing pipeline in several ways:

*  It tries several sequence lengths (powers of two up to 4096).  Shorter
   lengths are faster to explore, allowing many restarts; the best shape is
   interpolated to the next length.
*  The vector is forced to be *mirror‑symmetric* (a[i] = a[n‑1‑i]).  This halves

*  Phase 1 uses Adam on a *log‑sum‑exp* surrogate of the maximal convolution,
   but the temperature (the “α” scaling) is decreased gradually.
*  Phase 2 replaces the single‑peak sub‑gradient by a *top‑k* sub‑gradient:
   the gradient is averaged over the k largest convolution entries, which

   move.  The temperature schedule is exponential and the move size can be
   up to 40 % of the donor mass.
*  All steps respect the 1000 s budget via a shared deadline; the algorithm
   falls back to the current best construction if time runs out.

The function ``propose_candidate`` returns a plain Python list of floats

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on the smooth max (log‑sum‑exp) surrogate,
#            with enforced symmetry.
# ----------------------------------------------------------------------
def _adam_smoothing_sym(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alphas,
    lr0: float = 0.5,
    iters_per_alpha: int = 3000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # pre‑compute a length suitable for all convolutions in this phase
    L_fft = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: score={cur_score:.6f}")

            # ----- soft‑max weights (log‑sum‑exp) -----------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            s = w.sum()
            if s == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= s
            w_rev = w[::-1]

            # ----- gradient of smooth max -------------------------
            Af = np.fft.rfft(a, n=L_fft)
            Wrev = np.fft.rfft(w_rev, n=L_fft)
            corr = np.fft.irfft(Af * Wrev, n=L_fft)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update ------------------------------------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # enforce symmetry and simplex
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected top‑k sub‑gradient descent on the true max.
# ----------------------------------------------------------------------
def _subgrad_topk_sym(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_score: float,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the average of the top‑k peaks."""
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] top‑k sub‑grad improve: score={cur_score:.6f}")

        # pick the k largest convolution entries
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(idxs))

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # enforce symmetry and project onto simplex
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Symmetric simulated‑annealing polish.
# ----------------------------------------------------------------------
def _anneal_sym(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealing while keeping the sequence mirror‑symmetric."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.40     # up to 40 % of source mass

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # mirror the move
        src_mir = n - 1 - src
        dst_mir = n - 1 - dst
        if src_mir != src:
            a_prop[src_mir] -= eps
            a_prop[dst_mir] += eps

        if np.any(a_prop < -1e-12):
            continue  # reject moves that would create negative entries

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a probability vector that minimises ``evaluate_sequence``.
    The routine respects the time budget (≈1000 s) and returns a plain
    Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Length schedule – we try increasingly larger vectors.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]
    prev_best = None
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise – either upscale the previously best shape or random.
        # --------------------------------------------------------------
        if prev_best is not None:
            a0 = np.interp(
                np.linspace(0.0, 1.0, n),
                np.linspace(0.0, 1.0, prev_best.size),
                prev_best,
            )
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            # possible global starter ``height_sequence_1``
            try:
                starter = np.asarray(height_sequence_1, dtype=float)
                if starter.ndim == 1 and starter.size > 0:
                    a0 = np.interp(
                        np.linspace(0.0, 1.0, n),
                        np.linspace(0.0, 1.0, starter.size),
                        starter,
                    )
                    a0 = np.maximum(a0, 0.0)
                    a0 = a0 / a0.sum()
                else:
                    raise Exception
            except Exception:
                a0 = np.random.rand(n)
                a0 = a0 / a0.sum()

        # --------------------------------------------------------------
        # Phase 1 – Adam on smooth surrogate (≈10 % of remaining budget)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        phase1_deadline = time.time() + min(5.0, 0.10 * remaining)
        a_adam, score_adam = _adam_smoothing_sym(
            a0,
            n,
            phase1_deadline,
            start_time,
            alphas=[1.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            lr0=0.5,
            iters_per_alpha=2500,
        )
        if score_adam < overall_best_score:
            overall_best_score = score_adam
            overall_best_seq = a_adam.copy()
            print(
                f"[{time.time() - start_time:.2f}s] phase 1 best (n={n}): score={score_adam:.6f}"
            )

        # --------------------------------------------------------------
        # Phase 2 – top‑k sub‑gradient descent (major chunk of the budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            phase2_deadline = time.time() + min(20.0, 0.70 * remaining)
            a_sub, score_sub = _subgrad_topk_sym(
                a_adam,
                n,
                phase2_deadline,
                start_time,
                init_score=score_adam,
                top_k=5,
            )
            if score_sub < overall_best_score:
                overall_best_score = score_sub
                overall_best_seq = a_sub.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] phase 2 best (n={n}): score={score_sub:.6f}"
                )
        else:
            a_sub = a_adam
            score_sub = score_adam

        # --------------------------------------------------------------
        # Phase 3 – final symmetric annealing polish (use whatever is left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _anneal_sym(
                a_sub,
                n,
                deadline,
                start_time,
                cur_best_score=overall_best_score,
                cur_best_seq=overall_best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=150_000,
            )
            if score_final < overall_best_score:
                overall_best_score = score_final
                overall_best_seq = a_final.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] annealing best (n={n}): score={score_final:.6f}"
                )
        else:
            a_final = a_sub

        # keep the best shape for the next longer length
        prev_best = overall_best_seq.copy()

    # ------------------------------------------------------------------
    # Final normalisation – safeguard against rounding errors.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback: simple uniform vector of length 1024
        n = 1024
        overall_best_seq = np.full(n, 1.0 / n)

    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    total = overall_best_seq.sum()
    if total > 0:
        overall_best_seq /= total
    else:
        overall_best_seq = np.full_like(overall_best_seq, 1.0 / overall_best_seq.size)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in overall_best_seq]
