import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Adam on smoothed max‑convolution
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max over convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                     # shape (2n‑1,)

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # extract the centred part (length n)
    offset = len(w_rev) - 1
    grad = 2.0 * corr[offset:offset + n]
    return grad


def _adam_phase(a0: np.ndarray, n: int,
                deadline: float, start_time: float,
                best_score: float, best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam on the log‑sum‑exp surrogate.
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    # logits y such that softmax(y)=a
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # increasing smoothness; larger α → tighter approximation of max
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 12000                     # a bit more than the reference

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # recompute a from logits
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smooth surrogate
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # (the true objective value is 2·n·f_val)
            # compute gradient w.r.t a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule: grad_y = ∂f/∂a * ∂a/∂y ; ∂a/∂y = diag(a) - a aᵀ
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 400 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # ---- sub‑gradient of max(conv) ----
        max_val = conv.max()
        # treat values within 1e-12 relative tolerance as active
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Directed incremental simulated annealing
# ----------------------------------------------------------------------
def _contributions_to_peak(a: np.ndarray, k_max: int) -> np.ndarray:
    """Return a vector whose i‑th entry is a_i·a_{k_max‑i} if the pair exists, else 0."""
    n = a.shape[0]
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    contrib = np.zeros_like(a)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k_max - i
    contrib[i] = a[i] * a[j]
    return contrib


def _run_directed_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            total_steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that preferentially removes mass from indices
    that contribute most to the current peak of the convolution.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    T_start = 5e-2
    T_end = 1e-6

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = T_start * (T_end / T_start) ** (it / total_steps)

        # current peak index
        k_max = int(conv.argmax())
        contrib = _contributions_to_peak(a, k_max)
        prob = contrib.sum()
        if prob > 0:
            src = rng.choice(n, p=contrib / prob)
        else:
            src = rng.choice(n, p=a)           # fall back to mass‑proportional

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction of the source mass (up to 30 %)
        max_frac = 0.30 * (T / T_start) + 0.02   # a tiny fraction when temperature is low
        eps = a[src] * rng.random() * max_frac
        if eps <= 0:
            continue

        # ---- apply move (using *old* a for the convolution update) ----
        # linear‑time slice updates
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ---- reject: revert the move ----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine mixes Adam smoothing, exact sub‑gradient polishing and a
    directed annealing phase, all respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.8

    # Problem size and initialisation
    # --------------------------------------------------------------
    n = 4800

        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, starter.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, starter)
                a0 /= a0.sum()
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{0.0:6.2f}s] initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed objective
    # --------------------------------------------------------------
    a_adam, best_score, best_seq = _adam_phase(
        best_seq, n, deadline, start_time, best_score, best_seq
    )

    # --------------------------------------------------------------
    # Phase 2 – Exact projected gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # give half of the remaining time to polishing
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Directed simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_directed_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            total_steps=2_500_000,          # as many moves as the remaining time permits
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise (protect against numerical drift) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final safety check (diagnostic)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
