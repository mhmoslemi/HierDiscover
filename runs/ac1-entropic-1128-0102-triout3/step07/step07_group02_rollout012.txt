def _run_gradient_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    best_score = float('inf')
    best_a = a.copy()
    # epsilon schedule: start fraction = 0.5, decay linearly to 0.01 as time passes
    while time.time() < deadline:
        conv = _conv_self(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val
        # compute gradient for max_idx: g[i] = 2 * a[max_idx - i] if valid else 0.
        g = np.zeros_like(a)
        i_low = max(0, max_idx - (n - 1))
        i_high = min(max_idx, n - 1)
        i_range = np.arange(i_low, i_high + 1)
        g[i_range] = 2.0 * a[max_idx - i_range]
        # donor = argmax(g * a) maybe we want high contribution
        donor_candidates = np.where(a > 1e-12)[0]
        donor = donor_candidates[np.argmax(g[donor_candidates])]
        # receiver = argmin(g)
        receiver = np.argmin(g)
        if donor == receiver:
            # if gradient zero for all, break
            break
        # compute epsilon max fraction based on schedule
        elapsed = (time.time() - start_time) / (deadline - start_time)
        eps_frac = 0.5 * (1 - elapsed) + 0.01  # from 0.5 to 0.01
        eps = min(a[donor] * eps_frac, 0.1)
        # test move
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps
        # compute new score
        conv2 = _conv_self(a_prop)
        new_score = 2.0 * n * conv2.max()
        if new_score < cur_score or np.random.rand() < 0.001:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] grad_local improve: n={n}, score={new_score:.6f}")
        else:
            # optionally try different donor/receiver (random)
            pass
    return best_a, best_score

"""
Propose a step‑function that (approximately) minimizes the objective
    2·n·max(conv(a,a)) / (sum(a))²

The algorithm consists of four stages:

1. **Multi‑scale warm‑start** – we begin from the previous best
   construction (if available) and also from a few simple analytic
   seeds (uniform, triangular).  Each seed is up‑sampled progressively
   to larger lengths while staying on the probability simplex.

2. **Smooth optimisation** – for the intermediate lengths we run a
   projected gradient descent on a *soft‑max* approximation of the
   max‑convolution.  An exponential temperature schedule (down to 1e‑6)
   gives a smooth surrogate that is easy to differentiate; Nesterov
   momentum and an adaptive step size speed up convergence.

3. **Exact sub‑gradient descent** – on the final (largest) length we
   switch to the true max‑convolution and perform projected sub‑gradient
   descent (again with momentum).  A short smoothing step is applied
   after this phase to damp high‑frequency spikes.

4. **Deterministic gradient‑driven local search** – the remaining time
   is spent shaving the peak of the convolution.  For the current
   maximal entry we compute its exact gradient, move mass from the
   index with the largest gradient to the index with the smallest
   gradient, and accept the move if the objective improves (or with a
   tiny uphill probability).  The step size decays linearly with the
   elapsed time, allowing coarse moves early and fine adjustments later.

All updates are projected onto the probability simplex
`{x ≥ 0, sum x = 1}`.  The best sequence found across all stages is
returned as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())      # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Convolution of reversed a with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a soft‑max of the convolution (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k)·conv_k with w_k = soft‑max(conv/temperature).
    The factor 2 comes from the derivative of aᵀJ_k a (Toeplitz matrix).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))   # improve stability
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Phase A – projected (Nesterov) gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7,
                    T_start: float = 2.0e-2,
                    T_end: float = 5.0e-4) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the softened max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # exponential temperature decay
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()          # actual objective (simplified)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # temperature schedule (smoothly to T_end)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.5,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:

    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient of the active convolution entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        # momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper: short smoothing filter (used after sub‑gradient phase)
# ----------------------------------------------------------------------
def _smooth(seq: np.ndarray) -> np.ndarray:
    """Apply a tiny 3‑point smoothing filter and renormalise."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    sm = np.convolve(seq, kernel, mode='same')
    sm = np.maximum(sm, 0.0)
    sm_sum = sm.sum()
    if sm_sum > 0:
        sm /= sm_sum
    else:
        sm = np.full_like(sm, 1.0 / sm.size)
    return sm

# Phase C – deterministic gradient‑driven “peak shaving” local search

    with the elapsed time, allowing coarse early moves and fine final

    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

        it += 1

        g = np.zeros_like(a)
        i_low = max(0, idx_max - (n - 1))
        i_high = min(idx_max, n - 1)
        i_range = np.arange(i_low, i_high + 1)
        g[i_range] = 2.0 * a[idx_max - i_range]

        donor_candidates = np.where(a > 1e-14)[0]
        if donor_candidates.size == 0:
            break
        donor = donor_candidates[np.argmax(g[donor_candidates] * a[donor_candidates])]

        # receiver = index with smallest gradient (zero is ideal)
        receiver = int(np.argmin(g))

        # simple Metropolis acceptance (few uphill moves allowed)
        if delta < 0 or np.random.rand() < 0.001:
            a = a_prop
            # keep track of best (already done above)
        # else keep current a

    return best_a, best_score
