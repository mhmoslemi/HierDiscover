Hybrid optimizer for the step‑function attack.

The search consists of three complementary stages, each respecting the global
time budget (≈1000 s) and operating on a probability vector `a` (∑a=1,
a_i≥0).

1. **Smooth max minimisation with L‑BFGS‑B** – the objective
   f(a)=2·n·max(a*a) is approximated by the log‑sum‑exp of the convolution.
   The variables are the unconstrained logits `z` (a=softmax(z)).
   A schedule of increasing smoothing parameters α gradually tightens the
   approximation while L‑BFGS exploits the exact gradient (computed via FFT).

2. **Exact projected sub‑gradient polishing** – starting from the best
   sequence found above, the true non‑smoothed objective 2·n·max(conv) is
   minimised by a momentum‑based projected sub‑gradient descent on the
   probability simplex.

3. **Fast incremental simulated‑annealing** – a specialised annealing loop
   transfers a small ε‑fraction of mass between two indices, updating the
   convolution incrementally in O(n) using vectorised slice operations.
   This stage performs many cheap moves to escape possible local minima.

The best sequence encountered during the whole run is returned (normalised
to sum = 1).  All numbers are non‑negative, satisfying the checks in
``evaluate_sequence``.
"""

from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global parameters for the smoothed objective (used by the gradient fn)
# ----------------------------------------------------------------------
CURRENT_ALPHA = None   # smoothing parameter α
CURRENT_N = None       # length of the sequence
CURRENT_FFT_LEN = None # FFT length for cross‑correlation


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray):
    """Return (objective, gradient) for the current α (global)."""
    alpha = CURRENT_ALPHA
    n = CURRENT_N
    fft_len = CURRENT_FFT_LEN

    # a = softmax(z)
    maxz = np.max(z)
    expz = np.exp(z - maxz)
    a = expz / expz.sum()

    # convolution a*a
    conv = _conv_self(a)

    # log‑sum‑exp of α·conv
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    lse = max_scaled + np.log(sum_exp)

    # objective = 2·n·lse/α
    obj = (2.0 * n * lse) / alpha

    # gradient w.r.t a (gradient of the smoothed max)
    grad_f = _grad_smooth_max(a, conv, alpha, fft_len)          # ∂f/∂a
    grad_a = 2.0 * n * grad_f                                   # ∂(2·n·f)/∂a

    # chain rule through soft‑max
    grad_z = a * (grad_a - np.dot(grad_a, a))                   # ∂obj/∂z
    return obj, grad_z


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """
    Sub‑gradient descent on the true objective 2·n·max(conv).
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum + Nesterov‑style
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 1_000_000):
    """
    Mass‑transfer annealing where the convolution is updated incrementally.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # source chosen proportionally to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += eps * eps

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method mixes L‑BFGS on a smoothed max‑convolution, exact sub‑gradient
    polishing and a fast incremental simulated annealing, all respecting the
    supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # small safety margin

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800

    # Try to start from the previously discovered construction if available.
    a0 = None
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                a0 = starter.copy()
                if a0.shape[0] != n:
                    a0 = _upsample(a0, n)
        except Exception:
            a0 = None

    # Fallback: a smooth symmetric triangular shape.
    if a0 is None:
        mid = n // 2
        inc = np.arange(1, mid + 1, dtype=float)
        dec = inc[::-1] if n % 2 == 0 else inc[:-1][::-1]
        a0 = np.concatenate((inc, dec))
        a0 = a0 / a0.sum()

    # Normalise safely
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[0.00s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smoothed objective (log‑sum‑exp)
    # ------------------------------------------------------------------
    # Initialise logits for the soft‑max parametrisation
    eps = 1e-12
    z0 = np.log(np.maximum(a0, eps))

    # Global variables accessed by the gradient routine
    global CURRENT_ALPHA, CURRENT_N, CURRENT_FFT_LEN
    CURRENT_N = n
    CURRENT_FFT_LEN = _next_pow2(3 * n - 2)

    # A schedule of increasing smoothing parameters
    alpha_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 320.0, 1280.0]

    for alpha in alpha_schedule:
        if time.time() >= deadline:
            break
        CURRENT_ALPHA = alpha
        maxiter = 300                     # number of L‑BFGS iterations per α
        # Run L‑BFGS‑B; the objective returns both value and gradient.
        try:
            res = minimize(_smooth_obj_and_grad, z0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter,
                                    'disp': False,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12})
        except Exception as e:
            print(f"  L‑BFGS error (α={alpha}): {e}")
            break

        z0 = res.x                         # warm‑start for the next α
        a = np.exp(z0 - np.max(z0))
        a = a / a.sum()
        sc = float(evaluate_sequence(a.tolist()))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] L‑BFGS α={alpha:.1f} improve → {sc:.6f}")
        else:
            print(f"[{time.time() - start_time:6.2f}s] L‑BFGS α={alpha:.1f} finished, score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact projected sub‑gradient polishing (intense)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        # allocate roughly half of the remaining budget to polishing
        remaining = deadline - now
        polish_deadline = now + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.30,
            max_iter=5_000_000)
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] Polishing improve → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Fast incremental annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # the remaining half of the budget is spent in annealing
        total_moves = 2_000_000
        a_final, score_final = _run_incremental_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            total_steps=total_moves)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Annealing improve → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and final check
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    # Return as plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
