def solve_g_max_sum(a, M, active_tol=1e-3):
    n = a.shape[0]
    conv = np.convolve(a, a)
    active = np.where(conv >= M * (1 - active_tol))[0]
    K = active.size
    if K == 0:
        active = np.arange(conv.shape[0])
        K = active.size
    A = np.zeros((K, n), dtype=float)
    for row_idx, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for each i in [i_start,i_end], set column j=k-i.
        # So columns j = k - i.
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        A[row_idx, j_range] = a[i_range]
    b = np.full(K, M)
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        g = res.x
    else:
        g = np.zeros_like(a)
    return g

"""
Improved step‑function search with an additional linear‑programming mixing phase.

The algorithm consists of four stages:
1.  a short exponentiated‑gradient warm‑up on a smooth approximation of the max.
2.  a longer exact‑max projected sub‑gradient descent.
3.  a novel LP‑mixing step: maximise the total mass of a partner
    distribution g under the current convolution bound and mix a small
    amount of g into the current solution.  This is the “reference”
    iteration described in the paper and often yields a noticeable
    drop in the max‑convolution.
4.  a targeted local‑refinement that moves probability mass away from
    indices that contribute most to the current maximal convolution entry.

All stages respect the 1000 s time budget, log progress, and finally
return the best sequence found (as a plain list of floats).  The code
uses only NumPy and SciPy (HiGHS linear‑programming solver) and therefore
conforms to the execution environment restrictions.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : LP‑mixing (partner distribution) step
# ----------------------------------------------------------------------


def _solve_lp_for_g(a: np.ndarray, M: float, active_rel_tol: float = 1e-3):
    """
    Solve   max Σ g_j   s.t.  (a * g)[k] ≤ M   for all k with
    a * a close to the maximum (relative tolerance `active_rel_tol`).
    Returns the feasible vector g (may be all zeros if LP fails).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    active = np.where(conv >= M * (1.0 - active_rel_tol))[0]
    if active.size == 0:
        # fallback: use all constraints (more expensive)
        active = np.arange(conv.shape[0])

    K = active.size
    # Build the constraint matrix A (K × n)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        A[row, j] = a[i]

    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σg  → minimise -Σg
    bounds = [(0, None)] * n

    # HiGHS is the default method for recent SciPy releases and handles
    # moderate‑size LPs very efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # If the solver fails we simply return a zero vector – the caller
        # will ignore it.
        return np.zeros_like(a)


def _run_lp_mixing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 30,
                   init_t: float = 0.25,
                   min_t: float = 1e-4,
                   active_rel_tol: float = 1e-3):
    """
    Repeatedly (up to `max_iters`) solve the LP described above,
    then mix a small amount of the partner distribution into the current
    solution.  The mixing coefficient is reduced until a strict improvement
    of the maximal convolution entry is observed.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_max = conv.max()
    best_score = 2.0 * n * best_max
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        M = conv.max()

        g = _solve_lp_for_g(a, M, active_rel_tol=active_rel_tol)
        total_g = g.sum()
        if total_g <= 1e-12:
            # No feasible partner with positive mass – exit early.
            break

        # Normalise the partner to a probability vector.
        g_norm = g / total_g

        # Try a sequence of step sizes, halving each time.
        t = init_t
        improved = False
        while t >= min_t:
            a_candidate = (1.0 - t) * a + t * g_norm
            a_candidate /= a_candidate.sum()
            new_max = _conv_self(a_candidate).max()
            if new_max < M - 1e-12:
                a = a_candidate
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_score = 2.0 * n * new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] LP‑mix improve: "
                          f"iter={it:,}, t={t:.5f}, max={best_max:.8f}")
                improved = True
                break
            else:
                t *= 0.5

        if not improved:
            # Could not find an improving mixing coefficient – stop.
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : targeted local refinement (tiny moves focused on the max)
# ----------------------------------------------------------------------


def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         eps_frac: float = 0.025):
    """
    Moves a small fraction of mass away from coordinates that contribute
    most to the current maximal convolution entry.  Only moves that strictly
    lower the maximum are kept.  Destination indices are chosen proportionally
    to *low* contribution, which yields faster convergence than uniform
    sampling used previously.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        k_max = int(np.argmax(conv))

        # contribution of each index to this maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            prob_src = contrib / total_contrib          # high‑contrib → source
            prob_dst = (total_contrib - contrib) / total_contrib  # low‑contrib → destination
        else:
            prob_src = a / a.sum()
            prob_dst = np.full_like(a, 1.0 / n)

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # destination – biased towards low contribution indices
        dst = rng.choice(n, p=prob_dst)
        if dst == src:
            continue

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # Early exit check – keep the loop responsive to the time limit
        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a
