"""
Propose a step‑function (non‑negative list of heights) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1.  Adam optimisation on a smooth approximation of the max‐convolution.
    The smoothness parameter α is increased gradually, so the optimiser
    first sees a “soft‑max” of the convolution and later the true max.
    After each gradient step the vector is projected onto the probability
    simplex (non‑negative entries summing to one).

2.  A targeted refinement that repeatedly moves a tiny amount of probability
    mass from indices that contribute most to the current maximal
    convolution entry (chosen proportionally to the sub‑gradient) to indices
    that have the smallest sub‑gradient.  Only moves that strictly lower the
    maximal convolution value are kept.

3.  Final normalisation and conversion to a plain Python list.

All helper functions are top‑level, the search respects a 1000 s time
budget and returns the best sequence found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose value
    lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """
    The evaluation objective when the vector is already normalised to sum=1:
        2 * n * max(conv)
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase‑1 : Adam on a smooth max (soft‑max)
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray,
              deadline: float,
              start_time: float,
              alpha_schedule,
              iters_per_alpha: int = 3000,
              lr: float = 0.15,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best vector (normalised) and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    # FFT length needed for the correlation a * w_rev
    L_corr = _next_pow2(3 * n - 2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = _objective(a)

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"α={alpha:.1f}, score={cur_score:.6f}")

            # --- soft‑max weights ---
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # --- gradient of the smoothed max ---
            Af = np.fft.rfft(a, n=L_corr)
            Wrev = np.fft.rfft(w_rev, n=L_corr)
            corr = np.fft.irfft(Af * Wrev, n=L_corr)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # --- Adam update + simplex projection ---
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps) * m_hat
            a = a - step
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : targeted local refinement
# ----------------------------------------------------------------------


def _run_targeted_refine(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         eps_frac: float = 0.02,
                         dest_pool: int = 12,
                         max_iters: int = 10 ** 7):
    """
    Greedy mass‑transfer refinement.
    Each iteration moves a small ε from a source index (chosen with probability
    proportional to its sub‑gradient contribution to the current maximal
    convolution entry) to a destination with the smallest sub‑gradient.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # sub‑gradient of the current maximal entry
        grad = _subgradient_full(a, conv)

        # source – probability ∝ gradient magnitude
        prob_src = grad + 1e-20
        prob_src /= prob_src.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # destination – pick among the indices with smallest gradient
        # (i.e. where moving mass is least harmful)
        low_idx = np.argpartition(grad, dest_pool)[:dest_pool]
        if src in low_idx:
            # replace with a random alternative to keep src ≠ dst
            alt = rng.integers(0, n)
            while alt == src:
                alt = rng.integers(0, n)
            low_idx = np.append(low_idx, alt)
        dst = rng.choice(low_idx)

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-13:          # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-13:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

    return best_a

    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(30.0, 0.70 * remaining)

    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                      80.0, 160.0, 320.0, 640.0]    # from smooth to sharp

    a_adam, score_adam = _run_adam(
        a0,
        phase1_deadline,
        start_time,
        alpha_schedule=alpha_schedule,
        iters_per_alpha=2500,   # ≈ 27 k updates total → still well within time
        lr=0.15,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] phase 1 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted refinement (focus on the active max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_targeted_refine(
            best_seq,
            deadline,
            start_time,
            eps_frac=0.018,
            dest_pool=12,
        )
        score_ref = _objective(a_refined)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
