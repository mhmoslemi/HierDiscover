def _generate_candidate(rng, n):
    typ = rng.integers(0,5)
    if typ == 0:
        return np.full(n, 1.0/n)
    elif typ == 1: # dirichlet
        alpha = rng.choice([0.1,0.3,0.5,1.0,2.0])
        vals = rng.gamma(alpha, 1.0, size=n)
        vals /= vals.sum()
        return vals
    elif typ == 2: # sparse random subset
        m = rng.integers(int(0.2*n), int(0.8*n)+1)
        idx = rng.choice(n, size=m, replace=False)
        vals = np.zeros(n)
        # assign random positive weights via gamma
        weights = rng.exponential(scale=1.0, size=m) # positive
        vals[idx] = weights
        if vals.sum() == 0:
            vals[idx] = 1.0
        vals /= vals.sum()
        return vals
    elif typ == 3: # V shape simple
        p = rng.uniform(0.5, 2.0)
        i = np.arange(n)
        v = 1.0/(i+1)**p + 1.0/(n-i)**p
        v /= v.sum()
        return v
    else: # also random expo decay
        r = rng.uniform(0.95,0.999)
        v = np.array([r**i for i in range(n)], dtype=float)
        v /= v.sum()
        return v

    eps_small = 1e-20
    y = np.log(np.maximum(a0, eps_small))
    n = a0.shape[0]
    fft_len = _next_pow2(3*n - 2)
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 4000   # maybe small
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_seq = a0.copy()
    best_score = _score_of(best_seq)
    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()
            conv = _conv_self(a)
            # compute smooth objective
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + math.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # gradient

            grad_y = a * (grad_a - np.dot(grad_a, a))
            # Adam
            t += 1
            m = beta1*m + (1-beta1)*grad_y
            v = beta2*v + (1-beta2)*(grad_y**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr_t = lr0 * (math.sqrt(1 - beta2**t) / (1 - beta1**t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if i % 250 == 0:

                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam improve score {score:.6f}")

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

The new search consists of three stages that respect the global time
budget:

1. **Diverse random seeding** – many candidate vectors are generated
   (uniform, Dirichlet with various concentrations, random sparse
   subsets and a simple “inverse‑triangular” shape).  The 5–10 best
   candidates (lowest max‑convolution) are kept.  This stage consumes
   roughly 15 % of the total time and already discovers vectors whose
   score is often below the previous record.

2. **Smooth Adam refinement** – each retained seed is optimized with
   Adam on the smooth log‑sum‑exp approximation of the max‑convolution.
   The smoothing parameter α is increased gradually (continuation) and
   the objective is projected onto the probability simplex via the
   Euclidean projection routine.  The number of Adam steps is
   dynamically limited by the remaining time.

3. **Targeted greedy peak‑reduction + simulated annealing** – a cheap
   deterministic move pushes mass from the coefficient that most
   contributes to the current peak towards the smallest coefficient.
   The best vector is then refined by a classic annealing scheme that
   repeatedly transfers a random ε‑fraction of mass between two indices.
   The schedule is exponential (high temperature → low temperature) and
   runs until the time budget is exhausted.

All helper functions are top‑level; no closures or lambdas are used.
The algorithm works purely with NumPy and stays well inside the 1000 s
limit.  The final sequence is normalised (sum = 1) before being
returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# ----- Low‑level helpers ------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a) = (1/α)·log Σ_k exp(α·conv_k)
    using the soft‑max weights w_k = exp(α·conv_k)/Σ.
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)

    # keep only the needed part (length 3·n‑2) and extract the gradient
    needed = 3 * n - 2
    corr = corr[:needed]
    grad = 2.0 * corr[(len(w_rev) - 1) - np.arange(n)]

    return grad


def _score_of(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (sum is assumed >0)."""
    n = a.shape[0]
    conv = _conv_self(a)
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:                     # protects against degenerate vectors
        return math.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# ----- Candidate generation ---------------------------------------------
# ----------------------------------------------------------------------


def _random_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Return a random non‑negative vector of length n.
    Five different families are mixed:
        0 – uniform
        1 – Dirichlet (concentration α drawn uniformly)
        2 – random sparse subset with exponential weights
        3 – inverse‑triangular (high at the ends, low in the middle)
        4 – exponential decay
    """
    typ = rng.integers(0, 5)
    if typ == 0:                              # uniform
        a = np.full(n, 1.0 / n)

    elif typ == 1:                            # Dirichlet
        # concentration between 0.05 and 5.0 (log‑uniform)
        log_alpha = rng.uniform(np.log(0.05), np.log(5.0))
        alpha = math.exp(log_alpha)
        g = rng.gamma(alpha, 1.0, size=n)
        a = g / g.sum()

    elif typ == 2:                            # sparse subset
        # size between 0.2 n and 0.9 n
        m = rng.integers(int(0.2 * n), int(0.9 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        w = rng.exponential(scale=1.0, size=m)
        a = np.zeros(n)
        a[idx] = w
        # avoid a completely zero vector
        if a.sum() == 0.0:
            a[idx[0]] = 1.0
        a /= a.sum()

    elif typ == 3:                            # inverse‑triangular (V‑shape)
        p = rng.uniform(0.5, 2.0)             # shape exponent
        i = np.arange(n, dtype=float)
        v = 1.0 / (i + 1) ** p + 1.0 / (n - i) ** p
        a = v / v.sum()

    else:                                     # exponential decay
        r = rng.uniform(0.95, 0.999)
        v = np.array([r ** i for i in range(n)], dtype=float)
        a = v / v.sum()

    return a


# ----------------------------------------------------------------------
# ----- Smooth Adam optimisation (log‑sum‑exp) ---------------------------
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth approximation of the max‑convolution.
    Returns the best vector seen and its exact (non‑smooth) score.
    """
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))               # logits
    n = a0.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    # continuation over smoothing parameters
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0,
                      20.0, 40.0, 80.0, 160.0, 320.0]
    # number of Adam steps per α – will be trimmed by the deadline
    steps_per_alpha = 4000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a0.copy()
    best_score = _score_of(best_a)

    for alpha in alpha_schedule:
        for it in range(steps_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + math.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                    # smooth approx. of max
            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (math.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))

            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation of the true objective
            if it % 250 == 0:
                cur_score = _score_of(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve – score={cur_score:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# ----- Deterministic greedy peak‑reduction -----------------------------
# ----------------------------------------------------------------------


def _peak_reduction(a0: np.ndarray, deadline: float, start_time: float,
                    max_no_improve: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy move of a fraction of the coefficient that contributes most
    to the current max‑convolution peak towards the smallest coefficient.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]                     # ordered contributions
        src = i_idx[contrib.argmax()]                     # most “responsible” index

        # destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps_move = a[src] * 0.25                         # move a quarter of src mass
        if eps_move <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] Peak‑reduce – new max={new_max:.6f}")
        else:
            no_improve += 1

    final_score = _score_of(a)
    return a, final_score


# ----------------------------------------------------------------------
# ----- Simulated annealing ---------------------------------------------
# ----------------------------------------------------------------------


def _annealing(a0: np.ndarray, deadline: float, start_time: float,
               start_temp: float = 5e-2, end_temp: float = 1e-6,
               steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Classic annealing: transfer a random ε‑fraction of mass from a
    source (chosen proportionally to its current weight) to a random
    destination.  Moves are accepted according to the Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _score_of(a)

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source index with probability proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (up to 30 %) of src's mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_of(a_prop)
        delta = new_score - best_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal improve – score={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# ----- Main driver ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step function that minimises ``evaluate_sequence``.
    The routine consists of a fast random‑search phase, a smooth Adam
    polishing phase for the best few seeds and a final greedy+annealing
    refinement.  All work is confined to the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5                 # tiny safety margin

    # problem size – the reference size works well and keeps all FFTs cheap
    n = 4800

    # ------------------------------------------------------------------
    # Phase 1 – generate many random candidates, keep the best few
    # ------------------------------------------------------------------
    top_k = 7
    best_candidates = []                # list of (score, vector) tuples

    phase1_deadline = start_time + 0.15 * budget_s   # ~150 s
    while time.time() < phase1_deadline:
        a = _random_candidate(rng, n)
        sc = _score_of(a)

        if len(best_candidates) < top_k:
            best_candidates.append((sc, a))
            best_candidates.sort(key=lambda x: x[0])
        else:
            # replace the worst if the new one is better
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, a)
                best_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:6.2f}s] Phase 1 complete – "
          f"best seed score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth Adam refinement for each retained seed
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = math.inf

    # allocate the remaining time equally among the seeds
    remaining = deadline - time.time()
    per_seed_deadline = time.time() + max(0.0, remaining / max(len(best_candidates), 1))

    for idx, (seed_score, seed_vec) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Optimising seed #{idx + 1}")

        # Adam optimisation (may finish earlier than the per‑seed deadline)
        a_opt, sc_opt = _adam_optimize(seed_vec, per_seed_deadline, start_time)

        # deterministic greedy peak reduction (cheap)
        if time.time() < deadline:
            a_opt, sc_opt = _peak_reduction(a_opt, deadline, start_time)

        if sc_opt < global_best_score:
            global_best_score = sc_opt
            global_best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST *** "
                  f"score={global_best_score:.6f}")

        # move the per‑seed deadline forward for the next seed
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + max(0.0, remaining / max(len(best_candidates) - idx - 1, 1))

    # ------------------------------------------------------------------
    # Phase 3 – final annealing on the overall best vector
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – uniform distribution
        global_best_seq = np.full(n, 1.0 / n)

    if time.time() < deadline:
        a_final, sc_final = _annealing(global_best_seq, deadline, start_time)
        if sc_final < global_best_score:
            global_best_seq = a_final.copy()
            global_best_score = sc_final
            print(f"[{time.time() - start_time:6.2f}s] Final anneal improved to {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (protect against tiny numerical drift) and return
    # ------------------------------------------------------------------
    best = np.maximum(global_best_seq, 0.0)
    total = best.sum()
    if total > 0.0:
        best /= total
    else:
        best = np.full_like(best, 1.0 / n)

    final_score = evaluate_sequence(best.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return a plain Python list of floats
    return [float(x) for x in best.tolist()]
