"""
Propose a step‑function that minimises the peak of its self‑convolution.

Algorithm
---------
1. Warm‑start from the previous best sequence (``height_sequence_1``) if it
   exists, otherwise start from the uniform distribution.
2. Linear‑programming refinement:
   * For the current shape compute the convolution and collect the K
     largest shifts (the active constraints where the peak is attained).
   * Solve a tiny LP (maximise Σ g_j) subject to the *linear* constraints
     f * g ≤ current_peak on those K shifts and g ≥ 0.
   * Mix the current shape towards the LP‑solution (convex combination,
     step size 0 < t ≤ 0.3) and re‑project onto the probability simplex.
   * Repeat while time permits – each LP has only a few rows, thus it
     stays cheap even for n≈10 000.
3. Smooth‑max (log‑sum‑exp) Adam gradient descent – a few thousand
   iterations on the full sequence to further flatten the convolution.
4. Targeted peak‑shaving walk (the Metropolis local search used in the
   reference code) for the remaining budget.
5. Return the best sequence seen (normalised to sum = √(2·n) is not
   required – the evaluation function is scale‑invariant).

The whole routine respects the 1000 s budget and prints progress
messages.  The final list is plain Python floats, as required by the
harness.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Helper: smooth max (log‑sum‑exp) gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of  (1/beta)·log Σ_k exp(beta·conv_k)   w.r.t. a.
    Returns a vector of length n.
    """
    # soft‑max weights
    maxc = conv.max()
    scaled = (conv - maxc) * beta
    exps = np.exp(scaled - np.max(scaled))   # stabilise
    w = exps / exps.sum()                    # Σ w_k = 1
    # gradient = 2 * (rev(a) * w)  (convolution)
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# Score (scale‑invariant)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a normalised to sum=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray,
             n: int,
             cur_max: float,
             active_idx: np.ndarray,
             mix: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ cur_max  on the rows in ``active_idx``,
    g ≥ 0.  Return the mixed point  (1‑mix)·f + mix·g after normalising g.
    """
    m = active_idx.size
    rows = []
    cols = []
    data = []

    # Build the sparse constraint matrix A (m × n)
    for r, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for each i in [i_start, i_end] we have coefficient f[i] at column j=k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(r)
            cols.append(j)
            data.append(f[i])

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(m, n), dtype=float)

    c = -np.ones(n)                 # maximise Σ g_j
    b_ub = np.full(m, cur_max, dtype=float)
    bounds = [(0, None)] * n

    # small tolerance to avoid numerical trouble
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

    if not res.success:
        return None

    g0 = res.x
    tot = g0.sum()
    if tot <= 0:
        return None
    g = g0 / tot                    # enforce sum = 1

    # convex combination and projection back onto the simplex
    new_f = (1.0 - mix) * f + mix * g
    new_f = _project_simplex(new_f)
    return new_f

# ----------------------------------------------------------------------
# LP‑based refinement (few hundred cheap LP solves)
# ----------------------------------------------------------------------
def _run_lp_refine(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 300,
                   mix: float = 0.25,
                   active_k: int = 20) -> tuple[np.ndarray, float]:
    """Iteratively improve ``a0`` by solving tiny LPs on the active shifts."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()

        # pick the K largest shift indices (active constraints)
        if active_k >= len(conv):
            active_idx = np.arange(len(conv))
        else:
            # argpartition gives unsorted top‑K indices efficiently
            active_idx = np.argpartition(-conv, active_k)[:active_k]

        new_a = _lp_step(a, n, cur_max, active_idx, mix)
        if new_a is None:
            break
        a = new_a
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter={it}, "
                  f"score={sc:.6f}")

        # optional reduction of mixing factor when progress stalls
        if it % 40 == 0 and sc > best_sc * 0.9999:
            mix = max(0.05, mix * 0.8)

    return best_a, best_sc

# ----------------------------------------------------------------------
# Adam optimisation on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.02,
              beta1: float = 0.9,
              beta2: float = 0.999,
              T_start: float = 1e-1,
              T_end: float = 5e-4,
              max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """Adam on the smooth approximation of max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)

        # temperature schedule (log‑sum‑exp smoothing)
        T = T_start * (T_end / T_start) ** (it / max_iter)
        beta = 1.0 / T                     # β in the soft‑max formula

        grad = _softmax_gradient(a, conv, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = _project_simplex(a)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve iter={it}, "
                  f"score={sc:.6f}")

    return best_a, best_sc

# ----------------------------------------------------------------------
# Targeted local search (same idea as the reference implementation)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis walk that shifts a small amount of mass from hot to cold spots."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_sc = _score(a)
    best_a = a.copy()
    best_sc = cur_sc

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_sc = _score(a)

        i = np.arange(n)
        j = idx_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_sc = _score(a_prop)
        delta = new_sc - cur_sc

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve iter={it}, "
                      f"score={new_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    The pipeline consists of a warm‑start, a small LP‑based refinement,
    Adam‑based smooth‑max descent and a final Metropolis local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a working length (the previous best used 9600)
    # ------------------------------------------------------------------
    n = 9600
    if starter is not None and starter.shape[0] == n:
        a_cur = starter.copy()
    else:
        # uniform distribution of length n
        a_cur = np.full(n, 1.0 / n)

    # ensure exact normalisation
    a_cur = _project_simplex(a_cur)

    best_a = a_cur.copy()
    best_sc = _score(best_a)
    print(f"[{time.time() - start_time:6.2f}s] start score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming refinement (≈ 25 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

                                 start_time, max_iter=300,
                                 mix=0.25, active_k=20)
    if sc_lp < best_sc:

    # ------------------------------------------------------------------

                                lr=0.02, beta1=0.9,
                                beta2=0.999,

                                max_iter=20000)
    if sc_adam < best_sc:

        print(f"[{time.time() - start_time:6.2f}s] after Adam: score={best_sc:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
