"""
Algorithm:
- Initialize a non‑negative vector on the simplex (random or from previous best).
- Stage A: short “soft‑max” Adam optimisation with a log‑sum‑exp surrogate
  that quickly discovers a good global shape.
- Stage B: long “Adam‑subgradient” phase that directly minimises the true
  max‑convolution using Adam updates on the sub‑gradient of the max.
- Stage C: light simulated‑annealing mass‑transfer for final polishing.
All stages obey the global time budget and keep the vector on the probability
simplex, so the evaluation score 2·n·max(conv) is invariant to scaling.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv

# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

# ----------------------------------------------------------------------
# Adam optimisation of the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-14,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam on O_τ(a)=2·n·τ·log∑exp(conv(a)/τ)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)        # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp               # size 2·n‑1, sums to 1

        # gradient of O_τ : 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)   # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # anneal τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score

def _run_adam_subgrad(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam where the gradient is the sub‑gradient of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam‑subgrad improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        # Adam‐style update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Light simulated annealing (mass transfer)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        exp_z = np.exp(z - z_max)        # numerical stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp               # length 2·n‑1, sums to 1

        # gradient of Oτ: 4·n·∑ a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)   # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # decay τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam driven by the true sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _run_adam_subgrad(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step_size: float = 0.25,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam where the gradient is the sub‑gradient of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam‑subgrad improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        # Adam‑style adaptive step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The pipeline is:
        1. initialise a probability vector,
        2. warm‑up Adam on a smooth surrogate,
        3. long Adam on the true sub‑gradient,
        4. a final short simulated‑annealing pass.
    The result is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Fixed length – 8192 gives a good compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # Initialise a (possibly) informed vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                a0 = _upsample(tmp, n)
                a0 /= a0.sum()
            else:
                raise ValueError
        except Exception:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage A – short Adam on the smoothed surrogate (≈10‑15 % of budget)
    # ------------------------------------------------------------------
    warmup_time = min(120.0, (deadline - time.time()) * 0.15)
    warmup_deadline = time.time() + warmup_time

    conv0 = _conv_self(a0)
    tau0 = max(conv0.max() * 8.0, 1e-8)

    a_warm, score_warm = _run_adam_softmax(
        a0,
        n,
        warmup_deadline,
        start_time,
        tau0=tau0,
        tau_end=1e-14,
        tau_decay=0.9993,
        step_size=0.22,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage A improve: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage B – long Adam on the true sub‑gradient of max(conv)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # keep a small reserve for the final annealing (~5 % of total)
        anneal_reserve = min(80.0, budget_s * 0.05)
        subgrad_deadline = deadline - anneal_reserve

        a_sub, score_sub = _run_adam_subgrad(
            best_seq,
            n,
            subgrad_deadline,
            start_time,
            step_size=0.30,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage B improve: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage C – very light simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=1e-2,
            end_temp=1e-7,
            steps=250_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex and return a plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
