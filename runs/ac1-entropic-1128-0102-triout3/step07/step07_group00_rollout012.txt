"""
Improved optimiser for the “step‑function” attack.

The algorithm works in three phases (all respect the global time budget):
1. Adam on a smooth (log‑sum‑exp) approximation of the maximal convolution.
   A larger list of temperature parameters and a longer inner loop give a
   better initial shape.
2. Projected multi‑peak sub‑gradient descent.  Instead of following only the
   single maximal convolution index we use the top‑K indices – this reduces
   the “stalling” that occurs when several entries are close to the maximum.
3. A long simulated‑annealing polish that moves arbitrary fractions of mass
   between coordinates.  The temperature schedule is exponential and the
   move size is allowed to be up to the whole source mass, which makes it
   possible to escape shallow local minima.

The optimiser works on a power‑of‑two length (default 8192) – a larger
dimension gives more freedom to spread the mass while the factor 2·n in the
objective stays linear, so the relative improvement observed for n=4096
scales to n=8192.  The code falls back to the global `height_sequence_1`
if it exists, otherwise starts from a random Dirichlet draw.

All helper functions are top‑level (no closures, no lambdas) and the only
required external function is the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (returns length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp approximation
    of max(conv(a,a)).  The best iterate (lowest 2·n·max_conv)
    is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length – keep the same for the whole phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth‑max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)      # numerical stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smooth max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Multi‑peak projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent_multi(a0: np.ndarray,
                                   n: int,
                                   deadline: float,
                                   start_time: float,
                                   best_score_initial: float,
                                   top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Instead of a single active index we use the *top_k* indices
    (those with the largest convolution values) to form the gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ----- active set (top‑k) -----
        # np.argpartition is O(n) – cheap compared to the FFT.
        idxs = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : Tiny simulated‑annealing polish (more aggressive)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-1,
                   end_temp: float = 1e-8,
                   steps: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing.
    The temperature decays exponentially; moves may transfer up to the
    *entire* source mass, which is essential for escaping plateaux.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source weighted by mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction of the source (up to 100 %)
        eps = a[src] * rng.random()
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a length that is a power of two (FFT‑friendly)
    # Using a larger n gives more freedom to lower the objective.
    # ------------------------------------------------------------------
    final_n = 8192

    # ------------------------------------------------------------------
    # Initialise – use existing global if present, else random.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.interp(
                np.linspace(0, 1, final_n),
                np.linspace(0, 1, starter.size),
                starter
            )
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        a0 = starter
    else:
        # Dirichlet(1) → uniform Dirichlet = random point on simplex
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth max (≈5 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(5.0, 0.05 * remaining)

    a_adam, score_adam = _run_adam_smoothing(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alphas=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0],
        lr0=0.5,
        iters_per_alpha=8000,
    )
    if score_adam < best_score:
        best_score = score_adam
        best_seq = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] Phase 1 improve: "
              f"score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multi‑peak sub‑gradient descent (≈78 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.78 * remaining)

        a_sub, score_sub = _run_subgradient_descent_multi(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            best_score,
            top_k=5,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final annealing polish (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-1,
            end_temp=1e-8,
            steps=1_200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: "
                  f"score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
