def simulated_annealing(a, deadline, start_time):
    n = len(a)
    cur_seq = a.copy()
    cur_score = objective(cur_seq)  # 2*n*conv_max

    best_seq = cur_seq.copy()
    best_score = cur_score

    # compute schedule: start_temp maybe 0.01? but objective measured in absolute scores ~1.5, delta typical maybe 0.001. temperature may be around delta magnitude.
    # We start with T0 = 0.001 maybe and decay to 1e-8.

    while time.time() < deadline:
        # choose donor and recipient randomly weighted maybe by a_i magnitude
        # but we can just choose random i, j.
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i: j += 1

        if cur_seq[i] <= 1e-12:
            continue
        # epsilon from i: proportion of a_i; maybe sample exponential distribution
        eps = cur_seq[i] * np.random.rand() * 0.5  # up to half of donor mass
        # propose new sequence
        cand = cur_seq.copy()
        cand[i] -= eps
        cand[j] += eps

        # compute objective
        cand_score = _objective(cand)

        # compute temperature based on elapsed fraction
        elapsed = (time.time() - start_time) / (deadline - start_time)
        # Use schedule: T = T0 * (T_end/T0)**elapsed
        T0 = 5e-2
        T_end = 1e-6
        T = T0 * (T_end / T0) ** elapsed

        # acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: score={cand_score:.6f}")

    return best_seq, best_score

"""
Simulated‑Annealing optimiser for the “step‑function’’ challenge.

The algorithm proceeds as follows:

1. **Warm‑start** – if the global variable ``height_sequence_1`` exists it
   is used (normalised to sum = 1); otherwise a random point on the simplex
   is generated.

2. **Annealing phase** – the bulk of the time budget (~80 %) is spent in a
   simulated‑annealing loop that repeatedly transfers a random fraction of the
   mass of a randomly chosen donor index to a random recipient index.  The
   acceptance probability follows the usual Metropolis rule with a temperature
   schedule decreasing from 5×10⁻² to 10⁻⁶.  The objective being minimised is
   the exact evaluation function (i.e. 2·n·max(a∗a) because the sum of the
   coefficients is kept equal to 1).

3. **Permutation refinement** – any remaining time is used to try random
   permutations of the current multiset of heights; because the evaluation
   depends on the ordering, a different permutation can lower the maximal
   convolution value.

4. **Return** – the best sequence found is re‑normalised to the probability
   simplex and emitted as a plain list.

The implementation uses only NumPy and the standard library, works in a
single‑CPU environment and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the max of a self‑convolution
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq*seq)_k using an FFT (linear convolution)."""
    n = seq.shape[0]
    # length of the FFT – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(F * F, fft_len)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper: exact objective (sum of the sequence is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a normalised sequence."""
    n = seq.shape[0]
    max_conv = _max_convolution(seq)
    return 2.0 * n * max_conv

                         deadline: float,

    randomly chosen donor index is transferred to a randomly chosen recipient.
    The Metropolis rule with a geometric temperature schedule decides acceptance.
    """
    a = a0.copy()
    n = a.shape[0]

    # Normalise to the probability simplex (the objective is scale‑invariant)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a[:] = 1.0 / n

    cur_score = _objective(a)
    best_seq = a.copy()
    best_score = cur_score

    # temperature schedule parameters
    T_start = 5e-2
    T_end = 1e-6

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ---- pick donor (probability ∝ current weight) ----
        if a.max() == 0:
            break
        donor = np.random.choice(n, p=a / a.sum())

        # ensure the donor actually has mass to give
        if a[donor] <= 1e-15:
            continue

        # ---- pick recipient (any other index) ----
        recipient = np.random.randint(n - 1)
        if recipient >= donor:
            recipient += 1

        # ---- amount of mass to move (up to 80 % of donor) ----
        fraction = np.random.rand() * 0.80
        eps = a[donor] * fraction
        if eps <= 0.0:
            continue

        # ---- propose the move (in‑place) ----
        a[donor] -= eps
        a[recipient] += eps

        # ---- evaluate the new candidate ----
        new_score = _objective(a)

        # ---- temperature (geometric decay) ----
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        # ---- Metropolis acceptance ----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: score={new_score:.6f}")
        else:
            # revert the move
            a[donor] += eps
            a[recipient] -= eps

    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation refinement – tries random re‑orderings of the same multiset
# ----------------------------------------------------------------------
def _permutation_refinement(seq: np.ndarray,
                            deadline: float,
                            start_time: float,
                            current_best: float) -> tuple[np.ndarray, float]:
    """
    While time permits, generate random permutations of ``seq``.
    Because the objective depends only on the ordering, a better permutation
    can be discovered.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = current_best

    while time.time() < deadline:
        perm = np.random.permutation(n)
        cand = seq[perm]
        cand_score = _objective(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:6.2f}s] perm improve: score={cand_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine warm‑starts from the historic best construction (``height_sequence_1``)
    if available, then performs a long simulated‑annealing phase followed by
    a short permutation‑search phase.  The best sequence found is returned
    as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start (previous best construction, if any)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Target length – use the length of the warm‑start or a sensible default
    # ------------------------------------------------------------------
    n_target = starter.shape[0] if starter is not None else 9600

    # ------------------------------------------------------------------
    # Initialise the sequence
    # ------------------------------------------------------------------
    if starter is not None:
        a0 = starter.copy()
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase A – simulated annealing (≈80 % of the total budget)
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    anneal_deadline = start_time + 0.80 * total_time
    best_seq, best_score = _simulated_annealing(a0, anneal_deadline, start_time)

    # ------------------------------------------------------------------
    # Phase B – permutation refinement (use the remaining time, if any)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _permutation_refinement(
            best_seq, deadline, start_time, best_score
        )

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may appear because of floating‑point ops)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain list for the evaluation harness
    return [float(x) for x in best_seq]
