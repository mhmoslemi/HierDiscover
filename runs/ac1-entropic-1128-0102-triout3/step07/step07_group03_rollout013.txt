        if init.ndim != 1 or init.size == 0:
            raise ValueError
        a = init.copy()
        a /= a.sum()
    except Exception:

        n = 9600
        a = np.random.rand(n)
        a /= a.sum()
    
    n = a.shape[0]
    # Precompute convolution via FFT
    def conv_fft(v):
        n = v.shape[0]
        fft_len = 1 << ((2 * n - 1).bit_length())
        Af = np.fft.rfft(v, n=fft_len)
        return np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    conv = conv_fft(a)
    
    def score_from_conv(c):
        return float(2.0 * n * c.max())
    
    best_a = a.copy()
    best_score = score_from_conv(conv)
    print(f"[0.00s] initial score: {best_score:.6f}")
    
    # Helper to apply move efficiently
    def apply_move(a_vec, conv_vec, donor, recipient, eps):
        di = -eps
        dr = eps
        # linear part (use current a before updating)
        conv_vec[donor : donor + n] += 2.0 * di * a_vec
        conv_vec[recipient : recipient + n] += 2.0 * dr * a_vec
        # quadratic corrections
        conv_vec[2 * donor] += di * di
        conv_vec[2 * recipient] += dr * dr
        conv_vec[donor + recipient] += 2.0 * di * dr
        # update a
        a_vec[donor] += di
        a_vec[recipient] += dr
    
    # Simulated annealing loop
    iteration = 0
    start_temp = 5e-2
    end_temp = 1e-6
    while time.time() < deadline:
        iteration += 1
        # compute current max and its index
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        idx_max = int(np.argmax(conv))
        # contributions of each index to the current max shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = contrib.sum()
        # donor: sample proportional to contribution, fallback uniform
        if total_contrib > 0:
            donor = np.random.choice(n, p=contrib / total_contrib)
        else:
            donor = np.random.randint(n)

    print(f"final reported score: {final_score:.6f}")

"""
Local‑search optimizer for the “self‑convolution peak” problem.

Algorithm
---------
1. Initialise a probability vector (sum = 1).  If a previous best
   (`height_sequence_1`) is available, use it, otherwise start from a
   random uniform vector of length 9600.
2. Compute its full linear self‑convolution (FFT) – this gives the exact
   objective value 2·n·max(conv).
3. Repeatedly perform a tiny “mass‑transfer” move:
   * pick a *donor* index proportionally to its contribution to the
     current maximal convolution entry,
   * pick a *recipient* from indices that contribute little,
   * move up to 30 % of the donor’s mass to the recipient,
   * update the convolution incrementally (O(n) work) and accept the move
     if it improves the objective or with a Metropolis probability
     (annealed temperature).
4. Keep the best vector seen.  Every few thousand iterations the vector
   is re‑normalised and the convolution is recomputed from scratch to
   drain accumulated rounding errors.
5. Return the best vector (as a plain list).  The routine respects the
   1000 s budget and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution (full linear self‑convolution via FFT)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Incremental update of convolution after moving ε mass from i→j
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray,
                donor: int, recipient: int, eps: float) -> None:
    """
    Apply a mass transfer of size `eps` from `donor` to `recipient`.
    The vector `a` and its full self‑convolution `conv` are updated in‑place.
    """
    n = a.shape[0]
    di = -eps          # change to donor entry
    dr = eps           # change to recipient entry

    # linear part (uses the *current* a, i.e. before the update)
    conv[donor: donor + n] += 2.0 * di * a
    conv[recipient: recipient + n] += 2.0 * dr * a

    # quadratic corrections
    conv[2 * donor] += di * di
    conv[2 * recipient] += dr * dr
    conv[donor + recipient] += 2.0 * di * dr

    # finally update the probability vector
    a[donor] += di
    a[recipient] += dr


# ----------------------------------------------------------------------
# Helper: objective from a convolution array
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) as a float."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Uses a fast simulated‑annealing style local search that updates the
    convolution incrementally, allowing millions of cheap moves within the
    1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Initialise the probability vector (warm‑start if possible)
    # --------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        a = np.maximum(init, 0.0)            # protect against negatives
        s = a.sum()
        if s > 0:
            a /= s
        else:
            raise ValueError
    except Exception:
        # fall back to a fresh random vector of length 9600
        n = 9600
        a = np.random.rand(n)
        a /= a.sum()
    else:
        n = a.shape[0]

    # --------------------------------------------------------------
    # Compute the exact convolution and initialise bookkeeping
    # --------------------------------------------------------------
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = _score_from_conv(conv, n)
    print(f"[0.00s] initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the simulated‑annealing walk
    # --------------------------------------------------------------
    start_temp = 5e-2          # relatively high to allow uphill steps early
    end_temp = 1e-6            # cool down to near‑greedy behaviour
    renorm_interval = 5000    # recompute everything every few thousand moves

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- current maximal shift and contributions -----
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        idx_max = int(np.argmax(conv))

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = contrib.sum()

        # ----- pick donor (high contribution) -----
        if total_contrib > 0:
            donor = np.random.choice(n, p=contrib / total_contrib)
        else:
            donor = np.random.randint(n)

        # ----- pick recipient (low contribution) -----
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_idx)

        if donor == recipient or a[donor] <= 1e-15:
            continue

        # ----- propose a mass transfer -----
        eps = a[donor] * np.random.rand() * 0.3       # up to 30 % of donor

        # keep a cheap backup for a possible rejection
        a_backup = a.copy()
        conv_backup = conv.copy()

        _apply_move(a, conv, donor, recipient, eps)

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # ----- annealed Metropolis acceptance -----
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta <= 0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best (iter {iteration}): {best_score:.6f}")
        else:
            # reject: restore the previous state
            a[:] = a_backup
            conv[:] = conv_backup

        # ----- occasional re‑normalisation & full recompute -----
        if iteration % renorm_interval == 0:
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)
            conv = _conv_self(a)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] renorm improve: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (tiny drift may have accumulated)
    # --------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    # --------------------------------------------------------------
    # Report the final score using the provided evaluator (optional)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"final reported score: {final_score:.6f}")
    except Exception:
        # if the evaluator is unavailable for some reason, just print the internal value
        print(f"final internal score: {_score_from_conv(_conv_self(best_a), best_a.shape[0]):.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_a]
