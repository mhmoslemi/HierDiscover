def _generate_exponential(n, r):
   a = np.power(r, np.arange(n))
   a = a / np.sum(a)
   return a

def _generate_gaussian(n, sigma):
   x = np.arange(n) - (n-1)/2
   a = np.exp(-(x**2)/(2*sigma**2))
   a = a / np.sum(a)
   return a

def _generate_powerlaw(n, p):
   a = 1.0 / np.power(np.arange(1, n+1), p)
   a = a / np.sum(a)
   return a

def _generate_triangular_power(n, alpha):
   half = n // 2
   i = np.arange(n)
   left = np.power(i, alpha)
   right = np.power(n-1-i, alpha)
   a = np.where(i <= half, left, right)
   a = a / np.sum(a)
   return a

"""
Propose a low‑peak self‑convolution step function.

Algorithm
---------
1. **Fast parametric scan** – generate a variety of simple families
   (uniform, triangular‑power, exponential, Gaussian, power‑law) for several
   lengths (2000…12000).  The best candidate (smallest 2·n·max(conv)) is kept.
2. **Upsample** the best candidate to a target length (the longest length
   tried in the scan) using linear interpolation and re‑normalise.
3. **Refinement** – three stages on the up‑sampled vector:
   * a soft‑max (log‑sum‑exp) gradient descent with Nesterov momentum,
   * a true max‑convolution sub‑gradient descent,
   * a targeted Metropolis “peak‑shaving” local search.
   The three stages are run sequentially on the remaining time budget.
4. Return the final vector as a plain list.  All entries are non‑negative and
   the sum is forced to be >0.01, so `evaluate_sequence` will accept it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple convex helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv) (sum is assumed >0)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Families of parametric candidates
# ----------------------------------------------------------------------
def _gen_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _gen_triangular_power(n: int, alpha: float) -> np.ndarray:
    """Triangular shape with power α (α=0 gives uniform)."""
    i = np.arange(n, dtype=float)
    half = (n - 1) / 2.0
    left = np.power(i, alpha)
    right = np.power((n - 1) - i, alpha)
    a = np.where(i <= half, left, right)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _gen_exponential(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=float))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _gen_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _gen_powerlaw(n: int, p: float) -> np.ndarray:
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Gradient‑descent helpers (soft‑max approximation)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Grad of Σ_k softmax_k(conv/T)·conv_k w.r.t. a."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))     # stabilise
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov GD on smooth max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    T_start, T_end = 2.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.5,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis walk that transfers mass from high‑contrib. to low‑contrib. indices."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp, end_temp = 5e-2, 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient from low‑contribution tail
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta < 0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

# Fast parametric scan

def _scan_initial_candidates(deadline: float) -> tuple[np.ndarray, float]:
    """Explore a handful of parametric families; return best sequence & score."""
    best_seq = None
    best_score = float("inf")

    lengths = [2000, 4000, 8000, 9600, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Uniform (baseline)
        seq = _gen_uniform(n)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()

        # Triangular‑power
        for alpha in np.linspace(0.1, 3.0, 13):
            if time.time() >= deadline:
                break
            seq = _gen_triangular_power(n, alpha)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

        # Exponential
        for r in np.linspace(0.5, 0.99, 20):
            if time.time() >= deadline:
                break
            seq = _gen_exponential(n, r)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

        # Gaussian
        for sigma_frac in np.linspace(0.05, 0.5, 12):
            if time.time() >= deadline:
                break
            sigma = sigma_frac * n
            seq = _gen_gaussian(n, sigma)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

        # Power‑law
        for p in np.linspace(0.5, 3.0, 12):
            if time.time() >= deadline:
                break
            seq = _gen_powerlaw(n, p)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    The routine runs a quick parametric scan, then refines the best candidate
    with gradient descent and a targeted local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Fast parametric scan (≈20 % of the budget)
    # --------------------------------------------------------------
    scan_deadline = start_time + budget_s * 0.20
    best_seq, best_score = _scan_initial_candidates(scan_deadline)
    if best_seq is None:
        # fallback – uniform vector of modest size
        best_seq = _gen_uniform(200)
        best_score = _score(best_seq)

    print(f"[{time.time() - start_time:6.2f}s] after scan: len={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Upsample to the largest length tried during scanning
    # --------------------------------------------------------------
    target_len = max(2000, best_seq.size)  # at least 2000 for decent resolution
    if best_seq.size != target_len:
        best_seq = _upsample(best_seq, target_len)

    # Ensure simplex normalisation (sum=1)
    if best_seq.sum() > 0:
        best_seq = best_seq / best_seq.sum()
    else:
        best_seq = _gen_uniform(target_len)

    # --------------------------------------------------------------
    # 3️⃣  Refine with three optimisation phases using the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        # out of time – just return what we have
        return [float(x) for x in best_seq]

    # allocate time slices
    softmax_frac = 0.4
    subgrad_frac = 0.35
    local_frac   = max(0.0, 1.0 - softmax_frac - subgrad_frac)

    # ----- soft‑max GD -----
    soft_deadline = time.time() + remaining * softmax_frac
    a_soft, sc_soft = _run_softmax_gd(
        best_seq, target_len,
        soft_deadline, start_time,
        step0=0.6, beta=0.9, max_iter=10 ** 7
    )
    if sc_soft < best_score:
        best_score, best_seq = sc_soft, a_soft.copy()
        print(f"[{time.time() - start_time:6.2f}s] soft‑max final improve: score={sc_soft:.6f}")

    # ----- sub‑gradient GD -----
    remaining = deadline - time.time()
    if remaining > 0:
        sub_deadline = time.time() + remaining * subgrad_frac
        a_sub, sc_sub = _run_subgradient_gd(
            best_seq, target_len,
            sub_deadline, start_time,
            step0=0.5, beta=0.5, max_iter=10 ** 7
        )
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad final improve: score={sc_sub:.6f}")

    # ----- targeted local search -----
    remaining = deadline - time.time()
    if remaining > 0:
        a_loc, sc_loc = _run_targeted_local_search(
            best_seq, target_len,
            deadline, start_time
        )
        if sc_loc < best_score:
            best_score, best_seq = sc_loc, a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] local final improve: score={sc_loc:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final sanity check and return
    # --------------------------------------------------------------
    # Guard against sum being too small (evaluate_sequence would reject)
    if best_seq.sum() < 0.01:
        best_seq = best_seq + 1e-8
    best_seq = best_seq / best_seq.sum()   # normalise again

    # Convert to plain Python list
    final_list = [float(x) for x in best_seq]
    print(f"[{time.time() - start_time:6.2f}s] returning sequence of length {len(final_list)} with score {best_score:.6f}")
    return final_list
