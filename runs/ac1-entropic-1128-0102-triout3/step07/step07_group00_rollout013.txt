"""
Multi‑restart local optimisation for the step‑function evaluation problem.

The optimiser repeatedly:
  * draws a random probability vector (Dirichlet),
  * refines it using a short Adam phase on a log‑sum‑exp surrogate,
  * runs projected sub‑gradient descent on the true max‑convolution objective,
  * finishes with a relatively long simulated‑annealing polish.

Each run keeps the best iterate it sees; the globally best across all runs
is returned.  All phases respect the global time budget (≈1000 s) and
fall back to the uniform distribution if no valid candidate is found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)).  a is assumed to be non‑negative and sum=1."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth max surrogate
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                alphas=(0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0),
                iters_per_alpha=3000,
                lr0=0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best iterate (lowest objective) found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # pre‑compute safe FFT length for the whole phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            # current convex surrogate (not used for update)
            conv = _conv_self(a)
            cur_score = _objective(a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: iter={it:,}, score={cur_score:.6f}")

            # soft‑max weighting of convolution entries
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smoothed max
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)  # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       init_best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a simple 1/√t step‑size with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # Find all indices attaining the maximum (rarely more than one)
        max_idx = np.flatnonzero(conv == cur_max)
        # build sub‑gradient as average over the active set
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(max_idx))

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal_phase(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-6,
                  steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Light‑weight annealing that moves a random fraction of mass between
    two coordinates.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = _objective(a, n)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportional to current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 80 % of the source mass)
        eps = a[src] * rng.random() * 0.80
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop, n)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Local search wrapper
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Run the three optimisation phases sequentially on a single start.
    Returns the best sequence found and its objective value.
    """
    # Phase 1 – Adam (use at most 10 % of the allocated time)
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(2.0, 0.10 * remaining)
    a1, score1 = _adam_phase(a0, n, phase1_deadline, start_time)

    # Phase 2 – sub‑gradient (major chunk)
    remaining = deadline - time.time()
    phase2_deadline = time.time() + max(5.0, 0.70 * remaining)
    a2, score2 = _subgradient_phase(a1, n, phase2_deadline, start_time, score1)

    # Phase 3 – annealing (use whatever time is left)
    a3, score3 = _anneal_phase(a2, n, deadline, start_time, score2, a2)

    # Pick the best among the three phases
    best_seq = a0
    best_score = _objective(a0, n)
    for seq, sc in [(a1, score1), (a2, score2), (a3, score3)]:
        if sc < best_score:
            best_score = sc
            best_seq = seq

    return best_seq, best_score

"""
Multi‑restart local optimisation for the step‑function evaluation problem.

The optimiser repeatedly:
  * draws a random probability vector (Dirichlet with α<1 → sparse start),
  * refines it using a short Adam phase on a log‑sum‑exp surrogate of the
    max‑convolution,
  * runs projected sub‑gradient descent on the exact objective,
  * finishes with a relatively long simulated‑annealing polish.

Each run keeps the best iterate it sees; the globally best across all runs
is returned.  All phases respect the global time budget (≈1000 s) and fall
back to a uniform distribution if no valid candidate is found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)).  a is assumed to be non‑negative and sum≈1."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                alphas=(0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0),
                iters_per_alpha=3000,
                lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best iterate (lowest objective) found before ``deadline``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length that accommodates all convolutions in this phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a)
            cur_score = _objective(a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: iter={it:,}, score={cur_score:.6f}")

            # soft‑max weighting of convolution entries
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smoothed max (via correlation)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       init_best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum and a 1/√t step‑size schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # active set of max indices (normally a single index)
        max_idx = np.flatnonzero(conv == cur_max)
        grad = np.zeros_like(a)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(max_idx))

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal_phase(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-6,
                  steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Light annealing that moves a random fraction of mass between two indices.
    Temperature decays exponentially.
    """
    a = a0.copy()
    score = _objective(a, n)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 80 % of the source mass
        eps = a[src] * rng.random() * 0.80
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop, n)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local search wrapper (three phases in order)
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Execute Adam → sub‑gradient → annealing on a single start.
    Returns the best sequence discovered and its objective value.
    """
    # Phase 1 – Adam (≈10 % of the allotted time)
    remaining = deadline - time.time()
    phase1_deadline = time.time() + max(2.0, 0.10 * remaining)
    a1, s1 = _adam_phase(a0, n, phase1_deadline, start_time)

    # Phase 2 – sub‑gradient (major chunk)
    remaining = deadline - time.time()
    phase2_deadline = time.time() + max(5.0, 0.70 * remaining)
    a2, s2 = _subgradient_phase(a1, n, phase2_deadline, start_time, s1)

    # Phase 3 – annealing (use whatever is left)
    a3, s3 = _anneal_phase(a2, n, deadline, start_time, s2, a2)

    # pick the best among the three phases
    best_seq = a0
    best_score = _objective(a0, n)
    for seq, sc in [(a1, s1), (a2, s2), (a3, s3)]:
        if sc < best_score:
            best_score = sc
            best_seq = seq
    return best_seq, best_score
