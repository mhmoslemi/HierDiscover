"""
Hybrid refinement for the step‑function minimisation problem.

Algorithm
---------
* Initialise a candidate distribution by one of several simple shapes
  (uniform Dirichlet, power‑law, triangular) or by the previously found
  construction `height_sequence_1` when it exists.
* Run a smooth “soft‑max’’ gradient descent (Phase A) on the full size
  (≈ 9600 entries).  The temperature is driven quickly to a very low value
  so that the gradient focuses on the true maximal convolution entry.
* Follow with a long projected sub‑gradient descent (Phase B).  The step
  size is larger than in the reference implementation and the iteration
  budget is increased.
* Perform an aggressive peak‑shaving local search (Phase C).  The donor
  index is chosen as the one that contributes most to the current maximal
  convolution entry and the recipient is taken from the lowest‑contribution
  region.  The update magnitude is larger and the annealing schedule is
  tighter.
* The three phases are repeated (sub‑grad ➜ local search) until the time
  budget expires, keeping the best sequence seen (evaluated with the
  supplied ``evaluate_sequence``).
* The final sequence is normalised to the probability simplex and returned
  as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (full length = 2·len(a)‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Convolution of rev(a) with a weight vector w (used for soft‑max gradient)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the temperature‑softened max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    maxc = conv.max()
    # avoid overflow – shift by max before scaling
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Phase A – projected gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    T_start: float = 2.0e-2,
                    T_end: float = 5.0e-5,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n},"
                  f" iter={it}, score={cur_score:.6f}")

        # exponential temperature decay (faster than the reference)
        T = T_start * (T_end / T_start) ** (it / 15000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.9,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n},"
                  f" iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient of the active convolution entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – aggressive peak‑shaving local search
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               start_temp: float = 4e-2,
                               end_temp: float = 5e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)

        # contribution of each index to the maximal entry (ordered pairs)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor = index with the largest contribution (most harmful)
        donor = int(np.argmax(contrib))

        # recipient = an index that currently contributes little (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            recipient = np.argmin(a)          # fallback to the smallest weight
        else:
            recipient = int(np.random.choice(low_idx))

        if donor == recipient or a[donor] <= 1e-15:
            continue

        # move a modest fraction of the donor mass (up to 30 % of its weight)
        eps = a[donor] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature schedule (slow annealing)
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n},"
                      f" iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple initialisers (random Dirichlet, power‑law, triangular, warm‑start)
# ----------------------------------------------------------------------
def _initial_sequence(pattern: str, n: int, rng: np.random.Generator, warm: np.ndarray | None) -> np.ndarray:
    if pattern == "warm" and warm is not None:
        return _upsample(warm, n)

    if pattern == "random":
        v = rng.random(n)
    elif pattern == "tent":
        v = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    elif pattern.startswith("power"):
        # pattern = "power_-2", "power_-1", "power_0", "power_1", "power_2"
        exp = float(pattern.split("_")[1])
        inds = np.arange(1, n + 1, dtype=float)
        v = inds ** exp
    else:
        # fallback to uniform
        v = np.ones(n, dtype=float)

    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v = np.full(n, 1.0 / n)

    return v


# ----------------------------------------------------------------------
# Upsample helper (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs a soft‑max gradient phase, a long sub‑gradient phase and
    an aggressive peak‑shaving local search, repeatedly refining the best
    candidate within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start (previous best construction) if available
    # ------------------------------------------------------------------
    warm = None
    try:
        warm_raw = np.asarray(height_sequence_1, dtype=float)
        if warm_raw.ndim == 1 and warm_raw.size > 0:
            warm = warm_raw / warm_raw.sum()
    except Exception:
        warm = None

    # target length – the length that gave the best historic result
    target_n = 9600

    # keep the best overall candidate
    best_seq = None
    best_obj = float("inf")

    # a small pool of initial patterns
    patterns = ["warm", "random", "tent", "power_-2", "power_-1", "power_0",
                "power_1", "power_2"]

    # ------------------------------------------------------------------
    # Main optimisation loop – keep producing refined candidates until time expires
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for finalisation
        # pick a pattern (warm‑start used only once at the very beginning)
        if best_seq is None and warm is not None:
            pattern = "warm"
        else:
            pattern = rng.choice(patterns[1:])   # avoid "warm" after first use

        init_seq = _initial_sequence(pattern, target_n, rng, warm)

        # split the remaining budget into three phases
        remaining = deadline - time.time()
        if remaining < 30.0:
            break

        # allocate roughly 30 % softmax, 45 % sub‑gradient, 25 % local search
        soft_time = remaining * 0.30
        sub_time = remaining * 0.45
        local_time = remaining - (soft_time + sub_time)

        # Phase A – soft‑max gradient descent
        soft_deadline = time.time() + soft_time
        a_soft, _ = _run_softmax_gd(init_seq, target_n,
                                    soft_deadline, start_time,
                                    step0=0.8, beta=0.9,
                                    T_start=2.0e-2, T_end=5.0e-5)

        # Phase B – sub‑gradient descent (longer and more aggressive)
        sub_deadline = time.time() + sub_time
        a_sub, _ = _run_subgrad_gd(a_soft, target_n,
                                   sub_deadline, start_time,
                                   step0=0.9, beta=0.9)

        # Phase C – local peak‑shaving refinement
        local_deadline = time.time() + local_time
        a_loc, _ = _run_targeted_local_search(a_sub, target_n,
                                               local_deadline, start_time)

        # --------------------------------------------------------------
        # Evaluate the candidate with the official scorer
        # --------------------------------------------------------------
        try:
            obj = evaluate_sequence(a_loc.tolist())
        except Exception:
            obj = float("inf")

        if obj < best_obj:
            best_obj = obj
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST "
                  f"obj={best_obj:.6f} (pattern={pattern}) ***")

        # optional quick escape if we already reach the target bound
        if best_obj <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but we keep a safe default
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
