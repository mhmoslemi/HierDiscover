    n = len(a_current)
    conv = _conv_self(a_current)

    # Build matrix A with shape (2n-1, n)

    # Use broadcasting with outer difference?
    idx_i = np.arange(2*n-1)[:, None]  # shape (2n-1,1)
    idx_k = np.arange(n) # shape (n,)
    diff = idx_i - idx_k  # shape (2n-1, n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = 2 * a_current[diff[mask]]

"""
Propose a step‑function (a non‑negative vector summing to 1) that minimizes
the objective used by ``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the previously found best sequence
   ``height_sequence_1`` (if available) or from a random simplex point.
   Use a modest length (≈2400) for the heavy optimisation phases.

2. **Convex‑concave linearisation (CCP)** – at the current point a,
   linearise each convolution entry
        conv_i(a) ≈ 2·∑_k a_k·a_{i‑k}^{old} − conv_i(old)
   and solve a linear programme

        minimise   M
        subject to a ≥ 0 , 1ᵀa = 1
                   2·∑_k a_{i‑k}^{old} a_k  ≤  M + conv_i(old)      (active i)

   Only the largest convolution entries (the current active set) are kept
   as constraints, making the LP small (a few hundred rows).  The LP
   solution yields a new a with a provably smaller upper bound M.  The
   process is repeated a handful of times (CCP iterations).

3. **Upsampling** – the improved short vector is interpolated to a longer
   length (≈9600) while keeping the sum equal to 1.  A larger support
   offers a finer “mass‑distribution” that can further lower the max‑convolution.

4. **Final refinement** – a long simulated‑annealing walk with tiny
   mass‑transfer moves (and occasional smoothing) is performed on the
   large vector.  The annealing schedule is geometric and stops when the
   time budget expires.

All vectors are projected onto the probability simplex after each
operation.  The code respects the 1000 s wall‑clock limit and prints the
current best bound throughout the run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – used for the linear‑programming phase.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:  # pragma: no cover
    _HAS_CVX = False

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1} (Euclidean distance)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution (full, linear convolution a*a).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear interpolation to a new length (keeps the sum = 1).
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient descent (optional, quick warm‑up).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the temperature‑smoothed soft‑max of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov‑style GD on the soft‑max surrogate of max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    T_start, T_end = 5.0e-2, 5.0e-4
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")
        # temperature schedule
        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (optional refinement).
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer + occasional smoothing).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (convex‑concave procedure).
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               max_iters: int = 6,
               active_frac: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Repeatedly linearise the max‑convolution and solve a small LP.
    ``active_frac`` determines the fraction of convolution entries kept as
    constraints (the largest ones).  Returns the best vector seen and its
    surrogate score 2·n·max(conv).
    """
    if not _HAS_CVX:
        print("cvxpy not available – skipping LP refinement.")
        return a0, 2.0 * n * _conv_self(a0).max()

    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        # Choose active constraints – top ``active_frac`` of entries
        K = max(1, int(active_frac * conv.size))
        active_idx = np.argpartition(-conv, K - 1)[:K]
        # Build the matrix for the active constraints
        A_rows = np.zeros((K, n), dtype=np.float64)
        for r, i in enumerate(active_idx):
            lo = max(0, i - n + 1)
            hi = min(i + 1, n)
            ks = np.arange(lo, hi)
            A_rows[r, ks] = 2.0 * a[i - ks]
        b_vec = conv[active_idx]

        # CVXPY model
        a_var = cp.Variable(n)
        M_var = cp.Variable()
        constraints = [
            a_var >= 0,
            cp.sum(a_var) == 1,
            A_rows @ a_var <= M_var + b_vec
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        a_var.value = a  # warm start
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except Exception as e:  # pragma: no cover
            print(f"LP solve failed on iteration {it}: {e}")
            break

        if a_var.value is None:
            print(f"LP iteration {it} returned no solution – aborting LP.")
            break

        a_new = np.maximum(a_var.value, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.full_like(a_new, 1.0 / n)

        a = a_new
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it+1}: score={cur_score:.6f}")

    return best_a, best_score

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the objective used
    by ``evaluate_sequence``.  The routine respects the given wall‑clock
    budget and combines a quick soft‑max gradient warm‑up, a convex‑concave
    LP refinement, up‑sampling to a large resolution and a final
    simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin for the harness.
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Load a starter sequence if the global ``height_sequence_1`` exists.
    # ------------------------------------------------------------------
    starter = None
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            starter = hs / hs.sum()
    except Exception:  # pragma: no cover
        starter = None

    # ------------------------------------------------------------------
    # 1. Choose a modest working size for the heavy optimisation phases.
    # ------------------------------------------------------------------
    n_small = 2400                      # enough to capture shape, cheap LP
    if starter is not None:
        a0 = _upsample(starter, n_small)
    else:
        a0 = np.random.rand(n_small)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # 2. Quick smooth‑gradient warm‑up (only a few seconds).
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 40.0)
    a_warm, _ = _run_softmax_gd(
        a0, n_small, warm_deadline, start_time,
        step0=0.6, beta=0.9, max_iter=200_000
    )

    # ------------------------------------------------------------------
    # 3. Convex‑concave linearisation (LP) – main improvement driver.
    # ------------------------------------------------------------------
    lp_deadline = min(deadline, start_time + 300.0)
    a_lp, lp_score = _lp_refine(
        a_warm, n_small, lp_deadline, start_time,
        max_iters=8, active_frac=0.22
    )

    # ------------------------------------------------------------------
    # 4. Optional exact sub‑gradient refinement at the small scale.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, start_time + 380.0)
    if time.time() < subgrad_deadline:
        a_sub, sub_score = _run_subgradient_gd(
            a_lp, n_small, subgrad_deadline, start_time,
            step0=0.12, beta=0.9, max_iter=300_000, rel_eps=0.008
        )
        if sub_score < lp_score:
            a_lp = a_sub
            lp_score = sub_score

    # ------------------------------------------------------------------
    # 5. Upsample to a large resolution for the final annealing phase.
    # ------------------------------------------------------------------
    n_large = 9600
    a_big = _upsample(a_lp, n_large)

    # ------------------------------------------------------------------
    # 6. Final annealing – allocate all remaining time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:  # need at least a few seconds for the walk
        # Roughly 1500 annealing proposals per second (empirically fast)
        steps = int(min(4_000_000, max(500_000, remaining * 1500)))
        cur_score = 2.0 * n_large * _conv_self(a_big).max()
        a_final, final_score = _run_annealing(
            a_big, n_large,
            deadline, start_time,
            cur_score, a_big,
            start_temp=0.3, end_temp=1e-7,
            steps=steps
        )
        a_big = a_final
        print(f"[{time.time() - start_time:6.2f}s] final anneal score: {final_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Normalise once more (tiny numerical drift possible).
    # ------------------------------------------------------------------
    a_big = np.maximum(a_big, 0.0)
    s = a_big.sum()
    if s > 0:
        a_big /= s
    else:
        a_big = np.full_like(a_big, 1.0 / a_big.size)

    # Return a plain Python list – the harness will call evaluate_sequence.
    return [float(x) for x in a_big]
