"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

The idea is very simple: the objective 2·n·max(a*a)/(sum(a)^2) is
scale‑invariant, so we can restrict ourselves to probability vectors
(sum = 1).  For a uniform vector the score is 2.  By tilting the
distribution – e.g. an exponential or power‑law decay – the aperiodic
autocorrelation (the linear convolution a*a) becomes more “flat” and
the maximum entry drops dramatically.  A tiny parametric sweep over a few
families (exponential, power‑law, linear) is performed first; the best
candidate is then polished with a short run of the incremental simulated
annealing that was used in the reference implementation.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found (as a plain list of
floats).  All numbers are non‑negative, so the security checks in
``evaluate_sequence`` are satisfied.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (used for fast evaluation of the objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise can produce tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast version of the evaluation used during the search.
    The vector is assumed to be normalised (sum ≈ 1).  Returns the value
    2·n·max(conv) which equals the true score because the denominator
    (sum²) is 1.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Candidate generators (parametric families)
# ----------------------------------------------------------------------
def _exp_family(beta: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-beta·i)  (i = 0 … n‑1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * i)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _power_family(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (i = 0 … n‑1)."""
    i = np.arange(1, n + 1, dtype=np.float64)  # 1‑based to avoid division by zero
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _linear_family(alpha: float, n: int) -> np.ndarray:
    """
    Linear decay from 1.0 at i=0 to ``alpha`` at i=n‑1.
    ``alpha`` ∈ (0, 1]; alpha=1 gives the uniform vector.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - (1.0 - alpha) * i / (n - 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Incremental simulated annealing (the same routine used in the reference)
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               deadline: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Fast mass‑transfer annealing.  The convolution is updated incrementally,
    allowing millions of cheap moves.
    """
    n = a0.shape[0]
    a = a0.copy()
    conv = _conv_self_fft(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # source proportional to its mass – focuses moves where they matter
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (linear‑time slice updates) -----
        # the update formulas are exact for a linear convolution
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # reject – revert the move
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += eps * eps

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The algorithm works in three stages:

    1. **Parametric sweep** over a few simple families (exponential,
       power‑law, linear).  The score is evaluated with a fast FFT‑based
       convolution; the best candidate is kept.
    2. **Optional refinement** – a short simulated‑annealing run starting
       from the best candidate found in step 1.  This cleans up any
       remaining irregularities.
    3. Normalise the final vector (to guard against tiny numerical drift)
       and return it as a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the outer harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (use the length of the reference construction if available)
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the surrounding harness
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.shape[0]
        else:
            raise Exception
    except Exception:
        n = 4800

    # ------------------------------------------------------------------
    # Initial uniform baseline
    # ------------------------------------------------------------------
    a_uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = a_uniform.copy()
    best_score = _fast_score(best_seq)
    print(f"[{0.0:6.2f}s] uniform baseline (fast) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families
    # ------------------------------------------------------------------
    # exponential decay (log‑spaced beta)
    for beta in np.logspace(-6, -1, num=80):
        a = _exp_family(beta, n)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp   beta={beta:.2e} → score={sc:.6f}")

    # power law (linear spaced exponent)
    for p in np.linspace(0.0, 5.0, num=80):
        a = _power_family(p, n)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] power p={p:.3f}   → score={sc:.6f}")

    # linear decay (alpha = last/first)
    for alpha in np.linspace(0.2, 1.0, num=30):
        a = _linear_family(alpha, n)
        sc = _fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] linear α={alpha:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # give the annealer about 30 % of the remaining budget
        anneal_deadline = time.time() + 0.3 * (deadline - time.time())
        # the fast annealer works directly on the probability simplex
        best_seq, best_score = _run_incremental_annealing(
            best_seq,
            anneal_deadline,
            best_score,
            best_seq,
            total_steps=2_000_000,
        )

    # ------------------------------------------------------------------
    # 3️⃣  Final safety normalisation and official evaluation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
