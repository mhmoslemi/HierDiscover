def gradient_L2(a):
    C = _conv_self(a)
    a_rev = a[::-1]
    # compute convolution via FFT:
    fft_len = 1 << ((len(C) + len(a_rev) - 1).bit_length())
    A = np.fft.rfft(C, n=fft_len)
    B = np.fft.rfft(a_rev, n=fft_len)
    conv_rev = np.fft.irfft(A*B, n=fft_len)[:len(C)+len(a_rev)-1]
    g = 4 * conv_rev[n-1:n-1+len(a)]
    return C, g

def _run_adam_l2(a0, n, deadline, start,
                 lr=0.1, max_iter=10**7):

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        C = _conv_self(a)
        max_c = C.max()
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time()-start:.2f}s] Adam L2 improve: n={n} iter={t} score={score:.6f}")

        # but we do L2 gradient.
        # We'll compute g = 4 * conv(C, a_rev)[mid:mid+n]
        a_rev = a[::-1]
        # conv via FFT
        conv_len = len(C) + len(a_rev) - 1
        fft_len = 1 << (conv_len - 1).bit_length()
        A = np.fft.rfft(C, n=fft_len)
        B = np.fft.rfft(a_rev, n=fft_len)

def _run_targeted_local_search(a0, n, deadline, start, cur_best_score):
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    start_temp=0.05; end_temp=1e-6
    while time.time() < deadline:
        it += 1
        # compute conv and max index
        C = _conv_self(a)
        k = int(np.argmax(C))
        cur_score = 2.0 * n * C.max()
        # compute contributions at this k
        # for i: if 0 <= j = k - i < n -> contribution = a[i] * a[j]
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = contrib.sum()
        # donor
        if total_contrib > 0:
            prob = contrib / total_contrib
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)
        # recipient: low contribution indices (bottom 20%)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size > 0:
            recipient = np.random.choice(low_idx)
        else:
            recipient = np.random.randint(n)
        if donor == recipient or a[donor] < 1e-12:
            continue
        # move eps mass
        eps = a[donor] * np.random.rand() * 0.5  # up to 50%
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps
        # ensure nonneg and project onto simplex maybe not needed
        # but we maintain sum by moving mass.
        # compute new max
        C_new = _conv_self(a_prop)
        new_score = 2.0 * n * C_new.max()
        delta = new_score - cur_score
        # cooling schedule
        elapsed_frac = (time.time() - start) / (deadline - start + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed_frac
        if delta < 0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start:.2f}s] local improve: n={n} iter={it} score={new_score:.6f}")

    return best_a, best_score

"""
Propose a step‑function that minimizes `evaluate_sequence`.

The search proceeds in two phases:

1. **L2‑flattening (Adam)** – minimise the squared ℓ₂‑norm of the
   self‑convolution (∑_k (a∗a)_k²).  This pushes the convolution
   towards a flat shape, which in turn reduces its maximum entry.
   Adam with a simplex projection provides fast, stable updates.

2. **Targeted peak‑shaving** – a Metropolis walk that moves mass from
   indices that contribute heavily to the current maximal convolution
   entry to indices that contribute little.  The temperature is
   annealed, allowing occasional uphill moves early on.

Both phases respect the 1000 s budget, continuously reporting
intermediate scores.  The final vector is projected onto the probability
simplex (non‑negative, sum = 1) and returned as a plain Python list.
"""
# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Upsample (linear interpolation) preserving total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `seq` to length `new_n` by linear interpolation, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Full linear self‑convolution via FFT (real signals)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Gradient of the L2 loss  S(a)=∑_k (a*a)_k²
# ----------------------------------------------------------------------
def _grad_L2(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the self‑convolution C = a*a and the gradient of S(a)=||C||².
    Returns (C, grad) where
        grad_i = 4 * Σ_m C_{i+m} * a_m  (i = 0 … n‑1)
    """
    n = a.shape[0]
    # C = a * a
    C = _conv_self(a)

    # compute convolution of C with the reversed a (a_rev)
    a_rev = a[::-1]
    conv_len = len(C) + len(a_rev) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(C, n=fft_len)
    B = np.fft.rfft(a_rev, n=fft_len)
    conv_rev = np.fft.irfft(A * B, n=fft_len)[:conv_len]

    # gradient (slice the central part)
    grad = 4.0 * conv_rev[n - 1 : n - 1 + n]
    return C, grad

# ----------------------------------------------------------------------
# Adam optimiser on the L2 loss, with simplex projection
# ----------------------------------------------------------------------
def _run_adam_l2(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 lr: float = 0.3,
                 max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Adam on the L2 flattening objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    while time.time() < deadline and t < max_iter:
        t += 1
        C = _conv_self(a)                     # full convolution
        cur_score = 2.0 * n * C.max()        # exact evaluation (sum(a)=1 after projection)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam L2 improve: "
                  f"n={n} iter={t} score={cur_score:.6f}")

        # gradient of Σ C_k² (same as gradient of Σ (C_k-avg)²)
        _, grad = _grad_L2(a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to the simplex
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Targeted local search (mass‑shaving) with annealed Metropolis rule
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float) -> tuple[np.ndarray, float]:
    """Move mass away from indices that contribute most to the current peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_best_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        C = _conv_self(a)
        idx_max = int(np.argmax(C))
        cur_score = 2.0 * n * C.max()

        # contribution of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        # recipient: preferably a low‑contribution index (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size > 0:
            recipient = np.random.choice(low_idx)
        else:
            recipient = np.random.randint(n)

        if donor == recipient or a[donor] < 1e-15:
            continue

        # transfer a random fraction (≤30 %) of the donor mass
        eps = a[donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps

        # evaluate the candidate
        C_new = _conv_self(a_prop)
        new_score = 2.0 * n * C_new.max()
        delta = new_score - cur_score

        # annealed temperature
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n} iter={it} score={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine consists of a fast Adam‑based L2‑flattening phase followed by
    a Metropolis‑style peak‑shaving local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length (the previous best used 9600)
    # ------------------------------------------------------------------
    n = 9600

    # ------------------------------------------------------------------
    # Warm‑start from the previous construction if available
    # ------------------------------------------------------------------
    a0 = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            a0 = _upsample(tmp / tmp.sum(), n)
    except Exception:
        a0 = None

    if a0 is None:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – Adam on L2 loss (≈ 65 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.65
    best_seq, best_score = _run_adam_l2(a0, n, phase1_deadline, start_time)

    # ------------------------------------------------------------------
    # Phase 2 – targeted local search (use the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_targeted_local_search(
            best_seq, n, deadline, start_time, best_score
        )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (project onto the simplex to clean up numerical drift)
    best_seq = np.asarray(best_seq, dtype=float)
    best_seq = _project_simplex(best_seq)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
