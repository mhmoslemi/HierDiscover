"""
Propose a step‑function that (empirically) beats the previous 1.512563 bound.
The algorithm follows the “alternating LP‑update” idea from the paper
outline:

1. Start from a feasible simplex vector a (sum = 1, a_i≥0).  Use the
   previously‑found construction if it exists, otherwise a random start.
2. Repeatedly:
   * Compute the current maximum convolution value M = max(a*a).
   * Build a tiny “active” set of convolution indices (the K largest)
     and solve the linear program

        max   Σ_j b_j
        s.t.  (a * b)_k ≤ M   for every k in the active set,
              b_j ≥ 0.

     This LP finds a new distribution b that is feasible for the
     same bound M while having maximal total mass.
   * Mix the old and new vectors:
        a ← (1‑t)·a + t·b̂      (b̂ = b / Σ b)
     The mixture stays on the simplex and usually reduces the true
     objective 2·n·max(a*a).
   * If no improvement for several iterations, perturb a randomly
     to escape a possible local trap.

3. When the time budget is almost exhausted, run a fast “peak‑shaving”
   simulated‑annealing local search (the routine from the previous
   submission) to polish the result.

All heavy work uses NumPy/FFT (O(n log n)) and a tiny LP with at most
~30 constraints, so many iterations fit comfortably into the 1000 s
budget.  The final candidate is returned as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: compute the evaluation score used by the harness
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) (equivalent to evaluate_sequence for sum = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Helper: build the (dense) constraint matrix for a selected set of
# convolution indices.
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, idx_list) -> np.ndarray:
    """
    For each k in idx_list construct a row representing the linear form
    (f * b)_k = Σ_j f[k‑j]·b_j (coefficients that are 0 outside the
    overlapping range).  Returns a matrix of shape (len(idx_list), n).
    """
    n = f.shape[0]
    rows = np.zeros((len(idx_list), n), dtype=float)
    for r, k in enumerate(idx_list):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        rows[r, js] = f[k - js]
    return rows


# ----------------------------------------------------------------------
# Helper: solve the LP “max Σ b_j s.t. (f * b)_k ≤ M for k in C”.
# ----------------------------------------------------------------------
def _solve_lp_for_f(f: np.ndarray, M: float, C_initial, max_outer_iter: int = 5):
    """
    Solve the LP using a cutting‑plane style active‑set.  Starts from the
    indices in C_initial, checks the full convolution for violations,
    and adds the most violating indices (up to a small limit).
    Returns a feasible vector b (or None on failure).
    """
    n = f.shape[0]
    # start with the supplied indices, guarantee at least one constraint
    C = set(int(k) for k in C_initial)
    if not C:
        C.add(int(np.argmax(_conv_self(f))))   # fallback

    for outer in range(max_outer_iter):
        A = _build_constraint_matrix(f, list(C))
        b_ub = np.full(A.shape[0], M, dtype=float)

        # maximise Σ b_j  <=> minimise -Σ b_j
        c = -np.ones(n, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            return None

        b = res.x

        # verify feasibility against *all* convolutions
        conv_fb = np.convolve(f, b)                 # dense but cheap (FFT not needed)
        viol = np.where(conv_fb > M + 1e-12)[0]

        if viol.size == 0:
            return b

        # Add a few of the most violating indices and retry
        to_add = viol[np.argsort(conv_fb[viol])[-min(5, viol.size):]]
        C.update(int(v) for v in to_add)

        # safety stop if the active set became too large
        if len(C) > 250:
            break

    # final attempt (may be slightly infeasible – still returned)
    return b


# ----------------------------------------------------------------------
# Targeted local search (peak shaving) – same idea as the previous run
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing style peak‑shaving."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)

        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor proportional to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # receiver: a low‑contribution index
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine mixes a cutting‑plane LP update with a final simulated‑annealing
    polishing.  It respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 8000                                   # ~8k terms (fits easily in memory)

    # ------------------------------------------------------------------
    # Warm‑start (previous best construction, if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        # up‑sample / down‑sample to required length
        if starter.shape[0] != n:
            # linear interpolation preserving total mass
            old_x = np.linspace(0.0, 1.0, starter.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            starter = np.interp(new_x, old_x, starter)
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0:
            starter = np.full(n, 1.0 / n)
        else:
            starter = starter / starter.sum()
        a = starter.copy()
    except Exception:
        a = np.random.rand(n)
        a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score(best_a)
    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Core iterative improvement using the LP‑based update
    # ------------------------------------------------------------------
    no_improve = 0
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        conv = _conv_self(a)
        M = conv.max()

        # pick a small active set – the indices that are near the maximum
        K = 30
        idx_desc = np.argsort(-conv)[:K]

        b = _solve_lp_for_f(a, M, idx_desc)
        if b is None:
            # LP failed – fallback to a small random perturbation
            eps = 1e-3
            pert = np.random.randn(n) * eps
            a = a + pert
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            continue

        # normalise the LP solution to the simplex
        if b.sum() == 0:
            b = np.full_like(b, 1.0 / n)
        else:
            b = b / b.sum()

        # convex combination (mixing) – step size t slowly decays
        t = 0.6
        a_new = (1.0 - t) * a + t * b
        a_new = _project_simplex(a_new)

                a = best_a + jitter
                a = _project_simplex(a)
                no_improve = 0

    return [float(x) for x in best_a]
