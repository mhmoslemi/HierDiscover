def _solve_lp_for_direction(a, M):
    n = a.shape[0]
    rows = np.arange(n, dtype=int)
    # Actually we need to build matrix shape (2*n-1, n)
    total_rows = 2*n-1
    data = []
    row_idx = []
    col_idx = []

def _fw_step(a, n, t, K, deadline):
    conv = _conv_self(a)
    M = conv.max()
    # pick K largest indices
    if K >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx_unsorted = np.argpartition(-conv, K-1)[:K]
        idx = np.sort(idx_unsorted)
    # build constraints for these idx only
    # row count = len(idx)
    rows = idx
    # Build A as sparse matrix (len(idx), n)
    row_list = []
    col_list = []
    data_list = []
    for row in rows:
        # for each output shift row
        # a_i * g_{row-i} ; i must satisfy 0 <= row - i < n => i ∈ [row-(n-1), row]
        low = max(0, row - (n - 1))
        high = min(row, n - 1)
        # indices i = low..high
        i_vals = np.arange(low, high + 1)
        # corresponding column j = row - i
        j_vals = row - i_vals
        # coefficient a[i_vals] for columns j_vals.
        row_list.append(np.full_like(j_vals, row, dtype=int))
        col_list.append(j_vals)
        data_list.append(a[i_vals])
    row_arr = np.concatenate(row_list)
    col_arr = np.concatenate(col_list)
    data_arr = np.concatenate(data_list)

"""
Optimise a non‑negative step‑function (probability vector) to minimise
the evaluation function

    2·n·max(conv(a, a)) / (sum(a)**2),

where ``conv`` is the linear self‑convolution.  The optimiser combines
four complementary heuristics:

1. **Adam** on a log‑sum‑exp smooth approximation of the max‑convolution.
   This rapidly discovers a good global shape while staying on the
   probability simplex.

2. **Frank–Wolfe‑style LP step** – a tiny linear programme that,
   for the current vector ``a``, finds a feasible direction ``g`` with the
   largest ℓ₁‑norm while keeping the convolution bound unchanged.
   A convex combination ``(1‑t)a + t·g`` lowers the peak of the
   convolution.

3. **Projected sub‑gradient descent** on the true objective
   (using the gradient at the currently maximal convolution index).

4. **Simulated‑annealing polish** that repeatedly moves a small amount
   of mass between coordinates.

The routine respects a 1000 s time budget (≈1 000 000 ms) and always
returns the best sequence found so far.
"""

import time
import math
import numpy as np

# optional SciPy – used only for the tiny LP in the Frank–Wolfe step
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) self‑convolution a * a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                       # numerically unlikely, but safe
        w = np.full_like(v, 1.0 / n)
    return w


def _score_fast(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) (sum(a) is 1 after normalisation)."""
    n = a.shape[0]
    max_b = _conv_self(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Phase‑1 : Adam on a smooth max surrogate
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float,
              start_time: float,
              alphas, lr0: float = 0.5,
              iters_per_alpha: int = 4000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_fast(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    L = _next_pow2(3 * n - 2)                     # common FFT length

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # --- soft‑max weights (log‑sum‑exp) ------------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smoothed max (see original code)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Frank–Wolfe style LP direction
# ----------------------------------------------------------------------
def _run_frank_wolfe(a0: np.ndarray, n: int, deadline: float,
                     start_time: float,
                     max_steps: int = 30,
                     t: float = 0.2,
                     top_k: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny LP that finds a feasible direction g
    with maximal ℓ₁‑norm while keeping the current max‑convolution bound.
    The LP is built only for the ``top_k`` largest convolution entries,
    which yields a very small constraint matrix.
    """
    if not _SCIPY_OK:                     # fallback – nothing to do
        return a0, _score_fast(a0)

    a = a0.copy()
    best_a = a.copy()
    best_score = _score_fast(a)

    step = t
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()
        # pick the indices with the largest values (ties are fine)
        if top_k >= conv.size:
            idx = np.arange(conv.size)
        else:
            idx = np.argpartition(-conv, top_k - 1)[:top_k]
            idx = np.sort(idx)

        # build sparse matrix A (|idx| × n) where A[row, col] = a[row-col]
        rows = []
        cols = []
        datas = []
        for row in idx:
            low = max(0, row - (n - 1))
            high = min(row, n - 1)
            i_vals = np.arange(low, high + 1, dtype=int)
            j_vals = row - i_vals                     # column index for g
            rows.append(np.full_like(j_vals, row, dtype=int))
            cols.append(j_vals)
            datas.append(a[i_vals])
        row_arr = np.concatenate(rows)
        col_arr = np.concatenate(cols)
        data_arr = np.concatenate(datas)
        A = _sp.csr_matrix((data_arr, (row_arr, col_arr)),
                           shape=(len(idx), n))

        # LP: maximise sum(g)  (i.e. minimise -sum(g))
        c = -np.ones(n)
        b_ub = np.full(len(idx), M)
        bounds = [(0, None)] * n
        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=bounds, method='highs',
                               options={'presolve': True,
                                        'time_limit': max(0.0,
                                                          deadline - time.time())})
        except Exception:                 # safety net – skip this step
            continue
        if not res.success:
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            continue
        g_norm = g / sum_g

        a_candidate = (1.0 - step) * a + step * g_norm
        a_candidate = _project_simplex(a_candidate)
        cand_score = _score_fast(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] FW improve: "
                  f"step={it}, score={cand_score:.6f}")

        # always move a a little towards the direction – this keeps the
        # optimisation moving even if the score does not improve immediately.
        a = a_candidate
        # gradually shrink the mixing weight
        step = max(0.01, step * 0.92)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : exact max – projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float,
                 start_time: float,
                 best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int, deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   min_steps: int = 0) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer annealing; runs until the deadline is reached."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    steps = 0
    while time.time() < deadline:
        steps += 1
        # exponential temperature schedule
        progress = steps / (steps + 1.0)                     # <1, grows slowly
        T = start_temp * (end_temp / start_temp) ** progress

        # propose moving a small fraction of mass from src -> dst
        src = rng.choice(n, p=a)           # probability proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={steps:,}, score={score:.6f}")

    return best_seq, best_score
