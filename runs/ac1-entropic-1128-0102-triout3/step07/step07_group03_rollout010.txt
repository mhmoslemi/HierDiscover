def adam_opt(a0, n, deadline, start_time, temperature_schedule, alpha=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**9):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f'... best update')
        # temperature based on schedule: maybe T = max(T_end, T_start * (T_end/T_start)**(it/total_iters_est))
        T = temperature_schedule(it)
        # compute softmax weights
        maxc = conv.max()
        scaled = (conv - maxc) / T
        exps = np.exp(scaled - np.max(scaled))  # numeric stable
        w = exps / exps.sum()
        # gradient: conv derivative: 2 * convolution(rev(a), w)
        rev_a = a[::-1]
        # convolution rev_a and w using FFT
        fft_len = 1 << ((len(rev_a) + len(w) - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[:len(rev_a) + len(w) - 1]
        grad = 2.0 * full[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)
    return best_a, best_score

def _cosine_shape(n, beta):
    # a_i = alpha + beta * np.cos(np.pi * i/(n-1))
    i = np.arange(n)
    a = 1.0 + beta * np.cos(np.pi * i/(n-1))
    # Ensure nonnegative: if any a<0 set to 0
    a = np.maximum(a, 0.0)
    if a.sum() == 0:

    return a

"""
Propose a non‑negative step‑function (probability vector) that minimises

    2·n·max(conv) / (sum(a)**2)

where conv = a * a is the linear self‑convolution.
The algorithm consists of three phases:

1.  Multi‑stage Adam‑based gradient descent on a smooth
    log‑sum‑exp approximation of the max‑convolution.  The vector is
    gradually up‑sampled (800 → 2400 → 4800 → 9600) while the
    temperature is annealed.

2.  Adam‑based sub‑gradient descent on the true max‑convolution
    (only the active shift contributes to the gradient).

3.  A fast targeted local‑search that moves a small amount of mass
    from “hot” indices (large contribution to the current peak) to
    “cold” indices.  A simple annealed Metropolis rule permits occasional
    uphill moves.

All iterates are projected onto the probability simplex after each
update.  The routine respects the 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weight vector
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Return a smooth‑max weight vector of the same length as ``conv``."""
    maxc = conv.max()
    # scale so the maximal exponent is 0 → avoid overflow
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """∇_a Σ_k softmax_k(conv/T)·conv_k."""
    w = _softmax_weights(conv, temperature)
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max objective
# ----------------------------------------------------------------------
def _adam_softmax_opt(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      T_start: float = 2.0e-2,
                      T_end: float = 5.0e-4,
                      alpha: float = 0.12,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smoothed max–convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax Adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature based on overall progress (linear in elapsed fraction)
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        grad = _softmax_gradient(a, conv, T)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser for the true (non‑smooth) max‑convolution
# ----------------------------------------------------------------------
def _adam_subgrad_opt(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alpha: float = 0.07,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser that follows the sub‑gradient of the exact max‑conv."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx_max]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad Adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient for the active shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _targeted_local_search(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑shifting walk that tries to lower the peak of the self‑convolution."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for the Metropolis rule
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the current peak
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – pick proportionally to its contribution (focus on hot spots)
        if contrib.sum() > 0.0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – pick from low‑contribution indices
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of the donor mass
        eps = a[i_donor] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed_frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed_frac
        accept = (delta < 0.0) or (math.exp(-delta / (T + 1e-12)) > np.random.rand())

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple parametric families (used for cheap diversification)
# ----------------------------------------------------------------------
def _cosine_shape(n: int, beta: float) -> np.ndarray:
    """U‑shaped cosine: a_i = 1 + beta·cos(π·i/(n‑1)), projected onto simplex."""
    i = np.arange(n)
    a = 1.0 + beta * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _edge_uniform_mix(n: int, p: float) -> np.ndarray:
    """Mix a small edge mass p (split equally) with a uniform interior."""
    if n < 3:
        return np.full(n, 1.0 / n)
    interior = (1.0 - p) / (n - 2)
    a = np.full(n, interior, dtype=float)
    a[0] = a[-1] = p / 2.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines an Adam‑based smoothed‑max descent,
    an Adam sub‑gradient refinement, and a final peak‑shaving walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Progressive up‑sampling (soft‑max Adam)
    # ------------------------------------------------------------------
    stage_lengths = [500, 1500, 3000, 6000, 9600]   # final length 9600
    total_len_weight = sum(stage_lengths)

    total_time = deadline - start_time
    softmax_budget = total_time * 0.60   # 60 % of the budget
    subgrad_budget = total_time * 0.35   # 35 %
    local_budget  = total_time - softmax_budget - subgrad_budget  # remaining ≈5 %

    best_seq = None
    best_score = float("inf")

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # Allocate a share of the soft‑max budget proportional to length
        stage_frac = n / total_len_weight
        stage_time = max(1.0, softmax_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # Initialise a0 for this length
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Run Adam on the smoothed max‑convolution
        a_opt, score_opt = _adam_softmax_opt(
            a0, n, stage_deadline, start_time,
            T_start=2.0e-2, T_end=5.0e-4,
            alpha=0.12, beta1=0.9, beta2=0.999
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # Update remaining soft‑max budget
        elapsed = time.time() - start_time
        softmax_budget = max(0.0, deadline - start_time - subgrad_budget - local_budget - elapsed)

    # ------------------------------------------------------------------
    # Small parametric diversification (cosine / edge‑mix) on final size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        # try a few cosine shapes
        for beta in (0.2, 0.4, 0.6, 0.8):
            cand = _cosine_shape(n_final, beta)
            conv = _conv_self(cand)
            cand_score = 2.0 * n_final * conv.max()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] cosine beta={beta:.2f} improve: "
                      f"score={cand_score:.6f}")

        # try a few edge‑mixes
        for p in (0.03, 0.06, 0.09, 0.12):
            cand = _edge_uniform_mix(n_final, p)
            conv = _conv_self(cand)
            cand_score = 2.0 * n_final * conv.max()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] edge‑mix p={p:.2f} improve: "
                      f"score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (Adam) on the final length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        subgrad_deadline = time.time() + subgrad_budget
        a_ref, score_ref = _adam_subgrad_opt(
            best_seq, n_final, subgrad_deadline, start_time,
            alpha=0.07, beta1=0.9, beta2=0.999
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Targeted local search (peak shaving) – uses the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list
    return [float(x) for x in best_seq]
