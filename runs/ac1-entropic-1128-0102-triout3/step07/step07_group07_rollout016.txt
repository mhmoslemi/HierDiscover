def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4800

    # Build start points
    starts = []
    # arcsine
    x = (np.arange(n)+0.5) / n
    a_arcsine = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a_arcsine = np.maximum(a_arcsine, 0.0)
    a_arcsine /= a_arcsine.sum()
    starts.append(a_arcsine)
    # uniform
    starts.append(np.full(n, 1.0/n))
    # exponential
    r = 0.98
    exp_vec = np.power(r, np.arange(n))
    exp_vec /= exp_vec.sum()
    starts.append(exp_vec)
    # random Dirichlet
    rng = np.random.default_rng()
    random_dir = rng.gamma(1.0, 1.0, size=n)
    random_dir /= random_dir.sum()
    starts.append(random_dir)
    # previous best if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.shape[0] != n:
                starter = _upsample(starter, n)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
                starts.append(starter)
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")

    # define epoch schedule: α values and iteration count adaptively based on remaining time
    base_alphas = [0.5, 1, 2, 5, 10, 20, 40, 80, 160, 320, 640]
    iters_per_alpha = 12000   # fairly large

    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1}")

        # Softmax parametrisation
        eps = 1e-12
        y = np.log(np.maximum(a0, eps))

        fft_len = _next_pow2(3 * n - 2)

        # Adam hyperparameters
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr0 = 0.12
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0

        # keep best for this start
        best_local_seq = a0.copy()
        best_local_score = _score(best_local_seq)

        for alpha in base_alphas:
            for i in range(iters_per_alpha):
                if time.time() >= deadline:
                    break

                exp_y = np.exp(y - maxy)

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Diverse initialisations** – we generate a few
   candidate vectors of length ``n = 4800``:
   * the classic uniform distribution,
   * a distribution that approximates the continuous arcsine law
     (heavy weight near the ends, light in the centre),
   * a random Dirichlet‑sample,
   * the previously best known construction (if it exists).

2. **Smooth Adam optimisation** – each start is represented in the
   log‑domain (soft‑max).  A schedule of increasing smoothing
   parameters ``α`` (from 0.5 up to 640) is used and for every ``α`` a
   long Adam run (≈12 000 updates) is performed.  The gradient of the
   smooth‑max objective is obtained analytically via FFT, which makes
   the optimisation fast even for 4800 variables.

3. **Projected sub‑gradient polishing** – after the Adam phase we
   switch to the exact non‑smooth objective ``max_k (a*a)[k]``.  A
   sub‑gradient is built from the (few) convolution entries that are
   within a relative tolerance of the current maximum and an Euclidean
   projection onto the probability simplex is applied.  This step
   continues until the time budget for the current start is exhausted.

4. **Deterministic peak‑reduction** – a cheap greedy pass that moves a
   fraction of the coefficient that contributes most to the current
   peak towards the currently smallest coefficient.  The move is kept
   only when the peak strictly decreases.

5. **Fast simulated annealing** – the final refinement performs a
   large number of random mass‑transfer proposals.  The convolution
   update caused by a proposal is computed with an O(n) delta‑formula
   (no full FFT per move).  An exponential cooling schedule is used
   and any improvement (or a Metropolis‑type uphill move) is accepted.

All phases respect the global 1000 s budget.  The best sequence found
by any start is returned as a plain list of floats.

The implementation uses only NumPy and the provided ``evaluate_sequence``
function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # gradient = 2·(a⋆w)   (cross‑correlation)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _score(a: np.ndarray) -> float:
    """Exact score 2·n·max(conv) for a probability vector a."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for f(a)=max_k (a*a)[k].
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient from a few nearly‑maximal entries ----------
        # use a relative tolerance to keep the active set small
        rel_tol = 1e-6
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        if active.size == 0:
            g = np.zeros_like(a)
        else:
            # average sub‑gradient of the active peaks
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size
            # gradient = 2·a⋆w_rev   (FFT for speed)
            w_rev = w[::-1]
            L = _next_pow2(n + len(w_rev) - 1)
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            g = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy peak reduction
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 1500) -> tuple[np.ndarray, float]:
    """Greedy loop that moves mass from the strongest‑contributing index to the smallest coefficient."""
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        # indices that generate this peak
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]                     # ordered contribution
        src = i_idx[contrib.argmax()]                    # source of most mass

        # destination = currently smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: max={new_max:.6f}")
        else:
            no_improve += 1

    score = _score(a)
    return a, score


# ----------------------------------------------------------------------
# Fast simulated‑annealing refinement (delta‑updates)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 0.1,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Classic annealing: move a random ε‑fraction of mass between two indices.
    The convolution update is performed with an O(n) delta‑formula, avoiding a full FFT per move.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise full convolution and current score
    conv = _conv_self(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    # pre‑allocate temporary arrays for speed
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportionally to its mass and a random destination
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30  # up to 30 % of the source mass

        # ---- delta update of the convolution ---------------------------------
        # delta = 2 * (a*δ) + δ*δ   where δ has -eps at src and +eps at dst
        delta = np.zeros_like(conv)

        # linear part (a*δ)
        # contribution from moving *out* of src
        start_k = src
        end_k = src + n - 1
        delta[start_k:end_k + 1] -= 2.0 * eps * a[:n]  # a[k-src]
        # contribution from moving *into* dst
        start_k = dst
        end_k = dst + n - 1
        delta[start_k:end_k + 1] += 2.0 * eps * a[:n]  # a[k-dst]

        # quadratic part (δ*δ)
        delta[2 * src]     += eps * eps
        delta[2 * dst]     += eps * eps
        delta[src + dst]   -= 2.0 * eps * eps

        # new convolution and new max
        conv_prop = conv + delta
        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score < 0 or math.exp(-delta_score / T) > rng.random():
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            cur_score = new_score
            cur_max = new_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={best_score:.6f}")

    return best_seq, best_score
