"""
Propose a near‑optimal non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.  The optimiser works in several stages:

1.  **Random multi‑start small‑scale search** – a few quick Adam + sub‑gradient
    runs on a modest size (n≈1024) starting from different random seeds.
    The best candidate is up‑sampled to the final resolution.

2.  **Large‑scale refinement** – long Adam optimisation on a smooth
    log‑sum‑exp surrogate of the max‑convolution, using many α‑values
    (up to 640) and many iterations per α.

3.  **Projected sub‑gradient descent** on the true objective
    (max‑convolution) with a slowly decaying step‑size
    (t⁻⁰·⁶) and momentum.

4.  **Mass‑transfer simulated annealing** to move tiny amounts of mass
    between coordinates.

5.  **Swap‑annealing** – random swaps of two positions (changing the
    ordering of the heights) accepted with a Metropolis rule.  Ordering
    strongly influences the convolution, so these swaps can break remaining
    local minima.

All phases respect the overall time budget (≈1000 s) and keep the best
sequence found.  The final vector is normalised to sum‑to‑one and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_a(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on a smoothed max‑convolution
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        iters_per_alpha: int = 5000,
                        lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest exact score) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_a(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length needed for the gradient (convolution a * w_rev)
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # current conv and exact score (for bookkeeping)
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smoothed max weights -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)  # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex (Euclidean projection)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))/sum(a)².
    Uses momentum and a t⁻⁰·⁶ step decay.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.92
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # Determine all active indices (within tiny tolerance)
        active = np.where(np.abs(conv - max_val) < 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active))

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / (it ** 0.6)          # slower decay than 1/√t
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : Simulated annealing – tiny mass transfers
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-8,
                   steps: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Lightweight mass‑transfer annealing.
    Moves a random fraction (up to 30 %) of the source mass to a random target.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportionally to its mass (helps escaping plateaus)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑4 : Swap‑annealing – random re‑ordering of heights
# ----------------------------------------------------------------------
def _run_swap_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 2e-2,
                        end_temp: float = 1e-8,
                        steps: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing where a move consists of swapping two entries.
    Ordering has a large impact on the convolution, so this can reduce the
    max‑convolution further.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = rng.choice(n, size=2, replace=False)
        a_prop = a.copy()
        a_prop[i], a_prop[j] = a[j], a[i]

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] swap‑anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper: quick optimisation on a small problem size (used for multi‑start)
# ----------------------------------------------------------------------
def _quick_optimize_small(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap optimisation on a reduced size (n≈1024).  Runs a short Adam
    phase followed by a short sub‑gradient phase.
    """
    # Short Adam run
    a_adam, score_adam = _run_adam_smoothing(
        a0,
        n,
        deadline,
        start_time,
        alphas=[0.5, 2.0, 5.0, 10.0, 20.0],
        iters_per_alpha=1500,
        lr0=0.7,
    )
    # Reserve a little time for sub‑gradient descent
    if time.time() < deadline:
        sub_deadline = deadline - 0.5
        a_sub, score_sub = _run_subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score_initial=score_adam,
        )
        if score_sub < score_adam:
            return a_sub, score_sub
    return a_adam, score_adam


def _upsample_to(a_small: np.ndarray, final_n: int) -> np.ndarray:
    """Linear interpolation from a smaller vector to the final length."""
    if a_small.shape[0] == final_n:
        return a_small.copy()
    xs = np.linspace(0, 1, a_small.shape[0])
    xf = np.linspace(0, 1, final_n)
    a_big = np.interp(xf, xs, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    return a_big

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # small safety margin

    # --------------------------------------------------------------
    # Determine a usable starter (global if available)
    # --------------------------------------------------------------
    try:
        starter_global = np.asarray(height_sequence_1, dtype=float)
        if starter_global.ndim == 1 and starter_global.size > 0:
            starter_global = np.maximum(starter_global, 0.0)
            starter_global /= starter_global.sum()
        else:
            starter_global = None
    except Exception:
        starter_global = None

    # --------------------------------------------------------------
    # Multi‑start small‑scale exploration
    # --------------------------------------------------------------
    small_n = 1024
    best_small_seq = None
    best_small_score = float("inf")
    num_random_starts = 3

    # Use the global starter as the first start if it exists
    start_candidates = []
    if starter_global is not None:
        # down‑sample to small_n
        a0 = np.interp(
            np.linspace(0, 1, small_n),
            np.linspace(0, 1, starter_global.size),
            starter_global,
        )
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        start_candidates.append(a0)

    # add random starts
    rng = np.random.default_rng(seed + 12345)
    for _ in range(num_random_starts - len(start_candidates)):
        a0 = rng.random(small_n)
        a0 /= a0.sum()
        start_candidates.append(a0)

    for idx, a0 in enumerate(start_candidates):
        if time.time() >= deadline:
            break
        # allocate roughly 30 s per small run (or the remaining time if tiny)
        phase_deadline = min(deadline, time.time() + 30.0)
        a_opt, score_opt = _quick_optimize_small(
            a0, small_n, phase_deadline, start_time
        )
        # up‑sample to the final resolution for scoring
        a_big = _upsample_to(a_opt, 4096)
        big_score = _score_from_a(a_big)
        if big_score < best_small_score:
            best_small_score = big_score
            best_small_seq = a_big.copy()
            print(f"[{time.time() - start_time:.2f}s] small‑scale improve: "
                  f"candidate={idx}, score={big_score:.6f}")

    # If for any reason we have no candidate (e.g. out of time), fall back
    if best_small_seq is None:
        # uniform distribution as a last resort
        best_small_seq = np.full(4096, 1.0 / 4096)

    # --------------------------------------------------------------
    # Large‑scale refinement on the chosen candidate
    # --------------------------------------------------------------
    n_final = best_small_seq.shape[0]
    a_current = best_small_seq.copy()
    best_score = _score_from_a(a_current)
    best_seq = a_current.copy()

    # Phase‑1 : Adam on a smooth surrogate (big portion of the budget)
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + max(10.0, 0.15 * remaining)
        a_adam, score_adam = _run_adam_smoothing(
            a_current,
            n_final,
            phase1_deadline,
            start_time,
            alphas=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0],
            iters_per_alpha=5000,
            lr0=0.5,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={score_adam:.6f}")

    # Phase‑2 : exact‑max sub‑gradient descent (major chunk)
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + max(20.0, 0.65 * remaining)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            n_final,
            phase2_deadline,
            start_time,
            best_score_initial=best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_sub:.6f}")

    # Phase‑3 : mass‑transfer annealing
    if time.time() < deadline:
        a_anneal, score_anneal = _run_annealing(
            best_seq,
            n_final,
            deadline,
            start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-8,
            steps=150_000,
        )
        if score_anneal < best_score:
            best_score = score_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"score={score_anneal:.6f}")

    # Phase‑4 : swap‑annealing (order optimisation)
    if time.time() < deadline:
        a_swap, score_swap = _run_swap_annealing(
            best_seq,
            n_final,
            deadline,
            start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=2e-2,
            end_temp=1e-8,
            steps=100_000,
        )
        if score_swap < best_score:
            best_score = score_swap
            best_seq = a_swap.copy()
            print(f"[{time.time() - start_time:.2f}s] swap‑anneal improve: "
                  f"score={score_swap:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n_final, 1.0 / n_final)

    return [float(x) for x in best_seq]
