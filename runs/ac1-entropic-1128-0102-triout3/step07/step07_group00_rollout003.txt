def _adam_smoothing(a0,n,deadline,start_time):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    L = _next_pow2(3*n-2)
    for alpha in [0.5,2,5,10,20,40,80,160]:
        for _ in range(2000):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1
            conv = _conv_self(a)
            cur_score = 2.0*n*conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam iter={it} score={cur_score:.6f}")
            # compute softmax weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum
            w_rev = w[::-1]
            # gradient
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            lr = 0.5 / math.sqrt(it + 1)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)
    return best_a, best_score

def _subgrad_descent(a0, n, deadline, start_time, init_score):
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={it} score={cur_score:.6f}")
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # contributions from pairs (i, k-i)
        grad[idx] = 2.0 * a[k - idx]
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score

def _annealing_polish(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                      start_temp=0.5, end_temp=1e-6, steps=200_000):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        # pick source and destination uniformly
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.5  # move up to 50% of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal iter={it} best={best_score:.6f}")
    return best_seq, best_score

"""
Multi‑restart heuristic optimizer for the max‑auto‑convolution problem.

The algorithm works in three stages for each restart:

  1. Adam on a log‑sum‑exp (soft‑max) surrogate of max(conv(a,a)).
  2. Projected sub‑gradient descent on the exact objective
     f(a)=2·n·max(conv(a,a)) (the same as evaluate_sequence for a normalized a).
  3. An aggressive simulated‑annealing polish that moves larger mass chunks.

Several random initialisations (uniform, exponential, reversed exponential,
Dirichlet) are tried for each of a few sequence lengths (default 4096, also
2048/8192 when time permits).  The best sequence found before the time
budget expires is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numeric safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smoothing(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best iterate (lowest 2·n·max(conv)) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    # FFT length sufficient for all convolutions in this phase
    L = _next_pow2(3 * n - 2)

    # a schedule of smoothing parameters (larger α → sharper surrogate)
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    iters_per_alpha = 2500

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam iter={it:,} score={cur_score:.6f}")

            # ----- soft‑max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = 0.5 / math.sqrt(it)    # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # stay on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Exact max – single‑peak sub‑gradient descent
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     init_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum and a 1/√t step‑size schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={it:,} score={cur_score:.6f}")

        # index where the convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # contribution of each a[idx] to the maximal entry
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Aggressive simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing_polish(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray,
                      start_temp: float = 5e-1,
                      end_temp: float = 1e-6,
                      steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Light‑weight annealing that moves up to 50 % of the source mass.
    Temperature decays exponentially; accepts worsening moves with the
    usual Metropolis probability.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to 50 %) of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a non‑negative step function (list of heights) that minimises
    evaluate_sequence under the given time budget.  The routine uses several
    random restarts, each consisting of Adam smoothing → sub‑gradient descent
    → annealing, and keeps the best solution found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # Sequence lengths to explore (power‑of‑two values are FFT‑friendly)
    lengths = [4096, 2048, 8192]

    for n in lengths:
        # check whether we still have enough time for another length
        if time.time() >= deadline - 30:
            break

        # estimate how many restarts we can afford for this length
        remaining = deadline - time.time()
        # one restart needs roughly 300‑400 s; adjust conservatively
        max_restarts = max(1, int(remaining // 350))
        for restart in range(max_restarts):
            # ------- initialise a candidate distribution -------
            init_type = np.random.choice(['uniform', 'exp', 'revexp', 'dirichlet'])
            if init_type == 'uniform':
                a0 = np.full(n, 1.0 / n)
            elif init_type == 'exp':
                lam = np.random.uniform(0.0005, 0.005)
                a0 = np.exp(-lam * np.arange(n))
                a0 /= a0.sum()
            elif init_type == 'revexp':
                lam = np.random.uniform(0.0005, 0.005)
                a0 = np.exp(-lam * np.arange(n)[::-1])
                a0 /= a0.sum()
            else:  # Dirichlet
                conc = np.random.uniform(0.2, 2.0)
                a0 = np.random.dirichlet([conc] * n)

            # optionally inject the globally stored construction (if any)
            if restart == 0 and 'height_sequence_1' in globals():
                try:
                    hs = np.asarray(height_sequence_1, dtype=float)
                    if hs.ndim == 1 and hs.size > 0:
                        a0 = np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, hs.size),
                                       hs)
                        a0 = np.maximum(a0, 0.0)
                        a0 /= a0.sum()
                except Exception:
                    pass

            # --------------------------------------------------
            # Phase 1 – Adam on the smooth surrogate
            # --------------------------------------------------
            phase1_deadline = min(deadline, time.time() + max(5.0, 0.07 * (deadline - time.time())))
            a_adam, score_adam = _adam_smoothing(a0, n, phase1_deadline, start_time)
            if score_adam < best_score:
                best_score = score_adam
                best_seq = a_adam.copy()
                print(f"[{time.time() - start_time:.2f}s] restart {restart} n={n} phase‑1 best={score_adam:.6f}")

            # --------------------------------------------------
            # Phase 2 – exact‑max projected sub‑gradient descent
            # --------------------------------------------------
            if time.time() < deadline:
                phase2_deadline = min(deadline, time.time() + max(20.0, 0.75 * (deadline - time.time())))
                a_sub, score_sub = _subgrad_descent(best_seq, n, phase2_deadline, start_time, best_score)
                if score_sub < best_score:
                    best_score = score_sub
                    best_seq = a_sub.copy()
                    print(f"[{time.time() - start_time:.2f}s] restart {restart} n={n} phase‑2 best={score_sub:.6f}")

            # --------------------------------------------------
            # Phase 3 – final annealing polish
            # --------------------------------------------------
            if time.time() < deadline:
                a_fin, score_fin = _annealing_polish(best_seq, n, deadline, start_time,
                                                     best_score, best_seq,
                                                     start_temp=5e-1,
                                                     end_temp=1e-6,
                                                     steps=200_000)
                if score_fin < best_score:
                    best_score = score_fin
                    best_seq = a_fin.copy()
                    print(f"[{time.time() - start_time:.2f}s] restart {restart} n={n} final best={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the (unlikely) event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096)

    # final normalisation and conversion to plain Python list
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
