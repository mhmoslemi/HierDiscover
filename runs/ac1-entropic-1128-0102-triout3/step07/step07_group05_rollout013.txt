    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()

        T = start_temp * (end_temp/start_temp) ** (iter/total_moves)

        # choose source weighted by a
        src = rng.choice(n, p=a)

def _search_parametric(n, deadline, start_time):
    best_score = float('inf')
    best_seq = None
    # list of shape generators
    def hann_shape():
        i = np.arange(n)
        w = np.sin(np.pi * i / (n-1))**2 # sin^2 shape
        return w
    def hamming_shape():
        i...
    def blackman_shape():
        ...
    def kaiser_shape(beta):
        return np.kaiser(n, beta)
    def gaussian_shape(sigma):
        i = np.arange(n)
        center = (n-1)/2.0
        w = np.exp(-0.5 * ((i - center)/sigma)**2)
        return w
    def exponential_shape(lam):
        i = np.arange(n)
        center = (n-1)/2.0
        w = np.exp(-lam * np.abs(i - center))
        return w
    def power_shape(p):
        i = np.arange(n)
        w = ((i+1)*(n-i))**p
        return w

    shape_fns = [(hann_shape, None), (hamming_shape, None), (blackman_shape, None)]
    # kaiser betas
    for beta in np.linspace(0.5, 14.0, 15):
        shape_fns.append((lambda n, beta=beta: np.kaiser(n, beta), None))

    # gaussian sigma 0.1n to 0.5n
    for sigma_frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        shape_fns.append((lambda n, sigma_frac=sigma_frac: np.exp(-0.5*((np.arange(n)-(n-1)/2)/(sigma_frac*n))**2), None))

    # exponential lambdas
    for lam in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        shape_fns.append((lambda n, lam=lam: np.exp(-lam*np.abs(np.arange(n)-(n-1)/2)), None))

    # power shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        shape_fns.append((lambda n, p=p: ((np.arange(n)+1)*(n-np.arange(n)))**p, None))

    # default uniform
    shape_fns.append((lambda n: np.ones(n), None))

    # initial random shape
    shape_fns.append((lambda n: np.random.rand(n), None))

    for fn, _ in shape_fns:
        if time.time() >= deadline:
            break
        try:
            a = fn(n)
        except Exception as e:
            continue
        a = np.maximum(a, 0)
        if a.sum() == 0:
            continue
        a = a / a.sum()
        # evaluate
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] parametric init improvement: shape={fn.__name__ if hasattr(fn,'__name__') else 'lambda'}, score={sc:.6f}")
    # If best_seq is None, fallback to random
    if best_seq is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a.copy()
        best_score = evaluate_sequence(a.tolist())
    return best_seq, best_score

"""
Hybrid optimiser for the step‑function “attack”.

The algorithm works in two stages within the 1000 s budget.

1. **Parametric initialisation**
   – A small library of analytic windows (Hann, Hamming, Blackman,
     Kaiser, Gaussian, exponential and power‑law) is sampled with a few
     parameters.  Each candidate is normalised to sum = 1 and evaluated
     with ``evaluate_sequence``.  The best candidate from this cheap sweep
     is used as a starting point.

2. **Targeted simulated annealing**
   – The convolution ``b = a * a`` is maintained incrementally (O(n) per
     move).  At each step we locate the index `k_star` where the
     convolution attains its maximum.  The contribution of every entry
     `a[i]` to this maximum equals `a[i] * a[k_star-i]` (when defined);
     the entry with the largest contribution is taken as the *source*
     of mass.  A small fraction ε of that mass is transferred to a
     uniformly chosen destination.  The move is kept if it lowers the
     objective; otherwise it is undone using the same O(n) update.
   – Temperature is decreased geometrically; uphill moves are occasionally
     accepted according to the Metropolis criterion.
   – Every 200 000 moves the full convolution is recomputed with FFT to
     eliminate numerical drift.

When the time budget expires the best sequence found is re‑normalised
(and clipped to non‑negative) and returned.  All helper functions are
top‑level and have no external dependencies beyond ``numpy`` and ``math``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep of simple windows
# ----------------------------------------------------------------------
def _parametric_initialisation(n: int, deadline: float, start_time: float):
    """
    Scan a few classic windows (Hann, Hamming, Blackman, Kaiser, Gaussian,
    exponential, power‑law) and keep the best.  Returns (seq, score).
    """
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # window generators – each returns a non‑negative vector of length n
    # ------------------------------------------------------------------
    def _hann(_n):
        i = np.arange(_n)
        return np.sin(np.pi * i / (_n - 1)) ** 2

    def _hamming(_n):
        i = np.arange(_n)
        return 0.54 - 0.46 * np.cos(2 * np.pi * i / (_n - 1))

    def _blackman(_n):
        i = np.arange(_n)
        return (
            0.42
            - 0.5 * np.cos(2 * np.pi * i / (_n - 1))
            + 0.08 * np.cos(4 * np.pi * i / (_n - 1))
        )

    def _kaiser_factory(beta):
        return lambda _n: np.kaiser(_n, beta)

    def _gaussian_factory(sigma_frac):
        def gen(_n):
            i = np.arange(_n)
            centre = (_n - 1) / 2.0
            sigma = sigma_frac * _n
            return np.exp(-0.5 * ((i - centre) / sigma) ** 2)

        return gen

    def _exp_factory(lam):
        def gen(_n):
            i = np.arange(_n)
            centre = (_n - 1) / 2.0
            return np.exp(-lam * np.abs(i - centre))

        return gen

    def _power_factory(p):
        def gen(_n):
            i = np.arange(_n)
            return ((i + 1) * (_n - i)) ** p

        return gen

    # ------------------------------------------------------------------
    # collection of (generator, description) pairs
    # ------------------------------------------------------------------
    generators = [
        (_hann, "hann"),
        (_hamming, "hamming"),
        (_blackman, "blackman"),
    ]

    for beta in np.linspace(0.5, 14.0, 15):
        generators.append((_kaiser_factory(beta), f"kaiser(beta={beta:.1f})"))

    for sigma_frac in [0.05, 0.10, 0.15, 0.20, 0.30, 0.40]:
        generators.append(
            (_gaussian_factory(sigma_frac), f"gaussian(σ={sigma_frac:.2f})")
        )

    for lam in [0.0005, 0.001, 0.002, 0.005, 0.010]:
        generators.append((_exp_factory(lam), f"exp(λ={lam:.4f})"))

    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        generators.append((_power_factory(p), f"power(p={p:.1f})"))

    # a single random initialisation (useful as a fallback)
    generators.append(
        (lambda _n: rng.random(_n), "random")
    )

    # ------------------------------------------------------------------
    # evaluate every generator (stop early if time runs out)
    # ------------------------------------------------------------------
    for gen, name in generators:
        if time.time() >= deadline:
            break
        try:
            a = gen(n).astype(float)
        except Exception:
            continue
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] parametric init "
                f"improve: {name:20s} score={score:.6f}"
            )
    # safety net – if nothing succeeded fall back to pure random
    if best_seq is None:
        a = rng.random(n)
        a /= a.sum()
        best_seq = a
        best_score = evaluate_sequence(a.tolist())
        print(f"[{time.time() - start_time:6.2f}s] fallback random init score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Targeted simulated annealing / mass‑transfer optimisation
# ----------------------------------------------------------------------
def _targeted_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
):
    """
    Move mass from the index that contributes most to the current maximal
    convolution towards a random destination.  The convolution is updated
    incrementally; the move is kept if it improves the objective,
    otherwise it is reverted (using the same O(n) update).  Temperature
    follows a geometric schedule to allow occasional uphill steps.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    iteration = 0

    # annealing schedule parameters
    total_time = deadline - start_time
    start_temp = 5e-2
    end_temp = 1e-8

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        # geometric cooling based on elapsed fraction
        T = start_temp * ((end_temp / start_temp) ** (elapsed / total_time))

        # ---- locate the current maximal convolution entry ----
        k_star = int(conv.argmax())

        # ---- compute contributions a[i] * a[k_star-i] ----
        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # source = index with largest contribution (break ties randomly)
        src_candidates = np.flatnonzero(contrib == contrib.max())
        src = int(rng.choice(src_candidates))

        # avoid pathological zero‑mass sources
        if a[src] < 1e-12:
            # fallback to a mass‑weighted random choice
            src = int(rng.choice(n, p=a))

        dst = rng.integers(n)
        if src == dst:
            continue

        # ---- choose transfer amount (up to 30 % of source mass) ----
        max_transfer = min(0.30, a[src])  # a[src] ≤ 1, so this caps at 0.30
        eps = max_transfer * rng.random()
        if eps <= 0.0:
            continue

        # ---- store pre‑move values for a quick revert ----
        a_src_before = a[src]
        a_dst_before = a[dst]

        # ---- apply the incremental convolution update (using a before move) ----
        conv[src : src + n] -= 2.0 * eps * a
        conv[dst : dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # ---- update the probability vector ----
        a[src] -= eps
        a[dst] += eps

        # ---- evaluate new objective ----
        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score

        # ---- Metropolis acceptance ----
        accept = delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random()
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] anneal improve: "
                    f"iter={iteration:,}, score={cur_score:.6f}"
                )
        else:
            # ---- revert both a and conv (exact inverse of the update) ----
            a[src] = a_src_before
            a[dst] = a_dst_before
            # after restoring a, perform the opposite update
            conv[src : src + n] += 2.0 * eps * a
            conv[dst : dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # cur_score unchanged

        # Periodically recompute the full convolution to kill drift
        if iteration % 200_000 == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine first tries a quick parametric sweep, then refines the best
    candidate with a targeted simulated‑annealing phase that runs until the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – the reference used n = 4800, which offers a good
    # balance between solution quality and speed
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # 1️⃣  initialise from a library of analytic windows (or fall back
    #     to the previously discovered construction if it exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                a0 = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, a0.shape[0]),
                    a0,
                )
        else:
            raise Exception
    except Exception:
        a0 = None

    if a0 is None:
        a0, init_score = _parametric_initialisation(n, deadline, start_time)
    else:
        init_score = evaluate_sequence(a0.tolist())
        print(f"[{0.0:6.2f}s] using provided height_sequence_1, score={init_score:.6f}")

    best_seq = a0.copy()
    best_score = init_score

    # ------------------------------------------------------------------
    # 2️⃣  targeted annealing – run until the time budget expires
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _targeted_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_seq,
            best_score,
        )

    # ------------------------------------------------------------------
    # final safety normalisation (protect against tiny numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # ------------------------------------------------------------------
    # final check (optional but useful for logging)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
