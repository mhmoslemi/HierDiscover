    best_a = a.copy()
    best_score = compute_score(a)

    # parameters
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # schedule temperature linearly decreasing from T0 to Tf
        elapsed = (time.time() - start_time) / total_time
        T = 0.05 * (0.001 / 0.05) ** elapsed   # exponential from 0.05 to 0.001

        # propose move:
        # choose donor i weighted by a_i, receiver j weighted by (1 - a_j?) maybe uniform.
        donor = np.random.choice(n, p=a)  # probability proportional to mass

        eps = a[donor] * np.random.rand() * 0.5
        if eps < 1e-12:
            continue

        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        # reproject onto simplex? Actually after moving mass we keep sum = 1 and nonnegative, yes.

        # compute score
        score = compute_score(a_prop)

        cur_score = compute_score(a)

        delta_cur = score - cur_score

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Warm‑start** – load the previously found construction (``height_sequence_1``)
   if it exists, otherwise start from a random simplex point.
2. **Target length** – work on a fairly large vector (≈ 2·10⁴ entries).  This
   gives enough degrees of freedom to lower the maximal self‑convolution.
3. **Phase A – Adam on a smooth soft‑max of the convolution**.  
   A smoothed version of the max‑convolution (soft‑max) is differentiable;
   we run Adam with an exponential temperature schedule to shape the
   distribution quickly.
4. **Phase B – exact sub‑gradient descent**.  
   Using the true max‑convolution we perform a projected sub‑gradient step
   (Nesterov momentum) to further shrink the peak.
5. **Phase C – aggressive simulated‑annealing (SA) “peak‑shaving’’**.  
   Moves consist of transferring a small fraction of mass from a donor index
   (chosen proportionally to its current weight) to a receiver index.
   The convolution is updated incrementally (O(n) per move) and the move is
   accepted with a Metropolis rule under a temperature schedule.
   The SA phase explores many tiny perturbations and often finds a
   lower maximum than the gradient phases alone.
6. **Return** the best sequence seen, projected onto the probability simplex
   (ensuring non‑negativity and unit sum).

All phases respect the supplied time budget (≈ 1000 s).  The code is
self‑contained, uses only NumPy and the standard library, and follows the
required API.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper – convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring – 2·n·max(conv)  (equivalent to evaluate_sequence after normalising)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Soft‑max gradient (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the convolution.
    The factor 2 comes from the derivative of aᵀJ_k a (Toeplitz matrix).
    """
    # Stabilise soft‑max
    scaled = (conv - conv.max()) / temperature
    exps = np.exp(scaled - np.max(scaled))  # extra shift for safety
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block
    return grad


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth soft‑max of the convolution
# ----------------------------------------------------------------------
def _run_adam_softmax_gd(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         lr: float = 0.12,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max approximation of the max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.5,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution entry."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # Exact sub‑gradient of the active convolution entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        # Momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – Simulated Annealing “Peak Shaving’’
# ----------------------------------------------------------------------
def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Aggressive SA that moves a small fraction of mass from a donor (chosen
    proportionally to its weight) to a receiver (uniform).  Convolution is
    updated incrementally (O(n) per move).  Temperature decays exponentially.
    """
    a = a0.copy()
    conv = _conv_self(a)                     # full convolution, length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    best_conv = conv.copy()

    # temperature schedule
    T_start = 5.0e-2
    T_end = 1.0e-7

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # donor proportional to current mass
        donor = np.random.choice(n, p=a)
        # receiver uniform among the other indices
        rec = np.random.randint(n - 1)
        if rec >= donor:
            rec += 1

        # amount to transfer – up to 30 % of donor's mass
        eps = a[donor] * np.random.rand() * 0.30
        if eps <= 1e-15:
            continue

        # Incrementally update the convolution (linear change)
        #   Δconv = -2·eps·shift(donor) + 2·eps·shift(receiver)
        # shift vectors are just the current a (before the move)
        shift = a
        conv_new = conv.copy()
        conv_new[donor:donor + n] -= 2.0 * eps * shift
        conv_new[rec:rec + n]     += 2.0 * eps * shift

        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score

        # elapsed fraction for temperature
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        accept = delta < 0.0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            # commit move
            a[donor] -= eps
            a[rec]    += eps
            conv = conv_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: n={n}, "
                      f"iter={it}, score={new_score:.6f}")

        # occasional re‑projection to fight numerical drift
        if it % 5000 == 0:
            a = _project_simplex(a)

    # final clean‑up – project back onto the simplex (ensures exact sum = 1)
    best_a = _project_simplex(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` from its current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search combines Adam (soft‑max), sub‑gradient descent and a
    simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.4

    # ------------------- Warm‑start -------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = np.maximum(0.0, starter)
            starter = starter / starter.sum()          # normalise to simplex
    except Exception:
        starter = None

    # ------------------- Target length -------------------
    # 19200 gives a good trade‑off between expressive power and speed.
    n_target = 19200

    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _score(best_seq)
    print(f"[0.00s] initial score {best_score:.6f}")

    # ------------------- Time budget split -------------------
    total_time = deadline - start_time
    # 30 % soft‑max Adam, 30 % exact sub‑gradient, 40 % SA polishing
    phase_a_dead = start_time + total_time * 0.30
    phase_b_dead = start_time + total_time * 0.60
    phase_c_dead = deadline

    # ------------------- Phase A – Adam (soft‑max) -------------------
    a_phase_a, score_a = _run_adam_softmax_gd(
        best_seq, n_target, phase_a_dead, start_time,
        lr=0.12, beta1=0.9, beta2=0.999, eps=1e-8
    )
    if score_a < best_score:
        best_score = score_a
        best_seq = a_phase_a.copy()
        print(f"[{time.time() - start_time:6.2f}s] after Adam phase score {best_score:.6f}")

    # ------------------- Phase B – exact sub‑gradient -------------------
    if time.time() < phase_b_dead:
        a_phase_b, score_b = _run_subgrad_gd(
            best_seq, n_target, phase_b_dead, start_time,
            step0=0.5, beta=0.9
        )
        if score_b < best_score:
            best_score = score_b
            best_seq = a_phase_b.copy()
            print(f"[{time.time() - start_time:6.2f}s] after sub‑grad phase score {best_score:.6f}")

    # ------------------- Phase C – Simulated Annealing -------------------
    if time.time() < phase_c_dead:
        a_phase_c, score_c = _run_simulated_annealing(
            best_seq, n_target, phase_c_dead, start_time
        )
        if score_c < best_score:
            best_score = score_c
            best_seq = a_phase_c.copy()
            print(f"[{time.time() - start_time:6.2f}s] after SA phase score {best_score:.6f}")

    # ------------------- Final normalisation -------------------
    best_seq = _project_simplex(best_seq)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
