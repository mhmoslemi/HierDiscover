"""
Propose a step–function minimizing
    2·n·max(conv(a,a)) / (sum(a))**2

The search starts from the best known construction (``height_sequence_1``
if it exists) and then runs a fast local‑refinement that repeatedly
moves a small amount of probability mass from indices that contribute
most to the current maximal convolution entry to a better destination.
For each move several random destinations are tried and the one that
gives the largest decrease of the maximal convolution value is kept.
The update of the convolution is done analytically (no FFT per move),
so millions of tiny moves can be performed inside the 1000 s budget.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of ``seq`` to length ``new_n`` while preserving mass.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Targeted refinement – fast incremental convolution updates
# ----------------------------------------------------------------------
def _run_targeted_refine_adaptive(a0: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  start_time: float,
                                  initial_eps_frac: float = 0.025,
                                  min_eps_frac: float = 1e-4,
                                  decay_rate: float = 0.9995,
                                  cand_per_iter: int = 5,
                                  recompute_every: int = 20000):
    """
    Tiny moves focused on the current maximal convolution entry.
    At each step a source index is chosen proportionally to its
    contribution to the maximal entry and several destination indices
    are sampled; the move that yields the greatest strict decrease of
    the max‑convolution is performed.
    The convolution is updated analytically, which makes the loop
    extremely fast.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val

    rng = np.random.default_rng()
    eps_frac = initial_eps_frac
    it = 0
    # Helper: recompute convolution from scratch (keeps numeric drift in check)
    def recompute():
        return _conv_self(a)

    while time.time() < deadline:
        it += 1

        # ---- choose source based on contribution to current max ----
        k_max = int(np.argmax(conv))
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total = contrib.sum()
        if total > 0:
            prob_src = contrib / total
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- propose several random destinations ----
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # sample destinations (avoid src)
        cand_dst = rng.integers(0, n - 1, size=cand_per_iter)
        cand_dst = np.where(cand_dst >= src, cand_dst + 1, cand_dst)

        a_old = a.copy()          # needed for the linear update formula
        best_candidate = None
        best_candidate_max = max_val

        for dst in cand_dst:
            # build a temporary copy of the convolution
            new_conv = conv.copy()

            # linear part: 2*eps*(shift_dst - shift_src)
            new_conv[dst:dst + n] += 2.0 * eps * a_old
            new_conv[src:src + n] -= 2.0 * eps * a_old

            # quadratic part: eps^2 at 2*dst, 2*src and -2*eps^2 at src+dst
            idx2_dst = 2 * dst
            idx2_src = 2 * src
            idx_src_dst = src + dst
            if idx2_dst < new_conv.size:
                new_conv[idx2_dst] += eps * eps
            if idx2_src < new_conv.size:
                new_conv[idx2_src] += eps * eps
            if idx_src_dst < new_conv.size:
                new_conv[idx_src_dst] -= 2.0 * eps * eps

            # safety – keep non‑negative (tiny round‑off may appear)
            if new_conv.min() < -1e-15:
                continue

            new_max = new_conv.max()
            if new_max < best_candidate_max - 1e-14:
                best_candidate_max = new_max
                best_candidate = dst

        # ---- apply the best improving move (if any) ----
        if best_candidate is not None:
            dst = best_candidate
            # update a
            a[src] -= eps
            a[dst] += eps

            # update conv using the same linear/quadratic formula (with a_old)
            conv[dst:dst + n] += 2.0 * eps * a_old
            conv[src:src + n] -= 2.0 * eps * a_old
            if 2 * dst < conv.size:
                conv[2 * dst] += eps * eps
            if 2 * src < conv.size:
                conv[2 * src] += eps * eps
            if src + dst < conv.size:
                conv[src + dst] -= 2.0 * eps * eps

            # minor clean‑up
            max_val = conv.max()
            if max_val < best_max - 1e-14:
                best_max = max_val
                best_a = a.copy()
                objective = 2.0 * n * best_max
                print(f"[{time.time() - start_time:.2f}s] improve: "
                      f"iter={it:,}, max={best_max:.8f}, obj={objective:.6f}")

        # ---- periodic housekeeping ----
        if it % recompute_every == 0:
            # recompute convolution from scratch to erase accumulated error
            conv = recompute()
            max_val = conv.max()

        # decay epsilon fraction slowly
        eps_frac = max(eps_frac * decay_rate, min_eps_frac)

        # occasional renormalisation of tiny entries (keeps active set small)
        if it % 50000 == 0:
            tiny = a < 1e-12
            if tiny.any():
                a[tiny] = 0.0
                total = a.sum()
                if total > 0:
                    a /= total
                else:
                    a[:] = 1.0 / n
                conv = recompute()
                max_val = conv.max()

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.  The routine
    loads the best known construction (``height_sequence_1`` if present),
    upsamples it to 4800 points, and then performs an aggressive
    targeted refinement using the fast incremental convolution updates.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # spare margin

    # --------------------------------------------------------------
    # initialise – use previously found construction if available
    # --------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = _upsample(starter, final_n)
    except Exception:
        starter = None

    if starter is None:
        # fallback: random probability vector
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()
    else:
        a0 = starter / starter.sum()

    # --------------------------------------------------------------
    # run the adaptive targeted refinement for the bulk of the budget
    # --------------------------------------------------------------
    a_best = _run_targeted_refine_adaptive(
        a0,
        final_n,
        deadline,
        start_time,
        initial_eps_frac=0.025,
        min_eps_frac=5e-5,
        decay_rate=0.9998,
        cand_per_iter=7,
        recompute_every=15000,
    )

    # final safety – clip negativity, renormalise
    a_best = np.maximum(a_best, 0.0)
    s = a_best.sum()
    if s > 0:
        a_best /= s
    else:
        a_best = np.full(final_n, 1.0 / final_n)

    # report final objective for logging purposes
    final_conv = _conv_self(a_best)
    final_max = final_conv.max()
    final_obj = 2.0 * final_n * final_max
    print(f"[{time.time() - start_time:.2f}s] final objective: {final_obj:.6f}")

    return [float(x) for x in a_best]
