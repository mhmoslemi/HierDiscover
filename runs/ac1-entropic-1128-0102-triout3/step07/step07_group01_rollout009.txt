def _run_gradient_refine(a0, n, deadline, start_time, max_iters=5e6, eps_frac=0.03, tol=1e-12):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    it = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute subgradient
        g = _subgradient_full(a, conv, rel_tol=1e-12)
        src = int(g.argmax())
        dst = int(g.argmin())
        if src == dst:
            # choose random
            src = rng.integers(n)
            dst = (src + rng.integers(1, n)) % n
        # choose epsilon
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        conv_new = _conv_self(a_new)
        new_max = conv_new.max()
        if new_max < max_val - 1e-12:
            a = a_new
            conv = conv_new
            max_val = new_max
            no_improve = 0
            if new_max < best_max - 1e-12:

                best_a = a.copy()

"""
Multi‑phase optimizer for the step‑function bound

The objective to minimise is

    score = 2 * n * max( a ⊛ a )          (a is a probability vector)

because the provided ``evaluate_sequence`` normalises the sum to 1.
The algorithm consists of

1.  **Exponential‑gradient warm‑up** – a smooth‑max (log‑sum‑exp)
    surrogate is minimised using a multiplicative update.

2.  **Projected sub‑gradient descent** – the exact sub‑gradient of the
    (non‑smooth) max is computed and the iterate is projected onto the
    probability simplex.

3.  **Gradient‑guided local refinement** – at each step the index whose
    sub‑gradient is largest (most responsible for the current maximal
    convolution entry) donates a tiny amount of probability mass to the
    index with the smallest sub‑gradient.  The move is accepted only if
    the maximal convolution value strictly decreases, which guarantees
    monotone improvement while aggressively attacking the active
    constraints.

Hyper‑parameters are chosen so that the three phases comfortably fit
into the 1000 s budget.  The routine returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    All indices whose convolution value lies within `rel_tol` of the max
    are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]          # the symmetric contribution
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed max (log‑sum‑exp) gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update (mirror descent with KL‑divergence)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.30,
                             beta: float = 0.93,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : gradient‑guided local refinement
# ----------------------------------------------------------------------
def _run_gradient_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         eps_frac: float = 0.018,
                         tol: float = 1e-12) -> np.ndarray:
    """
    Repeatedly move a tiny amount of mass from the index with the
    largest sub‑gradient (most responsible for the current maximal
    convolution entry) to the index with the smallest sub‑gradient.
    The move is kept only when it strictly lowers the maximal
    convolution value.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # sub‑gradient of the current maximum
        g = _subgradient_full(a, conv, rel_tol=tol)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # sanity checks – if src==dst or src has negligible mass, fallback
        if src == dst or a[src] < 1e-15:
            src = rng.integers(n)
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

        # amount to relocate (a small fraction of the source mass)
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()

        if new_max < max_val - 1e-13:          # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            no_improve = 0
            if new_max < best_max - 1e-13:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] grad‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")
        else:
            no_improve += 1
            # occasional random jitter to escape tiny plateaus
            if no_improve % 5000 == 0:
                src = rng.integers(n)
                dst = (src + rng.integers(1, n)) % n
                eps = a[src] * rng.random() * eps_frac * 0.5
                if eps > 0:
                    a_tmp = a.copy()
                    a_tmp[src] -= eps
                    a_tmp[dst] += eps
                    conv_tmp = _conv_self(a_tmp)
                    if conv_tmp.max() < max_val - 1e-13:
                        a = a_tmp
                        conv = conv_tmp
                        max_val = conv_tmp.max()
                        no_improve = 0

        # early exit if we are stuck for a long time
        if it % 200000 == 0 and time.time() >= deadline:
            break

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Initialise – try to reuse the previously found construction
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase‑1 – warm‑up (exponentiated gradient)
    # --------------------------------------------------------------
    phase1_time = min(20.0, 0.07 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0],
        eta0=0.4,
        steps_per_alpha=8000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] phase‑1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Phase‑2 – projected sub‑gradient descent (bulk of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve about 25 % of the remaining budget for the final refinement
        phase2_time = max(30.0, 0.74 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.30,
            beta=0.93,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 improve: score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Phase‑3 – gradient‑guided local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_gradient_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10**7,
            eps_frac=0.018,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise and output
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Return a plain Python list – the caller will feed it to evaluate_sequence
    return [float(x) for x in best_seq]
