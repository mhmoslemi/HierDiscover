"""
Optimiser for the step‑function bound.

The idea is to start from a simple parametric “U‑shaped” family of
distributions (a discrete Beta density with exponents < 1).  These
distributions are already close to flat self‑convolution, so a short
grid search over the shape parameter together with a convex combination
with the uniform distribution gives a good initial point.

From that point a fast projected sub‑gradient descent on the true
objective  maxₖ(a⋆a)[k]  is performed.  The sub‑gradient is obtained from
the active (almost‑maximal) convolution indices, the iterate is
projected onto the probability simplex and the step size decays as
1/√t.  Finally a brief simulated‑annealing phase moves a small amount of
mass between two indices to escape possible plateaus.

All heavy operations (convolutions, gradients) use FFT‑based O(n log n)
code, so thousands of iterations easily fit into the 1000 s budget.
The routine returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, simplex projection, scoring)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """Score 2·n·max(conv) without the overhead of evaluate_sequence."""
    n = a.shape[0]
    max_conv = _conv_self_fft(a).max()
    return 2.0 * n * max_conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns a sub‑gradient of f(a)=max_k (a*a)[k] and the current max value.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    cur_max = conv.max()
    # indices that are (practically) maximal
    rel_tol = 1e-12
    thresh = cur_max * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # ordered contributions (diagonal counted twice automatically)
        g[i] += a[j]
        g[j] += a[i]
    return g, cur_max


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgrad_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent with step size  step0/√t .
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        g, _ = _subgrad_max(a)
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * g)

        # evaluate current score
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: it={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-7,
                   max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    score = _fast_score(a)
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # linear cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # source weighted by current mass, destination uniform
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * rng.random() * 0.25  # move up to 25 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _fast_score(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: it={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Parametric “U‑shaped” families (Beta‑type densities)
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta(α,β) density on the discrete grid {0,…,n‑1}.
    Returns a probability vector (sum = 1, all entries ≥ 0).
    """
    x = (np.arange(n) + 0.5) / n                # points in (0,1)
    # avoid overflow/under‑flow by using logs
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    # numerical safety
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals = np.full(n, 1.0 / n)
    return vals

# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The strategy:
      1.  Coarse grid over a symmetric Beta family (α,β) and a convex
          combination with the uniform distribution.
      2.  Local random walk on (α,β) to fine‑tune the shape.
      3.  Projected sub‑gradient descent on the exact max‑convolution.
      4.  A brief simulated‑annealing polishing phase.
    All steps are time‑aware and stop before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – 4800 is a good compromise between quality and speed
    n = 4800

    # ------------------------------------------------------------------
    # 1. Coarse grid search over (α,β) and mixing weight λ
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = _fast_score(best_seq)
    best_alpha = 0.5
    best_beta = 0.5
    best_lambda = 0.0  # 0 → pure uniform, 1 → pure Beta

    alphas = np.linspace(0.2, 1.2, 11)   # 0.2 … 1.2
    betas = np.linspace(0.2, 1.2, 11)
    lambdas = np.linspace(0.0, 1.0, 6)   # mixing fractions

    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                break
            beta_vec = _beta_shape(n, a, b)
            for lam in lambdas:
                # convex combination with the uniform distribution
                seq = (1.0 - lam) * (1.0 / n) + lam * beta_vec
                # re‑normalise (numerical drift)
                seq = np.maximum(seq, 0.0)
                seq /= seq.sum()
                sc = _fast_score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_alpha, best_beta, best_lambda = a, b, lam
                    print(f"[{time.time() - start_time:6.2f}s] grid improve: α={a:.3f}, β={b:.3f}, λ={lam:.2f}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Local random walk on (α,β,λ) to refine the shape
    # ------------------------------------------------------------------
    cur_alpha, cur_beta, cur_lambda = best_alpha, best_beta, best_lambda
    rng = np.random.default_rng()

    while time.time() < deadline:
        # shrinking Gaussian proposals
        remaining = deadline - time.time()
        sigma = max(0.02, (remaining / budget_s) * 0.2)   # 0.2 → initial width
        cand_alpha = np.clip(cur_alpha + rng.normal(scale=sigma), 0.05, 2.0)
        cand_beta = np.clip(cur_beta + rng.normal(scale=sigma), 0.05, 2.0)
        cand_lambda = np.clip(cur_lambda + rng.normal(scale=0.1), 0.0, 1.0)

        beta_vec = _beta_shape(n, cand_alpha, cand_beta)
        seq = (1.0 - cand_lambda) * (1.0 / n) + cand_lambda * beta_vec
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = _fast_score(seq)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            cur_alpha, cur_beta, cur_lambda = cand_alpha, cand_beta, cand_lambda
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: α={cand_alpha:.3f}, β={cand_beta:.3f}, λ={cand_lambda:.2f}, score={sc:.6f}")
        # simple “accept‑if‑better” walk
        else:
            # occasional acceptance of worse proposals to avoid trivial stagnation
            if rng.random() < 0.03:
                cur_alpha, cur_beta, cur_lambda = cand_alpha, cand_beta, cand_lambda

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the true L∞ objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly half of the remaining time to polishing
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.45 * remaining
        polished_seq, polished_score = _run_subgrad_descent(
            best_seq,
            subgrad_deadline,
            start_time,
            step0=0.6,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad polishing improve: score={polished_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing polishing (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        final_seq, final_score = _run_annealing(
            best_seq,
            anneal_deadline,
            start_time,
            start_temp=4e-2,
            end_temp=1e-7,
            max_steps=200_000,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time (guard against numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Report the true score using the provided evaluate_sequence
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # return a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
