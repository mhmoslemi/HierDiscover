    n = a.shape[0]
    conv = _conv_self(a)
    target = 1.0 / (2 * n - 1)  # since sum a = 1
    error = conv - target
    rev_a = a[::-1]

    full = np.fft.irfft(A * E, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]

# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1} using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _mse_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the MSE loss  ∑_k (conv_k – μ)^2  w.r.t. a,
    where μ = 1/(2n‑1) is the target constant (for a probability vector).
    """
    n = a.shape[0]
    target = 1.0 / (2 * n - 1)
    err = conv - target                # length 2n‑1
    rev_a = a[::-1]                    # length n
    m = rev_a.shape[0] + err.shape[0] - 1
    fft_len = 1 << (m.bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    E = np.fft.rfft(err, n=fft_len)
    full = np.fft.irfft(A * E, n=fft_len)[: n + err.shape[0] - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]  # central n entries
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        rel_eps: float = 0.01,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # ---- active set -------------------------------------------------
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:               # safety fallback
            active = np.array([conv.argmax()], dtype=int)

        # weight vector (average over active indices)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        m = rev_a.shape[0] + w.shape[0] - 1
        fft_len = 1 << (m.bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + w.shape[0] - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a tiny fraction of mass; occasional local smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:                     # simple mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25       # transfer ≤25 % of the source mass
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # occasional local smoothing
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method first flattens the self‑convolution via an MSE‑Adam phase,
    then refines the maximum with sub‑gradient descent and finishes with a
    simulated‑annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # Determine a working dimension (a compromise between quality and speed).
    # ------------------------------------------------------------------
    n = 9600                                       # final resolution
    # Try to reuse a previously discovered construction if available.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample / down‑sample to the desired length and normalise.
            if starter.size != n:
                old_x = np.linspace(0.0, 1.0, starter.size)
                new_x = np.linspace(0.0, 1.0, n)
                starter = np.interp(new_x, old_x, starter)
            starter /= starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    if starter is not None:
        a = starter.copy()
    else:
        a = np.full(n, 1.0 / n)                    # start from the uniform distribution

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    # ------------------------------------------------------------------
    # Phase 1 – flatten the convolution (MSE) using Adam.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.2                                   # base learning rate
    t = 0
    flatten_deadline = start_time + (deadline - start_time) * 0.45

    while time.time() < flatten_deadline:
        t += 1
        conv = _conv_self(a)
        grad = _mse_gradient(a, conv)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

        # occasional score check
        if t % 200 == 0:
            cur_score = 2.0 * n * _conv_self(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] flatten improve: "
                      f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max‑convolution.
    # ------------------------------------------------------------------
    subgrad_deadline = start_time + (deadline - start_time) * 0.80
    a_sub, sub_score = _run_subgradient_gd(
        best_a, n, subgrad_deadline, start_time,
        step0=0.12, beta=0.9, rel_eps=0.008, max_iter=10 ** 7
    )
    if sub_score < best_score:
        best_score = sub_score
        best_a = a_sub.copy()
        print(f"[{time.time() - start_time:6.2f}s] subgrad final improve: "
              f"score={sub_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, ann_score = _run_annealing(
            best_a, n,
            deadline, start_time,
            best_score, best_a,
            start_temp=0.25, end_temp=1e-7,
            steps=8_000_000
        )
        if ann_score < best_score:
            best_score = ann_score
            best_a = a_ann.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal final improve: "
                  f"score={ann_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation and conversion to plain list.
    # ------------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    return [float(x) for x in best_a]
