# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent on max‑convolution (projected Nesterov)
# ----------------------------------------------------------------------
def _run_subgradient_gd(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.5,
    beta: float = 0.5,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_max = conv[idx_max]
        cur_score = 2.0 * n * cur_max  # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                f"iter={it}, score={cur_score:.6f}"
            )

        # sub‑gradient of the active shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # momentum & step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming correction (active‑set)
# ----------------------------------------------------------------------
def _lp_correction(
    a: np.ndarray,
    M: float,
    active_shifts: np.ndarray,
    upper_bound: float = 10.0,
) -> np.ndarray:
    """
    Solve a tiny LP:
        maximise   sum(b)
        subject to A_active @ b <= M
                    0 <= b_i <= upper_bound
        where A_active[j,i] = a_i if the pair (i, j-i) is valid.
    Returns the LP solution (projected onto the simplex; if LP fails,
    returns the original a).
    """
    n = a.shape[0]
    K = active_shifts.size
    if K == 0:
        return a

    # Build dense A matrix (K × n)
    A = np.zeros((K, n), dtype=float)
    for row, j in enumerate(active_shifts):
        i_start = max(0, j - (n - 1))
        i_end = min(j, n - 1)
        # the contribution of a[i] to conv[j] for each i in the slice
        A[row, i_start : i_end + 1] = a[i_start : i_end + 1]

    bounds = [(0.0, upper_bound) for _ in range(n)]
    c = -np.ones(n)  # maximise sum(b)  <=> minimise -sum(b)

    res = linprog(
        c,
        A_ub=A,
        b_ub=np.full(K, M),
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return a  # fall back to the current point

    b = res.x
    if b.sum() > 0:
        b = b / b.sum()
    else:
        b = np.full_like(b, 1.0 / n)
    return b


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Mass‑shifting walk that reduces the maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: pick proportionally to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: favour low‑contribution indices
        low_thr = np.percentile(contrib, 20)  # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3  # move up to 30 % of donor
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] local improve: n={n}, "
                    f"iter={it}, score={new_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine consists of:
      * warm‑start & progressive up‑sampling,
      * sub‑gradient descent on the true max‑convolution,
      * a few active‑set LP corrections,
      * and a final peak‑shaving local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Progressive up‑sampling stages
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]

    # allocate roughly 60 % of the total budget to the gradient stage
    total_time = deadline - start_time
    grad_budget = total_time * 0.60
    lp_budget   = total_time * 0.25
    local_budget = total_time - grad_budget - lp_budget

    # keep track of what has been spent
    grad_spent = 0.0
    lp_spent = 0.0

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time left for this stage
        remaining = deadline - time.time()
        # a proportion of the remaining gradient budget
        stage_grad_time = max(1.0, grad_budget * (n / sum(stage_lengths)))
        stage_deadline = time.time() + min(stage_grad_time, remaining * 0.6)

        # initialise a0 for this length
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Sub‑gradient descent (main optimisation phase)
        # --------------------------------------------------------------
        a_opt, score_opt = _run_subgradient_gd(
            a0,
            n,
            stage_deadline,
            start_time,
            step0=0.5,
            beta=0.5,
            max_iter=10 ** 7,
        )
        grad_spent += time.time() - start_time - grad_spent - lp_spent

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage n={n} subgrad improve: "
                f"score={score_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Active‑set LP correction (quick refinements)
        # --------------------------------------------------------------
        # give each stage a small slice of the LP budget
        lp_stage_deadline = time.time() + min(lp_budget * (n / sum(stage_lengths)), remaining * 0.3)
        a_lp = best_seq.copy()
        it_lp = 0
        while time.time() < lp_stage_deadline and it_lp < 8:
            it_lp += 1
            conv = _conv_self(a_lp)
            M = conv.max()
            # pick the K most critical shifts (K between 10 and 30)
            K = min(20, conv.size)
            active = np.argsort(-conv)[:K]
            b = _lp_correction(a_lp, M, active, upper_bound=10.0)
            # mix a little bit (step size gradually decreasing)
            t = 0.3 / (it_lp + 1.0)
            a_new = (1.0 - t) * a_lp + t * b
            a_new = _project_simplex(a_new)

            cur_score = 2.0 * n * _conv_self(a_new).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] LP improve: n={n}, "
                    f"iter={it_lp}, score={cur_score:.6f}"
                )
            a_lp = a_new

        lp_spent += time.time() - start_time - grad_spent - lp_spent

    # ------------------------------------------------------------------
    # Targeted local search with the remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final local improve: "
                f"score={score_loc:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (the harness expects a list)
    return [float(x) for x in best_seq]
