"""
LP+Gradient optimization for minimizing max self-convolution.
Stage 0: solve a series of linear programs on a modest size (n≈1024)
         to find a low‑peak shape.
Stage 1: up‑sample to the target size (≈8192) and polish with
         projected sub‑gradient descent and a light simulated‑annealing.

import time, math, numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

def _conv_self(a):
    # same as before using FFT
    n = a.shape[0]
    fft_len = 1 << ((2*n-2).bit_length())  # next pow2 >= 2n-1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv

def _build_conv_matrix_sparse(f):
    n = f.shape[0]
    # create data with shape (n, n) but stored as list of diagonals
    # each diagonal j with value f[j] repeated n times
    data = [np.full(n, f_j) for f_j in f]
    offsets = list(range(n))
    # shape (2*n-1, n)
    m = 2*n - 1
    A = diags(data, offsets, shape=(m, n), format='csr')
    return A

def _subgradient_max(a, conv, rel_tol=1e-12):
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

def _run_subgradient_descent(a0, deadline, start_time, max_iter=10**7, beta=0.94, step0=0.13):
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad iter {it:,} score {cur_score:.6f}")
        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # project onto simplex
        # Euclidean projection onto simplex
        v = a - np.max(a)  # shift for numeric stability
        # Use standard algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(u)) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.where(cond)[0][-1]
            theta = cssv[rho] / (rho + 1)
            a = np.maximum(v - theta, 0)
        else:
            a = np.full_like(v, 1.0 / len(v))
        # ensure sum=1; due to projection already ensures that
    return best_a, best_score

def _run_annealing(a0, deadline, start_time, start_temp=2e-2, end_temp=1e-7, steps=200_000):
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()
    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        # choose source proportional to mass
        src = rng.choice(n, p=a)  # a sums to 1
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv_new = _conv_self(a_prop)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] anneal iter {it:,} best {best_score:.6f}")
    return best_seq, best_score

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

"""
LP + sub‑gradient optimisation for the “max self‑convolution” problem.

Stage 0 – small‑scale linear programmes (n≈1024).  Starting from a random
          simplex vector (or a down‑sampled known good solution) we repeatedly

          maximise Σ b_j  subject to  b_j≥0  and  (a * b)_k ≤ M,
          where M=‖a*a‖_∞.  The solution b is normalised back to the simplex
          and mixed with a (a←(1‑t)a+tb).  This quickly drives the peak
          of the self‑convolution down.

Stage 1 – linear interpolation of the best small‑scale vector to the target
          length (≈8192).

Stage 2 – projected sub‑gradient descent on the true objective
          F(a)=2·n·max(conv(a)).  The sub‑gradient is the averaged gradient
          over all indices where the maximum is attained.

Stage 3 – a very light simulated‑annealing “mass‑transfer” move to escape
          any remaining tiny traps.

All stages respect the overall time budget.  The final sequence is projected
onto the probability simplex, guaranteeing non‑negative entries and unit sum.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the LP stage)
# ----------------------------------------------------------------------
try:
    from scipy.sparse import diags          # type: ignore
    from scipy.optimize import linprog      # type: ignore
    _HAVE_SCIPY = True
except Exception:  # pragma: no cover
    _HAVE_SCIPY = False


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, length‑2 n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Build the (2 n‑1) × n convolution matrix for a given vector a.
# The matrix implements conv(a, x) = A @ x.
# ----------------------------------------------------------------------
def _build_conv_matrix_sparse(a: np.ndarray):
    """Sparse convolution matrix (shape (2n‑1, n))."""
    n = a.shape[0]
    # each diagonal j (offset j) contains the constant a[j]
    data = [np.full(n, a_j) for a_j in a]
    offsets = list(range(n))
    m = 2 * n - 1
    A = diags(data, offsets, shape=(m, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the polishing stage)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a≥0, Σa=1)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* objective
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.94,
                             step0: float = 0.13,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on  F(a)=2·n·max(conv(a))."""
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it:,} "
                  f"score {cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)            # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter {it:,} "
                  f"best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _downsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to a **smaller** length `new_n`."""
    return _upsample(seq, new_n)

# Main entry point required by the harness
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # Stage 0 – LP refinement on a modest problem size (n≈1024)
    # ------------------------------------------------------------------
    n_small = 1024
    best_small = None

    # Try to obtain a starter from the global variable if present
    starter = None

            starter = None

                a = np.random.rand(n_small)

                # build the sparse convolution matrix (2n‑1)×n

    # ------------------------------------------------------------------
