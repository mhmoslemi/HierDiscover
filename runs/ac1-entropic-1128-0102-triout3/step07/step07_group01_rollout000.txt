"""
A multi‑stage optimizer for the “step‑function” challenge.

The routine builds on the three‑phase approach that produced a score
≈1.5125, but adds a few orthogonal ideas that often push the bound a
bit lower:

* **Symmetry enforcement** – the optimal sequence is essentially
  symmetric; after each update we mirror the vector which reduces the
  search space and helps the optimizer stay near good candidates.

* **Finer exponential‑gradient schedule** – we raise the soft‑max
  temperature (α) up to 400, giving a tighter approximation of the
  true max and a sharper descent near the end of the warm‑up phase.

* **Adaptive projected sub‑gradient descent** – a Nesterov‑style
  momentum term with a step‑size that decays as 1/√t, together with a
  projection onto the probability simplex.  The sub‑gradient is
  computed only on the “active” indices whose convolution value lies
  within a relative tolerance of the maximum.

* **Simulated‑annealing local search** – random mass‑exchange moves
  (source → destination) are proposed.  Moves that strictly improve
  the maximal self‑convolution are always accepted; otherwise they
  are accepted with a temperature‑controlled probability.  The
  temperature decays, allowing the algorithm to escape shallow plateaus
  early on while performing fine‑grained refinement later.

* **Targeted refinement** – as in the reference code, a final pass
  performs tiny deterministic moves away from the indices that
  contribute most to the current maximum.

The whole pipeline respects a 1000 s wall‑clock budget, logs progress,
and finally returns a normalized (sum = 1) non‑negative list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Convolution of a with itself (full, length 2·len(a)−1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a ← (a + a[::-1]) / 2."""
    return (a + a[::-1]) * 0.5


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` to length `new_n` while preserving total mass.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated gradient on a smoothed max
# ----------------------------------------------------------------------


def _exp_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    alpha_sched,
    eta0: float = 0.3,
    steps_per_alpha: int = 4000,
):
    """
    Multiplicative updates using a soft‑max approximation of the max.
    Returns the best vector seen and its (unnormalised) score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)  # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_sched)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}"
            )

        alpha = alpha_sched[min(alpha_idx, n_alpha - 1)]

        # ----- smooth‑max gradient -----
        # w_i = exp(alpha * conv_i) / sum_j exp(alpha * conv_j)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * a * (w reversed) convolved
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + w_rev.size - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ----- multiplicative update -----
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        # normalise to simplex and symmetrise
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n
        a = _symmetrize(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Phase 2 – Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _subgradient_full(a: np.ndarray, conv: np.ndarray, rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose value lies
    within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _projected_subgradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.3,
    beta: float = 0.92,
):
    """
    Standard sub‑gradient descent with momentum, projection onto the simplex,
    and compulsory symmetrisation after each step.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}"
            )

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
        a = _symmetrize(a)

    return best_a, best_score

def _local_mass_exchange(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_eps: float = 0.04,
    final_eps: float = 0.003,
    init_temp: float = 1e-3,
    final_temp: float = 1e-6,
    max_iters: int = 10 ** 7,
):
    """
    Randomly moves a fraction of mass from a source index to a destination.
    Moves that lower the maximum convolution are always accepted; otherwise
    they are accepted with probability exp(-(Δ)/T).  Both the step size
    (eps) and temperature decay exponentially.
    Returns the best vector observed.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    it = 0

    # geometric decay of eps and temperature
    total_time = deadline - start_time
    eps = init_eps
    temp = init_temp

    while time.time() < deadline and it < max_iters:
        it += 1
        # linearly map progress to [0,1] for exponential decay
        elapsed = time.time() - start_time
        prog = elapsed / total_time
        eps = init_eps * (final_eps / init_eps) ** prog
        temp = init_temp * (final_temp / init_temp) ** prog

        # ---- pick a source weighted by its contribution to the current max ----
        k_max = int(np.argmax(conv))
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # destination: uniform over all other indices
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # amount to move
        delta = a[src] * rng.random() * eps
        if delta <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        a_prop = _symmetrize(a_prop)

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        delta_max = new_max - cur_max

        accept = False
        if delta_max < -1e-12:
            accept = True
        else:
            # Metropolis acceptance
            if temp > 0:
                prob = math.exp(-delta_max / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] local exchange improve: iter={it:,}, max={best_max:.8f}"
                )

    return best_a


# ----------------------------------------------------------------------
# Phase 4 – Targeted tiny‑mass refinement (as in the reference code)
# ----------------------------------------------------------------------


def _targeted_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    max_iters: int = 10 ** 7,
    eps_frac: float = 0.018,
):
    """
    Moves a tiny fraction of probability mass away from coordinates that
    contribute most to the current maximal convolution entry.  Only moves
    that strictly lower the maximum are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        k_max = int(np.argmax(conv))

        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
        else:
            prob_src = a / a.sum()

        src = rng.choice(n, p=prob_src)
        if a[src] < 1e-14:
            continue

        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _symmetrize(a_prop)

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] refine improve: iter={it:,}, max={best_max:.8f}"
                )

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # length of the final candidate – 4800 worked well for the reference
    final_n = 4800

    # try to start from the globally‑available construction if present
    starter = globals().get("height_sequence_1", None)
    if isinstance(starter, (list, tuple, np.ndarray)):
        try:
            starter = np.asarray(starter, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = starter / starter.sum()
                a0 = _upsample(starter, final_n)
        except Exception:
            a0 = None
    else:
        a0 = None

    # fallback: random symmetric Dirichlet
    if a0 is None:
        half = final_n // 2 + 1
        rand = np.random.rand(half)
        rand = rand / rand.sum()
        a0 = np.concatenate([rand, rand[-2 if final_n % 2 == 0 else -1 : 0 : -1]])
        a0 = _symmetrize(a0)

    a0 = _symmetrize(a0)
    a0 = a0 / a0.sum()  # enforce exact simplex

    best_seq = a0.copy()
    best_score = 2.0 * final_n * _conv_self(a0).max()
    print(f"[{time.time() - start_time:.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exponentiated gradient warm‑up (≈10 % of time)
    # ------------------------------------------------------------------
    phase1_time = min(20.0, 0.08 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time
    a_warm, score_warm = _exp_gradient(
        a0,
        final_n,
        phase1_deadline,
        start_time,
        alpha_sched=[0.5, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0],
        eta0=0.3,
        steps_per_alpha=4000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] after phase‑1: score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈70 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_time = max(30.0, 0.70 * remaining)
        phase2_deadline = time.time() + phase2_time

        a_sub, score_sub = _projected_subgradient(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.3,
            beta=0.92,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] after phase‑2: score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing local mass‑exchange search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_local = _local_mass_exchange(
            best_seq,
            final_n,
            deadline,
            start_time,
            init_eps=0.04,
            final_eps=0.003,
            init_temp=1e-3,
            final_temp=1e-6,
            max_iters=10 ** 7,
        )
        conv_local = _conv_self(a_local)
        score_local = 2.0 * final_n * conv_local.max()
        if score_local < best_score:
            best_score = score_local
            best_seq = a_local.copy()
            print(f"[{time.time() - start_time:.2f}s] after phase‑3: score={score_local:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted tiny‑mass refinement (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10 ** 7,
            eps_frac=0.018,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and output the final candidate
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
