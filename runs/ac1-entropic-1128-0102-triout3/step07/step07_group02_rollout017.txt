def _run_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    it = 0
    start_temp = 0.5
    end_temp = 1e-6
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        idx_max = int(np.argmax(conv))
        # Compute contributions for each i in the max convolution shift
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        if contrib.sum() == 0:
            donor = np.random.randint(n)
        else:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        # Choose recipient randomly from low contribution indices
        low = np.where(contrib <= np.percentile(contrib, 30))[0]
        if low.size:
            rec = np.random.choice(low)
        else:
            rec = np.random.randint(n)
        if donor == rec or a[donor] <= 1e-12:
            continue
        eps = a[donor] * np.random.rand() * 0.2
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[rec] += eps
        new_conv = _conv_self(a_new)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = (delta < 0) or (np.random.rand() < np.exp(-delta / (T+1e-12)))
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

"""
Propose a step–function (non‑negative sequence) that minimizes
``evaluate_sequence``.

The algorithm consists of several independent restarts, each restart
contains three phases:

1. **Smooth Adam optimisation** – minimise a temperature‑softened
   approximation of the max‑convolution using Adam updates and Euclidean
   projection onto the probability simplex.

2. **Exact sub‑gradient on the true max** – compute a sub‑gradient that
   takes *all* (almost) active convolution entries into account
   (instead of a single index).  A simple momentum scheme is used.

3. **Targeted peak‑shaving local search** – repeatedly move a small
   amount of mass from indices that contribute most to the current
   maximal convolution entry to low‑contribution indices.  An annealed
   Metropolis acceptance rule allows occasional uphill moves.

All phases respect the global time budget (default 1000 s).  The best
candidate seen over all restarts is returned as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear convolution helpers (FFT based, full length)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of reversed a with weight vector w (both real)."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights for a given temperature
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Return a probability vector w = softmax(conv / temperature)."""
    # shift for numerical stability
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (2 * aᵀ J_k a)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the smooth‑max objective."""
    w = _softmax_weights(conv, temperature)
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)               # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1 : n - 1 + n]             # central block
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lr0: float = 0.12,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     T_start: float = 2e-2,
                     T_end: float = 5e-4,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam with temperature‑softened max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam smooth improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (geometric)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam state update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (all active shifts)
# ----------------------------------------------------------------------
def _run_subgrad_active(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        lr0: float = 0.5,
                        beta: float = 0.9,
                        tol: float = 1e-12,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent using the full active set of maximal convolution entries."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad active improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        max_val = conv.max()
        active = np.where(conv >= max_val - tol)[0]
        # uniform weight over the active set
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        # sub‑gradient = 2 * (rev_a * w) restricted to the original length
        rev_a = a[::-1]
        full = _conv_rev_weight(rev_a, w)
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = lr0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted peak‑shaving local search (simulated annealing)
# ----------------------------------------------------------------------
def _run_local_search(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from high‑contributing to low‑contributing indices."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 30)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size:
            receiver = np.random.choice(low_cands)
        else:
            receiver = np.random.randint(n)

        if donor == receiver or a[donor] <= 1e-15:
            continue

        eps = a[donor] * np.random.rand() * 0.25
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # linear temperature decay
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12))

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _initial_vector(n: int, kind: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting probability vector of length n."""
    if kind == 0:                       # uniform
        return np.full(n, 1.0 / n)
    if kind == 1:                       # Dirichlet‑1 (random simplex point)
        v = rng.gamma(1.0, 1.0, size=n)
        v /= v.sum()
        return v
    # kind == 2: linear ramp (more mass at the beginning)
    ramp = np.arange(1, n + 1, dtype=float)
    ramp /= ramp.sum()
    return ramp


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine performs several restarts,
    each consisting of (i) Adam optimisation on a soft‑max surrogate,
    (ii) exact sub‑gradient descent with a full active set and
    (iii) a targeted local‑search polishing step.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine the final length – largest length gives the smallest possible bound.
    # ------------------------------------------------------------------
    final_n = 19200                     # same as the previous best construction
    # A few intermediate sizes help the optimiser to find a good shape.
    stage_lengths = [4800, 9600, final_n]

    # ------------------------------------------------------------------
    # Warm‑start (previous best) if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")
    restart = 0
    max_restarts = 6

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline and restart < max_restarts:
        # allocate a roughly equal time slice for each restart
        remaining_time = deadline - time.time()
        time_for_this = remaining_time / (max_restarts - restart)
        restart_deadline = time.time() + time_for_this

        # --------------------------------------------------------------
        # 1) Build an initial vector (may be warm‑started) and upsample gradually
        # --------------------------------------------------------------
        if restart == 0 and starter is not None:
            a = _upsample(starter, stage_lengths[0])
        else:
            a = _initial_vector(stage_lengths[0], restart % 3, rng)

        # --------------------------------------------------------------
        # 2) Run the three optimisation phases on each intermediate size
        # --------------------------------------------------------------
        for n in stage_lengths:
            if time.time() >= restart_deadline:
                break

            # phase A – smooth Adam
            phase_deadline = time.time() + (restart_deadline - time.time()) * 0.45
            a_opt, s_opt = _run_adam_smooth(
                a, n, phase_deadline, start_time,
                lr0=0.12, beta1=0.9, beta2=0.999,
                T_start=2e-2, T_end=5e-4, max_iter=10 ** 7
            )
            a = a_opt
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart {restart} phase A improve: "
                      f"n={n}, score={s_opt:.6f}")

            # phase B – exact sub‑gradient with full active set
            phase_deadline = time.time() + (restart_deadline - time.time()) * 0.35
            a_opt, s_opt = _run_subgrad_active(
                a, n, phase_deadline, start_time,
                lr0=0.5, beta=0.9, max_iter=10 ** 7
            )
            a = a_opt
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart {restart} phase B improve: "
                      f"n={n}, score={s_opt:.6f}")

            # phase C – local peak‑shaving
            phase_deadline = time.time() + (restart_deadline - time.time()) * 0.20
            a_opt, s_opt = _run_local_search(
                a, n, phase_deadline, start_time
            )
            a = a_opt
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] restart {restart} phase C improve: "
                      f"n={n}, score={s_opt:.6f}")

            # Upsample to the next stage (if any)
            nxt_idx = stage_lengths.index(n) + 1
            if nxt_idx < len(stage_lengths):
                a = _upsample(a, stage_lengths[nxt_idx])

        restart += 1

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
