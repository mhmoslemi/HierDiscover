def _lp_refine(a, n, t, active_ratio, max_iter):
    best_a = a.copy()
    best_score = objective(a)
    for it in range(max_iter):
        conv_self = _conv_self(a)
        M = conv_self.max()
        # Determine active constraints
        active_idx = np.where(conv_self >= M * active_ratio)[0]
        if len(active_idx) == 0:
            break
        # Build A_ub: each row k maps g to conv(f,g)[k]
        rows = []
        for k in active_idx:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            i = np.arange(lo, hi+1)  # indices of a
            j = k - i  # indices of g
            row = np.zeros(n)
            row[j] = a[i]
            rows.append(row)
        A_ub = np.vstack(rows)  # shape (m, n)
        b_ub = np.full(len(active_idx), M)
        # Solve LP
        res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        g_norm = g / sum_g
        # Update f
        a_new = (1 - t) * a + t * g_norm
        # Evaluate new objective
        new_score = evaluate_obj(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new
        # optionally adjust t
    return best_a
```

def _run_lp_refinement(a0: np.ndarray, n: int, deadline: float, start_time: float,
                       max_iters: int = 2000,
                       active_ratio: float = 0.999,
                       t: float = 0.1) -> np.ndarray:
    """
    Alternating linear-program refinement.
    At each iteration solve a small LP (max sum(g) s.t. convolution with a <= M)
    using only constraints that are close to the current maximum.
    Then move a fraction t toward the normalized LP solution.
    """
    from scipy.optimize import linprog
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv_self = _conv_self(a)
        M = conv_self.max()
        # active constraints
        active = np.where(conv_self >= M * active_ratio)[0]
        # cap number of constraints for speed
        max_cons = 150
        if active.size > max_cons:
            # take the top values
            idx = np.argpartition(conv_self, -max_cons)[-max_cons:]
            active = idx
        # Build A_ub matrix
        m = active.size
        if m == 0:
            break
        rows = np.zeros((m, n))
        for row_i, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            rows[row_i, j] = a[i]
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=float)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=rows, b_ub=np.full(m, M), bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # convex combination
        a = (1 - t) * a + t * g_norm
        # optional slight re-normalisation to avoid drift
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        # evaluate
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {it:,} improve: score={cur_score:.6f}")
        # optionally decay t
        if it % 200 == 0:
            t = max(t * 0.9, 0.01)
    return best_a

"""
Optimisation of a non‑negative step function for the objective

    E(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm combines three ideas:

1. A short exponential‑gradient warm‑up on a smooth approximation of
   the max‑convolution (log‑sum‑exp).
2. Exact projected sub‑gradient descent on the true max‑convolution.
3. An *active‑set linear‑program* refinement:
   for the current vector f we solve
        max   Σ g_i
        s.t.  (f * g)_k ≤ M   ∀ k in an active set,
              g_i ≥ 0,
   where M = max(f*f).  The result g is normalised to a probability
   vector and we move the current iterate toward it.
   Only constraints that are within a tiny relative tolerance of the
   current maximum are kept, hence each LP is tiny and can be solved
   thousands of times in the remaining budget.
4. Finally a targeted local‑refinement that moves a small amount of
   probability mass away from the coordinates that contribute most to
   the current maximal convolution entry (the “refine” phase of the
   reference code).

All stages respect the 1000 s time budget and keep the best sequence
found.  The returned list is non‑negative and automatically normalised
to sum = 1 (the evaluation function is homogeneous, any scaling is fine).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # eliminate tiny negative round‑off errors
    conv[conv < 0] = 0.0
    return conv


def _conv_cross(f: np.ndarray, g: np.ndarray) -> np.ndarray:
    """Cross‑convolution f * g (length 2·len(f)‑1)."""
    n = f.shape[0]
    L = _next_pow2(2 * n - 1)
    Ff = np.fft.rfft(f, n=L)
    Fg = np.fft.rfft(g, n=L)
    conv = np.fft.irfft(Ff * Fg, n=L)[: 2 * n - 1]
    conv[conv < 0] = 0.0
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence (up to the security clipping)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on exact max
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : active‑set linear‑program refinement
# ----------------------------------------------------------------------
def _run_lp_refinement(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       max_iters: int = 10000,
                       active_ratio: float = 0.999,
                       t: float = 0.1) -> np.ndarray:
    """
    Alternating LP refinement.
    We maximise Σ g_i under the active constraints
        (a * g)_k ≤ M
    where M = max(a*a).  A tiny convex combination with step size `t`
    guarantees monotone improvement while the LP stays tiny (≤~150 rows).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        M = conv.max()

        # active constraints – those near the current maximum
        active = np.where(conv >= M * active_ratio)[0]
        # keep the number of rows manageable
        max_rows = 150
        if active.size > max_rows:
            # pick the `max_rows` largest values
            idx = np.argpartition(conv, -max_rows)[-max_rows:]
            active = idx

        m = active.size
        if m == 0:
            break

        # build A_ub where each row implements (a * g)_k ≤ M
        rows = np.zeros((m, n), dtype=np.float64)
        for row_i, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            rows[row_i, j] = a[i]

        # linear programme: maximise Σ g_i  ⟺ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=rows, b_ub=np.full(m, M),
                      bounds=bounds, method='highs')
        if not res.success:
            # if the LP fails for any reason we stop the refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # move a little towards the LP solution
        a = (1 - t) * a + t * g_norm
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {it:,} improve: score={cur_score:.6f}")

        # slowly decay the step size so that we converge
        if it % 200 == 0:
            t = max(t * 0.9, 0.01)

    return best_a


# ----------------------------------------------------------------------
# Phase‑4 : targeted local refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _run_targeted_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 10**7,
                         eps_frac: float = 0.025):
    """
    Moves a small fraction of probability mass away from the coordinates that
    contribute most to the current maximal convolution entry.
    Only moves that strictly lower the maximal convolution value are kept.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        k_max = int(np.argmax(conv))

        # contribution of each index to this maximal entry
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            prob_src = a / a.sum()
        else:
            prob_src = contrib / total_contrib

        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # random destination different from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:       # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        if it % 50000 == 0 and time.time() >= deadline:
            break

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if it exists.
    # --------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.size != final_n:
                starter = _upsample(starter, final_n)
    except Exception:
        starter = None

    if starter is not None:
        a0 = starter.copy()
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = _objective(best_seq)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – short exponential‑gradient warm‑up
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + min(12.0, 0.07 * (deadline - time.time()))
        a_warm, score_warm = _run_exponential_gradient(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.4,
            steps_per_alpha=2000,
        )
        if score_warm < best_score:
            best_score, best_seq = score_warm, a_warm.copy()
            print(f"[{time.time() - start_time:.2f}s] after phase 1: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (major part of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remain = deadline - time.time()
        phase2_deadline = time.time() + max(30.0, 0.70 * remain)
        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score, best_seq = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] after phase 2: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – active‑set LP refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_lp = _run_lp_refinement(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=20000,
            active_ratio=0.999,
            t=0.1,
        )
        score_lp = _objective(a_lp)
        if score_lp < best_score:
            best_score, best_seq = score_lp, a_lp.copy()
            print(f"[{time.time() - start_time:.2f}s] after LP stage: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final targeted local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref = _run_targeted_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            max_iters=10**7,
            eps_frac=0.018,
        )
        score_ref = _objective(a_ref)
        if score_ref < best_score:
            best_score, best_seq = score_ref, a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[{time.time() - start_time:.2f}s] best final score = {best_score:.6f}")
    return [float(x) for x in best_seq]
