```python
import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

def _subgradient_full(a, conv, rel_tol=1e-12):
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g

def _project_simplex(v):
    """Euclidean projection onto simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0,1,old_n)
    new_x = np.linspace(0,1,new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0/new_n)
    return up
```

```python
def _run_exponential_gradient(a0, n, deadline, start_time, alpha_schedule, eta0=0.4, steps_per_alpha=6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3*n-2)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha-1)]

        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev)-1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1
    return best_a, best_score
```

```python
def _run_subgradient_descent(a0, n, deadline, start_time, step0=0.25, beta=0.94, max_iter=10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score
```

"""
Multi‑phase optimizer for the L∞ self‑convolution problem.

The algorithm improves on the reference implementation by
* adding an Adam‑style projected gradient phase (faster convergence),
* using a more aggressive targeted refinement that evaluates several
  destination candidates per move and shrinks the step size adaptively,
* performing a few independent restarts (different seeds) within the
  1000 s budget and keeping the best sequence found.

All helper functions are defined at module level; the entry point is
`propose_candidate`, which returns a list of non‑negative heights that
minimise `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp trick)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Adam‑style projected gradient descent
# ----------------------------------------------------------------------
def _run_adam_subgradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.1,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve: iter={t:,}, "
                  f"score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        step = step0 / math.sqrt(t)          # modest decay
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - update)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : aggressive targeted local refinement
# ----------------------------------------------------------------------
def _run_targeted_refine_improved(a0: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  start_time: float,
                                  init_eps_frac: float = 0.05,
                                  min_eps_frac: float = 5e-6,
                                  eps_decay: float = 0.90,
                                  max_iters: int = 10**7,
                                  tolerance: float = 1e-12,
                                  candidates_per_iter: int = 6):
    """
    Move a tiny amount of mass away from coordinates that contribute most
    to the current maximal convolution entry.  For each source we test a
    handful of destination indices (random, sparsest, opposite side) and
    keep the move that yields the greatest reduction of the max.  The
    step‑size (eps_frac) is shrunk whenever no improvement is found.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    eps_frac = init_eps_frac
    it = 0
    check_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1

        # index of the current (possibly non‑unique) maximum
        max_idxs = np.where(conv == max_val)[0]
        k_max = int(rng.choice(max_idxs)) if max_idxs.size > 1 else int(np.argmax(conv))

        # contribution of each position to this peak
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        prob_src = contrib / total_contrib if total_contrib > 0 else a / a.sum()
        src = rng.choice(n, p=prob_src)

        if a[src] < 1e-14:
            continue

        # build a small pool of destination candidates
        dest_set = set()
        while len(dest_set) < candidates_per_iter - 2:
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1
            dest_set.add(int(dst))
        # add the sparsest index
        min_idx = int(np.argmin(a))
        if min_idx != src:
            dest_set.add(min_idx)
        # add an index roughly opposite on the circle (to spread mass)
        far_idx = (src + n // 2) % n
        if far_idx != src:
            dest_set.add(far_idx)

        best_local_max = max_val
        best_move = None

        for dst in dest_set:
            eps = a[src] * rng.random() * eps_frac
            if eps <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            if a_prop[src] < -1e-15:          # safeguard against numerical drift
                continue
            conv_prop = _conv_self(a_prop)
            new_max = conv_prop.max()
            if new_max < best_local_max - tolerance:
                best_local_max = new_max
                best_move = (a_prop, conv_prop)

        if best_move is not None:               # accept the best candidate
            a, conv = best_move
            max_val = best_local_max
            if max_val < best_max - tolerance:
                best_max = max_val
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")
        else:
            eps_frac = max(eps_frac * eps_decay, min_eps_frac)

        if it % check_interval == 0 and time.time() >= deadline:
            break

    return best_a

        2. A long Adam‑style projected gradient descent.
        3. An aggressive targeted refinement.
    Several independent restarts are performed; the best sequence overall
    is returned.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    final_n = 4800

    # ------------------------------------------------------------------
    # Initialise – use the global reference if present.
    # ------------------------------------------------------------------
    try:

    else:

        a0 /= a0.sum()

    best_score = float("inf")

    # ------------------------------------------------------------------
    # Perform a few independent runs (different RNG seeds)
    # ------------------------------------------------------------------
    for run_id, run_seed in enumerate([seed, seed + 1, seed + 2]):

        rng = np.random.default_rng(run_seed)
        # jitter the start to obtain a distinct trajectory
        a_start = best_seq.copy() + rng.random(final_n) * 1e-6
        a_start = _project_simplex(a_start)

        # ------------------- Phase‑1 : warm‑up -------------------
        remaining = deadline - time.time()
        warm_time = min(12.0, 0.07 * remaining)
        warm_deadline = time.time() + warm_time

        a_warm, score_warm = _run_exponential_gradient(
            a_start,

        if score_warm < best_score:
