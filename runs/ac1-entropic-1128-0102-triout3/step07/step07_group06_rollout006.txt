"""
Search for a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The algorithm consists of four phases
respecting a 1000 s wall‑clock budget:

1. **Fast random‑parameter search** on a small vector (n≈1200).  Several
   simple families (power‑law, logistic, mixture of two Gaussian bumps)
   are sampled; the best candidate is kept and later up‑sampled.
2. **Soft‑max gradient descent** (a smooth surrogate of the max‑convolution)
   with Nesterov momentum, projected onto the probability simplex.
3. **Exact sub‑gradient descent** on the true max‑convolution.
4. **Extended simulated annealing** with targeted mass‑transfer moves,
   occasional local smoothing and a geometric cooling schedule.

The routine adapts the time spent on each phase, starts from the
previously discovered `height_sequence_1` if it exists, and finally
returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution (temperature‑smoothed).
    Returns a vector of length len(a).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # Use log‑sum‑exp trick for numerical stability
    e_shift = np.max(scaled)
    exps = np.exp(scaled - e_shift)
    w = exps / exps.sum()          # softmax weights (length 2n‑1)
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """Projected Nesterov‑style GD on the soft‑max surrogate of the max‑conv."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    # temperature schedule: from 5e‑2 down to 5e‑4 (geometric)
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # temperature decay (geometric)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.3,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 9,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv) using an active set."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it}, score={cur_score:.6f}")

        # ----- active set -------------------------------------------------
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # Build weight vector w (averaged over active indices)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # Gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1: n - 1 + n]

        # Nesterov‑style momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.5,
                   end_temp: float = 1e-7,
                   steps: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk with tiny mass‑transfer moves and occasional smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # ----- move proposal ------------------------------------------------
        if np.random.rand() < 0.85:                 # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25   # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                        # smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random‑parameter generators for the cheap initial phase
# ----------------------------------------------------------------------
def _gen_powerlaw(n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p or reversed, with p∈[-3,3]."""
    p = np.random.uniform(-3.0, 3.0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    if np.random.rand() < 0.5:
        a = a[::-1]
    a += 1e-12
    return a / a.sum()


def _gen_logistic(n: int) -> np.ndarray:
    """Logistic‑shaped vector a_i = 1/(1+exp(-k·(i-mid)))."""
    k = np.random.uniform(-8.0, 8.0)    # steepness
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(-k * i / (n / 2.0)))
    a += 1e-12
    return a / a.sum()


def _gen_mixture_gauss(n: int) -> np.ndarray:
    """Mixture of two Gaussian bumps."""
    idx = np.arange(n, dtype=np.float64) / (n - 1)
    pos1 = np.random.rand()
    pos2 = np.random.rand()
    sigma1 = np.random.uniform(0.02, 0.2)
    sigma2 = np.random.uniform(0.02, 0.2)
    w1 = np.random.rand()
    gauss1 = np.exp(-0.5 * ((idx - pos1) / sigma1) ** 2)
    gauss2 = np.exp(-0.5 * ((idx - pos2) / sigma2) ** 2)
    a = w1 * gauss1 + (1.0 - w1) * gauss2
    a += 1e-12
    return a / a.sum()


def _generate_random_small(n: int) -> np.ndarray:
    """Pick one of the families at random."""
    r = np.random.rand()
    if r < 0.35:
        return _gen_powerlaw(n)
    elif r < 0.70:
        return _gen_logistic(n)
    else:
        return _gen_mixture_gauss(n)

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs:
      * a fast random‑parameter search on a small vector,
      * up‑sampling to the final resolution,
      * projected soft‑max gradient descent,
      * exact sub‑gradient descent,
      * a long simulated‑annealing refinement.
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    total_budget = deadline - start_time

    # ------------------------------------------------------------------
    # Phase 1 – cheap random exploration on a small ``n_small``.
    # ------------------------------------------------------------------
    n_small = 1200
    best_seq_small = None
    best_score_small = float("inf")
    phase1_end = start_time + total_budget * 0.15
    it_small = 0

    while time.time() < phase1_end:
        a = _generate_random_small(n_small)
        conv = _conv_self(a)
        cur_score = 2.0 * n_small * conv.max()
        if cur_score < best_score_small:
            best_score_small = cur_score
            best_seq_small = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] random‑init improve: "
                  f"n={n_small}, iter={it_small}, score={cur_score:.6f}")
        it_small += 1

    if best_seq_small is None:
        best_seq_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the final resolution.
    # ------------------------------------------------------------------
    n_final = 9600
    a0 = _upsample(best_seq_small, n_final)

    # Try the historic warm‑start if available and better.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.shape[0] != n_final:
                starter = _upsample(starter, n_final)
            starter_score = 2.0 * n_final * _conv_self(starter).max()
            a0_score = 2.0 * n_final * _conv_self(a0).max()
            if starter_score < a0_score:
                a0 = starter
                print(f"[{time.time() - start_time:6.2f}s] warm‑start from "
                      f"`height_sequence_1` improved: score={starter_score:.6f}")
    except Exception:
        pass

    best_seq = a0.copy()
    best_score = 2.0 * n_final * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:6.2f}s] after up‑sampling: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Determine time slices for the remaining phases.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate 15 % softmax, 30 % sub‑gradient, rest (≈55 %) for annealing
    softmax_budget = remaining * 0.15
    subgrad_budget = remaining * 0.30
    anneal_budget = remaining - (softmax_budget + subgrad_budget)

    # ------------------------------------------------------------------
    # Phase 3 – soft‑max gradient descent.
    # ------------------------------------------------------------------
    soft_deadline = time.time() + softmax_budget
    a_opt, score_opt = _run_softmax_gd(
        best_seq, n_final, soft_deadline, start_time,
        step0=0.8, beta=0.9, max_iter=10 ** 9
    )
    if score_opt < best_score:
        best_seq = a_opt.copy()
        best_score = score_opt
        print(f"[{time.time() - start_time:6.2f}s] after softmax GD: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – exact sub‑gradient descent.
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + subgrad_budget
    a_opt2, score_opt2 = _run_subgradient_gd(
        best_seq, n_final, subgrad_deadline, start_time,
        step0=0.4, beta=0.9, max_iter=10 ** 9, rel_eps=0.01
    )
    if score_opt2 < best_score:
        best_seq = a_opt2.copy()
        best_score = score_opt2
        print(f"[{time.time() - start_time:6.2f}s] after sub‑gradient GD: "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated annealing refinement.
    # ------------------------------------------------------------------
    anneal_deadline = deadline  # use whatever time is left
    a_ref, score_ref = _run_annealing(
        best_seq, n_final,
        anneal_deadline, start_time,
        best_score, best_seq,
        start_temp=0.5, end_temp=1e-7,
        steps=10_000_000
    )
    if score_ref < best_score:
        best_seq = a_ref.copy()
        best_score = score_ref
        print(f"[{time.time() - start_time:6.2f}s] final annealing improvement: "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & conversion to plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
