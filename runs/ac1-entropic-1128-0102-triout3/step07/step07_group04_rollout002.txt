def _build_active_constraints(f, active_inds):
    n = len(f)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_inds):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i from lo..hi inclusive
        idx_range = np.arange(lo, hi+1)
        rows.extend([row_idx] * len(idx_range))
        cols.extend(idx_range.tolist())
        data.extend(f[idx_range].tolist())
    A = coo_matrix((data, (rows, cols)), shape=(len(active_inds), n)).tocsr()
    return A

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = conv.max()
        # active indices:
        eps = max(active_tol, M * 1e-9)
        active_inds = np.where(conv >= M - eps)[0]
        if active_inds.size == 0:
            break
        # limit number of active rows to e.g., 100
        if active_inds.size > 100:
            # pick indices with largest conv values
            sorted_inds = np.argsort(-conv[active_inds])
            active_inds = active_inds[sorted_inds[:100]]
        # Build sparse matrix
        A = _build_active_constraints(a, active_inds)
        b = np.full(A.shape[0], M)
        # variable bounds per g_i
        eps_f = 1e-12

        c = -np.ones(n)  # minimization
        try:

        except Exception as e:

            break
        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed: {res.message}")
            break

        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-9:

            print(f"[{time.time() - start_time:6.2f}s] LP no improvement (sum {sum_g0})")

"""
propose_candidate – hybrid optimizer for the max‑self‑convolution problem.

The routine builds on three ideas:

1. **Smooth Adam optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution, using FFT for cheap gradients.  This quickly yields a
   good shape for a modest size vector (n≈2048) and then for a larger vector
   (n≈8192).

2. **Projected sub‑gradient descent** – polish the solution on the true
   objective 2·n·max(conv) while keeping the vector on the probability
   simplex.

3. **Active‑set linear programming** – given the current vector `a`,
   compute the set of convolution indices that are (almost) maximal.
   Solve a tiny LP that maximises the total mass of a new vector `g`
   subject to the linear constraints `a * g ≤ M` on those active rows
   (together with simple per‑coordinate upper bounds).  Mixing the old
   vector with the normalised LP solution can strictly decrease the
   max‑convolution.  Only a handful of constraints are used, so the LP
   is solved in a fraction of a second even for n=8192.

Finally a light simulated‑annealing pass moves small amounts of mass
between coordinates to escape tiny local traps.

All stages respect the global time budget (default 1000 s).  The returned
sequence is a plain Python list of non‑negative floats whose sum is 1,
so it can be fed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Optional SciPy imports (LP solver and sparse matrices)
# -------------------------------------------------------------
try:

    from scipy.sparse import coo_matrix

except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# -------------------------------------------------------------
# Basic linear‑algebra helpers
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# -------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# -------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# -------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the final polishing stage)
# -------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# -------------------------------------------------------------
# Adam optimisation of the smooth log‑sum‑exp surrogate
# -------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed objective
        Oτ(a)=2·n·τ·log Σ exp(conv(a)/τ)
    with Adam and Euclidean projection onto the simplex.
    Returns the best vector seen and its true (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ----- soft‑max weights (log‑sum‑exp) -----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability trick
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape 2·n‑1, sums to 1

        # ----- gradient of Oτ -----
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # ∑ a_j·p_{i+j}
        grad = 4.0 * n * inner

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # ----- cooling -----
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# -------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# -------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classic Nesterov‑style projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (∑a=1, a≥0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# -------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# -------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# -------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# -------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# -------------------------------------------------------------
# Active‑set LP improvement utilities
# -------------------------------------------------------------
def _build_active_constraints(f: np.ndarray, active_inds: np.ndarray) -> coo_matrix:
    """Return a sparse matrix A such that (A @ g)_k = (f * g)_k for k in active_inds."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_inds):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        rows.extend([row_idx] * idx.size)
        cols.extend(idx.tolist())
        data.extend(f[idx].tolist())
    A = coo_matrix((data, (rows, cols)), shape=(len(active_inds), n))
    return A


def _run_lp_improvement(f: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        max_iters: int = 10,
                        rel_eps: float = 1e-9,
                        active_tol_abs: float = 1e-12,
                        t_candidates: tuple = (0.1, 0.25, 0.5, 0.75)) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the active‑set LP improvement described in the
    statement.  Returns the best sequence and its true score.
    """
    if not _SCIPY_AVAILABLE:
        print(f"[{time.time() - start_time:6.2f}s] SciPy unavailable – skipping LP stage.")
        conv = _conv_self(f)
        return f, 2.0 * n * conv.max()

    a = f.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    best_seq = a.copy()

    eps_f = 1e-12

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()
        # indices where conv is (almost) maximal
        eps = max(active_tol_abs, M * rel_eps)
        active = np.where(conv >= M - eps)[0]

        if active.size == 0:
            break

        # keep the set small to keep the LP cheap
        max_active = 100
        if active.size > max_active:
            # keep rows with largest conv values
            order = np.argsort(-conv[active])[:max_active]
            active = active[order]

        A = _build_active_constraints(a, active).tocsr()
        b = np.full(A.shape[0], M)

        # per‑coordinate upper bounds derived from a_i * g_i ≤ M
        ub = M / (a + eps_f)
        # clamp to a reasonable max to avoid numerical issues
        ub = np.minimum(ub, 1e6)

        bounds = [(0.0, ub_i) for ub_i in ub]

        c = -np.ones(n)                      # minimise –sum(g)  ≡ maximise sum(g)
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:               # pragma: no cover
            print(f"[{time.time() - start_time:6.2f}s] LP exception: {e}")
            break

        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-9:
            # cannot increase total mass – stop
            print(f"[{time.time() - start_time:6.2f}s] LP iteration {it}: sum(g)={sum_g0:.6f} (no gain)")
            break

        # normalise to keep total mass = 1
        g = g0 / sum_g0

        # try a few mixing coefficients and keep the best reduction
        best_local_score = float("inf")
        best_a_candidate = None
        for t in t_candidates:
            a_new = (1.0 - t) * a + t * g
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_local_score:
                best_local_score = new_score
                best_a_candidate = a_new

        if best_local_score < best_score - 1e-12:
            a = best_a_candidate
            best_score = best_local_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter {it}: score={best_score:.6f}")
        else:
            # no improvement with any mixing factor – terminate
            print(f"[{time.time() - start_time:6.2f}s] LP iteration {it}: no improvement")
            break

    return best_seq, best_score

# Main entry point required by the harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method combines Adam smoothing, sub‑gradient polishing,
    a short active‑set LP refinement and a tiny annealing phase.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------
    # Optional starter – the previously found construction
    # ---------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ---------------------------------------------------------
    # Stage 0 – several short Adam runs on a modest size (n≈2048)
    # ---------------------------------------------------------
    n0 = 2048
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    best_seq = None
    best_score = float("inf")
    restarts = 4
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 10.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-12,
            tau_decay=0.9995,
            step_size=0.25,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={score:.6f}")

    # fallback – extremely unlikely
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ---------------------------------------------------------
    # Stage 1 – up‑sample to a larger size (≈8192) and continue Adam
    # ---------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)

    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 10.0, 1e-12)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.30 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_adam_softmax(
        a1, n1, grad_deadline1, start_time,
        tau0=tau0_1,
        tau_end=1e-13,
        tau_decay=0.9994,
        step_size=0.12,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ---------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (shorter than before)
    # ---------------------------------------------------------
    if time.time() < deadline:
        # allocate roughly half of the remaining budget to sub‑gradient work
        subgrad_budget = max(30.0, 0.45 * (deadline - time.time()))
        subgrad_deadline = time.time() + subgrad_budget

        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n1, subgrad_deadline, start_time,
            beta=0.94,
            step0=0.13,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ---------------------------------------------------------
    # Stage 3 – active‑set linear programming refinement
    # ---------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = deadline - 8.0  # reserve a few seconds for annealing
        a_lp, score_lp = _run_lp_improvement(
            best_seq, n1, lp_deadline, start_time,
            max_iters=8,
            rel_eps=1e-9,
            active_tol_abs=1e-12,
            t_candidates=(0.1, 0.25, 0.5, 0.75),
        )
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 LP improve: score={score_lp:.6f}")

    # ---------------------------------------------------------
    # Stage 4 – very light simulated‑annealing refinement
    # ---------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            n1,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ---------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # ---------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
