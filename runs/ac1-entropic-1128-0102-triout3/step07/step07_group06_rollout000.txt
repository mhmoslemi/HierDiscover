"""
Propose a step‑function that (approximately) minimises the
``evaluate_sequence`` objective.

Algorithm Overview
------------------
1. **Warm‑start** – generate a small set of parametric candidates
   (uniform, linear, cosine, triangular, power‑law, symmetric power‑law,
   a few random Dirichlet vectors) and pick the best according to the
   true objective 2·n·max(conv(a)).

2. **Progressive up‑sampling** – the best candidate is up‑sampled to
   a series of increasing resolutions (800 → 2400 → 4800 → 9600).

3. **Adam‑style soft‑max gradient descent** – minimise a
   temperature‑smoothed version of max‑convolution using Adam
   optimisation and projection onto the probability simplex.

4. **Exact sub‑gradient descent** – switch to the true non‑smooth
   objective with an active‑set sub‑gradient method (tighter active
   tolerance than the reference).

5. **Guided simulated annealing** – perform a Metropolis walk that
   moves mass either randomly or **guided by the current sub‑gradient**
   (mass is transferred from a high‑gradient index to a low‑gradient
   one).  Occasional local smoothing moves are also applied.
   The temperature schedule is geometric and the number of steps is
   calibrated to the remaining wall‑clock budget.

All stages respect the 1000 s wall‑clock limit; the routine falls back
to a uniform distribution if nothing is found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max of convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution (temperature‑smoothed)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # subtract max for numerical stability
    scaled -= np.max(scaled)
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max surrogate
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.10,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Adam on the temperature‑smoothed max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # temperature schedule (soft‑max)
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # temperature (soft‑max smoothing)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (active set)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        rel_eps: float = 0.001,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv) with a tight active set."""
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, it={it}, score={cur_score:.6f}")

        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        m_len = w.shape[0]
        fft_len = 1 << ((n + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Guided simulated annealing
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   max_steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk with gradient‑guided mass transfers and occasional smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # Pre‑allocate space for sub‑gradient (recomputed only every few steps)
    grad = None

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # ---- decide move type ----
        if np.random.rand() < 0.75:
            # mass‑transfer move – sometimes guided
            if it % 30 == 0:
                # recompute a cheap sub‑gradient to guide the move
                conv = _conv_self(a)
                max_val = conv.max()
                eps = max_val * 0.01
                active = np.where(conv >= max_val - eps)[0]
                if active.size == 0:
                    active = np.array([conv.argmax()], dtype=int)
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size
                rev_a = a[::-1]
                m_len = w.shape[0]
                fft_len = 1 << ((n + m_len - 1).bit_length())
                A = np.fft.rfft(rev_a, n=fft_len)
                B = np.fft.rfft(w, n=fft_len)
                full = np.fft.irfft(A * B, n=fft_len)[: n + m_len - 1]
                grad = 2.0 * full[n - 1:n - 1 + n]
                i = np.argmax(grad)          # index that most increases the peak
                j = np.argmin(grad)          # index that most decreases it
            else:
                i, j = np.random.randint(0, n, size=2)

            if i == j or a[i] == 0.0:
                continue

            # transfer up to 50 % of the source mass
            eps_mass = a[i] * np.random.rand() * 0.5
            a_prop = a.copy()
            a_prop[i] -= eps_mass
            a_prop[j] += eps_mass
        else:
            # occasional local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, step={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Candidate initialisation (parametric families + random)
# ----------------------------------------------------------------------
def _initial_candidates(n: int):
    """Return a list of plausible initial vectors for length ``n``."""
    i = np.arange(n, dtype=float)

    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Linear ramp
    ramp = i + 1.0
    cand.append(ramp / ramp.sum())

    # Symmetric triangular (peak in the centre, zero at the ends)
    tri = 1.0 - np.abs(i - (n - 1) / 2.0) / (n / 2.0)
    tri[tri < 0] = 0.0
    cand.append(tri / tri.sum())

    # Cosine‑shaped (positive half‑wave)
    cosine = np.sin(np.pi * (i + 0.5) / n)
    cand.append(cosine / cosine.sum())

    # Power‑law decreasing
    for p in (0.2, 0.5, 1.0, 2.0):
        pw = (i + 1.0) ** (-p)
        cand.append(pw / pw.sum())

    # Symmetric power‑law (peaked in centre)
    for p in (0.5, 1.0, 2.0):
        pw_sym = ((i + 1.0) * (n - i)) ** p
        cand.append(pw_sym / pw_sym.sum())

    # A few random Dirichlet draws
    for _ in range(5):
        rnd = np.random.rand(n)
        cand.append(rnd / rnd.sum())

    return cand


def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a)).  Assumes sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, performs progressive up‑sampling,
    Adam‑softmax gradient descent, exact sub‑gradient descent and a guided
    simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – gradually increase resolution.
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]

    # Allocate roughly 60 % of the total time to deterministic optimisation,
    # the remaining 40 % to stochastic annealing.
    total_time = deadline - start_time
    grad_budget = total_time * 0.60
    anneal_budget = total_time - grad_budget

    # ------------------------------------------------------------------
    # Warm‑start: try to reuse a previously discovered construction if
    # the global variable ``height_sequence_1`` exists.
    # ------------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Iterate over resolutions.
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time slice for the current stage (proportional to its length)
        stage_time = grad_budget * (n / sum(stage_lengths))
        stage_deadline = time.time() + stage_time

        # --------------------------------------------------------------
        # Initialise a vector for this resolution.
        # --------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # generate a few parametric candidates and keep the best one
            candidates = _initial_candidates(n)
            best_local = None
            best_local_score = float("inf")
            for cand in candidates:
                score = _objective(cand)
                if score < best_local_score:
                    best_local_score = score
                    best_local = cand
            a0 = best_local.copy()

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimiser on the soft‑max surrogate.
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam_softmax(
            a0, n, stage_deadline, start_time,
            step0=0.10, beta1=0.9, beta2=0.999, eps=1e-8,
            max_iter=10 ** 7
        )

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient descent.
        # ------------------------------------------------------------------
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _run_subgradient_gd(
                a_opt, n, stage_deadline, start_time,
                step0=0.15, beta=0.9, rel_eps=0.001,
                max_iter=10 ** 7
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # --------------------------------------------------------------
        # Keep the global best.
        # --------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # Update remaining gradient budget.
        elapsed = time.time() - start_time
        grad_budget = max(0.0,
                          deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works on the final resolution obtained.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        remaining = deadline - time.time()
        # Rough estimate: ~500 steps per second for the annealer.
        max_steps = max(1_000, int(remaining * 500))
        # cap to avoid excessive loops
        max_steps = min(max_steps, 6_000_000)

        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-7,
            max_steps=max_steps
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # ------------------------------------------------------------------
    # Normalise (tiny drift may have accumulated) and return a plain list.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
