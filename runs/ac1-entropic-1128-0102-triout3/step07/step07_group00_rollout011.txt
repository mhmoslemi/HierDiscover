def _adam_smooth(a0, n, deadline, start_time, alpha_start, alpha_end, max_iters=20000, lr0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    beta1_pow=1.0; beta2_pow=1.0
    L = _next_pow2(3*n-2)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it}, score={cur_score:.6f}")
        # compute alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # softmax weights
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]
        # gradient
        Af = np.fft.rfft(a, n=L)
        Wrev = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it}, score={cur_score:.6f}")
        # gradient at max index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]  # a[k-idx] works within range
        # update momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score

def _annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
               start_temp=5e-2, end_temp=1e-6, steps=150_000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    for it in range(1, steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        src = rng.choice(n, p=a)  # source by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it}, best={best_score:.6f}")
    return best_seq, best_score

"""
Optimiser for the ``evaluate_sequence`` objective.

The algorithm proceeds in three layers:

1. **Multiple random restarts** – each trial starts from a fresh Dirichlet
   sample (or the supplied global starter).  A short Adam optimisation on a
   smooth log‑sum‑exp approximation of the maximal convolution is performed.
   The smoothing parameter α grows from a small to a large value inside the
   run.

2. **Projected sub‑gradient descent** – after the smooth phase a pure
   sub‑gradient descent on the exact objective (2·n·max(conv(a,a))) is run.
   Momentum and a 1/√t step‑size are used and the iterate is projected onto
   the probability simplex after every step.

3. **Light simulated‑annealing polish** – finally a cheap annealing that moves a
   tiny amount of mass between two coordinates is applied.  The temperature
   decays exponentially.

All phases respect the global time budget (≈1000 s).  The best sequence over
all trials is returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns a vector w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 alpha_start: float = 0.5,
                 alpha_end: float = 200.0,
                 max_iters: int = 12000,
                 lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on f_α(a)= 1/α·log∑exp(α·conv(a,a)).
    α is increased linearly from ``alpha_start`` to ``alpha_end``.
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial score
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    L_grad = _next_pow2(3 * n - 2)          # length for the gradient FFTs

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # evaluate current objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # increasing smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        # soft‑max weights over the convolution entries
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w /= w.sum()
        w_rev = w[::-1]

        # gradient of the smoothed objective
        Af = np.fft.rfft(a, n=L_grad)
        Wrev = np.fft.rfft(w_rev, n=L_grad)
        corr = np.fft.irfft(Af * Wrev, n=L_grad)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 / math.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact max – single‑peak sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum (β≈0.9) and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")

        # gradient at the (unique) maximal index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-6,
               steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Very light mass‑transfer annealing.  Temperature decays exponentially.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    The routine performs many short optimisation trials, keeps the best
    solution, and finishes with a longer refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid overrun
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 4096 is FFT‑friendly and was used by the reference.
    # --------------------------------------------------------------
    n = 4096

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a global starter exists, evaluate it first.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            cur = 2.0 * n * _conv_self(starter).max()
            best_seq = starter.copy()
            best_score = cur
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Repeated short trials (random restarts)
    # --------------------------------------------------------------
    min_trial_time = 90.0          # leave ~30 s for the final polish
    final_margin = 30.0

    while True:
        remaining = deadline - time.time()
        if remaining < min_trial_time + final_margin:
            break

        trial_deadline = time.time() + min_trial_time

        # random initial vector on the simplex
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # ---- Phase‑1 (Adam smooth) ---------------------------------
        a_adam, score_adam = _adam_smooth(
            a0,
            n,
            trial_deadline,
            start_time,
            alpha_start=0.5,
            alpha_end=200.0,
            max_iters=8000,
            lr0=0.5,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial Adam improve: score={score_adam:.6f}")

        # ---- Phase‑2 (sub‑gradient) --------------------------------
        sub_deadline = trial_deadline - 5.0   # reserve a few seconds for a tiny anneal
        a_sub, score_sub = _subgradient_descent(
            a_adam,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial sub‑grad improve: score={score_sub:.6f}")

        # ---- Phase‑3 (quick anneal) --------------------------------
        a_fin, score_fin = _annealing(
            a_sub,
            n,
            trial_deadline,
            start_time,
            best_score,
            a_sub,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=40_000,
        )
        if score_fin < best_score:
            best_score = score_fin
            best_seq = a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] Trial anneal improve: score={score_fin:.6f}")

    # ------------------------------------------------------------------
    # Final refinement on the best candidate found so far.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()

    # Phase‑2 (long sub‑gradient) with the remaining time
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.70
        a_sub, score_sub = _subgradient_descent(
            best_seq,
            n,
            sub_deadline,
            start_time,
            best_score,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Final sub‑grad improve: score={score_sub:.6f}")

    # Final simulated annealing polish (uses whatever time is left)
    if time.time() < deadline:
        a_final, score_final = _annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and cast to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as a list of floats – the harness will feed it to evaluate_sequence().
    return [float(x) for x in best_seq]
