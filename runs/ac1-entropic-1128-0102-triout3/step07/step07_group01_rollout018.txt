def _run_fast_refine(a0, n, deadline, start_time, eps_frac_start=0.1, eps_frac_min=0.005, decay_rate=0.98):

    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    eps_frac = eps_frac_start
    iter_count = 0
    # Compute tolerance for improvement
    tolerance = 1e-12
    while time.time() < deadline:
        iter_count += 1
        # Identifying max index
        k_max = int(np.argmax(conv))
        # Compute contributions
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        # contributions for each i (both sides)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        total_contrib = contrib.sum()
        # Choose source index
        src = int(np.argmax(contrib))
        if a[src] <= 0:
            # fallback
            src = int(np.argmax(a))
        # Choose destination - low contribution and not src
        # Option - choose index with minimal a
        dst_candidates = np.where(np.arange(n) != src)[0]
        # Choose dst with minimal contribution (or random from lowest top)
        # Let's compute contributions minimal
        # Could sample from few low contribution indices.
        # For speed, pick dest = argmin(contrib) if not src else second argmin.
        dst = int(np.argmin(contrib))
        if dst == src:
            # pick second smallest
            sorted_idx = np.argsort(contrib)
            dst = int(sorted_idx[1])
        # Ensure dst != src
        if dst == src:
            continue
        # Determine epsilon
        eps = a[src] * eps_frac
        if eps <= 0:
            eps = a[src] * 0.5  # fallback
        # maximum epsilon limited by a[src]
        if eps > a[src]:
            eps = a[src]
        # Ensure eps not too tiny
        if eps < 1e-12:
            continue
        # Try to apply move; allow reduction of epsilon if not improving
        for attempt in range(5):
            # compute conv_new via incremental update
            conv_new = conv.copy()
            # add contributions from shift
            # conv_new[dst:dst+n] += 2*eps*a
            # conv_new[src:src+n] -= 2*eps*a

            conv_new[dst:dst+n] += 2*eps*a
            conv_new[src:src+n] -= 2*eps*a

                a[src] -= eps
                a[dst] += eps

                    break

"""
Fast local refinement for the step‑function attack.

The algorithm starts from the best known construction (available as the
global ``height_sequence_1``) and then performs a long series of cheap,
gradient‑guided moves that strictly decrease the maximal value of the
self‑convolution.  Each move transfers a fraction of the mass from an
index that contributes the most to the current peak to an index that
contributes the least.  The convolution is updated incrementally, so a
single iteration costs only O(n) operations and many more iterations can
be performed within the 1000 s budget.  The final sequence is normalised
to unit sum before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1).  Uses numpy's
    direct convolution – only called once per run, so the O(n²) cost is
    negligible.
    """
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Sub‑gradient descent (coarse phase)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.4,
                             beta: float = 0.9) -> np.ndarray:
    """
    Global sub‑gradient descent with Euclidean projection onto the simplex.
    Runs until the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _conv_self(a).max()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, max={best_max:.8f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a


# ----------------------------------------------------------------------
# Fast targeted refinement (fine phase)
# ----------------------------------------------------------------------
def _run_fast_refine(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     eps_start: float = 0.12,
                     eps_min: float = 0.004,
                     decay: float = 0.97) -> np.ndarray:
    """
    Repeatedly moves a small amount of mass from a source that
    contributes most to the current peak to a destination that contributes
    least.  The convolution is updated incrementally, so each iteration
    costs only O(n).  Only moves that strictly lower the peak are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    eps_frac = eps_start
    it = 0
    tol = 1e-12

    # pre‑allocate a working copy for the trial convolution
    conv_trial = np.empty_like(conv)

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Identify the lag that realises the current maximum
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))

        # contributions of each index to this lag
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[i[valid]] * a[j[valid]]

        # --------------------------------------------------------------
        # Choose source (largest contribution) and destination (smallest)
        # --------------------------------------------------------------
        src = int(np.argmax(contrib))
        if a[src] <= 0.0:                     # safety fallback
            src = int(np.argmax(a))

        # destination – try to pick a low‑contributing index that is not src
        dst = int(np.argmin(contrib))
        if dst == src:
            # second smallest if the very smallest is src
            sorted_idx = np.argsort(contrib)
            dst = int(sorted_idx[1])
        if dst == src:
            # final fallback: random different index
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

        # --------------------------------------------------------------
        # Amount to move – start with a relatively aggressive fraction
        # --------------------------------------------------------------
        eps = a[src] * eps_frac
        if eps <= 0.0:
            continue
        # Clamp to the available mass at src
        eps = min(eps, a[src])

        # --------------------------------------------------------------
        # Try the move (possibly shrinking eps if it does not improve)
        # --------------------------------------------------------------
        accepted = False
        for _ in range(5):
            # copy current convolution
            conv_trial[:] = conv

            # linear part: 2*eps*(shift(dst) - shift(src))
            conv_trial[dst:dst + n] += 2.0 * eps * a
            conv_trial[src:src + n] -= 2.0 * eps * a

            # quadratic eps² terms
            conv_trial[2 * dst] += eps * eps
            conv_trial[2 * src] += eps * eps
            conv_trial[src + dst] -= 2.0 * eps * eps

            new_max = conv_trial.max()
            if new_max < cur_max - tol:
                # accept the move
                a[src] -= eps
                a[dst] += eps
                conv[:] = conv_trial
                cur_max = new_max
                accepted = True
                if new_max < best_max - tol:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] fast refine improve: "
                          f"iter={it:,}, max={best_max:.8f}")
                break
            # otherwise halve eps and retry
            eps *= 0.5
            if eps < 1e-14:
                break

        # decay epsilon fraction slowly to allow finer adjustments later
        if it % 1000 == 0:
            eps_frac = max(eps_frac * decay, eps_min)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Initialise – use the previously discovered construction if present
    # --------------------------------------------------------------
    final_n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample / down‑sample to the target length while preserving mass
            old_n = starter.shape[0]
            if old_n != final_n:
                old_x = np.linspace(0.0, 1.0, old_n)
                new_x = np.linspace(0.0, 1.0, final_n)
                a0 = np.interp(new_x, old_x, starter)
            else:
                a0 = starter.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 /= s
            else:
                a0 = np.full(final_n, 1.0 / final_n)
        else:
            raise ValueError
    except Exception:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    # --------------------------------------------------------------
    # Phase 1 – a short sub‑gradient descent to escape the worst region
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + min(15.0, 0.07 * (deadline - time.time()))
        a_mid = _run_subgradient_descent(a0, phase1_deadline, start_time,
                                         step0=0.4, beta=0.9)
    else:
        a_mid = a0

    # --------------------------------------------------------------
    # Phase 2 – fast targeted refinement for the bulk of the time budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_best = _run_fast_refine(a_mid, deadline, start_time,
                                  eps_start=0.12, eps_min=0.004, decay=0.97)
    else:
        a_best = a_mid

    # --------------------------------------------------------------
    # Final normalisation (evaluate_sequence expects a list)
    # --------------------------------------------------------------
    a_best = np.maximum(a_best, 0.0)
    total = a_best.sum()
    if total > 0:
        a_best /= total
    else:
        a_best = np.full(final_n, 1.0 / final_n)

    # optional debug print of final objective
    conv_final = _conv_self(a_best)
    final_score = float(2.0 * final_n * conv_final.max())
    print(f"[{time.time() - start_time:.2f}s] final score: {final_score:.6f}")

    return [float(x) for x in a_best]
