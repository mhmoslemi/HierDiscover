def _run_flatten_gd(a0, n, deadline, start_time, step0=0.1, beta=0.9, max_iter=20000):
    a = a0.copy()
    momentum = np.zeros_like(a)
    target = np.full(2*n-1, 1.0/(2*n-1))
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        diff = conv - target
        # gradient of mean squared error: 4 * (rev(a) * diff)
        rev_a = a[::-1]
        m = diff.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(diff, n=fft_len)
        grad_full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 4.0 * grad_full[n-1:n-1+n]
        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / np.sqrt(it)
        a = _project_simplex(a - step * momentum)
        # compute objective
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] flatten improve: n={n}, iter={it}, score={cur_score:.6f}")
    return best_a, best_score
```

"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The method is deliberately different from the
reference implementation:

1. **Warm‑start** – if a previous construction ``height_sequence_1`` is
   available it is linearly up‑sampled to the final resolution (9600).  If not,
   a random Dirichlet vector is used.

2. **Flattening gradient descent** – the objective
   ``max(conv)`` is reduced by minimising the squared deviation of the *self‑
   convolution* from a uniform distribution.  The gradient of the loss
   ``‖conv−uniform‖²`` is computed via a single FFT, Nesterov‑style momentum
   and a diminishing step size are applied, and every iterate is projected onto
   the probability simplex.

3. **Simulated‑annealing refinement** – a long Metropolis walk moves a small
   fraction of mass between two random bins (or performs a short smoothing
   convolution).  The temperature follows a geometric schedule; worsening moves
   are accepted with probability ``exp(−Δ/ T)``.  The walk stops when the time
   budget is exhausted.

4. The best sequence found (lowest ``evaluate_sequence`` value) is returned as
   a plain Python list.

The whole routine respects a user supplied wall‑clock budget (default 1000 s)
and never exceeds it.  The final length is 9600, which gives enough degrees of
freedom to push the score below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (full length = 2·n−1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear up‑sampling (simple interpolation) and re‑normalisation.
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient descent that flattens the self‑convolution towards uniform.
# ----------------------------------------------------------------------
def _run_flatten_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.20,
                    beta: float = 0.90,
                    max_iter: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov‑style gradient descent on the loss
        L(a) = || a*a - u ||²,
    where u is the uniform distribution of total mass 1.
    Returns the best vector seen and the corresponding evaluate_sequence score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    # uniform target for the convolution (sum a = 1 ⇒ sum conv = 1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    best_a = a.copy()
    # initialise best score with the current vector
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)                     # O(n log n)
        diff = conv - target                     # deviation from uniform

        # gradient of L = 4 * rev(a) * diff
        rev_a = a[::-1]
        m = diff.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(diff, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 4.0 * full[n - 1:n - 1 + n]       # same shape as a

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

        # track the true objective
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] flatten improve: "
                  f"iter={it}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (tiny mass transfers + occasional smoothing)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.30,
                   end_temp: float = 1e-7,
                   steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that perturbs the distribution while trying to lower max(conv)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25        # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a step‑function (list of non‑negative heights) that attempts
    to minimise ``evaluate_sequence`` within the allotted wall‑clock time.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Target size – a length around 10 k usually gives a good trade‑off.
    # ------------------------------------------------------------------
    n_target = 9600

    # ------------------------------------------------------------------
    # Initialise a candidate (warm start if possible).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – flattening gradient descent.
    # ------------------------------------------------------------------
    # give roughly 25 % of the total budget to the gradient phase
    grad_deadline = time.time() + (budget_s * 0.25)
    if grad_deadline > deadline:
        grad_deadline = deadline

    a_best, best_score = _run_flatten_gd(
        a0, n_target, grad_deadline, start_time,
        step0=0.20, beta=0.90, max_iter=100_000
    )

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing refinement (uses the remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Estimate a safe number of annealing steps (≈2500 moves per second).
        remaining = deadline - time.time()
        est_steps = int(max(0, remaining * 2500))
        steps = min(5_000_000, max(10_000, est_steps))

        a_best, best_score = _run_annealing(
            a_best, n_target,
            deadline, start_time,
            cur_best_score=best_score,
            cur_best_seq=a_best,
            start_temp=0.30,
            end_temp=1e-7,
            steps=steps
        )

    # ------------------------------------------------------------------
    # Safety net – if nothing was found, fall back to the uniform vector.
    # ------------------------------------------------------------------
    if a_best is None or len(a_best) == 0:
        a_best = np.full(n_target, 1.0 / n_target)

    # Final normalisation (the optimisation may incur tiny rounding drift)
    a_best = np.asarray(a_best, dtype=float)
    s = a_best.sum()
    if s > 0.0:
        a_best /= s
    else:
        a_best = np.full_like(a_best, 1.0 / a_best.size)

    # Return a plain Python list for the evaluation harness.
    return [float(x) for x in a_best]
