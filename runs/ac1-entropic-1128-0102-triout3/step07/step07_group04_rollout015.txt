def _random_init(n):
    # mixture of a few Gaussians plus a uniform background
    x = np.arange(n)
    a = np.random.rand(n) * 0.1  # small uniform noise
    num_comp = np.random.randint(1,4)

        mu = np.random.uniform(0, n-1)
        sigma = np.random.uniform(n*0.05, n*0.3)

    a = np.maximum(a, 0)

    if s > 0:
        a /= s
    else:

"""
Adaptive multi‑stage optimisation for the max‑self‑convolution objective.

The procedure consists of:
  1. Many short Adam runs on a modest size (n≈2048) with diverse
     random initialisations (mixture of Gaussians).  The best candidate
     is kept.
  2. Upsample the best candidate to a larger size (≈8192 or 16384) and
     continue the Adam optimisation with a tighter temperature schedule.
  3. Projected sub‑gradient descent on the true max‑convolution to polish
     the solution.
  4. A few cheap smoothing steps (moving‑average) that often lower the
     peak of the convolution.
  5. A light simulated‑annealing “mass‑transfer’’ refinement.

All stages respect the supplied time budget (default 1000 s) and the
final sequence is projected onto the probability simplex, which makes
the evaluation invariant to scaling.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used for polishing)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0, Σ a = 1).
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9997,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise   Oτ(a)=2·n·τ·log∑exp(conv(a)/τ)
    using Adam with projection onto the probability simplex.
    Returns the best vector seen and its *true* (non‑smoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # true (non‑smoothed) score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ----- soft‑max weights (log‑sum‑exp) -----
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # length 2·n‑1, sums to 1

        # ----- gradient of Oτ -----
        # grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # ∑_j a_j·p_{i+j}
        grad = 4.0 * n * inner

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cooling schedule for τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on
        F(a)=2·n·max(conv(a))   (Σ a = 1, a ≥ 0).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source proportional to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Random initialiser (mixture of a few Gaussians)
# ----------------------------------------------------------------------


def _random_init(n: int) -> np.ndarray:
    """Generate a non‑negative vector of length n and normalize it."""
    x = np.arange(n, dtype=float)
    a = np.random.rand(n) * 0.01  # tiny uniform background
    num_comp = np.random.randint(1, 4)  # 1‒3 Gaussian blobs
    for _ in range(num_comp):
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(n * 0.05, n * 0.3)
        a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

def _smooth_once(a: np.ndarray, kernel: np.ndarray) -> np.ndarray:
    """One smoothing pass with the given 1‑D kernel (odd length)."""
    conv = np.convolve(a, kernel, mode='same')
    s = conv.sum()
    if s > 0:
        conv /= s
    return conv


# ----------------------------------------------------------------------
# Helper to compute the exact objective (avoids the public evaluate_sequence)

def _score(seq: np.ndarray) -> float:
    """Exact 2·n·max(conv(seq))/sum(seq)^2."""
    seq = np.asarray(seq, dtype=float)
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter (previous best) – normalise automatically
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0 and tmp.sum() > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    stage0_budget = min(0.30 * budget_s, 250.0)          # at most 30 % of total time
    stage0_deadline = start_time + stage0_budget
    restarts = 12

    for r in range(restarts):
        if time.time() >= deadline or time.time() >= stage0_deadline:
            break

        # allocate a slice of the remaining stage‑0 budget
        remaining_stage0 = stage0_deadline - time.time()
        slice_time = max(5.0, remaining_stage0 / (restarts - r))
        slice_deadline = time.time() + slice_time

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = _random_init(n0)

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)

        a_opt, score = _run_adam_softmax(
            a0, n0, slice_deadline, start_time,
            tau0=tau0,
            tau_end=1e-14,
            tau_decay=0.9993,
            step_size=0.25,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Stage 1 – upsample to a larger size and continue Adam
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # pick a larger size only if we have enough time left
        n1 = 16384 if remaining > 350 else 8192
        a1 = _upsample(best_seq, n1)

        conv1 = _conv_self(a1)
        tau0_1 = max(conv1.max() * 8.0, 1e-10)

        # reserve ~60 s for the later stages
        slice_deadline = deadline - 60.0
        if slice_deadline - time.time() > 10.0:
            a1_opt, score1 = _run_adam_softmax(
                a1, n1, slice_deadline, start_time,
                tau0=tau0_1,
                tau_end=1e-15,
                tau_decay=0.9994,
                step_size=0.12,
            )
            if score1 < best_score:
                best_score = score1
                best_seq = a1_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: "
                      f"n={n1}, score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        n2 = best_seq.shape[0]
        sub_deadline = deadline - 30.0
        a2_opt, score2 = _run_subgradient_descent(
            best_seq, n2, sub_deadline, start_time,
            beta=0.94,
            step0=0.15,
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – a few cheap smoothing passes (often helpful)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        smoothing_deadline = deadline - 10.0
        kernel = np.array([0.25, 0.5, 0.25])        # simple 3‑point moving average
        a = best_seq.copy()
        cur_score = best_score
        while time.time() < smoothing_deadline:
            a_new = _smooth_once(a, kernel)
            new_score = _score(a_new)
            if new_score < cur_score - 1e-8:      # accept only strict improvements
                a = a_new
                cur_score = new_score
                best_seq = a.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:6.2f}s] smoothing improve: "
                      f"score={cur_score:.6f}")
            else:
                break

    # ------------------------------------------------------------------
    # Stage 4 – very light simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=1e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the probability simplex and return a plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
