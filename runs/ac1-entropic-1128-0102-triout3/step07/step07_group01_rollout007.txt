    conv = _conv_self(a)
    # compute softmax weights w
    scaled = beta * conv
    max_scaled = np.max(scaled)
    expVals = np.exp(scaled - max_scaled)
    w = expVals / np.sum(expVals)
    # compute gradient
    a_rev = a[::-1]
    # convolution via FFT of w and a_rev
    L = _next_pow2(w.shape[0] + a_rev.shape[0] - 1)
    Wf = np.fft.rfft(w, n=L)
    Afr = np.fft.rfft(a_rev, n=L)
    conv_wa = np.fft.irfft(Wf * Afr, n=L)[: w.shape[0] + a_rev.shape[0] - 1]  # length 3n-2
    # gradient values for i: conv_wa[i + n - 1]
    grad = 2.0 * conv_wa[n-1 : n-1 + a.shape[0]]
    return grad, conv.max()

def _run_smart_refine(a0, n, deadline, start_time, max_iters=5e6, eps_initial=0.02, eps_decay=0.9, topk_src=15, botk_dst=15):
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    it = 0
    eps_frac = eps_initial
    while time.time() < deadline and it < max_iters:
        it += 1

        # compute location of maximum
        k_max = int(np.argmax(conv))

        # compute contributions for each i
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        # choose source indices
        src_candidates = np.argpartition(-contrib, topk_src-1)[:topk_src]  # topk largest
        # choose destination indices based on low partner value a_{k_max-d}
        partner_val = np.full(n, np.inf)
        # for each possible dest, partner index is k_max - dest
        jdest = k_max - np.arange(n)
        mask = (jdest >= 0) & (jdest < n)
        partner_val[mask] = a[jdest[mask]]
        dst_candidates = np.argpartition(partner_val, botk_dst-1)[:botk_dst]  # smallest
        # randomly pick src and dst among candidates
        src = rng.choice(src_candidates)
        dst = rng.choice(dst_candidates)
        if src == dst: continue
        if a[src] <= 0: continue

        # move epsilon mass
        eps = a[src] * rng.random() * eps_frac
        if eps < 1e-16: continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-15:
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smart refine improve: iter={it:,}, max={best_max:.8f}")

        # adapt epsilon gradually: reduce when no improvement for a while
        if it % 50000 == 0:
            eps_frac *= eps_decay
    return best_a
```

"""
Optimise step‑function heights for the benchmark ``evaluate_sequence``.

The strategy consists of three stages:

1. **Warm‑up (exponentiated gradient)** – minimise a smooth
   approximation of the maximum self‑convolution using an
   exponential‑gradient (mirror) update.  The smoothing parameter
   (α) is increased slowly, giving a coarse‑to‑fine optimisation.

2. **Projected sub‑gradient descent** – use the exact sub‑gradient of the
   max‑convolution (computed from all active constraints) together with
   Euclidean projection onto the probability simplex.

3. **Smart local refinement** – repeatedly move a tiny amount of mass
   from indices that contribute most to the current maximal convolution
   entry to indices whose counterpart in that entry is small.  The step
   size decays over time and only strictly improving moves are kept.
   This focuses the search on the “active set” of the current optimum
   and typically yields a few extra digits of improvement.

All stages respect the 1000 s time budget.  The best sequence found
throughout the run is normalised to sum = 1 and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of ``max(conv)`` w.r.t. ``a`` using
    all entries whose value lies within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each pair (i,j) contributes a[j] to g[i] and a[i] to g[j]
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of ``v`` onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    """
    Mirror‑descent (exponentiated‑gradient) on the
    ``softmax`` approximation of the maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smooth‑max weights ------------------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        # --- gradient of the smooth max (mirror‑descent) -----------------------
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Momentum (exponential averaging) is employed to stabilise the path.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : targeted local refinement (smart moves)
# ----------------------------------------------------------------------
def _run_smart_refine(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      max_iters: int = 10**7,
                      eps_frac_start: float = 0.025,
                      eps_decay: float = 0.99,
                      topk_src: int = 15,
                      botk_dst: int = 15):
    """
    Focused mass‑transfer moves.  In each iteration we look at the
    current maximal convolution entry ``k_max`` and:
        * pick a source index with probability proportional to its
          contribution ``a[i] * a[k_max‑i]``,
        * pick a destination where the partner ``a[k_max‑dst]`` is tiny,
        * move a small fraction of the source mass to the destination.
    The fraction ``eps_frac`` is exponentially decayed, allowing large
    steps early on and fine‑grained tweaking later.
    Only strictly improving moves (lowering the maximal convolution)
    are kept, guaranteeing monotone progress.
    """
    a = a0.copy()
    conv = _conv_self(a)
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    eps_frac = eps_frac_start
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        k_max = int(np.argmax(conv))

        # --- source candidates (large contribution to the current max) ---
        i = np.arange(n)
        j = k_max - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        if contrib.sum() == 0:
            # fallback: any index proportional to current mass
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)
        else:
            prob_src = contrib / contrib.sum()
            src = rng.choice(n, p=prob_src)

        # --- destination candidates (small partner value a[k_max‑dst]) ----------
        jdest = k_max - np.arange(n)
        mask = (jdest >= 0) & (jdest < n)
        partner = np.full(n, np.inf)
        partner[mask] = a[jdest[mask]]
        # select a few (botk_dst) indices with smallest partner value
        dst_candidates = np.argpartition(partner, botk_dst - 1)[:botk_dst]
        dst = int(rng.choice(dst_candidates))
        if dst == src:
            continue
        if a[src] <= 1e-14:
            continue

        # --- move a tiny fraction of mass ------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = _conv_self(a_prop)
        new_max = conv_prop.max()
        if new_max < max_val - 1e-12:           # strict improvement
            a = a_prop
            conv = conv_prop
            max_val = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smart refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # decay epsilon occasionally
        if it % 50000 == 0:
            eps_frac = max(eps_frac * eps_decay, 1e-6)

        # safety exit if we have exhausted the time‑budget
        if it % 100000 == 0 and time.time() >= deadline:
            break

    return best_a

    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – precise sub‑gradient descent (majority of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve at least 30 s for the final refinement

    if time.time() < deadline:

    best_seq = np.maximum(best_seq, 0.0)

    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # Return as plain Python list (the caller will feed it to evaluate_sequence)
    return [float(x) for x in best_seq]
