"""
Proposes a low‑value step‑function by a long simulated‑annealing (SA) phase.
The SA works directly on the self‑convolution using an incremental O(n)
update, so millions of tiny mass‑transfer moves can be tried within the
1000 s budget.  The algorithm starts from the best known sequence
(`height_sequence_1` if available, otherwise a random Dirichlet vector),
keeps the vector on the simplex (mass is only moved, never created or
destroyed) and tracks the best objective
    2·n·max(conv(a,a))
encountered.  Temperature cools linearly from 2 % of the current objective
to ~10⁻⁷, allowing uphill moves early on and fine‑grained exploitation
later.  Periodic full recomputation of the convolution fixes accumulated
numeric drift.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------
# Helper: fast self‑convolution via FFT (real -> real)
# ---------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution) using FFT. Length = 2·len(a)-1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # -----------------------------------------------------------
    # Choose a length – 4096 is a good FFT‑friendly compromise.
    # -----------------------------------------------------------
    n = 4096

    # -----------------------------------------------------------
    # Initialise from the global best if it exists, else random.
    # -----------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Interpolate to the required length
            a = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        # Random Dirichlet‑like start (α = 0.5 gives a slightly spiky shape)
        a = rng.random(n) ** 0.5
        a /= a.sum()

    # -----------------------------------------------------------
    # Pre‑compute the convolution and objective for the start.
    # -----------------------------------------------------------
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] initial score: {cur_score:.6f}")

    # -----------------------------------------------------------
    # Simulated‑annealing loop (incremental convolution update)
    # -----------------------------------------------------------
    # Temperature schedule (linear from T0 to T_end)
    T0 = 0.02            # start temperature (≈ 2 % of typical Δscore)
    T_end = 1e-7        # final temperature
    # How often we recompute the full convolution from scratch (to avoid drift)
    RECOMPUTE_EVERY = 250_000

    it = 0
    accepted = 0
    last_recompute = 0

    # Small‑move vs occasional big‑move probabilities
    BIG_MOVE_PROB = 0.005
    MAX_MOVE_START = 0.5   # early moves may shift up to 50 % of source mass
    MIN_MOVE = 0.01        # later moves shrink to about 1 %

    while time.time() < deadline:
        it += 1

        # Linear cooling based on elapsed *fraction* of total budget
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 0.5)
        T = T0 * (1.0 - frac) + T_end * frac

        # -------------------------------------------------------
        # Choose a source (mass‑weighted) and a destination (uniform)
        # -------------------------------------------------------
        src = rng.choice(n, p=a)               # heavy indices are tried more often
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # -------------------------------------------------------
        # Choose the amount of mass to move (eps)
        # -------------------------------------------------------
        if rng.random() < BIG_MOVE_PROB:
            move_limit = 0.9
        else:
            # Move limit shrinks with temperature
            move_limit = MIN_MOVE + (MAX_MOVE_START - MIN_MOVE) * (T - T_end) / (T0 - T_end)
        eps = a[src] * rng.random() * move_limit
        if eps <= 0.0:
            continue

        # -------------------------------------------------------
        # Incremental update of the self‑convolution (O(n) work)
        # -------------------------------------------------------
        # Work on a copy to be able to reject the move cheaply
        conv_new = conv.copy()

        # core linear terms: -2·eps·a shifted by src, +2·eps·a shifted by dst
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        # quadratic correction terms (only three indices change)
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        # -------------------------------------------------------
        # Evaluate the new objective
        # -------------------------------------------------------
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - cur_score

        # -------------------------------------------------------
        # Metropolis acceptance test
        # -------------------------------------------------------
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            # Accept the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score
            accepted += 1

            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] new best: {best_score:.6f}")

        # -------------------------------------------------------
        # Every RECOMPUTE_EVERY **accepted** moves recompute conv from scratch
        # to keep numeric errors under control.
        # -------------------------------------------------------
        if accepted - last_recompute >= RECOMPUTE_EVERY:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            last_recompute = accepted

    # -----------------------------------------------------------
    # Final clean‑up: re‑normalise (protect against tiny drift)
    # -----------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list – the harness expects this format.
    return [float(x) for x in best_seq]
