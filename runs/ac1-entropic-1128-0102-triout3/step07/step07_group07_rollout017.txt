def _annealing_incremental(a, conv, deadline, start_time):
    n = a.shape[0]
    best_a = a.copy()
    best_max = conv.max()
    cur_max = best_max
    rng = np.random.default_rng()
    steps = 0
    max_steps = int(5e6)  # define a large number

    start_temp = 5e-2
    end_temp = 1e-6

    while time.time() < deadline and steps < max_steps:
        # compute temperature schedule
        T = start_temp * (end_temp / start_temp) ** (steps / max_steps)

        # select src vectored by a
        if a.sum() == 0:
            break
        src = rng.choice(n, p=a)   # a sums to 1
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            steps += 1
            continue

        eps = a[src] * rng.random() * 0.4  # up to 40%
        # apply move
        arr_before = a.copy()
        # update conv
        conv[dst:dst+n] += 2*eps*arr_before
        conv[src:src+n] -= 2*eps*arr_before
        conv[2*src] += eps*eps
        conv[2*dst] += eps*eps
        conv[src+dst] -= 2*eps*eps

        a[src] -= eps
        a[dst] += eps

        new_max = conv.max()
        delta = new_max - cur_max

        accept = False
        if delta <= 0:
            accept = True
        else:
            if math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: max={cur_max:.8f}")
        else:
            # revert
            conv[dst:dst+n] -= 2*eps*arr_before
            conv[src:src+n] += 2*eps*arr_before
            conv[2*src] -= eps*eps
            conv[2*dst] -= eps*eps
            conv[src+dst] += 2*eps*eps
            a[:] = arr_before

        steps += 1

        # optionally periodic recompute conv via FFT
        if steps % 10000 == 0:
            conv = _conv_self_fft(a)
            cur_max = conv.max()

    return best_a, best_max

def _annealing_incremental(a: np.ndarray, conv: np.ndarray,
                           deadline: float, start_time: float,
                           steps_limit: int = 500000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing with O(n) incremental convolution updates."""
    n = a.shape[0]
    best_a = a.copy()
    best_max = conv.max()
    cur_max = best_max
    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-6
    steps = 0
    recompute_interval = 5000  # recompute conv from scratch periodically

    while time.time() < deadline and steps < steps_limit:
        # recompute periodically to avoid drift
        if steps % recompute_interval == 0:
            conv = _conv_self_fft(a)
            cur_max = conv.max()

        # temperature schedule (linear in log scale)
        T = start_temp * (end_temp / start_temp) ** (steps / steps_limit)

        # choose source weighted by a, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            steps += 1
            continue

        eps = a[src] * rng.random() * 0.4  # up to 40 % of source mass
        a_before = a.copy()

        conv[src:src + n] -= 2 * eps * a_before
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2 * eps * eps

        a[dst] += eps

"""
Incremental annealing optimizer for the max‑convolution objective.

The optimizer works in three phases inside the 1000 s time budget:

1. **U‑shaped initial distribution** – a deterministic shape that places
   most mass at the ends of the sequence and very little in the centre.
   This already reduces the central convolution peak compared with a uniform
   distribution.

2. **Greedy peak‑reduction** – repeatedly moves a fraction of the coefficient
   that contributes most to the current maximal convolution entry towards the
   globally smallest coefficient.  The convolution is updated incrementally
   (O(n) vector operations) instead of recomputing it from scratch.

3. **Long incremental simulated‑annealing** – mass‑transfer moves are performed
   many times (hundreds of thousands).  Each move updates the convolution
   using the same O(n) incremental formula, which allows a very large number of
   proposals while staying well inside the time limit.  A simple exponential
   cooling schedule is used; accepted moves that improve the maximal entry are
   kept and the best sequence seen so far is recorded.

The final sequence is normalised to sum 1 (the objective is scale invariant)
and returned as a plain Python list.  The routine prints progress messages
including intermediate maximal convolution values and the final score obtained
by ``evaluate_sequence``.
"""

    Returns the best sequence seen and its maximal convolution entry.

    # ------------------------------------------------------------------
    if time.time() < deadline:
        peak_deadline = time.time() + 0.10 * (deadline - time.time())
        a, max_after_peak = _greedy_peak_reduction(
            a, conv, peak_deadline, start_time, max_no_improve=1500
        )
        # recompute convolution for the (possibly) new best sequence
        conv = _conv_self_fft(a)
        if max_after_peak < best_max:
            best_max = max_after_peak
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] after peak‑reduction max={best_max:.8f}")

    # ------------------------------------------------------------------
    # 3) Long incremental simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, max_final = _annealing_incremental(
            a, conv, deadline, start_time, steps_limit=1_200_000
        )
        if max_final < best_max:
            best_max = max_final
            best_a = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] after annealing max={best_max:.8f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and reporting
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
