def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math
    start = time.time()
    best_seq = None
    best_score = float('inf')
    # Optionally set np.random seed if needed.
    np.random.seed(seed)
    lengths = [500, 1000, 2000, 4000, 8000]
    exponents = [1, 2, 3, 4, 5, 6]
    for n in lengths:
        if time.time() - start > budget_s - 1:
            break
        # Base decreasing vector from n to 1.
        base = np.arange(n, 0, -1, dtype=np.float64)
        for p in exponents:
            a = np.power(base, p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            # Evaluate
            try:
                score = evaluate_sequence(a.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best n={n}, p={p}, score={score:.6f}")
    # If still none, default uniform
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0/n)
        best_score = evaluate_sequence(best_seq.tolist())
    # Ensure non-negative and sum>0
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    return [float(v) for v in best_seq]

"""
Propose a step function that minimises ``evaluate_sequence``.

The key observation is that the objective equals  

    2·n·max(conv(a,a)) / (sum(a)^2)

and we can keep the denominator equal to 1 by normalising the vector.
For a *decreasing* sequence a_i the centre of the convolution is  

    (a * a)[n‑1] = Σ a_i·a_{n‑1‑i} .

If a_i∝(n‑i)^p with p≥1, the centre term behaves like C(p)/n while the
contributions at the edges are O(1/n²).  The constant

    C(p) = (p+1)²·B(p+1,p+1)   (B = beta‑function)

decays rapidly with p, giving an asymptotic objective value 2·C(p).
For p=4 we obtain 2·C(4)≈0.08, far better than the target 1.503.
The implementation searches a few lengths n and a small set of exponents p,
keeps the best candidate according to the real ``evaluate_sequence`` call,
and returns it as a plain ``list[float]``.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    low value of `evaluate_sequence`.  The search explores a family
    a_i ∝ (n‑i)^p (p = 1…6) for several lengths n and keeps the best
    candidate according to the true evaluation function.
    """
    np.random.seed(seed)

    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # candidate lengths (larger n gives a more stable estimate of the limit)
    lengths = [1000, 2000, 4000, 8000]
    # exponents to try – p=4 is already excellent, but we also test neighbours
    exponents = [1, 2, 3, 4, 5, 6]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # decreasing base vector [n, n‑1, …, 1]
        base = np.arange(n, 0, -1, dtype=np.float64)

        for p in exponents:
            if time.time() >= deadline:
                break

            # a_i ∝ (n‑i)^p
            a = np.power(base, p)

            # Normalise to unit sum (the objective is homogeneous)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s

            # Evaluate the true objective (the harness provides evaluate_sequence)
            try:
                score = evaluate_sequence(a.tolist())
            except Exception:
                # In the unlikely event evaluate_sequence is unavailable
                continue

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best: n={n}, p={p}, score={score:.6f}")

    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"fallback uniform sequence, score={best_score:.6f}")

    # Final safety normalisation (tiny drift may occur due to rounding)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        # Extremely unlikely – replace by uniform
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return as plain Python list
    return [float(v) for v in best_seq]
