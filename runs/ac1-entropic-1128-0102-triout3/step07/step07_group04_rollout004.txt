"""
Propose a non‑negative step function that (hopefully) beats the previous

import math

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used by the line‑search stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (a ≥ 0)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max weights
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)   # stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp          # shape (2n‑1,), sums to 1

        # gradient of Oτ  (see derivation in the comment above)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)            # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]          # size n
        grad = 4.0 * n * inner

        # Adam update on the *a* variables (followed by simplex projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑Adam (Adam on the log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _run_mirror_adam_softmax(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             tau0: float,
                             tau_end: float = 1e-12,
                             tau_decay: float = 0.9997,
                             step_size: float = 0.2,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *log* parameters w, where a = softmax(w).
    The update respects the simplex implicitly (mirror descent).
    """
    # initialise log‑weights safely (avoid log(0))
    w = np.log(np.clip(a0, 1e-16, None))
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    best_a = None
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # recover a from w
        exp_w = np.exp(w - np.max(w))          # stabilise
        a = exp_w / np.sum(exp_w)

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror‑adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # soft‑max over conv
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        p = exp_z / np.sum(exp_z)

        # gradient w.r.t. a (same as in classic Adam)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)
        inner = conv_rev[n - 1:n - 1 + n]
        grad_a = 4.0 * n * inner

        # chain rule: ∂O/∂w_i = a_i * (grad_a_i - ⟨a, grad_a⟩)
        avg = np.dot(a, grad_a)
        grad_w = a * (grad_a - avg)

        # Adam on w
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        w = w - step * m_hat

        # temperature schedule
        tau = max(tau_end, tau * tau_decay)

    # fallback – in the unlikely case that we never updated best_a
    if best_a is None:
        best_a = np.exp(w - np.max(w))
        best_a /= best_a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with line‑search (stage 2)
# ----------------------------------------------------------------------


def _run_subgradient_line_search(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float,
                                 max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on F(a)=2·n·max(conv(a)) with an explicit line‑search.
    The direction is the (zero‑mean) negative sub‑gradient; a small set of
    trial step lengths is evaluated and the best improvement is taken.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # a few factors to try for the step length (relative to the feasible max)
    step_factors = np.array([1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01])

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad‑line improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_max(a, conv)

        # make the direction zero‑mean so that sum stays unchanged
        direction = -grad + np.mean(grad)  # sum(direction)=0

        # maximal feasible step before any a_i would become negative
        neg = direction < 0
        if not np.any(neg):
            # direction points only outward – cannot move while staying feasible
            break
        alpha_max = np.min(a[neg] / (-direction[neg]))
        if alpha_max <= 0:
            break

        # evaluate a handful of trial steps
        best_local_a = a
        best_local_score = cur_score
        for factor in step_factors:
            alpha = factor * alpha_max
            a_candidate = a + alpha * direction
            # numerical safety – clip tiny negatives that may appear because of rounding
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate /= a_candidate.sum()  # re‑normalise (preserves feasibility)

            conv_cand = _conv_self(a_candidate)
            score_cand = 2.0 * n * conv_cand.max()
            if score_cand < best_local_score:
                best_local_score = score_cand
                best_local_a = a_candidate

        # accept the best candidate (might be the original a if none improved)
        a = best_local_a

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very light SA that moves a random fraction of mass between two cells."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and renormalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The method combines cheap exploratory Adam runs,
    a mirror‑Adam refinement on a larger grid, a sub‑gradient line‑search
    stage and a final light annealing phase.
    Returns a plain Python list of floats (sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin to avoid overshooting the budget
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter (previous best) – will be normalised automatically
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short cheap runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    # allocate roughly 30 % of the total budget to this stage
    stage0_deadline = start_time + (deadline - start_time) * 0.30
    restarts = 8
    for r in range(restarts):
        if time.time() >= stage0_deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # initialise τ from the current max‑convolution
        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 8.0, 1e-8)

        # half the restarts use classic Adam, half use mirror‑Adam
        if r % 2 == 0:
            a_opt, score = _run_adam_softmax(
                a0, n0, stage0_deadline, start_time,
                tau0=tau0,
                tau_end=1e-12,
                tau_decay=0.9994,
                step_size=0.22,
            )
        else:
            a_opt, score = _run_mirror_adam_softmax(
                a0, n0, stage0_deadline, start_time,
                tau0=tau0,
                tau_end=1e-12,
                tau_decay=0.9993,
                step_size=0.18,
            )

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback – extremely unlikely
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – upscale to a larger size and continue mirror‑Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)

    conv1 = _conv_self(a1)
    tau0_1 = max(conv1.max() * 8.0, 1e-10)

    # allocate around 35 % of the remaining budget to this stage
    remaining = deadline - time.time()
    stage1_deadline = time.time() + remaining * 0.35

    a1_opt, score1 = _run_mirror_adam_softmax(
        a1, n1, stage1_deadline, start_time,
        tau0=tau0_1,
        tau_end=1e-13,
        tau_decay=0.9993,
        step_size=0.12,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent with line‑search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = best_seq.shape[0]
        # reserve a few seconds for the final annealing stage
        subgrad_deadline = deadline - 2.0
        a2_opt, score2 = _run_subgradient_line_search(
            best_seq, n2, subgrad_deadline, start_time)
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – very light simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-7,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Debug output: final score according to the provided evaluator
    try:
        # The evaluator is assumed to be present in the global namespace.
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
