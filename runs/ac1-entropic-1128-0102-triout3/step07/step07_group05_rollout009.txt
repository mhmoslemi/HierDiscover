Implementation typical algorithm:

def project_onto_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Now gradient_descent:

def gradient_descent(a, n, deadline, start_time, best_seq, best_score):
    # Parameters
    beta = 0.9
    step0 = 0.3
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        active = np.where(conv >= M * (1 - 1e-12))[0]

        # compute subgradient
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1 - beta) * g
        step = step0 / math.sqrt(it)

        a_new = a - step * momentum
        a_proj = _project_simplex(a_new)

        # compute new conv
        conv_new = _conv_self(a_proj)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_seq = a_proj.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: iter={it:,}, score={score_new:.6f}")

        a = a_proj

"""Hybrid optimizer for the step‑function attack.

The method consists of three stages that respect the 1000 s budget.

1. **Fast Adam on a smooth max‑convolution** – a short Adam run on the
   log‑sum‑exp approximation of the max‑convolution, keeping the vector on
   the probability simplex via a soft‑max parametrisation.
2. **Heavy‑weight projected sub‑gradient descent** – the exact max‑convolution
   is minimised using a momentum sub‑gradient method with Euclidean projection
   onto the simplex.  The loop runs as long as time permits (up to millions
   of iterations).
3. **Greedy incremental transfer (deterministic annealing)** – a mass is
   transferred from the index that contributes most to the current maximal
   convolution entry toward the index that contributes least.  The convolution
   is updated incrementally in O(n) using NumPy slice operations, so many
   millions of moves can be performed in the remaining time.

The best sequence seen throughout the whole run is returned (normalised to
sum = 1).  All numbers are non‑negative, satisfying the checks in
``evaluate_sequence``.  The algorithm starts from the supplied
``height_sequence_1`` when available, otherwise uses a random start.  """

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam on smoothed max‑convolution (short run)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Short Adam run on the log‑sum‑exp surrogate.
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # a compact schedule – we only need a few dozen epochs
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    iters_per_alpha = 3000   # much lighter than the original run

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max to keep a on the simplex
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # objective and gradient (smooth)
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t. logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 500 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                          f"α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (long run)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          beta: float = 0.9,
                          step0: float = 0.35) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Momentum sub‑gradient descent on the true max‑convolution.
    Returns the best sequence and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1

        # --- sub‑gradient of the max entry ---------------------------------
        M = conv.max()
        # accept any entry that is numerically equal to the maximum
        active = np.where(conv >= M * (1.0 - 1e-12))[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # --- momentum update -----------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g

        # --- step size -------------------------------------------------------
        step = step0 / math.sqrt(it)

        # --- projected update ------------------------------------------------
        a_tmp = a - step * momentum
        a = _project_onto_simplex(a_tmp)

        # --- recompute convolution -------------------------------------------
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: iter={it:,}, score={cur_score:.6f}")

        # occasional early exit when improvements become rare
        if it % 200_000 == 0 and time.time() >= deadline:
            break

    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Incremental greedy transfer (deterministic annealing)
# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer `eps` mass from `src` to `dst`, updating `conv` in‑place.
    The update uses the old values of `a` (before the move) exactly as in
    the original incremental scheme.
    """
    n = a.shape[0]

    # the convolution updates – everything is done with the *old* a
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # now modify the probability vector
    a[src] -= eps
    a[dst] += eps


def _greedy_annealing(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy mass‑transfer loop.
    The index that contributes most to the current maximal convolution entry
    loses a fraction of its mass, which is given to the index that contributes
    least.  Moves are accepted only if they strictly improve the objective.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1

        # compute (sub‑)gradient of the active maximal entries
        M = conv.max()
        active = np.where(conv >= M * (1.0 - 1e-12))[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] < 1e-14:
            # simple random fallback when gradient is flat
            src = np.random.choice(n, p=a)
            dst = np.random.randint(n)
            if src == dst or a[src] < 1e-14:
                continue

        # choose a modest transfer fraction (15 % of the source mass)
        eps = a[src] * 0.15
        if eps <= 0.0:
            continue

        # ----- apply move ----------------------------------------------------
        _apply_transfer(a, conv, src, dst, eps)

        new_M = conv.max()
        new_score = 2.0 * n * new_M
        if new_score < cur_score:
            # improvement – keep it
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={new_score:.6f}")
        else:
            # reject – revert the move (swap src/dst)
            _apply_transfer(a, conv, dst, src, eps)
            # cur_score unchanged

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine follows the three‑stage hybrid scheme described in the
    module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = np.maximum(starter, 0.0)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            if a0.shape[0] != n:
                a0 = _upsample(a0, n)
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[0.00s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short Adam on smoothed objective
    # ------------------------------------------------------------------
    a_adam, best_score, best_seq = _adam_phase(
        best_seq, n, deadline, start_time, best_score, best_seq
    )

    # ------------------------------------------------------------------
    # Phase 2 – heavy‑weight projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        gd_deadline = time.time() + 0.55 * remaining   # a bit more than half
        a_gd, best_score, best_seq = _run_gradient_descent(
            best_seq, n, gd_deadline, start_time, best_seq, best_score,
            beta=0.9, step0=0.35
        )
        if best_score < best_score:
            # (the function already updates best_score)
            pass

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, best_score = _greedy_annealing(
            best_seq, n, deadline, start_time, best_seq, best_score
        )
        best_seq = a_final

    # ------------------------------------------------------------------
    # Normalise and final safety check
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
