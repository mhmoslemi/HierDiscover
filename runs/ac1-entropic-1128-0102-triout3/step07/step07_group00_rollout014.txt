def _run_subgradient_multi(a0, n, deadline, start_time, best_score_initial, top_k=5, tol_ratio=0.02):

    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)

        # Find indices within (1 - tol) * max
        thresh = max_val * (1 - tol_ratio)
        idxs = np.where(conv >= thresh)[0]
        # maybe limit to top_k largest indices
        if idxs.size > top_k:
            # sort by conv descending and take top_k
            idxs = idxs[np.argsort(-conv[idxs])[:top_k]]
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_range = np.arange(lo, hi + 1, dtype=int)
            grad[idx_range] += 2.0 * a[k - idx_range]
        grad /= len(idxs)  # average

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        it += 1
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a,best_score

def _lp_step(f, M):
    n = len(f)
    A_data = []
    A_rows = []
    A_cols = []
    # We'll build sparse matrix
    for i in range(2*n - 1):
        lo = max(0, i - (n-1))
        hi = min(n-1, i)
        # columns j from lo..hi; correspond to g_j at those j
        # the coefficient for g_j is f_{i-j}
        for j in range(lo, hi+1):
            coeff = f[i - j]
            if coeff != 0:
                A_data.append(coeff)
                A_rows.append(i)
                A_cols.append(j)
    A = scipy.sparse.csr_matrix((A_data, (A_rows, A_cols)), shape=(2*n-1, n))

    # constraints: A @ g <= M
    bounds = [(0, None)] * n
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)   # SciPy's linprog minimizes c^T x
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

"""
Propose a near‑optimal non‑negative step‑function that minimises
``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initialisation** – use the global ``height_sequence_1`` if it exists,
   otherwise start from a random probability vector of length 4096.
2. **Linear‑programming boost** – down‑sample the vector to 1024 entries,
   solve a tiny LP that maximises the total mass under the current
   convolution bound, blend the result back and up‑sample.
3. **Phase 1 – Adam on a smooth‐max surrogate** – many iterations of Adam
   on the log‑sum‑exp approximation of ``max(conv(a,a))``.
4. **Phase 2 – Multi‑peak sub‑gradient descent** – instead of using a
   single maximising index, the gradient is averaged over the top few
   convolution entries (more stable descent).
5. **Phase 3 – Simulated‑annealing polish** – a long temperature‑controlled
   random mass‑transfer walk.
6. **Post‑processing** – enforce symmetry, a few cheap Gaussian smoothing
   passes and a last tiny random‑restart sub‑gradient run.

All phases respect the global time budget (≈1000 s).  The routine returns
a plain Python list of non‑negative heights whose sum is 1, which is
exactly the scale used by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional linear‑programming boost
try:
    import scipy.optimize
    import scipy.sparse
    _HAS_SCIPY = True
except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a via FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum=1, v≥0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # all zero – return uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 6000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best iterate (lowest true objective) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0

    # FFT length for all convolutions in this phase
    L = _next_pow2(3 * n - 2)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # current objective (exact, for logging only)
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----------- smooth‑max weights ------------
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stable soft‑max
            w /= w.sum()
            w_rev = w[::-1]

            # ----------- gradient of smooth max ----------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # -------------- Adam update -----------------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)   # simple decay
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Multi‑peak sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_multi(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_score_initial: float,
                           top_k: int = 5,
                           tol_ratio: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the average gradient of the
    top‑``top_k`` convolution entries (within ``tol_ratio`` of the max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.94
    step0 = 0.25
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad multi improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # indices whose convolution value is close to the maximum
        thresh = max_val * (1.0 - tol_ratio)
        cand = np.where(conv >= thresh)[0]
        if cand.size > top_k:
            # keep only the largest ``top_k`` entries
            cand = cand[np.argsort(-conv[cand])[:top_k]]

        # average sub‑gradient over selected indices
        grad = np.zeros_like(a)
        for k in cand:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(cand))

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : Simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealing with exponential temperature decay.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # source weighted by mass
        dst = rng.integers(n)             # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming boost (small‑scale)
# ----------------------------------------------------------------------
def _lp_maximize_sum(f: np.ndarray, M: float):
    """
    Solve   max sum(g)   subject to   conv(f,g) ≤ M  and g ≥ 0.
    The problem is linear because conv(f,g) is linear in g.
    Returns the optimal g (as a 1‑D NumPy array) or ``None`` on failure.
    """
    if not _HAS_SCIPY:
        return None

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # build the constraint matrix A where A @ g = conv(f,g)
    for i in range(2 * n - 1):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        # column indices for this row
        cols_row = np.arange(lo, hi + 1, dtype=int)
        # coefficients are f[i - j]
        coeffs = f[i - cols_row]
        rows.extend([i] * len(cols_row))
        cols.extend(cols_row.tolist())
        data.extend(coeffs.tolist())

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                     # minimise –sum(g)  == maximise sum(g)
    bounds = [(0, None)] * n
    b = np.full(2 * n - 1, M)

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None
