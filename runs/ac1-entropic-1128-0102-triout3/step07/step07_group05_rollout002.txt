"""
Hybrid optimizer for the step‑function attack.

The algorithm follows three stages, all respecting the supplied wall‑time
budget (≈ 1000 s).

1.  Adam on a smooth log‑sum‑exp approximation of the maximal convolution.
    The parameters are kept on the probability simplex via a soft‑max
    (logits).  The smoothing factor α is increased gradually.

2.  Projected sub‑gradient descent on the exact objective
    (2·n·max(conv(a,a))) with Euclidean projection onto the simplex.
    This phase runs until the time budget forces us to stop.

3.  A long targeted simulated‑annealing phase.
    *Mass is transferred* from a source index to a destination index.
    The source is chosen preferentially from the indices that contribute
    to the current maximal convolution entry, which gives a much stronger
    descent direction than a completely random move.
    The convolution is updated incrementally in O(n) using slice operations;
    we periodically (every 200 k moves) recompute the full convolution with
    FFT to avoid drift.
    The temperature schedule is exponential and the acceptance rule is the
    standard Metropolis criterion.

All numbers are kept non‑negative and finally renormalised to sum = 1, so
the sequence passes the security checks of ``evaluate_sequence``.
The best sequence encountered during the whole run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # centre the result (the resulting array has length n+len(w_rev)-1)
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w
    else:
        # All zero case – return uniform distribution
        return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimisation (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Runs Adam on the smoothed log‑sum‑exp objective.
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # Gradually tighten the smooth approximation
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 12000   # a bit more than the previous run

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max to obtain a probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smooth objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            # (the factor 2·n is omitted here – only used for diagnostics)

            # gradient of the smooth objective w.r.t a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t logits (soft‑max chain rule)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits (ensure feasibility)
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv(a,a)) with momentum.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing (targeted moves)
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               max_eps: float = 0.30,
                               recalibrate_interval: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Targeted annealing where mass is transferred between indices.
    The source is often taken from the set of indices that contribute
    to the current maximal convolution entry.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline:
        it += 1

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / (deadline - start_time + 1e-9))

        # ------------------------------------------------------------
        # choose source – either from the current max‑conv window
        # or simply weighted by mass (fallback)
        # ------------------------------------------------------------
        if rng.random() < 0.55:
            kmax = np.argmax(conv)
            i_low = max(0, kmax - (n - 1))
            i_high = min(n - 1, kmax)
            idx_window = np.arange(i_low, i_high + 1, dtype=int)
            contrib = a[idx_window] * a[kmax - idx_window]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                probs = contrib / total_contrib
                src = rng.choice(idx_window, p=probs)
            else:
                src = rng.choice(n, p=a)  # fallback
        else:
            src = rng.choice(n, p=a)  # weighted by current mass

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * max_eps
        if eps <= 0:
            continue

        # ----- apply move (using the *old* a for the convolution update) -----
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            score = new_score
            # accept – nothing else to do
        else:
            # ----- reject: restore previous state (note a now holds the old values) -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps    # *** bug‑fix: add back the full 2·ε² ***

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

        # occasional full recomputation to avoid drift
        if it % recalibrate_interval == 0:
            conv = _conv_self(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal recalibrate: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The method mixes Adam smoothing, exact projected sub‑gradient polishing and
    a long targeted simulated‑annealing, all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin at the end
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    n = 4800
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = starter / starter.sum()
            if a0.shape[0] != n:
                a0 = _upsample(a0, n)
        else:
            raise Exception
    except Exception:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # track the best sequence found so far
    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[0.0s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed objective
    # ------------------------------------------------------------------
    a_adam, best_score, best_seq = _adam_phase(
        best_seq, n, deadline, start_time, best_score, best_seq
    )

    # ------------------------------------------------------------------
    # Phase 2 – Exact projected sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # use half of the remaining budget for polishing, half for annealing
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Extended targeted annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_incremental_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_eps=0.30,
            recalibrate_interval=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final safety check (optional)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
