"""
Hybrid LP + projected sub‑gradient optimiser.

The search works on a modest problem size (n≈300) so that many
iterations fit into the ~1000 s budget.

1. Initialise a random probability vector a (∑a=1, a_i≥0).
2. Repeatedly:
   * Compute M = max(a∗a).
   * Build the linear constraints   (a∗g)_k ≤ M   (k=0..2n‑2)   which are linear
     in g.
   * Solve the LP   maximise 1ᵀg   subject to the constraints and g≥0.
   * Normalise g to sum = 1 and move a a little towards g
     (convex combination with a small step‑size).
   * Keep the best a found so far (according to evaluate_sequence).

The LP step eliminates the worst “peak” of the convolution.
After the LP loop a short projected sub‑gradient descent on the exact
max‑convolution is performed to polish the solution.

All vectors are kept on the probability simplex, hence the scaling
invariance of the objective is respected automatically.
The routine respects the global time budget and returns the best
sequence discovered.
"""

import time
import math
import numpy as np
import scipy.sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full, length 2·n‑1)."""
    return np.convolve(a, a)


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k  with respect to a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    max_val = conv.max()
    # consider all indices whose value is essentially maximal
    rel_tol = 1e-12
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


# ----------------------------------------------------------------------
# Linear‑programming step (maximise mass under the current bound)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Given the current vector a and an upper bound M on its self‑convolution,
    solve the LP  maximise 1ᵀg   s.t.   (a∗g)_k ≤ M   for all k,   g ≥ 0.
    The returned g is normalised to sum = 1.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # build the (2n‑1)×n constraint matrix A where
    #   (A·g)_k = Σ_j a_{k‑j}·g_j   =   (a∗g)_k
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # maximise 1ᵀg  ⇔ minimise –1ᵀg
    c = -np.ones(n)
    ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # use the high‑performance HiGHS interior‑point solver
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fallback: return the current vector unchanged
        return a.copy()
    g = res.x
    if g.sum() == 0:
        return a.copy()
    return _project_to_simplex(g)

# Main optimisation driver
# ----------------------------------------------------------------------
