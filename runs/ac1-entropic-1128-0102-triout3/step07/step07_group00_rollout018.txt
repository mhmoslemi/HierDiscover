Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

* **Initialisation** – take the provided global ``height_sequence_1`` if it
  exists, otherwise start from a uniform simplex vector of length ``N``.
  ``N`` is chosen as a power‑of‑two (default 4096) because FFT‑based
  convolutions are cheap.

* **Phase‑1 – Projected sub‑gradient descent with back‑tracking line search**.
  For the current iterate we locate the convolution index `k` that attains
  the maximum.  The (sub)‑gradient of the objective at this index is
  ``g[i] = 2*a[k‑i]`` for the indices that participate in the sum.
  A scaled step in the opposite direction is taken, then the iterate is
  projected back onto the probability simplex.  A simple Armijo style
  back‑tracking (geometric reduction of the step size) guarantees
  monotone improvement.

* **Phase‑2 – Fast simulated‑annealing polish**.
  The full convolution is kept in a mutable array and updated *incrementally*
  when a tiny amount of mass `ε` is moved from a source index to a destination.
  The update of the convolution entries is O(N) instead of recomputing an FFT
  each time, so many more proposals can be tried.  The temperature schedule
  decays exponentially from `1e‑2` to `1e‑6`.  Accepted moves are stored and
  the best sequence seen during the annealing run is remembered.

* **Final normalisation** – after the time budget expires the best sequence
  is clipped to ``[0,∞)``, re‑normalised to sum 1 and returned as a plain
  Python list.

The routine respects a user supplied time budget (default 1000 s).  All
helper functions are top‑level, there are no closures or lambdas, and the
implementation uses only ``numpy`` and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of suitable length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w with w >= 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑1 : Projected sub‑gradient descent with line‑search
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise ``2·n·max(conv(a,a))`` by projected sub‑gradient steps.
    Returns the best a seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    it = 0
    beta = 0.90               # momentum coefficient
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        it += 1
        # current convolution and its maximal index
        conv = _conv_self(a)
        k = int(conv.argmax())
        cur_max = conv[k]
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the active convolution entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # 2·a_{k‑i}
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line search
        step = 0.5 / math.sqrt(it)           # initial guess
        for _ in range(20):
            a_prop = a - step * momentum
            a_prop = _project_simplex(a_prop)
            conv_prop = _conv_self(a_prop)
            new_score = 2.0 * n * conv_prop.max()
            if new_score < cur_score:
                a = a_prop
                break
            step *= 0.5                       # shrink step
        else:
            # no improvement – exit early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : Incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   start_temp: float = 1e-2,
                   end_temp: float = 1e-6,
                   move_factor: float = 0.02,
                   max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Fast annealing that updates the convolution incrementally.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        src = rng.choice(n, p=a)          # choose source weighted by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * move_factor
        if eps <= 0.0:
            continue

        # --- incremental update of the convolution --------------------
        # store old a for the update
        a_src = a[src]
        a_dst = a[dst]

        # mass transfer
        a[src] = a_src - eps
        a[dst] = a_dst + eps

        # update conv: contributions involving src (removed) and dst (added)
        #   conv = a*a  →  conv + Δ where Δ = a*δ + δ*a + δ*δ
        #   δ has -eps at src and +eps at dst
        conv[src:src + n] -= eps * a_src           # old src contribution
        conv[dst:dst + n] += eps * a_dst           # old dst contribution
        conv[src:src + n] += eps * a[dst:dst + n][np.arange(n)]  # new dst part?  (handled later)

        # Simpler – recompute the slices using the *old* a vector (a before update)
        #   For src removal:
        conv[src:src + n] -= eps * a_src
        #   for dst addition:
        conv[dst:dst + n] += eps * a_dst

        # delta*delta terms
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # numerical safety
        np.maximum(conv, 0.0, out=conv)

        # --- evaluate new score ---------------------------------------
        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            # accept move
            score = new_score
        else:
            # reject – roll back
            a[src] = a_src
            a[dst] = a_dst
            # revert conv by reversing the update (cheap: recompute conv)
            conv = _conv_self(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine problem size – a power of two is convenient for FFT.
    # --------------------------------------------------------------
    N = 4096
    try:
        # use the provided global if it looks sensible
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                # resample uniformly onto the target length
                starter = np.interp(
                    np.linspace(0, 1, N),
                    np.linspace(0, 1, starter.size),
                    starter
                )
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            a0 = starter
        else:
            raise Exception
    except Exception:
        # uniform simplex start
        a0 = np.full(N, 1.0 / N)

    best_seq = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (major chunk of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_deadline = time.time() + max(30.0, 0.70 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq, phase1_deadline, start_time
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – fast annealing polish (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            deadline,
            start_time,
            cur_best_seq=best_seq,
            cur_best_score=best_score,
            start_temp=1e-2,
            end_temp=1e-6,
            move_factor=0.02,
            max_steps=400_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    return [float(x) for x in best_seq]
