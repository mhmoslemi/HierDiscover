"""
Propose a non‑negative step function (probability vector) that minimizes
evaluate_sequence.  The algorithm consists of:

1.  Gaussian initialisation – a smooth discrete normal shape on the chosen
    length (default 9600) and normalised to sum = 1.
2.  Soft‑max Adam optimisation – a smooth approximation of the max‑convolution
    is minimised with an Adam‑style adaptive step and an exponential temperature
    schedule.
3.  Tight sub‑gradient descent – the true max‑convolution is tackled using an
    active‑set (all points within a small relative tolerance of the current
    maximum).  A Nesterov‑style momentum with a diminishing step size refines the
    result.
4.  Long simulated‑annealing – tiny mass‑transfer moves (or occasional local
    smoothing) are accepted according to a geometric cooling schedule.
    The annealing phase receives the bulk of the time budget.
5.  Optional final sub‑gradient fine‑tuning if time permits.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1).  The routine respects the provided wall‑clock budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (full length)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFTs."""
    n = a.shape[0]
    m = b.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    c = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself (full length = 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate seq to length new_n and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Objective value (exact – 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gaussian initialiser
# ----------------------------------------------------------------------
def _gaussian_init(n: int, sigma_frac: float = 0.18) -> np.ndarray:
    """Return a normal‑shaped vector of length n, normalised to sum = 1."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth surrogate of max‑convolution)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Stable soft‑max weights for a given temperature."""
    maxc = conv.max()
    # Shift by maxc to avoid overflow, then scale
    scaled = (conv - maxc) / temperature
    # Numerical stability: subtract max(scaled) (which is 0)
    exps = np.exp(scaled - np.max(scaled))
    w = exps / exps.sum()
    return w


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the self‑convolution (to be subtracted)."""
    w = _softmax_weights(conv, temperature)
    rev_a = a[::-1]
    grad_full = _conv_fft(rev_a, w)
    n = a.shape[0]
    # Extract the part that corresponds to the gradient w.r.t. a
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max optimisation with Adam (projected onto simplex)
# ----------------------------------------------------------------------
def _softmax_optimize_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    T_start: float = 0.1,
    T_end: float = 1e-5,
    T_decay_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """Adam‑style projected optimisation of the soft‑max surrogate."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax‑Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (exponential decay)
        T = T_start * (T_end / T_start) ** (it / T_decay_iters)

        # gradient of the smoothed max
        grad = _softmax_gradient(a, conv, T)

        # Adam update (descent – we subtract the step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with active set
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.15,
    beta: float = 0.9,
    rel_eps: float = 0.001,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # ----- active set -----
        eps = rel_eps * max_val
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Nesterov‑style momentum & diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 0.4,
    end_temp: float = 1e-7,
    max_steps: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Metropolis‑type walk that moves a small fraction of mass."""
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # current score
    score = _objective(a)

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # ---- propose move ----
        if np.random.rand() < 0.85:          # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            # transfer up to a fraction that shrinks with temperature
            max_frac = min(0.4, T / start_temp + 0.05)
            eps = a[i] * np.random.rand() * max_frac
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                 # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode="same")
            a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 9600                     # final resolution
    # allocate ~40 % of the time to gradient phases, 60 % to annealing
    total_time = deadline - start_time
    grad_time = total_time * 0.40
    anneal_time = total_time - grad_time

    # ------------------------------------------------------------------
    # 1️⃣  Initialise with a smooth Gaussian shape
    # ------------------------------------------------------------------
    a0 = _gaussian_init(n, sigma_frac=0.18)

    # ------------------------------------------------------------------
    # 2️⃣  Soft‑max Adam optimisation (gradient phase)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + grad_time * 0.55   # ~22 % of total budget
    a_opt, score_opt = _softmax_optimize_adam(
        a0, n,
        deadline=phase1_deadline,
        start_time=start_time,
        lr0=0.5,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
        T_start=0.12,
        T_end=5e-5,
        T_decay_iters=80000,
    )

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient descent (tightening phase)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + grad_time * 0.35   # remaining gradient budget
        a_opt, score_opt = _subgradient_optimize(
            a_opt, n,
            deadline=phase2_deadline,
            start_time=start_time,
            step0=0.12,
            beta=0.9,
            rel_eps=0.001,
        )

    # ------------------------------------------------------------------
    # Keep the best so far
    # ------------------------------------------------------------------
    best_seq = a_opt.copy()
    best_score = score_opt

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing refinement (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _annealing_optimize(
            best_seq, n,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=0.4,
            end_temp=1e-7,
            max_steps=800_000,          # generous but bounded; the loop checks the deadline
        )
        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Tiny final sub‑gradient polishing (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        final_deadline = deadline
        a_opt, score_opt = _subgradient_optimize(
            best_seq, n,
            deadline=final_deadline,
            start_time=start_time,
            step0=0.08,
            beta=0.9,
            rel_eps=5e-4,
        )
        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (tiny drift may have occurred)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # Return a plain Python list for the harness
    print(f"[{time.time() - start_time:6.2f}s] DONE – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
