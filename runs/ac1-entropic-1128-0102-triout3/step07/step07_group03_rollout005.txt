"""
Propose a step‑function (non‑negative sequence) with a low
self‑convolution peak.

Algorithm
---------
1. Warm‑start from the previously found best sequence (if available).
2. Exploration phase – many short simulated‑annealing runs starting from
   random perturbations of the current best.  Moves transfer a fraction
   of the mass from a “hot” index (large contribution to the current
   maximal convolution entry) to a “cold” index.  The temperature
   schedule is aggressive (1 → 1e‑6) to escape shallow local minima.
3. Refined convex‑subgradient descent.  The objective 2·n·max(conv) is a
   convex, piecewise‑linear function of the vector; its sub‑gradient at
   the active shift is 2·a_{s‑i}.  A Polyak step size
   (step = (f(x)‑f*)/‖g‖²) with the known theoretical lower bound
   f* = 2·n/(2·n‑1) yields fast progress.
4. Final intensive local‑search (peak‑shaving) with a slower cooling
   schedule and larger mass transfers.
5. Return the best probability vector seen (normalised to sum = 1).

All helpers are defined at the top level; the routine respects a 1000 s
budget and never returns an empty or invalid sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers ---------------------------------------------------
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def _score_of(a: np.ndarray) -> float:
    """Score = 2·n·max(conv) (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


def _active_shift_info(a: np.ndarray):
    """Return (gradient, idx_of_max, max_val) for the active shift."""
    n = a.shape[0]
    conv = _conv_self(a)
    idx = int(np.argmax(conv))
    max_val = conv[idx]
    i = np.arange(n)
    j = idx - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[i[mask]] = 2.0 * a[j[mask]]
    return grad, idx, max_val


# ----------------------------------------------------------------------
# Simulated annealing – short exploratory runs -------------------------
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   max_seconds: float,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing aiming to reduce the max‑convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score_of(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (aggressive)
    T_start = 1.0
    T_end = 1e-6

    it = 0
    while time.time() < deadline and (time.time() - start_time) < max_seconds:
        it += 1
        # current active shift
        conv = _conv_self(a)
        idx = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[idx]

        # contributions of each index to the active shift
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        # donor – proportional to contribution (hot spots)
        if contrib.sum() > 0.0:
            prob = contrib / contrib.sum()
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)

        # recipient – from the cold tail
        low_thr = np.percentile(contrib, 20)  # bottom 20 %
        low_cand = np.where(contrib <= low_thr)[0]
        if low_cand.size == 0:
            recipient = np.random.randint(n)
        else:
            recipient = np.random.choice(low_cand)

        if donor == recipient or a[donor] <= 1e-15:
            continue

        # move up to 40 % of the donor's mass
        eps = a[donor] * np.random.rand() * 0.4
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[recipient] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        accept = delta < 0.0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Polyak sub‑gradient descent (convex) ---------------------------------
# ----------------------------------------------------------------------
def _run_subgradient_polyak(a0: np.ndarray,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with Polyak step size."""
    a = a0.copy()
    n = a.shape[0]
    lower_bound = 2.0 * n / (2.0 * n - 1.0)          # theoretical infimum of the score
    best_a = a.copy()
    best_score = _score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, idx, _ = _active_shift_info(a)
        cur_score = _score_of(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] polyak improve: iter={it}, score={cur_score:.6f}")

        # Polyak step size (note cur_score >= lower_bound)
        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            break
        step = (cur_score - lower_bound) / (norm2 + 1e-12)
        step = max(step, 0.0) * 0.5            # a small damping factor for stability
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Helper: generate a fresh start vector --------------------------------
# ----------------------------------------------------------------------
def _random_start(base: np.ndarray, n: int) -> np.ndarray:
    """Blend the current best with a fresh random vector."""
    rand_vec = np.random.rand(n)
    mix = np.random.rand()
    a0 = mix * base + (1.0 - mix) * rand_vec
    return _project_simplex(a0)
