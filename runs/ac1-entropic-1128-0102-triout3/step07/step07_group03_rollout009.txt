"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search consists of three layers:

1. **Progressive up‑sampling + Adam‑style mirror descent** – start from a short
   vector (or from the supplied ``height_sequence_1``), repeatedly up‑sample
   and run a few hundred Adam updates on the *sub‑gradient* of the active
   convolution entry.  The iterate is always projected onto the probability
   simplex, guaranteeing non‑negativity and unit L¹ norm.

2. **Aggressive peak‑shaving local walk** – after the global optimisation a fast
   O(n) incremental update of the full convolution is used.  Mass is moved
   from indices that contribute most to the current maximal convolution entry
   to indices that contribute little.  The exact effect on the convolution is
   applied using the analytical formula

        conv' = conv − 2·ε·shift(donor) + 2·ε·shift(recipient)
                + ε²·δ_{2·donor} + ε²·δ_{2·recipient}
                − 2·ε²·δ_{donor+recipient},

   which allows millions of cheap moves within the remaining time budget.

3. **Random restarts** – when time permits a few short, cheap restarts with
   different lengths are performed; the best result over all phases is kept.

The whole routine respects the 1000 s budget, prints progress, and finally
returns a plain Python list (the harness expects a list).
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling (preserves mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` by linear interpolation, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  init_lr: float = 0.6,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam updates on the active shift; always stay on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(conv.argmax())
        cur_score = 2.0 * n * conv[idx_max]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # sub‑gradient of the active entry (negative direction for descent)
        grad = np.zeros_like(a)
        j_arr = idx_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        grad[mask] = a[j_arr[mask]]                     # ∇ max = a_partner
        g = -grad                                        # descent direction

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = init_lr / (np.sqrt(v_hat) + 1e-12)
        a = a + step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive local search with exact O(n) convolution updates
# ----------------------------------------------------------------------
def _local_search_incremental(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    """Peak‑shaving walk that uses the analytical update of the convolution."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    # ingredient for the “hot‑spot / cold‑spot” selection
    low_percentile = 20          # bottom 20 % are considered cold
    max_transfer = 0.6           # up to 60 % of donor's mass may be moved

    while time.time() < deadline:
        it += 1
        idx = int(conv.argmax())
        cur_score = 2.0 * n * conv[idx]

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] local improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # contribution of each index to the current maximal entry
        i_arr = np.arange(n)
        j_arr = idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – pick proportionally to contribution (hot spot)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – pick from a low‑contribution region (cold spot)
        thr = np.percentile(contrib, low_percentile)
        low_idxs = np.where(contrib <= thr)[0]
        if low_idxs.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idxs)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # amount to move
        eps = a[i_donor] * np.random.rand() * max_transfer

        # store current a for the exact update
        a_before = a.copy()

        # ----- exact convolution update (linear part) -----
        # subtract donor contribution
        conv[i_donor : i_donor + n] -= 2.0 * eps * a_before
        # add recipient contribution
        conv[i_rec : i_rec + n] += 2.0 * eps * a_before

        # ----- quadratic correction (δ‑terms) -----
        pos = 2 * i_donor
        if pos < conv.size:
            conv[pos] += eps * eps
        pos = 2 * i_rec
        if pos < conv.size:
            conv[pos] += eps * eps
        pos = i_donor + i_rec
        if pos < conv.size:
            conv[pos] -= 2.0 * eps * eps

        # ----- apply the mass move to a -----
        a[i_donor] -= eps
        a[i_rec] += eps

        # occasional renormalisation to avoid drift
        if it % 1000 == 0:
            s = a.sum()
            if s > 0:
                a /= s
                conv = _conv_self(a)   # recompute full conv to wipe accumulated error

    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The algorithm combines progressive Adam optimisation, an exact peak‑shaving
    local walk and a few cheap random restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Warm‑start (if a previous best is available)
    # --------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Progressive up‑sampling + Adam optimisation
    # --------------------------------------------------------------
    # a ladder of lengths – we finish at 9600 (the length used previously)
    stage_lengths = [1024, 2048, 4096, 8192, 9600]
    total_stage_time = (deadline - start_time) * 0.60   # 60 % of budget
    # time left for the final walk and possible restarts
    remaining_time = deadline - start_time - total_stage_time

    cur_seq = None
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time proportional to the stage length
        stage_fraction = n / sum(stage_lengths)
        stage_budget = max(2.0, total_stage_time * stage_fraction)
        stage_deadline = time.time() + stage_budget

        # initialise the vector for this stage
        if cur_seq is not None:
            a0 = _upsample(cur_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Adam sub‑gradient optimisation on the active shift
        a_opt, score_opt = _adam_subgrad(a0, n, stage_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            cur_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # update remaining stage budget (in case we finished early)
        elapsed = time.time() - start_time
        total_stage_time = max(0.0,
                               deadline - start_time - remaining_time - elapsed)

    # --------------------------------------------------------------
    # Aggressive peak‑shaving local walk on the final length
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        # reserve half of the remaining budget for the local walk
        local_deadline = deadline - (remaining_time * 0.5)
        a_loc, score_loc = _local_search_incremental(
            best_seq, n_final, local_deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # --------------------------------------------------------------
    # Cheap random restarts (if any time left)
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:    # keep a couple of seconds as safety
        # pick a length in the same range but not always the same
        n = int(np.random.choice([2048, 4096, 6144, 8192, 9600]))
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # a very short Adam run (few hundred iterations ≈ couple of seconds)
        sub_deadline = time.time() + min(4.0, deadline - time.time())
        a_opt, score_opt = _adam_subgrad(a0, n, sub_deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] random‑restart improve: "
                  f"n={n}, score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may appear)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list for the harness
    return [float(x) for x in best_seq]
