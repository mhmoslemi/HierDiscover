"""
Propose a step‑function with a lower bound for ``evaluate_sequence``.

The search consists of three stages:

1️⃣  Exponentiated‑gradient warm‑up on a smooth approximation of the max
    (identical to the reference implementation).

2️⃣  Projected sub‑gradient descent on the exact max (again as before).

3️⃣  **Massive incremental refinement** – instead of recomputing the whole
    self‑convolution after every tiny move, we update the convolution
    array analytically (Δconv[k] = 2·ε·(a_{k‑dst} – a_{k‑src})).  This makes a
    single iteration O(n) instead of O(n log n) and allows dozens of
    millions of moves in the remaining time budget.  The source index is
    chosen proportionally to its contribution to the current maximal
    convolution entry, while the destination is biased towards indices that
    contribute little.  An epsilon proportional to the source mass (with a
    schedule that shrinks over time) is moved; the move is kept only if the
    maximal convolution value strictly drops.

The whole routine respects the 1000 s budget, returns the best sequence
found and works for any length (the default is 4800, but the code can be
changed easily).  All helper functions are top‑level and no I/O is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a using all entries whose
    value lies within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Phase‑1 : exponentiated‑gradient on a smoothed max
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smoothed‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# ----------------------------------------------------------------------
def _run_incremental_refine(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eps_start: float = 0.04,
                           eps_end: float = 0.002,
                           rel_improve: float = 1e-12):
    """
    Perform millions of tiny mass‑shifts away from the coordinates that
    contribute most to the current maximal convolution entry.
    Updates of the convolution are done analytically, which makes a single
    iteration O(n) and allows a huge number of iterations inside the budget.
    """
    a = a0.copy()
    conv = _conv_self(a)                 # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()

    start_refine = time.time()
    total_time = deadline - start_refine
    it = 0

    while time.time() < deadline:
        # shrinking epsilon fraction as the search proceeds
        elapsed = time.time() - start_refine
        eps_frac = max(eps_end, eps_start * (1.0 - elapsed / total_time))

        # current maximal index
        k_max = int(np.argmax(conv))

        # ---- choose source (high contribution to the max bin) ----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]
        tot_contrib = contrib.sum()
        if tot_contrib > 0:
            prob_src = contrib / tot_contrib
            src = rng.choice(i_range, p=prob_src)
        else:
            prob_src = a / a.sum()
            src = rng.choice(n, p=prob_src)

        if a[src] < 1e-15:
            continue

        # ---- choose destination (low contribution to the max bin) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        dest_contrib = np.zeros(n)
        dest_contrib[valid] = a[k_max - idx[valid]]
        # Inverse weighting – smaller contribution → larger probability
        inv_weight = 1.0 / (dest_contrib + 1e-12)
        inv_weight[src] = 0.0                         # never move to itself
        prob_dst = inv_weight / inv_weight.sum()
        dst = rng.choice(n, p=prob_dst)

        # ---- amount to move -------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ---- propose new convolution (O(n) update) -------------------------
        conv_new = conv.copy()
        # slices where the source / destination appear in the sum indices
        slice_src = slice(src, src + n)
        slice_dst = slice(dst, dst + n)
        conv_new[slice_src] -= 2.0 * eps * a
        conv_new[slice_dst] += 2.0 * eps * a

        new_max = conv_new.max()
        if new_max < max_val - rel_improve:
            # accept move
            conv = conv_new
            a[src] -= eps
            a[dst] += eps
            max_val = new_max
            it += 1
            if new_max < best_max - rel_improve:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] inc‑refine improve: "
                      f"iter={it:,}, max={best_max:.8f}")

        # occasional early exit if we are out of time (handled by while condition)
    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise – use an existing global if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    final_n = 4800                       # can be changed if desired
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – warm‑up (exponential gradient)
    # ------------------------------------------------------------------
    stage1_time = min(12.0, 0.07 * (deadline - time.time()))
    stage1_deadline = time.time() + stage1_time

    a_warm, score_warm = _run_exponential_gradient(
        a0,
        final_n,
        stage1_deadline,
        start_time,
        alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        eta0=0.4,
        steps_per_alpha=6000,
    )
    if score_warm < best_score:
        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + max(30.0, 0.78 * remaining)

        a_sub, score_sub = _run_subgradient_descent(
            best_seq,
            final_n,
            stage2_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – massive incremental local refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_refined = _run_incremental_refine(
            best_seq,
            final_n,
            deadline,
            start_time,
            eps_start=0.04,
            eps_end=0.002,
        )
        conv_ref = _conv_self(a_refined)
        score_ref = 2.0 * final_n * conv_ref.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] final refine improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
