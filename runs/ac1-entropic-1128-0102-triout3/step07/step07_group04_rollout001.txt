def random_spectral_init(n, seed=None):
    # random phases for half-spectrum
    rng = np.random.default_rng(seed)
    # magnitude: set all to 1 (i.e., constant), DC set to 1 (normalized later)
    # create complex array of length n
    A = np.zeros(n, dtype=complex)
    # DC component random amplitude? We'll set magnitude 1, random phase 0 because it's real.
    A[0] = n  # choose amplitude to later satisfy sum=1 after iFFT
    # For 1..n/2-1 random phases
    half = n // 2
    phases = rng.random(half-1) * 2*np.pi
    mags = np.ones(half-1)  # constant magnitude
    A[1:half] = mags * np.exp(1j * phases)
    if n % 2 == 0:  # Nyquist freq has to be real
        A[half] = np.real(rng.normal())
    else:
        # for odd we have symmetric partner
        pass
    # Set symmetric conjugate for negative frequencies
    A[half+1:] = np.conj(A[1:half][::-1])
    a = np.fft.ifft(A).real
    # shift to nonnegative
    a -= a.min()  # make min zero
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full_like(a, 1.0 / n)
    return a

"""
Multistage optimizer for the “max self‑convolution” objective.

* Stage 0 – many short Adam runs on a modest size (n≈2048) using a smooth
  log‑sum‑exp surrogate of the max‑convolution.
* Stage 1 – up‑sample the current best vector to n≈8192 and continue Adam.
* Stage 2 – up‑sample to n≈16384 and run Adam again (more degrees of freedom).
* Stage 3 – projected sub‑gradient descent on the true max‑convolution.
* Stage 4 – a light simulated‑annealing mass‑transfer refinement.

All stages respect a global time budget (default 1000 s) and keep the
vector on the probability simplex (non‑negative, Σa=1).  The final
sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    fft_len = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in the polishing stage)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv) w.r.t. a (non‑negative entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam on the smooth log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      tau0: float,
                      tau_end: float = 1e-12,
                      tau_decay: float = 0.9995,
                      step_size: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        O_τ(a) = 2·n·τ·log∑exp(conv(a)/τ)
    using Adam with Euclidean projection onto the simplex.
    Returns the best vector seen (still on the simplex) and its *true*
    (unsmoothed) score 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1

        # exact (unsmoothed) score – for bookkeeping only
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}, τ={tau:.2e}")

        # ------- soft‑max weights -------
        z = conv / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)          # numerical stability
        sum_exp = np.sum(exp_z)
        p = exp_z / sum_exp                # shape (2·n‑1), sums to 1

        # ------- gradient of O_τ -------
        # grad_i = 4·n·∑_j a_j·p_{i+j}
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)      # length 3·n‑2
        inner = conv_rev[n - 1:n - 1 + n]    # exactly the required slice
        grad = 4.0 * n * inner

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step_size / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat
        a = _project_simplex(a)

        # cool τ
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             beta: float = 0.93,
                             step0: float = 0.15,
                             max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on
        F(a) = 2·n·max(conv(a)),  a ≥ 0, Σa = 1.
    Uses a simple momentum scheme.
    Returns the best vector seen and its true score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass transfers)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very light SA that transfers a random fraction of mass between two cells.
    Only runs while time permits.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective.  The routine follows the multistage
    strategy described in the module docstring and respects the global
    time budget.

    Returns:
        list[float]: a sequence (sum≈1) to be submitted to the evaluator.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter (previous best) – will be normalised automatically
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                tmp = np.maximum(0.0, tmp)
                s = tmp.sum()
                if s > 0:
                    starter = tmp / s
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    # allocate up to ~80 s or 15 % of the total budget (whichever is smaller)
    slice_time0 = min(80.0, 0.15 * (budget_s - (time.time() - start)))
    slice_deadline0 = time.time() + slice_time0

    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        conv0 = _conv_self(a0)
        tau0 = max(conv0.max() * 10.0, 1e-8)

        a_opt, _ = _run_adam_softmax(
            a0, n0, slice_deadline0, start,
            tau0=tau0,
            tau_end=1e-13,
            tau_decay=0.9995,
            step_size=0.25,
        )
        cur_score = 2.0 * n0 * _conv_self(a_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start:6.2f}s] stage 0 improve: restart={r}, "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to n≈8192 and continue Adam
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n1 = 8192
        if best_seq is None:
            best_seq = np.full(n0, 1.0 / n0)
        a1 = _upsample(best_seq, n1)

        conv1 = _conv_self(a1)
        tau0_1 = max(conv1.max() * 10.0, 1e-10)

        # allocate roughly half of the remaining time (capped at 200 s)
        remaining = deadline - time.time()
        adam_time1 = min(200.0, 0.5 * remaining)
        adam_deadline1 = time.time() + adam_time1

        a1_opt, _ = _run_adam_softmax(
            a1, n1, adam_deadline1, start,
            tau0=tau0_1,
            tau_end=1e-13,
            tau_decay=0.9994,
            step_size=0.15,
        )
        cur_score = 2.0 * n1 * _conv_self(a1_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a1_opt.copy()
            print(f"[{time.time() - start:6.2f}s] stage 1 improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to n≈16384 and run Adam again
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = 16384
        a2 = _upsample(best_seq, n2)

        conv2 = _conv_self(a2)
        tau0_2 = max(conv2.max() * 10.0, 1e-10)

        remaining = deadline - time.time()
        adam_time2 = min(250.0, 0.5 * remaining)
        adam_deadline2 = time.time() + adam_time2

        a2_opt, _ = _run_adam_softmax(
            a2, n2, adam_deadline2, start,
            tau0=tau0_2,
            tau_end=1e-14,
            tau_decay=0.9993,
            step_size=0.12,
        )
        cur_score = 2.0 * n2 * _conv_self(a2_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start:6.2f}s] stage 2 improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        subgrad_deadline = deadline - 1.0                # leave a second for finalisation
        a3_opt, _ = _run_subgradient_descent(
            best_seq, n, subgrad_deadline, start,
            beta=0.94,
            step0=0.15,
        )
        cur_score = 2.0 * n * _conv_self(a3_opt).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a3_opt.copy()
            print(f"[{time.time() - start:6.2f}s] sub‑grad final improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – very light simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        a_ref, _ = _run_annealing(
            best_seq,
            n,
            deadline,
            start,
            best_score,
            best_seq,
            start_temp=1e-2,
            end_temp=1e-8,
            steps=100_000,
        )
        cur_score = 2.0 * n * _conv_self(a_ref).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_ref.copy()
            print(f"[{time.time() - start:6.2f}s] anneal final improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
