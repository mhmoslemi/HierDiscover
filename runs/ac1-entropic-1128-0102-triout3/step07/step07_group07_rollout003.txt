"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

The algorithm is a three‑phase heuristic that stays well inside the
1000 s budget:

1. **Fast Adam optimisation on a soft‑max parametrisation** – a sequence
   is represented by logits `y` and the smooth objective
   `f(a)= (1/α)·log Σ exp(α·(a*a))` is minimised for a schedule of
   increasing smoothing parameters `α`.  Adam with momentum is used and
   the best iterate is kept.

2. **Projected sub‑gradient descent on the true max‑convolution**
   objective.  The sub‑gradient is formed from the (almost) active
   convolution entries and a Euclidean projection onto the probability
   simplex enforces the constraints.

3. **Long simulated‑annealing with a targeted‑move policy** – a very
   cheap incremental “mass‑transfer’’ move is repeatedly applied:
   * with probability 0.6 the move is random,
   * otherwise the source is taken from the indices that contribute the
     most to the current convolution peak and the destination is the
     currently smallest coefficient.
   The move is accepted either if it strictly reduces the score or
   with Metropolis probability `exp(-Δ/T)`.  The temperature decays
   exponentially from `0.2` to `1e‑6`.  The annealing runs for as many
   iterations as the remaining time permits (typically several hundred
   thousand steps).

All stages share a single time budget, and the best sequence seen
overall is returned (as a plain Python list).  The routine works for
any length `n` (default 4800) and makes no assumptions about the
external variable `height_sequence_1` – it is used when available as an
additional start point.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # the relevant part of the cross‑correlation
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserve sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact projected gradient descent on the true (max) objective
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with momentum on the objective
        f(a) = max_k (a*a)[k]      (the max of the self‑convolution).

    Returns the best vector seen (probability vector) and its score
    2·n·max_conv .
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient of the max (active set) --------------------
        rel_tol = 1e-12
        thresh = cur_max * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each ordered pair contributes a[j] to the gradient w.r.t a[i]
            g[i] += a[j]
            g[j] += a[i]          # diagonal counted twice – harmless

        # momentum (Nesterov‑style) and step‑size decay
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement with a targeted‑move policy
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   start_temp: float = 2e-1,
                   end_temp: float = 1e-6,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Long annealing loop.  Two kinds of moves are used:
    * random (source drawn w.r.t. a, destination uniform),
    * targeted: source comes from the indices that contribute most to the
      current convolution peak, destination is the smallest coefficient.
    The move is accepted if it improves the score or with Metropolis
    probability exp(-Δ/T).  The best sequence seen is returned.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (exponential cooling)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # decide move type
        if rng.random() < 0.6:          # random move
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
        else:                           # targeted move
            # locate current convolution peak
            k_max = conv.argmax()
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = k_max - i_idx
            contrib = a[i_idx] * a[j_idx]
            src = i_idx[contrib.argmax()]    # index that contributes most
            # pick destination = currently smallest coefficient (different from src)
            dst_candidates = np.delete(np.arange(n), src)
            dst = dst_candidates[a[dst_candidates].argmin()]

        # propose a mass transfer
        eps = a[src] * rng.random() * 0.30   # at most 30 % of src mass
        if eps <= 1e-14:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction (very cheap)
# ----------------------------------------------------------------------


def _run_peak_reduction(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_no_improve: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy loop that moves a fraction of the coefficient that contributes
    most to the current maximum convolution peak towards the smallest
    coefficient.  The move is kept only if it strictly lowers the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = a[src] * 0.25
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            no_improve = 0
            print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: "
                  f"max={new_max:.6f}")
        else:
            no_improve += 1

    score = 2.0 * n * _conv_self(a).max()
    return a, score

    np.random.seed(seed)
    start_time = time.time()
    # tiny margin to avoid cutting the last print statement
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4800 works well; we keep it but allow a modest up‑scale
    # if time permits (not required for the current budget).
    # ------------------------------------------------------------------
    n = 4800

    # ------------------------------------------------------------------
    # Build a small pool of diverse starting points.
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best construction, when available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            if starter.shape[0] != n:
                starter = _upsample(starter, n)
            starts.append(starter)
    except Exception:
        pass

    # 2) uniform distribution
    starts.append(np.full(n, 1.0 / n))

    # 3) exponential decay (geometric)
    r = 0.985
    exp_vec = np.array([r ** i for i in range(n)], dtype=float)
    starts.append(exp_vec / exp_vec.sum())

    # 4) symmetric V‑shape (linear increase towards the ends)
    lin = np.linspace(0.0, 1.0, n // 2 + 1)
    vshape = np.concatenate([lin, lin[-2::-1]])[:n]
    starts.append(vshape / vshape.sum())

    # 5) random smooth vector (Gaussian‑filtered noise)
    rng = np.random.default_rng()
    noise = rng.normal(size=n)
    window = 15
    smooth = np.convolve(noise, np.ones(window) / window, mode='same')
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() == 0:
        smooth = np.full_like(smooth, 1.0 / n)
    else:
        smooth /= smooth.sum()
    starts.append(smooth)

    # ------------------------------------------------------------------
    # Keep track of the overall best candidate.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Process each start independently, sharing the global time budget.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] start #{idx + 1}")

        # --------------------------------------------------------------
        # 1. Smooth Adam optimisation (soft‑max parametrisation)
        # --------------------------------------------------------------
        eps = 1e-16
        y = np.log(np.maximum(a0, eps))

        fft_len = _next_pow2(3 * n - 2)

        # a richer schedule of smoothing parameters
        alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
        iters_per_alpha = 12000      # a bit more than before

        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr0 = 0.12
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0

        local_best_seq = a0.copy()
        local_best_score = float(evaluate_sequence(local_best_seq.tolist()))

        for alpha in alpha_schedule:
            for i in range(iters_per_alpha):
                if time.time() >= deadline:
                    break

                # soft‑max to obtain a valid probability vector
                maxy = np.max(y)
                exp_y = np.exp(y - maxy)
                a = exp_y / exp_y.sum()

                # smoothed objective (log‑sum‑exp)
                conv = _conv_self(a)
                scaled = alpha * conv
                max_scaled = scaled.max()
                lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
                f_val = lse / alpha
                obj = 2.0 * n * f_val

                # gradient w.r.t. a
                grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

                # chain rule to logits
                grad_y = a * (grad_a - np.dot(grad_a, a))

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_y
                v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
                y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

                # occasional exact evaluation
                if i % 300 == 0:
                    sc = evaluate_sequence(a.tolist())
                    if sc < local_best_score:
                        local_best_score = sc
                        local_best_seq = a.copy()
                        print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                              f"α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

            if time.time() >= deadline:
                break

        # --------------------------------------------------------------
        # 2. Projected gradient descent on the true max‑convolution objective
        # --------------------------------------------------------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            # allocate roughly half of what is left to polishing
            polish_deadline = time.time() + 0.5 * remaining
            a_polish, score_polish = _run_gradient_descent(
                local_best_seq,
                polish_deadline,
                start_time,
                beta=0.94,
                step0=0.25,
            )
            if score_polish < local_best_score:
                local_best_score = score_polish
                local_best_seq = a_polish.copy()
                print(f"[{time.time() - start_time:6.2f}s] polishing improve: "
                      f"score={score_polish:.6f}")

        # --------------------------------------------------------------
        # 3. Deterministic greedy peak‑reduction (cheap)
        # --------------------------------------------------------------
        if time.time() < deadline:
            peak_deadline = time.time() + 0.2 * (deadline - time.time())
            a_peak, score_peak = _run_peak_reduction(
                local_best_seq,
                peak_deadline,
                start_time,
                max_no_improve=1500,
            )
            if score_peak < local_best_score:
                local_best_score = score_peak
                local_best_seq = a_peak.copy()
                print(f"[{time.time() - start_time:6.2f}s] peak‑reduce improve: "
                      f"score={score_peak:.6f}")

        # --------------------------------------------------------------
        # 4. Long simulated‑annealing refinement (uses remaining time)
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_final, score_final = _run_annealing(
                local_best_seq,
                deadline,
                start_time,
                start_temp=2e-1,
                end_temp=1e-6,
                steps=800_000,          # as many steps as time allows
            )
            if score_final < local_best_score:
                local_best_score = score_final
                local_best_seq = a_final.copy()
                print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                      f"score={score_final:.6f}")

        # --------------------------------------------------------------
        # Update the global best if this start beat it.
        # --------------------------------------------------------------
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST *** "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (guard against drift) and final evaluation.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    # Return a plain Python list
    return [float(x) for x in best_seq.tolist()]
