def _run_adam_smooth(a0, n, deadline, start_time, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, T_start=5e-1, T_end=5e-4, max_iter=10**7):

   a = a0.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   best_a = a.copy()
   best_score = float('inf')
   t = 1

   while time.time() < deadline and t <= max_iter:
       conv = _conv_self(a)
       max_conv = conv.max()
       score = 2.0 * n * max_conv
       if score < best_score:
           best_score = score
           best_a = a.copy()
           print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={t}, score={score:.6f}")

       # temperature schedule: geometric decay with exponent
       T = T_start * (T_end/T_start) ** (t / max_iter)   # maybe slower

       # compute softmax weights:
       scaled = (conv - max_conv) / T
       # subtract max to avoid overflow
       max_scaled = scaled.max()
       w = np.exp(scaled - max_scaled)
       w_sum = w.sum()
       if w_sum == 0:
           w = np.full_like(w, 1.0 / w.shape[0])
       else:
           w /= w_sum

       # gradient
       rev_a = a[::-1]
       mlen = len(w)
       fft_len = 1 << ((n + mlen - 1).bit_length())
       A = np.fft.rfft(rev_a, n=fft_len)
       B = np.fft.rfft(w, n=fft_len)
       full = np.fft.irfft(A * B, n=fft_len)[: n + mlen - 1]
       grad = 2.0 * full[n - 1:n - 1 + n]   # shape n

       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad ** 2)
       m_hat = m / (1 - beta1 ** t)
       v_hat = v / (1 - beta2 ** t)

       step = step0 * m_hat / (np.sqrt(v_hat) + eps)

       a = _project_simplex(a - step)

       t += 1

   return best_a, best_score

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of max(conv)
# ----------------------------------------------------------------------
def _run_adam_softmax(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    T_start: float = 5e-2,
    T_end: float = 5e-5,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser that minimises a temperature‑smoothed
    max‑convolution.  The temperature is geometrically annealed from
    ``T_start`` to ``T_end``.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: "
                f"n={n}, iter={it}, score={cur_score:.6f}"
            )

        # geometric temperature decay
        T = T_start * (T_end / T_start) ** (it / max_iter)

        # soft‑max weights
        scaled = (conv - cur_max) / T
        # stabilise the exponentials
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient = 2 * (rev(a) * w)
        rev_a = a[::-1]
        mlen = w.shape[0]
        fft_len = 1 << ((n + mlen - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + mlen - 1]
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (active set)
# ----------------------------------------------------------------------
def _run_subgradient_gd(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.1,
    beta: float = 0.9,
    max_iter: int = 10 ** 7,
    rel_eps: float = 0.005,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv) using a tight active set."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                f"n={n}, iter={it}, score={cur_score:.6f}"
            )

        # active set: all entries within rel_eps of the current maximum
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        rev_a = a[::-1]
        mlen = w.shape[0]
        fft_len = 1 << ((n + mlen - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + mlen - 1]
        grad = 2.0 * full[n - 1 : n - 1 + n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 0.2,
    end_temp: float = 1e-7,
    steps: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """Metropolis walk with tiny mass‑transfer, swaps and occasional smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        move_type = np.random.rand()
        if move_type < 0.75:
            # tiny mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                eps = a[i] * np.random.rand() * 0.2  # up to 20 % of mass at i
                a_prop = a.copy()
                a_prop[i] -= eps
                a_prop[j] += eps
            else:
                continue
        elif move_type < 0.90:
            # swap move (preserves the sum exactly)
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                a_prop = a.copy()
                a_prop[i], a_prop[j] = a_prop[j], a_prop[i]
            else:
                continue
        else:
            # local smoothing move
            k = np.random.choice([3, 5, 7])
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode="same")
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the given time budget.
    The algorithm follows the four‑phase scheme described in the module
    docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found construction.
    # ------------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage schedule (lengths).  The final length is 9600 – the same as
    # the baseline – but the budget allocation is more aggressive.
    # ------------------------------------------------------------------
    stage_ns = [800, 2400, 4800, 9600]          # progressive resolution
    total_n = sum(stage_ns)

    total_time = deadline - start_time
    # 80 % of the budget for optimisation phases, 20 % for annealing
    opt_time_budget = total_time * 0.80
    anneal_time_budget = total_time - opt_time_budget

    # ------------------------------------------------------------------
    # Optimisation phases
    # ------------------------------------------------------------------
    for n in stage_ns:
        if time.time() >= deadline:
            break

        # allocate a proportional slice of the remaining optimisation budget
        remaining_opt = max(0.0, deadline - start_time - anneal_time_budget - (time.time() - start_time))
        stage_time = remaining_opt * (n / total_n)
        stage_deadline = time.time() + stage_time

        # initialise the vector at this resolution
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None and starter.size == n:
            a0 = starter.copy()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ----- Phase 1: Adam on soft‑max surrogate --------------------
        a_opt, score_opt = _run_adam_softmax(
            a0,
            n,
            stage_deadline,
            start_time,
            step0=0.2,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            T_start=5e-2,
            T_end=5e-5,
            max_iter=10 ** 7,
        )

        # ----- Phase 2: exact sub‑gradient descent -------------------
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _run_subgradient_gd(
                a_opt,
                n,
                stage_deadline,
                start_time,
                step0=0.1,
                beta=0.9,
                max_iter=10 ** 7,
                rel_eps=0.005,
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # keep the best across stages
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Final annealing on the highest‑resolution vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq,
            n_final,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=0.2,
            end_temp=1e-7,
            steps=10_000_000,
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list
    return [float(x) for x in best_seq]
