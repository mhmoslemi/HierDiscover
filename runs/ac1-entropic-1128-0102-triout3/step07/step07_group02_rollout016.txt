"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the previous best construction (``height_sequence_1``) if it
   exists, otherwise initialise a random Dirichlet vector.
2. Upsample the warm‑start to a moderately large size (≈2·10⁴ terms) and
   normalise it to the probability simplex.
3. Two‑phase descent on a smooth approximation of the max‑convolution:
   * Phase A – projected Adam on the soft‑max of the convolution (fast
     optimisation of the shape).
   * Phase B – projected sub‑gradient descent on the true max‑convolution
     (refinement of the active constraints).
4. Heavy‑weight stochastic local optimisation that mixes three move types:
   * “Peak‑shaving’’ (transfer a small amount of probability mass from
     indices that contribute most to the current maximal convolution entry).
   * Pairwise swaps of coefficients (explores the permutation space of a
     given multiset of values).
   * Random re‑shuffling of the whole vector (large‑scale jumps).
   The moves are accepted with a Metropolis rule under a slowly cooling
   temperature.
5. Keep the best candidate seen throughout the run, project it onto the
   simplex one last time and return it as a plain list.

The routine respects the 1000 s budget, printing progress messages so that
the harness can see the evolving upper bound.
"""

# ----------------------------------------------------------------------
# Imports

import time
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – centre of simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of reversed a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution (used in Phase A)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k softmax(conv_k / temperature) * conv_k.
    The factor 2 comes from the derivative of aᵀT_k a.
    """
    # stable soft‑max
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # extra stability
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)               # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]               # centre block of length n
    return grad

# Optimiser: Adam on the soft‑max (Phase A)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      lr: float = 0.6,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth max‑convolution.
    Returns the best vector seen and its (exact) score 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 5.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (slow exponential decay)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser: projected sub‑gradient descent on the exact max‑convolution (Phase B)
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.5,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑convolution entry."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        idx_max = int(np.argmax(conv))

        # exact sub‑gradient of the active convolution entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j_arr[mask]]

        # momentum & diminishing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stochastic local search – combines peak‑shaving, swaps and full reshuffles
# ----------------------------------------------------------------------
def _run_heavy_local_search(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Mixed local search:
       * move a small fraction of mass from high‑contribution indices to low‑contribution ones,
       * swap two coefficients,
       * occasionally replace the whole vector with a random shuffle.
    Uses a Metropolis acceptance rule with a cooling temperature.
    Returns the best sequence observed.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    start_T = 5e-2
    end_T = 1e-6

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # decide which move type to apply
        move_type = np.random.rand()
        if move_type < 0.5:
            # ---------- peak‑shaving ----------
            conv = _conv_self(a)
            idx_max = int(np.argmax(conv))

            i_arr = np.arange(n)
            j_arr = idx_max - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            contrib = np.zeros(n)
            contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

            if contrib.sum() > 0:
                donor = np.random.choice(n, p=contrib / contrib.sum())
            else:
                donor = np.random.randint(n)

            # pick a low‑contribution recipient
            low_thr = np.percentile(contrib, 20)
            low_idxs = np.where(contrib <= low_thr)[0]
            if low_idxs.size == 0:
                rec = np.random.randint(n)
            else:
                rec = np.random.choice(low_idxs)

            if donor == rec or a[donor] <= 1e-15:
                continue

            eps = a[donor] * np.random.rand() * 0.3
            a_prop = a.copy()
            a_prop[donor] -= eps
            a_prop[rec]   += eps

        elif move_type < 0.85:
            # ---------- swap two coefficients ----------
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        else:
            # ---------- full random reshuffle ----------
            a_prop = np.random.permutation(a)

        # evaluate proposed vector
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        # temperature based acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_T * (end_T / start_T) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={iteration}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a smooth Adam optimiser, an exact sub‑gradient descent
    and a heavy‑weight stochastic local optimiser.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Warm‑start (previous best construction, if available)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # Target length – a compromise between expressiveness and speed
    # --------------------------------------------------------------
    n_target = 19200                      # roughly the size used in the previous run

    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        # Dirichlet with modest concentration – yields many small entries
        a0 = np.random.gamma(0.5, 1.0, size=n_target)
        a0 /= a0.sum()

    best_a = a0.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase A – Adam on soft‑max (smooth max‑convolution)
    # --------------------------------------------------------------
    phase_a_deadline = min(deadline,
                           start_time + 0.2 * (deadline - start_time))
    a_opt, score_opt = _run_adam_softmax(best_a, n_target,
                                         phase_a_deadline, start_time)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] Phase‑A improve: "
              f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Phase B – exact sub‑gradient descent
    # --------------------------------------------------------------
    phase_b_deadline = min(deadline,
                           start_time + 0.45 * (deadline - start_time))
    a_opt, score_opt = _run_subgrad_gd(best_a, n_target,
                                       phase_b_deadline, start_time)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] Phase‑B improve: "
              f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Phase C – heavy stochastic local search (swap / peak‑shave / shuffle)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _run_heavy_local_search(best_a, n_target,
                                                   deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Final tidy‑up – ensure we are still on the simplex
    # --------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    # Return the plain Python list required by the harness
    return [float(x) for x in best_a]
```

This implementation keeps the spirit of the previous approach but adds:

* **Adam** optimisation for a smoother and potentially faster descent on the soft‑max surrogate.
* A **mixed stochastic local search** that not only shaves peaks but also swaps coefficients and occasionally reshuffles the whole vector – a cheap way of exploring the permutation space of a fixed multiset of values.
* A **careful time budget split** (≈20 % Adam, ≈25 % sub‑gradient, ≈55 % stochastic search) to make the most of the 1000 s limit.
* Detailed progress prints so you can see the bound improvement in real time.

The algorithm respects all constraints of the evaluation harness and returns a valid non‑negative step‑function ready for scoring.
