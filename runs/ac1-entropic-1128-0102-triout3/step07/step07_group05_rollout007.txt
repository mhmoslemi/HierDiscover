    if a_new_sum == 0:
        a_new = np.full_like(a, 1.0 / a.shape[0])

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

where conv(a,a) is the linear self‑convolution.
The optimizer consists of three stages, each respecting the overall
budget (≈1000 s):

1.  *Smoothed exponential‑gradient* – minimise a log‑sum‑exp
    approximation of the maximal convolution.  The gradient of the
    smooth objective is obtained analytically via FFT and the update
    uses exponentiated‑gradient (mirror descent on the simplex).

2.  *Exact max‑sub‑gradient* – after the smoothing phase we switch to
    the true max‑convolution.  The sub‑gradient is built from all
    “almost‑tight’’ constraints (indices where the convolution is equal
    to its maximum up to a tiny tolerance) and an exponentiated‑gradient
    step is applied.

3.  *Fast local annealing* – a short Metropolis‑type annealing loop
    performs tiny mass transfers between two indices, recomputing the
    convolution with FFT only when a move is accepted.

The best sequence encountered throughout the run is returned (normalised
to sum = 1).  All numbers are non‑negative, satisfying the checks in
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of f(a)= (1/α)·log Σ_k exp(α·conv_k)  (smooth max).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    L = _next_pow2(n + len(w_rev) - 1)
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _eg_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) update on the simplex."""
    # work in log‑space for numerical stability
    loga = np.log(a + 1e-300) - eta * grad
    loga -= np.max(loga)                     # shift to avoid overflow
    a_new = np.exp(loga)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method mixes a smooth exponential‑gradient phase, an exact
    sub‑gradient phase and a short annealing refinement, all respecting the
    supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid overruns
    deadline = start_time + budget_s - 0.8

    # --------------------------------------------------------------
    # Problem size and initialisation
    # --------------------------------------------------------------
    n = 4800

        if starter.ndim == 1 and starter.size > 0:

    if s == 0.0:

    total = best_seq.sum()
    if total > 0.0:

    else:

    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")
