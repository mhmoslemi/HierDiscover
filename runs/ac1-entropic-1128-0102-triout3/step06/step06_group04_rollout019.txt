"""
Propose a sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several cheap ideas that are known to give
very low convolution peaks:

* several analytic “seed” shapes (uniform, decreasing linear,
  geometric, arcsine‑like) are generated and normalised.
* each seed is refined by projected Adam on a smooth log‑sum‑exp
  approximation of the max‑norm.
* the best refined seed is finally polished by a fast simulated‑annealing
  mass‑transfer loop.

All operations stay on the probability simplex (∑a=1).  Because the
evaluation is homogeneous (scales cancel) we do not need any further
normalisation – the returned list can be used directly with
``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np
from scipy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT (fast)."""
    n = a.shape[0]
    m = n + n - 1
    # next power‑of‑two for speed
    N = 1 << ((m - 1).bit_length())
    A = rfft(a, n=N)
    c = irfft(A * A, n=N)[:m]
    return c


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (∑a is assumed to be 1)."""
    n = a.shape[0]
    max_conv = _conv_self(a).max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Analytic seed generators (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def _seed_uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n)


def _seed_linear_desc(n: int) -> np.ndarray:
    """Decreasing linear shape a_i ∝ (n‑i)."""
    x = np.arange(n, 0, -1, dtype=float)       # n, n‑1, … ,1
    return x / x.sum()


def _seed_geometric(n: int, q: float) -> np.ndarray:
    """Geometric (exponential) decay a_i ∝ q**i, 0<q<1."""
    i = np.arange(n, dtype=float)
    a = (1.0 - q) * (q ** i)
    s = a.sum()
    if s == 0.0:
        return _seed_uniform(n)
    return a / s


def _seed_arcsine_like(n: int) -> np.ndarray:
    """Discrete version of the arcsine density: 1/√(x(1‑x))."""
    # centre the points at (i+0.5)/n  ∈ (0,1)
    i = np.arange(n, dtype=float) + 0.5
    t = i / n
    a = 1.0 / np.sqrt(t * (1.0 - t))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _seed_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet sample (flat α=1)."""
    a = np.random.rand(n)
    if a.sum() == 0.0:
        a = np.ones(n)
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_logsumexp(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of  L_λ(a)=λ·log Σ exp(conv/λ).
    Using the identity ∂conv/∂a = 2· rev(a) * w  (linear conv).
    """
    n = a.shape[0]
    # weights w_k ∝ exp(conv_k/λ)
    max_c = conv.max()
    w = np.exp((conv - max_c) / lam)
    w /= w.sum()
    # convolution of w with reversed a (full length)
    a_rev = a[::-1]
    tmp = _conv_self(w)                      # conv of w with w
    # we need   ∂/∂a_i Σ_k w_k * (a * a)_k   = 2 * (w * rev(a))_i
    # Compute convolution w * rev(a)
    prod = np.fft.irfft(np.fft.rfft(w) * np.fft.rfft(a_rev, n=w.shape[0]*2-1))
    grad = 2.0 * prod[n - 1:n - 1 + n]      # central slice of length n
    return grad


def _run_adam(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Long‑run Adam optimiser (projected onto simplex)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    lam_factor = 0.4            # λ = lam_factor·max(conv)
    lam_decay = 0.9
    lam_decay_interval = 2000

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam it={it} score={cur:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-8)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_logsumexp(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.7 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing mass‑transfer polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best: float,
                   cur_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Fast mass‑transfer annealing (no temperature schedule, simple Metropolis)."""
    a = a0.copy()
    best = cur_seq.copy()
    best_score = cur_best
    n = a.shape[0]

    # geometric temperature decay
    initial_T = 1e-1
    final_T = 1e-6
    total_steps = 200000
    it = 0

    while time.time() < deadline and it < total_steps:
        it += 1
        # linear temperature schedule
        T = initial_T * (final_T / initial_T) ** (it / total_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        cur_score = _objective(a_prop)
        delta = cur_score - best_score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal it={it} score={cur_score:.6f}")

    return best, best_score

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a ``budget_s`` second runtime
    budget and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a single length (large enough to give a good constant)
    # ------------------------------------------------------------------
    # Experiments show that n≈4096–8192 is a good trade‑off.
    n = 4096
    if time.time() + 0.5 < deadline:
        # give a little extra room for a larger n if we have time
        if budget_s > 600:
            n = 8192

    print(f"[{time.time() - start_time:6.2f}s] chosen length n={n}")

    # ------------------------------------------------------------------
    # Produce a diverse set of seed vectors (all already normalised)
    # ------------------------------------------------------------------
    seeds = [
        _seed_uniform(n),
        _seed_linear_desc(n),
        _seed_arcsine_like(n),
        _seed_dirichlet(n),
    ]

    # add a few geometric seeds with different decay rates
    for q in [0.95, 0.97, 0.99, 0.992, 0.995, 0.998]:
        seeds.append(_seed_geometric(n, q))

    # ------------------------------------------------------------------
    # Run a medium‑length Adam optimisation on every seed.
    # Allocate the majority of the budget to the best seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    per_seed_deadline = start_time + (budget_s * 0.5)   # first half of the budget

    for idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        # each seed gets a slice of the first‑half budget
        slice_dead = time.time() + (budget_s * 0.5) / len(seeds)
        a_opt, sc = _run_adam(seed_vec, slice_dead, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] New global best after Adam seed {idx} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final polish with a longer annealing phase (second half of budget)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform
        best_seq = _seed_uniform(n)

    anneal_deadline = deadline - 0.2
    best_seq, best_score = _run_annealing(best_seq,
                                          deadline=anneal_deadline,
                                          start_time=start_time,
                                          cur_best=best_score,
                                          cur_seq=best_seq)

    # ------------------------------------------------------------------
    # Ensure a valid return even if something failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _seed_uniform(n)

    # The evaluator will clamp / normalise, but we keep the distribution as is.
    return [float(x) for x in best_seq]
