"""
Propose a high‑quality step‑function sequence.

The algorithm consists of three stages:

1. **Exploratory random search (low dimension)**  
   – Sample many analytic families (product‑power, Gaussian, Beta,
     arcsine, power‑law, two‑end, Dirichlet) for n=256.  
   – Keep the best shape.

2. **Progressive up‑sampling + entropic mirror descent**  
   – Increase the length (256 → 512 → … → 32768).  
   – Upsample the current best vector, enrich it with a short random‑shape
     sweep, then run a KL‑projected (exponentiated‑gradient) descent on
     the smooth soft‑max surrogate of the convolution ∞‑norm.  
   – The temperature τ is annealed from ~5/n down to ~5·10⁻⁶·1/n,
     which effectively reaches the true max‑norm.

3. **Fine‑tuning**  
   – A short hard‑max sub‑gradient descent (exact max instead of soft‑max).  
   – A cheap mass‑transfer simulated‑annealing polish.

All updates preserve non‑negativity and the ℓ₁‑normalisation, so the
objective `2 n·max(conv)/sum²` is invariant under scaling.  The routine
stops automatically after the allotted budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution of u and v via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring (exact – scale invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalized vector (∑a = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation, then renormalise)
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Analytic parametric families
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta density on [0,1] discretised to n points."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape: a_i ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_powerlaw(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·(n-i)^{-q} (both ends heavy when p,q>0)."""
    i = np.arange(1, n + 1, dtype=float)
    left = i ** (-p)
    right = (n - i + 1) ** (-q)
    a = left * right
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_two_end(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p} – two symmetric heavy ends."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector and normalise."""
    draws = np.random.gamma(alpha, 1.0, size=n)
    s = draws.sum()
    if s > 0.0:
        draws /= s
    else:
        draws = np.full(n, 1.0 / n)
    return draws


# ----------------------------------------------------------------------
# Entropic (soft‑max) mirror descent on the simplex
# ----------------------------------------------------------------------


def entropic_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Minimise max(a*a) via a smoothed soft‑max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule: start ≈5/n, end ≈5e‑6/n
    tau = 5.0 / n
    tau_end = tau * 1e-3
    decay = 0.9995

    eta0 = 0.7
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # Stable soft‑max weights
        shift = max_val
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of smooth surrogate: 2 * (w * a_rev) convolution
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)           # length = 3n‑2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Exponentiated‑gradient step (KL‑projection)
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)  # avoid overflow
        a = a * np.exp(expo)

        # Project back onto simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # Anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Project‑gradient steps using the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    eta0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        mask = conv == max_val
        cnt = mask.sum()
        if cnt > 0:
            w = mask.astype(float) / cnt
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)

        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 80000,
) -> tuple[np.ndarray, float]:
    """Cheap move‑based annealing: transfer a small ε from i→j."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Use a previously found construction if available
    # ------------------------------------------------------------------
    global_a = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            global_a = tmp / s
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – intensive random search at low dimension (n=256)
    # ------------------------------------------------------------------
    n_low = 256
    low_deadline = min(deadline, start_time + max(30.0, (deadline - start_time) * 0.15))
    a_low_best = np.random.rand(n_low)
    a_low_best /= a_low_best.sum()
    best_low_score = score_of(a_low_best)

    low_families = [
        "product_power",
        "gaussian",
        "beta",
        "arcsine",
        "powerlaw",
        "two_end",
        "dirichlet",
    ]

    while time.time() < low_deadline:
        kind = random.choice(low_families)
        if kind == "product_power":
            p = np.random.uniform(-6.0, -0.05)
            cand = shape_product_power(n_low, p)
        elif kind == "gaussian":
            sigma = np.random.uniform(n_low / 30.0, n_low / 2.0)
            cand = shape_gaussian(n_low, sigma)
        elif kind == "beta":
            alpha = np.random.uniform(0.05, 0.9)
            beta_par = np.random.uniform(0.05, 0.9)
            cand = shape_beta(n_low, alpha, beta_par)
        elif kind == "arcsine":
            cand = shape_arcsine(n_low)
        elif kind == "powerlaw":
            p = np.random.uniform(0.5, 3.0)
            q = np.random.uniform(0.5, 3.0)
            cand = shape_powerlaw(n_low, p, q)
        elif kind == "two_end":
            p = np.random.uniform(0.5, 3.0)
            cand = shape_two_end(n_low, p)
        elif kind == "dirichlet":
            alpha = np.random.uniform(0.05, 2.0)
            cand = shape_dirichlet(n_low, alpha)
        else:
            cand = np.random.rand(n_low)
            cand /= cand.sum()

        sc = score_of(cand)
        if sc < best_low_score:
            best_low_score = sc
            a_low_best = cand
            print(
                f"[{time.time() - start_time:6.2f}s] low‑dim shape {kind} score={sc:.6f}"
            )

    if global_a is None:
        global_a = a_low_best.copy()
        best_score = best_low_score
        best_seq = global_a.copy()

    # ------------------------------------------------------------------
    # Stage 2 – progressive length increase with entropic descent
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # Upsample the current best vector
        if global_a.shape[0] != n:
            a0 = upsample(global_a, n)
        else:
            a0 = global_a.copy()

        # Allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_time = max(5.0, remaining * 0.12)  # at least a few seconds
        slice_deadline = time.time() + slice_time

        # --------------------------------------------------------------
        # Random parametric warm‑starts for this length
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(4.0, slice_time * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = random.choice(low_families)
            if kind == "product_power":
                p = np.random.uniform(-8.0, -0.02)
                cand = shape_product_power(n, p)
            elif kind == "gaussian":
                sigma = np.random.uniform(n / 40.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == "beta":
                alpha = np.random.uniform(0.05, 0.9)
                beta_par = np.random.uniform(0.05, 0.9)
                cand = shape_beta(n, alpha, beta_par)
            elif kind == "arcsine":
                cand = shape_arcsine(n)
            elif kind == "powerlaw":
                p = np.random.uniform(0.5, 3.5)
                q = np.random.uniform(0.5, 3.5)
                cand = shape_powerlaw(n, p, q)
            elif kind == "two_end":
                p = np.random.uniform(0.5, 3.5)
                cand = shape_two_end(n, p)
            elif kind == "dirichlet":
                alpha = np.random.uniform(0.05, 1.5)
                cand = shape_dirichlet(n, alpha)
            else:
                cand = np.random.rand(n)
                cand /= cand.sum()

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Entropic mirror‑descent optimisation
        # --------------------------------------------------------------
        grad_deadline = time.time() + max(2.0, slice_deadline - time.time() - 6.0)
        a_opt, sc_opt = entropic_gradient_descent(
            local_best_a, n, grad_deadline, start_time
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] new global after entropic n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Hard‑max sub‑gradient fine‑tuning (quick)
        # --------------------------------------------------------------
        subgrad_deadline = time.time() + 2.0
        if time.time() < subgrad_deadline:
            a_sub, sc_sub = subgradient_descent(
                a_opt, n, subgrad_deadline, start_time
            )
            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] new global after subgrad n={n} score={sc_sub:.6f}"
                )

        # --------------------------------------------------------------
        # Cheap annealing polish
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + min(6.0, deadline - time.time())
            a_ref, sc_ref = annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] new global after anneal n={n} score={sc_ref:.6f}"
                )

        # Propagate the current best to the next length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 1.0 / 256)

    # Convert to plain Python list for the external evaluator
    return [float(x) for x in best_seq]
