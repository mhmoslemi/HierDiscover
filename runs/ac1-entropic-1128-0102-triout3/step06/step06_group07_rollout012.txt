def _smooth_gradient(a, conv, alpha):
    max_val = conv.max()
    # compute shifted conv for stability
    shift = conv - max_val
    # compute exp(alpha * shift) where shift <=0.
    # may underflow; but that's okay.
    exp_alpha = np.exp(alpha * shift)
    w = exp_alpha / exp_alpha.sum()
    # compute cross-corr via convolution w_rev with a.
    w_rev = w[::-1]
    # convolution length = len(a) + len(w_rev) - 1 = n + (2n-1) - 1 = 3n-2
    fft_len = 1 << ((len(a) + len(w_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    cross = np.fft.irfft(A * W, n=fft_len)
    offset = len(w_rev) - 1  # which equals len(conv)-1
    grad = 2 * cross[offset:offset + len(a)]
    return grad

"""
A multi‑stage optimiser for the “minimise max(self‑convolution)” problem.

Stage 0 – many short Adam‑optimisation runs on a modest size (n≈1024)
          with random restarts and a simplex projection.

Stage 1 – up‑sample the best vector to a medium size (≈4096) and run
          a longer Adam optimisation.

Stage 2 – up‑sample again to a large size (≈12288) and continue the
          Adam run.

Stage 3 – light simulated‑annealing (mass‑transfer moves) for fine‑tuning.

Stage 4 – deterministic local refinement:  repeatedly pick the shift
          where the convolution is maximal and transfer a small amount
          of mass away from the most offending pair.

All stages respect the global time budget (default 1000 s).  The final
vector is projected onto the probability simplex (non‑negative, sum = 1)
– the scale does not affect `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} (O(n) algorithm)."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct diagonal handling.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimisation (projected onto the simplex) using the sub‑gradient
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.12,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for  F(a)=2·n·max(conv(a))  with a ≥ 0, Σ a = 1.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_max(a, conv)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected update
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Light annealing that moves a fraction of mass between two indices."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic local refinement – attack the current maximal shift
# ----------------------------------------------------------------------
def _run_local_refine(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray,
                      max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Repeatedly pick the shift where the convolution attains its maximum,
    take the pair of indices that contributes most to this value and move
    a small fraction of its mass to a random location.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        s = int(conv.argmax())
        max_val = conv[s]

        # indices participating in this shift
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        if i_start > i_end:
            continue

        idx = np.arange(i_start, i_end + 1)
        jdx = s - idx
        products = a[idx] * a[jdx]

        # the most offending index (pick the larger side of the pair)
        offending = idx[np.argmax(products)]

        # propose moving a fraction away from the offending index
        eps = a[offending] * 0.20
        if eps < 1e-14:
            continue
        dst = rng.integers(n)
        if dst == offending:
            continue

        a_prop = a.copy()
        a_prop[offending] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] local refine improve: "
                  f"n={n}, iter={it:,}, score={new_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
