```
import scipy.sparse as sp

def build_constraint_matrix(a, active_indices):
    n = len(a)
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_indices):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # columns j=lo..hi
        cols_range = np.arange(lo, hi+1, dtype=int)
        rows.extend([idx]*(hi-lo+1))
        cols.extend(cols_range.tolist())
        data.extend(a[lo:hi+1].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_indices), n))
    return A
```

def _run_incremental_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq, total_steps=500_000):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    it = 0
    start_temp = 5e-2
    end_temp = 1e-7

    while time.time() < deadline and it < total_steps:
        it += 1
        T = start_temp * (end_temp/start_temp)**(it/total_steps)

        # choose src weighted by a
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue
        if a[src] < 1e-12:
            continue

        # propose epsilon
        eps = a[src] * rng.random() * 0.30  # up to 30% of source mass

        # apply move to conv (using old a)
        # move updates: conv[src:src+n] -= 2*eps*a ; conv[dst:dst+n] += 2*eps*a
        conv[src:src+n] -= 2.0 * eps * a
        conv[dst:dst+n] += 2.0 * eps * a
        # delta*delta terms
        conv[2*src] += eps * eps
        conv[2*dst] += eps * eps
        conv[src+dst] -= 2.0 * eps * eps

        # apply move to a
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # decide accept
        if delta < 0 or math.exp(-delta / T) > rng.random():
            # accept
            score = new_score
        else:
            # reject: revert changes
            # revert a
            a[src] += eps
            a[dst] -= eps
            # revert conv (inverse operation using reverted a = original a)
            conv[src:src+n] += 2.0 * eps * a
            conv[dst:dst+n] -= 2.0 * eps * a
            conv[2*src] -= eps * eps
            conv[2*dst] -= eps * eps
            conv[src+dst] += 2.0 * eps * eps
            # score unchanged

        # update best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

"""
Hybrid optimiser for the step‑function attack.

The algorithm consists of three stages, each respecting the overall time budget
(≈1000 s).  The same sequence length as the reference (n = 4800) is used – this
offers a good trade‑off between solution quality and speed.

1. **Adam on a smoothed max‑convolution** – the objective
   f(a)=2·n·max(a*a) is approximated by a log‑sum‑exp of the convolution.
   The parameters are kept on the simplex via a soft‑max (logits y).  The
   smoothing parameter α is increased gradually, yielding a more accurate
   gradient as optimisation proceeds.

2. **Exact projected‑gradient polishing** – after the Adam phase the true
   non‑smoothed maximal convolution is minimised by sub‑gradient descent on
   the probability simplex (Nesterov momentum, diminishing steps).

3. **Fast incremental simulated‑annealing** – a specialised annealing loop
   transfers a small ε‑fraction of mass from a source index to a destination
   index.  The convolution is updated *incrementally* in O(n) using slice
   operations, so many millions of moves can be performed within the remaining
   time.  A geometric temperature schedule accepts uphill moves with a
   probability exp(−Δ / T).

The best sequence encountered during the whole run is returned (normalised to
sum = 1).  All numbers are non‑negative, satisfying the security checks in
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, deadline: float,
                start_time: float, best_score: float,
                best_seq: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Runs Adam on the smoothed log‑sum‑exp objective.
    Returns (a, best_score, best_seq).
    """
    a = a0.copy()
    y = np.log(np.maximum(a, 1e-16))

    fft_len = _next_pow2(3 * n - 2)

    # schedule: higher α gives a tighter approximation of max
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 8000

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution / smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val  # not used directly, only for diagnostics

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient wrt logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # final a from logits
    maxy = np.max(y)
    exp_y = np.exp(y - maxy)
    a = exp_y / exp_y.sum()
    return a, best_score, best_seq


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """
    Simple projected gradient descent with momentum on the exact max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _run_incremental_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               total_steps: int = 1_000_000):
    """
    Mass‑transfer annealing where the convolution is updated incrementally.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    start_temp = 5e-2
    end_temp = 1e-7

    it = 0
    while time.time() < deadline and it < total_steps:
        it += 1
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)

        # pick a source proportional to its mass (helps focus on heavy positions)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30

        # ----- apply move (using the *old* a for the convolution update) -----
        # linear‑time slice updates
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        # update the probability vector
        a[src] -= eps
        a[dst] += eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            score = new_score
        else:
            # ----- reject: revert the move (note a now holds the *old* values) -----
            a[src] += eps
            a[dst] -= eps
            conv[src:src + n] += 2.0 * eps * a
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # score unchanged

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    """

    print(f"[{time.time() - start_time:6.2f}s] final reported score: {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
