            best_a = a.copy()

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The search proceeds in three phases:

1. **Progressive up‑sampling + gradient refinement** – start from a short
   vector (800 entries) and repeatedly double its length up to 9600.
   For each length a smooth soft‑max gradient descent is performed,
   followed by exact sub‑gradient descent, both projected onto the
   probability simplex.

2. **Long incremental simulated‑annealing** – the best vector found is
   refined by a Metropolis walk that moves a small fraction of the mass
   between two positions.  The convolution is updated incrementally
   (O(n) per move) and the temperature follows a slow geometric schedule.
   Periodically the full convolution is recomputed to avoid drift.

3. **Final polish** – a short sub‑gradient descent run cleans up the
   tail‑end of the annealed solution.

All phases respect the given time budget (default 1000 s).  If a previous
construction ``height_sequence_1`` exists it is used as a warm start;
otherwise a few random restarts are performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`
    and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) gradient descent on the soft‑max objective."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.1,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()  # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental simulated‑annealing (fast O(n) updates).
# ----------------------------------------------------------------------
def _run_incremental_anneal(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           cur_best_score: float,
                           cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions.
    The convolution is updated incrementally (O(n) per move)."""
    a = a0.copy()
    conv = _conv_self(a)               # full convolution, length 2*n-1
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    steps = 0
    max_move_frac = 0.20                # move up to 20 % of the source mass
    recompute_interval = max(2000, n)   # full recomputation to keep numerical stability

    start_temp = 0.2
    end_temp = 1e-7
    # a generous horizon for the temperature schedule
    horizon_steps = 5_000_000

    while time.time() < deadline:
        steps += 1

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (steps / horizon_steps)

        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * max_move_frac
        if eps == 0.0:
            continue

        # keep a copy of the current a to compute the delta efficiently
        a_old = a.copy()

        # ----- incrementally update the convolution -----
        # conv[i:i+n] -= 2·eps·a_old
        conv[i:i + n] -= 2.0 * eps * a_old
        # conv[j:j+n] += 2·eps·a_old
        conv[j:j + n] += 2.0 * eps * a_old
        # quadratic eps² terms
        conv[i + j] -= 2.0 * eps * eps
        conv[2 * i] += eps * eps
        conv[2 * j] += eps * eps

        # ----- apply the move to a -----
        a[i] = a_old[i] - eps
        a[j] = a_old[j] + eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # ----- Metropolis acceptance -----
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            cur_score = new_score
        else:
            # reject – roll back both a and conv
            a[i] = a_old[i]
            a[j] = a_old[j]
            conv[i:i + n] += 2.0 * eps * a_old
            conv[j:j + n] -= 2.0 * eps * a_old
            conv[i + j] += 2.0 * eps * eps
            conv[2 * i] -= eps * eps
            conv[2 * j] -= eps * eps

        # ----- record best -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, step={steps}, score={cur_score:.6f}")

        # ----- occasional full recomputation for safety -----
        if steps % recompute_interval == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] recompute improve: "
                      f"n={n}, step={steps}, score={cur_score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    try:

        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final incremental annealing phase
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_incremental_anneal(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Very short polishing sub‑gradient descent (just in case)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_polish, score_polish = _run_subgradient_gd(
            best_seq, best_seq.shape[0],
            deadline, start_time,
            step0=0.05, beta=0.9, max_iter=200_000
        )
        if score_polish < best_score:
            best_seq = a_polish.copy()
            best_score = score_polish
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
