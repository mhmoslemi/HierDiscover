    # a length n, w length 2n-1
    a_rev = a[::-1]
    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())

"""
propose_candidate – Multi‑phase optimizer for the self‑convolution max problem.

* Phase 0 – start from the previously discovered sequence (or a random one)
  and up‑sample to a relatively large size (default 16384).

* Phase 1 – “soft‑max” gradient descent with Adam on the simplex (log‑its).
  The objective  τ·log ∑exp(conv/τ) smoothly approximates the max.
  τ is tied to the current maximum value so the weighting adapts automatically.
  This phase exploits a smooth gradient and converges quickly.

* Phase 2 – classic projected sub‑gradient descent on the true max‑objective.
  Nesterov‑style momentum with a √t‑step schedule further lowers the bound.

* Phase 3 – light simulated‑annealing (mass‑transfer moves) for final polishing.

All phases respect the global time budget (default 1000 s) and the final
vector is returned as a plain Python list of non‑negative floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (used in Phase 2)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used in Phase 2)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Weighted gradient for the smooth (soft‑max) objective
#    G(a) = τ·log Σ exp(conv/τ)
# ----------------------------------------------------------------------
def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k·conv_k where w are non‑negative weights (∑w=1).
    The formula is 2·(a_rev * w)[i+n-1] for i=0…n‑1.
    """
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())           # length for linear conv a_rev*w
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    full = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]
    # slice the central n entries
    grad = full[n - 1: n - 1 + n]
    return 2.0 * grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective (Phase 1)
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its true (max‑conv) score.
    """
    # initialise logits (log‑softmax parameters)
    u = np.log(a0 + 1e-14)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                               # base learning rate

    best_a = a0.copy()
    best_score = 2.0 * n * _conv_self(a0).max()

    it = 0
    while time.time() < deadline:
        it += 1

        # probability vector from logits
        a = np.exp(u - np.max(u))
        a /= a.sum()

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam smooth iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature proportional to current maximum (adaptive smoothing)
        tau = max(max_val * 0.15, 1e-7)

        # soft‑max weights (numerically stable)
        exp_vals = np.exp((conv - max_val) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a
        grad_a = _gradient_weighted(a, w)

        # chain rule to logits (softmax Jacobian)
        grad_u = a * (grad_a - np.dot(a, grad_a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        u -= step * m_hat

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Phase 2, same as the reference code)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.92,
                          step0: float = 0.24,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise F(a)=2·n·max(conv(a)) with Nesterov momentum."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (Phase 3)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine follows a multi‑phase scheme (smooth Adam → sub‑gradient →
    annealing) while respecting the supplied time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # Initialise from the previous construction if available
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Target size – large enough to approach the continuous limit,
    # but still cheap for many FFTs.
    # ------------------------------------------------------------------
    n_target = 16384

    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate (≈70 % of the budget)
    # ------------------------------------------------------------------
    total_budget = deadline - start_time
    phase1_deadline = start_time + 0.70 * total_budget
    a_phase1, best_score = _run_adam_smooth(a0, n_target,
                                            phase1_deadline, start_time)

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (≈25 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = start_time + 0.95 * total_budget   # keep ~5 % for annealing
        a_phase2, score2 = _run_gradient_descent(
            a_phase1,
            n_target,
            phase2_deadline,
            start_time,
            beta=0.94,
            step0=0.22,
        )
        if score2 < best_score:
            best_score = score2
            a_phase1 = a_phase2.copy()

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            a_phase1,
            n_target,
            deadline,
            start_time,
            best_score,
            a_phase1,
            start_temp=2e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            a_phase1 = a_final.copy()

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    a_phase1 = np.asarray(a_phase1, dtype=float)
    total = a_phase1.sum()
    if total > 0:
        a_phase1 /= total
    else:
        a_phase1 = np.full_like(a_phase1, 1.0 / a_phase1.size)

    print(f"[{time.time() - start_time:6.2f}s] final score={best_score:.6f}")
    return [float(x) for x in a_phase1]
