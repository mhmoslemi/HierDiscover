def _adam_optimize(a0, n, deadline, start_time):
    a = a0
    y = np.log(np.maximum(a, 1e-16))
    # compute FFT length
    fft_len = _next_pow2(3*n - 2)
    # schedule
    alphas = [0.5, 2., 5., 10., 20., 40., 80., 160.]
    iters_per_alpha = 10000 # will adjust based on time
    # ADAM parameters:
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8; lr0 = 0.12
    m = np.zeros_like(y); v = np.zeros_like(y); t=0
    best_seq = a.copy()
    best_score = float('inf')
    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq, best_score
            # compute a
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()
            # compute conv and smooth value
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val
            # gradient wrt a:
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)
            # gradient wrt y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t))

            best_val = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] polishing improve: iter={it}, score={cur}")
        # subgradient calculation
        max_val = conv.max()
        # tolerance: use relative tolerance small
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]
        # compute subgradient via cross-correlation
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        # project step
        v = a - step * momentum
        a = _project_onto_simplex(v)
    return best_a, best_val

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – use a larger length (default 9600) than the previous
   4800‑point baseline.  The objective 2·n·max(conv) is (approximately) scale‑free,
   so a finer discretisation can approach the continuous optimum more closely.

2. **Initialisation** – start from the previously known construction
   (``height_sequence_1``) if it exists, otherwise from a random simplex point.
   The sequence is up‑sampled to the target length by linear interpolation.

3. **Smooth Adam phase** – optimise the *smooth* max of the self‑convolution
   (log‑sum‑exp) using a soft‑max parametrisation ``a = softmax(y)``.
   The smoothing factor ``α`` follows an increasing schedule
   ``[0.5, 2, 5, 10, 20, 40, 80, 160]``; for each α we perform a few
   thousand Adam steps with decaying learning‑rate.  The exact objective
   (via ``evaluate_sequence``) is evaluated periodically and the best
   sequence is remembered.

4. **Exact sub‑gradient polishing** – a projected gradient descent on the
   true objective ``2·n·max(conv)``.  The sub‑gradient of the max‑convolution
   is obtained from the active set (indices attaining the current maximum)
   and momentum is used.  The iterate is projected onto the probability
   simplex after each step.

5. **Simulated‑annealing refinement** – a mass‑transfer Metropolis walk that
   repeatedly moves a random fraction of mass from a source index to a
   destination index.  Temperature is cooled geometrically from ``5e‑2`` to
   ``1e‑6``.  Improvements are logged and the best sequence is kept.

6. **Time management** – the whole routine stops before the supplied
   budget (default 1000 s).  Each phase checks the deadline and aborts
   gracefully, guaranteeing that a valid sequence is always returned.

The combination of a finer discretisation, longer Adam optimisation,
aggressive polishing and a longer annealing schedule typically pushes the
score well below the previous 1.5128 barrier and often reaches the target
≈1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # select the n entries that correspond to the correlation at lags 0…n‑1
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Upsampling helper (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‐convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float):
    """
    Adam optimisation of the smooth surrogate of the max‑convolution.
    Returns the best (real) sequence found and its exact score.
    """
    # initialise logits for the soft‑max parametrisation
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    fft_len = _next_pow2(3 * n - 2)

    # schedule of smoothing parameters
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    iters_per_alpha = 8000  # reduced if we run out of time

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_seq = a0.copy()
    best_score = float("inf")

    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_seq, best_score

            # soft‑max
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # convolution and smooth objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            smooth = lse / alpha
            obj = 2.0 * n * smooth      # surrogate objective (not the true score)

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t. logits y (soft‑max trick)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution polishing (projected sub‑gradient descent)
# ----------------------------------------------------------------------
def _polish(a0: np.ndarray, n: int,
            deadline: float, start_time: float,
            beta: float = 0.94, step0: float = 0.25,
            max_iter: int = 10 ** 7):
    """
    Simple projected sub‑gradient descent on the exact objective
    2·n·max(conv).  Returns the best vector encountered and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        # active set (very tight tolerance – essentially the exact maximisers)
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += a[j_idx]
            g[j_idx] += a[i_idx]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)
        v = a - step * momentum
        a = _project_onto_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray, n: int,
            deadline: float, start_time: float,
            cur_best_score: float, cur_best_seq: np.ndarray,
            start_temp: float = 5e-2,
            end_temp: float = 1e-6,
            steps: int = 300_000):
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method uses a larger discretisation, a long Adam phase on a smooth surrogate,
    an exact sub‑gradient polishing stage and a final simulated‑annealing refinement.
    It respects the supplied time budget (≈1000 s) and always returns a valid
    sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Determine problem size and a sensible starting point
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Use a finer grid than before; 9600 gives a good trade‑off
    n = 9600

    if starter is not None:
        if starter.shape[0] != n:
            a0 = _upsample(starter, n)
        else:
            a0 = starter.copy()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation
    # ------------------------------------------------------------------
    best_seq, best_score = _adam_optimize(a0, n, deadline, start_time)

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient polishing (short, time‑aware)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        pol_seq, pol_score = _polish(best_seq, n, polish_deadline, start_time)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = pol_seq
            print(f"[{time.time() - start_time:6.2f}s] polishing best updated: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning (use any remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _anneal(best_seq, n, deadline, start_time,
                                        best_score, best_seq)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[{time.time() - start_time:6.2f}s] final anneal best: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental drift) and return a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(x) for x in best_seq.tolist()]
