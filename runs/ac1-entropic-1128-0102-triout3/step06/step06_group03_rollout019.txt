    x = np.arange(n)

    hann /= hann.sum()
    inits.append(hann)

    rand = np.random.rand(n)
    rand /= rand.sum()
    inits.append(rand)
    # previous best if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # upsample/downsample to length n
            if prev.size != n:

                new_x = np.linspace(0.0, 1.0, n)
                prev_up = np.interp(new_x, old_x, prev)
                prev_up = np.maximum(prev_up, 0)
                prev_up /= prev_up.sum()
                prev = prev_up
            else:
                prev = prev / prev.sum()
            inits.append(prev)
    except Exception:
        pass

    best_seq = None

    total_time = deadline - start
    grad_deadline = start + total_time * 0.70
    # divide gradient time equally among initial candidates
    per_init = grad_deadline - start
    per_init = per_init / len(inits) if len(inits) > 0 else 0
    for idx, a0 in enumerate(inits):
        if time.time() >= grad_deadline:
            break
        # define time for this init
        init_deadline = start + (idx + 1) * per_init
        # run exponentiated gradient descent
        a_opt, score_opt = exp_gradient_descent(

    if best_seq is not None and time.time() < deadline:

    else:

"""
Algorithm: multi‑start sub‑gradient descent on the max‑convolution
followed by a cheap simulated‑annealing local walk.
We build a few diverse initial step‑functions (triangular,
Hann window, a sparse “comb”, a random Dirichlet and the optional
previous best), then for each one minimise
    score = 2·n·max(a∗a)
by repeated evaluation of the self‑convolution, computing a
sub‑gradient of the maximal entry and projecting back onto the
probability simplex.  After the gradient phase the best candidate
is polished with a local mass‑shifting walk that accepts occasional
up‑hill moves.  The whole routine respects the 1000 s time budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length = 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())            # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return conv.real


# ----------------------------------------------------------------------
# Simple “project onto the simplex” – we use rescaling after clipping,
# which is sufficient for our non‑negative updates.
# ----------------------------------------------------------------------
def _renorm_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0 and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Sub‑gradient descent for  score = 2·n·max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         base_step: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    # ensure it lies on the simplex
    a = _renorm_simplex(a)

    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        k = int(conv.argmax())          # index where the maximum occurs

        # sub‑gradient of the max entry:
        #   g_i = 2 * a_{k-i}   if   0 ≤ k‑i < n , else 0
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = base_step / math.sqrt(it)

        # tentative move
        a_new = a - step * g
        a_new = _renorm_simplex(a_new)

        # evaluate
        new_score = 2.0 * n * _conv_self(a_new).max()

        # accept only if we improve (monotone descent)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] subgrad improve score={new_score:.6f} it={it}")

        # occasional jitter to escape flat regions
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = _renorm_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Local search – cheap mass‑shifting walk with Metropolis acceptance
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of probability mass from a
    ``donor’’ index (chosen proportionally to its contribution to the
    current maximal convolution entry) to a ``receiver’’ that contributes
    little.  Accept uphill moves with a slowly decreasing temperature.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()
    best_score = cur_score

    it = 0
    # temperature schedule – from 5·10⁻² down to 10⁻⁶
    T_start, T_end = 5e-2, 1e-6
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        k = int(conv.argmax())

        # contribution of each index to conv[k]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        if contrib.sum() > 0:
            donor = np.random.choice(n, p=contrib / contrib.sum())
        else:
            donor = np.random.randint(n)

        # pick a receiver from the low‑contribution tail
        low_thr = np.percentile(contrib, 20)      # bottom 20 %
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = np.random.randint(n)
        else:
            receiver = np.random.choice(low_candidates)

        if donor == receiver or a[donor] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of donor mass
        eps = a[donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        # Metropolis acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] local improve score={new_score:.6f} it={it}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _init_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised to sum = 1."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([inc, inc[::-1]])
    else:
        tri = np.concatenate([inc, inc[-2::-1]])
    return tri / tri.sum()


def _init_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window, normalised to sum = 1."""
    x = np.arange(n, dtype=float)
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    return w / w.sum()


def _init_comb(n: int, nb_nonzero: int = 64) -> np.ndarray:
    """Sparse ‘comb’ – nb_nonzero equally spaced spikes, normalised."""
    a = np.zeros(n, dtype=float)
    positions = np.linspace(0, n - 1, nb_nonzero, dtype=int)
    a[positions] = 1.0
    return a / a.sum()


def _init_random(n: int) -> np.ndarray:
    a = np.random.rand(n)
    return a / a.sum()

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    start_time = time.time()

    try:

def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass away from hot spots, accepting
    uphill moves with a slowly cooling temperature."""
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()
    best_score = cur_score
    it = 0
    T_start, T_end = 5e-2, 1e-6                # temperature schedule

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())

        # contribution of each index to conv[k]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        if contrib.sum() > 0.0:
            donor = np.random.choice(n, p=contrib / contrib.sum())
        else:
            donor = np.random.randint(n)

        # pick a receiver from the low‑contribution tail
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            receiver = np.random.randint(n)
        else:
            receiver = np.random.choice(low_candidates)

        if donor == receiver or a[donor] < 1e-14:
            continue

        # transfer up to 30 % of donor mass
        eps = a[donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[receiver] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        # Metropolis acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed
        accept = delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12))
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] local improve "
                      f"score={new_score:.6f} it={it}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def _init_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, normalised to sum = 1."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([inc, inc[::-1]])
    else:
        tri = np.concatenate([inc, inc[-2::-1]])
    return tri / tri.sum()


def _init_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window, normalised to sum = 1."""
    x = np.arange(n, dtype=float)
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    return w / w.sum()


def _init_comb(n: int, nb_nonzero: int = 128) -> np.ndarray:
    """Sparse “comb” – nb_nonzero equally spaced spikes, normalised."""
    a = np.zeros(n, dtype=float)
    positions = np.linspace(0, n - 1, nb_nonzero, dtype=int)
    a[positions] = 1.0
    return a / a.sum()


def _init_random(n: int) -> np.ndarray:
    """Uniform Dirichlet‑type random vector."""
    a = np.random.rand(n)
    return a / a.sum()

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine builds several starter shapes, refines each with projected
    sub‑gradient descent, and finally polishes the best result with a
    Metropolis local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 0.5

    # Length of the candidate (large enough to give a good bound)
    n = 9600

    # ------------------------------------------------------------------
    # Build starter sequences
    # ------------------------------------------------------------------
    starters = [
        _init_triangular(n),
        _init_hann(n),
        _init_comb(n, nb_nonzero=128),
        _init_random(n),
    ]

    # Include the previous best if the global ``height_sequence_1`` exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = _renorm_simplex(prev)
            starters.append(prev)
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Gradient phase – 70 % of the time budget
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    grad_deadline = start_time + total_time * 0.70
    per_start_deadline = (grad_deadline - start_time) / max(1, len(starters))

    for idx, s in enumerate(starters):
        if time.time() >= grad_deadline:
            break
        this_deadline = start_time + (idx + 1) * per_start_deadline
        a_opt, score_opt = _subgradient_descent(
            s, n, this_deadline, start_time, base_step=1.0
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] starter {idx} "
                  f"best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Local polishing – remaining 30 % of the budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_loc, score_loc = _local_search(best_seq, n, deadline, start_time)
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:.2f}s] final local best "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be triggered)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Final normalisation – protects against tiny drift
    best_seq = _renorm_simplex(best_seq)

    # Return a plain Python list for the evaluation harness
    return [float(v) for v in best_seq]
