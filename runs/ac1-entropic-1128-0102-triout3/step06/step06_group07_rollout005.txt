def _solve_lp_for_b(f, M):
    n = len(f)
    # Build matrix A_ub shape (2*n-1, n)
    # Use vectorized approach: for i from 0 to n-1, A_ub[i:i+n, i] = f
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = f
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    ub = np.full(rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return zero vector
        return np.zeros(n)
    b = res.x
    return b

"""
Algorithm: 2‑stage optimisation for the self‑convolution L∞‑norm.

Stage 0 – Small dimensional LP updates.
    * Keep the vector on the probability simplex (sum = 1, all ≥0).
    * For the current vector f compute M = max(f*f).
    * Solve a linear program: maximise Σ b_j  subject to
          f * b ≤ M  (component‑wise, linear in b)  and  b ≥ 0 .
      Only variables with f_j>0 are kept – otherwise the LP would be
      unbounded.
    * Mix the obtained direction b (normalised to sum = 1) into f with a
      small step size t (line‑search over a few candidates).  The mix
      stays on the simplex.
    * Repeat until the time budget is exhausted or no improvement is
      observed.

Stage 1 – Projected gradient descent (Nesterov) for fine‑tuning.

The two stages are cheap (the LP for n≈1024 has ~2 · 10⁶ coefficients)
and together fit well inside the 1000 s budget.  The LP step drives the
solution quickly to a region where the sub‑gradient becomes small;
the gradient stage then polishes the result.

The routine respects the global time budget, prints occasional progress
messages and finally returns the best vector (as a plain Python list)
found during the run.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent (Nesterov) on the simplex
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.3,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Minimise 2·n·max(conv(a)) with a∈simplex."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programme: maximise Σ b  subject to  f∗b ≤ M ,  b ≥ 0 .
# Only variables with f_i > 0 are kept – otherwise the LP would be
# unbounded.
# ----------------------------------------------------------------------
def _solve_lp_for_b(f: np.ndarray, M: float,
                    max_var: float = 1e6) -> np.ndarray:
    """Return a feasible b (same length as f) solving the LP."""
    n = f.shape[0]
    # keep only indices with non‑negligible mass
    act = np.where(f > 1e-12)[0]
    if act.size == 0:
        return np.zeros_like(f)

    rows = 2 * n - 1
    cols = act.size
    A = np.zeros((rows, cols), dtype=float)

    # each column i (corresponding to variable b_i) is a shifted copy of f
    for col_pos, i in enumerate(act):
        A[i:i + n, col_pos] = f

    c = -np.ones(cols)               # maximise Σb  <=> minimise -Σb
    b_ub = np.full(rows, M)          # f∗b ≤ M
    bounds = [(0.0, max_var)] * cols

    # the HiGHS interior‑point method is fast for dense problems of this size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    b = np.zeros_like(f)
    if res.success:
        b[act] = res.x
    else:
        # fall back to the zero vector (no improvement)
        pass
    return b


# ----------------------------------------------------------------------
# Light local search (mass transfer) – used after gradient descent
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      cur_best_score: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing on the simplex."""
    best_a = a.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < 200_000:
        it += 1
        # choose source proportional to mass, destination uniformly
        src = rng.choice(n, p=best_a)
        dst = rng.integers(n)
        if src == dst or best_a[src] < 1e-12:
            continue
        eps = best_a[src] * rng.random() * 0.25  # move up to 25 % of src mass
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_simplex(cand)

        score = 2.0 * n * _conv_self(cand).max()
        if score < best_score:
            best_score = score
            best_a = cand
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) preserving the simplex constraint
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The method combines a specialised linear‑program
    based update with projected gradient descent and a short hill‑climbing
    phase, all respecting the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so the harness does not cut us exactly at the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Starting vector – use the provided construction if available,
    # otherwise random.
    # ------------------------------------------------------------------
    n0 = 1024                     # modest size for the LP stage
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
                a = _upsample(starter, n0)
            else:
                raise ValueError
        except Exception:
            a = np.random.rand(n0)
            a /= a.sum()
    else:
        a = np.random.rand(n0)
        a /= a.sum()

    best_seq = a.copy()
    best_score = 2.0 * n0 * _conv_self(a).max()

    # ------------------------------------------------------------------
    # Stage 0 – LP based improvement loop
    # ------------------------------------------------------------------
    lp_iters = 0
    max_lp_iters = 250
    t_candidates = (0.02, 0.05, 0.1, 0.2, 0.3, 0.4)

    while time.time() < deadline - 5.0 and lp_iters < max_lp_iters:
        lp_iters += 1
        f = best_seq.copy()
        conv = _conv_self(f)
        M = conv.max()

        b = _solve_lp_for_b(f, M)
        sumb = b.sum()
        if sumb < 1e-12:          # LP did not produce a useful direction
            # random restart to avoid stagnation
            f = np.random.rand(n0)
            f /= f.sum()
            continue

        b_norm = b / sumb

        improved = False
        for t in t_candidates:
            cand = (1.0 - t) * f + t * b_norm
            cand = _project_simplex(cand)   # keep the simplex property
            cand_score = 2.0 * n0 * _conv_self(cand).max()
            if cand_score < best_score - 1e-9:
                best_score = cand_score
                best_seq = cand
                improved = True
                print(f"[{time.time() - start_time:6.2f}s] LP step iter={lp_iters}, "
                      f"t={t:.2f}, score={cand_score:.6f}")
                break
        if not improved:
            # No improvement with the tested steps – try a tiny random jitter
            jitter = np.random.rand(n0) * 1e-4
            cand = _project_simplex(best_seq + jitter)
            cand_score = 2.0 * n0 * _conv_self(cand).max()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand

    # ------------------------------------------------------------------
    # Stage 1 – Projected gradient descent (Nesterov) for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        grad_deadline = time.time() + min(remaining, 250.0)   # at most 250 s here
        a_opt, score_opt = _run_gradient_descent(
            best_seq, n0, grad_deadline, start_time,
            beta=0.92,
            step0=0.28,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] gradient polish improve: "
                  f"score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Very light hill‑climbing refinement (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _local_hill_climb(
            best_seq, n0, deadline, start_time, best_score
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final hill‑climb improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
