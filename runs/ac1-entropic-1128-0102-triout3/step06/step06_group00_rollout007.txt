def _smooth_gradient(a, conv, alpha, fft_len):
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad
```

                    step0=0.25, beta=0.94):

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The search proceeds in four stages:

1. **Diverse seeding** – generate a handful of analytically‑motivated
   candidates (Gaussian, exponential, triangular, cosine, Dirichlet,
   etc.) on a modest length (≈1500).  Keep the best one.

2. **Fast smooth optimisation on the small problem** – minimise a
   smooth approximation of the max‑convolution using Adam on the
   simplex.  The smoothing parameter α is increased from ≈0.5 to ≈200
   during the phase.  This yields a high‑quality shape cheaply.

3. **Upsample & exact optimisation** – linearly upsample the shape to
   the final length (4800), then continue optimisation on the true
   objective using (i) another Adam run on the smooth approximation,
   followed by (ii) a projected sub‑gradient descent with momentum.
   The best iterate is retained throughout.

4. **Final polish** – a short simulated‑annealing pass moves tiny
   amounts of mass between entries to escape shallow local minima.

All updates stay on the probability simplex (non‑negative, sum = 1).
The routine respects the 1000‑second budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all near‑max entries."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
        fα(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∇fα(a) (same shape as a).
    """
    # soft‑max weights
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)      # subtract max for stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]                     # reversed for correlation
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.size - 1]
    grad = 2.0 * corr[w_rev.size - 1 - np.arange(a.shape[0])]
    return grad


def _compute_obj(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Candidate generation on a given length
# ----------------------------------------------------------------------
def _generate_candidates(n: int, rng: np.random.Generator):
    """Return the best analytically‑motivated candidate for length n."""
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Gaussian bells (different widths)
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 30.0, n / 3.0, 6):
        i = np.arange(n, dtype=float)
        a = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
        candidates.append(a)

    # Exponential decays
    for lam in np.linspace(1e-4, 1.5e-3, 6):
        a = np.exp(-lam * np.arange(n, dtype=float))
        candidates.append(a)

    # Linear ramps (up and down)
    a = np.arange(1, n + 1, dtype=float)
    candidates.append(a)
    candidates.append(a[::-1])

    # Triangular shape
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(float)
    candidates.append(tri)

    # Cosine bell (half‑wave)
    x = np.arange(n, dtype=float)
    cos_shape = 0.5 * (1.0 + np.cos(np.pi * (x - centre) / centre))
    candidates.append(cos_shape)

    # Dirichlet‑like samples (Gamma draws)
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(a)

    best_a = None
    best_score = float("inf")
    for a in candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a = a / s
        score = _compute_obj(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

            if src == dst or cur_seq[src] < 1e-12:
                continue
            eps = cur_seq[src] * rng.random() * 0.30
            prop = cur_seq.copy()
            prop[src] -= eps
            prop[dst] += eps

            new_score = _compute_obj(prop)
            delta = new_score - cur_score
            if delta < 0 or math.exp(-delta / T) > rng.random():
                cur_seq = prop
                cur_score = new_score

            if cur_score < best_anneal_score:
                best_anneal_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

        best_a = best_seq
        best_score = best_anneal_score

    # ------------------------------------------------------------------
    # Normalise final answer and return as a Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / final_n)

    return [float(x) for x in best_a]
