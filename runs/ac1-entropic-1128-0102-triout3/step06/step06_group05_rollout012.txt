"""
Hybrid annealing optimizer for the self‑convolution L∞ problem.

The routine works on the probability simplex (∑a_i = 1) and
uses a cheap incremental update of the linear convolution
when a small amount of mass ε is transferred from one index
to another.  This makes a simulated‑annealing “mass‑transfer”
procedure fast enough to perform hundreds of thousands of
moves on a vector of several thousand entries within the
1000 s time budget.

The algorithm proceeds in two phases:
  1. optimise a “medium” size vector (default n = 4096) with
     aggressive annealing.
  2. up‑sample the best vector found to a larger size (default
     n = 8192) and polish it with a second annealing run.

Both phases keep the vector symmetric by occasional mirroring
(a ← (a + a[::-1]) / 2) and periodically recompute the full
convolution from scratch to avoid numerical drift.
The best distribution seen during the whole run is returned as a
plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities ------------------------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT (fast for all lengths)."""
    n = a.shape[0]
    # length of linear convolution = 2n-1
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)          # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Incremental convolution update (move ε from i → j) -------------------
# ----------------------------------------------------------------------


def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, eps: float) -> None:
    """
    Transfer a small amount ``eps`` from index i to index j.
    ``a`` and ``conv`` are updated in‑place.
    The update uses the exact formula

        conv_new = conv_old
                  + 2·eps·(shift(a, j) – shift(a, i))
                  + eps²·δ_{2i} + eps²·δ_{2j} – 2·eps²·δ_{i+j}

    where ``shift(a, s)`` is the a‑vector placed at indices s … s+n‑1.
    """
    n = a.shape[0]

    # slice updates (use the *old* a; a is unchanged until the end)
    delta = 2.0 * eps * a          # a is the pre‑move vector
    conv[i:i + n] -= delta
    conv[j:j + n] += delta

    # the quadratic (δ*δ) contribution
    conv[2 * i] += eps * eps
    conv[2 * j] += eps * eps
    conv[i + j] -= 2.0 * eps * eps

    # finally update the probability vector
    a[i] -= eps
    a[j] += eps

# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
