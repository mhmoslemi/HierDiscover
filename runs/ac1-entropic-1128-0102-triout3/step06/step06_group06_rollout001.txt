Propose a low‑maximum self‑convolution step function.

Algorithm
---------
1. **Warm‑start** – Randomly sample simple analytic shapes
   (product‑power, Gaussian, Beta, arcsine) and keep the best one.
2. **Smooth surrogate optimisation** – minimise
   τ·log ∑exp(conv/τ) with an Adam optimiser on the
   soft‑max parametrisation a_i = exp(y_i)/∑exp(y).  The gradient of

   finally converges to the true ∞‑norm problem.

# FFT‑based convolutions
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (exact objective for sum‑to‑1 vectors)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families for rapid warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_two_gaussians(n: int, sigma1: float, sigma2: float, mix: float) -> np.ndarray:
    """Mixture of two symmetric Gaussians centred at the ends."""
    i = np.arange(n)
    left = np.exp(-0.5 * ((i) / sigma1) ** 2)
    right = np.exp(-0.5 * ((n - 1 - i) / sigma2) ** 2)
    a = mix * left + (1.0 - mix) * right
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (τ·log ∑exp(conv/τ)) and its gradient
# ----------------------------------------------------------------------


def smooth_surrogate_and_grad(a: np.ndarray, tau: float):
    """
    Returns (value, grad_a) where
      value = τ * log ∑_k exp(conv_k / τ)
      grad_a = ∂value/∂a   (same shape as a)
    The gradient is computed via a second convolution.
    """
    n = a.shape[0]
    conv = conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights
    w = np.exp((conv - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient w.r.t a:  grad_i = 2 * Σ_k w_k * a_{k-i}
    # compute correlation w * a_rev
    a_rev = a[::-1]
    corr = conv_fft(w, a_rev)               # length = 3n‑2
    idx = (n - 1) - np.arange(n)            # pick the needed slice
    grad_a = 2.0 * corr[idx]

    # surrogate value (for possible logging)
    val = tau * (np.log(w_sum) + max_c / tau)  # = max_c + τ·log ∑exp((conv-max)/τ)
    return val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation
# ----------------------------------------------------------------------


def adam_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth surrogate using Adam on the logits y,
    where a = softmax(y).  τ is annealed from a modest start to 1e‑5.
    Returns the best exact‐objective vector seen.
    """
    # initialise logits from a0
    eps = 1e-12
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                       # base learning rate for logits

    # temperature schedule
    # start with τ ≈ 0.5·max(conv) (smooth enough) and decay geometrically
    conv0 = conv_self(a0)
    tau0 = max(conv0.max() * 0.5, 1e-6)
    tau_end = 1e-5
    decay = 0.9995

    best_a = a0.copy()
    best_sc = score_of(a0)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- obtain probabilities a from logits -----
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # ----- compute exact score occasionally (cheap) -----
        if it % 20 == 0:
            cur_sc = score_of(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_sc:.6f}"
                )

        # ----- temperature for the surrogate -----
        tau = max(tau_end, tau0 * (decay ** it))

        # ----- smooth surrogate value and gradient w.r.t a -----
        _, grad_a = smooth_surrogate_and_grad(a, tau)

        # ----- convert gradient to logits space (softmax Jacobian) -----
        mean = np.dot(a, grad_a)          # ⟨a,∂f/∂a⟩
        grad_y = a * (grad_a - mean)      # ∂f/∂y

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        y -= step

    # final exact evaluation of the last iterate
    final_a = np.exp(y - np.max(y))
    final_a /= final_a.sum()
    final_sc = score_of(final_a)
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = final_a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random ε from one index to another, accepting with a
    geometric temperature schedule.  Returns the best vector found.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found construction
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            starter = tmp / s
    except Exception:
        starter = None

    # Length schedule – grow gradually, shuffle to avoid bias if we stop early
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    random.shuffle(lengths)

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initial vector for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Allocate a time slice for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # allocate a fraction that grows with n
        frac = 0.12 / (len(lengths) - idx)
        slice_deadline = time.time() + max(8.0, remaining * frac)

        # --------------------------------------------------------------
        # Random parametric warm‑starts (quick exploration)
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3, 4])
            if kind == 0:      # product‑power family
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:    # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:    # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 3:    # arcsine‑like family
                cand = shape_arcsine(n)
            else:              # mixture of two Gaussians
                sigma1 = np.random.uniform(n / 30.0, n / 5.0)
                sigma2 = np.random.uniform(n / 30.0, n / 5.0)
                mix = np.random.rand()
                cand = shape_two_gaussians(n, sigma1, sigma2, mix)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Adam‑based smooth‑surrogate optimisation
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = adam_descent(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Cheap annealing polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # Keep the best vector for the next up‑sample step
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as a plain Python list (the external evaluator will clip/normalise)
    return [float(x) for x in best_seq]
