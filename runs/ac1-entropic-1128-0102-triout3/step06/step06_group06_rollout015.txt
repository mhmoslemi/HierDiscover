def adam_mirror_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    # init logit vector
    logit = np.log(np.maximum(a, 1e-12))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter = 0
    # temperature schedule
    tau_start = 0.1
    tau_end = 1e-4
    tau = tau_start
    # assume we want to reduce temperature exponentially based on iteration count; we will compute decay factor each iteration as (tau_end/tau_start) ** (1/ max_iters)
    # but we don't know max_iters beforehand; adapt: multiply by 0.999 each iteration
    decay = 0.9995

    while time.time() < deadline:
        iter += 1

        # compute conv
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={iter} score={cur_sc:.6f}")

        # compute softmax weights
        shift = conv.max()
        w = np.exp((conv - shift) / max(tau, 1e-12))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (smooth surrogate)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)  # length 3n-2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # compute gradient wrt logits (mirror)
        avg = np.dot(a, grad)
        dlogit = a * (grad - avg)

        # Adam update for logit
        m = beta1 * m + (1 - beta1) * dlogit
        v = beta2 * v + (1 - beta2) * (dlogit * dlogit)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        lr = 0.1  # base learning rate
        logit -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute new a via softmax
        # subtract max for numerical stability
        logit_shift = logit - np.max(logit)
        a = np.exp(logit_shift)
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            a = np.full(n, 1.0/n)
        else:
            a /= sum_a

        # anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc

"""Optimisation of a non‑negative step function for the objective

    evaluate_sequence(sequence) = 2·n·max(conv(sequence,sequence))/sum(sequence)²

The algorithm proceeds in increasing lengths (256 → … → 16384) and
maintains a probability vector (∑a = 1).  For each length it

1. generates several warm‑starts from simple analytic families,
2. refines the best warm‑start with Adam‑accelerated KL‑mirror‑descent
   on the smooth surrogate  τ·log ∑exp(conv/τ) (the temperature τ is
   annealed from 0.1 to 1e‑4),
3. finishes with a cheap mass‑transfer simulated‑annealing polish.

All operations are performed with FFT‑based convolutions (O(n log n)).
The best sequence seen overall is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1):
    2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families for quick warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_triangular(n: int) -> np.ndarray:
    """Simple symmetric triangle, peak at the centre."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential(n: int, lam: float) -> np.ndarray:
    """Decreasing exponential a_i ∝ exp(‑λ i)."""
    i = np.arange(n)
    a = np.exp(-lam * i)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Adam‑accelerated KL‑mirror‑descent on the simplex
# ----------------------------------------------------------------------


def adam_mirror_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) by minimising the smooth surrogate
        τ·log∑exp(conv/τ)
    using Adam on the log‑odds (KL‑mirror) representation.
    Returns the best vector seen and its exact score.
    """
    # initialise logits (log‑softmax) and Adam moments
    a = a0.copy()
    logit = np.log(np.maximum(a, 1e-12))
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = score_of(a)

    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.1

    # temperature schedule (soft‑max)
    tau = 0.1          # start relatively hot
    tau_end = 1e-4
    tau_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # exact score for tracking
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---- soft‑max weights for the smooth surrogate ----
        conv = conv_self(a)
        shift = conv.max()
        # avoid overflow/underflow
        arg = (conv - shift) / max(tau, 1e-12)
        arg = np.clip(arg, -700.0, 700.0)
        w = np.exp(arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate w.r.t. a ----
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)       # extract needed slice
        grad = 2.0 * corr[idx]

        # ---- mirror‑descent gradient (w.r.t. logits) ----
        avg = np.dot(a, grad)
        dlogit = a * (grad - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * dlogit
        v = beta2 * v + (1.0 - beta2) * (dlogit * dlogit)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logit -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- project back onto the simplex (soft‑max) ----
        # numerical stabilisation
        logit_shift = logit - np.max(logit)
        a = np.exp(logit_shift)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---- temperature anneal ----
        tau = max(tau_end, tau * tau_decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights of a step function that (hopefully)
    improves on the current best upper bound.  The routine respects the
    1000 s time budget, performs a multi‑scale optimisation and finally
    returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable height_sequence_1)
    # ------------------------------------------------------------------
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        starter = tmp / s if s > 0.0 else None
    except Exception:
        starter = None

    # Length schedule – increase gradually, up‑sample the current best.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    # Keep a deterministic order; random shuffling is unnecessary here.
    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Allocate a time slice for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # give a bit more time to larger n (proportional to log n)
        slice_budget = remaining * (math.log2(n) / sum(math.log2(m) for m in lengths))
        slice_deadline = time.time() + max(8.0, slice_budget)

        # --------------------------------------------------------------
        # Random parametric warm‑starts
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.2)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.randint(0, 5)
            if kind == 0:   # product‑power family
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:  # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 3:  # arcsine‑like family
                cand = shape_arcsine(n)
            else:            # triangular family
                cand = shape_triangular(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Adam mirror‑descent optimisation (soft‑max surrogate)
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.7
        a_opt, sc_opt = adam_mirror_descent(
            local_best_a, n, grad_deadline, start_time
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Cheap annealing polish (if any time left)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # propagate the best vector to the next scale
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the external evaluator will clip/normalise)
    return [float(x) for x in best_seq]
