def optimize(n, init_vec, time_budget):
    a = init_vec
    best = a.copy()
    best_score = objective(a)
    it=0
    start = time.time()
    while time.time() - start < time_budget:
        it+=1
        # compute conv and max
        conv = _conv_self(a)
        max_val = conv.max()
        # compute active mask where conv close to max
        tol = 1e-12
        active = np.abs(conv - max_val) <= tol
        # subgradient w.r.t a:
        w = active.astype(float) / active.sum()
        a_rev = a[::-1]
        # compute gradient using convolution of w with reversed a
        grad = 2 * _conv_fft(w, a_rev)[n-1:n-1+n]
        # exponentiated gradient update
        eta = 0.5 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # renorm
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= a_sum
        # maybe symmetrize
        if np.random.rand() < 0.1:
            a = 0.5 * (a + a[::-1])
        # compute score
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[...] exponentiated improve n={n} it={it} score={cur_score:.6f}")
    return best, best_score
```

def _run_expgrad(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    eta0 = 0.5
    it = 0
    while time.time() < deadline:

        grad, max_val = _grad_max_conv(a)

            a /= a.sum()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgrad improve n={n} it={it} score={cur_score:.6f}")

def _run_annealing_simple(a0, n, deadline, start_time, cur_best_score):
    a = a0.copy()
    conv = _conv_self(a)
    score = 2 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_best_score

    # set annealing schedule
    start_temp = 0.05
    end_temp = 1e-5
    total_steps = 50000
    for it in range(1, total_steps+1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / total_steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # up to 10% transfer
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        conv_prop = _conv_self(a_prop)
        new_score = 2 * n * conv_prop.max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")
    return best_seq, best_score

"""
Hybrid multi‑scale optimizer for the self‑convolution L∞ problem.

The optimizer works on the probability simplex (∑a_i = 1) because the
objective returned by ``evaluate_sequence`` is scale‑invariant.
It proceeds through increasing vector lengths, warm‑starting each size
by linear interpolation, and for every size runs three phases:

1. **Exponentiated‑gradient descent** on the exact max‑convolution
   (sub‑gradient of the L∞ norm).  The update a_i ← a_i·exp(−η·g_i)
   automatically respects non‑negativity and the simplex after
   renormalisation.

2. **Projected sub‑gradient descent** (Euclidean) for a short
   refinement pass.

3. **Targeted annealing** that moves a small ε from one index to
   another and accepts worsening moves with a decaying temperature.

All phases respect the overall time budget (default 1000 s) and
continually keep the best vector encountered.  The final vector is
returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based) ------------------------------------
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # clip tiny negatives caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set) -------------------
# ----------------------------------------------------------------------


def _grad_max_conv(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.

    Returns (grad, max_value) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_val = conv.max()
    # active indices (within a tiny tolerance)
    tol = 1e-12
    active = np.where(np.abs(conv - max_val) <= tol)[0]
    if active.size == 0:                     # fallback – single max entry
        active = np.array([conv.argmax()], dtype=int)

    grad = np.zeros_like(a)
    inv = 1.0 / active.size                    # equal weight for each active index
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev_idx = k - idx
        grad[idx] += 2.0 * inv * a[rev_idx]
    return grad, max_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution ----------
# ----------------------------------------------------------------------


def _run_expgrad(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of the exact max‑convolution.
    Returns the best vector seen and its exact objective value
    (2·n·max(conv) because the vector stays on the simplex).
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    eta0 = 0.7                     # initial learning rate
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _grad_max_conv(a)

        # step size decays like η₀/√t
        eta = eta0 / math.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional tiny random perturbation to avoid stagnation
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgrad improve n={n} it={it} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic projected sub‑gradient descent -------------------------
# ----------------------------------------------------------------------


def _run_subgrad(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _grad_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} "
                  f"score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing -------------------------------
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination index.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_best_score

    start_temp = 5e-2
    end_temp = 1e-5
    steps = 150_000

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------

        # allocate a slice of the remaining time for this length

        if remaining <= 0.0:

        # ------------------------------------------------------------------

            a_sub, score_sub = _run_subgrad(best_local_a, n, sub_deadline, start_time)
            if score_sub < best_score:

                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad "

        # ------------------------------------------------------------------
        # Phase 3 – final annealing polishing
        # ------------------------------------------------------------------
