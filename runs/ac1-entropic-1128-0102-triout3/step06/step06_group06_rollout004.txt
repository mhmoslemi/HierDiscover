"""
Propose a low‑max‑self‑convolution step‑function.

The algorithm works in three stages:

1. **Parametric warm‑starts** – several analytic families (product‑power,
   Gaussian, Beta, sine, arcsine) are sampled at random for the current
   length.  The best of these is kept as a start point.

2. **Projected sub‑gradient descent** – the exact objective
   `max(conv(a,a))` is minimised by repeatedly moving a step in the
   sub‑gradient (the gradient of the currently maximal convolution entry)
   and projecting back onto the probability simplex with an
   exponentiated‑gradient update.

3. **Cheap mass‑transfer annealing** – a few thousand random transfers
   of a small amount of probability mass are tried and accepted
   according to a geometric temperature schedule.

The search proceeds over a schedule of increasing lengths
(512 → 1024 → 2048 → 4096).  The best vector found for a length is
upsampled and used as the start for the next (larger) length, giving a
smooth continuation of the shape.  All operations run in O(n log n) time
thanks to FFT‑based convolutions, and the whole routine obeys the
`budget_s` time limit supplied by the runner.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Return the smallest power‑of‑two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Parametric families
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.maximum(term, 1e-12)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.maximum(term, 1e-12)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_sine(n: int) -> np.ndarray:
    """Half‑sine shape a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) using projected exponentiated‑gradient updates.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # Current convolution and its maximal index
        conv = conv_self(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # Sub‑gradient of the max entry w.r.t a
        # (∂/∂a_i) conv[max_idx] = 2 * a_{max_idx-i}   (if index in range)
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # indices where a_{max_idx-i} is defined
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[max_idx - idx]

        # Step‑size schedule (decreasing like 1/√it)
        eta = 0.7 / math.sqrt(it)

        # Exponentiated‑gradient update (KL‑projection)
        expo = -eta * grad
        expo = np.clip(expo, -700.0, 700.0)   # avoid overflow
        a = a * np.exp(expo)

        # Renormalise onto the simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # Track best
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based simulated annealing: move a random ε from one
    index to another.  Accept moves that improve the score or with a
    Metropolis criterion based on a geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # transfer up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

            kind = np.random.choice([0, 1, 2, 3, 4])
            if kind == 0:            # product‑power
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:          # Gaussian
                sigma = np.random.uniform(n / 40.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:          # Beta
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 3:          # arcsine‑like
                cand = shape_arcsine(n)
            else:                    # half‑sine
                cand = shape_sine(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] warm‑start improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # 2. Sub‑gradient descent on the exact max‑norm
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = subgradient_descent(
            local_best_a, n, grad_deadline, start_time
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # 3. Cheap annealing polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.20)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=40000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # Use the current best as warm‑start for the next length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list; the external evaluator will clip/normalise.
    return [float(x) for x in best_seq]
