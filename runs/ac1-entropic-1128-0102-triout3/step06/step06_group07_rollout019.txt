The optimiser is a projected Adam (adaptive‐momentum) algorithm that works

"""

# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} (O(n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct handling of the diagonal term.
# The gradient of conv[k] w.r.t a_i is 2*a_{k-i} when the index is in range.
# For several (almost) maximal k we average the gradients.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.30,
              beta1: float = 0.90,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with simplex projection.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the current maximum convolution entry
        grad = _subgradient_max(a, conv)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a decaying learning rate (∝ 1/√it) works well for this problem
        lr = lr0 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Light annealing using elementary mass‑transfer moves.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric schedule for temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportionally to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (up to 30%) of the source mass
        eps_mass = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps_mass
        a_prop[dst] += eps_mass

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    `evaluate_sequence`.  The routine works as follows:

    1. Initialise a vector of length n≈12288 – use the globally provided
       `height_sequence_1` when available, otherwise start from a random point.
    2. Run a long Adam optimisation on the simplex (projected gradient
       descent with adaptive learning rates) until the time budget is
       exhausted or a large number of iterations have been performed.
    3. If time permits, perform a short simulated‑annealing pass to polish the
       result.
    4. Return the best vector (as a plain Python list) after normalising it
       to sum = 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose the working length.  12288 yields a good trade‑off between
    # expressiveness and speed on the 1000 s budget.
    # ------------------------------------------------------------------
    n = 12288

    # ------------------------------------------------------------------
    # Try to initialise from the previously published best construction.
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    if starter is not None and starter.shape[0] == n:
        a0 = starter.copy()
    elif starter is not None:
        # linearly interpolate to the required length
        old_x = np.linspace(0.0, 1.0, starter.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, starter)
        total = a0.sum()
        if total > 0:
            a0 /= total
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Adam optimisation (most of the time budget is spent here)
    # ------------------------------------------------------------------
    best_seq, best_score = _run_adam(
        a0,
        n,
        deadline,
        start_time,
        lr0=0.30,
        beta1=0.90,
        beta2=0.999,
    )

    # ------------------------------------------------------------------
    # Light annealing phase if there is time left
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
