"""
Mirror‑descent + soft‑max smoothing optimizer for the
self‑convolution L∞‑norm problem.

The algorithm proceeds in three stages:

1.  Small‑scale Adam projected‑gradient runs (many restarts) on a
    modest size (≈2 k).  This quickly finds a decent basin.
2.  Upsample the best vector to a medium size (≈4 k) and continue with
    Adam for a longer period.
3.  Upsample again to a large size (≈12 k) and run a mirror‑descent
    optimiser that minimises a smooth approximation of the max‑convolution
    via a soft‑max (temperature annealed to near‑zero).  Finally a light
    simulated‑annealing refinement moves mass between coordinates.

All stages respect the global time budget (default 1000 s) and the
final vector is normalised to sum = 1 (the scaling does not affect the
evaluation function).  The implementation uses only NumPy and the
provided `evaluate_sequence` for scoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (used by Adam)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Averaged sub‑gradient of the max‑convolution."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= float(max_inds.size)
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (additive updates)
# ----------------------------------------------------------------------
def _run_adam_descent(a0: np.ndarray, n: int,
                      deadline: float, start_time: float,
                      lr: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Adam with simplex projection to minimise 2·n·max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max smoothing of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax_max(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   τ·log(∑_k exp(conv[k]/τ))   w.r.t. a.
    Returns a vector of shape (n,).
    """
    n = a.shape[0]
    # stable soft‑max on conv
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / tau)
    w = exp_vals / exp_vals.sum()                # shape (2n-1,)

    # compute  g_i = 2 * Σ_k w_k * a_{k-i}
    # This is a cross‑correlation of a and w.
    w_rev = w[::-1]                              # length 2n‑1
    L = w_rev.shape[0]
    N = 1 << ((n + L - 1).bit_length())          # FFT length
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w_rev, N)
    corr_full = np.fft.irfft(A * W, N)[: n + L - 1]   # length n+L‑1
    offset = L - 1
    grad = 2.0 * corr_full[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent optimiser using the soft‑max surrogate
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        step0: float = 0.5,
                        tau0: float = None,
                        tau_factor: float = 0.9997,
                        max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Entropic (exponentiated) mirror descent on the soft‑max
    approximation of max(conv).  The temperature τ is annealed
    multiplicatively to approach the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # initial temperature – if not supplied use the current max value
    current_conv = _conv_self(a)
    if tau0 is None:
        tau = max(1e-6, current_conv.max())
    else:
        tau = tau0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Mirror improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of smooth surrogate
        grad = _gradient_softmax_max(a, conv, tau)

        # step size – classic 1/√t schedule
        step = step0 / math.sqrt(it)

        # exponentiated (multiplicative) update, guarded against overflow
        exp_arg = -step * (grad - grad.max())
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # project back onto the simplex
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # anneal temperature
        tau = max(tau * tau_factor, 1e-9)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (same as the previous implementation)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Light mass‑transfer SA; keeps the best sequence seen."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample by linear interpolation and renormalise to sum = 1
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
