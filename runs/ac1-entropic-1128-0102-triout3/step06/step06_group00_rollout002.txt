"""
Adam‑softmax optimizer for the max‑convolution bound.

The objective  f(a)=2·n·max(conv(a,a))  (the denominator of evaluate_sequence
is 1 after normalising the vector to a probability distribution).  We
optimise directly on the simplex by representing a as a soft‑max of an
unconstrained vector `z`.  The gradient of f w.r.t. `a` is the exact
sub‑gradient of the max‑convolution (computed on the active set).  This
gradient is back‑propagated through the soft‑max (∂a/∂z) and fed to an
Adam optimiser.  The algorithm runs until the time budget (≈1000 s) is
exhausted, keeping the best sequence seen.  A tiny simulated‑annealing
polish is performed with any remaining time.

The routine respects all security checks of ``evaluate_sequence`` by
producing a non‑negative vector that sums to 1 (all entries are far
below the 1000 cap).  It also uses the global ``height_sequence_1`` if
available as a warm start.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    It sums contributions from all indices whose convolution value is
    within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = z - np.max(z)
    expz = np.exp(shift)
    return expz / expz.sum()

                    new_x = np.linspace(0.0, 1.0, n)
                    a0 = np.interp(new_x, old_x, a0)
        except Exception:          # pragma: no cover – fallback below
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.size == 0:
        a0 = np.random.rand(n)

    # --------------------------------------------------------------
    # Normalise to a probability vector and add a tiny random jitter to
    # avoid being stuck exactly at a (potentially) flat saddle.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    jitter = 0.001 * np.random.randn(n)
    a0 = a0 + jitter
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    # --------------------------------------------------------------
    # Work in the unconstrained log‑space (soft‑max parameterisation).
    # --------------------------------------------------------------
    eps = 1e-12
    z = np.log(a0 + eps)

    # --------------------------------------------------------------
    # Adam hyper‑parameters
    # --------------------------------------------------------------
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # --------------------------------------------------------------
    # Track the best sequence seen so far.
    # --------------------------------------------------------------
    conv0 = _conv_self(a0)
    best_seq = a0.copy()
    best_score = 2.0 * n * conv0.max()
    it = 0

    # --------------------------------------------------------------
    # Main optimisation loop (Adam on the simplex via soft‑max).
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # Soft‑max to obtain a feasible probability vector.
        a = _softmax(z)

        # Convolution and current objective value.
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # Keep the best iterate.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        # Exact sub‑gradient of max(conv) w.r.t. a.
        g = _subgradient_full(a, conv)

        # Gradient w.r.t. the log‑parameters (soft‑max Jacobian).
        dot = np.dot(g, a)
        grad_z = a * (g - dot)

        # Adam moment updates.
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)

        # Bias‑correction.
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Parameter step.
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # (optional) early termination if we have exhausted the budget.
        # The loop condition already checks the deadline.

    # --------------------------------------------------------------
    # Final tiny simulated‑annealing polish (uses any remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        score = best_score
        rng = np.random.default_rng()
        # Number of attempted moves – limited by the remaining budget.
        max_moves = 150_000
        for mv in range(1, max_moves + 1):
            if time.time() >= deadline:
                break
            # Temperature schedule (very gentle cooling).
            T = 5e-2 * (1e-6 / 5e-2) ** (mv / max_moves)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            eps_move = a[src] * rng.random() * 0.30
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_conv = _conv_self(a_prop)
            new_score = 2.0 * n * new_conv.max()
            delta = new_score - score

            if delta < 0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                score = new_score

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                      f"move={mv:,}, score={score:.6f}")

    # --------------------------------------------------------------
    # Normalise once more (protect against tiny numerical drift) and
    # return the sequence as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]
