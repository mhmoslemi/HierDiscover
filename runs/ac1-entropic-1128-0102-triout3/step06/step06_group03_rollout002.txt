"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm (≈1000 s total)
-------------------------
1. **Initialisation** – start from the previously found best sequence if
   available, otherwise random.  Upsample (linear interpolation) to a
   relatively large size (n = 9600) so that we have many degrees of
   freedom.

2. **Fast projected gradient descent** – minimise the ∞‑norm of the
   self‑convolution using a smooth soft‑max surrogate at the beginning
   (temperature ≈ 10⁻²) and switch to the exact max‑gradient once the
   temperature becomes negligible.  The gradient step is combined with
   Nesterov momentum and projection onto the probability simplex.
   Only a single FFT (the self‑convolution) is needed after the switch,
   which allows > 200 k iterations in the allocated time.

3. **Targeted local “peak‑shaving’’ search** – after the global optimisation
   we repeatedly move a small fraction of probability mass from indices
   that contribute most to the current maximal convolution entry to indices
   that contribute little.  A simple Metropolis rule with a cooling schedule
   accepts occasional uphill moves.

4. **Return** the best sequence (renormalised to sum = 1) as a plain list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: hard‑max (sub‑gradient) of the max‑convolution
# ----------------------------------------------------------------------
def _hard_max_gradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Gradient of max(conv) when the maximum occurs at `max_idx`."""
    n = a.shape[0]
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) gradient descent.
    Uses a soft‑max surrogate while the temperature is > 1e-6 and
    switches to the exact max‑gradient thereafter (no extra FFT needed).
    Returns best sequence seen and its raw score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # hyper‑parameters
    step0 = 1.0
    beta = 0.9
    T_start = 0.01          # start temperature for the soft‑max
    T_end = 1e-7            # final temperature (essentially exact max)
    T_switch = 1e-6         # when to stop using the expensive soft‑max gradient
    max_it = 10 ** 9        # safety upper bound

    while time.time() < deadline and it < max_it:
        it += 1

        # ------------------------------------------------------------------
        # 1. evaluate current score
        # ------------------------------------------------------------------
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2. temperature schedule
        # ------------------------------------------------------------------
        T = T_start * (T_end / T_start) ** (it / 200000.0)  # exponential decay

        # ------------------------------------------------------------------
        # 3. gradient (soft‑max or hard‑max)
        # ------------------------------------------------------------------
        if T > T_switch:
            # soft‑max weight vector (stable)
            scaled = (conv - cur_max) / T
            # subtract max to avoid overflow (max(scaled) == 0)
            exps = np.exp(scaled)
            w = exps / exps.sum()
            rev_a = a[::-1]
            full = _conv_rev_weight(rev_a, w)                 # length = 3n‑2
            grad = 2.0 * full[n - 1:n - 1 + n]                # centre block
        else:
            # exact sub‑gradient using the (unique) max index
            idx_max = int(np.argmax(conv))
            grad = _hard_max_gradient(a, idx_max)

        # ------------------------------------------------------------------
        # 4. Nesterov momentum + diminishing step
        # ------------------------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Targeted local search (peak shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass away from indices that
    contribute most to the current maximal convolution entry.
    A slowly cooling Metropolis acceptance rule permits occasional
    uphill moves to escape shallow traps.
    Returns the best sequence and its (raw) score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – probability proportional to its contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – pick from low‑contribution tail (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        # evaluate proposal
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # cooling schedule
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed

        # Metropolis acceptance
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine combines a fast projected
    gradient descent (soft‑max → exact max) with a dedicated peak‑shaving
    local search.  It respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load optional starter (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()          # normalise to the simplex
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Target size – larger n gives more freedom (we use 9600)
    # ------------------------------------------------------------------
    n_target = 9600

    # ------------------------------------------------------------------
    # Initialise a0
    # ------------------------------------------------------------------
    if starter is not None:
        a0 = _upsample(starter, n_target)
    else:
        a0 = np.random.rand(n_target)
        a0 /= a0.sum()

    # Small random jitter to avoid pathological symmetry
    jitter = 1e-4 * (np.random.rand(n_target) - 0.5)
    a0 = np.clip(a0 + jitter, 0, None)
    if a0.sum() == 0:
        a0 = np.full(n_target, 1.0 / n_target)
    else:
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Allocate time: 80 % gradient descent, 20 % local search
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    grad_deadline = start_time + 0.80 * total_time
    local_deadline = deadline

    # ------------------------------------------------------------------
    # Gradient descent phase
    # ------------------------------------------------------------------
    best_seq, best_score = _run_gradient_descent(
        a0, n_target, grad_deadline, start_time
    )
    print(f"[{time.time() - start_time:6.2f}s] after GD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if time.time() < local_deadline:
        best_seq, best_score = _run_targeted_local_search(
            best_seq, n_target, local_deadline, start_time
        )
        print(f"[{time.time() - start_time:6.2f}s] after local search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (tiny drift may happen because of rounding)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
```

This implementation follows the same overall strategy as the previous code but introduces several improvements:

* **Temperature schedule** that drives the soft‑max to the exact max, after which the gradient becomes an O(n) sub‑gradient (no extra FFT).
* **Larger target length (9600)** giving more degrees of freedom.
* **More aggressive allocation of the time budget** (≈ 80 % to global optimisation, ≈ 20 % to fine‑grained local search).
* **Simplified gradient computation after the switch** which allows many more iterations within the time limit, leading to a tighter bound on the maximal convolution and, consequently, a lower score.
