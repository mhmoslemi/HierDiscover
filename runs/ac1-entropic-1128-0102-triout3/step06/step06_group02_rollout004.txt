def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, numpy as np

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine final length, adapt based on budget
    base_n = 9600
    # maybe increase if time left >some threshold
    remaining_for_grad = deadline - start_time
    if remaining_for_grad > 600: # enough time
        n = int(base_n * 1.5)  # 14400
    else:
        n = base_n

    # choose initial a (vector)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            a = _upsample(prev, n)
        else:
            raise Exception
    except Exception:
        # build Hann window
        i = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))  # Hann
        a += np.random.rand(n) * 0.001  # small noise
        a = np.maximum(a, 0)
        a /= a.sum()

    # define simple functions if not defined earlier (but we will include definitions earlier)

    # Adam optimizer
    max_iters = int(1e6)  # but break early based on time
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_score = float('inf')
    T0 = 5e-2
    T_end = 1e-5
    for it in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break
        # compute conv
        conv = _conv_self(a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{now-start_time:.2f}s] Adam improve: n={n} it={it} score={score:.6f}")

        # temperature schedule
        T = T0 * (T_end / T0) ** (it / max_iters)

        # compute gradient of softmax approximation
        # note: conv - max_conv
        scaled = (conv - max_conv) / T
        # to avoid overflow, subtract max before exp.
        # Already scaling may produce large negative values -> exp underflows to 0
        exps = np.exp(scaled - np.max(scaled))  # actually max of scaled = 0, so skip?
        # but we want exp((conv - max_conv) / T) which is <=1
        # Let's compute safely: exp_vals = np.exp((conv - max_conv) / T)
        exp_vals = np.exp((conv - max_conv) / T)
        w = exp_vals / exp_vals.sum()
        # conv length = 2n-1, w length same
        # compute gradient: 2 * rev(a) * w (central n entries)
        rev_a = a[::-1]
        # convolution of rev_a and w
        n_a = a.shape[0]
        m_len = w.shape[0]
        fft_len = 1 << ((n_a + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n_a + m_len - 1]
        grad = 2.0 * full[n_a - 1: n_a - 1 + n_a]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)

"""
Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

The method consists of four phases:

1. **Progressive up‑sampling** – start from a short vector and repeatedly
   double its length, keeping the distribution normalised (sum = 1).

2. **Soft‑max (log‑sum‑exp) gradient descent** – minimise a smooth
   approximation of the max‑convolution.  The temperature is driven
   from 0.1 down to 1e‑8 and an Adam‑style adaptive step is used.
   Momentum and a projection onto the probability simplex keep the
   iterates feasible.

3. **Exact sub‑gradient descent** – after the smooth phase we descend
   on the true max‑convolution using the explicit sub‑gradient.

4. **Simulated‑annealing refinement** – a long Metropolis walk that
   transfers a small fraction of mass between two positions.
   The temperature schedule is very slow, allowing the walk to escape
   shallow local minima.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.  If a previous construction
(`height_sequence_1`) is present it is used as a warm start; otherwise
several random restarts are performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`
    and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of the max‑convolution.
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max (log‑sum‑exp) of the convolution."""
    maxc = conv.max()
    # weights w_i ∝ exp((conv_i - maxc) / T)
    exp_vals = np.exp((conv - maxc) / temperature)
    w = exp_vals / exp_vals.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – soft‑max (log‑sum‑exp) gradient descent with Adam.
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.4,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    T_start: float = 0.1,
                    T_end: float = 1e-8,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam‑style projected gradient descent on the soft‑max objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    eps = 1e-12

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        # temperature schedule – geometric decay to T_end
        T = T_start * (T_end / T_start) ** (it / max_iter)

        # gradient of the log‑sum‑exp approximation
        grad = _softmax_gradient(a, conv, T)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – we keep a modest base step and let Adam scale it
        step = step0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.05,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a (see analysis)
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k :: -1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# Phase 3 – simulated‑annealing refinement.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 30 % of the mass at i (larger moves early on)
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

            max_iter=5_000_000
        )

        # ==== Phase 2: exact sub‑gradient descent ==========================
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _run_subgradient_gd(
                a_opt, n, stage_deadline, start_time,
                step0=0.05, beta=0.9,
                max_iter=2_000_000
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # refresh the remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works with the final length we ended up with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        # allocate the whole remaining time to annealing
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-7,
            steps=4_000_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
