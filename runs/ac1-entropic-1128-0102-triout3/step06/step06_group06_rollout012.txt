def adam_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    # logit representation
    logits = np.log(a + 1e-30)  # avoid log0.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1
    tau = 5.0 / n
    tau_end = tau * 1e-3
    decay = 0.9995
    t = 0
    best_a = a.copy()
    best_sc = score_of(a)
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = conv_self(a)
        max_val = conv.max()
        # Softmax weights
        shift = max_val
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a (surrogate)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev) # length 3n-2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]
        # compute gradient w.r.t logits
        mean_grad = np.dot(grad, a)
        g_logits = a * (grad - mean_grad)
        # Adam update
        m = beta1 * m + (1 - beta1) * g_logits
        v = beta2 * v + (1 - beta2) * (g_logits * g_logits)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # descent step (subtract)
        logits -= eta * m_hat / (np.sqrt(v_hat) + eps)
        # recompute a
        # to improve numerical stability, subtract max logit
        max_logit = np.max(logits)
        exp_logits = np.exp(logits - max_logit)
        a = exp_logits / exp_logits.sum()
        # track exact score
        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam improve n={n} it={t} score={sc:.6f}")
        # anneal temperature
        tau = max(tau_end, tau * decay)
    return best_a, best_sc
```

"""
search for a step‑function with a low self‑convolution peak.

The core ideas are:

1.  **Diverse warm‑starts** – a small set of analytic families
    (product‑power, Gaussian, Beta, arcsine) plus a few random *sparse*
    candidates (equal mass on a random subset of indices).  Sparse
    starts encourage very low convolution peaks.

2.  **ADAM on the soft‑max surrogate** – we minimise the smooth
    approximation τ·log ∑exp(conv/τ) using an ADAM optimiser on the
    *log‑it* (KL‑mirror) parametrisation of the simplex.  The gradient
    of the surrogate is obtained analytically via a second FFT
    (the same trick as the original entropic mirror‑descent).

3.  **Cheap annealing polish** – after the ADAM run a mass‑transfer
    simulated‑annealing step is performed, allowing zero entries to
    become non‑zero and fine‑tuning the best vector.

The routine respects the 1000 s wall‑time budget, works for any length
(up to a few‑tens of thousands) and returns a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring (exact objective for a normalised vector)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector with sum 1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Parametric families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.maximum(term, eps)
    a = term ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.maximum(term, eps)
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# ADAM optimiser on the soft‑max surrogate (entropy mirror‑descent)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    ADAM optimiser on τ·log ∑exp(conv/τ) with KL‑projection onto the simplex.
    Returns the best exact‑score vector seen.
    """
    # initialise logits (log‑it representation)
    a = a0.copy()
    # avoid exact zeros for the log
    eps = 1e-30
    logits = np.log(a + eps)

    # ADAM parameters
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    eta = 0.2                 # base step size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    # start with a relatively smooth temperature based on the first conv
    conv0 = conv_self(a)
    tau = max(conv0.max() * 0.5, 1e-4)   # keep it positive
    tau_end = tau * 1e-3
    decay = 0.9995

    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- compute convolution & soft‑max weights -----
        conv = conv_self(a)
        max_val = conv.max()
        shift = max_val
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. a -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)            # length = 3n‑2
        idx = (n - 1) - np.arange(n)         # pick needed segment
        grad = 2.0 * corr[idx]

        # ----- gradient w.r.t logits (KL‑mirror) -----
        mean_grad = np.dot(grad, a)          # Σ_i a_i * grad_i
        g_logits = a * (grad - mean_grad)    # ∇_{logits} S

        # ----- ADAM update (descent) -----
        m = beta1 * m + (1 - beta1) * g_logits
        v = beta2 * v + (1 - beta2) * (g_logits * g_logits)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        logits -= eta * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- re‑normalise to the simplex -----
        max_logit = np.max(logits)
        exp_logits = np.exp(logits - max_logit)
        a = exp_logits / exp_logits.sum()

        # ----- exact (non‑smoothed) score tracking -----
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ADAM improve n={n} it={it} score={cur_sc:.6f}")

        # ----- anneal temperature -----
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
