Implementation of simplex projection:

def proj_simplex(v):
    """Project onto probability simplex."""
    # v is 1-D array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # largest index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Consider numeric stability; ensure that if all values same e.g., v zeros, then rho = n, keep zeros.

"""
Low‑maximum self‑convolution search (re‑implemented).

The previous implementation used a soft‑max surrogate whose gradient was
computed incorrectly (the index extraction was reversed).  Here we:

* compute the exact sub‑gradient of the true max‑norm,
* use an exponentiated‑gradient (mirror‑descent) step with a
  step‑size that decays as 1/√t,
* keep a rich set of parametric warm‑starts,
* up‑sample the best vector when moving to larger problem sizes,
* finish each length with a cheap mass‑transfer simulated‑annealing
  polish.

All operations respect the non‑negativity and ∑a = 1 constraints required
by `evaluate_sequence`.  The algorithm dynamically splits the remaining
budget among a schedule of increasing lengths and logs progress.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (self‑convolution and cross‑convolution)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (exact objective for a normalized vector)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalized vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term <= 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term <= 0] = eps
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Gradient of the true max‑norm (sub‑gradient)
# ----------------------------------------------------------------------


def max_conv_and_gradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (max_val, grad) where max_val = max(a*a) and
    grad is a sub‑gradient of the max‑norm w.r.t. a.
    """
    n = a.shape[0]
    conv = conv_self(a)
    max_val = conv.max()

    # active set – indices whose value is within a tiny tolerance of the max
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(float)          # length 2n‑1

    # gradient = 2 * (mask * a_rev) convolved, sliced to length n
    a_rev = a[::-1]
    corr = conv_fft(mask, a_rev)                         # length 3n‑2
    grad = 2.0 * corr[n - 1 : n - 1 + n]                  # length n
    return max_val, grad


def exp_gradient_step(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """Exponentiated‑gradient update on the simplex."""
    expo = -step * grad
    expo = np.clip(expo, -700.0, 700.0)  # avoid overflow in exp
    a_new = a * np.exp(expo)
    s = a_new.sum()
    if s == 0.0 or np.isnan(s):
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing polish
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another. Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Core optimisation (exponentiated‑gradient on the true max‑norm)
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑norm using exponentiated updates.
    Returns the best vector encountered and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # step‑size schedule
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        max_val, grad = max_conv_and_gradient(a)
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] exp‑GD improve n={n} it={it} score={cur_sc:.6f}"
            )

        step = step0 / math.sqrt(it)
        a = exp_gradient_step(a, grad, step)

    return best_a, best_sc

            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # keep the current best for up‑sampling to the next length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the external evaluator will clip/normalise)
    return [float(x) for x in best_seq]
