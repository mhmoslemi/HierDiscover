def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety

    # Determine base length n0
    n0 = 4800
    try:
        starter = np.asarray(height_sequence_1)
        if starter.ndim == 1 and starter.size > 0:
            # normalize to sum=1
            starter = starter.astype(float)
            starter /= starter.sum()
    except Exception:
        starter = None

    if starter is not None:
        if starter.shape[0] != n0:
            a_cur = _upsample(starter, n0)
        else:
            a_cur = starter.copy()
    else:
        a_cur = rng.random(n0)
        a_cur /= a_cur.sum()

    best_seq = a_cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    # Stage parameters
    stages = [
        {'n': n0, 'iters_per_alpha': 12000, 'alpha_schedule':[0.5,2,5,10,20,40,80,160,320], 'polish_frac': 0.4, 'anneal_steps':500_000},
        # later larger n's maybe
    ]
    # you could add second stage if time left
    total_time_spent = time.time() - start_time
    for stage in stages:
        n = stage['n']
        # if current best_seq not same length, upsample
        if best_seq.shape[0] != n:
            a_cur = _upsample(best_seq, n)
        else:
            a_cur = best_seq.copy()

        # run Adam
        seq, sc = _run_adam(a_cur, n, deadline, start_time, stage)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"After Adam stage n={n}: best_score={best_score:.6f}")

        # polishing
        remaining = deadline - time.time()
        if remaining > 1.0:
            polish_deadline = time.time() + 0.5 * remaining
            a_polish, sc_polish = _run_gradient_descent(best_seq, n, polish_deadline,
                                                        start_time, beta=0.94, step0=0.25)
            if sc_polish < best_score:
                best_score = sc_polish
                best_seq = a_polish.copy()
                print(f"Polish improved: {best_score:.6f}")

        # annealing
        if time.time() < deadline:
            a_ann, sc_ann = _run_annealing(best_seq, n, deadline, start_time,
                                          best_score, best_seq,
                                          start_temp=5e-2, end_temp=1e-8,
                                          steps=int(300_000))
            if sc_ann < best_score:
                best_score = sc_ann
                best_seq = a_ann.copy()
                print(f"Anneal improved: {best_score:.6f}")

        # early exit if time almost out
        if time.time() >= deadline - 1.0:
            break

    # final output handling: ensure non-negative and normalized to sum 1
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    print(f"Final best score: {best_score:.6f}")
    return best_seq.tolist()
```

"""
Propose a step‑function with a very low evaluate_sequence() score.

The algorithm works in three phases and repeats them while time allows:

1. **Adam on a smooth log‑sum‑exp approximation of the max‐convolution**
   – the coefficients are kept on the probability simplex via a softmax
   parametrisation.  Alpha (smoothing) is increased gradually and the learning
   rate is adapted automatically (Adam).  The schedule is longer and reaches
   higher α values than the reference implementation.

2. **Exact sub‑gradient polishing** – a projected (Nesterov‑style) gradient
   descent on the true max‑convolution objective.  It receives a larger share
   of the remaining time and uses a step size that decays with √iteration.

3. **Simulated‑annealing fine‑tuning** – random mass‑transfer moves with a
   geometric cooling schedule.  The number of annealing steps is increased
   compared to the baseline.

If time is still left after the first optimisation at n=4800, the
best‑found shape is up‑sampled to a larger grid (n≈7200 and then n≈9600) and
the three phases are repeated.  All sequences are kept non‑negative and are
renormalised to sum to 1 before being returned (the objective is scale‑invariant
under the clipping performed by ``evaluate_sequence``).

The implementation is completely self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` function, and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max‑convolution objective
# ----------------------------------------------------------------------


def _run_adam(a0: np.ndarray, n: int, deadline: float, start_time: float,
              global_best_score: float, rng: np.random.Generator):
    """
    Adam optimisation of the smoothed objective.
    Returns (best_sequence, best_score) where the score is the exact
    evaluate_sequence value.
    """
    # initialise logits for the softmax parametrisation
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    # schedule of smoothing parameters (alpha) and iterations
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    iters_per_alpha = 15000          # more iterations than the reference
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_seq = a0.copy()
    best_score = global_best_score

    fft_len = _next_pow2(3 * n - 2)

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # ----- softmax -----
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # ----- smoothed objective -----
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # stable log‑sum‑exp
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # ----- gradient w.r.t a -----
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # ----- gradient w.r.t logits y (softmax chain rule) -----
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- occasional exact evaluation -----
            if i % 500 == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient projected gradient descent (polishing)
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7):
    """
    Projected gradient descent using the exact sub‑gradient of max(conv).
    Returns (best_vector, best_score) where the score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # active set (indices where conv is (almost) maximal)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of max(conv)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_onto_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-7,
                   steps: int = 300_000):
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------


def _optimise_stage(a_start: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    global_best_score: float,

    Returns (best_sequence, best_score).
    """

    # 1) Adam (smooth) optimisation
    # ------------------------------------------------------------------
    best_seq, best_score = _run_adam(a_start, n, deadline, start_time,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
