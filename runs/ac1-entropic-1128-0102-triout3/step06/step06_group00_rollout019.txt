"""
Propose a candidate step‑function (non‑negative heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm consists of three phases and respects the 1000 s time budget.

1. **Initialisation** – a deterministic “square‑root” shape
   a_i ∝ C_i = (2i choose i) / 4^i  (the coefficients of (1‑x)^-½).
   For an infinite series a*a is exactly constant, so the truncated version
   already has a very flat self‑convolution.

2. **Projected sub‑gradient descent** – an AdaGrad‑style descent on the true
   non‑smooth objective.  At each iteration we compute the convolution by FFT,
   find the index where it attains its maximum and move the probability vector
   in the opposite direction of the exact sub‑gradient of that entry.
   After the step we Euclidean‑project back onto the probability simplex.
   The run is limited only by the remaining wall‑clock time, therefore it
   performs several million cheap iterations.

3. **Simulated‑annealing polish** – a long geometric‑cooling annealing pass that
   transfers a tiny amount of mass between two randomly chosen positions.
   The temperature schedule is very slow (≈10⁶ trials) which lets the search
   escape shallow plateaus and often reduces the bound a little further.

The best vector seen during the whole run is returned as a plain Python list.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _initial_binomial(n: int) -> np.ndarray:
    """Deterministic start: C_i = (2i choose i) / 4^i  (i=0..n‑1)."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence C_{i} = C_{i-1} * (2i-1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()               # bring onto the probability simplex
    return a


# ----------------------------------------------------------------------
# Phase 1 – warm‑up with exponentiated‑gradient on a smooth max
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 8000):
    """Multiplicative updates based on a log‑sum‑exp smooth of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # smoothing parameter
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ---- smooth‑max gradient ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = rfft(a, n=fft_len)
        Wrev = rfft(w_rev, n=fft_len)
        corr = irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

        # gradient of smoothed objective
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # ---- multiplicative update ---------------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

        # increase α slowly
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – AdaGrad projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _run_subgradient_adagrad(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.3,
                             beta: float = 0.98,
                             eps: float = 1e-12):
    """
    Projected sub‑gradient descent with AdaGrad step‑size adaptation.
    Returns the best vector encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = a.shape[0]

    # AdaGrad accumulator (per‑coordinate)
    G = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of the *single* maximal entry
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]               # see derivation in the analysis

        # AdaGrad update (scalar step0 with per‑coordinate scaling)
        G += grad * grad
        adjusted_step = step0 / np.sqrt(G + eps)   # element‑wise step size
        a = a - adjusted_step * grad

        # Optional momentum (helps a little on noisy gradients)
        # a = a - step0 * grad / np.sqrt(G + eps)

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – long simulated‑annealing polish
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   steps: int = 1_000_000,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-6):
    """
    Tiny mass‑transfer moves with geometric cooling.
    The move is accepted iff it improves the objective or with a
    Metropolis probability exp(-Δ/ T).
    """
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling in log‑space
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # source biased by current mass
        dst = rng.integers(n)             # destination uniformly
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move at most 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # recompute the objective (full convolution – cheap for n≈5k)
        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

        best_score = score_warm
        best_seq = a_warm.copy()
        print(f"[{time.time() - start_time:.2f}s] stage 1 improve: "
              f"score={score_warm:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long AdaGrad projected sub‑gradient descent (≈85 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        stage2_deadline = time.time() + max(20.0, 0.84 * remaining)

        a_sub, score_sub = _run_subgradient_adagrad(
            best_seq,
            stage2_deadline,
            start_time,
            step0=0.30,
            beta=0.98,
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] stage 2 improve: "
                  f"score={score_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – extended annealing polish (use any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # allocate the rest of the budget; 1 M steps is usually well within the
        # remaining time (each step costs only an FFT of size ≈ 16384)
        remaining = deadline - time.time()
        steps = int(min(2_000_000, remaining * 2000))   # roughly 0.5 ms per step
        a_final, score_final = _run_annealing(
            best_seq,
            deadline,
            start_time,
            best_score,
            best_seq,
            steps=steps,
            start_temp=4e-2,
            end_temp=1e-6,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={score_final:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (protect against numerical drift) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    # sanity check – optional (can be commented out in production)
    # print("final bound =", evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
