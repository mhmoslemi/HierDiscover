def propose_candidate(seed=0, budget_s=1000):
   np.random.seed(seed)
   start_time = time.time()
   deadline = start_time + budget_s - 0.5

   # use known best if available
   try:
        a0 = np.array(height_sequence_1, dtype=float)
        if a0.ndim == 1 and a0.size>0:
            a0 = np.maximum(a0,0)
            if a0.sum()>0:
               a0 = a0 / a0.sum()
            else:
               a0 = None
   except:
        a0 = None

   # define length
   n = 4096
   if a0 is None or a0.shape[0] != n:
        a0 = np.random.rand(n)
        a0 /= a0.sum()
   # optional: use multiple restarts of random Dirichlet for half the time
   best_seq = a0.copy()
   best_score = _objective(best_seq)

   # we will run a loop doing three types of moves until time runs out
   while time.time() < deadline:
        remaining = deadline - time.time()
        # decide move type based on random
        move_type = np.random.choice(['adam','sg','swap','anneal'])
        # each move takes limited time slice
        slice_deadline = time.time() + min(remaining*0.2, 30.0)
        if move_type == 'adam':
            # run some steps of adam
            a_opt, sc = _run_adam_opt(best_seq, n, slice_deadline, start_time, symmetrize=False)
        elif move_type == 'sg':
            a_opt, sc = _subgradient_descent(best_seq, n, slice_deadline, start_time)
        elif move_type =='swap':
            a_opt = _run_lp_improve(best_seq, n, step=0.2, max_iters=3, deadline=slice_deadline, start_time=start_time)
            sc = _objective(a_opt)
        else:
            a_opt, sc = _run_annealing(best_seq, n, slice_deadline, start_time, best_score, best_seq,
                                       start_temp=0.05, end_temp=1e-6, steps=200000)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time()-start_time:6.2f}s] new global best {best_score:.6f}")

   return best_seq.tolist()

"""
Improved search for a step‑function with a small self‑convolution peak.

The algorithm works in three stages:

1. **Quick parametric sweep** – a few families of analytic shapes
   (uniform, power‑law, exponential, sine) are generated, normalised
   and evaluated.  The best of them is used as a warm start.

2. **Projected Adam** on the smooth log‑sum‑exp surrogate of the peak.
   The Adam loop runs as long as the overall time budget permits
   (≈ 60‒70 % of the remaining time).  A very small λ‑factor is used
   and decayed slowly so that the optimiser focusses on the true max.

3. **Polishing** – a short sub‑gradient descent followed by a
   mass‑transfer simulated‑annealing phase.  Both keep the total mass
   fixed (the sequence stays on the probability simplex).

All intermediate best sequences are remembered; the final return
is the best one found within the 1000 s budget.  The code uses only
numpy and the helper functions already supplied in the environment.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (simple, no external dependencies)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real)."""
    n = a.shape[0] + a.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(A * A, n=nfft)[:n]
    return c


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation used by the competition (sum(a) = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth log‑sum‑exp surrogate of the peak
# ----------------------------------------------------------------------


def _grad_logsumexp(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ).
    The derivation uses the convolution theorem.
    """
    n = a.shape[0]
    # soft‑max weights
    w = np.exp((conv - conv.max()) / lam)
    w /= w.sum()
    # convolve w with reversed a
    a_rev = a[::-1]
    tmp = np.fft.rfft(w, n=2 * n - 1)
    rev = np.fft.rfft(a_rev, n=2 * n - 1)
    g = np.fft.irfft(tmp * rev, n=2 * n - 1)
    # extract the central n entries (indices n‑1 … 2n‑2)
    grad = 2.0 * g[n - 1 : n - 1 + n]
    return grad


def _run_adam(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp surrogate. Returns best seen."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # start with a relatively aggressive λ‑factor and decay slowly
    lam_factor = 0.5
    lam_decay = 0.98
    decay_every = 2000
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve it={it} score={cur_score:.6f}")

        if it % decay_every == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _grad_logsumexp(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (simple 1/√t decay)
        step = 0.6 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _run_subgrad(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # indices where the max is attained
        max_idx = np.flatnonzero(conv == M)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for k in max_idx:
            # all i with j = k-i in range
            i_start = max(0, k - (a.shape[0] - 1))
            i_end = min(a.shape[0] - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]          # contribution from a_i * a_j
            grad[j_range] += a[i_range]          # symmetric term

        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve it={it} score={cur:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------


def _run_anneal(a0: np.ndarray, deadline: float, start_time: float,
                cur_best_score: float, cur_best_seq: np.ndarray,
                start_T: float = 5e-2, end_T: float = 1e-6,
                steps: int = 150000) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a small amount of mass between two indices."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    n = a.shape[0]
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        T = start_T * (end_T / start_T) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper to generate a few analytic initial shapes
# ----------------------------------------------------------------------


def _candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a list of plausible starting vectors (all normalised)."""
    xs = np.arange(n, dtype=float)

    # uniform
    cand = [np.full(n, 1.0 / n)]

    # power‑law a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.0, 2.0, 13):
        w = 1.0 / np.power(xs + 1.0, p)
        w /= w.sum()
        cand.append(w)

    # exponential decay a_i ∝ exp(-λ·i)
    for lam in np.linspace(0.0, 5.0 / n, 9):
        w = np.exp(-lam * xs)
        w /= w.sum()
        cand.append(w)

    # single sine bump (positive half‑wave)
    s = np.sin(np.pi * (xs + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # triangular (linearly decreasing then increasing)
    tri = np.concatenate([np.linspace(1, 0, n // 2, endpoint=False),
                          np.linspace(0, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    cand.append(tri)

    return cand

    for idx, a in enumerate(init_pool):

        if sc < best_score:

            best_seq = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] init {idx} score={sc:.6f}")

    # ------------------------------------------------------------------
