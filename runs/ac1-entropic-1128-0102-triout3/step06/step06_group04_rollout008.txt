def _conv_fft(a, b):
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c

"""
Hybrid exponential‑gradient + LP + annealing optimizer.

* Works directly on the simplex (∑a_i = 1, a_i≥0) via the
  multiplicative (exponentiated) gradient descent on the exact
  max‑convolution objective.
* Each iteration obtains the exact sub‑gradient of the max‑norm
  through a fast FFT convolution of the mask of maximal indices.
* A short linear‑programming “swap” phase (classic LP attack) is
  invoked when time permits.
* Final polishing with a light simulated‑annealing mass‑transfer.
* The routine respects a 1000 s budget and returns the best
  sequence found.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Objective (exact evaluation, sum(a)=1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a) (assumes ∑a_i = 1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – keeps the simplex property
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exponential‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                               n: int,
                               slice_deadline: float,
                               start_time: float,
                               symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent directly on
    max(conv) over the simplex.  Returns the best sequence seen.
    """
    a = np.maximum(a0, 1e-30)                # avoid zeros for log
    a = a / a.sum()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.2                               # initial step size
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # mask of positions achieving the maximum (with tolerance)
        mask = np.isclose(conv, M, atol=1e-12).astype(np.float64)

        # gradient = 2 * (mask * a_rev)  (central slice of convolution)
        a_rev = a[::-1]
        grad = 2.0 * _conv_fft(mask, a_rev)[n - 1:n - 1 + n]

        eta = eta0 / math.sqrt(it)

        # multiplicative update (log‑space to keep positivity)
        grad_clip = np.clip(grad, -50.0, 50.0)          # guard against overflow
        log_a = np.log(a) - eta * grad_clip
        a = np.exp(log_a)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap” step (the classic LP attack)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A_{k,i} = f_i   if   i ≤ k < i + n,
                = 0      otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 2,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """
    Classic LP‑based improvement: maximise total mass while keeping the
    convolution bounded by the current peak.
    """
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise –sum(b) → maximise sum(b)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = np.maximum(a_cur, 0.0)
        a_cur = a_cur / a_cur.sum()

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
