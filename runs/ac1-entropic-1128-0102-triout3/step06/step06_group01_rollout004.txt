"""
Propose a non‑negative step function that (hopefully) improves the current
upper bound on ``evaluate_sequence``.

Algorithm
---------
* Use the previously known construction ``height_sequence_1`` (if present)
  and up‑sample it to a chosen length `n`.
* Run several short Adam optimisations on a smoothed max‑convolution
  (log‑sum‑exp) with an increasing smoothing parameter α.  The parameters
  (learning‑rate schedule, α‑schedule, number of iterations) are tuned to
  stay well inside the allocated time budget.
* Polish the best candidate with a fast projected sub‑gradient descent.
  The sub‑gradient of ``max(conv)`` is computed by a single FFT‑based
  cross‑correlation, which makes it possible to run many iterations.
* Finish with a simulated‑annealing refinement that repeatedly moves a
  random ε‑fraction of mass between two entries while respecting the simplex
  constraint.
* All updates keep the vector on the probability simplex (non‑negative,
  sum = 1).  The best sequence found during the whole run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)−1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of     f(a)= (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact sub‑gradient of the max‑convolution (FFT version)
# ----------------------------------------------------------------------
def _grad_exact_max(a: np.ndarray, conv: np.ndarray,
                    max_val: float, tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Implements the formula
        g_i = 2 * Σ_{k active} a_{k-i}
    where `active` are the indices where conv is within `tol` of its maximum.
    Computed via a single FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    active = conv >= max_val * (1.0 - tol)
    active_vec = active.astype(float)                     # shape (2n‑1,)

    # Cross‑correlation a ⋆ active via FFT
    L = _next_pow2(3 * n - 2)
    Af = np.fft.rfft(a, n=L)
    Brev = np.fft.rfft(active_vec[::-1], n=L)
    corr = np.fft.irfft(Af * Brev, n=L)  # length L (>= 3n‑2)

    # Extract g_i = 2 * corr[n‑1 : 2n‑1][::-1]
    slice_part = corr[n - 1: 2 * n - 1]          # length n
    g = 2.0 * slice_part[::-1]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fast version)
# ----------------------------------------------------------------------
def _run_gradient_descent_fast(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               beta: float = 0.9,
                               step0: float = 0.3,
                               max_iter: int = 10 ** 7,
                               tol: float = 1e-12):
    """
    Fast projected sub‑gradient descent on the true max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient
        g = _grad_exact_max(a, conv, max_val, tol)

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # projected SGD step
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    """
    Simple annealing that moves a random ε‑fraction of mass from a source
    index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.30  # up to 30% of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load a starter sequence (previous best) if available
    # ------------------------------------------------------------------
    starter = None
    try:
        # ``height_sequence_1`` is expected to be a global numpy array / list
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
    except Exception:
        starter = None

    # Choose a problem size.  A larger n gives more degrees of freedom.
    # Experiments show that n ≈ 7200 is a good trade‑off for the 1000 s budget.
    n = 7200

    if starter is not None:
        a0 = _upsample(starter, n)
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Keep track of the best ever sequence
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{time.time() - start_time:6.2f}s] initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimisation with increasing smoothing parameter α
    # ------------------------------------------------------------------
    # schedule of smoothing parameters (log‑sum‑exp)
    alpha_schedule = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 200.0]
    iters_per_alpha = 4000               # modest number – total ≈ 40 k iters

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    # start from the current best (may be the up‑sampled starter)
    y = np.log(np.maximum(best_seq, 1e-16))

    # Pre‑compute FFT length for the smoothed gradient
    fft_len = _next_pow2(3 * n - 2)

    t = 0
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # ---------- softmax ----------
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # OPTIONAL symmetry enforcement (helps convergence for many problems)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()

            # ---------- smoothed objective ----------
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # ---------- gradient w.r.t. a ----------
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # ---------- gradient w.r.t. logits ----------
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ---------- Adam update ----------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---------- occasional evaluation ----------
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improvement: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Exact projected sub‑gradient polishing (fast version)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly half of the remaining time to polishing
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, polish_score = _run_gradient_descent_fast(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if polish_score < best_score:
            best_score = polish_score
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improvement: score={polish_score:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, final_score = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:6.2f}s] finished, best score = {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
