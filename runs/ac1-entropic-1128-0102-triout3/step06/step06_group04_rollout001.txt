"""
Propose a low‑peak self‑convolution step function.

Algorithm outline
-----------------
1. Choose a target length (default 4096, smaller if time is tight).
2. Build a diverse set of initial shapes:
   • uniform, sinusoid, triangle, Hann, Hamming, Blackman,
   • exponential and power‑law profiles with random parameters,
   • random Dirichlet vectors, and the optional warm‑start `height_sequence_1`.
3. For each start run two complementary optimisers on the smooth‑max surrogate
   Lλ(a)=λ·log∑exp(conv/λ) (λ≈max(conv)·c):
   * **Mirror descent** on the probability simplex (exponentiated gradient).
   * **Projected Adam** (as in the previous baseline).
   The best intermediate vector is kept.
4. Refine the best vector with a short LP “swap” step, a sub‑gradient descent
   on the true max‑norm and a mass‑transfer simulated‑annealing polishing.
5. A final Adam pass with symmetry is performed if time permits.
   The returned list is a Python list of non‑negative numbers (the evaluator
   will renormalise automatically).

All heavy work respects the global time budget (`budget_s`).  The code
contains no closures, uses only top‑level functions and works on a single
CPU.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum(a)≈1).  Returns 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Gradient of smooth‑max surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of Lλ(a)=λ·log Σ exp(conv/λ).

    Returns ∇_a Lλ.
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable softmax weights
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # gradient is 2 * conv(w, reversed a) restricted to the central slice
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)                 # length = 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]       # central n entries
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the simplex
# ----------------------------------------------------------------------


def _run_mirror_opt(a0: np.ndarray,
                    n: int,
                    slice_deadline: float,
                    start_time: float,
                    lam_factor: float = 0.4,
                    eta0: float = 1.0,
                    symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent) on the smooth‑max surrogate.
    Returns best encountered (vector on simplex, objective value).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror improve it={it} score={cur_score:.6f}")

        # smoothing parameter
        lam = max(conv.max() * lam_factor, 1e-12)

        # gradient of surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # step size
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (avoid overflow)
        # shift grad to have zero max before exponentiation
        shift = grad.max()
        factor = np.exp(-eta * (grad - shift))
        a = a * factor
        a = _project_simplex(a)

        if symmetrize:
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam optimiser (same as the baseline)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth‐max surrogate; returns best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming swap step (the classic LP attack)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A_{k,i} = f_i   if   i ≤ k < i + n,
                = 0      otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 3,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """Repeated LP‑based improvement of a."""
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve step={step:.2f} score={sc:.6f}")

    return a_cur

# Sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve it={it} score={cur_score:.6f}")

    return best_a, best_score

# Cheap annealing (mass transfer) for final polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve it={it} score={score:.6f}")

    return best_seq, best_score

def _uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _sinusoid_shape(n: int) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    return s / s.sum()


def _triangular_shape(n: int) -> np.ndarray:
    half = n // 2
    first = np.linspace(1, half, half, endpoint=False)
    second = np.linspace(half, 1, n - half)
    tri = np.concatenate([first, second])
    return tri / tri.sum()


def _hann_shape(n: int) -> np.ndarray:
    i = np.arange(n)
    h = np.sin(np.pi * i / (n - 1)) ** 2
    return h / h.sum()


def _hamming_shape(n: int) -> np.ndarray:
    i = np.arange(n)
    h = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return h / h.sum()


def _blackman_shape(n: int) -> np.ndarray:
    i = np.arange(n)
    h = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4 * np.pi * i / (n - 1)))
    return h / h.sum()


def _exp_shape(n: int, alpha: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    a = np.exp(-alpha * np.abs(np.arange(n) - centre))
    return a / a.sum()


def _power_shape(n: int, p: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.abs(np.arange(n) - centre)) ** p
    return a / a.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    a = np.random.rand(n)
    return a / a.sum()


def _make_initial_variants(base: np.ndarray, n: int) -> list[np.ndarray]:
    """Generate a collection of plausible starts from a base vector."""
    variants = [base.copy()]

    # deterministic windows
    variants.append(_sinusoid_shape(n))
    variants.append(_triangular_shape(n))
    variants.append(_hann_shape(n))
    variants.append(_hamming_shape(n))
    variants.append(_blackman_shape(n))

    # random parametrised families
    for _ in range(3):
        alpha = 10 ** np.random.uniform(-4, -1)   # 1e-4 … 1e-1
        variants.append(_exp_shape(n, alpha))

    for _ in range(3):
        p = np.random.uniform(0.5, 3.0)
        variants.append(_power_shape(n, p))

    # random Dirichlet
    for _ in range(2):
        variants.append(_random_dirichlet(n))

    # optionally a uniform fallback
    variants.append(_uniform_shape(n))

    # force symmetry on all variants (helps the objective)
    for i in range(len(variants)):
        v = variants[i]
        variants[i] = (v + v[::-1]) * 0.5
        variants[i] = _project_simplex(variants[i])

    return variants

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
