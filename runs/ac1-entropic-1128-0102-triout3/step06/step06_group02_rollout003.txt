def subgradient_max_conv(a):
    conv = _conv_self(a)
    max_idx = conv.argmax()
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idx < n:
        grad[:max_idx+1] = 2.0 * a[max_idx::-1]
    else:
        start = max_idx - n + 1
        grad[start:] = 2.0 * a[start:][::-1]
    max_val = conv[max_idx]
    return grad, max_val
```

def _lp_step(a, n, M):
    # Solve LP maximize sum(b) subject to a*b <= M, b>=0.
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    # Build A_ub shape (2n-1, n) sparse.
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # each j such that i = k - j in [0, n-1]
        # j in [max(0, k - (n-1)), min(n-1, k)]
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # i = k - j, a_i
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    # Convert to CSR matrix
    import scipy.sparse as sp
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    c = -np.ones(n)  # minimize -sum(b)
    # Solve with highss
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 60})
    if res.success:
        b = res.x
        # Clip negative small values
        b = np.maximum(b, 0)
        # Renormalize to sum 1 if possible
        total = b.sum()
        if total > 0:
            b = b / total
        else:
            # fallback to uniform distribution
            b = np.full_like(b, 1.0 / n)
        return b
    else:
        # if LP fails, return a unchanged
        return a.copy()

"""
Hybrid search for the step‑function that minimises ``evaluate_sequence``.

The algorithm
---------------
1. **Warm‑start** – use a previous construction if available,
   otherwise start from a random simplex point.
2. **Progressive up‑sampling** – work with a cascade of lengths
   (800 → 1600 → 3200 → 6400 → 9600 → 12800).  At each length the current
   best vector is interpolated and renormalised.
3. **LP‑direction (small sizes only)** – for n ≤ 2000 we solve a tiny
   linear program that maximises the total mass of a vector *b* under the
   current max‑convolution bound.  The resulting *b* is combined with the
   current *a* (simple line search) to obtain a better starting point.
4. **Two gradient phases**
   * **Exponentiated (multiplicative) gradient** – a KL‑type update
     ``a_i ← a_i·exp(-η·g_i)`` where *g* is the sub‑gradient of the
     max‑convolution.  The iterate is projected onto the probability
     simplex.
   * **Adam‑style sub‑gradient descent** – a classic adaptive step‑size
     method applied to the same sub‑gradient.
   The best iterate of both phases is kept.
5. **Long simulated‑annealing walk** – a Metropolis move that transfers a
   small fraction of mass between two random positions, with a slow
   cooling schedule.  The walk uses the exact max‑convolution, so it can
   escape shallow minima.
6. **Return** the best simplex vector found (list of floats).

All phases respect the global 1000 s budget; the time is split
approximately 15 % for the LP stage, 70 % for gradient descent and
15 % for annealing.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray):
    """Return (grad, max_val) where grad is a sub‑gradient of max(a*a)."""
    conv = _conv_self(a)
    k = int(conv.argmax())
    n = a.shape[0]
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k :: -1]
    else:
        start = k - n + 1
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, conv[k]


# ----------------------------------------------------------------------
# Exponentiated (multiplicative) gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_expgrad(a0: np.ndarray, n: int, deadline: float, start_time: float,
                 eta0: float = 0.8, decay: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _subgradient_max_conv(a)
        eta = eta0 * (decay ** it)
        # log‑space update to avoid underflow
        loga = np.log(a + 1e-15)
        loga -= eta * grad
        a = np.exp(loga)
        a = _project_simplex(a)
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n} it={it} score={cur_score:.6f}")
    return best_a, best_score

# Adam‑style sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, start_time: float,
              lr0: float = 0.12, beta1: float = 0.9,
              beta2: float = 0.999, eps: float = 1e-8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _subgradient_max_conv(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / math.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(np.maximum(a, 0.0))
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n} it={t} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming direction (only for small n)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, n: int, M: float) -> np.ndarray:
    """
    Solve   max 1ᵀb   s.t. a * b ≤ M (componentwise), b ≥ 0.
    Return the feasible vector b normalised to sum = 1.
    If SciPy is unavailable or the LP fails, return a copy of a.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:          # pragma: no cover – SciPy may be missing
        return a.copy()

    # Build sparse matrix A_ub where (A_ub)_{k,j} = a_{k‑j} if 0≤k‑j<n, else 0
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    c = -np.ones(n)                     # minimise -∑b  ⇔ maximise ∑b
    bounds = [(0, None)] * n
    # Restrict the solver to a short time limit (≈30 s) to stay safe.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 30})
    if not res.success:
        return a.copy()
    b = np.maximum(res.x, 0.0)
    total = b.sum()
    if total > 0:
        b = b / total
    else:
        b = np.full_like(b, 1.0 / n)
    return b


def _best_combo(a: np.ndarray, b: np.ndarray, n: int) -> np.ndarray:
    """
    Simple line‑search over a few mixing coefficients t ∈ [0,1]
    and return the convex combination (1‑t)a + t b that yields the
    smallest value of 2·n·max((a+t(b‑a))*(a+t(b‑a))).
    """
    best = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    for t in (0.0, 0.05, 0.1, 0.2, 0.4, 0.8, 1.0):
        cand = (1 - t) * a + t * b
        cand = _project_simplex(cand)
        cur = 2.0 * n * _conv_self(cand).max()
        if cur < best_score:
            best_score = cur
            best = cand.copy()
    return best


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (exact max‑convolution)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int, deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 0.3, end_temp: float = 1e-8,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 20 % of the mass at i
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n} it={it} score={score:.6f}")

    return best_seq, best_score

        if n <= 2000 and time.time() < stage_deadline:
            M = _conv_self(a0).max()
            b_lp = _lp_step(a0, n, M)
            a0 = _best_combo(a0, b_lp, n)

        # ----------------------------------------------------------------
        # Phase 1 – exponentiated gradient (half of the stage time)
        # ----------------------------------------------------------------
        half_deadline = time.time() + stage_time * 0.5
        a_opt, score_opt = _run_expgrad(a0, n, half_deadline, start_time,
                                        eta0=0.8, decay=0.999)

        # ----------------------------------------------------------------
        # Phase 2 – Adam sub‑gradient (remaining time of the stage)
        # ----------------------------------------------------------------
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _run_adam(a_opt, n, stage_deadline, start_time,
                                           lr0=0.12)
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # Keep the best vector seen so far
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # Update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – spend the rest of the budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        remaining = deadline - time.time()
        # scale the number of annealing steps to the remaining time (≈ 5 µs per step)
        steps = int(min(3_000_000, max(200_000, remaining * 200_000)))
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-8,
            steps=steps
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(2000, 1.0 / 2000)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
