"""
Hybrid optimizer for the self‑convolution L∞ problem.

The search proceeds in three phases for a single large length (the
largest power‑of‑two that comfortably fits in the time budget):

1. **Parametric warm‑start** – a rapid random sweep over several analytic
   families (product‑power, Gaussian, Beta, arcsine) yields a good
   initial shape.

2. **Entropic mirror‑descent** – we minimise the smooth surrogate
   τ·log ∑exp(conv/τ) on the probability simplex with exponentiated‑gradient
   updates.  The temperature τ is annealed from a large value (≈max(conv))
   down to 10⁻⁶·max(conv) while the step‑size follows η/√t.  The gradient
   of the surrogate is obtained analytically via one FFT‑based cross‑correlation.

3. **Mass‑transfer simulated annealing** – a cheap but effective move set
   (transfer a random ε from one index to another) is applied for a long
   period, accepting worsening moves with a geometric temperature schedule.
   This polish often pushes the objective a few percent lower.

The algorithm is fully deterministic given the seed, respects the
non‑negativity and sum‑to‑one constraints required by `evaluate_sequence`,
and stops cleanly before the supplied time budget expires.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring – exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Return 2·n·max(conv_self(a)) for a normalised vector."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Parametric families (quick warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Entropic mirror‑descent on the simplex
# ----------------------------------------------------------------------


def entropic_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    log_prefix: str = "",
) -> tuple[np.ndarray, float]:
    """
    Minimise the max self‑convolution via a smoothed surrogate.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = score_of(a)

    # Initialise temperature to a few times the current max convolution.
    cur_conv = conv_self(a)
    cur_max = cur_conv.max()
    tau = max(cur_max * 5.0, 1e-8)          # start relatively high
    tau_end = max(cur_max * 1e-4, 1e-8)    # final temperature (≈0)
    decay = 0.995

    eta0 = 0.6  # base step‑size for exponentiated‑gradient

    it = 0
    while time.time() < deadline:
        it += 1

        # --- evaluation / best‑track ---------------------------------
        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] {log_prefix} improve it={it} score={cur_sc:.6f}"
            )

        # --- soft‑max weights ----------------------------------------
        # stable soft‑max using the current maximum as shift
        shift = max_val
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed surrogate ----------------------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)       # extract required slice
        grad = 2.0 * corr[idx]

        # --- exponentiated‑gradient update ---------------------------
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        # Prevent overflow/underflow in the exponent
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)

        # --- projection onto the simplex -----------------------------
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # --- temperature annealing -----------------------------------
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def anneal_polish(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    cur_best_sc: float,
    cur_best_seq: np.ndarray,
    log_prefix: str = "",
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random ε from one index to another.
    Accept moves that improve the exact objective or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    sc = score_of(a)
    best_seq = cur_best_seq.copy()
    best_sc = cur_best_sc

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, a.size, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_sc = score_of(a_prop)
        delta = new_sc - sc

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            sc = new_sc

        if sc < best_sc:
            best_sc = sc
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] {log_prefix} anneal improve it={it} score={sc:.6f}"
            )

    return best_seq, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The function returns a plain Python list of heights (sum is normalised to 1).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a single working length – a power of two that allows fast FFT.
    # Larger lengths give finer discretisation of the optimal continuous shape.
    # ------------------------------------------------------------------
    # We pick 8192 unless the time budget is extremely tight.
    # If the remaining time after warm‑up is < 2 min we fall back to 4096.
    length = 8192
    if budget_s < 120:
        length = 4096

    n = length

    # ------------------------------------------------------------------
    # 1️⃣ Parametric warm‑start: random sweep over analytic families.
    # ------------------------------------------------------------------
    # Guarantee at least a minimal time for the warm‑up.
    warmup_deadline = min(deadline, start_time + 10.0)  # 10 s max
    best_a = np.full(n, 1.0 / n)
    best_sc = score_of(best_a)

    while time.time() < warmup_deadline:
        kind = np.random.choice([0, 1, 2, 3])
        if kind == 0:  # product‑power
            p = np.random.uniform(-2.0, 3.0)
            cand = shape_product_power(n, p)
        elif kind == 1:  # Gaussian
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            cand = shape_gaussian(n, sigma)
        elif kind == 2:  # Beta
            alpha = np.random.exponential(1.0) + 0.05
            beta_par = np.random.exponential(1.0) + 0.05
            cand = shape_beta(n, alpha, beta_par)
        else:  # arcsine
            cand = shape_arcsine(n)

        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            print(
                f"[{time.time() - start_time:6.2f}s] warm‑up improve kind={kind} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣ Entropic mirror‑descent optimisation (smooth surrogate).
    # ------------------------------------------------------------------
    # Allocate the bulk of the remaining budget to this phase.
    remaining = deadline - time.time()
    if remaining > 5.0:
        grad_deadline = time.time() + remaining * 0.70
        best_a, best_sc = entropic_descent(
            best_a,
            grad_deadline,
            start_time,
            log_prefix=f"n={n}",
        )
        print(
            f"[{time.time() - start_time:6.2f}s] after entropic n={n} best_score={best_sc:.6f}"
        )
    else:
        print(
            f"[{time.time() - start_time:6.2f}s] Skipping entropic phase (time too low)."
        )

    # ------------------------------------------------------------------
    # 3️⃣ Mass‑transfer simulated annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_a, best_sc = anneal_polish(
            best_a,
            anneal_deadline,
            start_time,
            cur_best_sc=best_sc,
            cur_best_seq=best_a,
            log_prefix=f"n={n}",
            start_temp=5e-2,
            end_temp=1e-5,
            steps=200_000,
        )
        print(
            f"[{time.time() - start_time:6.2f}s] final polished score={best_sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Return the sequence as a plain Python list (evaluate_sequence
    # will clip/normalise it if necessary).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)   # ensure non‑negativity
    s = best_a.sum()
    if s == 0 or np.isnan(s):
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    return [float(v) for v in best_a]
