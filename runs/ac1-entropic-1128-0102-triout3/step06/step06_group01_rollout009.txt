"""
Propose a step‑function (non‑negative list) that minimizes the given
evaluate_sequence() objective.

Algorithm
---------
* Multiple random restarts (default three) – each run works on a fixed
  length `n` (default 4800) but the length may be changed later.
* Smooth Adam optimisation on the simplex:
    – a = softmax(y) enforces a_i ≥ 0 and Σa_i = 1.
    – a smoothed max‑convolution objective is built with a log‑sum‑exp
      (parameter α).  α is increased gradually (0.5 → 320) and the
      Adam schedule is run for many iterations.
* Exact sub‑gradient polishing:
    – The true max‑convolution is minimised by projected gradient descent
      with Nesterov‑style momentum, using an FFT‑based sub‑gradient
      (O(n log n) per iteration).
* Simulated‑annealing fine‑tuning:
    – mass‑transfer moves (ε‑fraction of a source entry to a random
      destination) are accepted with a standard Metropolis criterion.
    – The temperature decays geometrically from 5·10⁻² to 10⁻⁶.
* Time‑management: the whole routine respects the supplied time budget
  (≈1000 s).  The best sequence found by any run is returned.

The combination of a long smooth Adam phase, aggressive exact polishing,
and a large annealing run often pushes the bound below the target
≈1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT / convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) via FFT.
    a is assumed to be a probability vector (Σa = 1).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of        f(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return grad


def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray,
                      fft_len: int, rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Uses the same FFT‑based correlation trick as _grad_smooth_max.
    """
    max_val = conv.max()
    # indices where conv is (almost) maximal
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    if active.size == 0:                     # should never happen
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv)
        w[active] = 1.0

    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + len(w_rev) - 1]
    g = 2.0 * corr[len(w_rev) - 1 - np.arange(a.shape[0])]
    return g


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Upsampling / interpolation of a probability vector
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n`, preserving the sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact sub‑gradient projected gradient descent (polishing)
# ----------------------------------------------------------------------


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7):
    """
    Projected gradient descent on the true max‑convolution.
    Returns the best vector found and its score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        g = _subgrad_max_conv(a, conv, fft_len)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _proj_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000,
                   max_eps_frac: float = 0.5):
    """
    Small‑scale simulated annealing that transfers a random ε‑fraction
    of mass from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    # pre‑compute FFT length for cheap convolution during moves
    fft_len = _next_pow2(2 * n - 1)

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * rng.random() * max_eps_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main optimisation routine (one seed / one length)
# ----------------------------------------------------------------------


def _optimise_one(seed: int,
                  n: int,
                  deadline: float,
                  start_time: float,
                  starter: np.ndarray | None) -> tuple[list[float], float]:
    """
    Optimises a single candidate (given a seed and length `n`).
    Returns (best_sequence_as_list, best_score).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # initialise a0 ----------------------------------------------------
    if starter is not None:
        if starter.shape[0] != n:
            a0 = _upsample(starter, n)
        else:
            a0 = starter.copy()
    else:
        a0 = rng.random(n)
        a0 /= a0.sum()

    # ---------------------------------------------------------------
    # 1) Adam optimisation on the smoothed max‑convolution
    # ---------------------------------------------------------------
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    fft_len = _next_pow2(3 * n - 2)

    # α schedule – we go up to 320 for a very sharp approximation
    alpha_schedule = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    # iteration budget per α – generous but will be truncated by the deadline
    iters_per_alpha = 12_000

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # softmax to get a
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # gradient wrt a (smooth)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # chain rule to logits
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (math.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # ---------------------------------------------------------------
    # 2) Exact sub‑gradient polishing (short but intensive)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        # give half of remaining budget to polishing
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ---------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
            max_eps_frac=0.5,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # final projection onto simplex (just in case of numerical drift)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final official evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        print(f"[{time.time() - start_time:6.2f}s] final evaluate improvement: score={final_score:.6f}")

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs several independent optimisation runs (different seeds)
    and returns the best sequence found within the supplied time budget.
    """
    start_time = time.time()
    # keep a small safety margin for the host environment
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Load a starter from the global variable if it exists
    # ------------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Decide on a set of seeds / lengths to explore
    # ------------------------------------------------------------------
    # we will try three different seeds; the length is kept at 4800
    # (the reference size) – for larger lengths the overhead grows
    # and the benefit is modest.
    seeds = [seed, seed + 17, seed + 31]

    best_overall_seq = None
    best_overall_score = float('inf')

    for s in seeds:
        if time.time() >= deadline:
            break

        # each run receives a fraction of the remaining time
        remaining = deadline - time.time()
        # guard against allocating too little time
        run_deadline = time.time() + max(30.0, 0.3 * remaining)

        seq, sc = _optimise_one(s, n=4800, deadline=run_deadline,
                                start_time=start_time,
                                starter=starter)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_seq = seq
            print(f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST (seed={s}) score={sc:.6f}")

    # Fallback – if nothing succeeded (should never happen), return a uniform
    if best_overall_seq is None:
        best_overall_seq = [1.0 / 4800] * 4800

    return best_overall_seq
