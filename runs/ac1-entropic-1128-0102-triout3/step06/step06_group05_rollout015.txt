def propose_candidate(seed=0, budget_s=1000):
    set random seed.
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None; best_score = inf.

    length_schedule = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]  # maybe up to 65536
    random shuffle for early stopping if needed.

    prev_a = None

    for n in length_schedule:
        if time.time()>=deadline: break

        if prev_a is not None:
            a0 = upsample(prev_a, n)
        else:
            a0 = np.full(n, 1.0/n)  # uniform as fallback

        # generate initial candidates:
        init_vecs = []
        init_vecs.append(a0.copy())
        # random Dirichlet
        init_vecs.append(np.random.dirichlet(np.ones(n))*1.0)
        # linear decreasing: weight ~ (n-i)
        lin = np.arange(n, 0, -1, dtype=float)
        lin /= lin.sum()
        init_vecs.append(lin)
        # exponential decreasing: r=0.99
        r=0.9995
        exp = r**np.arange(n); exp/= exp.sum(); init_vecs.append(exp)
        # maybe random Gaussian shaped
        x = np.linspace(-1,1,n)
        gauss = np.exp(- (x**2)/0.2)
        gauss/=gauss.sum()
        init_vecs.append(gauss)

        # Run optimizer for each candidate limited time slice:
        remaining = deadline - time.time()
        slice_time = min(remaining * 0.2, 30.0) # allocate up to 30s per length piece

        slice_deadline = time.time() + slice_time

        for init in init_vecs:
            if time.time()>=slice_deadline: break
            a_opt, score_opt = run_adam_opt_no_sym(init, n, slice_deadline, start_time)
            if score_opt < best_score:
                best_score= score_opt
                best_seq = a_opt.copy()
                print(...)

            # refinement via exponentiated gradient descent
            # allocate some time for refinement
            refine_deadline = time.time() + 5.0
            a_ref, score_ref = run_exponentiated_grad(a_opt, n, refine_deadline, start_time)
            if score_ref < best_score:
                best_score = score_ref

"""
Hybrid search for the self‑convolution L∞ problem (no enforced symmetry).

The algorithm proceeds in stages over a geometric schedule of lengths.
For each length:

1. Warm‑start from the best solution found for a shorter length
   (linear interpolation and renormalisation).

2. Several initial vectors are generated (uniform, linear decay,
   exponential decay, random Dirichlet, Gaussian bump).

3. Each vector is refined by Adam optimisation on a soft‑max surrogate
   of the convolution peak.  The surrogate temperature λ is decreased
   continuously until it is essentially zero, giving an accurate
   approximation of the true L∞ norm.  The iterates are kept on the
   probability simplex via Euclidean projection – no symmetrisation
   is performed, allowing genuinely asymmetric solutions.

4. The best Adam result is further refined by a multiplicative
   (exponentiated‑gradient) descent on the exact max‑convolution
   sub‑gradient.  This mirror‑descent step respects the simplex
   constraints naturally.

5. A short simulated‑annealing polishing phase moves a small ε of
   mass between random indices, accepting uphill moves with an
   exponential temperature schedule.

All stages respect the overall time budget (default 1000 s) and the
best sequence encountered is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -----------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) ------------------------------------
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """ Linear convolution a * a using a real FFT. """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """ Linear convolution of two real vectors via FFT. """
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """ Linear interpolation from old length to new_n and re‑normalise. """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the smoothed L∞ surrogate (soft‑max) --------------------
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) w.r.t. a.
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)     # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]   # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set) -------------------
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)               # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate ----------------------------
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best (a, score) observed in the allotted time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule
    lam_factor = 0.4          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.97
    lam_min_factor = 1e-9

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # sum(a)=1 ⇒ denominator =1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam it={it} n={n} score={cur_score:.6f}")

        # update λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base step decreasing like 1/√it
        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution ----------
# ----------------------------------------------------------------------
def _run_exp_grad(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Multiplicative (mirror) descent on the exact max‑convolution.
    Returns best (a, score) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    eta0 = 0.7
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        eta = eta0 / math.sqrt(it)

        # multiplicative update (avoid overflow)
        upd = np.exp(-eta * grad)
        a = a * upd
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad it={it} n={n} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing) ---------------------------
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination.
    Returns (best_seq, best_score) observed during the run.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal it={it} n={n} score={score:.6f}")

    return best_seq, best_score

# Initial shape generators ---------------------------------------------

def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce several plausible starting vectors derived from a base distribution.
    All returned vectors are normalised to sum = 1.
    """
    variants = []
    n = base_seq.shape[0]

    # 1. warm‑start copy (as‑is)
    variants.append(base_seq.copy())

    # 2. linear decay (a_i ∝ n‑i)
    lin = np.arange(n, 0, -1, dtype=float)
    lin /= lin.sum()
    variants.append(lin)

    # 3. exponential decay (r≈0.9995)
    r = 0.9995
    exp = r ** np.arange(n, dtype=float)
    exp /= exp.sum()
    variants.append(exp)

    # 4. Gaussian bump centred near the left edge
    x = np.linspace(-1, 1, n)
    gauss = np.exp(- (x ** 2) / 0.2)
    gauss /= gauss.sum()
    variants.append(gauss)

    # 5. random Dirichlet (concentration 0.8 → slightly sparse)
    rnd = np.random.dirichlet(np.full(n, 0.8))
    variants.append(rnd)

    return variants
