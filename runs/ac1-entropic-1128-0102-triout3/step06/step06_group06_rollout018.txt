def fft_len_needed(m):
    return 1 << ((m - 1).bit_length())

def conv_self(a):
    n = a.shape[0]
    m = 2*n-1
    L = fft_len_needed(m)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:m]
    return conv

def conv_fft(u, v):
    m = len(u) + len(v) - 1

def entropic_optimize(a_start, n, deadline, start_time):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # Temperature schedule
    tau = 5.0 / n
    tau_end = tau * 0.001
    decay = 0.999

    eta0 = 0.5
    it = 0

    a_rev = a[::-1]  # will be updated each iteration

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] entropic n={n} it={it} score={cur_sc:.6f}")

        # compute softmax weights
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        a_rev = a[::-1]  # update after a changes
        corr = conv_fft(w, a_rev)  # length = 3n-2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # update
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # anneal tau
        tau = max(tau_end, tau * decay)

    return best_a, best_sc

"""
Optimal step‑function search (≤ 1000 s)

The evaluator is scale‑invariant:
    score(sequence) = 2·n·max(conv(sequence,sequence))/sum(sequence)².
We therefore work with probability vectors (sum = 1) and minimise
    2·n·max(conv).

Approach
--------
1.  Build a hierarchy of lengths (256 … 8192).  For each length a
    candidate is up‑sampled from the best vector found for the previous
    length.
2.  Warm‑start the candidate with many analytic families
    (product‑power, Gaussian, Beta, arcsine, tent, sine‑power) and keep the
    best one.
3.  Refine the warm‑start with an entropic mirror‑descent (soft‑max
    surrogate τ·log ∑exp(conv/τ)).  The gradient of the surrogate is
    obtained by a single FFT‑based correlation; the update is an
    exponentiated‑gradient step on the probability simplex.
4.  Finish each length with very cheap mass‑transfer simulated‑annealing
    polish.
5.  All timings are checked aggressively – the routine always finishes
    before the global deadline, otherwise the grader returns –∞.

The implementation is fully self‑contained, uses only numpy / math /
random / time and follows the required API.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    L = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Cross‑convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    L = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (vectors are assumed normalised to sum = 1)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) → new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Analytic families for warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5) raised to p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i·(n‑i)) – very heavy at the ends."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(x - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin⁽ᵖ⁾(π·(i+0.5)/n)."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n) ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Entropic (soft‑max) mirror‑descent on the simplex
# ----------------------------------------------------------------------


def entropic_optimize(a0: np.ndarray, n: int,
                      deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) via a smooth log‑sum‑exp surrogate.
    Returns the best probability vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule – start a few multiples of 1/n, end ≈ 1e‑3/n
    tau = 5.0 / n
    tau_end = tau * 1e-3
    decay = 0.9995

    # step size for exponentiated‑gradient (mirrored on simplex)
    eta0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        # exact objective (track the best we see)
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---- soft‑max weights ----
        conv = conv_self(a)
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)            # length = 3n‑2
        idx = (n - 1) - np.arange(n)         # pick the n relevant entries
        grad = 2.0 * corr[idx]

        # ---- exponentiated‑gradient update ----
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)   # avoid overflow in exp()
        a = a * np.exp(expo)

        # re‑normalise onto the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # ---- anneal temperature ----
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def anneal_polish(a0: np.ndarray, n: int,
                  deadline: float, start_time: float,
                  cur_best_score: float, cur_best_seq: np.ndarray,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-5,
                  steps: int = 60000) -> tuple[np.ndarray, float]:
    """
    Cheap move‑based annealing: transfer a random ε from one index to another.
    Accept moves that improve the exact score or according to a geometric schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

    """

    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a globally stored construction
    # ------------------------------------------------------------------
    global_a = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            global_a = tmp / s
    except Exception:
        pass

    # Length schedule (ascending – larger vectors deserve more budget)
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    total_len = sum(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Allocate time roughly proportionally to length
        frac = n / total_len

        # Initialisation for this length

        if global_a is not None:

        # --------------------------------------------------------------
        best_local_a = a0.copy()

        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.15)
        while time.time() < shape_deadline:
            kind = random.choice([0, 1, 2, 3, 4, 5])
            if kind == 0:  # product‑power

            elif kind == 1:  # Gaussian

            elif kind == 2:  # Beta

        total_len -= n  # adjust remaining allocation
