"""
Hybrid local‐search for the self‑convolution L∞ problem.

The routine builds a sequence on the probability simplex (∑a_i = 1)
and iteratively reduces the peak of the linear convolution a*a.
It works in three stages:

1.  Multi‑scale warm‑starts – start from a tiny vector and repeatedly
    up‑sample to a larger length while retaining the shape.
2.  Projected sub‑gradient descent on the exact ‖a*a‖∞ (the
    max‑convolution).  This is an inexpensive O(n log n) update
    because the sub‑gradient can be computed by a single FFT.
3.  A cheap mass‑transfer simulated annealing that polishes the
    result.

The search is time‑budget aware – each length receives a slice of the
remaining time, and the algorithm stops well before the global deadline
to avoid a hard timeout.  The final answer is returned as a plain list
of floats, ready for ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (fast FFT implementation)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)  # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution (exact)
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12):
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    # gradient = 2 * (w * a_rev) convolution
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _run_subgradient_refine(a0: np.ndarray,
                            n: int,
                            dead: float,
                            start_time: float,
                            log_interval: int = 500):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its exact score (2·n·max_conv).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    step0 = 1.0  # a slightly larger initial step than the previous 0.5
    it = 0
    while time.time() < dead:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)          # stay on the simplex
        # optional: enforce a tiny symmetry bias – not forced
        if random.random() < 0.05:
            a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
        # occasional logging even without improvement
        if it % log_interval == 0:
            print(f"[{time.time() - start_time:6.2f}s] subgrad n={n} it={it} cur_score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (final polishing)
# ----------------------------------------------------------------------


def _run_annealing_refine(a0: np.ndarray,
                         n: int,
                         dead: float,
                         start_time: float,
                         cur_best_score: float,
                         cur_best_seq: np.ndarray,
                         start_temp: float = 5e-2,
                         end_temp: float = 1e-5,
                         steps: int = 250000):
    """
    Simple simulated annealing that moves a fraction of mass from one index
    to another.  Returns the best sequence observed and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= dead:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to ~12 % of the source mass
        eps = a[i] * np.random.rand() * 0.12
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape generators (used as warm‑starts)
# ----------------------------------------------------------------------


def _shape_uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    a = np.full(n, 1.0 / n)
    return a


def _shape_exponential(n: int, decay: float = 0.9) -> np.ndarray:
    """Monotonically decreasing exponential shape."""
    x = np.arange(n)
    a = np.exp(-x / (n * (1.0 - decay)))
    a /= a.sum()
    return a


def _shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    t = (np.arange(n) + 0.5) / n
    eps = 1e-12
    a = np.exp((alpha - 1) * np.log(t + eps) + (beta - 1) * np.log(1 - t + eps))
    a /= a.sum()
    return a


def _initial_candidates(n: int) -> list[np.ndarray]:
    """A handful of plausible starting vectors for a given length."""
    cand = [_shape_uniform(n)]

    # exponential decays with a few different rates
    for d in [0.7, 0.85, 0.95]:
        cand.append(_shape_exponential(n, decay=d))

    # a few beta‑distributions (symmetric and asymmetric)
    for (a, b) in [(2.0, 2.0), (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (1.5, 0.7)]:
        cand.append(_shape_beta(n, a, b))

    # a random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    cand.append(rnd)

    return cand
