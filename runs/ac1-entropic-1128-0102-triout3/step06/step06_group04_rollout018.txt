def propose_candidate(seed=0, budget_s=1000, **kwargs):

    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # list of n candidates
    lengths = [4096, 8192] maybe trying larger.

    for n in lengths:
        if time.time() >= deadline:
            break
        # search alpha values
        alphas = np.linspace(1.0, 4.0, 31) # inclusive maybe
        for alpha in alphas:
            if time.time() >= deadline:
                break
            if alpha >= n:
                continue
            r = 1.0 - alpha / n
            if r <= 0.0:
                continue
            # compute weights
            # use log to avoid underflow: arr = np.exp(np.arange(n) * np.log(r))
            log_r = np.log(r)
            exponent = np.arange(n) * log_r
            arr = np.exp(exponent, dtype=np.float64)
            # ensure sum >= maybe small; normalize
            s = arr.sum()
            if s < 1e-12:
                continue
            arr = arr / s
            # Evaluate
            score = evaluate_sequence(arr.tolist())
            if score < best_score:
                best_score = score
                best_seq = arr.copy()
                print(f"[{time.time()-start_time:6.2f}s] alpha={alpha:.3f} n={n} score={score:.6f}")
                # If below target break early maybe
                if best_score < 1.503:
                    break
        # if target reached break outer loop
        if best_score < 1.503:
            break

    # If no improvement found fallback to uniform
    if best_seq is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n)

    return [float(x) for x in best_seq]

def generate_exponential(alpha, n):
    if alpha <=0 or alpha >= n:
        return None
    d = alpha / n
    r = 1.0 - d
    # compute raw weights
    exp_part = np.exp(np.arange(n) * np.log(r))
    # normalization factor
    total = exp_part.sum()
    if total == 0:
        # fallback uniform
        seq = np.full(n, 1.0 / n)
    else:
        seq = exp_part / total
    return seq

"""
Propose a candidate step‑function using a simple parametric family.
The family is an exponentially decreasing sequence

    a_i(α) = r^i / sum_{j=0}^{n-1} r^j ,     r = 1 - α / n ,   α ∈ (0 , n)

For a fixed length n the objective 2·n·max(conv) varies smoothly with α.
By scanning α in a modest interval we can locate a value that beats the
previous best bound (≈1.513).  After the scan a tiny simulated‑annealing
phase (mass‑transfer moves) is run to polish the solution.

The whole procedure respects the 1000 s time‑budget and returns a plain
Python list ready for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helpers for the exponential family
# ----------------------------------------------------------------------


def _exp_sequence(alpha: float, n: int) -> np.ndarray:
    """
    Return the normalized sequence a_i = r**i  (i=0..n-1) with
        r = 1 - alpha/n   (0 < alpha < n).
    The sum of the returned array is exactly 1.
    """
    if alpha <= 0.0 or alpha >= n:
        # fall back to uniform – never happens in the search
        return np.full(n, 1.0 / n, dtype=np.float64)

    r = 1.0 - alpha / n
    # use logarithms to avoid loss of precision when r is very close to 1
    log_r = math.log(r)
    exponents = np.arange(n, dtype=np.float64) * log_r
    raw = np.exp(exponents)               # r**i
    total = raw.sum()
    if total == 0.0:                      # safety net – should not occur
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / total


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            cur_best_score: float,
            cur_best_seq: np.ndarray,
            start_temp: float = 2e-2,
            end_temp: float = 1e-6,
            steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing that moves a small fraction of mass
    from one index to another.  The objective is re‑evaluated with the
    original `evaluate_sequence` function.
    """
    a = seq.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    n = a.size

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move at most 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = evaluate_sequence(a_prop.tolist())
        if not math.isfinite(new_score):
            continue

        delta = new_score - best_score
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal improve score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The search uses a one‑parameter exponential family and a tiny annealing
    polish.  It finishes well within the 1000 s budget and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # short safety margin

    # ------------------------------------------------------------------
    # Lengths to try – start with the size that gave the previous best
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 8192]
    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] trying length n={n}")

        # Scan a focused range of α.
        # Empirically values around 0.7 – 1.1 give the smallest bound.
        alphas = np.linspace(0.6, 1.2, 121)   # step ≈0.005
        for alpha in alphas:
            if time.time() >= deadline:
                break

            seq = _exp_sequence(alpha, n)
            score = evaluate_sequence(seq.tolist())
            if not math.isfinite(score):
                continue

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best n={n} α={alpha:.4f} score={score:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Light polishing with simulated annealing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        best_seq, best_score = _anneal(best_seq,
                                       deadline=deadline - 0.2,
                                       start_time=start_time,
                                       cur_best_score=best_score,
                                       cur_best_seq=best_seq,
                                       start_temp=2e-2,
                                       end_temp=1e-6,
                                       steps=120000)

    # ------------------------------------------------------------------
    # Fallback – return a uniform distribution if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list as required by the platform
    return [float(x) for x in best_seq]
