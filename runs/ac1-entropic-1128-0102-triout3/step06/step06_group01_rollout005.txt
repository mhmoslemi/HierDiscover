"""
Propose a step function that (approximately) minimises
evaluate_sequence(sequence).

Algorithm
---------
1.  **Coarse shape search** – work on a small problem (n≈800) and
    sweep a few simple parametric families (symmetric Beta PDFs,
    inverse‑power tails) plus a handful of random Dirichlet seeds.
    The best shape is kept.

2.  **Upsampling** – the chosen coarse shape is linearly interpolated
    to the target length (n=4800) and re‑normalised to sum = 1.

3.  **Adam optimisation on the full size** – the smooth log‑sum‑exp
    approximation of the max‑convolution is minimised with Adam.
    The smoothing parameter α follows an increasing schedule, and
    the optimiser runs as long as the time budget permits.

4.  **Exact sub‑gradient polishing** – a short projected momentum
    gradient descent on the true nonsmooth objective (2·n·max(conv))
    refines the solution.

5.  **Simulated‑annealing fine‑tuning** – mass‑transfer moves
    (move a random ε‑fraction of probability mass from a source
    index to a destination index) are accepted with a standard
    Metropolis criterion.  The annealing schedule is long enough
    to explore the neighbourhood of the polished solution.

All stages respect the overall time budget (≈1000 s).  Progress
is printed to stdout.  The final sequence is normalised,

# FFT helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Simple parametric families for the coarse search
# ----------------------------------------------------------------------
def _beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric (or not) Beta density on n points, normalised to sum = 1."""
    # mid‑point rule to avoid exact zeros at the boundaries
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    pdf = (np.maximum(x, eps) ** (a - 1)) * (np.maximum(1 - x, eps) ** (b - 1))
    s = pdf.sum()
    if s == 0:
        pdf = np.full_like(pdf, 1.0 / n)
    else:
        pdf /= s
    return pdf


def _inverse_power_sequence(n: int, p: float) -> np.ndarray:
    """U‑shaped distribution proportional to (i+½)^(-p)+(n‑i‑½)^(-p)."""
    i = np.arange(n)
    left = (i + 0.5) / n
    right = (n - i - 0.5) / n
    vals = left ** (-p) + right ** (-p)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7):
    """
    Simple projected momentum gradient descent on 2·n·max(conv).
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # active set for the max (relative tolerance)
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score

# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Small‑scale annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30      # up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper for the coarse phase
# ----------------------------------------------------------------------
def _objective_small(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) under the assumption sum(a)=1."""
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine consists of:
        * a rapid parametric sweep on a small problem,
        * up‑sampling,
        * Adam optimisation with increasing smoothness,
        * exact sub‑gradient polishing,
        * simulated‑annealing fine‑tuning.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety buffer
    deadline = start + budget_s - 0.8

    # --------------------------------------------------------------
    # Determine the target length (use the reference size if possible)
    # --------------------------------------------------------------
    n_full = 4800
    # try to reuse a previously discovered construction
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # Stage 1 – coarse shape search on a small problem
    # --------------------------------------------------------------
    n_small = 800
    best_small = None
    best_obj_small = float("inf")

    # 1) symmetric Beta families
    for p in np.linspace(0.2, 3.0, 30):
        if time.time() >= deadline:
            break
        seq = _beta_sequence(n_small, p, p)
        obj = _objective_small(seq)
        if obj < best_obj_small:
            best_obj_small = obj
            best_small = seq

    # 2) inverse‑power tails
    for p in [0.5, 1.0, 1.5, 2.0]:
        if time.time() >= deadline:
            break
        seq = _inverse_power_sequence(n_small, p)
        obj = _objective_small(seq)
        if obj < best_obj_small:
            best_obj_small = obj
            best_small = seq

    # 3) a few random Dirichlet draws
    for _ in range(5):
        if time.time() >= deadline:
            break
        seq = np.random.rand(n_small)
        seq /= seq.sum()
        obj = _objective_small(seq)
        if obj < best_obj_small:
            best_obj_small = obj
            best_small = seq

    # 4) down‑sample a known construction if it exists
    if starter is not None:
        cand = _upsample(starter, n_small)
        obj = _objective_small(cand)
        if obj < best_obj_small:
            best_obj_small = obj
            best_small = cand

    # Fallback: uniform distribution
    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # --------------------------------------------------------------
    # Upsample to the full size
    # --------------------------------------------------------------
    a0 = _upsample(best_small, n_full)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    # --------------------------------------------------------------
    # Stage 2 – Adam optimisation on the full size
    # --------------------------------------------------------------
    eps_logit = 1e-16
    y = np.log(np.maximum(a0, eps_logit))

    fft_len = _next_pow2(3 * n_full - 2)

    # smoothing schedule (α grows, approximating the true max)
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    # keep the best exact score seen so far
    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    for alpha in alphas:
        i = 0
        # adapt the iteration budget to the remaining time
        while i < 20000 and time.time() < deadline:
            # softmax to obtain a probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective and its gradient
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj_val = 2.0 * n_full * f_val

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n_full)

            # chain rule to get gradient w.r.t. logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

            i += 1

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Stage 3 – exact sub‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n_full,
            polish_deadline,
            start,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start:6.2f}s] polishing improve: score={score_polish:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – simulated‑annealing fine tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_fine, score_fine = _run_annealing(
            best_seq,
            n_full,
            deadline,
            start,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_fine < best_score:
            best_score = score_fine
            best_seq = a_fine.copy()
            print(f"[{time.time() - start:6.2f}s] final anneal improve: score={score_fine:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n_full)

    return [float(x) for x in best_seq.tolist()]
