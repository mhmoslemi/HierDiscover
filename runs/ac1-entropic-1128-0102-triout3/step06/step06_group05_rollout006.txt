"""
Hybrid search for the self‑convolution ℓ∞ problem.

The algorithm proceeds in three stages:

1. **Parametric warm‑start** – we generate a family of symmetric beta‑type
   shapes a_i ∝ (x_i (1‑x_i))^{-p} (the arcsine distribution corresponds to
   p = 0.5) for several values of p and keep the best one (according to the
   exact objective 2·n·max(conv)).  A few extra random Dirichlet draws are also
   tested.

2. **Refinement** – the best warm‑start is refined by three complementary
   optimisation loops that respect the simplex constraint:
      * Adam on a soft‑max surrogate of the ℓ∞ norm.
      * Projected sub‑gradient descent on the exact max‑convolution.
      * Exponentiated‑gradient descent (multiplicative updates) on the exact
        sub‑gradient.
   Each loop runs as long as the remaining time budget allows.

3. **Polish** – a cheap simulated‑annealing mass‑transfer scheme is finally
   applied to squeeze out any remaining improvement.

All updates are projected onto the probability simplex and (optionally)
symmetrised; the final sequence is returned as a plain Python list of
non‑negative floats.  The routine respects the supplied `budget_s` time
limit (default 1000 s) and never exceeds it.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (≥0, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for any length)."""
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)[:size]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c

# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp (soft‑max) surrogate.
    Returns dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation of weights with reversed a
    full = _conv_fft(w, a[::-1])               # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad

def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12):
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.
    Returns (grad, max_val).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    full = _conv_fft(w, a[::-1])               # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val

# ----------------------------------------------------------------------
# Parameterised symmetric beta‑type shape
# ----------------------------------------------------------------------
def _shape_arcsine(n: int, p: float) -> np.ndarray:
    """
    Return a symmetric density a_i ∝ (x_i (1‑x_i))^{-p},
    where x_i = (i+0.5)/n.
    """
    x = (np.arange(n) + 0.5) / n
    vals = (x * (1.0 - x) + 1e-12) ** (-p)
    vals /= vals.sum()
    return vals

# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False):
    """Adam optimiser of the soft‑max surrogate. Returns best (a, score)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.6          # initial λ as a fraction of max(conv)
    lam_decay = 0.985
    lam_min_factor = 1e-9
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # update smoothing parameter λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.7 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Deterministic projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float):
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    step0 = 0.6

    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative updates)
# ----------------------------------------------------------------------
def _run_expgd(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float):
    """Multiplicative (exponentiated) gradient descent on the exact max‑conv."""
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    eta0 = 0.5

    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        eta = eta0 / math.sqrt(it + 1.0)
        # multiplicative update, keep positivity
        a = a * np.exp(-eta * grad)
        # renormalise to the simplex
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)
        # optional symmetrisation
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGD improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Cheap mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000):
    """Single‑source / single‑dest mass‑transfer annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12  # move up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a Python list of floats (the heights).  The routine respects the
    supplied time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # tiny safety margin to avoid overshooting the limit
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine working length (larger lengths give higher resolution but
    # also slower iterations).  We pick 32768 if enough time, otherwise a
    # smaller power of two.
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    if total_time >= 400:
        n = 32768
    elif total_time >= 200:
        n = 16384
    elif total_time >= 100:
        n = 8192
    else:
        n = 4096

    # ------------------------------------------------------------------
    # Warm‑start candidates: a small parametric sweep over beta‑type shapes
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')

    # try a few exponents p (including uniform p=0)
    p_vals = [0.0] + list(np.linspace(0.1, 1.2, 13))
    for p in p_vals:
        a = _shape_arcsine(n, p)
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Warm‑start p={p:.3f} score={cur_score:.6f}")

    # add a plain sine shape
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_shape = sin_shape / sin_shape.sum()
    cur_score = 2.0 * n * _conv_self(sin_shape).max()
    if cur_score < best_score:
        best_score = cur_score
        best_a = sin_shape.copy()
        print(f"[{time.time() - start_time:6.2f}s] Warm‑start sine score={cur_score:.6f}")

    # add some random Dirichlet draws
    for _ in range(8):
        a = np.random.rand(n)
        a = a / a.sum()
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Warm‑start random score={cur_score:.6f}")

    # if a previous construction is available, up‑sample it
    global_a = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                global_a = starter.copy()
    except Exception:
        global_a = None

    if global_a is not None:
        # linear interpolation to current length
        old_n = global_a.shape[0]
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, n)
        up = np.interp(new_x, old_x, global_a)
        up /= up.sum()
        # compare with current best
        cur_score = 2.0 * n * _conv_self(up).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = up.copy()
            print(f"[{time.time() - start_time:6.2f}s] Warm‑start global score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Time slicing – allocate remaining budget among optimisation phases
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0:
        return [float(x) for x in best_a]

    # Adam phase (~30 % of remaining time)
    adam_deadline = time.time() + max(5.0, time_left * 0.30)
    a_adam, score_adam = _run_adam_opt(best_a, n, adam_deadline,
                                       start_time, symmetrize=False)
    if score_adam < best_score:
        best_score, best_a = score_adam, a_adam.copy()
        print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam score={score_adam:.6f}")

    # Sub‑gradient phase (~30 % of remaining time)
    time_left = deadline - time.time()
    if time_left > 8.0:
        sub_deadline = time.time() + max(5.0, time_left * 0.30)
        a_sub, score_sub = _run_subgradient(best_a, n, sub_deadline, start_time)
        if score_sub < best_score:
            best_score, best_a = score_sub, a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after sub‑grad score={score_sub:.6f}")

    # Exponentiated‑gradient phase (~20 % of remaining time)
    time_left = deadline - time.time()
    if time_left > 8.0:
        exp_deadline = time.time() + max(5.0, time_left * 0.20)
        a_exp, score_exp = _run_expgd(best_a, n, exp_deadline, start_time)
        if score_exp < best_score:
            best_score, best_a = score_exp, a_exp.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after ExpGD score={score_exp:.6f}")

    # Final annealing polish (use whatever time is left)
    time_left = deadline - time.time()
    if time_left > 5.0:
        anneal_deadline = time.time() + time_left
        a_ann, score_ann = _run_annealing(best_a, n, anneal_deadline,
                                          start_time, best_score, best_a,
                                          start_temp=5e-2, end_temp=1e-5,
                                          steps=150000)
        if score_ann < best_score:
            best_score, best_a = score_ann, a_ann.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal score={score_ann:.6f}")

    # ------------------------------------------------------------------
    # Return the best-discovered sequence as a plain list
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
