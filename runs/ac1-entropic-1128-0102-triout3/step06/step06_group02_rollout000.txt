"""
Propose a step‑function (non‑negative vector) that minimises ``evaluate_sequence``.

Approach
--------
* Work on the *simplex* (∑a = 1).  The evaluation then equals ``2·n·max(conv(a,a))``,
  i.e. we only have to minimise the maximum of the self‑convolution.
* Progressive up‑sampling: start with a very short vector, optimise it and
  repeatedly double the resolution.  Higher resolutions give a finer approximation
  of the continuous optimum.
* Optimisation consists of three phases per resolution:
    1. A few soft‑max gradient steps (smooth approximation of ``max``) to get
       into a good basin.
    2. Projected sub‑gradient descent on the exact ``max`` – the objective is
       convex, the sub‑gradient is cheap to compute.
    3. After the resolution loop a long simulated‑annealing walk that moves a
       small amount of mass from a high‑value entry to a low‑value one.
* The vector is *symmetrised* after each update: ``a ← (a + a[::-1]) / 2``.
  The objective and the constraints are symmetric, so this reduces the
  search space and typically yields a lower maximal convolution.
* Timing: 75 % of the budget is spent on the progressive optimisation,
  the remaining 25 % on the final annealing.  The code respects the deadline
  and always returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: a_i = a_{n-1-i}."""
    return (v + v[::-1]) * 0.5


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n and simplex projection."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = _project_simplex(up)
    up = _symmetrize(up)
    return up


# ----------------------------------------------------------------------
# Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    # nearest power of two not smaller than 2*n-1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient – for a quick warm‑start
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of soft‑max of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))       # improve stability
    w = exps / exps.sum()
    rev_a = a[::-1]

    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.7,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov gradient descent on the soft‑max approximation."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (geometric)
    T_start, T_end = 2e-2, 5e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max improve: n={n}, it={it}, score={cur_score:.6f}")

        # temperature decays slowly
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.12,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, it={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a.
        # The contribution is 2·a_{k-i} for i such that 0≤i<n and 0≤k-i<n.
        grad = np.zeros_like(a)
        if k < n:
            # i = 0 … k   →   partner index = k-i
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            # i runs from k-n+1 … n-1
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement – mass transfer moves
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.25,
                   end_temp: float = 1e-7,
                   steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass from a heavy to a light entry."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # Pre‑compute a list of indices sorted by current mass (updated lazily)
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # Choose donor among the top 10 % mass entries, recipient among bottom 10 %
        # This focuses the moves where they are most useful.
        top_k = max(2, n // 10)
        low_k = max(2, n // 10)

        # cheap sampling without sorting every iteration:
        i = np.random.choice(np.argpartition(a, -top_k)[-top_k:])
        j = np.random.choice(np.argpartition(a, low_k)[:low_k])

        if i == j or a[i] == 0.0:
            continue

        # move a small fraction (up to 15 %) of mass from i to j
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # maintain simplex (sum unchanged) and symmetry
        a_prop = _symmetrize(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score

            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Final normalisation – numerical drift may have occurred.
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the harness
    return [float(x) for x in best_seq]
