The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `*` denotes the ordinary (zero‑padded) convolution.
Because the expression is scale‑invariant we may restrict ourselves to
probability vectors (`∑a_i = 1`).  
For a uniform vector the value of `F` is exactly 2; the best known
construction gives about 1.512 – we aim to push this a little further.

The implementation follows a **multi‑scale, hybrid optimisation**:

1. **Coarse optimisation** – start with a short exponentiated‑gradient
   (smooth log‑sum‑exp approximation of the max) on a small problem
   ( 600 entries).  
2. **Refinement on larger scales** – repeatedly up‑sample the current
   candidate to the next size ( 1200 → 2400 → 4800 ) and run a fresh
   warm‑up + projected sub‑gradient descent.  
   Several random restarts per scale increase the chance to escape shallow
   local minima.
3. **Final polishing** – a fast simulated‑annealing pass moves a tiny
   fraction of mass between entries while always staying on the simplex.

All phases respect the global 1000 s time budget.  The code is completely
self‑contained – only `numpy` and the supplied `evaluate_sequence` are
required.

```python
"""
Multi‑scale hybrid optimiser for the step‑function bound.

The algorithm works on four resolutions (600 → 1200 → 2400 → 4800).  For
each resolution it runs a short exponentiated‑gradient warm‑up on the
log‑sum‑exp smooth approximation of the max‑convolution, followed by a
long projected sub‑gradient descent on the true objective.  Several
random restarts are performed at the coarse levels; the best candidate
is up‑sampled to the next resolution.  After the final resolution a
tiny simulated‑annealing polishing is performed.

The whole procedure stops after the allotted 1000 s and returns the best
sequence found (normalised to sum = 1, all entries ≥ 0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (sum of contributions of
    all near‑max entries)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a (sum is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 6000):
    """Multiplicative updates based on a log‑sum‑exp smooth of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # enough for a·w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient (log‑sum‑exp)
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # stable scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a[:] = 1.0 / n

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exact‑max projected sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5,
                             beta: float = 0.95,
                             rel_tol: float = 1e-12):
    """Projected sub‑gradient descent (with momentum) on the true objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv, rel_tol=rel_tol)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    """Mass‑transfer Metropolis moves with exponential cooling."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # linear cooling in log‑space
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Initialise – use the global ``height_sequence_1`` if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # Hierarchical sizes (coarse → fine)
    final_n = 4800
    levels = [600, 1200, 2400, final_n]

    # Initialise at the coarsest level
    if starter is not None:
        a_current = _upsample(starter, levels[0])
    else:
        a_current = np.full(levels[0], 1.0 / levels[0])

    best_seq = a_current.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Multi‑scale optimisation
    # ------------------------------------------------------------------
    for lvl_idx, n in enumerate(levels):
        now = time.time()
        if now >= deadline:
            break

        # Allocate time for this level
        if n != final_n:
            # a small fixed budget for the coarse levels (max 30 s each)
            level_time = min(30.0, 0.10 * (deadline - now))
        else:
            level_time = deadline - now
        level_deadline = now + level_time

        # Number of random restarts (more for coarse levels)
        n_restarts = 4 if n != final_n else 1
        local_best_seq = None
        local_best_score = float("inf")

        for r in range(n_restarts):
            # ----------------------------------------------------------------
            # Initialise this restart
            # ----------------------------------------------------------------
            if r == 0:
                a0 = a_current.copy()
            else:
                a0 = np.random.rand(n)
                a0 /= a0.sum()

            # ------------------- warm‑up (exp‑grad) ------------------------
            warm_deadline = time.time() + max(2.0, 0.05 * level_time)
            a_warm, score_warm = _run_exponential_gradient(
                a0,
                n,
                warm_deadline,
                start_time,
                alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
                eta0=0.6,
                steps_per_alpha=2000,
            )

            # ------------------- exact max sub‑gradient -------------------
            sub_deadline = level_deadline
            a_sub, score_sub = _run_subgradient_descent(
                a_warm,
                n,
                sub_deadline,
                start_time,
                step0=0.5,
                beta=0.95,
                rel_tol=1e-12,
            )

            # Pick the better result of this restart
            if score_sub < local_best_score:
                local_best_score = score_sub
                local_best_seq = a_sub.copy()
            elif score_warm < local_best_score:
                local_best_score = score_warm
                local_best_seq = a_warm.copy()

        # --------------------------------------------------------------
        # Update global best and move to the next resolution
        # --------------------------------------------------------------
        if local_best_seq is not None and local_best_score < best_score:
            best_seq = local_best_seq.copy()
            best_score = local_best_score
            print(f"[{time.time() - start_time:.2f}s] level {n} improve: "
                  f"score={best_score:.6f}")

        # Upsample for the next level (except after final level)
        if n != final_n:
            next_n = levels[lvl_idx + 1]
            a_current = _upsample(best_seq, next_n)
        else:
            a_current = best_seq.copy()

    # ------------------------------------------------------------------
    # Final simulated‑annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    now = time.time()
    if now < deadline:
        a_final, score_final = _run_annealing(
            a_current,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_seq = a_final.copy()
            best_score = score_final
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"Finished – best objective = {best_score:.6f}")
    return [float(x) for x in best_seq]
```
