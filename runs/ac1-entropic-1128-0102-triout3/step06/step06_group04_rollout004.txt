"""
A multi‑stage optimizer for the self‑convolution peak.
Strategy
---------
1. Warm‑start from a previously discovered sequence (if any) or a random Dirichlet.
2. Grow the support length gradually; each size gets a slice of the time budget.
3. For each length:
   • Several restarts of projected Adam on a smooth log‑sum‑exp surrogate
     (symmetrised after every projection).
   • A short series of LP “mass‑swap’’ steps – the classic LP‑based attack –
     kept on the simplex and symmetrised.
   • A brief projected sub‑gradient descent on the exact max‑norm.
   • A simulated‑annealing polishing phase (mass‑transfer moves).
4. After the last length a final heavy Adam + annealing polishing is performed.
All iterates stay on the probability simplex (∑a_i = 1) and are forced to be
symmetric, which empirically yields a lower ℓ∞ self‑convolution.
The routine respects the supplied time budget and returns the best sequence
found as a plain Python list.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Mirrored‑descent onto the probability simplex)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution helpers (FFT – O(n log n))
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum value of the self‑convolution (a normalized to sum=1)."""
    return _conv_self(a).max()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (a is assumed to sum to 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)                # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]      # central slice of length n
    return grad


# ----------------------------------------------------------------------
# Projected Adam on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation on the log‑sum‑exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.5          # λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 2000

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)

        lam = max(conv.max() * lam_factor, 1e-12)
        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap’’ step (the classic LP attack)
# ----------------------------------------------------------------------
def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A for constraints (f * b)_k ≤ M.
    Each row k contains f_i for i ≤ k < i + n, 0 elsewhere.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_improve(a: np.ndarray,
                n: int,
                step: float = 0.2,
                max_iters: int = 8,
                deadline: float = None,
                start_time: float = None) -> np.ndarray:
    """Repeated LP‑based mass‑swap improvement."""
    a_cur = a.copy()
    for it in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σb ⇒ minimise –Σb
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        # enforce symmetry and simplex
        a_cur = (a_cur + a_cur[::-1]) * 0.5
        a_cur = _project_simplex(a_cur)
        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP iter={it+1} n={n} score={sc:.6f}")
    return a_cur


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑norm
# ----------------------------------------------------------------------
def _subgradient_opt(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            # contribution a_i * a_{k-i}
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            # the symmetric term a_j * a_{i-k}
            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        step = 0.6 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal_opt(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                cur_best_score: float,
                cur_best_seq: np.ndarray,
                start_temp: float = 5e-2,
                end_temp: float = 1e-5,
                steps: int = 150000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing; keeps the best encountered sequence."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.10   # up to 10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # enforce symmetry after the move
        a_prop = (a_prop + a_prop[::-1]) * 0.5
        a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and simple shape generators
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution on {0,…,n‑1}."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
        tri = np.concatenate([left, right])
    else:
        left = np.arange(1, half + 1)
        right = left[::-1]
        tri = np.concatenate([left, [half + 1], right])
    tri = tri.astype(float)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    return tri

# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered sequence if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – start modest and grow.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    random.shuffle(lengths)

    best_seq = None
    best_score = float('inf')
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm‑start + a few variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = [
            a0,
            _triangular(n),
            np.random.rand(n)  # random Dirichlet
        ]
        init_candidates = [c / c.sum() for c in init_candidates]

        # --------------------------------------------------------------
        # Allocate a time slice for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # Later (larger) lengths receive a larger fraction.
        fraction = 0.08 + 0.42 * (idx / max(1, len(lengths) - 1))
        slice_deadline = time.time() + max(4.0, remaining * fraction)

        # --------------------------------------------------------------
        # Adam optimisation (multiple restarts)
        # --------------------------------------------------------------
        local_best_a = None
        local_best_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            a_opt, sc_opt = _adam_opt(start_vec, n,
                                      slice_deadline, start_time,
                                      symmetrize=True)
            if sc_opt < local_best_score:
                local_best_score = sc_opt
                local_best_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam local best n={n} score={sc_opt:.6f}")

        if local_best_a is None:
            continue

        # --------------------------------------------------------------
        # Optional LP “swap’’ phase (only for moderate sizes to keep memory low)
        # --------------------------------------------------------------
        if n <= 2048:
            lp_deadline = min(slice_deadline,
                              time.time() + min(8.0, (deadline - time.time()) * 0.15))
            local_best_a = _lp_improve(local_best_a, n,
                                        step=0.2,
                                        max_iters=6,
                                        deadline=lp_deadline,
                                        start_time=start_time)
            local_best_score = _objective(local_best_a)

        # --------------------------------------------------------------
        # Sub‑gradient refinement (quick, cheap)
        # --------------------------------------------------------------
        if time.time() < deadline:
            sg_deadline = time.time() + min(5.0, (deadline - time.time()) * 0.10)
            a_sg, sc_sg = _subgradient_opt(local_best_a, n,
                                           sg_deadline, start_time)
            if sc_sg < local_best_score:
                local_best_score = sc_sg
                local_best_a = a_sg.copy()
                print(f"[{time.time() - start_time:6.2f}s] Sub‑grad improve n={n} score={sc_sg:.6f}")

        # --------------------------------------------------------------
        # Annealing polishing for this length
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.20)
            a_ref, sc_ref = _anneal_opt(local_best_a, n,
                                        deadline=anneal_deadline,
                                        start_time=start_time,
                                        cur_best_score=local_best_score,
                                        cur_best_seq=local_best_a,
                                        start_temp=2e-2,
                                        end_temp=1e-5,
                                        steps=200000)
            if sc_ref < local_best_score:
                local_best_score = sc_ref
                local_best_a = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} score={sc_ref:.6f}")

        # remember the best for the next up‑sample
        global_a = local_best_a.copy()
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final heavy polishing on the longest length we managed to keep.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        final_deadline = deadline - 0.5
        a_final, sc_final = _adam_opt(best_seq, n,
                                      final_deadline,
                                      start_time,
                                      symmetrize=True)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final Adam improve score={sc_final:.6f}")

        if time.time() < deadline - 0.5:
            anneal_deadline = deadline - 0.2
            best_seq, best_score = _anneal_opt(best_seq, n,
                                               deadline=anneal_deadline,
                                               start_time=start_time,
                                               cur_best_score=best_score,
                                               cur_best_seq=best_seq,
                                               start_temp=1e-2,
                                               end_temp=1e-5,
                                               steps=300000)

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return even if everything fails.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Convert to plain Python list for the evaluator.
    return [float(x) for x in best_seq]
