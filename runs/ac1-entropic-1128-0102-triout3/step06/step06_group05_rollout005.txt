    np.random.seed(seed); random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine n; could try multiple lengths but for time we go with n_max = 32768; or choose length that fits within time.

    # Set target lengths: maybe small initial passes at small n (256, 1024) to warm up?
    # But we can skip; big passes will dominate.

    # If too slow, reduce.
    # Build initial a: arcsine shape
    t = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    a = a / a.sum()
    a = (a + a[::-1]) * 0.5

    best_a = a.copy()
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()

    # Phase 1: Adam smoothing
    phase_deadline = start_time + (deadline - start_time) * 0.55
    a_current = best_a.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a_current)
    v = np.zeros_like(a_current)
    lam_factor = 0.5
    lam_decay = 0.99
    lam_min_factor = 1e-9
    iteration = 0
    while time.time() < phase_deadline:
        iteration += 1
        conv = _conv_self(a_current)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score; best_a = a_current.copy()
            print(f"[{time.time()-start_time:6.2f}s] Adam improved it={iteration} score={cur_score:.6f}")

        # Update λ
        lam = max(cur_max * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient from surrogate
        # compute softmax weights
        max_val = cur_max
        weights = np.exp((conv - max_val) / lam)
        weights_sum = weights.sum()
        if weights_sum == 0:
            # fallback to uniform?
            weights = np.ones_like(conv) / conv.size
        else:
            weights = weights / weights_sum

        # gradient: 2 * (weights * a reversed conv)
        grad = 2.0 * _conv_fft(weights, a_current[::-1])[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step_size = 0.5 / math.sqrt(iteration) # may adjust
        a_current = a_current - step_size * m_hat / (np.sqrt(v_hat) + eps)
        # symmetrize
        a_current = (a_current + a_current[::-1]) * 0.5
        # project
        a_current = _project_simplex(a_current)

    # Phase 2: subgradient descent
    phase2_deadline = start_time + (deadline - start_time) * 0.85
    a_current = best_a.copy()
    step0 = 0.5
    it = 0
    while time.time() < phase2_deadline:
        it += 1
        grad, cur_max = _gradient_max_conv(a_current)
        step = step0 / math.sqrt(it + 1)
        a_current = a_current - step * grad
        a_current = _project_simplex(a_current)
        a_current = (a_current + a_current[::-1]) * 0.5
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_current.copy()
            print(f"[{time.time()-start_time:6.2f}s] Subgrad improved it={it} score={cur_score:.6f}")

    # Phase 3: annealing/hill climbing
    phase3_deadline = deadline
    a_current = best_a.copy()
    cur_score = best_score
    while time.time() < phase3_deadline:
        # random transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j or a_current[i] == 0:

        eps_transfer = a_current[i] * random.random() * 0.2 # up to 20%
        a_prop = a_current.copy()
        a_prop[i] -= eps_transfer
        a_prop[j] += eps_transfer
        # reduce round-off errors
        a_prop = _project_simplex(a_prop) # but values still sum to 1; should be close.
        # symmetrize maybe
        # Actually maintain symmetry by moving both i and its mirror?
        # Let's enforce symmetry: if we move mass at i, also move at n-1-i accordingly to keep near symmetric shape
        # Simpler: after move, symmetrize
        a_prop = (a_prop + a_prop[::-1]) * 0.5

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < cur_score:
            a_current = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_current.copy()
                print(f"[{time.time()-start_time:6.2f}s] Anneal improved score={cur_score:.6f}")
    # end

    return best_a.tolist()

"""
Hybrid Adam – Subgradient – Hill‑Climbing optimizer.

The routine works directly on the probability simplex (∑a_i = 1) because the
objective returned by ``evaluate_sequence`` is scale‑invariant.
It proceeds in three phases:

1. **Adam on a soft‑max surrogate** – quickly brings the distribution into a
   region where the L∞‑norm of the self‑convolution is already low.
2. **Long deterministic sub‑gradient descent** on the exact maximum
   convolution value.  The step size follows a 1/√k schedule and the vector
   is kept symmetric by averaging with its reverse.
   Random perturbations are injected from time to time to escape plateaus.
3. **Simple hill‑climbing (mass‑transfer moves)** – cheap local moves are
   accepted only when they improve the exact objective, providing a final
   polishing stage.

All iterates are projected onto the simplex after every update.
The best vector seen during the whole run is returned as a plain Python
list.  The algorithm respects the supplied ``budget_s`` time limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) ------------------------------------
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for any length)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = x.shape[0] + y.shape[0] - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set) --------------------
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12):
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.
    Returns (grad, max_value), where grad has shape (n,).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    # if nothing is exactly active (numerical), fall back to the exact max entry
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float)
    w /= w.sum()
    # cross‑correlation of w with reversed a
    full = _conv_fft(w, a[::-1])                     # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (log‑sum‑exp) ----------------------
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) = λ·log Σ exp(conv/λ).

    Parameters
    ----------
    a   – current probability vector (sum = 1)
    conv – self‑convolution of a (length 2n‑1)
    lam – smoothing temperature (λ > 0)

    Returns
    -------
    grad – dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # cross‑correlation of soft‑max weights with reversed a
    full = _conv_fft(w, a[::-1])
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function that minimizes `evaluate_sequence`.

    The routine respects a total wall‑clock budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin to avoid being cut off exactly at the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a finer discretisation of the optimal continuous shape.
    # We pick 32768 (a power of two) – fast enough for many FFTs in the time budget.
    # ------------------------------------------------------------------
    n = 32768

    # ------------------------------------------------------------------
    # Initialise a symmetric arcsine‑like distribution (the classical good start)
    # ------------------------------------------------------------------
    t = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    a /= a.sum()
    a = (a + a[::-1]) * 0.5                     # enforce symmetry

    # keep the best solution ever seen
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate (10 % of total budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.10
    a_cur = a.copy()
    m = np.zeros_like(a_cur)
    v = np.zeros_like(a_cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    lam_factor = 0.5          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.99
    lam_min_factor = 1e-9

    while time.time() < phase1_deadline:
        it += 1
        conv = _conv_self(a_cur)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_cur.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve it={it} score={cur_score:.6f}")

        # temperature for the soft‑max surrogate
        lam = max(cur_max * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of the smooth surrogate
        grad = _softmax_gradient(a_cur, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing base step
        base_step = 0.8 / math.sqrt(it + 1.0)

        a_cur = a_cur - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry and feasibility
        a_cur = (a_cur + a_cur[::-1]) * 0.5
        a_cur = _project_simplex(a_cur)

    # ------------------------------------------------------------------
    # Phase 2 – Long deterministic sub‑gradient descent (≈75 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + (deadline - start_time) * 0.85
    a_cur = best_seq.copy()
    step0 = 0.5
    it = 0
    last_improve = 0

    while time.time() < phase2_deadline:
        it += 1
        grad, cur_max = _gradient_max_conv(a_cur)
        step = step0 / math.sqrt(it + 1.0)
        a_next = a_cur - step * grad
        a_next = _project_simplex(a_next)
        a_next = (a_next + a_next[::-1]) * 0.5

        # compute exact score for the candidate
        conv = _conv_self(a_next)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            last_improve = it
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve it={it} score={cur_score:.6f}")

        a_cur = a_next

        # every few thousand iterations, inject a small random perturbation if we have been stagnant
        if it % 2000 == 0 and (it - last_improve) > 5000:
            # perturb the current best – perturbation size decays with remaining time
            remaining = deadline - time.time()
            total = deadline - start_time
            noise_scale = 0.02 * (remaining / total)               # at most a few percent of the mass
            a_cur = best_seq + noise_scale * np.random.randn(n)
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = _project_simplex(a_cur)
            a_cur = (a_cur + a_cur[::-1]) * 0.5
            last_improve = it

    # ------------------------------------------------------------------
    # Phase 3 – Simple hill‑climbing (mass transfer) (remaining time)
    # ------------------------------------------------------------------
    a_cur = best_seq.copy()
    cur_score = best_score
    steps_done = 0
    while time.time() < deadline:
        steps_done += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a_cur[i] == 0.0:
            continue
        # move up to 15 % of the source mass
        eps_transfer = a_cur[i] * np.random.rand() * 0.15
        a_prop = a_cur.copy()
        a_prop[i] -= eps_transfer
        a_prop[j] += eps_transfer
        # enforce symmetry and feasibility
        a_prop = (a_prop + a_prop[::-1]) * 0.5
        a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < cur_score:
            cur_score = new_score
            a_cur = a_prop
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_cur.copy()
                print(f"[{time.time() - start_time:6.2f}s] Anneal improve step={steps_done} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:6.2f}s] final best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
