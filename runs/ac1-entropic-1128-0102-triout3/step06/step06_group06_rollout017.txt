Project Simplex algorithm:

def simplex_projection(v):
    """Project onto probability simplex (non-negative, sum=1)"""
    # If all v >=0 and sum=1, return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# FFT based convolutions (linear, not circular)
# ----------------------------------------------------------------------


def _next_pow_two(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    L = _next_pow_two(m)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT."""
    m = len(u) + len(v) - 1
    L = _next_pow_two(m)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of Duchi et al. (2008)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length and re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families – warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term <= 0, 1e-12, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term <= 0, 1e-12, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def shape_triangle_left(n: int) -> np.ndarray:
    """Linear decrease from left to right (left‑heavy)."""
    a = np.arange(n, 0, -1, dtype=float)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def shape_dirichlet(n: int, conc: float = 0.5) -> np.ndarray:
    """Random Dirichlet sample – useful for stochastic exploration."""
    g = np.random.gamma(conc, 1.0, size=n)
    s = g.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return g / s


# ----------------------------------------------------------------------
# Entropic (soft‑max) mirror descent on the simplex
# ----------------------------------------------------------------------


def entropic_mirror_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    tau0: float = None,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Minimises the true max‑norm via a smooth surrogate.
    Returns the best vector encountered and its exact (non‑smooth) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule – start a bit higher than the default and cool very slowly
    if tau0 is None:
        tau = 8.0 / n          # start a little hotter than before
    else:
        tau = tau0
    tau_end = tau * 1e-4
    decay = 0.9997            # slower decay than the reference implementation

    it = 0
    while time.time() < deadline:
        it += 1

        # softened max via soft‑max weights
        conv = conv_self(a)
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate : 2 * (w ★ a_rev)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                     # length = 3n‑2
        idx = (n - 1) - np.arange(n)                 # pick the slice that aligns with a
        grad = 2.0 * corr[idx]

        # exponentiated‑gradient update
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)           # avoid overflow
        a = a * np.exp(expo)

        # project back onto the simplex
        a = project_simplex(a)

        # track best exact score
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic it={it} n={n} score={cur_sc:.6f}"
            )

        # anneal temperature
        if tau > tau_end:
            tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (tiny temperature, true max)
# ----------------------------------------------------------------------


def hard_subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    eta0: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    After the entropic phase we can refine with the exact sub‑gradient
    of the ∞‑norm.  The update is a projected gradient step on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        k = conv.argmax()          # index that attains the max

        # sub‑gradient of conv[k] wrt a  -> 2 * a shifted by k
        grad = np.zeros_like(a)
        j = k - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        step = eta0 / math.sqrt(it)
        a = a - step * grad
        a = project_simplex(a)

        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] hard‑sgd it={it} n={n} score={cur_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing (mass transfer) – cheap polish
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    steps: int = 200000,
    start_temp: float = 1e-1,
    end_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to the
    current temperature.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 30 % of the source mass – a bit more aggressive than before
        eps = a[i] * np.random.rand() * 0.3

        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists.
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            starter = tmp / s
    except Exception:
        starter = None

    # We work with a single length that is large enough to give a good
    # constant factor (the objective is length‑invariant).  4096 offers a
    # nice trade‑off between resolution and speed.
    n = 4096

    # --------------------------------------------------------------
    # Initialisation – a pool of candidate shapes
    # --------------------------------------------------------------
    pool = []

    # include starter if present
    if starter is not None:
        pool.append(starter)

    # uniform – always a solid baseline
    pool.append(np.full(n, 1.0 / n))

    # parametric families
    for _ in range(8):
        kind = np.random.choice(["product", "gauss", "beta", "arcsine", "tri", "dirich"])
        if kind == "product":
            p = np.random.uniform(-3.0, 3.0)
            cand = shape_product_power(n, p)
        elif kind == "gauss":
            sigma = np.random.uniform(n / 40.0, n / 2.0)
            cand = shape_gaussian(n, sigma)
        elif kind == "beta":
            alpha = np.random.exponential(1.0) + 0.05
            beta_par = np.random.exponential(1.0) + 0.05
            cand = shape_beta(n, alpha, beta_par)
        elif kind == "arcsine":
            cand = shape_arcsine(n)
        elif kind == "tri":
            cand = shape_triangle_left(n)
        else:  # dirichlet
            conc = np.random.uniform(0.2, 2.0)
            cand = shape_dirichlet(n, conc)
        pool.append(cand)

    # random Dirichlet draws for extra diversity
    for _ in range(6):
        pool.append(shape_dirichlet(n, conc=np.random.uniform(0.1, 1.0)))

    # ------------------------------------------------------------------
    # Keep the best warm‑start according to the true objective.
    # ------------------------------------------------------------------
    local_best_a = None
    local_best_sc = float("inf")
    for cand in pool:
        sc = score_of(cand)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best_a = cand
            print(
                f"[{time.time() - start_time:6.2f}s] warm‑start improve score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # Entropic mirror‑descent phase (soft‑max surrogate)
    # ------------------------------------------------------------------
    ent_deadline = time.time() + 0.45 * (deadline - time.time())
    a_ent, sc_ent = entropic_mirror_descent(
        local_best_a, n, ent_deadline, start_time, tau0=8.0 / n, eta0=0.7
    )
    if sc_ent < best_score:
        best_score = sc_ent
        best_seq = a_ent.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] post‑entropic global improve score={sc_ent:.6f}"
        )

    # ------------------------------------------------------------------
    # Hard‑max sub‑gradient polish (tiny temperature)
    # ------------------------------------------------------------------
    hard_deadline = time.time() + 0.20 * (deadline - time.time())
    a_hard, sc_hard = hard_subgradient_descent(a_ent, n, hard_deadline, start_time, eta0=0.1)
    if sc_hard < best_score:
        best_score = sc_hard
        best_seq = a_hard.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] post‑hard‑sgd global improve score={sc_hard:.6f}"
        )

    # ------------------------------------------------------------------
    # Final cheap annealing polish (mass‑transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_fin, sc_fin = annealing(
            a_hard,
            n,
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            steps=300000,
            start_temp=1e-1,
            end_temp=1e-6,
        )
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve score={sc_fin:.6f}"
            )

    # ------------------------------------------------------------------
    # Safety net – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain list for the external evaluator
    return [float(x) for x in best_seq]
