    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
    return conv

def subgradient_descent(a0, n, deadline, start_time, step0=0.5, beta=0.0, max_iter=10**7):
   a = a0.copy()
   best_a = a.copy()
   best_max = conv.max()
   momentum = np.zeros_like(a) if beta>0 else None
   it = 0
   while time.time() < deadline and it < max_iter:
      it+=1
      conv = conv_self_fft(a)
      max_val = conv.max()
      idx_max = conv.argmax()
      if max_val < best_max:
         best_max = max_val
         best_a = a.copy()
         # print progress
      # compute gradient
      j = idx_max - np.arange(n)
      mask = (j>=0) & (j<n)
      grad = np.zeros(n)
      grad[mask] = 2*a[j[mask]]
      if beta>0:
         momentum = beta * momentum + (1-beta) * grad
         grad_step = momentum

         grad_step = grad
      step = step0 / math.sqrt(it)

   sum_best = best_a.sum()
   if sum_best>0:
        best_a /= sum_best
   else:
        best_a = np.full_like(best_a, 1.0/n)

    a = a0.copy()
    best_a = a.copy()
    conv = conv_self_fft(a)
    best_max = conv.max()
    # compute initial cur_max
    cur_max = best_max
    momentum = np.zeros_like(a) if beta>0 else None

    it = 0
    while time.time() < deadline and it < max_iter:
        it+=1
        # compute conv
        conv = conv_self_fft(a)
        max_val = conv.max()
        idx_max = np.argmax(conv)

        if max_val < best_max:
            best_max = max_val
            best_a = a.copy()

        # compute gradient

        j_indices = idx_max - np.arange(n)
        mask = (j_indices >=0) & (j_indices < n)
        grad = np.zeros(n)

        # project onto simplex

    # final best
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s>0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0/n)
    return best_a, best_max

"""
Multi‑stage subgradient optimiser for the max‑convolution objective.

1.  Optional warm‑start from the global variable `height_sequence_1`.
2.  A sequence of up‑sampling stages (small → large n) gives a coarse‑to‑fine
    search space while keeping the runtime affordable.
3.  At each stage a projected sub‑gradient descent on the exact
    `max(conv(a,a))` is performed.  The gradient of the active
    convolution entry is `2 * shift_rev(a)`.  After every step the vector
    is projected onto the probability simplex (non‑negative, sum = 1).
4.  The best vector from the final stage is refined by a stochastic
    “peak‑shaving’’ local search that moves a small fraction of mass away
    from indices that contribute most to the currently maximal convolution
    entry.
5.  The returned list is normalised to sum = 1 (the evaluation function
    is scale‑invariant as long as the clamp limits are not hit).

The routine respects the `budget_s` time budget and prints progress
information useful for debugging.
"""

# Imports

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent on  max(conv)  (sum of a is kept = 1)
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.5,
                        beta: float = 0.0,
                        max_iter: int = 10 ** 7):
    """
    Projected (optionally momentum‑augmented) sub‑gradient descent
    for the objective  max_k (a*a)[k]  with a ∈ simplex.
    Returns the best vector seen and the associated max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = conv_self_fft(a)
    best_max = conv.max()

    momentum = np.zeros_like(a) if beta > 0 else None
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = conv_self_fft(a)
        cur_max = conv.max()
        idx_max = int(conv.argmax())

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd iter={it:5d} "
                  f"n={n:5d} max={cur_max:.8f} score={2*n*cur_max:.6f}")

        # gradient of the active convolution entry
        j = idx_max - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[j[mask]]

        if beta > 0:
            momentum = beta * momentum + (1.0 - beta) * grad
            step_dir = momentum
        else:
            step_dir = grad

        step = step0 / math.sqrt(it)
        a = a - step * step_dir
        a = project_simplex(a)

    # final projection / normalisation
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    return best_a, best_max

# Targeted stochastic local search (peak‑shaving)
# ----------------------------------------------------------------------
def targeted_local_search(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float):
    """
    Repeatedly move a small fraction of mass from indices that contribute
    heavily to the current maximal convolution entry to indices that
    contribute little.  A simple Metropolis rule with a cooling schedule
    enables occasional uphill moves.
    Returns the best vector seen and its maximal convolution entry.
    """
    a = a0.copy()
    conv = conv_self_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature schedule (log‑linear)
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        # recompute convolution and locate the maximal entry
        conv = conv_self_fft(a)
        idx_max = int(conv.argmax())
        cur_max = conv.max()

        # contribution of each index to this entry (a_i * a_{idx_max-i})
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # pick a donor proportionally to its contribution
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # pick a recipient from the low‑contribution tail
        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_idxs = np.where(contrib <= low_thr)[0]
        if low_idxs.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idxs)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a fraction of the donor mass (up to 30 %)
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        # evaluate the proposal
        new_conv = conv_self_fft(a_prop)
        new_max = new_conv.max()
        delta = new_max - cur_max

        # temperature based acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n} score={2*n*new_max:.6f}")

    # final normalisation (should already sum to 1)
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    return best_a, best_max

    descent and a final stochastic local refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage configuration (progressive up‑sampling)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2000, 4000, 8000]   # final length ≈ 8000
    total_len_weight = sum(stage_lengths)

    total_time = deadline - start_time
    grad_budget = total_time * 0.70          # 70 % for gradient descent
    local_budget = total_time - grad_budget  # remaining for local search

    best_seq = None
    best_max = float("inf")

    # --------------------------------------------------------------
    # Gradient descent phase (projected sub‑gradient)
    # --------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining gradient budget
        stage_frac = n / total_len_weight
        stage_time = max(2.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this length
        if best_seq is not None:
            a0 = upsample(best_seq, n)
        elif starter is not None:
            a0 = upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        a_opt, max_opt = subgradient_descent(
            a0, n, stage_deadline, start_time,
            step0=0.5, beta=0.7, max_iter=10 ** 7
        )

        if max_opt < best_max:
            best_max = max_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"max={max_opt:.8f} score={2*n*max_opt:.6f}")

        # update the remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # --------------------------------------------------------------
    # Targeted stochastic local search (peak‑shaving)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, max_loc = targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if max_loc < best_max:
            best_max = max_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"max={max_loc:.8f} score={2*n_final*max_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
