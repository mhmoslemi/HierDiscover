def _run_adam_l2(a0, n, slice_deadline, start_time, symmetrize=False):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    m=np.zeros_like(a); v=np.zeros_like(a)
    it=0
    while time.time()<slice_deadline:
        it+=1
        # compute gradient using L2
        grad = _grad_l2(a)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step_size = 0.5 / math.sqrt(it+1)  # adjust step size
        a = a - step_size*m_hat/(np.sqrt(v_hat)+eps)
        if symmetrize:

        cur_score = _objective(a)
        if cur_score < best_score:

            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam-L2 improve n={n} it={it} score={cur_score:.6f}")

"""
Hybrid L2‑flattening + max‑surrogate optimisation.

The search proceeds in stages of increasing length.  For each length
a warm‑started distribution is refined by:

1. Projected Adam on the L2 loss Σ(conv‑c)² (c = average convolution).
   This flattens the self‑convolution and dramatically reduces the
   ∞‑norm of the autocorrelation.
2. A short Adam run on the log‑sum‑exp surrogate of the max‑norm
   (the classic “smooth max’’) to polish the peak.
3. Light simulated‑annealing mass‑transfer polishing.

All updates stay on the probability simplex (Σaᵢ=1) so the evaluation
`2·n·max(conv)` is scale‑invariant.  The routine respects a 1000 s
budget and returns the best sequence found.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation used for logging (sum(a)=1 ⇒ 2·n·max(conv))."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# L2‑flattening Adam optimiser
# ----------------------------------------------------------------------


def _run_adam_l2(a0: np.ndarray,
                 n: int,
                 slice_deadline: float,
                 start_time: float,
                 symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising Σ(conv‑c)² where c=1/(2n‑1)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    c_target = 1.0 / (2 * n - 1)

    while time.time() < slice_deadline:
        it += 1

        # current self‑convolution and gradient of the L2 loss
        conv = _conv_self(a)
        diff = conv - c_target                     # length 2n‑1
        a_rev = a[::-1]
        grad_full = _conv_fft(diff, a_rev)        # length 3n‑2
        grad = 4.0 * grad_full[n - 1:n - 1 + n]   # central slice

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] L2‑Adam improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Log‑sum‑exp (smooth‑max) Adam optimiser (from the previous code)
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[n - 1:n - 1 + n]    # central slice
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] log‑SE Adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

# Simulated‑annealing polish
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % transfer
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation and renormalisation to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Produce a few plausible starts from `base_seq`."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # sinusoidal (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # random Dirichlet‑like
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants
