def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    
    # Determine n length
    n = 4800
    try: starter = np.asarray(height_sequence_1, dtype=float); if starter.ndim==1 and starter.size>0: ..., else: starter=None
    except: starter=None

    def evaluate(a):
        return evaluate_sequence(a.tolist())
    
    best_seq = None
    best_score = float('inf')
    
    # Phase 1: multiple random restarts with Adam+smooth
    # allocate 30% of time for restarts
    time_for_restarts = 0.3 * (deadline - start_time)
    restart_deadline = start_time + time_for_restarts
    restart_cnt = 0
    
    while time.time() < restart_deadline:
        restart_cnt += 1
        # initial distribution: random or upsample starter with some noise
        if restart_cnt == 1 and starter is not None:
            a0 = starter.copy()
            # add small random perturbation
            a0 += 0.01 * np.random.rand(n)
            a0 = np.maximum(a0, 0)
            a0 /= a0.sum()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        # run smoothed Adam
        seq, score = _run_adam_opt(a0, n, deadline)   # custom function with time check
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time()-start_time:.2f}s] restart {restart_cnt} improve: score={score:.6f}")
    
    # Phase 2: polishing via projected gradient descent
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.4 * remaining
        a_polish, score_polish = _run_gradient_descent(best_seq, n, polish_deadline, start_time, beta=0.94, step0=0.25)
        if score_polish < best_score:
           best_score = score_polish
           best_seq = a_polish.copy()
           print(f"[{time.time()-start_time:.2f}s] polishing improve: score={score_polish:.6f}")
    
    # Phase 3: long annealing
    if time.time() < deadline:
        a_final, score_final = _run_annealing(best_seq, n, deadline, start_time, best_score, best_seq,
                                             start_temp=1e-1, end_temp=1e-8, steps=int(5e5))
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time()-start_time:.2f}s] final anneal improve: score={score_final:.6f}")
    
    # final normalization
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0: best_seq /= total
    else: best_seq = np.full_like(best_seq, 1/n)
    print(f"Best score achieved: {best_score:.6f}")

"""
Enhanced search for a non‑negative step‑function that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – use the length of the supplied starter
   (``height_sequence_1``) or fall back to n = 4800.

2. **Soft‑max (log‑domain) parameterisation** – the vector ``a`` lives on the
   probability simplex.  Gradients are computed w.r.t. the logits ``y``.

3. **Smoothed objective** – minimise a smoothed version of the
   max‑convolution using a log‑sum‑exp (parameter ``α``) and Adam
   optimisation.  The schedule of ``α`` is prolonged and the number of
   iterations per value is increased compared with the reference code.

4. **Exact sub‑gradient polishing** – short projected‑gradient descent on the
   true max‑convolution (momentum, Polyak step‑size).

5. **Simulated‑annealing fine‑tuning** – mass‑transfer moves with a much
   higher starting temperature and a slower cooling schedule, allowing the
   search to escape shallow local minima.

6. **Time budgeting** – the total runtime is split 25 % Adam,
   35 % polishing, 40 % annealing (the exact split adapts to the remaining
   time).  All phases stop as soon as the deadline is reached.

The routine returns the best sequence found (normalised to sum = 1) before the
deadline; if the time budget expires the current best is still returned,
avoiding the “negative‑infinity” penalty.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _run_adam_opt(init_a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective.
    Returns (best_vector, best_score) where score = evaluate_sequence().
    """
    start_time = time.time()
    eps = 1e-16
    y = np.log(np.maximum(init_a, eps))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – a longer list and more iterations per α
    alpha_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    iters_per_alpha = 8000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = init_a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # ----- softmax -----
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            sum_exp = exp_y.sum()
            a = exp_y / sum_exp

            # ----- convolution & smoothed obj -----
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            # stable log‑sum‑exp
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val  # exact objective (not used for gradient)

            # ----- gradient w.r.t. a -----
            grad_a = 2.0 * n * _grad_smooth_max(a, conv, alpha, fft_len)

            # ----- gradient w.r.t logits y (mirror‑descent form) -----
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- occasional exact evaluation -----
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the true max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # --- sub‑gradient of max(conv) ---
        max_val = conv.max()
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        # Polyak‑type step size (more aggressive than 1/√t)
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-1,
                   end_temp: float = 1e-8,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that transfers a random ε‑fraction of mass from a
    source index to a destination index.  A higher starting temperature and
    a slower cooling schedule give the algorithm more freedom to escape
    local minima.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score
