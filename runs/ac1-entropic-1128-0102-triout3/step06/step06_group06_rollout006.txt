"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. Choose a fairly large length (default: 8192).  Larger n gives a lower
   theoretical bound (≈2·n/(2·n‑1) → 1) but still allows fast FFT
   convolutions.
2. Generate many random “warm‑starts’’ from several analytic families
   (uniform, Gaussian, Beta, arcsine, product‑power) and keep the best
   according to the exact objective.
3. Optimise the chosen start with an entropic (soft‑max) mirror‑descent
   on the simplex:
      * soft‑max temperature τ starts relatively large (0.05) and is
        annealed geometrically to a very small value (1e‑6).
      * the gradient of the smooth surrogate is obtained via a second
        FFT‑based convolution.
      * updates are performed with an exponentiated‑gradient (KL‑projection)
        step size η∝1/√t; a tiny Adam‑like momentum correction is also
        applied for robustness.
4. When time runs low, finish with a cheap simulated‑annealing polishing
   that repeatedly transfers a random fraction of mass between two indices.
5. Return the best vector found (normalised to sum‑1).  All values are
   non‑negative, satisfying the checks in `evaluate_sequence`.

The implementation respects the 1000 s time budget and uses only NumPy
(and the standard library).  No file or network I/O is performed.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Return the smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Parametric families for warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_random(n: int) -> np.ndarray:
    """Simple Dirichlet‑draw (uniform random on the simplex)."""
    a = np.random.rand(n)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Entropic mirror‑descent (with optional Adam‑like momentum)
# ----------------------------------------------------------------------


def entropic_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) using a smooth log‑sum‑exp surrogate.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule
    tau = 0.05                     # fairly large start → smooth surrogate
    tau_end = 1e-6                 # final near‑exact max
    decay = 0.9995

    # Adam‑style accumulators (optional, improve stability)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # base step for exponentiated‑gradient
    eta0 = 0.5

    while time.time() < deadline:
        t_iter += 1

        # ---- exact score (for tracking) ----
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if verbose:
                print(
                    f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={t_iter} score={cur_sc:.6f}"
                )

        # ---- soft‑max weights ----
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum <= 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate ----
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                 # length = 3n‑2
        idx = (n - 1) - np.arange(n)              # extract the needed slice
        grad = 2.0 * corr[idx]

        # ---- Adam‑style momentum ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # ---- exponentiated‑gradient update ----
        step = eta0 / math.sqrt(t_iter)
        update = m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step * update)

        # ---- project back onto the simplex (ensures non‑negativity) ----
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---- anneal temperature ----
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    max_steps: int = 60000,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, a.shape[0], size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            if verbose:
                print(
                    f"[{time.time() - start_time:6.2f}s] anneal improve n={a.shape[0]} it={it} score={score:.6f}"
                )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a Python list of heights.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a working length.
    # 8192 fits comfortably into FFT‑based convolutions and gives a good
    # theoretical bound (~1.0002).  If the remaining time is tiny we fall
    # back to a smaller size.
    # ------------------------------------------------------------------
    n = 8192
    if deadline - start_time < 30.0:   # not enough time → smaller problem
        n = 2048

    # ------------------------------------------------------------------
    # Warm‑start: explore several analytic families quickly.
    # ------------------------------------------------------------------
    best_a = None
    best_sc = float("inf")
    warm_deadline = time.time() + min(5.0, (deadline - time.time()) * 0.1)

    families = [
        lambda: shape_product_power(n, random.uniform(-2.0, 3.0)),
        lambda: shape_gaussian(n, random.uniform(n / 30.0, n / 2.0)),
        lambda: shape_beta(
            n,
            random.exponential(1.0) + 0.05,
            random.exponential(1.0) + 0.05,
        ),
        lambda: shape_arcsine(n),
        lambda: shape_random(n),
    ]

    while time.time() < warm_deadline:
        cand = random.choice(families)()
        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            # progress output – not required but helpful
            print(
                f"[{time.time() - start_time:6.2f}s] warm‑start improve n={n} score={sc:.6f}"
            )

    # Fallback to uniform if nothing found (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Entropic mirror‑descent optimisation (the heavy part)
    # ------------------------------------------------------------------
    grad_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.75)
    opt_a, opt_sc = entropic_descent(
        best_a, grad_deadline, start_time, verbose=True
    )

    if opt_sc < best_sc:
        best_sc = opt_sc
        best_a = opt_a.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] post‑entropic improve n={n} score={opt_sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Final polishing with cheap simulated‑annealing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        polished_a, polished_sc = annealing(
            best_a,
            anneal_deadline,
            start_time,
            best_sc,
            best_a,
            start_temp=5e-2,
            end_temp=1e-5,
            max_steps=80000,
            verbose=True,
        )
        if polished_sc < best_sc:
            best_sc = polished_sc
            best_a = polished_a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve n={n} score={polished_sc:.6f}"
            )

    # ------------------------------------------------------------------
    # Return as plain Python list (the external evaluator will clip/normalise)
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
