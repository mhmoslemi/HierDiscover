"""
Advanced search for a non‑negative step function that minimises
`evaluate_sequence`.  The method consists of three stages:

1. **Fast parametric warm‑starts** – random sampling of a few
   low‑dimensional families (product‑power with shift, Beta,
   arcsine‑uniform mixtures, powered‑arcsine).  The best shape for a
   modest size (n≈512) is kept.

2. **Entropic mirror‑descent with momentum** – a smooth surrogate
   τ·log∑exp(conv/τ) of the ∞‑norm is minimised using exponentiated‑gradient
   updates on the probability simplex.  Momentum (exponential moving
   average of the gradient) and an adaptive step‑size improve convergence
   over the plain version.

3. **Mass‑transfer simulated annealing polish** – a cheap move‑based
   annealer transfers a random ε from one index to another, accepting
   uphill moves with a geometric temperature schedule.

The algorithm runs a length schedule (512 → 1024 → 2048 → 4096 → 8192)
and up‑samples the current best vector before each new length.
All operations respect the non‑negativity and unit‑sum constraints
required by `evaluate_sequence`.  Progress and intermediate scores are
printed for monitoring.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant, assumes sum(a)=1)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families (low‑dimensional)
# ----------------------------------------------------------------------


def shape_product_power_shifted(n: int, p: float, shift: float) -> np.ndarray:
    """
    a_i ∝ ((i+0.5)*(n-i-0.5) + shift) ** p .
    Works for negative and positive p.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i) + shift
    # protect against non‑positive terms when p is not integer
    term = np.maximum(term, 1e-12)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_product_power(n: int, p: float) -> np.ndarray:
    """Legacy family without shift (kept for completeness)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.maximum(term, eps)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.maximum(term, eps)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_powered_arcsine(n: int, q: float) -> np.ndarray:
    """
    Raise the arcsine shape to the power q (0<q≤1) and renormalise.
    q<1 flattens the heavy edges.
    """
    base = shape_arcsine(n)
    # Avoid exact zeros – add a very small epsilon before power
    eps = 1e-12
    a = (base + eps) ** q
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_mix_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """
    Convex combination of arcsine and uniform distribution.
    lam=0 → pure arcsine, lam=1 → pure uniform.
    """
    arcs = shape_arcsine(n)
    uniform = np.full(n, 1.0 / n)
    a = (1.0 - lam) * arcs + lam * uniform
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Entropic mirror‑descent with momentum on the simplex
# ----------------------------------------------------------------------


def entropic_mirror_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimises `max(a*a)` (equivalently the exact objective)
    via an entropic (soft‑max) surrogate.
    Uses exponentiated‑gradient updates with momentum.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule (start relatively high, decay towards a tiny value)
    tau = 5.0 / n          # initial temperature
    tau_end = 1e-6 / n     # final temperature – essentially hard max
    decay = 0.9996

    # step‑size schedule (initial η)
    eta0 = 0.7
    it = 0

    # momentum parameters
    mom = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline:
        it += 1

        # ----- exact objective (for bookkeeping) -----
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ----- soft‑max weights -----
        shift = conv.max()
        # use stable soft‑max
        w = np.exp((conv - shift) / max(tau, 1e-12))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fall back to one‑hot at the current maximum index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the smooth surrogate -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                      # length = 3n‑2
        idx = (n - 1) - np.arange(n)                  # extraction indices
        grad = 2.0 * corr[idx]

        # ----- momentum update -----
        mom = beta * mom + (1.0 - beta) * grad

        # ----- exponentiated‑gradient (mirror‑descent) -----
        step = eta0 / math.sqrt(it)
        expo = -step * mom
        # protect against overflow
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)

        # ----- project back onto the probability simplex -----
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- anneal temperature -----
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing_polish(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    max_steps: int = 120000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing that moves a random ε from one index to another.
    Accepts uphill moves according to a geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # transfer up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights (the caller will normalise/clamp).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # Give a small safety margin before the hard deadline.
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from previous run (global variable height_sequence_1)
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            starter = tmp / s
    except Exception:
        starter = None

    # Length schedule – start small (cheap exploration) and grow.
    lengths = [512, 1024, 2048, 4096, 8192]
    random.shuffle(lengths)  # avoid systematic bias if we stop early

    # We'll keep the best vector encountered so far and up‑sample it for
    # the next length.
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Allocate a slice of the remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # Give a modest share to early lengths, more to later (larger) ones.
        share = 0.12 / (len(lengths) - idx)
        slice_deadline = time.time() + max(8.0, remaining * share)

        # --------------------------------------------------------------
        # 1️⃣  Fast parametric warm‑starts (small n = 512)
        # --------------------------------------------------------------
        # For very small n we can afford a dense random search over

        n_small = min(512, n)
        shape_deadline = time.time() + min(2.0, slice_deadline - time.time() * 0.2)
        local_best_a = shape_arcsine(n_small) if n_small == n else a0[:n_small].copy()
        local_best_sc = score_of(local_best_a)

        while time.time() < shape_deadline:

            if kind == 0:  # product‑power with shift

            elif kind == 2:  # Gaussian

                cand = shape_gaussian(n_small, sigma)
            elif kind == 3:  # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n_small, alpha, beta_par)
            elif kind == 4:  # powered arcsine (0 < q ≤ 1)
                q = np.random.uniform(0.2, 1.0)
                cand = shape_powered_arcsine(n_small, q)
            else:  # mixture arcsine‑uniform
                lam = np.random.uniform(0.0, 0.6)
                cand = shape_mix_arcsine_uniform(n_small, lam)

            sc = score_of(cand)
            if sc < local_best_sc:

        if time.time() < deadline:

            a_ref, sc_ref = annealing_polish(

            )
