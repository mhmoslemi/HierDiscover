"""
Propose a non‑negative step‑function (list of heights) that minimizes the
given ``evaluate_sequence`` objective.

Algorithm
---------

1. **Initialisation** – start from the previously discovered sequence
   (global ``height_sequence_1``) if available, otherwise from a random
   probability vector.  Down‑sample it to a modest size (≈ 1024) to keep the
   linear programmes tractable.

2. **Linear‑programming refinement** – repeatedly solve the LP

       maximise   Σ g_j
       subject to  (f * g)_k ≤ M   (for all k)
                   g_j ≥ 0

   where ``f`` is the current vector, ``M = max(f*f)`` and ``*`` denotes
   discrete convolution.  This LP finds a direction that can increase the
   total mass while keeping the convolution bound fixed.  After solving we
   rescale the solution to unit sum, form a convex combination
   ``f←(1‑t)f + t·g`` (with a back‑tracking line‑search on ``t``) and repeat.
   The procedure is guaranteed to lower the objective for sufficiently
   small ``t``.

3. **Upsampling** – linearly interpolate the refined vector to the target
   length (4800).  The shape of the distribution is preserved, and the
   objective scales almost invariantly with the length.

4. **Fine polishing** – a short projected sub‑gradient descent on the true
   objective ``2·n·max(conv)`` (with Euclidean projection onto the simplex)
   is applied to squeeze out any remaining slack.

All phases respect the overall 1000 s time budget.  The final list of
heights is guaranteed to be non‑negative and renormalised to sum to one,
so ``evaluate_sequence`` can be called safely.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all near‑max entries."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (used for a final polish)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    """
    Classic (momentum‑accelerated) projected sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  Returns the best vector encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score

            break

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 0:
            print(f"[{time.time() - start_time:.2f}s] LP returned zero mass – stopping.")
            break

        # Rescale to unit sum (the LP was solved without any sum constraint)
        g = g_raw / sum_g

        # ------------------------------------------------------------------
        # Convex combination step with simple back‑tracking on t
        # ------------------------------------------------------------------
        improved = False
        t = t_initial
        while t > 1e-4:
            candidate = (1.0 - t) * best_seq + t * g
            cand_conv = _conv_self(candidate)
            cand_score = 2.0 * n * cand_conv.max() / (candidate.sum() ** 2)
            if cand_score < best_score_small - 1e-12:
                best_seq = candidate
                best_score_small = cand_score
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter + 1}, t={t:.3f}, "
                      f"score={cand_score:.6f}")
                break
            t *= 0.5

        if not improved:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {lp_iter + 1} gave no improvement – aborting.")
            break

        lp_iter += 1
        # gradually shrink the step size for later iterations
        t_initial = max(t_initial * 0.9, 0.05)

    # --------------------------------------------------------------
    # Phase 3 – Upsample to the final length
    # --------------------------------------------------------------
    seq_large = _upsample(best_seq, final_n)
    conv_large = _conv_self(seq_large)
    best_score = 2.0 * final_n * conv_large.max() / (seq_large.sum() ** 2)
    best_seq = seq_large
    print(f"[{time.time() - start_time:.2f}s] after up‑sampling (n={final_n}) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Short projected sub‑gradient polish (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:          # need a few seconds to be worthwhile
        sub_deadline = time.time() + min(remaining * 0.9, 300.0)   # cap at ~5 min
        a_sub, sub_score = _run_subgradient_descent(
            best_seq,
            final_n,
            sub_deadline,
            start_time,
            step0=0.25,
            beta=0.94,
        )
        # sub_score already equals 2·n·max(conv)
        if sub_score < best_score:
            best_seq = a_sub
            best_score = sub_score
            print(f"[{time.time() - start_time:.2f}s] final sub‑grad improve: score={sub_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return (ensuring the harness sees a proper list)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    return [float(x) for x in best_seq]
