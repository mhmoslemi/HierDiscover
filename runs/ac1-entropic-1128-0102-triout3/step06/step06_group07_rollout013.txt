"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm works in three phases:

1. **Smoothed Adam descent** – the non‑smooth objective
   `max(conv(a))` is approximated by a log‑sum‑exp (smooth max).  The
   gradient of this surrogate is computed analytically via FFT‑based
   convolutions and an Adam optimiser with projection onto the simplex
   drives the search.

2. **Upsampling** – the best vector from a moderate length (≈4096) is
   linearly interpolated to a larger length (≈12288) and the Adam stage
   is repeated.  Upsampling keeps the overall shape while allowing a
   finer discretisation.

3. **Fine‑grained annealing** – a cheap mass‑transfer simulated‑annealing
   refines the final vector.

All stages respect the overall time budget (default 1000 s).  The best
candidate found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input)
# ----------------------------------------------------------------------
def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Return linear convolution of real vectors u and v using a real FFT."""
    size = u.shape[0] + v.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()   # next power of two
    U = np.fft.rfft(u, n=nfft)
    V = np.fft.rfft(v, n=nfft)
    conv = np.fft.irfft(U * V, n=nfft)[:size]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Score: 2·n·max(conv)  (equal to evaluate_sequence when Σa=1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Smoothed Adam descent (log‑sum‑exp surrogate for max)
# ----------------------------------------------------------------------
def _run_smooth_adam(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lr: float = 0.08,
                     alpha_start: float = 1.0,
                     alpha_end: float = 500.0) -> tuple[np.ndarray, float]:
    """
    Minimise the true objective using Adam on a smoothed approximation.
    Returns the best vector encountered and its true score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # pre‑allocate reversed a for gradient computation (updated each iter)
    while time.time() < deadline:
        it += 1

        # ---- true score for logging ----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ---- schedule for smoothing parameter α ----
        t_frac = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha_start + t_frac * (alpha_end - alpha_start)

        # ---- smoothed max (log‑sum‑exp) weights ----
        maxc = conv.max()
        # shift to avoid overflow
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()               # shape (2·n‑1,)

        # ---- gradient of smoothed max ----
        a_rev = a[::-1]
        conv_wr = _fft_convolve(w, a_rev)           # length 3·n‑2
        grad = 2.0 * conv_wr[n - 1:n - 1 + n]        # slice of length n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Light annealing that moves ε mass from a source index to a random
    destination.  The temperature decays geometrically.
    """
    a = a0.copy()
    score = _score(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # choose a source proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
