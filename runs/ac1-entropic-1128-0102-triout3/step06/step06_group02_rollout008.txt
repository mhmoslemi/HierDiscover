"""
Hybrid optimiser for the step‑function max‑convolution problem.

The algorithm works in three passes:

1. **Multiscale warm‑start** – optimise on a short vector,
   then repeatedly up‑sample (linear interpolation) to a longer one,
   keeping the mass normalised.

2. **Smooth Adam descent** – minimise a temperature‑smoothed
   soft‑max of the convolution using the Adam optimiser on the simplex.
   The temperature is exponentially cooled so that the objective
   gradually approaches the true max‑convolution.

3. **Exact sub‑gradient + long annealing** – switch to the true
   max‑convolution sub‑gradient (again with Adam) to sharpen the solution,
   then perform a Metropolis simulated‑annealing walk that moves tiny
   fractions of mass between two positions.

All updates respect the non‑negativity and sum‑to‑one constraints
by projection onto the probability simplex.  The routine respects the
time budget (default 1000 s) and returns the best sequence it finds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (full length = 2·len(a)−1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from `seq` (length m) to `new_n` points,
    followed by normalisation to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Soft‑max weights w_i = exp((c_i‑max)/T) / Σ exp((c‑max)/T)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    return w


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the temperature‑smoothed max‑convolution w.r.t. `a`.
    Implements  2·rev(a)  *  w   (central n entries of the convolution).
    """
    w = _softmax_weights(conv, temperature)
    rev_a = a[::-1]

    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the probability simplex
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    *,
    base_lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.2,
    temp_end: float = 5e-4,
    max_iter: int = 10 ** 7,
    use_softmax: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the (soft‑)max‑convolution objective.

    If *use_softmax* is True, the smoothed max (soft‑max) is used with a
    temperature schedule; otherwise the exact max (hard) is used.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"{'softmax' if use_softmax else 'subgrad'} improve: "
                f"n={n}, iter={it}, score={cur_score:.6f}"
            )

        # temperature schedule (only matters for the soft‑max case)
        if use_softmax:
            # exponential cooling
            progress = min(1.0, it / 60000.0)          # ~60k it → final temperature
            T = temp_start * (temp_end / temp_start) ** progress
            grad = _softmax_gradient(a, conv, T)
        else:
            # exact sub‑gradient of max(conv)
            k = conv.argmax()
            grad = np.zeros_like(a)
            if k < n:
                grad[: k + 1] = 2.0 * a[k::-1]
            else:
                start = k - n + 1
                grad[start:] = 2.0 * a[start:][::-1]

        # Adam update (move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (math.sqrt(it) + 1.0)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    *,
    start_temp: float = 0.2,
    end_temp: float = 1e-7,
    max_steps: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Metropolis walk moving a small fraction of mass between two positions.
    """
    a = a0.copy()
    score = cur_best_score
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 10 % of the mass at i
        eps = a[i] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n}, iter={it}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The optimiser works on a hierarchy of resolutions, using Adam on a
    smoothed max‑convolution, followed by exact sub‑gradient Adam and a
    long annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Initialise from a previous run if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:                     # noqa: BLE001
        starter = None

    # ------------------------------------------------------------------
    # Multiscale schedule – progressively larger vectors
    # ------------------------------------------------------------------
    stage_ns = [1024, 2048, 4096, 8192, 9600]   # final length matches the historic best
    total_weight = sum(stage_ns)
    remaining = deadline - start_time

    # allocate ~15 % of total time for the final annealing phase
    anneal_budget = max(30.0, 0.15 * remaining)
    grad_budget = remaining - anneal_budget

    for n in stage_ns:
        if time.time() >= deadline:
            break

        # portion of the gradient budget for this stage
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this resolution
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ---- Phase 1: smooth Adam descent (soft‑max) ----
        a_opt, score_opt = _adam_optimize(
            a0,
            n,
            stage_deadline,
            start_time,
            base_lr=0.6,
            temp_start=0.2,
            temp_end=5e-4,
            use_softmax=True,
        )

        # ---- Phase 2: exact sub‑gradient Adam descent ----
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _adam_optimize(
                a_opt,
                n,
                stage_deadline,
                start_time,
                base_lr=0.1,
                use_softmax=False,
                temp_start=1e-6,  # unused when use_softmax=False
                temp_end=1e-6,
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # keep the best across stages
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # recompute remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – uses the longest resolution reached.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline  # whole remaining time
        best_seq, best_score = _annealing_refine(
            best_seq,
            n_final,
            anneal_deadline,
            start_time,
            best_seq,
            best_score,
            start_temp=0.2,
            end_temp=1e-7,
            max_steps=5_000_000,
        )
        print(
            f"[{time.time() - start_time:6.2f}s] final anneal result: score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
