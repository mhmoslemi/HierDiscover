def adam_optimize(a0, n, tau_init, tau_final, steps, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1  # base learning rate
    t = 0
    tau = tau_init
    # linear or exponential schedule for tau
    for iteration in range(steps):
        if time.time() >= deadline:
            break
        t += 1
        # compute conv and max
        conv = conv_self(a)
        cur_sc = 2 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve n={n} it={t} score={cur_sc:.6f}")

        # compute w (softmax)
        shift = conv.max()
        # tau may become very small; ensure not zero
        # if tau < 1e-12: treat as hard max: w = indicator vector for max positions.
        if tau < 1e-12:
            # w = unit mass at max positions
            w = np.zeros_like(conv)
            w[conv == shift] = 1.0
            w_sum = w.sum()
            w = w / w_sum
        else:
            w = np.exp((conv - shift) / tau)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0/conv.size)
                w_sum = 1.0
            w /= w_sum

        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)  # length 3n-2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update

        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

```python
import numpy as np
import random
import math
import time

def _fft_len_needed(m):
    return 1 << ((m - 1).bit_length())

def conv_self(a):
    n = a.shape[0]
    m = 2*n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv

def conv_fft(u, v):
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv

def upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero?
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def score_of(a):
    """Exact objective for a normalized vector (sum = 1). Returns 2*n*max(conv)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()

# ------- parametric families -------------------------------------------------
def shape_product_power(n, p):
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0/n)
    return a

def shape_gaussian(n, sigma):
    i = np.arange(n)
    centre = (n-1)/2.0
    a = np.exp(-0.5 * ((i-centre)/sigma)**2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0/n)
    return a

def shape_beta(n, alpha, beta):
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps)**(alpha-1)) * (np.maximum(1.0-x, eps)**(beta-1))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum

    return a

def shape_arcsine(n):

    return a

def adam_smooth_descent(a0, n, deadline, start_time,
                        tau_start=0.1, tau_end=1e-6,
                        max_iter=200000, eta=0.2):
    """
    Minimise max self‑convolution using a soft‑max surrogate.
    Optimisation performed with Adam + projection onto the simplex.
    Returns the best sequence (numpy array) and its exact score.
    """
    a = a0.copy()
    # Ensure a sums to 1
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score_of(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    tau = tau_start
    decay = 0.9995  # geometric cooling of tau
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # current objective
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_sc:.6f}")

        # soft‑max weights for smooth approximation
        shift = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[conv == shift] = 1.0
            w_sum = w.sum()
            w = w / w_sum
        else:
            w = np.exp((conv - shift) / tau)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0/conv.size)
                w_sum = 1.0
            w /= w_sum

        # gradient of smoothed objective
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)    # length = 3n‑2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update (minimise → move opposite to gradient)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # Project back onto simplex
        a = project_onto_simplex(a)

        # Anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc

def annealing_polish(a0, n, deadline, start_time,
                     start_temp=5e-2, end_temp=1e-5,
                     steps=60000):
    """
    Simple mass‑transfer simulated annealing. Returns best sequence and score.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = a.copy()
    best_score = score
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = score_of(a_prop)
        delta = new_score - score
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")
    return best_seq, best_score

"""
Low‑maximum self‑convolution search.

The routine proceeds through a schedule of increasing lengths.  For each
length it generates a handful of analytic warm‑starts (product‑power,
Gaussian, Beta, arcsine), keeps the best, and then refines it with a
smooth‑max surrogate minimised by Adam + simplex‑projection (an
entropic‑mirror‑descent variant with adaptive learning rates).  After the
gradient phase a cheap mass‑transfer simulated‑annealing polish is run.
All operations stay on the probability simplex, so the final vector can
be returned directly to `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (self‑ and cross‑)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) via a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def score_of(a: np.ndarray) -> float:
    """
    Exact evaluation for a normalised vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()

# Analytic families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

# Optimisation core – Adam on a smooth max‑surrogate
# ----------------------------------------------------------------------


def adam_smooth_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    tau_start: float = 0.1,
    tau_end: float = 1e-7,
    max_iter: int = 200_000,
    eta: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Minimise max self‑convolution via a soft‑max surrogate.
    The surrogate  τ·log∑exp(conv/τ) is differentiable;
    Adam updates are performed on the probability simplex.
    Returns the best vector (numpy array) seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score_of(a)

    # Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    tau = tau_start
    decay = 0.9995

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # current objective (exact)
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_sc:.6f}"
            )

        # smooth weights (soft‑max)
        shift = conv.max()
        if tau < 1e-12:
            w = np.zeros_like(conv)
            w[conv == shift] = 1.0
            w /= w.sum()
        else:
            w = np.exp((conv - shift) / tau)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
                w_sum = 1.0
            w /= w_sum

        # gradient of the smooth surrogate
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)               # length = 3n‑2
        idx = (n - 1) - np.arange(n)           # pull out the n required entries
        grad = 2.0 * corr[idx]

        # Adam update (minimise → move opposite to gradient)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_onto_simplex(a)

        # anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing_polish(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple move‑based annealing: transfer a random ε from one index to another.
    Accept moves that improve the exact score or according to a geometric temperature schedule.
    Returns the best sequence found and its exact score.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = a.copy()
    best_score = score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The algorithm uses a length schedule, cheap analytic warm‑starts,
    Adam‑based smooth‑max descent and a final annealing polish.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional global starter from a previous run (if present)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – increase gradually.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)   # avoid systematic bias when stopping early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise vector for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Slice of the remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give each remaining length an almost equal share (minimum 8 s)
        slice_deadline = time.time() + max(8.0, remaining / (len(lengths) - idx))

        # --------------------------------------------------------------
        # Quick parametric warm‑starts
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3])
            if kind == 0:                     # product‑power family
                p = np.random.uniform(-2.0, 4.0)
                cand = shape_product_power(n, p)
            elif kind == 1:                   # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:                   # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            else:                             # arcsine‑like family
                cand = shape_arcsine(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Adam‑based smooth‑max descent
        # --------------------------------------------------------------
        # reserve most of the slice for optimisation, leave a bit for polishing
        remaining_slice = slice_deadline - time.time()
        grad_deadline = time.time() + remaining_slice * 0.55

        a_opt, sc_opt = adam_smooth_descent(
            local_best_a,
            n,
            grad_deadline,
            start_time,
            tau_start=0.1,
            tau_end=1e-7,
            max_iter=200_000,
            eta=0.3,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after Adam n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # Cheap annealing polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            # use the remainder of the slice for polishing
            anneal_deadline = time.time() + remaining_slice * 0.25
            a_ref, sc_ref = annealing_polish(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80_000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # Prepare up‑sample for the next length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list; the external evaluator will clip/normalise.
    return [float(x) for x in best_seq]
