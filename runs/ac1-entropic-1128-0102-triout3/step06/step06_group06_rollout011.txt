Let's implement project_onto_simplex:

def project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # ensure v is array
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Low‑max‑self‑convolution step‑function search (improved).

The algorithm builds on the previous pipeline but adds a
second‑order fine‑tuning stage:
  * Rich random warm‑starts from several analytic families.
  * Entropic (soft‑max) mirror‑descent on the simplex.
  * A dedicated sub‑gradient descent that targets the true ∞‑norm
    (binary max‑mask instead of a soft‑max) and uses an Euclidean
    projection onto the probability simplex.
  * A cheap mass‑transfer simulated‑annealing polish.

The sub‑gradient stage often removes the remaining few‑percent
gap left by the entropic phase and is fast thanks to FFT‑based
gradients.  The whole search stops after the allotted budget
(≈ 1000 s) and returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Parametric families for quick warm‑starts

def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Entropic mirror‑descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def entropic_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Minimise max(a*a) via a softened surrogate and exponentiated‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule – start relatively high, end very low
    tau = 5.0 / n                 # start temperature
    tau_end = tau * 1e-3          # final temperature
    decay = 0.9995                # exponential decay factor

    eta0 = 0.5                    # initial step‑size scaling
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---- soft‑max weights ----
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate ----
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)        # length = 3n‑2
        idx = (n - 1) - np.arange(n)     # extract the needed slice
        grad = 2.0 * corr[idx]

        # ---- exponentiated‑gradient update ----
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)   # avoid overflow
        a = a * np.exp(expo)

        # ---- projection back onto the simplex ----
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---- anneal temperature ----
        tau = max(tau_end, tau * decay)

    return best_a, best_sc

# Sub‑gradient descent targeting the exact ∞‑norm
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Pure sub‑gradient method for min max(a*a).  Uses an
    Euclidean projection onto the simplex after each additive step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    step0 = 0.8  # initial step size (tuned empirically)

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        M = conv.max()
        cur_sc = 2.0 * n * M
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )

        # binary mask of (near‑)maximal convolution entries
        tol = max(1e-12, M * 1e-6)
        w = (conv >= M - tol).astype(float)

        # gradient = 2 * (w ⊛ a_rev)   (⊛ = cross‑correlation)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # diminishing additive step
        eta = step0 / math.sqrt(it)

        a_candidate = a - eta * grad
        a = project_onto_simplex(a_candidate)

    return best_a, best_sc

# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """Very cheap move‑based annealing: transfer a random ε from one index to another."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2  # up to 20 % transfer
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
