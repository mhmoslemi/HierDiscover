"""
Propose a low‑value step‑function for ``evaluate_sequence``.

The algorithm works in four phases (all respect a 1000 s wall‑clock budget):

1. **Initialisation** – start from the globally supplied construction,
   otherwise build a few random Dirichlet vectors (different concentration
   parameters) and keep the best after a short exponential‑gradient warm‑up.

2. **Linear‑programming boost** – for the current vector `a` solve a small
   LP that maximises the ℓ₁‑mass under the current max‑convolution bound.
   The LP is built only for the most critical convolution indices
   (the largest entries).  The solution `b` is blended with `a` and
   projected back onto the probability simplex.  A few such LP steps
   dramatically lower the peak of the self‑convolution.

3. **Projected sub‑gradient descent** – classic momentum sub‑gradient
   optimisation of `f(a)=2·n·max(a*a)`.  The step size follows a 1/√t schedule
   and the iterate is projected onto the simplex after each update.
   Because we work with a modest dimension (≈2000‑3000) many millions of
   iterations fit into the remaining time budget.

4. **Final simulated‑annealing polish** – a cheap “mass‑transfer’’ anneal
   that can escape the last plateaus.

All helper routines are top‑level (no closures) and use only ``numpy`` and
``scipy.optimize.linprog``.  The routine returns a plain Python list of
non‑negative floats.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all entries within rel_tol."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase 1 – a quick exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)          # length for a⋆w_rev
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # smooth‑max gradient
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a /= a.sum()                       # stay on simplex

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – linear‑programming boost (few active constraints)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                max_iters: int = 4,
                max_constraints: int = 600) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP that maximises the ℓ₁‑mass while keeping
    the current max‑convolution bound `M`.  Only the `max_constraints`
    largest convolution entries are kept as constraints.
    """
    best_a = a.copy()
    best_score = float("inf")

    for lp_it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_it}: "
                  f"score={cur_score:.6f}")

        # pick the most critical convolution indices
        k = min(max_constraints, conv.size)
        active_idx = np.argpartition(-conv, k - 1)[:k]   # indices of top‑k entries

        rows = active_idx.size
        A_ub = np.zeros((rows, n))
        # fill each active constraint row
        for r, kk in enumerate(active_idx):
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            i_vals = np.arange(lo, hi + 1, dtype=int)
            j_vals = kk - i_vals
            A_ub[r, j_vals] = a[i_vals]

        # linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        b = res.x
        if b.sum() == 0:
            break
        b = b / b.sum()                     # keep total mass = 1

        # convex combination (t = 0.5 works well)
        t = 0.5
        a = _project_simplex((1.0 - t) * a + t * b)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    and a final anneal.
    """

    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 2400   # chosen to be fast enough for many iterations

    # --------------------------------------------------------------
    # Initialise – use the global height_sequence_1 if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:

            a0 = a0 / a0.sum()
        else:

    except Exception:

        alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
        a0 = np.full(n, 1.0 / n)
        for a in alphas:
            cand = np.random.gamma(a, 1.0, size=n)

    best_seq = a0.copy()
    best_score = float("inf")

        if sc_warm < best_score:

    # --------------------------------------------------------------

        a_sub, sc_sub = _run_subgradient_descent(
            best_seq,
            n,
            stage3_deadline,
            start_time,

            beta=0.94,

            print(f"[{time.time() - start_time:.2f}s] after sub‑grad: score={sc_sub:.6f}")

    # --------------------------------------------------------------
