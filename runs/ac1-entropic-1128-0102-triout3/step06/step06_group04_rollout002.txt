A multi‑scale, hybrid optimisation for the self‑convolution peak.

6. **Projected sub‑gradient descent** on the true max‑norm.
7. **Mass‑transfer simulated annealing** for final polishing.
8. **Iterate** over the length schedule, always keeping the best sequence

9. **Last heavy polish** – a final Adam run (with symmetry) followed by a

All operations stay on the simplex (∑aᵢ = 1, aᵢ ≥ 0) so the evaluation
``evaluate_sequence`` is scale‑invariant.  The routine respects the

import numpy as np

# ----------------------------------------------------------------------
# Simplex projection, convolution helpers and objective
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (sum(a)≈1). Returns 2·n·max(conv)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimisation on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central slice, same length as a
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the log‑sum‑exp surrogate.
    Returns the best sequence seen and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing λ – start relatively large, decay geometrically
    lam_factor = 0.5          # λ = lam_factor * max(conv)
    lam_decay = 0.7
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # decay λ from time to time
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.8 / math.sqrt(it + 1.0)   # a bit larger than before

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap” phase (classic LP attack)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A for constraints (f * g)[k] ≤ M.
    A_{k,i} = f_i if i ≤ k < i + n, else 0.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.3,
                    max_iters: int = 4,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """
    Perform a few LP‑based mass‑inflation steps.
    The LP maximises total mass under the current convolution bound.
    """
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 15.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur

def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            # contribution from pairs (i, j) with i+j = k
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            grad[i_range] += a[j_range]
            grad[j_range] += a[i_range]   # symmetric contribution

        step = 0.4 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 2e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing; always respects the simplex."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 10 % of source mass (a bit more aggressive)
        eps = a[i] * np.random.rand() * 0.10
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Return a collection of plausible starting vectors for a given length."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # sinusoidal (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # random Dirichlet (flat Dirichlet with concentration = 1)
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    # geometric families (exponential decay)
    for r in [0.90, 0.95, 0.97, 0.985, 0.995, 0.998, 0.999]:
        geo = (1.0 - r) * np.power(r, np.arange(n))
        s = geo.sum()
        if s > 0.0:
            geo /= s
        else:
            geo = np.full(n, 1.0 / n)
        variants.append(geo)

    # mirror of a geometric (mass concentrated at the right end)
    for r in [0.90, 0.95, 0.97, 0.985, 0.995, 0.998, 0.999]:
        geo = (1.0 - r) * np.power(r, np.arange(n))[::-1]
        s = geo.sum()
        if s > 0.0:
            geo /= s
        else:
            geo = np.full(n, 1.0 / n)
        variants.append(geo)

    return variants

            # occasionally enforce a mild symmetry (helps for some lengths)
            sym = (np.random.rand() < 0.25)
            a_opt, score_opt = _run_adam_opt(start_vec, n,
                                             slice_deadline,
                                             start_time,
                                             symmetrize=sym)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam local best n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # --------------------------------------------------------------
        # LP “swap” phase – a few cheap iterations
        # --------------------------------------------------------------
        lp_deadline = min(slice_deadline, time.time() + 7.0)
        best_local_a = _run_lp_improve(best_local_a, n,
                                       step=0.3,
                                       max_iters=4,
                                       deadline=lp_deadline,
                                       start_time=start_time)
        best_local_score = _objective(best_local_a)

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after LP n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient refinement (if time allows)
        # --------------------------------------------------------------
        if time.time() < deadline:
            sg_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.12)
            a_sg, score_sg = _subgradient_descent(best_local_a, n,
                                                  slice_deadline=sg_deadline,
                                                  start_time=start_time)
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad n={n} score={score_sg:.6f}")

        # --------------------------------------------------------------
        # Annealing polish for this length
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.18)
            a_ref, score_ref = _run_annealing(best_local_a, n,
                                              deadline=anneal_deadline,
                                              start_time=start_time,
                                              cur_best_score=best_score,
                                              cur_best_seq=best_seq,
                                              start_temp=2e-2,
                                              end_temp=1e-5,
                                              steps=180000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # keep the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Heavy final polish on the largest length we managed to keep.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        final_deadline = deadline - 0.5
        a_final, score_final = _run_adam_opt(best_seq, n,
                                             slice_deadline=final_deadline,
                                             start_time=start_time,
                                             symmetrize=True)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final Adam improve score={score_final:.6f}")

        if time.time() < deadline - 0.5:
            anneal_deadline = deadline - 0.2
            best_seq, best_score = _run_annealing(best_seq, n,
                                                  deadline=anneal_deadline,
                                                  start_time=start_time,
                                                  cur_best_score=best_score,
                                                  cur_best_seq=best_seq,
                                                  start_temp=1e-2,
                                                  end_temp=1e-5,
                                                  steps=250000)

    # ------------------------------------------------------------------
    # Fallback – always return a valid non‑negative sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the external evaluator will apply its clamps)
    return [float(x) for x in best_seq]
