def _conv_fft(x, y):
    L = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << ((L-1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:L]
    return conv

def _exp_update(a, grad, eta):
    # ensure grad is finite
    a_new = a * np.exp(-eta * grad)
    total = a_new.sum()
    if total <= 0:
        # fallback: add epsilon
        a_new = np.maximum(a_new, 1e-12)
        total = a_new.sum()
    return a_new / total

"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The algorithm mixes several optimisation ideas:

* **Randomised restarts** on a modest size (n≈1024) to explore many basins.
* **Exponentiated‑gradient descent** on the simplex using a smooth
  approximation of the max‑convolution (log‑sum‑exp).  The smoothing
  parameter α is increased gradually (continuation) and the learning
  rate decays as 1/√t.  This often finds better minima than pure
  sub‑gradient descent.
* **Upsampling** of the best vector to larger dimensions (≈4096, ≈12288)
  with linear interpolation and further refinement.
* **Fallback to the original projected‑gradient and annealing** code for
  fine‑tuning (the helpers from the previous solution are kept unchanged).
* The whole search respects a global time‑budget (default 1000 s).

The code is fully self‑contained, uses only `numpy`/`math`/`time` and
exposes a single entry point `propose_candidate`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    fft_len = 1 << ((L - 1).bit_length())
    Af = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Af * Af, fft_len)[:L]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors via real FFT (full length)."""
    L = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:L]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct diagonal handling.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent (Nesterov style) – kept from the previous
# solution for final polishing.
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Deterministic sine‑shaped starter (often a good basin)
# ----------------------------------------------------------------------
def _sine_start(n: int) -> np.ndarray:
    """Return a smooth sine‑like distribution on the simplex."""
    x = np.arange(n) + 0.5
    a = np.sin(np.pi * x / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex using a soft‑max
# approximation of the max‑convolution.
# ----------------------------------------------------------------------
def _run_exp_gradient_descent(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_start: float = 1.0,
                              alpha_end: float = 200.0,
                              eta0: float = 0.5,
                              beta: float = 0.9,
                              max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a*a) using a smooth (log‑sum‑exp) surrogate.
    The update is multiplicative (exponentiated gradient) which
    automatically respects non‑negativity; a simplex projection is
    performed by normalisation.
    """
    a = a0.copy()
    mom = np.zeros_like(a)          # momentum on the gradient (optional)

    best_a = a.copy()
    best_score = float("inf")
    step = 0

    while time.time() < deadline and step < max_steps:
        step += 1

        # --- evaluate current objective ---
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, "
                  f"iter={step:,}, score={cur_score:.6f}")

        # --- smooth max (log‑sum‑exp) ---
        # α is increased linearly from α_start to α_end
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, step / max_steps)

        # stabilise exponentials
        diff = conv - max_val                # ≤ 0
        w = np.exp(alpha * diff)             # all ≤ 1
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum                     # normalized weights (soft max)

        # --- gradient of the smooth surrogate ---
        # grad_i = 2 * Σ_k p_k * a_{k-i}
        rev_a = a[::-1]
        conv_rev = _conv_fft(p, rev_a)       # length = (2n-1)+n-1 = 3n-2
        grad = 2.0 * conv_rev[n - 1:2 * n - 1]

        # optional momentum on the (additive) gradient before exponentiation
        mom = beta * mom + (1.0 - beta) * grad

        # learning rate decreasing as 1/√t
        eta = eta0 / math.sqrt(step)

        # --- exponentiated (multiplicative) update ---
        a = a * np.exp(-eta * mom)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.maximum(a, 1e-12)
            s = a.sum()
        a = a / s

    return best_a, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = 12288
        a2 = _upsample(best_seq, n2)

        run_deadline = time.time() + min(0.20 * (deadline - time.time()), 40.0)
        a2_opt, sc2 = _run_exp_gradient_descent(
            a2, n2, run_deadline, start_time,
            alpha_start=5.0, alpha_end=200.0,
            eta0=0.35, beta=0.93,
            max_steps=120_000,
        )
        _register(a2_opt, sc2, "exp‑grad‑large")

        # final projected‑gradient polishing at large scale (shorter)
        if time.time() < deadline:
            pg_deadline = time.time() + min(12.0, deadline - time.time())
            a2_pg, sc2_pg = _run_gradient_descent(
                a2_opt, n2, pg_deadline, start_time,
                beta=0.94,
                step0=0.28,
                max_iter=150_000,
            )
            _register(a2_pg, sc2_pg, "pgd‑large")

        best_seq = a2_opt if sc2 < best_score else best_seq

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, sc_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        _register(a_ref, sc_ref, "final‑anneal")
        best_seq = a_ref

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
