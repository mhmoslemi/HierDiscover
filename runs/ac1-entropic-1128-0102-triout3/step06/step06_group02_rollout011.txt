"""
A fast, robust heuristic for the max‑convolution minimisation problem.
The search proceeds in stages of increasing resolution (800 → 2400 → 4800
→ 9600).  For each length we run three optimisation phases:

1. Adam‑based projected gradient descent on a soft‑max approximation of
   max(conv) (smooth, cheap and gives a good basin).
2. Adam‑based projected sub‑gradient descent on the exact max‑convolution
   (sharpens the solution).
3. A long simulated‑annealing walk that transfers a (large) fraction of
   mass between two coordinates *or* swaps two coordinates.  The walk
   uses a geometric temperature schedule and keeps the best ever vector.
   Finally a “permutation annealing” phase repeatedly swaps pairs of
   coordinates, aiming to decorrelate large weights in the index sum
   space.

All updates are projected onto the probability simplex (non‑negative,
unit‑sum).  The routine respects the 1000 s budget and returns the best
sequence found.  It starts from the previous global `height_sequence_1`
if present, otherwise from several sparse random restarts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the softmax of the convolution a*a.
    Uses the identity ∂/∂a (softmax(conv)) = 2·rev(a) * w,
    where w is the softmax weight vector over the convolution entries.
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # improve stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the soft‑max objective.
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.15,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max approximation of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    # temperature schedule – start relatively hot and cool down
    T_start = 5.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max Adam improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        # decreasing temperature for the soft‑max
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑based projected sub‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _run_adam_subgrad(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.08,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Adam optimiser on the exact max‑convolution (non‑smooth)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad Adam improve: "
                  f"n={n}, it={it}, score={cur_score:.6f}")

        k = conv.argmax()  # shift where the peak occurs

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass transfer + swaps).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-7,
                   steps: int = 1_000_000,
                   max_transfer: float = 0.30,
                   swap_prob: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Metropolis walk:
      * with probability `swap_prob` – swap two entries,
      * otherwise – move a random fraction (≤ max_transfer) of mass from i→j.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < swap_prob:
            # swap two positions
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]
        else:
            # mass transfer
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * max_transfer
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation‑only annealing (focus on decorrelating large weights).
# ----------------------------------------------------------------------
def _run_permutation_annealing(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               cur_best_score: float,
                               cur_best_seq: np.ndarray,
                               start_temp: float = 0.03,
                               end_temp: float = 1e-7,
                               steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Only swaps two coordinates; useful for breaking accidental sum‑collision patterns."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        a_prop = a.copy()
        a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] perm‑anneal improve: "
                  f"n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random sparse initialisation (helps to keep max‑conv low from the start).
# ----------------------------------------------------------------------
def _random_sparse_seq(n: int, m: int) -> np.ndarray:
    """Generate a length‑n vector with exactly `m` equal non‑zero entries."""
    seq = np.zeros(n)
    idx = np.random.choice(n, size=m, replace=False)
    seq[idx] = 1.0 / m
    return seq

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    two‑stage annealing walk (mass transfers + swaps, then pure swaps).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previous run.
    # ------------------------------------------------------------------
    try:

        if starter.ndim == 1 and starter.size > 0:

        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definition (progressively finer resolutions).
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]

    # allocate roughly 20 % of the budget to the final permutation anneal
    total_time = deadline - start_time
    final_perm_budget = max(20.0, 0.10 * total_time)
    remaining_budget = total_time - final_perm_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time for this stage proportionally to its size
        stage_time = remaining_budget * (n / sum(stage_lengths))
        stage_deadline = time.time() + stage_time

        # --------------------------------------------------------------
        # Initialise a0 for the current size.
        # --------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:

    # ------------------------------------------------------------------

        pass

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
