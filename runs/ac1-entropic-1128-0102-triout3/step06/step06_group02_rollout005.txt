"""
Propose a non‑negative step‑function (a probability vector) that
minimises the supplied ``evaluate_sequence`` objective.

Algorithm
---------
1.  Multiple short restarts are performed while time permits.
2.  Each restart starts from a random low‑support distribution (or an
    up‑sampled best‑so‑far) of a chosen length *n* (usually 2000‑12000).
3.  Three optimisation phases are applied to the candidate:
    * a smooth “soft‑max” gradient descent with a slowly cooling temperature,
    * a true sub‑gradient descent on the max‑convolution,
    * a long simulated‑annealing walk that transfers tiny mass packets.
4.  The best sequence seen over all restarts is returned.
   The code respects the 1000 s budget (a small safety margin is kept).

The implementation re‑uses the fast FFT‑based convolution utilities from the
previous version, adds a stricter temperature schedule and a few extra
iterations, and explores many random initialisations.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    # next power‑of‑two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` and re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the temperature‑softened max of the convolution.
    Uses the identity  d conv_i / d a = 2 * (rev(a) * e_i) where e_i are the
    soft‑max weights.
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))   # additional stabilisation
    w = exps / exps.sum()
    rev_a = a[::-1]

    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]

    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.6,
    beta: float = 0.9,
    t_start: float = 0.1,
    t_end: float = 1e-5,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov‑style gradient descent on the smooth soft‑max
    approximation of the max‑convolution.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # target number of iterations for the temperature schedule
    schedule_len = 300_000

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] softmax improve: "
                f"n={n}, it={it}, score={cur_score:.6f}"
            )

        # temperature decays very slowly (linear in log‑space)
        T = t_start * ((t_end / t_start) ** (it / schedule_len))

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_subgradient_gd(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.15,
    beta: float = 0.9,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov‑style sub‑gradient descent on the true
    max‑convolution objective (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                f"n={n}, it={it}, score={cur_score:.6f}"
            )

        k = conv.argmax()  # index of the current worst lag

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k :: -1]      # a[k‑i] for i=0..k
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 0.25,
    end_temp: float = 5e-8,
    steps: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that repeatedly moves a tiny fraction of mass
    from one position to another.  Temperature follows a geometric schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * ((end_temp / start_temp) ** (it / steps))

        # pick donor preferentially where mass exists
        i = np.random.randint(0, n)
        if a[i] == 0.0:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        # move a *random* fraction (up to 20 %) of the donor's mass
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n}, it={it}, score={score:.6f}"
            )

    return best_seq, best_score

            a0 = _upsample(best_seq, n)
        elif starter is not None and np.random.rand() < 0.15:
            a0 = _upsample(starter, n)
        else:
            # random low‑support start
            k = max(2, int(1.2 * math.sqrt(n)))           # ~1.2·√n non‑zero entries
            positions = np.random.choice(n, size=k, replace=False)
            weights = np.random.rand(k)
            a0 = np.zeros(n)
            a0[positions] = weights
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Phase 1 – smooth soft‑max gradient descent
        # --------------------------------------------------------------
        phase_deadline = time.time() + remaining * 0.45
        a_opt, score_opt = _run_softmax_gd(
            a0,
            n,
            phase_deadline,
            start_time,
            step0=0.6,
            beta=0.9,
            t_start=0.2,
            t_end=1e-5,
            max_iter=10 ** 7,
        )

        # --------------------------------------------------------------
        # Phase 2 – exact sub‑gradient descent (if time permits)
        # --------------------------------------------------------------
        if time.time() < phase_deadline:
            a_opt2, score_opt2 = _run_subgradient_gd(
                a_opt,
                n,
                phase_deadline,
                start_time,
                step0=0.12,
                beta=0.9,
                max_iter=10 ** 7,
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # --------------------------------------------------------------
        # Phase 3 – long annealing walk
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_ref, score_ref = _run_annealing(
                a_opt,
                n,
                deadline,
                start_time,
                best_score,
                best_seq if best_seq is not None else a_opt,
                start_temp=0.25,
                end_temp=5e-8,
                steps=2_500_000,
            )
            if score_ref < score_opt:
                a_opt, score_opt = a_ref, score_ref

        # --------------------------------------------------------------
        # Record improvement
        # --------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST: n={n}, score={score_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # If we never managed to produce a candidate, fall back to a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Final normalisation (protect against tiny numerical drift)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain list for the evaluation harness
    return [float(x) for x in best_seq]
