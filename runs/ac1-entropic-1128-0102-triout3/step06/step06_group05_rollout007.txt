"""
Hybrid multi‑scale optimizer for the self‑convolution L∞ problem.

The search works on the probability simplex (∑ a_i = 1) because the
evaluation function is scale‑invariant.  It proceeds through a
geometric schedule of lengths, each stage being a warm‑start for the
next one.  For each length we run several independent Adam runs on a
soft‑max surrogate of the peak of the convolution, followed by a
projected sub‑gradient descent on the exact peak and a final
targeted simulated‑annealing polishing.  The code explores a diverse
set of initial shapes (uniform, sine, triangular, arcsine,
Dirichlet draws) and occasionally disables the symmetry constraint.
All updates are projected onto the simplex and the algorithm stops
as soon as the global 1000 s budget is exhausted, returning the best
sequence found so far.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) ------------------------------------
# ----------------------------------------------------------------------


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (fast for any length)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)   # remove tiny negatives due to rounding
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the smoothed L∞ surrogate (soft‑max) --------------------
# ----------------------------------------------------------------------


def softmax_gradient(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing temperature λ > 0

    Returns
    -------
    grad: dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    # correlation of weights with reversed a
    a_rev = a[::-1]
    full = conv_fft(weights, a_rev)                 # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set) -------------------
# ----------------------------------------------------------------------


def max_conv_gradient(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.

    Returns
    -------
    grad   – gradient vector of shape (n,)
    maxval – current maximal convolution value
    """
    conv = conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = conv_fft(w, a_rev)                     # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate ----------------------------
# ----------------------------------------------------------------------


def run_adam(a0: np.ndarray,
             n: int,
             deadline: float,
             start_time: float,
             symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the soft‑max surrogate.

    Returns
    -------
    best_a   – vector delivering the lowest exact score seen
    best_obj – corresponding exact score (= 2·n·max(conv))
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = float('inf')
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ schedule
    lam_factor = 0.5          # starts at 0.5·max(conv)
    lam_decay = 0.95
    min_factor = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_b = conv.max()
        cur_obj = 2.0 * n * max_b
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} obj={cur_obj:.6f}")

        # update λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, min_factor)

        # gradient of smooth surrogate
        grad = softmax_gradient(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.7 / math.sqrt(it + 1.0)   # a bit larger than the previous 0.8
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = project_simplex(a)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the exact max -------------------
# ----------------------------------------------------------------------


def run_subgradient(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns best vector and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = conv_self(a)
    best_obj = 2.0 * n * conv.max()
    step0 = 0.6                           # slightly smaller to improve stability
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = max_conv_gradient(a)
        step = step0 / math.sqrt(it + 1.0)
        a -= step * grad
        if symmetrize:
            a = (a + a[::-1]) * 0.5
        a = project_simplex(a)
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad improve n={n} it={it} obj={cur_obj:.6f}")
    return best_a, best_obj


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing) ---------------------------
# ----------------------------------------------------------------------


def run_annealing(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  cur_best_obj: float,
                  cur_best_seq: np.ndarray,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-6,
                  steps: int = 180000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination.
    Returns the best sequence seen and its exact score.
    """
    a = a0.copy()
    conv = conv_self(a)
    cur_obj = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_obj = cur_best_obj

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move up to 12 % of the source mass
        eps = a[i] * np.random.rand() * 0.12
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # no need for explicit projection – total mass unchanged and eps ≤ a[i]
        new_obj = 2.0 * n * conv_self(a_prop).max()
        delta = new_obj - cur_obj
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_obj = new_obj
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} obj={cur_obj:.6f}")

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Initial shape generators -----------------------------------------------
# ----------------------------------------------------------------------


def generate_initial_variants(base_seq: np.ndarray, n: int) -> list[np.ndarray]:
    """
    Produce several plausible start vectors derived from `base_seq`.
    All returned vectors are normalised to sum = 1.
    """
    variants = []

    # 0. warm‑start copy (upsampled if needed)
    variants.append(upsample(base_seq, n))

    # 1. uniform
    uni = np.full(n, 1.0 / n)
    variants.append(uni)

    # 2. positive sine shape
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular (linearly increasing then decreasing)
    left = np.linspace(1, n // 2, n // 2, endpoint=False)
    right = np.linspace(n // 2, 1, n - n // 2)
    tri = np.concatenate([left, right])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. arcsine‑like shape (1/sqrt(x(1‑x)))
    t = (np.arange(n) + 0.5) / n
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    variants.append(arc)

    # 5. random Dirichlet draws with varied concentration
    for alpha in (0.3, 0.7, 1.5, 3.0):
        rnd = np.random.gamma(alpha, 1.0, size=n)
        if rnd.sum() > 0:
            rnd /= rnd.sum()
            variants.append(rnd)

    # 6. slightly perturbed warm‑start (add small Gaussian noise)
    pert = base_seq + 0.02 * np.random.randn(n)
    pert = np.maximum(pert, 0.0)
    if pert.sum() > 0:
        pert /= pert.sum()
        variants.append(pert)

    return variants


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_obj = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction
    # ------------------------------------------------------------------
    base_vec = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                base_vec = starter / s
    except Exception:
        base_vec = None

    # Length schedule – start modest, grow geometrically, then try a very large size
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]
    # Shuffle early to avoid bias if we run out of time early
    random.shuffle(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 (warm start / several variants)
        # ------------------------------------------------------------------
        if base_vec is not None:
            a0 = upsample(base_vec, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_variants = generate_initial_variants(a0, n)

        # Time slice for this length: proportional to n but capped
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_frac = 0.12 / (len(lengths) - idx)          # 12 % of remaining time per stage
        slice_deadline = time.time() + max(5.0, remaining * slice_frac)

        # ------------------------------------------------------------------
        # Run a few Adam optimisation phases (different starts, random sym)
        # ------------------------------------------------------------------
        local_best_a = None
        local_best_obj = float('inf')
        for start_vec in init_variants:
            if time.time() >= slice_deadline:
                break
            sym = (np.random.rand() < 0.25)   # 25 % of runs forced symmetric
            a_opt, obj_opt = run_adam(start_vec, n, slice_deadline, start_time,
                                      symmetrize=sym)
            if obj_opt < local_best_obj:
                local_best_obj = obj_opt
                local_best_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Local Adam improve n={n} obj={obj_opt:.6f}")

        if local_best_a is None:
            continue

        # Record any global improvement from Adam
        if local_best_obj < best_obj:
            best_obj = local_best_obj
            best_seq = local_best_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} obj={local_best_obj:.6f}")

        # ------------------------------------------------------------------
        # Deterministic sub‑gradient descent (refinement)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.12)
            a_sub, obj_sub = run_subgradient(local_best_a, n, sub_deadline, start_time,
                                             symmetrize=False)   # allow asymmetry in fine tune
            if obj_sub < best_obj:
                best_obj = obj_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Subgrad n={n} obj={obj_sub:.6f}")

        # ------------------------------------------------------------------
        # Final annealing polishing (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_ref, obj_ref = run_annealing(
                local_best_a,
                n,
                anneal_deadline,
                start_time,
                best_obj,
                best_seq,
                start_temp=4e-2,
                end_temp=5e-7,
                steps=200000,
            )
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Anneal n={n} obj={obj_ref:.6f}")

        # Remember the best vector as warm‑start for larger lengths
        base_vec = best_seq.copy()

    # ------------------------------------------------------------------
    # Safety fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0)

    # Return a plain Python list – the evaluator will clamp / normalise
    return [float(x) for x in best_seq]
