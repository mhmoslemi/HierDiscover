"""
Low‑maximum self‑convolution search (enhanced).

The algorithm builds on the previous entropic mirror‑descent approach
but adds three new ingredients:

1. **Linear‑programming “mass‑balance” step** – for a fixed shape `a`
   we solve a small LP that maximises the total mass of a partner
   vector `g` while keeping the self‑convolution bound `max(a*a)`.
   Mixing the LP solution into `a` reduces the peak of the
   convolution much faster than pure gradient descent.

2. **Plain sub‑gradient descent on the true ℓ∞‑objective** – after the
   soft‑max phase we switch to the exact max‑norm surrogate.
   The gradient is obtained with a single FFT (the same trick used by
   the entropic method) and an exponentiated‑gradient (mirror‑descent)
   update.

3. **Adaptive time allocation** – longer sequences receive a
   proportionally larger slice of the total budget, improving the final
   bound for large `n`.

The code respects all security checks of ``evaluate_sequence``:
only non‑negative numbers are produced and the returned list is a
plain ``list[float]``.  The search stops well before the deadline, so
the evaluator never receives ``np.inf``.

"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog  # LP solver; if unavailable, LP step is skipped.

# ----------------------------------------------------------------------
# FFT‑based convolutions (self and cross)
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring utilities (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalized vector (∑a = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families for quick warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term <= 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine‑like density ∝ 1/√(i·(n‑i)).
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term <= 0] = eps
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Entropic (soft‑max) mirror descent
# ----------------------------------------------------------------------


def entropic_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Minimise max(a*a) via a smooth soft‑max surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # temperature schedule – start relatively high, decay slowly
    tau = max(conv_self(a).max() * 0.2, 1e-6)
    tau_end = tau * 1e-4
    decay = 0.9997

    eta0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] entropic improve n={n} it={it} score={cur_sc:.6f}")

        # soft‑max weights
        shift = conv.max()
        w = np.exp((conv - shift) / tau)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate (FFT‑based)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)       # extract needed part
        grad = 2.0 * corr[idx]

        # exponentiated‑gradient update
        step = eta0 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # anneal temperature
        tau = max(tau_end, tau * decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑norm objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on max(a*a) using the true ℓ∞‑surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        # indicator of active constraints (all peaks)
        w = (conv == max_val).astype(float)

        # gradient = 2 * (w * a_rev)   (FFT for speed)
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)          # length = 3n‑2
        idx = (n - 1) - np.arange(n)
        grad = 2.0 * corr[idx]

        # exponentiated‑gradient step
        step = 0.5 / math.sqrt(it)
        expo = -step * grad
        expo = np.clip(expo, -700.0, 700.0)
        a = a * np.exp(expo)

        # renormalise onto simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_sc = 2.0 * n * conv_self(a).max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑grad improve n={n} it={it} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming mass‑balance step
# ----------------------------------------------------------------------


def lp_optimize(
    a0: np.ndarray,
    n: int,
    max_iters: int,
    deadline: float,
    start_time: float,
    step: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP iterations:
    maximise Σ g_i   s.t.   g_i ≥ 0  and  (a * g)_k ≤ M  for every k,
    where M = max(a*a).  The solution is mixed back into `a`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    it = 0

    # sanity – SciPy may be missing; in that case we simply return the input.
    if 'linprog' not in globals():
        return a, best_sc

    while it < max_iters and time.time() < deadline:
        it += 1
        M = conv_self(a).max()

        # Build dense constraint matrix A (shape (2n‑1, n))
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i : i + n, i] = a[i]

        c = -np.ones(n, dtype=np.float64)           # maximise Σ g_i  → minimise −Σ g_i
        b = np.full(2 * n - 1, M, dtype=np.float64)  # RHS of each inequality

        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method="highs")
        if not res.success:
            # If the LP fails, break early.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g > 0:
            g = g / sum_g
        else:
            g = np.full_like(g, 1.0 / n)

        # convex blend
        a = (1.0 - step) * a + step * g

        # renormalise (should already be 1)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} it={it} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polish)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap move‑based annealing: transfer a random ε from one index
    to another.  Accept moves that improve the score or according to a
    geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine combines random‑family warm‑starts,
    a linear‑programming mass‑balance phase, entropic mirror‑descent,
    exact sub‑gradient descent and a short annealing polish.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s > 0.0:
            starter = tmp / s
    except Exception:
        starter = None

    # Length schedule – larger n gets a larger share of the remaining budget.
    lengths = [256, 512, 1024, 2048, 4096]
    random.shuffle(lengths)

    global_a = starter

    # Pre‑compute total length weight for adaptive time slicing.
    total_weight = sum(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Allocate a slice of the remaining budget (proportional to n)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_frac = n / total_weight
        slice_deadline = time.time() + max(10.0, remaining * slice_frac)

        # --------------------------------------------------------------
        # Quick parametric warm‑starts
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.5, (slice_deadline - time.time()) * 0.2)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3])
            if kind == 0:  # product‑power family
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:  # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            else:  # arcsine‑like family
                cand = shape_arcsine(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # Linear‑programming mass‑balance optimisation
        # --------------------------------------------------------------
        lp_deadline = time.time() + max(5.0, (slice_deadline - time.time()) * 0.25)
        a_lp, sc_lp = lp_optimize(
            local_best_a,
            n,
            max_iters=5,
            deadline=lp_deadline,
            start_time=start_time,
            step=0.2,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after LP n={n} score={sc_lp:.6f}"
            )

        # --------------------------------------------------------------
        # Entropic mirror‑descent optimisation (smooth surrogate)
        # --------------------------------------------------------------
        ent_deadline = time.time() + max(8.0, (slice_deadline - time.time()) * 0.3)
        a_ent, sc_ent = entropic_gradient_descent(
            a_lp, n, ent_deadline, start_time
        )
        if sc_ent < best_score:
            best_score = sc_ent
            best_seq = a_ent.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after entropic n={n} score={sc_ent:.6f}"
            )

        # --------------------------------------------------------------
        # Sub‑gradient descent on the exact max‑norm
        # --------------------------------------------------------------
        sub_deadline = time.time() + max(6.0, (slice_deadline - time.time()) * 0.25)
        a_sub, sc_sub = subgradient_descent(
            a_ent, n, sub_deadline, start_time
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after sub‑grad n={n} score={sc_sub:.6f}"
            )

        # --------------------------------------------------------------
        # Cheap annealing polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.1)
            a_ref, sc_ref = annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # keep the best distribution for the next up‑sample
        global_a = best_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] finished n={n}, best score so far={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the external evaluator will clip/normalise)
    return [float(x) for x in best_seq]
