"""
Hybrid up‑sample → Adam → annealing optimizer.

1. Load the previously discovered sequence (``height_sequence_1``) if it exists.
2. Optionally up‑sample it to a larger length (up to 8192) to give the optimizer
   more degrees of freedom – the objective 2·n·max(a*a) converges to a
   length‑independent limit, so a finer discretisation can improve the bound.
3. Run a long projected‑Adam optimisation on the smooth log‑sum‑exp surrogate
   of the max‑norm; we keep the best iterate seen.
4. Finish with a short simulated‑annealing “mass‑transfer” polish.
All steps respect the overall `budget_s` time limit and keep the vector on the
probability simplex (non‑negative, sum = 1).  The returned list is a candidate
step‑function for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np
from scipy.special import i0  # only needed for optional Kaiser windows


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)
    return c[:n]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(A * A, n=nfft)
    return c[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (∑a_i = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        L_λ(a) = λ·log Σ exp(conv/λ)
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)           # length 3n-2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]  # central slice
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth log‑sum‑exp surrogate.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Accepts a move if it improves or with Metropolis probability.
    """
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # cooling schedule (exponential)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up

    slice_deadline = deadline - annealing_margin
    if slice_deadline > time.time():
        a_opt, score_opt = _run_adam_opt(best_seq, target_n,
                                         slice_deadline, start_time,
                                         symmetrize=True)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam final improve = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing polish (light)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_annealing(best_seq, target_n,
                                              deadline=deadline,
                                              start_time=start_time,
                                              cur_best_score=best_score,
                                              cur_best_seq=best_seq,
                                              start_temp=2e-2,
                                              end_temp=1e-5,
                                              steps=80000)
        print(f"[{time.time() - start_time:6.2f}s] final anneal best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Ensure valid simplex point before returning
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(target_n, 1.0 / target_n)

    return [float(x) for x in best_seq]
