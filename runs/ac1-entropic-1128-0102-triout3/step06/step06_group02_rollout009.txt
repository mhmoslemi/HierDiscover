def _run_subgradient_gd(a0, n, deadline, start_time, step0=0.1, beta=0.9, max_iter=10**7, eps_frac=0.01):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")
        # active set tolerance: set tolerance as eps_frac * max_val
        eps = max_val * eps_frac
        active = np.where(conv >= max_val - eps)[0]
        # build weight vector
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        # gradient = 2 * conv_rev_a_w
        rev_a = a[::-1]
        # convolution rev_a with w using FFT
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n-1:n-1+n]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

   All indices whose value is within a small relative tolerance of the current
   maximum are treated as active; the gradient is the average of the individual
   sub‑gradients (computed efficiently via FFT).
4. **Extended simulated annealing** – tiny random mass‑transfer moves and
   occasional local smoothing are accepted with a geometric temperature
   schedule.  More time is devoted to this final refinement than in the
   previous baseline.

The routine works for any length up to a few‑ten‑thousands (the final length
used here is 9600).  All vectors are kept non‑negative and normalised so that
``sum(a)=1``; the final candidate is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``new_n`` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution (temperature‑smoothed)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))   # improve numerical stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected Nesterov‑style GD on the soft‑max surrogate of max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule: from 5·10⁻² down to 5·10⁻⁴
    T_start, T_end = 5.0e-2, 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # geometric temperature decay
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# Uses an active set (all entries within a small fraction of the maximum).
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.15,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7,
                        rel_eps: float = 0.01) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv) using an active set."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # ---- active set -------------------------------------------------
        eps = max_val * rel_eps
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:   # fallback (should not happen)
            active = np.array([conv.argmax()], dtype=int)

        # build weight vector w (averaged over active indices)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (rev(a) * w)   (linear convolution)
        rev_a = a[::-1]
        m = w.shape[0]
        fft_len = 1 << ((n + m - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
        grad = 2.0 * full[n - 1:n - 1 + n]

        # Nesterov‑style momentum & diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer + occasional smoothing).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7,
                   steps: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass and applies local smoothing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        if np.random.rand() < 0.85:                     # mass‑transfer move
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            eps = a[i] * np.random.rand() * 0.25        # up to 25 % of mass at i
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
        else:                                            # local smoothing move
            k = np.random.choice([3, 5, 7])              # small kernel width
            kernel = np.ones(k) / k
            a_prop = np.convolve(a, kernel, mode='same')
            a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs progressive up‑sampling,
    a smooth gradient phase, an active‑set sub‑gradient phase and ends with a
    relatively long simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previous construction if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage layout (lengths) – we keep the same schedule as the reference.
    # ------------------------------------------------------------------
    stage_ns = [800, 2400, 4800, 9600]               # growing resolution
    total_n = sum(stage_ns)

    total_time = deadline - start_time
    # Allocate ~70 % of the time to the gradient phases, the rest to annealing.
    grad_time_budget = total_time * 0.70
    anneal_time_budget = total_time - grad_time_budget

    for n in stage_ns:
        if time.time() >= deadline:
            break

        # time slice for the current stage (proportional to its length)
        stage_time = grad_time_budget * (n / total_n)
        stage_deadline = time.time() + stage_time

        # ------------------------------------------------------------------
        # Initialise a vector for this resolution.
        # ------------------------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ==== Phase 1: smooth (soft‑max) gradient descent =================
        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.8, beta=0.9, max_iter=10 ** 7
        )

        # ==== Phase 2: exact sub‑gradient descent =========================
        if time.time() < stage_deadline:
            a_opt2, score_opt2 = _run_subgradient_gd(
                a_opt, n, stage_deadline, start_time,
                step0=0.15, beta=0.9, max_iter=10 ** 7,
                rel_eps=0.01
            )
            if score_opt2 < score_opt:
                a_opt, score_opt = a_opt2, score_opt2

        # Keep the globally best sequence found so far.
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # re‑compute remaining gradient budget after this stage
        elapsed = time.time() - start_time
        grad_time_budget = max(0.0,
                               deadline - start_time - anneal_time_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – operates on the final resolution we ended up with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-7,
            steps=5_000_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be triggered).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # Final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness.
    return [float(x) for x in best_seq]
