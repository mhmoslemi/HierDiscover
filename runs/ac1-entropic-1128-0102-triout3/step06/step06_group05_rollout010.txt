def _lp_matrices(f):
    n = len(f)
    K = 2*n - 1
    A_ub = np.zeros((K, n+1), dtype=float)
    for k in range(K):
        # indices i from max(0,k-(n-1)) to min(n-1,k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # for j = i, coefficient f[k-i]
        # vector g indices range i_min..i_max
        # set A_ub[k, i_min:i_max+1] = f[k - np.arange(i_min,i_max+1)]
        idxs = np.arange(i_min, i_max+1)
        A_ub[k, idxs] = f[k - idxs]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    return A_ub, b_ub, A_eq, b_eq

"""
Hybrid optimizer for the self‑convolution sup‑norm problem.

The algorithm consists of three stages:

1. **LP‑based active‑set improvement** on a modest size (≈1024).
   For a fixed vector f we solve a linear program that finds the
   distribution g (∑g=1, g≥0) that minimises the maximal entry of
   the convolution f∗g.  Mixing f and g (f←(1‑τ)f+τg) strictly
   reduces the objective unless a fixed point has been reached.

2. **Scale‑up** – the best small‑scale distribution is interpolated
   to a much larger length (up to 65536).  The up‑sampled vector is
   symmetrised and re‑normalised; the theoretical lower bound then
   becomes very close to 1.

3. **Fine‑tuning** – a few thousand iterations of Adam on a
   soft‑max surrogate of the ℓ∞‑norm, followed by deterministic
   sub‑gradient descent and a short annealing polish.

All vectors are projected onto the probability simplex after each
update, and the distribution is kept symmetric (a←(a+a[::-1])/2)
because optimal solutions are believed to be symmetric.
The routine respects a 1000 s time budget and returns the best
sequence found.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities ----------------------------------------
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (fast for all lengths)."""
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(Af * Af, n=nfft)[:size]
    np.maximum(conv, 0.0, out=conv)   # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Upsample via linear interpolation ---------------------------------------
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (log‑sum‑exp) ----------------------
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """Gradient of L_λ(a) = λ·log∑exp(conv/λ) (smooth max)."""
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)          # length = n + (2n‑1) - 1 = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution -----------------------------
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Returns (grad, max_value) for the ℓ∞‑norm of a*a."""
    conv = _conv_self_fft(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate -----------------------------
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max surrogate. Returns best a seen and its exact score."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    lam_factor = 0.3
    lam_decay = 0.98
    lam_min_factor = 1e-9

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent --------------------------------------
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_score = 2.0 * n * conv.max()
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing -------------------------------
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Cheap annealing: re‑distribute a small ε from one index to another."""
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape generators -------------------------------------------------------
# ----------------------------------------------------------------------


def _shape_arcsine_like(n: int, p: float = 0.5, eps: float = 1e-12) -> np.ndarray:
    """
    Produce a symmetric U‑shaped distribution:
        a_i ∝ (t_i·(1‑t_i) + eps)^{-p}
    where t_i = (i+0.5)/n ∈ (0,1).
    The vector is normalised to sum = 1.
    """
    t = (np.arange(n) + 0.5) / n
    vals = (t * (1.0 - t) + eps) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming helper -----------------------------------------------
# ----------------------------------------------------------------------


def _build_lp_matrices(f: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """
    Build the matrices for:
        minimise   t
        s.t.       f * g ≤ t   (elementwise)
                    ∑ g = 1
                    g ≥ 0,  t ≥ 0
    Variables are (g_0,…,g_{n‑1}, t)  –  t is the last one.
    """
    n = f.size
    K = 2 * n - 1                     # length of the linear convolution
    A_ub = np.zeros((K, n + 1), dtype=float)
    for k in range(K):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = f[k - idx]      # contribution of g_i
        A_ub[k, -1] = -1.0              # -t ≤ 0  →  sum_i f_i g_{k-i} - t ≤ 0
    b_ub = np.zeros(K, dtype=float)

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=float)

    return A_ub, b_ub, A_eq, b_eq


def _solve_lp_for_g(f: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the LP described in _build_lp_matrices.
    Returns (g, t_opt).  If the solver fails, falls back to the current f.
    """
    A_ub, b_ub, A_eq, b_eq = _build_lp_matrices(f)
    n = f.size
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0                       # minimise t

    bounds = [(0.0, None)] * n + [(0.0, None)]   # g_i ≥0, t ≥0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})

    if not res.success:
        # fallback – return the current f (which is feasible)
        return f.copy(), (f * f).max()   # crude upper bound
    sol = res.x
    g = sol[:n]
    t_opt = sol[-1]
    # symmetrise the solution (helps later stages)
    g = (g + g[::-1]) * 0.5
    g = _project_simplex(g)
    return g, t_opt

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Small‑scale LP‑driven improvement (n ≈ 1024)
    # ------------------------------------------------------------------
    n_lp = 1024
    # start from an arcsine‑like shape (parameter p will be tuned shortly)
    best_lp_vec = None
    best_lp_score = float('inf')

    # quick sweep over a few exponents to obtain a decent start
    for p in np.linspace(0.3, 1.0, 15):
        a0 = _shape_arcsine_like(n_lp, p=p)
        # run a few LP iterations (at most 5) to polish
        a = a0.copy()
        for _ in range(5):
            g, t_opt = _solve_lp_for_g(a)
            if t_opt >= 0.9999 * (a * a).max():
                break
            # mix with τ = 0.5
            a = 0.5 * a + 0.5 * g
            a = _project_simplex(a)
        conv = _conv_self_fft(a)
        score = 2.0 * n_lp * conv.max()
        if score < best_lp_score:
            best_lp_score = score
            best_lp_vec = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP sweep p={p:.3f} score={score:.6f}")

    # If LP failed for some reason fallback to a random start
    if best_lp_vec is None:
        best_lp_vec = np.random.rand(n_lp)
        best_lp_vec = _project_simplex(best_lp_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Scale‑up to a large size (the final length)
    # ------------------------------------------------------------------
    # Choose a final length that comfortably fits in the remaining time.
    # 65536 works well on modern hardware; if we are already close to the
    # deadline we fall back to a smaller size.
    remaining = deadline - time.time()
    if remaining > 400:
        final_n = 65536
    elif remaining > 200:
        final_n = 32768
    elif remaining > 80:
        final_n = 16384
    else:
        final_n = 8192

    a = _upsample(best_lp_vec, final_n)
    a = (a + a[::-1]) * 0.5               # enforce symmetry
    a = _project_simplex(a)

    best_seq = a.copy()
    best_score = 2.0 * final_n * _conv_self_fft(a).max()
    print(f"[{time.time() - start_time:6.2f}s] after up‑sample n={final_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + max(8.0, (deadline - time.time()) * 0.45)
        a_adam, score_adam = _run_adam_opt(a, final_n, sub_deadline, start_time,
                                           symmetrize=True)
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:6.2f}s] after Adam score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Deterministic sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.30)
        a_sub, score_sub = _run_subgradient(best_seq, final_n,
                                            sub_deadline, start_time)
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] after sub‑grad score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, score_final = _run_annealing(best_seq, final_n,
                                              anneal_deadline, start_time,
                                              best_score, best_seq,
                                              start_temp=2e-2,
                                              end_temp=1e-6,
                                              steps=250000)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] after annealing score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    print(f"Best objective reached: {best_score:.6f}")
    return [float(x) for x in best_seq]
