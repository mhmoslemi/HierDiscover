def _run_subgradient_descent(a0, n, deadline, start_time,
                             step0=0.8, beta=0.9):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = evaluate(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_idx = int(np.argmax(conv))
        cur_max = conv[max_idx]
        cur_score = 2.0 * n * cur_max # sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # gradient for max index
        grad = np.zeros_like(a)
        # compute gradient contributions
        # For each i where j = max_idx - i in [0, n-1]
        # grad[i] = 2 * a[j]
        # Use vectorized operation:
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad[i[mask]] = 2.0 * a[j[mask]]

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # projection onto simplex to maintain non-negativity and sum=1
        a = _project_simplex(a)
    return best_a, best_score

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Helper: up‑sample a probability vector (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Soft‑max gradient (smooth max‑convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """∇_a Σ_k softmax_k(conv/temperature)·conv_k."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    # convolution of rev(a) with weight vector w
    n = a.shape[0]
    fft_len = 1 << ((n + len(w) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + len(w) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Return the best point seen and its (un‑normalised) score = 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] soft‑max improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.5,
                        beta: float = 0.5,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Project the iterates onto the simplex while descending the max‑conv."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_max = conv[idx_max]
        cur_score = 2.0 * n * cur_max          # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # gradient of the active shift (sub‑gradient)
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(a)
        grad[i_arr[mask]] = 2.0 * a[j_arr[mask]]

        # momentum & step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """Mass‑shifting walk that reduces the maximal convolution entry."""
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule for Metropolis acceptance
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contributions of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor: pick proportionally to contribution (focus on hot spots)
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: preferably from a low‑contribution region
        low_thr = np.percentile(contrib, 20)            # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        eps = a[i_donor] * np.random.rand() * 0.3      # up to 30 % of donor
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines a multi‑stage soft‑max gradient descent,
    a sub‑gradient refinement on the true max‑convolution,
    and a final peak‑shaving local walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Progressive up‑sampling stages (soft‑max GD)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]   # final length 9600
    total_len_weight = sum(stage_lengths)

    # allocate 70 % of the remaining time to the soft‑max phase
    total_time = deadline - start_time
    softmax_budget = total_time * 0.70
    local_budget = total_time - softmax_budget

    best_seq = None
    best_score = float("inf")

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # proportion of the remaining soft‑max budget
        stage_frac = n / total_len_weight
        stage_time = max(2.0, softmax_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this length
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.8, beta=0.9, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # update remaining soft‑max budget (account for elapsed time)
        elapsed = time.time() - start_time
        softmax_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # ------------------------------------------------------------------
    # Sub‑gradient refinement on the final length (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        subgrad_deadline = deadline - local_budget * 0.5   # keep half of local budget for final walk
        a_refined, score_refined = _run_subgradient_gd(
            best_seq, n_final, subgrad_deadline, start_time,
            step0=0.5, beta=0.5, max_iter=10 ** 7
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑gradient improve: "
                  f"score={score_refined:.6f}")

    # ------------------------------------------------------------------
    # Targeted local search (peak shaving) – uses the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list (the harness expects a list)
    return [float(x) for x in best_seq]
```

This implementation respects the time budget, explores several optimisation stages, and prints intermediate scores so that the progress can be monitored.
