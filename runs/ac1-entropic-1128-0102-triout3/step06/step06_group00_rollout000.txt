"""
Propose a low‑peak autocorrelation step function.

The objective returned by ``evaluate_sequence`` (after the internal
normalisation) equals

    2·n·max(conv(a,a))        (where sum(a)=1)

so we minimise the peak of the self‑convolution of a non‑negative
vector a.  The algorithm consists of three stages:

1. **Linear‑programming improvement** – for the current vector f we
   compute the maximal convolution value M = max(conv(f,f)).  Using only
   the constraints that are near this maximum we solve a linear program

        maximise   Σ b_i
        subject to conv(f,b) ≤ M   (active shifts only)
                    b_i ≥ 0

   The solution b is rescaled to a probability vector g and mixed into
   f (f←(1‑t)f+t·g).  Repeating this step quickly drives down the
   peak.

2. **Projected sub‑gradient descent** – a long run of (accelerated)
   sub‑gradient descent on the true nondifferentiable objective
   2·n·max(conv) while staying on the probability simplex.

3. **Tiny simulated‑annealing polish** – a short stochastic mass‑transfer
   phase that can escape shallow local minima.

All phases respect the overall time budget (≈1000 s).  The routine
returns a list of non‑negative heights (the vector is finally normalised
to sum = 1, which is irrelevant for the objective because of scale
invariance).  The algorithm works for any starting point; if a global
variable ``height_sequence_1`` exists it is used, otherwise a random
initialisation is generated.
"""

import time
import math
import numpy as np

try:
    # SciPy is optional – the LP phase will be skipped if unavailable.
    from scipy.optimize import linprog
    _HAVE_LP = True
except Exception:   # pragma: no cover
    _HAVE_LP = False


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a.
    All shifts whose values are within rel_tol of the maximum are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # contribution a[j] to positions i, and symmetrically
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Phase‑1 : linear‑programming driven improvement
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                t_mix: float,
                active_rel: float,
                max_outer_iters: int,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a small LP that maximises Σ b_i under the
    constraints conv(a,b) ≤ M for only the shifts whose convolution
    value is within `active_rel` of the current peak M.
    Returns the best vector found and its objective value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    for outer in range(max_outer_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()
        # identify “active” shifts (those close to the peak)
        thresh = M * (1.0 - active_rel)
        act_idx = np.where(conv >= thresh)[0]

        # limit the number of constraints to keep the LP small
        max_constraints = 400
        if act_idx.size > max_constraints:
            # keep the largest values
            top = np.argsort(-conv[act_idx])[:max_constraints]
            act_idx = act_idx[top]

        rows = act_idx.size
        if rows == 0:
            break

        # Build the constraint matrix A (rows × n)
        A = np.zeros((rows, n), dtype=float)
        for r, k in enumerate(act_idx):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            A[r, i] = a[k - i]

        b = np.full(rows, M, dtype=float)

        # Solve  maximise Σ b_i  <=> minimise −Σ b_i
        c = -np.ones(n, dtype=float)

        if _HAVE_LP:
            # Using the HiGHS interior‑point solver (default for recent SciPy)
            res = linprog(c,
                          A_ub=A,
                          b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})
        else:
            # If SciPy is missing, we cannot perform the LP step.
            break

        if not res.success:
            # Something went wrong – exit the LP phase.
            print(f"[{time.time()-start_time:.2f}s] LP failed (iter {outer}): {res.message}")
            break

        b_vec = res.x
        s = b_vec.sum()
        if s < 1e-12:
            break

        g = b_vec / s                     # normalise to a probability vector
        # mixing step
        a = (1.0 - t_mix) * a + t_mix * g
        a = _project_simplex(a)

        # evaluate
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] LP improve: score={best_score:.6f}")

        # early stop if improvement stalled
        if M - cur_score / (2.0 * n) < 1e-9:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.5,
                         beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classical momentum‑accelerated projected sub‑gradient descent.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] sub‑grad improve: iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

        # occasionally break if we are already close to the best known bound
        if it % 200000 == 0:
            # re‑estimate remaining time
            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing_polish(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray,
                      steps: int = 120_000,
                      start_temp: float = 5e-2,
                      end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer moves with exponential cooling.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)                 # choose a source proportional to mass
        dst = rng.integers(n)                    # uniform destination
        if src == dst or a[src] < 1e-12:
            continue

        # transfer a random fraction (up to 30%) of src's mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] anneal improve: iter={it:,} score={cur_score:.6f}")

    return best_seq, best_score
