"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the previous best construction if available,
   otherwise from a uniform distribution.
2. Progressive up‑sampling: 800 → 1600 → 3200 → 6400 → 9600.
   At each resolution run a *projected sub‑gradient descent* that
   directly minimises the quantity 2·n·max(conv) (the denominator is 1
   because we keep the vector on the probability simplex).
3. After the final resolution run an intensive *mass‑shifting* local search
   (biased Metropolis).
4. Finally mix the best candidate with a small proportion of the uniform
   distribution – a cheap one‑dimensional search that often smoothes
   remaining peaks.
The whole pipeline respects the 1000 s budget and prints progress
information.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should never happen; fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: upscale a simplex vector by linear interpolation (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient of 2·n·max(conv) (conv = a*a)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of 2·n·max(conv) w.r.t. a (a on simplex)."""
    n = a.shape[0]
    conv = _conv_self(a)
    k = int(np.argmax(conv))                # index of the current maximum
    idx = np.arange(n)
    mask = (k - idx >= 0) & (k - idx < n)   # indices that actually appear in the max term
    g = np.zeros_like(a)
    # derivative of max(conv) w.r.t a_i is 2·a_{k-i}
    g[mask] = 4.0 * n * a[k - idx[mask]]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with simple momentum)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.2,
                             beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Perform projected (Nesterov‑style) sub‑gradient descent on 2·n·max(conv).
    Returns the best point seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline:
        it += 1
        # evaluate current score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # compute sub‑gradient at the current maximiser
        grad = _subgradient_max_conv(a)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (mass‑shifting with Metropolis acceptance)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of probability mass away from indices that
    contribute most to the current maximal convolution entry.  A slowly cooling
    Metropolis rule accepts occasional uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (geometric cooling)
    T_start = 5e-2
    T_end = 1e-6

    it = 0
    while time.time() < deadline:
        it += 1

        # recompute convolution and its maximiser
        conv = _conv_self(a)
        k_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to this maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        # donor – biased towards large contributions
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient – choose a low‑contribution index
        low_thr = np.percentile(contrib, 20)        # bottom 20 %
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_candidates)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of donor's mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        # evaluate proposal
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        # compute current temperature
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T_start * (T_end / T_start) ** elapsed

        # Metropolis acceptance
        accept = (delta < 0) or (np.exp(-delta / (T + 1e-12)) > np.random.rand())
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

# Mix candidate with the uniform distribution – cheap 1‑D search
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, t_vals) -> tuple[np.ndarray, float]:
    """Return the mixture (1‑t)·a + t·uniform with the smallest score."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    best_score = float("inf")
    best_mix = a
    for t in t_vals:
        mix = (1.0 - t) * a + t * uniform
        mix /= mix.sum()
        score = 2.0 * n * _conv_self(mix).max()
        if score < best_score:
            best_score = score
            best_mix = mix
    return best_mix, best_score

            a0 = _upsample(starter, n)
        else:
            a0 = np.full(n, 1.0 / n)   # uniform start

        # run sub‑gradient descent
        a_opt, score_opt = _run_subgradient_descent(
            a0,
            deadline=stage_deadline,
            start_time=start_time,
            step0=0.2,
            beta=0.9,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_mix_budget - elapsed)

    # --------------------------------------------------------------
    # Targeted local search (mass‑shifting) + uniform mixing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]

        # --- local search -------------------------------------------------
        local_deadline = time.time() + local_mix_budget * 0.8
        a_loc, score_loc = _run_targeted_local_search(
            best_seq,
            deadline=local_deadline,
            start_time=start_time,
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] local search improve: score={score_loc:.6f}")

        # --- final mixing with uniform ------------------------------------
        mix_deadline = time.time() + local_mix_budget * 0.2
        if time.time() < mix_deadline:
            # try 21 mixing coefficients between 0 and 0.2 (small uniform boost)
            t_grid = np.linspace(0.0, 0.2, 21)
            a_mix, score_mix = _mix_with_uniform(best_seq, t_grid)
            if score_mix < best_score:
                best_score = score_mix
                best_seq = a_mix.copy()
                print(f"[{time.time() - start_time:6.2f}s] uniform‑mix improve: score={score_mix:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
