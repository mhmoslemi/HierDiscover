import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex; but we can use simpler renorm
    v = np.maximum(v, 0.0)
    s = v.sum()
    n = v.shape[0]
    if s > 0:
        return v / s
    else:
        return np.full_like(v, 1.0/n)

def _conv_self(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n - 1]
    return conv

def _subgrad_max_conv(a, idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end+1)
        j_range = idx - i_range
        grad[i_range] = 2.0 * a[j_range]
    return grad

def _run_subgradient_descent(a0, n, deadline, start_time, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx = int(np.argmax(conv))
        maxc = conv[idx]
        score = 2.0 * n * maxc
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it}, score={score:.6f}")
        grad = _subgrad_max_conv(a, idx)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)
    return best_a, best_score

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0/new_n)
    return up

def _run_targeted_local_search(a0, n, deadline, start_time):
    # copied from previous code (maybe adjust thresholds)
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)
        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

            a = a_prop
            cur_score = new_score

                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve iter {it}, score={new_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Subgradient‑descent optimizer for the max‑convolution objective.
    Starts from the previous best construction (if any), up‑samples to
    length 9600, runs a long subgradient descent on the exact L∞
    objective (max convolution), and finishes with a targeted mass‑shifting
    local search. The whole run respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Load previous construction if available and up‑sample to final length
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    n_final = 9600

    # Prepare initial candidate(s)
    init_candidates = []
    if starter is not None:
        init_candidates.append(_upsample(starter, n_final))
    # add a few random Dirichlet starts
    for _ in range(3):
        rand = np.random.rand(n_final)
        rand /= rand.sum()
        init_candidates.append(rand)

    # Allocate most of the time to subgradient descent
    total_time = deadline - start_time
    grad_time = total_time * 0.75
    local_time = total_time - grad_time
    grad_deadline = start_time + grad_time

    best_seq = None
    best_score = float('inf')
    # Run subgradient descent on each candidate within the remaining grad budget
    # share the budget equally among candidates (or until time runs out)
    per_candidate_time = (grad_deadline - time.time()) / max(1, len(init_candidates))
    for idx, a0 in enumerate(init_candidates):
        if time.time() >= grad_deadline:
            break
        sub_deadline = time.time() + per_candidate_time
        a_opt, score_opt = _run_subgradient_descent(
            a0, n_final, sub_deadline, start_time, step0=0.6
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] candidate {idx} improved: score={score_opt:.6f}")

    # If still time, continue refining the best sequence for the remaining gradient budget
    if best_seq is not None and time.time() < grad_deadline:
        a_ref, score_ref = _run_subgradient_descent(
            best_seq, n_final, grad_deadline, start_time, step0=0.4
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final refinement improved: score={score_ref:.6f}")

    # Local‑search phase
    if best_seq is not None and time.time() < deadline:
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_final, deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local search improved: score={score_loc:.6f}")

    # Fallback in case nothing was produced
    if best_seq is None:
        best_seq = np.full(n_final, 1.0 / n_final)

    # Final normalisation (tiny drift may happen)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]

Subgradient‑descent optimiser for the max‑convolution objective.

The algorithm:
1. Load a previous construction (if any) and up‑sample it to the final
   length (9600).  Additionally generate a few random Dirichlet starts.
2. Run a pure subgradient descent on the exact L∞ objective
   (2·n·max(a*a)) for each start, sharing the majority of the 1000 s budget.
   The subgradient is computed analytically for the active convolution index.
   After each update the vector is projected back onto the probability simplex.
3. Refine the current best sequence with a short additional subgradient run.
4. Finish with the targeted “peak‑shaving’’ local search that moves mass
   from indices heavily contributing to the maximal convolution entry to
   low‑contribution indices (Metropolis acceptance with annealed temperature).

All operations respect the non‑negative and sum‑to‑one constraints, so the

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    n = v.shape[0]
    if s > 0.0:
        return v / s
    else:
        return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Subgradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _subgrad_max_conv(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Gradient of f_k(a)=∑_i a_i a_{k-i} for the active index k=idx.
    Derivative: (∂/∂a_i) f_k = 2·a_{k-i} for all i where k-i is in range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1)
        j_range = idx - i_range            # satisfies 0 ≤ j_range < n
        grad[i_range] = 2.0 * a[j_range]
    return grad

# Subgradient descent on the exact L∞ objective
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.5) -> tuple[np.ndarray, float]:

    Euclidean subgradient descent with projection onto the simplex.
    Returns the best sequence seen and its surrogate score 2·n·max(conv).
    """

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx = int(np.argmax(conv))
        maxc = conv[idx]
        score = 2.0 * n * maxc

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad iter {it}, "
                  f"score={score:.6f}")

        grad = _subgrad_max_conv(a, idx)

        # diminishing step size (square‑root schedule)
        step = step0 / math.sqrt(it)

        # Euclidean update + simplex projection
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score

# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
