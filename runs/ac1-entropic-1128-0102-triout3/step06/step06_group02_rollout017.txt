"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the global ``height_sequence_1`` if it exists.
2. Build a large “Sidon‑like’’ set of indices (all pairwise sums distinct)
   and place equal mass on these positions – this often yields a very small
   max‑convolution because each convolution coefficient receives at most two
   products.
3. Pick the best of the warm‑start and the Sidon candidate.
4. Refine the chosen vector by (a) projected sub‑gradient descent on the
   true max‑convolution and (b) a long simulated‑annealing walk that moves
   small fractions of mass between two coordinates.
   The whole process respects the supplied time budget (default 1000 s).

The Sidon construction is cheap (O(m²) for a set of size *m*).  The
sub‑gradient phase uses a diminishing step size and momentum; the annealing
phase accepts downhill moves and occasional uphill moves with a geometric
cooling schedule.  All operations stay on the probability simplex
(i.e. the vector stays non‑negative and sums to 1), which leaves the
``evaluate_sequence`` value invariant under scaling.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.2,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # momentum term
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.3,
                   end_temp: float = 1e-7) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    it = 0

    while time.time() < deadline:
        it += 1
        # geometric temperature schedule (slow cooling)
        T = start_temp * (end_temp / start_temp) ** (it / 1e7)  # very slow cooling

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 5 % of the mass at i (small steps improve stability)
        eps = a[i] * np.random.rand() * 0.05
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sidon‑like set construction (random greedy)
# ----------------------------------------------------------------------
def _random_sidon_set(n: int, target_m: int, rng: np.random.Generator,
                      max_trials_per_insert: int = 5000) -> np.ndarray:
    """
    Try to construct a set S ⊂ {0,…,n‑1} of size ``target_m`` such that all
    unordered sums i+j (i≤j) are distinct.  Returns the array of indices or
    an empty array if the attempt failed.
    """
    S = []
    sums = set()
    trials = 0
    while len(S) < target_m and trials < max_trials_per_insert * target_m:
        x = int(rng.integers(0, n))
        if x in S:
            trials += 1
            continue
        # check for sum collisions
        coll = False
        for y in S:
            if (x + y) in sums:
                coll = True
                break
        if coll or (2 * x) in sums:
            trials += 1
            continue
        # accept x
        for y in S:
            sums.add(x + y)
        sums.add(2 * x)
        S.append(x)
        trials = 0        # reset after a success
    return np.array(S, dtype=int)

        target_deadline = start + time_limit * (target - 140 + 1) / (max_target - 140 + 1)
        while time.time() < target_deadline:
            S = _random_sidon_set(n, target, rng)
            if S.size < target:      # failed to hit the target size
                continue
            a = np.zeros(n, dtype=float)
            a[S] = 1.0 / S.size
            score = 2.0 * n * _conv_self(a).max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start:6.2f}s] Sidon candidate size={S.size}, score={score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, first tries a Sidon‑like
    construction, then refines the best candidate with sub‑gradient descent and
    simulated‑annealing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Warm start from a previous run if available
    # --------------------------------------------------------------
    try:
        starter_raw = np.asarray(height_sequence_1, dtype=float)
        if starter_raw.ndim == 1 and starter_raw.size > 0:
            starter = starter_raw / starter_raw.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # Fixed length for the search – 9600 matches the previous best.
    # --------------------------------------------------------------
    n = 9600

    # --------------------------------------------------------------
    # 1) Sidon‑like candidate search (≈20 % of the total budget)
    # --------------------------------------------------------------
    sidon_budget = max(30.0, 0.2 * (deadline - start_time))
    sidon_seq, sidon_score = _build_best_sidon_candidate(
        n, np.random.default_rng(seed + 12345), sidon_budget
    )

    # --------------------------------------------------------------
    # 2) Gather all available candidates
    # --------------------------------------------------------------
    candidates = []
    # a) previous best (if any)
    if starter is not None:
        if starter.size != n:
            # upscale / downscale to the target length
            old_n = starter.size
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            starter_up = np.interp(new_x, old_x, starter)
            starter_up /= starter_up.sum()
            starter = starter_up
        candidates.append(("starter", starter, 2.0 * n * _conv_self(starter).max()))
    # b) Sidon candidate (if we managed to build one)
    if sidon_seq is not None:
        candidates.append(("sidon", sidon_seq, sidon_score))

    # pick the best initial vector
    best_name, best_seq, best_score = min(candidates, key=lambda t: t[2])
    print(f"[{time.time() - start_time:6.2f}s] initial best: {best_name} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Refinement phase (sub‑gradient + annealing)
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # split remaining time: 60 % sub‑gradient, 40 % annealing
        subgrad_time = remaining * 0.6
        anneal_time = remaining * 0.4

        # ----- sub‑gradient descent -----
        sub_deadline = time.time() + subgrad_time
        a_opt, score_opt = _run_subgradient_gd(
            best_seq, n, sub_deadline, start_time,
            step0=0.2, beta=0.9, max_iter=10 ** 7
        )
        if score_opt < best_score:
            best_score, best_seq = score_opt, a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] post‑subgrad improve: score={best_score:.6f}")

        # ----- annealing -----
        anneal_deadline = time.time() + anneal_time
        a_ref, score_ref = _run_annealing(
            best_seq, n,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=0.3, end_temp=1e-7
        )
        if score_ref < best_score:
            best_score, best_seq = score_ref, a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] post‑anneal improve: score={best_score:.6f}")

        # If we have spent very little time (e.g. because we are close to deadline)
        # break to avoid an endless loop.
        if remaining < 5.0:
            break

    # --------------------------------------------------------------
    # 4) Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list
    return [float(x) for x in best_seq]
