def _subgradient_max(a, conv, n, m=1):
    max_val = conv.max()
    # find indices of top m conv values above threshold (based on max)
    # Use argpartition for speed
    # we want m largest values
    if conv.size <= m:
        top_idx = np.arange(conv.size)
    else:
        # argpartition returns unsorted but we can sort after
        part = np.argpartition(-conv, m-1)[:m]
        top_idx = part[np.argsort(-conv[part])]  # sorted descending
    g = np.zeros_like(a)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    g /= m
    return g

"""
Improved multi‑scale optimisation for the step‑function attack.

The algorithm proceeds in three resolution stages (≈2400 → 4800 → 9600 points).
For each stage we

1. initialise the current vector (upsampled best solution so far or a random start);
2. run a smooth Adam optimiser on a log‑softmax parametrisation, gradually
   increasing the smoothing parameter α;
3. polish the result with a projected sub‑gradient descent that works on the
   true (non‑smooth) max‑convolution;
4. finish with a short simulated‑annealing “mass‑transfer’’ refinement.

All steps are time‑aware and stop as soon as the global deadline (≈1000 s) is
reached.  The final vector is normalised to sum = 1, which is sufficient because
`evaluate_sequence` is homogeneous under scaling (as long as no entry exceeds
the internal 1000‑cap).  The routine returns the best sequence found.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a) = (1/α)·log Σ_k exp(α·conv_k)
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation, preserves the sum)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Objective used internally (identical to evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   alphas,
                   iters_per_alpha: int,
                   start_time: float,
                   deadline: float,
                   eps: float = 1e-16) -> tuple[np.ndarray, float]:
    """Smooth Adam optimisation (softmax parametrisation)."""
    n = a0.shape[0]
    y = np.log(np.maximum(a0, eps))
    fft_len = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = None
    best_score = np.inf

    for alpha in alphas:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a if best_a is not None else a0, best_score

            # softmax
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # smoothed objective
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha                       # = smooth max
            obj = 2.0 * n * f_val                     # (sum a = 1)

            # gradient wrt a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient wrt logits y
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact‑objective check
            if i % 200 == 0:
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={score:.6f}")

    if best_a is None:
        # fallback – use current a
        best_a = a
        best_score = _objective(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (average over top‑k peaks)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray, top_k: int = 1) -> np.ndarray:
    """Return a sub‑gradient of max(conv) (averaged over the top‑k peaks)."""
    n = a.shape[0]
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        # select the indices of the top‑k values
        part = np.argpartition(-conv, top_k - 1)[:top_k]
        idx = part[np.argsort(-conv[part])]   # sorted descending (optional)
    g = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    g = g / len(idx)                 # average over the selected peaks
    return g


# ----------------------------------------------------------------------
# Projected gradient descent using the exact max‑convolution
# ----------------------------------------------------------------------
def _run_projected_gradient(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           beta: float = 0.94,
                           step0: float = 0.25,
                           top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Simple projected (Euclidean) gradient descent with Nesterov‑style momentum.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        g = _subgradient_max(a, conv, top_k=top_k)

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """Tiny‑scale annealing that moves a random ε‑fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30          # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score
