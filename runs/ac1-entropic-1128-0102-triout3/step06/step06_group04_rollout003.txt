"""
Hybrid optimisation for the self‑convolution peak.

The search proceeds in stages over increasing lengths.
For each length we generate a small diverse pool of
parameterised shapes (sin‑powers, Gaussians, Beta‑pdfs,
random Dirichlet).  Each candidate is refined by a
projected Adam optimiser on a smooth log‑sum‑exp
surrogate of the max‑norm.  Afterwards a classic LP‑based
mass‑increase step is applied repeatedly (while the
problem size is modest) and a long projected sub‑gradient
descent follows.  Finally a cheap hill‑climbing/
simulated‑annealing phase polishes the best sequence.
The best distribution found is up‑sampled and used as a
warm‑start for the next larger length.  The routine respects
the 1000 s time budget and returns the best sequence obtained.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation used for internal comparison.
    Equivalent to evaluate_sequence when sum(a)=1.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimisation on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        L_λ(a) = λ·log Σ exp(conv/λ)
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # convolution of weight vector with reversed a gives the gradient components
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central slice of length n
    return grad


def _run_adam_opt(a0: np.ndarray,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False,
                  lam0: float = 0.5,
                  lam_decay: float = 0.85,
                  lam_decay_interval: int = 2500) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the log‑sum‑exp surrogate.
    Returns the best solution seen before slice_deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = lam0          # λ = lam_factor * max(conv)
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-14)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size decays slowly with iteration count
        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap” step (classic LP attack)
# ----------------------------------------------------------------------


_MAX_LP_N = 4096  # avoid quadratic blow‑up beyond this size


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A for constraints (f * g)_k ≤ M.
    Entry A_{k,i} = f_i if i ≤ k < i + n, else 0.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    slice_deadline: float,
                    start_time: float,
                    max_iters: int = 12,
                    step: float = 0.5) -> np.ndarray:
    """
    Repeated LP‑based improvement.
    Given a, solves max Σ b subject to b ≥ 0 and a * b ≤ M,
    where M = max(a*a).  The solution b is mixed with a.
    """
    n = a.shape[0]
    if n > _MAX_LP_N:
        return a  # skip LP for large problems

    a_cur = a.copy()
    it = 0
    while time.time() < slice_deadline and it < max_iters:
        it += 1
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ b_i

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 20.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # no mass increase possible
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        cur_score = _objective(a_cur)
        print(f"[{time.time() - start_time:6.2f}s] LP iter={it} step={step:.2f} score={cur_score:.6f}")

    return a_cur

def _subgradient_descent(a0: np.ndarray,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(a*a) with projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.shape[0]
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            # the symmetric term (same contribution, already counted)
            # because convolution is symmetric the above covers both sides

        step = 0.4 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Subgrad it={it} score={cur_score:.6f}")

    return best_a, best_score

def _hill_climb(a0: np.ndarray,
                slice_deadline: float,
                start_time: float,
                max_moves: int = 200000,
                move_frac: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random fraction of mass from i→j
    and keep the move if the objective improves.
    """
    a = a0.copy()
    best_score = _objective(a)
    n = a.shape[0]
    it = 0
    while time.time() < slice_deadline and it < max_moves:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        eps = a[i] * np.random.rand() * move_frac
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        # keep on simplex automatically (sum unchanged)
        score = _objective(a_prop)
        if score < best_score:
            best_score = score
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] HillClimb move={it} score={score:.6f}")

    return a, best_score

def _shape_sin_pow(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p, non‑negative."""
    x = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    x = np.maximum(x, 0.0) ** p
    s = x.sum()
    if s == 0:
        x = np.full(n, 1.0 / n)
    else:
        x /= s
    return x


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, truncated to [0,1]."""
    centre = (n - 1) / 2.0
    x = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    s = x.sum()
    if s == 0:
        x = np.full(n, 1.0 / n)
    else:
        x /= s
    return x


def _shape_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta(α,β) pdf on a discrete grid."""
    t = (np.arange(n) + 0.5) / n
    # use log‑beta to avoid overflow for extreme parameters
    logx = (a - 1) * np.log(t) + (b - 1) * np.log(1 - t)
    x = np.exp(logx - np.max(logx))
    s = x.sum()
    if s == 0:
        x = np.full(n, 1.0 / n)
    else:
        x /= s
    return x


def _random_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet draw (α_i = 1)."""
    x = np.random.rand(n)
    s = x.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return x / s


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:

    and a few parameterised families.
    """

    variants = [base_seq.copy()]

    # sin‑power exponents
    for p in (0.5, 1.0, 2.0, 4.0):
        variants.append(_shape_sin_pow(n, p))

    # Gaussians with different widths
    for sigma in (n / 6.0, n / 4.0, n / 8.0):
        variants.append(_shape_gaussian(n, sigma))

    # Beta distributions (various skewness)
    for a, b in ((0.5, 0.5), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)):
        variants.append(_shape_beta(n, a, b))

    # Random Dirichlet
    variants.append(_random_dirichlet(n))

    return variants


# ----------------------------------------------------------------------
# Upsampling utility
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() within the given time budget.

    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Global warm‑start (use previously found solution if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – start modest, grow, finish on the longest useful size.
    # ------------------------------------------------------------------
    # Choose a set of lengths (powers of two).  Shuffle so we do not get
    # stuck in a deterministic order.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)

    best_seq = None
    best_score = float('inf')
    global_a = starter  # used for warm‑starting the next scale

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm‑start + several variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # --------------------------------------------------------------
        # Slice of remaining budget for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate roughly 12 % of the remaining time per length, but never less than 5 s
        slice_deadline = time.time() + max(5.0, remaining * 0.12 / (len(lengths) - idx))

        # --------------------------------------------------------------
        # Adam optimisation on several starts
        # --------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # optionally force symmetry for a fraction of runs
            sym = (np.random.rand() < 0.3)
            a_opt, sc_opt = _run_adam_opt(start_vec,
                                          slice_deadline,
                                          start_time,
                                          symmetrize=sym,
                                          lam0=0.5,
                                          lam_decay=0.85,
                                          lam_decay_interval=2500)
            if sc_opt < best_local_score:
                best_local_score = sc_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam local best n={n} score={sc_opt:.6f}")

        if best_local_a is None:
            continue

        # --------------------------------------------------------------
        # LP “swap” phase (only for moderate lengths)
        # --------------------------------------------------------------
        lp_deadline = min(slice_deadline, time.time() + 8.0)
        best_local_a = _run_lp_improve(best_local_a,
                                       lp_deadline,
                                       start_time,
                                       max_iters=12,
                                       step=0.5)
        best_local_score = _objective(best_local_a)

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after LP n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient refinement
        # --------------------------------------------------------------
        if time.time() < deadline:
            sg_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_sg, sc_sg = _subgradient_descent(best_local_a, sg_deadline, start_time)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after SubGrad n={n} score={sc_sg:.6f}")

        # --------------------------------------------------------------
        # Hill‑climbing / annealing polishing for this length
        # --------------------------------------------------------------
        if time.time() < deadline:
            hc_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_hc, sc_hc = _hill_climb(best_local_a,
                                      hc_deadline,
                                      start_time,
                                      max_moves=150000,
                                      move_frac=0.12)
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after HillClimb n={n} score={sc_hc:.6f}")

        # remember for the next up‑sampling step
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final heavy polishing on the largest length we managed to keep.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.shape[0]
        final_deadline = deadline - 0.5
        # a longer Adam run with symmetry enforced (often helpful)
        a_final, sc_final = _run_adam_opt(best_seq,
                                          final_deadline,
                                          start_time,
                                          symmetrize=True,
                                          lam0=0.4,
                                          lam_decay=0.9,
                                          lam_decay_interval=2000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final Adam improve score={sc_final:.6f}")

        # a final hill‑climb
        if time.time() < deadline - 1.0:
            hc_deadline = deadline - 0.1
            best_seq, best_score = _hill_climb(best_seq,
                                               hc_deadline,
                                               start_time,
                                               max_moves=200000,
                                               move_frac=0.08)
            print(f"[{time.time() - start_time:6.2f}s] Final HillClimb improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return even if everything failed.
    # ------------------------------------------------------------------
    if best_seq is None:
        # uniform distribution of modest size as a safe fallback
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list for the evaluator (it will clip/normalise again)
    return [float(x) for x in best_seq]
