"""
Optimise a non‑negative step‑function for the evaluation metric

    score = 2·n·max(conv(a))

where conv(a) = a * a is the linear self‑convolution and a is
normalised to sum = 1.  The algorithm uses three stages:

* Stage 0 – many short runs on a modest size (n≈1024) with a
  *smooth* gradient descent.  The objective max(conv) is approximated
  by the log‑sum‑exp (soft‑max) with a slowly increasing temperature
  (γ).  The gradient of this smooth surrogate is obtained by a single
  FFT‑based correlation, which is cheap (≈2 FFTs per iteration).

* Stage 1 – up‑sample the best vector to n≈4096 and run a longer
  smooth descent.

* Stage 2 – up‑sample again to n≈12288 and continue the descent.
  A final short simulated‑annealing (mass‑transfer) refinement is
  performed if time permits.

All updates respect the simplex constraint via Euclidean projection.
The code respects the 1000 s budget, prints progress, and returns the
best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation‑based gradient for the smooth max‑convolution objective
# ----------------------------------------------------------------------
def _corr_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i w_{i+j} for all j (the gradient of the
    smoothed max w.r.t. a, *without* the outer factor 2·n).
    Both a (length n) and w (length 2n‑1) are real.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse w
    m = n + w_rev.shape[0] - 1          # = 3n‑2
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(Af * Wf, n=fft_len)[:m]
    start = w_rev.shape[0] - 1          # 2n‑2
    idx = start - np.arange(n)          # descending slice
    grad = 2.0 * conv[idx]              # 2·Σ a_i w_{i+j}
    return grad


# ----------------------------------------------------------------------
# Smooth gradient descent on the log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    *,
    beta: float = 0.9,
    step0: float = 0.3,
    gamma_start: float = 0.5,
    gamma_end: float = 200.0,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a)) using a smoothed objective
    (log‑sum‑exp) and Euclidean projection onto the simplex.
    Returns the best encountered vector and its true (unsmoothed) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # a simple linear schedule for γ, reaching gamma_end after ~50 k iters
    gamma_target_iters = 50000

    while time.time() < deadline and it < max_iter:
        it += 1

        # self‑convolution and true max‑value
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # temperature (γ) schedule
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, it / gamma_target_iters)

        # soft‑max weights for the current convolution values
        # (shift by max to avoid overflow)
        shifted = conv - max_val
        exp_vals = np.exp(gamma * shifted)
        w = exp_vals / exp_vals.sum()                     # shape 2n‑1

        # gradient of the smoothed objective (including the outer 2·n factor)
        grad_corr = _corr_gradient(a, w)                  # = 2·∑ a_i w_{i+j}
        grad = 2.0 * n * grad_corr                         # final Euclidean gradient

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size (√t schedule)
        step = step0 / math.sqrt(it)

        # projected gradient step
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    *,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Light simulated annealing that repeatedly transfers a small ε
    from a source index (chosen proportionally to its mass) to a random
    destination.  Acceptance follows the Metropolis rule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
