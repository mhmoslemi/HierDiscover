"""
Hybrid search for a low‑L∞ self‑convolution step function.

Key ideas over the previous version
----------------------------------
1. **No forced symmetry** – the optimal distribution is not known to be symmetric,
   therefore we keep the shape fully asymmetric throughout the whole run.

2. **Richer warm‑starts** – apart from sinusoidal and triangular seeds we also
   inject monotone, exponential and power‑law profiles which are known to give
   low values of the mirror‑convolution `∑ a_i a_{n−1−i}`.

3. **Longer length schedule** – we continue the multi‑scale warm‑start up to
   65536 entries, giving the optimiser more degrees of freedom.

4. **More aggressive Adam** – a larger base step and a slower λ‑decay give the
   optimiser more room to explore early on while still converging to the true
   max‑convolution.

5. **Refinement without symmetrisation** – both deterministic sub‑gradient
   descent and the final annealing keep the distribution asymmetric.

All updates are projected onto the probability simplex (∑a_i = 1, a_i ≥ 0)
so the `evaluate_sequence` objective is scale‑invariant.
The routine respects the supplied time‑budget and returns the best sequence
found before the deadline.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -----------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero or negative – uniform fallback
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions -------------------------------------------------
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)          # numerical safety
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using the FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the soft‑max L∞ surrogate (log‑sum‑exp) -------------------
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth log‑sum‑exp surrogate  L_λ(a) = λ·log Σ exp(conv/λ).
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length = 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]        # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution -----------------------------
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of  max_k (a * a)[k]  at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)               # length = 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val

# Adam optimiser on the smooth surrogate --------------------------------
# ----------------------------------------------------------------------

                  n: int,

    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector seen and its exact score (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.3            # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.99           # slower decay – more stable optimisation
    lam_min_factor = 1e-9

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base step (slightly larger than the original version)
        base_step = 1.0 / math.sqrt(it + 1.0)

        a -= base_step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent (exact max) ------------------------
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    step0 = 0.7
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing) ----------------------------
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300000) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing: move a small ε from a source index to a destination.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators -----------------------------------------------
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors derived from a base
    distribution.  All returned vectors are normalised to sum = 1.
    """
    variants = []
    n = base_seq.shape[0]

    # 1. warm‑start copy
    variants.append(base_seq.copy())

    # 2. random Dirichlet (α=1)
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    # 3. random Dirichlet (α=0.3) – gives a sparser, heavy‑tailed start
    rnd_sparse = np.random.gamma(0.3, 1.0, size=n)
    rnd_sparse /= rnd_sparse.sum()
    variants.append(rnd_sparse)

    # 4. positive sine shape
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 5. monotone decreasing linear shape (a_i ∝ n‑i)
    lin_dec = np.arange(n, 0, -1, dtype=float)
    lin_dec /= lin_dec.sum()
    variants.append(lin_dec)

    # 6. exponential decay
    exp_decay = np.exp(-np.arange(n) / (n / 10.0))
    exp_decay /= exp_decay.sum()
    variants.append(exp_decay)

    # 7. power‑law (inverse‑sqrt) – a_i ∝ 1/√(i+1)
    power = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    power /= power.sum()
    variants.append(power)

    # 8. arcsine‑like shape (1/√(t·(1‑t))) – classic benchmark
    t = (np.arange(n) + 0.5) / n
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    variants.append(arc)

    return variants


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # tiny safety margin – the runner will kill us after the budget anyway
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Try to start from a previously discovered construction, if present
    # ------------------------------------------------------------------
    global_a = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                global_a = starter.copy()
    except Exception:
        pass

    # Length schedule – start small, end big.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]
    random.shuffle(lengths)   # avoid bias if we run out of time early

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm start / several diverse variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # Time slice for the current length.
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # Give later (larger) lengths a larger fraction of the remaining budget.
        slice_frac = 0.12 / (len(lengths) - idx)   # roughly 12 % total
        slice_deadline = time.time() + max(6.0, remaining * slice_frac)

        # ------------------------------------------------------------------
        # Adam optimisation phase (several starts)
        # ------------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            a_opt, score_opt = _run_adam_opt(start_vec, n, slice_deadline, start_time)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # Record any global improvement from the Adam phase
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Deterministic sub‑gradient descent (refinement)
        # --------------------------------------------------------------
        if time.time() < deadline:
            subgrad_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.10)
            a_sub, score_sub = _run_subgradient(best_local_a, n, subgrad_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad n={n} score={score_sub:.6f}")

        # --------------------------------------------------------------
        # Final annealing polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.20)
            a_ref, score_ref = _run_annealing(
                best_local_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=250000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # Remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list – the evaluator will clamp / normalise as needed.
    return [float(x) for x in best_seq]
