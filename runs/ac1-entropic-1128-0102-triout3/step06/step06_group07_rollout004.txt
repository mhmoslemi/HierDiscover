def _smooth_gradient(a, p):
    conv = _conv_self(a)
    # Add epsilon to avoid zero:** We can treat conv^p safe for zero (zero gradient).
    conv_p = conv ** p
    sum_p = conv_p.sum()
    if sum_p == 0:
        weights = np.zeros_like(conv)
    else:
        weights = conv ** (p - 1) / (sum_p ** ((p - 1)/p))
    # gradient = 2 * correlation(weights, a) slice
    grad_full = np.correlate(weights, a, mode='full')

    n = a.shape[0]
    grad = 2.0 * grad_full[n-1:2*n-1]
    return grad

def _run_smooth_descent(a0, n, deadline, start_time, p_schedule=None, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        # p schedule: maybe p = min(p_max, p_start * (it ** factor)), or predetermined schedule.
        if p_schedule is not None:
            p = p_schedule(it)
        else:
            p = 32 # default

        conv = _conv_self(a)
        cur_score = 2 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth improve: n={n}, iter={it}, p={p}, score={cur_score:.6f}")

        # gradient from smooth p-norm
        grad = _smooth_gradient(a, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1)  # just decreasing step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

        # enforce symmetry optionally
        if symmetry:
            a = (a + a[::-1]) / 2

    return best_a, best_score

"""
Propose a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1.  Use a smooth “soft‑max” approximation of the max‑convolution:
        M(p, a) = (1/p)·log Σ_k exp(p·(a*a)[k]).
    Its gradient w.r.t. `a` can be written as a correlation
    between the soft‑max weights and `a`.  This gives a smooth,
    well‑behaved direction even when the true max is flat.

2.  Optimise the smooth objective with Adam (adaptive step‑size) on the
    probability simplex (Σ a_i = 1, a_i ≥ 0).  After each update the
    vector is projected onto the simplex and optionally symmetrised
    (a_i ← (a_i+a_{n‑1‑i})/2) – the optimum is known to be symmetric.

3.  Run several short random‑restart phases on a modest size
    (n≈1024).  The best vector is up‑sampled (linear interpolation)
    to larger sizes (≈4096 and ≈12288) and the Adam optimisation is
    continued with a schedule that slowly increases the soft‑max
    temperature `p` up to a large value (≈64), thereby approaching the
    true max‑convolution.

4.  Finally a light simulated‑annealing phase moves a small
    fraction of mass between two positions; this can escape the
    remaining local minima.

All phases respect the overall time budget (`budget_s`, default 1000 s).
The function returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation w * a using FFT (returns vector of length n)
# ----------------------------------------------------------------------
def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation of w (length 2n‑1) with a (length n)."""
    n = a.shape[0]
    L = 1 << ((len(w) + n - 1).bit_length())
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(Wf * Af, n=L)[: len(w) + n - 1]
    # slice that corresponds to indices i = 0..n‑1
    return corr[n - 1: 2 * n - 1]


# ----------------------------------------------------------------------
# Smooth gradient of the soft‑max approximation of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, p: float) -> np.ndarray:
    """
    Return an (approximate) gradient of
        M(p,a) = (1/p)·log Σ exp(p·conv[k])
    w.r.t. a, where conv = a*a.
    """
    conv = _conv_self(a)

    # soft‑max weights, numerically stable
    max_c = conv.max()
    exp_w = np.exp(p * (conv - max_c))
    sum_exp = exp_w.sum()
    if sum_exp == 0.0:
        w = np.zeros_like(conv)
    else:
        w = exp_w / sum_exp

    # gradient = 2 * correlation(w, a)
    grad = 2.0 * _corr_fft(w, a)
    return grad, conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with optional symmetry
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     p_start: float = 1.0,
                     p_max: float = 64.0,
                     p_inc: float = 0.0005,
                     lr: float = 0.1,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     symmetry: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate.
    Returns the best sequence seen and its exact (non‑smooth) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        p = min(p_max, p_start + p_inc * it)

        grad, conv_max = _smooth_gradient(a, p)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (math.sqrt(it) + 1.0)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto the simplex
        a = _project_simplex(a)

        # optional symmetry enforcement
        if symmetry:
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

        # exact score (no smoothing)
        cur_score = 2.0 * n * conv_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, p={p:.2f}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) and re‑normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Light simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Simple mass‑transfer annealing."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportionally to its mass (more mass → more moves)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Phase 0 – many short runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(25.0, 0.09 * (deadline - time.time()))   # ~9 % of budget
    slice_deadline = time.time() + slice_time

    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # small randomisation of Adam's hyper‑parameters for diversity
        lr = np.random.uniform(0.06, 0.12)
        p_inc = np.random.uniform(0.0003, 0.0008)

        a_opt, sc = _run_adam_smooth(
            a0, n0, slice_deadline, start_time,
            lr=lr,
            p_inc=p_inc,
            symmetry=True,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] phase 0 improve: "
                  f"restart={r}, score={sc:.6f}")

    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Phase 1 – upsample to medium size (≈4096) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)          # ~35 % of the rest
    grad_deadline1 = time.time() + grad_time1

    a1_opt, sc1 = _run_adam_smooth(
        a1, n1, grad_deadline1, start_time,
        lr=0.09,
        p_inc=0.0006,
        symmetry=True,
    )
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] phase 1 improve: score={sc1:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to large size (≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)          # ~30 % of the rest
    grad_deadline2 = time.time() + grad_time2

    a2_opt, sc2 = _run_adam_smooth(
        a2, n2, grad_deadline2, start_time,
        lr=0.07,
        p_inc=0.0008,
        symmetry=True,
    )
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] phase 2 improve: score={sc2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final annealing refinement (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, sc_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
