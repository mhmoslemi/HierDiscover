Optimisation of a non‑negative step function minimizing the
`evaluate_sequence` upper bound.  The routine runs four stages:

* Stage 0 – many short Adam‑optimisation runs on a small grid (n≈1024)
  with randomly chosen Dirichlet initialisation and a run‑time based
  stop‑condition.
* Stage 1 – up‑sample the best vector to n≈4096 and continue Adam.
* Stage 2 – up‑sample to n≈12288 and run a longer Adam optimisation.
* Stage 3 – a prolonged simulated‑annealing phase that performs
  mass‑transfer moves with a geometric temperature schedule.
All stages respect the overall time budget and keep the vector
normalised to sum = 1 via projection onto the probability simplex.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution a*a using real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Return an (averaged) sub‑gradient of max(conv).
    For each index k attaining (almost) the maximum we add the
    contribution 2*a_{k-i} (when valid) to the i‑th component.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad

def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.3,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for F(a) = 2 * n * max(conv(a)) under a ≥ 0, Σ a = 1.
    Returns best vector seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a small fraction of mass from a
    source index to a destination index.  Acceptance follows the
    Metropolis rule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    """

    if score1 < best_score:

        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

            best_seq,
            best_seq.shape[0],

    best_seq = np.asarray(best_seq, dtype=float)

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]

def _lp_improve(a: np.ndarray, n: int, M: float, deadline: float) -> np.ndarray:
    """
    Solve LP: maximize sum(b) subject to b >= 0, (a * b) <= M.
    Returns the normalised b (sum = 1) if a feasible solution with sum>0
    is found; otherwise returns the original a.
    """
    # Compute number of constraints
    conv_len = 2 * n - 1
    # Build sparse matrix A where A[k,i] = a[i] if 0 <= k-i < n else 0
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows for which column i appears: k = i .. i+n-1
        start = i
        end = i + n  # exclusive
        rows.extend(np.arange(start, end))
        cols.extend([i] * n)
        data.extend([a[i]] * n)
    A = coo_matrix((data, (rows, cols)), shape=(conv_len, n))
    # upper bound vector
    b_vec = np.full(conv_len, M, dtype=float)

    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n, dtype=float)

    # bounds for each variable: (0, None)
    bounds = [(0, None)] * n

    # Solve with time limit check
    # SciPy's linprog does not accept time limit directly; we simply check after solving.
    # For safety we limit the total number of constraints to avoid extremely long solves.
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if res.success and res.x is not None:
        b_opt = res.x
        s = b_opt.sum()
        if s > 0:
            return b_opt / s
    # Fall back
    return a.copy()

"""
Optimisation of a non‑negative step function that minimises the
`evaluate_sequence` upper bound.

The routine works in four stages while respecting a 1000 s time budget:

* Stage 0 – many short Adam‑optimisation runs on a modest grid (n≈1024)
  with random Dirichlet initialisations and random hyper‑parameters.
* Stage 1 – up‑sample the best vector to n≈4096 and continue Adam.
  After this, a tiny linear‑programming refinement (maximise Σb under
  a·b≤M) is attempted.
* Stage 2 – up‑sample to n≈12288 and run a longer Adam optimisation.
* Stage 3 – a final simulated‑annealing phase that transfers small
  amounts of mass between coordinates using a geometric temperature
  schedule.

All vectors are always projected onto the probability simplex
(Σa=1, a_i≥0) so the scaling invariance of the objective is respected.
The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray,
                     conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """
    Averaged sub‑gradient of max(conv).  For every (almost) maximal
    index k the contribution 2·a_{k-i} is added to the i‑th component.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]
    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.3,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for  F(a)=2·n·max(conv(a))  under a≥0, Σa=1.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_max(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step * m_hat)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing that moves a random fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _lp_improve(a: np.ndarray,
                n: int,
                M: float,
                deadline: float) -> np.ndarray:
    """
    Linear‑programming refinement.
    maximise Σb subject to  b≥0  and  a∗b ≤ M (elementwise).
    The returned vector is normalised to sum = 1.
    """
    time_left = deadline - time.time()
    if time_left < 1.0:          # not enough time for a solve
        return a

    conv_len = 2 * n - 1

    # Build the sparse matrix that represents the linear map b ↦ a∗b.
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel().astype(np.int32)
    cols = np.repeat(i, n).astype(np.int32)
    data = np.repeat(a, n).astype(np.float64)

    A = coo_matrix((data, (rows, cols)), shape=(conv_len, n))
    b_vec = np.full(conv_len, M, dtype=np.float64)

    # Objective: maximise Σb  → minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    # Variable bounds: b_i ≥ 0
    bounds = [(0, None)] * n

    try:
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_vec,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception:
        return a

    if res.success and res.x is not None:
        b_opt = res.x
        s = b_opt.sum()
        if s > 0.0:
            return b_opt / s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine follows a multi‑stage scheme
    (many Adam restarts → up‑sampling → LP refinement → final annealing)
    while staying inside the global time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest grid (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.10 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time
    restarts = 30
    for r in range(restarts):
        if time.time() >= deadline:
            break
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            alpha = np.random.uniform(0.2, 4.0)          # concentration for Dirichlet
            a0 = np.random.dirichlet(np.full(n0, alpha))

        lr0 = np.random.uniform(0.15, 0.45)
        beta1 = np.random.uniform(0.85, 0.95)
        beta2 = np.random.uniform(0.995, 0.9995)

        a_opt, score = _run_adam(
            a0, n0, slice_deadline, start_time,
            lr0=lr0, beta1=beta1, beta2=beta2,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to medium size (≈4096) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_adam(
        a1, n1, grad_deadline1, start_time,
        lr0=0.28, beta1=0.92, beta2=0.999,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Small LP refinement on the medium‑size vector
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        conv1 = _conv_self(best_seq)
        M1 = conv1.max()
        a1_lp = _lp_improve(best_seq, n1, M1, deadline)
        score_lp = 2.0 * n1 * _conv_self(a1_lp).max()
        if score_lp < best_score:
            best_score = score_lp
            best_seq = a1_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to large size (≈12288) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_adam(
        a2, n2, grad_deadline2, start_time,
        lr0=0.24, beta1=0.94, beta2=0.9995,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing refinement (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-7,
            steps=600_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
