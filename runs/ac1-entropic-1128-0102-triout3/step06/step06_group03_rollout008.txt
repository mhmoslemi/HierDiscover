def run_subgradient_gd(a0, n, deadline, start_time, step0=0.5):
    a = a0.copy()
    best = a.copy()
    best_obj = np.inf
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad improve: n={n}, iter={it}, score={obj:.6f}")
        # pick random max index if multiple with equal max
        max_idxs = np.where(conv == max_val)[0]
        k = np.random.choice(max_idxs)
        # compute gradient
        grad = np.zeros_like(a)
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        grad[idx] = 2.0 * a[k - idx]
        # step size schedule
        step = step0 / np.sqrt(it)  # can tune
        # mirror descent update:
        a = a * np.exp(-step * grad)
        # renorm to sum=1 (prevent underflow)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return best, best_obj

def _conv_self(a):
    n = a.shape[0]
    m = 2*n - 1
    fft_len = 1 << (m-1).bit_length()
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A*A, n=fft_len)[:m]
    return conv

"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.
The method combines cheap randomized restarts (small problem sizes),
a projected multiplicative sub‑gradient descent on the exact max‑convolution,
and a final stochastic “peak‑shaving’’ local search.
All updates keep the vector on the probability simplex (sum = 1),
so the scaling of the sequence does not affect the score.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto simplex (unused for the multiplicative updates but kept
# for possible fall‑backs)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max) with multiplicative simplex updates
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.3,
                    noise_every: int = 10,
                    noise_scale: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative sub‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # pick a (random) index attaining the maximum
        max_idxs = np.where(conv == max_val)[0]
        k = np.random.choice(max_idxs)

        # gradient of the max entry:
        #   g_i = 2 * a_{k-i}  for i such that 0 ≤ k-i < n, else 0
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # diminishing multiplicative step
        step = step0 / math.sqrt(it)

        # multiplicative (mirror‑descent) update – keeps positivity
        a = a * np.exp(-step * grad)

        # occasional tiny random perturbation to escape plateaus
        if noise_every and it % noise_every == 0:
            noise = (np.random.rand(n) - 0.5) * noise_scale * a
            a = a + noise
            a = np.maximum(a, 0.0)

        # re‑normalise to the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Fixed‑iteration variant used for quick cheap restarts (small n)
# ----------------------------------------------------------------------
def _run_subgrad_gd_fixed_iters(a0: np.ndarray,

            a = np.maximum(a, 0.0)

        if s > 0.0:

        else:

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from indices that contribute most to the
    current maximal convolution entry to indices that contribute little.
    Acceptance follows a slow annealing schedule.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (slow exponential cooling)
    start_T = 5e-2
    end_T = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        cur_score = 2.0 * n * max_val

        # contributions of each index to this maximal entry
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor – proportional to contribution (hot spots)
        if contrib.sum() > 0:
            donor = np.random.choice(n, p=contrib / contrib.sum())
        else:
            donor = np.random.randint(n)

        # recipient – pick from low‑contribution region (bottom 10 %)
        low_thr = np.percentile(contrib, 10)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size > 0:
            rec = np.random.choice(low_idx)
        else:
            rec = np.random.randint(n)

        if donor == rec or a[donor] <= 1e-15:
            continue

        # move up to 40 % of donor mass (random fraction)
        eps = a[donor] * np.random.rand() * 0.4
        a_prop = a.copy()
        a_prop[donor] -= eps
        a_prop[rec]   += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # annealed Metropolis acceptance
        elapsed_frac = (time.time() - start_time) / (deadline - start_time + 1e-12)

        accept = (delta < 0) or (np.random.rand() < math.exp(-delta / (T + 1e-12)))

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    The strategy is:

    2. **Upsample** the best small‑size candidate to the final length (≈9600).
    3. **Fine‑tune** the full‑size candidate with a long multiplicative
       sub‑gradient descent (exact max‑convolution gradient).
    4. **Peak‑shaving local search** that randomly re‑distributes mass from
       heavy‑contribution indices to light‑contribution ones.
    The whole pipeline respects the time budget (≈1000 s) and always
    returns a non‑negative sequence normalised to sum = 1.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load optional starter (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    N_FINAL = 9600                     # final length (same as previous best)
    N_SMALL = 2000                     # size used for cheap restarts
    MAX_RESTARTS = 5                   # number of cheap restarts (adjustable)
    RESTART_ITER = 4000                # iterations per small‑size run
    LOCAL_RESERVE_FRAC = 0.20          # fraction of total time kept for local search
    # ------------------------------------------------------------------
    # Phase 1 – cheap random restarts on a small problem
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for r in range(MAX_RESTARTS):
        if time.time() >= deadline:
            break
        # initialise a random Dirichlet vector (concentration <1 → more variance)
        a0 = np.random.dirichlet(np.full(N_SMALL, 0.5))
        # short sub‑gradient optimisation on the small size
        a_small, score_small = _run_subgrad_gd_fixed_iters(

    # If no restart succeeded (unlikely) fall back to uniform or starter

    # ------------------------------------------------------------------
    time_left = deadline - time.time()

        # reserve a slice of the remaining budget for the final local walk

        a_fine, fine_score = _run_subgrad_gd(
            best_seq, N_FINAL, local_deadline,
            start_time, step0=0.3, noise_every=15, noise_scale=5e-5
        )
        if fine_score < best_score:
            best_score = fine_score
            best_seq = a_fine.copy()
            print(f"[{time.time() - start_time:6.2f}s] fine‑tune improve: "
                  f"score={fine_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted local search (peak shaving)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_local, local_score = _run_targeted_local_search(
            best_seq, N_FINAL, deadline, start_time

        if local_score < best_score:
            best_score = local_score
            best_seq = a_local.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={local_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
