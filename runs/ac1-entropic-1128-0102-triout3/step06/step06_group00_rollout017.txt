Propose a candidate step function solving the minimisation of evaluate_sequence
by a short multiplicative warm‑up, a long projected Adam (sub‑gradient) phase
and a final heavy simulated‑annealing polishing.  The Adam phase uses the exact
sub‑gradient of the max‑convolution and an adaptive per‑coordinate learning‑rate,

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    else:

        # clip steps that could be excessively large
        step = np.clip(step, -5.0, 5.0)

        a_tmp = a - step * m_hat
        a = _project_simplex(a_tmp)

    # --------------------------------------------------------------
    # Phase 3 – final annealing polish (any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        # Rough estimate: ~1000 full convolutions per second on a typical CPU.
        steps = int(min(300_000, max(50_000, remaining * 1000)))
        print(f"[{time.time() - start_time:.2f}s] annealing start – {steps:,} steps")
        rng = np.random.default_rng(seed + 12345)

        a = best_seq.copy()
        best_local = a.copy()
        best_local_score = best_score
        start_temp = 5e-2
        end_temp = 1e-6

        for it in range(1, steps + 1):
            if time.time() >= deadline:
                break
            T = start_temp * (end_temp / start_temp) ** (it / steps)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            eps_move = a[src] * rng.random() * 0.30
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            conv_prop = _conv_self(a_prop)
            new_score = 2.0 * n * conv_prop.max()
            delta = new_score - best_local_score

            if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best: score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise (protect against possible drift) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
