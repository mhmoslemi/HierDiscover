"""
Multi‑stage optimizer for the step‑function bound.

The optimizer works on several candidate lengths (n).  For each length it
1) initialises a probability vector a (either from the supplied global
   `height_sequence_1` or from a random start),
2) runs a long Adam optimisation on a smoothed version of the
   max‑convolution (the smoothing parameter α is increased progressively
   up to 320 so that the gradient focuses on the true maximum),
3) refines the best Adam solution with a projected sub‑gradient descent
   on the exact max‑convolution,
4) finishes with a simulated‑annealing fine‑tuning that moves small
   fractions of mass between coordinates.

All phases respect the overall 1000 s time budget and keep the best
sequence found.  The returned sequence is normalised to sum to 1, is
non‑negative and can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,

                          start_time: float,

                          max_iter: int = 10 ** 7):
    """
    Simple projected sub‑gradient descent for 2·n·max(conv(a)).
    Returns (best_vector, best_score) where score = evaluate_sequence.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        max_val = conv.max()
        # active set (tight constraints)
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        # sub‑gradient of max(conv)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small‑scale simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000):
    """
    Simulated annealing that transfers a random ε‑fraction of mass from
    a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _optimize_length(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float):
    """
    Adam optimisation of the smoothed objective for a fixed length `n`.
    Returns (best_sequence, best_score) where score = evaluate_sequence.
    """
    eps = 1e-16
    y = np.log(np.maximum(a0, eps))

    fft_len = _next_pow2(3 * n - 2)

    # smoothing schedule – expands to fairly large α
    alpha_schedule = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                     40.0, 80.0, 160.0, 320.0]

    iters_per_alpha = 8000  # generous budget per α

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_seq = a0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline - 0.5:
                break

            # softmax → probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            conv = _conv_self(a)

            # smoothed max (log‑sum‑exp)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val    # not used for learning rate – only for logging

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t. logits y (softmax parametrisation)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if i % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline - 0.5:
            break

    # final exact evaluation of the best softmax vector
    best_score = float(evaluate_sequence(best_seq.tolist()))
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – orchestrates multi‑length search
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine explores several problem sizes,
    runs a long Adam phase with a progressive smoothing schedule,
    refines the result with projected sub‑gradient descent and a final
    annealing stage.  It respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overrunning the allocation
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Possible lengths – we try a handful of increasing sizes
    # ------------------------------------------------------------------
    length_choices = [2400, 3600, 4800, 7200, 9600]

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to obtain an initial vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n):
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = starter / starter.sum()
        except Exception:
            starter = None

        if starter is not None:
            if starter.shape[0] != n:
                return _upsample(starter, n)
            else:
                return starter.copy()
        # fallback: random Dirichlet‑like start
        a = np.random.rand(n)
        return a / a.sum()

    # ------------------------------------------------------------------
    # Loop over candidate lengths, allocating a slice of the total time
    # ------------------------------------------------------------------
    for idx, n in enumerate(length_choices):
        # compute a sub‑deadline that gives roughly equal time to each n
        # (the last length gets whatever is left)
        if idx < len(length_choices) - 1:
            sub_deadline = start_time + (budget_s * (idx + 1) / len(length_choices)) - 0.5
        else:
            sub_deadline = deadline

        if time.time() >= sub_deadline:
            break

        print(f"[{time.time() - start_time:6.2f}s] ---- Optimising length n={n} ----")

        a0 = _initial_vector(n)

        # ---- Adam stage (smooth max) ----
        seq_adam, sc_adam = _optimize_length(a0, n, sub_deadline, start_time)

        # keep the best candidate seen so far
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time() - start_time:6.2f}s] New overall best after Adam: n={n}, score={best_score:.6f}")

        # ---- Projected sub‑gradient polishing (if time permits) ----
        if time.time() < sub_deadline - 1.0:
            polish_deadline = time.time() + 0.4 * (sub_deadline - time.time())
            a_pol, sc_pol = _run_gradient_descent(seq_adam,
                                                  n,
                                                  polish_deadline,
                                                  start_time,
                                                  beta=0.95,
                                                  step0=0.22)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"[{time.time() - start_time:6.2f}s] Polishing improvement: n={n}, score={best_score:.6f}")

        # ---- Final annealing for this length (remaining time) ----
        if time.time() < sub_deadline - 0.5:
            a_ann, sc_ann = _run_annealing(best_seq,
                                           n,
                                           sub_deadline,
                                           start_time,
                                           best_score,
                                           best_seq,
                                           start_temp=5e-2,
                                           end_temp=1e-6,
                                           steps=200_000)
            if sc_ann < best_score:
                best_score = sc_ann
                best_seq = a_ann.copy()
                print(f"[{time.time() - start_time:6.2f}s] Annealing improvement: n={n}, score={best_score:.6f}")

        # if overall deadline is near, stop early
        if time.time() >= deadline - 1.0:
            break

    # ------------------------------------------------------------------
    # Global polishing with the remaining budget (if any)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_pol, sc_pol = _run_gradient_descent(best_seq,
                                              best_seq.size,
                                              polish_deadline,
                                              start_time,
                                              beta=0.96,
                                              step0=0.2)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"[{time.time() - start_time:6.2f}s] Final polish improvement: score={best_score:.6f}")

        # final annealing with any leftover time
        if time.time() < deadline - 0.2:
            a_fin, sc_fin = _run_annealing(best_seq,
                                           best_seq.size,
                                           deadline,
                                           start_time,
                                           best_score,
                                           best_seq,
                                           start_temp=2e-2,
                                           end_temp=1e-6,
                                           steps=300_000)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"[{time.time() - start_time:6.2f}s] Final anneal improvement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – uniform distribution of length 4800
        best_seq = np.full(4800, 1.0 / 4800)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq.tolist()]
