"""
Propose a non‑negative step function that (approximately) minimizes
``evaluate_sequence``.

Algorithm
---------
* **Problem size** – use the length of the supplied starter (if any) or 4800.
  Work on the probability simplex (∑a_i = 1) because the objective is
  scale‑invariant.

* **Multiple random restarts** – generate several symmetric Dirichlet
  initialisations, run a short Adam optimisation on a *smoothed* max‑convolution
  objective and keep the best result.

* **Polishing** – run a projected sub‑gradient descent (momentum + simplex
  projection) on the exact max‑convolution for the remaining time.

* **Fine‑tuning** – a simulated‑annealing phase that repeatedly moves a random
  ε‑fraction of mass from a source index to a destination index, accepting
  uphill moves with a geometric temperature schedule.

* **Time management** – the total budget (≈1000 s) is split into three stages
  (restarts ≈ 55 %, polishing ≈ 30 %, annealing ≈ 15 %).  The routine aborts
  gracefully if the deadline is reached.

All helper functions are top‑level, use only NumPy and the standard library,
and the final sequence is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  g(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂g/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max over convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(y0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alpha_schedule=None,
                   iters_per_alpha: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam runs on the smoothed objective.
    Returns best vector (on the simplex) and its exact score = 2·n·max(conv).
    """
    if alpha_schedule is None:
        alpha_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    fft_len = _next_pow2(3 * n - 2)

    best_a = None
    best_score = float("inf")

    for alpha in alpha_schedule:
        for i in range(iters_per_alpha):
            if time.time() >= deadline:
                break

            # ---- soft‑max to obtain a on the simplex ----
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # ---- smooth objective and its gradient ----
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            smooth_val = lse / alpha    # (1/α)·log‑sum‑exp
            obj = 2.0 * n * smooth_val

            # gradient w.r.t. a
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # gradient w.r.t. logits y (soft‑max chain rule)
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ---- Adam update ----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- cheap exact evaluation every few steps ----
            if i % 250 == 0:
                maxc = conv.max()
                score = 2.0 * n * maxc
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={score:.6f}")

        if time.time() >= deadline:
            break

    if best_a is None:
        # fall back to the final parameters
        maxy = np.max(y)
        exp_y = np.exp(y - maxy)
        best_a = exp_y / exp_y.sum()
        best_score = 2.0 * n * _conv_self(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent with momentum on the exact
    max‑convolution objective.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # ---- sub‑gradient of max(conv) ----
        max_val = conv.max()
        # active set (values within 1e-12 relative tolerance of the max)
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # each pair (i,j) contributes a_j to gradient of a_i and a_i to gradient of a_j
            g[i] += a[j]
            g[j] += a[i]

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto probability simplex
        v = a - step * momentum
        a = _project_onto_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The strategy consists of:
        1. a series of short Adam runs from random symmetric initialisations,
        2. a polishing projected‑gradient descent on the exact max,
        3. a final simulated‑annealing fine‑tuning.
    The whole procedure respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine problem size and possible starter sequence
    # ------------------------------------------------------------------
    try:

            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:

            else:

        else:

            n = 4800
    except Exception:
        starter = None
        n = 4800

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (2·n·max(conv)) – faster than
    # calling evaluate_sequence for intermediate checks.
    # ------------------------------------------------------------------
    def exact_score(a_vec: np.ndarray) -> float:
        return 2.0 * n * _conv_self(a_vec).max()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s

    best_score = float("inf")
