"""
Multi‑stage optimiser for the step‑function bound.

The algorithm works on a longer sequence (default 9 600 entries) and
contains three phases that respect the 1000 s budget:

1. **Soft‑max Adam on a smoothed max‑convolution**
   – a is parametrised as a soft‑max of logits y (automatically stays in the
     simplex).
   – the objective 2·n·max(conv) is approximated by a log‑sum‑exp with a
     smoothing parameter α, which is increased from 0.5 up to 200.
   – Adam updates the logits; the best exact score is recorded.

2. **Exact mirror‑descent polishing**
   – an exponentiated‑gradient (mirror‑descent) scheme works on the true
     max‑convolution (non‑smooth objective).  The sub‑gradient of the
     max‑convolution is the average of the gradients of all active
     constraints; the update `a_i ← a_i·exp(-η·g_i)` keeps a ≥ 0 and the
     subsequent normalisation projects back onto the probability simplex.
   – this phase exploits the remaining time after Adam.

3. **Simulated‑annealing fine‑tuning**
   – a simple mass‑transfer move (pull ε‑fraction from a source index to a
     destination) with geometric cooling further reduces the score.

The returned list contains non‑negative numbers that sum to one (the scale
does not matter for ``evaluate_sequence``).  By using a larger problem size
and a more aggressive polishing stage the method typically reaches values
close to the target 1.503, improving on the previous 1.5128 bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) polishing
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.2,
                     rel_tol: float = 1e-12,
                     eval_every: int = 500) -> tuple[np.ndarray, float]:
    """
    Runs an exponentiated‑gradient (mirror‑descent) on the exact
    max‑convolution objective.  Returns the best (a, score) pair where
    score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < 5_000_000:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active constraints (values within rel_tol of the maximum)
        active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # sub‑gradient: average of gradients of active constraints
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]          # derivative w.r.t. a_i is a_{k-i}
        if active.size:
            g = 2.0 * g / active.size   # factor 2 from symmetry
        else:
            g = np.zeros_like(a)

        # exponentiated‑gradient step
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * g)

        # projection onto the simplex (renormalisation)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional exact evaluation
        if it % eval_every == 0:
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000,
                   eval_every: int = 1_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

                break

            # ---- soft‑max ----
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            # ---- convolution & smoothed objective ----
            conv = _conv_self(a)
            scaled = alpha * conv
            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha
            obj = 2.0 * n * f_val

            # ---- gradient w.r.t. a ----
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            # ---- gradient w.r.t. logits y ----
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # ---- Adam update ----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- occasional exact evaluation ----
            if i % 100 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Exact mirror‑descent polishing (exponentiated gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = deadline
        a_polish, score_polish = _run_exp_gradient(
            best_seq,
            n,
            polish_deadline,
            start_time,
            step0=0.2,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(x) for x in best_seq.tolist()]
