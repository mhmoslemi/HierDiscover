"""
Improved search for a non‑negative step‑function minimizing `evaluate_sequence`.

Algorithm
---------
1.  Initialise a random (or supplied) distribution and normalise it to sum = 1.
2.  Optimise the distribution with a *continuation* Adam optimiser:
    *   For a sequence of norms p = 2, 4, 8, 16, 32, 64, 128 we minimise the
        smooth surrogate   Fₚ(a) = 2·n·( Σ conv(a)^p )¹ᐟᵖ .
        As p → ∞ this converges to the original objective
        2·n·max(conv(a)).
    *   The gradient of Fₚ is computed analytically using FFT‑based
        convolutions; a projection onto the probability simplex enforces
        a ≥ 0 and Σa = 1.
    *   Adam with bias‑corrected moments provides an adaptive step size.
3.  After the Adam phase the current best vector is *upsampled* (linear
    interpolation) to a larger size and the optimisation is repeated.
    Sizes used: 1024 → 4096 → 12288 (≈10⁴).  Larger vectors give a better
    bound but are much more expensive, therefore the continuation scheme
    is applied only on the final size for a short fine‑tuning phase.
4.  A light simulated‑annealing move set is performed on the remaining
    time budget to escape any residual local minima.
5.  The best sequence encountered is returned as a plain Python list.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible non‑negative sequence, so the harness never
receives `np.inf`.
"""

import time
import math
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – O(n)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT, length 2·n‑1
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real FFT, length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Correlation of w (length 2n‑1) with a (length n) – needed for the gradient.
#   corr[j] = Σ_i w[i] * a[i‑j]   (out‑of‑range indices are treated as zero)
# ----------------------------------------------------------------------
def _correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation of w with a (both real). Result length = len(w)+len(a)-1."""
    n = a.shape[0]
    m = w.shape[0]
    out_len = m + n - 1
    fft_len = 1 << ((out_len).bit_length())
    Wf = np.fft.rfft(w, n=fft_len)
    Af_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr = np.fft.irfft(Wf * Af_rev, n=fft_len)[:out_len]
    return corr


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed p‑norm surrogate.
# ----------------------------------------------------------------------
def _run_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    p_schedule,
    max_iter_per_p: int = 5000,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a (sum = 1, a≥0) using Adam on the surrogate
    Fₚ(a)=2·n·(∑ conv(a)^p)^{1/p}.  Returns the best vector found.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    for p in p_schedule:
        # 1‑step warm‑up for each p (optional)
        for it in range(1, max_iter_per_p + 1):
            if time.time() >= deadline:
                return best_a, best_score

            t += 1

            conv = _conv_self(a)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] p={p:<3d} iter={it:,} "
                    f"score={cur_score:.6f}"
                )

            # ----- gradient of the smoothed surrogate -----
            # S = Σ conv_i^p
            S = np.power(conv, p, out=conv).sum()
            if S == 0.0:
                # degenerate (should not happen for a non‑zero vector)
                break

            w = np.power(conv, p - 1, out=conv)    # reuse conv array
            # correlation w ⋆ a  -> length 3n‑2, we need the middle n entries
            corr_full = _correlate(w, a)
            # middle slice corresponds to indices j = 0 … n‑1
            correlation = corr_full[n - 1 : n - 1 + n]

            factor = 4.0 * n * (S ** (1.0 / p - 1.0))
            grad = factor * correlation

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = alpha * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # projection onto the probability simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsample via linear interpolation and renormalise.
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from one index to another."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    best_seq = None

    # ------------------------------------------------------------------

    starter = None

        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)

    slice_deadline = time.time() + slice_time

    for r in range(restarts):

    # fallback (unlikely)

    # ------------------------------------------------------------------

    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
