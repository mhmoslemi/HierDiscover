def _run_incremental_local_search(a0, n, deadline, start_time, max_iter=None):
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    it = 0
    start_temp = 5e-2
    end_temp = 1e-6
    # we can also fraction of iterations default unlimited; stop by time.

    while time.time() < deadline:
        it += 1
        # optionally limit to max_iter if given
        if max_iter and it > max_iter:
            break

        # get top shifts
        top_k = 5
        # argsort descending
        # Use np.argpartition to get top indices quickly
        idx_top = np.argpartition(conv, -top_k)[-top_k:]
        # compute contributions per index
        contrib = np.zeros(n)
        for idx in idx_top:
            # i range for this shift
            lo = max(0, idx - (n-1))
            hi = min(n-1, idx)
            # i slice length
            i_range = np.arange(lo, hi+1)
            j_range = idx - i_range
            contrib[i_range] += a[i_range] * a[j_range]

        # donor: choose index with probability proportional to contrib
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient: choose from low-contrib indices
        low_thr = np.percentile(contrib, 20)  # bottom 20%
        low_idxs = np.where(contrib <= low_thr)[0]
        if low_idxs.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idxs)

        if i_donor == i_rec or a[i_donor] < 1e-12:
            continue

        # amount: fraction of donor mass, scaled by temperature
        elapsed = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** elapsed
        # max fraction maybe 0.5 * T/start_temp? let's set fraction = min(0.5, T/start_temp)
        max_frac = min(0.5, T / start_temp)  # T decays from start_temp to end_temp.
        # random fraction up to max_frac
        frac = max_frac * np.random.rand()
        delta = a[i_donor] * frac
        if delta <= 0:
            continue

        # Apply move
        _apply_move(a, conv, i_donor, i_rec, delta)

        # compute new max
        new_max = conv.max()
        # accept if better or with probability
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] inc local improve: n={n}, iter={it}, max={new_max:.6f}")
        else:
            # metropolis acceptance with temperature T
            # delta_score = new_max - cur_max
            prob_accept = math.exp(-(new_max - cur_max) / (T + 1e-12))
            if np.random.rand() < prob_accept:
                cur_max = new_max  # accept
            else:
                # revert move: apply opposite move
                # To revert, we can apply move with delta negative? Simpler: revert by swapping donor and recipient with same delta.
                # Actually original a changed: a[i_donor] decreased, a[i_rec] increased.
                # We can revert by applying opposite move: inc donor by delta, dec recipient by delta.
                # But we also need revert conv changes.
                # Instead of revert, just recompute conv from scratch (costly but rare). We'll recompute conv for simplicity.
                # Recompute conv after revert: revert a manually.
                a[i_donor] += delta
                a[i_rec] -= delta
                conv = _conv_self(a)
                # continue (cur_max unchanged)
                continue

    # after loop, compute final best score
    best_score = 2.0 * n * best_max
    return best_a, best_score

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Warm‑start from the previous best construction (``height_sequence_1``) if it
   exists, otherwise from a random simplex point.
2. Progressive projected gradient descent on a smooth
   approximation of the max‑convolution (soft‑max).  Three stages
   (800 → 2400 → 4800) are performed; the last stage receives most of the
   gradient budget.
3. Fine‑grained incremental local search that moves a small amount of mass
   from indices that contribute heavily to the current largest convolution
   entries to indices that contribute little.  The move is applied by an
   *incremental* update of the full self‑convolution (O(n) per move) and
   accepted with a simulated‑annealing rule.
4. Return the best sequence found (normalised to sum = 1).

The incremental local search together with a longer gradient phase typically
pushes the upper bound below the previous 1.512882 → target 1.503.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of a soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of Σ_k w_k(conv/temperature)·conv_k with w = soft‑max."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))          # extra stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = n + (2n‑1) - 1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]         # centre block of length n
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected gradient descent with soft‑max gradient
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) gradient descent on the smooth max‑convolution.
    Returns the best point seen and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-5                # a bit lower than before for higher resolution

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Incremental convolution update (mass shift)
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer `delta` mass from index i to index j.
    Updates `a` and the full self‑convolution `conv` in‑place.
    """
    n = a.shape[0]

    # contribution of the *old* vector a to the convolution update
    # (a * delta_a) term – factor 2 because of symmetry
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # delta_a * delta_a term
    idx = 2 * i
    if idx < conv.shape[0]:
        conv[idx] += delta * delta
    idx = 2 * j
    if idx < conv.shape[0]:
        conv[idx] += delta * delta
    if i != j:
        idx = i + j
        if idx < conv.shape[0]:
            conv[idx] -= 2.0 * delta * delta

    # finally move the mass
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
# Targeted incremental local search (mass‑shifting)
# ----------------------------------------------------------------------
def _run_incremental_local_search(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float) -> tuple[np.ndarray, float]:
    """
    A fast hill‑climbing / simulated‑annealing walk that repeatedly shifts a
    small fraction of probability mass away from indices that contribute most
    to the current top convolution entries.
    Returns the best sequence seen and its (scaled) score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start_temp = 5e-2
    end_temp = 1e-6
    top_k = 5                     # number of shifts taken into account
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) Determine the indices that dominate the convolution
        # ------------------------------------------------------------------
        # pick the top‑k shift positions (largest convolution values)
        if top_k >= conv.shape[0]:
            idx_top = np.arange(conv.shape[0])
        else:
            idx_top = np.argpartition(conv, -top_k)[-top_k:]

        # contribution of each position to those top shifts
        contrib = np.zeros(n)
        for idx in idx_top:
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = idx - i_range
            contrib[i_range] += a[i_range] * a[j_range]

        # ------------------------------------------------------------------
        # 2) Choose donor (high contribution) and recipient (low contribution)
        # ------------------------------------------------------------------
        total_c = contrib.sum()
        if total_c > 0:
            prob = contrib / total_c
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        low_thr = np.percentile(contrib, 20)          # bottom 20 %
        low_idxs = np.where(contrib <= low_thr)[0]
        if low_idxs.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idxs)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # ------------------------------------------------------------------
        # 3) Propose a transfer amount (scaled by the annealing temperature)
        # ------------------------------------------------------------------
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        # maximal fraction shrinks together with the temperature
        max_frac = min(0.5, T / start_temp)
        frac = max_frac * np.random.rand()
        delta = a[i_donor] * frac
        if delta <= 0:
            continue

        # ------------------------------------------------------------------
        # 4) Apply the move incrementally
        # ------------------------------------------------------------------
        _apply_move(a, conv, i_donor, i_rec, delta)
        new_max = conv.max()

        # ------------------------------------------------------------------
        # 5) Accept / reject (Metropolis rule)
        # ------------------------------------------------------------------
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] inc local improve: "
                      f"n={n}, iter={it}, max={new_max:.6f}")
        else:
            # uphill move: accept with probability exp(-(Δ)/T)
            prob_accept = math.exp(-(new_max - cur_max) / (T + 1e-12))
            if np.random.rand() < prob_accept:
                cur_max = new_max
            else:
                # reject → revert the move (re‑compute convolution from scratch)
                a[i_donor] += delta
                a[i_rec] -= delta
                conv = _conv_self(a)          # expensive but rare
                # cur_max stays unchanged

    best_score = 2.0 * n * best_max
    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of a multi‑stage soft‑max gradient descent followed by
    a fine‑grained incremental local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Load optional starter (previous best construction)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()          # normalise to sum = 1
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage configuration (progressive up‑sampling)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800]          # final length = 4800
    weight_sum = sum(stage_lengths)

    total_time = deadline - start_time
    grad_budget = total_time * 0.65           # 65 % for gradient descent
    local_budget = total_time - grad_budget   # remainder for local search

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Gradient descent phase (soft‑max)
    # --------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a share of the remaining gradient budget
        stage_frac = n / weight_sum
        stage_time = max(2.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this length
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # run the smooth‑max gradient descent
        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.6, beta=0.9, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # update remaining gradient budget (takes elapsed time into account)
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # --------------------------------------------------------------
    # Incremental targeted local search (mass‑shifting)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        local_deadline = deadline
        a_loc, score_loc = _run_incremental_local_search(
            best_seq, n_final, local_deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:6.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
