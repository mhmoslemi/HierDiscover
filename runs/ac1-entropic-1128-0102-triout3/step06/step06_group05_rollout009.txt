"""
Mirror-descent optimizer for the self‑convolution peak problem.

The algorithm starts from a supplied or randomly generated probability
vector (sum = 1), repeatedly upsamples it to larger lengths, then runs a
projected mirror‑descent (exponential‑gradient) loop that directly
minimises the maximal entry of the self‑convolution.  The loop works
on the simplex using a KL‑divergence Bregman projection, which keeps
all coefficients non‑negative.  After each iteration the sequence is
symmetrised (a←(a+a[::-1])/2) because optimal solutions are believed to
be symmetric.  The routine returns the best vector found before the
time budget expires.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only for fallback safety
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – self convolution
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # Clip tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c

# ----------------------------------------------------------------------
# Upsample (linear interpolation) keeping the L1‑norm equal to 1
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up

# ----------------------------------------------------------------------
# Mirror‑descent optimizer (KL‑projection) for a given length
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        symmetrize: bool = True,
                        eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of exponentiated‑gradient (mirror) descent
    on the probability simplex to minimise the maximal entry of the
    self‑convolution.  Returns the best vector seen and its exact score
    (2·n·max(conv) because the sum is 1).
    """
    a = a0.copy()
    # ensure feasibility
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        idx = int(conv.argmax())           # index of the maximal entry
        # gradient of the max‑convolution (sub‑gradient at the active shift)
        g = np.zeros_like(a)
        # i such that 0 <= idx - i < n  → i ∈ [idx-n+1, idx]
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = idx - i_range
            # g_i = 2 * a_j (the derivative of a_i * a_j + a_j * a_i)
            g[i_range] = 2.0 * a[j_range]

        # step‑size schedule – diminishing with √t
        eta = eta0 / math.sqrt(it)

        # Exponential update (mirror descent with KL divergence)
        # loga = log(a) - η g   ⇒   a ← a * exp(-η g)
        loga = np.log(a + 1e-30) - eta * g
        a = np.exp(loga)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # renormalise onto the simplex
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # evaluate current objective
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror n={n} it={it} score={cur_score:.6f}")

        # occasional random jitter to escape flat plateaus (probability 0.001)
        if random.random() < 0.001:
            # add a tiny Gaussian noise and re‑normalise
            noise = np.abs(np.random.randn(n)) * 1e-4
            a = a + noise
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)

    return best_a, best_score

"""
Mirror‑descent optimiser for the self‑convolution peak problem.

The algorithm builds a probability vector (non‑negative, sum = 1) for a
small initial length, repeatedly upsamples it to larger lengths and,
for each length, runs a projected mirror‑descent (exponential‑gradient)
loop that directly minimises the maximal entry of the self‑convolution.
The update is performed on the probability simplex using a KL‑divergence
Bregman projection, which guarantees non‑negativity.  After each
iteration the vector is symmetrised (a←(a+a[::-1])/2) because optimal
solutions are believed to be symmetric.  The best vector seen before
the time budget expires is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (fallback safety)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Linear‑interpolation up‑sampling (preserves L1‑norm)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Upsample a probability vector to a new length by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up[:] = 1.0 / new_n
    return up

# Mirror‑descent optimiser (KL‑projection) for a given length
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        symmetrize: bool = True,
                        eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of exponentiated‑gradient (mirror) descent on
    the probability simplex to minimise max(self‑convolution).  Returns
    the best vector seen and its exact score 2·n·max(conv) (sum is 1).
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        idx = int(conv.argmax())
        max_val = conv.max()

        if cur_score < best_score:
