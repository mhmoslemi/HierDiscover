    arc = 1.0/np.sqrt(t*(1-t)+1e-12)
    arc /= arc.sum()
    candidates.append(arc)
    # triangular shape:
    left = np.linspace(1, n_target//2, n_target//2, endpoint=False)
    right = np.linspace(n_target//2, 1, n_target-n_target//2)
    tri = np.concatenate([left,right])
    tri = np.maximum(tri,0.0)
    tri /= tri.sum()
    candidates.append(tri)
    # uniform:
    uni = np.full(n_target, 1.0/n_target)
    candidates.append(uni)
    # random Dirichlet:
    rnd = np.random.rand(n_target)
    rnd /= rnd.sum()
    candidates.append(rnd)

    # Evaluate each candidate's exact score
    best_a = None
    best_score = float('inf')
    for cand in candidates:
        sc = 2.0 * n_target * _conv_self(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] initial best score {best_score:.6f} n={n_target}")

    # Phase fractions
    adam_frac = 0.55
    subgrad_frac = 0.25
    # compute deadlines
    adam_deadline = start_time + adam_frac * total_time
    subgrad_deadline = adam_deadline + subgrad_frac * total_time
    anneal_deadline = deadline

    # Adam Optimization
    a_adam, score_adam = _adam_optimize(best_a, n_target, adam_deadline, start_time, symmetrize=True)
    if score_adam < best_score:
        best_score = score_adam
        best_a = a_adam.copy()
        print(f"[{time.time() - start_time:.2f}s] after Adam best score {best_score:.6f}")

    # Subgradient descent
    if time.time() < subgrad_deadline:
        a_sub, score_sub = _subgradient_descent(best_a, n_target, subgrad_deadline, start_time)
        if score_sub < best_score:
            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] after subgrad best score {best_score:.6f}")

    # Annealing final polishing
    if time.time() < anneal_deadline:
        a_anneal, score_anneal = _annealing_refine(best_a, n_target, anneal_deadline, start_time, best_score, best_a)
        if score_anneal < best_score:
            best_score = score_anneal
            best_a = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] after anneal best score {best_score:.6f}")

    # Possibly try a secondary length if time remains: if time left>some threshold and n_target<65536, upsample to larger length and repeat refine.
    # But for simplicity, return best_a.

    # Convert to Python list for output
    return [float(x) for x in best_a]

"""
Hybrid refinement for minimizing the self‑convolution L∞ objective.

The algorithm works on the probability simplex (∑a_i = 1) because the
objective returned by `evaluate_sequence` is scale‑invariant.
It proceeds in three stages on a (large) target length:

1. **Warm‑start selection** – a few analytically‑motivated shapes
   (arcsine, triangle, uniform, random) and, if available, the
   previous best construction (`height_sequence_1`) are up‑sampled to the
   target size and the best among them is kept as the start vector.

2. **Adam optimisation on a log‑sum‑exp surrogate** of the maximum
   self‑convolution.  The surrogate is gradually sharpened (temperature
   decay) so that the algorithm converges to the true L∞ norm while
   staying smooth enough for reliable gradients.

3. **Projected sub‑gradient descent** on the exact max‑convolution,
   followed by a **simulated‑annealing polishing** phase that makes cheap
   mass‑transfer moves and accepts occasional uphill steps.

All iterates are projected onto the simplex and symmetrised
(a ← (a+a[::-1])/2) because optimal solutions are believed to be
symmetric.  The routine respects a user‑supplied time budget (default
1000 s) and returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed L∞ surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a (length 2n‑1)
    lam : smoothing temperature λ > 0

    Returns
    -------
    grad: dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    s = weights.sum()
    if s == 0.0:
        weights = np.full_like(weights, 1.0 / weights.size)
    else:
        weights /= s

    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)               # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]               # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        active = conv == max_val
    w = active.astype(float)
    w /= w.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                     # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) to a new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, renormalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its exact objective value (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule
    lam_factor = 0.3
    lam_decay = 0.98
    lam_min_factor = 1e-9

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam it={it} score={cur_score:.6f}")

        # temperature for the log‑sum‑exp surrogate
        lam = max(max_val * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    final_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, final_score


# ----------------------------------------------------------------------
# Deterministic sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()

    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination index.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # enforce symmetry and feasibility
        a_prop = (a_prop + a_prop[::-1]) * 0.5
        a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal step={it} score={score:.6f}")

    final_score = 2.0 * n * _conv_self(best_seq).max()
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # safety margin to avoid overrunning the evaluator timeout
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Choose a target length. 32768 gives a good trade‑off between quality
    # and per‑iteration cost; fall back to a smaller size if the budget is
    # very tight.
    # ------------------------------------------------------------------
    if total_time < 180.0:
        n_target = 8192
    else:
        n_target = 32768

    # ------------------------------------------------------------------
    # Build a few plausible starting vectors (all normalised to sum = 1)
    # ------------------------------------------------------------------
    candidates = []

    # 1) reuse a previously discovered construction if present
    try:
        base_seq = np.asarray(height_sequence_1, dtype=float)
        if base_seq.ndim == 1 and base_seq.size > 0:
            base_seq = np.maximum(base_seq, 0.0)
            s = base_seq.sum()
            if s > 0.0:
                base_seq = base_seq / s
                candidates.append(_upsample(base_seq, n_target))
    except Exception:
        pass

    # 2) arcsine‑like distribution (high at the ends)
    t = (np.arange(n_target) + 0.5) / n_target
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    candidates.append(arc)

    # 3) triangular shape
    left = np.linspace(1, n_target // 2, n_target // 2, endpoint=False)
    right = np.linspace(n_target // 2, 1, n_target - n_target // 2)
    tri = np.concatenate([left, right])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    candidates.append(tri)

    # 4) uniform distribution
    uni = np.full(n_target, 1.0 / n_target)
    candidates.append(uni)

    # 5) random Dirichlet draw
    rnd = np.random.rand(n_target)
    rnd /= rnd.sum()
    candidates.append(rnd)

    # ------------------------------------------------------------------
    # Pick the best among the cheap candidates
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for cand in candidates:
        sc = 2.0 * n_target * _conv_self(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[{time.time() - start_time:6.2f}s] Initial best score {best_score:.6f} (n={n_target})")

    # ------------------------------------------------------------------
    # Time allocation for the three refinement phases
    # ------------------------------------------------------------------
    adam_frac = 0.55
    subgrad_frac = 0.25
    adam_deadline = start_time + adam_frac * total_time
    subgrad_deadline = adam_deadline + subgrad_frac * total_time
    anneal_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth surrogate
    # ------------------------------------------------------------------
    a_adam, score_adam = _run_adam_opt(best_a, n_target,
                                       adam_deadline, start_time,
                                       symmetrize=True)
    if score_adam < best_score:
        best_score = score_adam
        best_a = a_adam.copy()
        print(f"[{time.time() - start_time:6.2f}s] After Adam best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < subgrad_deadline:
        a_sub, score_sub = _run_subgradient(best_a, n_target,
                                            subgrad_deadline, start_time)
        if score_sub < best_score:
            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[{time.time() - start_time:6.2f}s] After sub‑grad best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < anneal_deadline:
        a_anneal, score_anneal = _run_annealing(best_a, n_target,
                                                anneal_deadline, start_time,
                                                best_score, best_a,
                                                start_temp=5e-2,
                                                end_temp=1e-5,
                                                steps=120000)
        if score_anneal < best_score:
            best_score = score_anneal
            best_a = a_anneal.copy()
            print(f"[{time.time() - start_time:6.2f}s] After anneal best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety check – ensure we have a valid sequence
    # ------------------------------------------------------------------
    if best_a is None or best_a.size == 0:
        best_a = np.full(512, 1.0 / 512.0)

    # Return the sequence as a plain Python list (the evaluator will clamp
    # values as needed, but our vector already respects the constraints).
    return [float(x) for x in best_a]
