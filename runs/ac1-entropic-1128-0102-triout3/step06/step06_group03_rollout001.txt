assistantfinal**Solution Overview**

The objective is to minimise  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the non‑negative vector `a` with itself.
Because the score is invariant to a global scaling of `a`, we can safely
work on the probability simplex (`∑a_i = 1`).  
The algorithm below combines three complementary heuristics:

1. **Progressive soft‑max gradient descent** – fast early optimisation on
   moderately sized vectors (800 → 2400 → 4800).  The soft‑max smooths the
   `max`‑operator and yields a usable gradient.

2. **Hard‑max sub‑gradient descent** – after up‑sampling to the final length
   (9600) we switch to the exact sub‑gradient of the maximal convolution entry.
   This removes the smoothing bias and drives the bound lower.

3. **Targeted stochastic local search** – a Metropolis‑style walk that moves
   probability mass from indices that contribute most to the current peak
   towards indices that contribute little.  The temperature slowly cools,
   allowing occasional uphill steps.

The time budget (default ≈ 1000 s) is split automatically:
≈ 50 % for the soft‑max phases, the remainder for the hard‑max phase and a
final local‑search polishing step.  The best sequence seen during the whole
run is returned.

All helper functions are defined at module level, no closures or file I/O are
used, and the code works with the standard `evaluate_sequence` that the
harness provides.

```python
"""
Gradient / sub‑gradient + local‑search optimiser for the convolution bound.

* Warm‑start from the previous best sequence (height_sequence_1) if available.
* Three soft‑max GD stages (800 → 2400 → 4800) to obtain a good shape.
* Upsample to length 9600 and run a hard‑max sub‑gradient optimiser for the
  remainder of the time budget.
* End with a focused stochastic local‑search that shaves the remaining peaks.
* The best sequence seen (normalised to sum = 1) is returned.
"""

import time

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    # length for FFT – next power of two ≥ 2n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution rev(a) * w (full length)."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of Σ_k softmax(conv_k/temperature) * conv_k."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()               # soft‑max weights, sum to 1
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)   # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]   # central block of length n
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov gradient descent on the smoothed max‑convolution.
    Returns the best point seen and the corresponding (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (exponential)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent (exact max‑convolution gradient)
# ----------------------------------------------------------------------
def _run_hardmax_subgrad(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.5,
                         beta: float = 0.9,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the true maximal convolution entry.
    Returns best sequence and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hardmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # index of the current maximal entry
        idx_max = int(np.argmax(conv))

        # compute sub‑gradient of that entry
        grad = np.zeros(n)
        low = max(0, idx_max - (n - 1))
        high = min(n - 1, idx_max)
        i_range = np.arange(low, high + 1)
        j_range = idx_max - i_range
        grad[low:high + 1] = 2.0 * a[j_range]

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size decays like 1/√t
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserving total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Targeted stochastic local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of probability mass from indices that contribute
    most to the current maximal convolution entry to indices that contribute
    little.  Metropolis acceptance with a slowly cooling temperature.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        # recompute convolution and locate the maximal entry
        conv = _conv_self(a)
        idx_max = int(np.argmax(conv))
        cur_score = 2.0 * n * conv.max()

        # contribution of each index to that maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # donor = index with probability ∝ contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # recipient = index from the bottom 20 % of contributions
        low_thr = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size == 0:
            i_rec = np.random.randint(n)
        else:
            i_rec = np.random.choice(low_idx)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # move a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        # evaluate proposal
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # cooling schedule (linear in elapsed fraction)
        elapsed_frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed_frac

        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()
        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] local improve: "
                      f"n={n}, iter={it}, score={new_score:.6f}")

    return best_a, best_score

        if time.time() >= deadline:
            break
        # proportion of the softmax budget for this stage (weighted by n)
        stage_frac = n / sum(stage_lengths)
        stage_time = max(2.0, softmax_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.8, beta=0.9, max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # adjust remaining softmax budget
        elapsed = time.time() - start_time
        softmax_budget = max(0.0, total_time * 0.5 - elapsed)

    # --------------------------------------------------------------
    # Hard‑max sub‑gradient descent on the final length
    # --------------------------------------------------------------
    if time.time() < deadline:
        # upsample the current best to the final length
        if best_seq is not None:
            a0 = _upsample(best_seq, final_len)
        elif starter is not None:
            a0 = _upsample(starter, final_len)
        else:
            a0 = np.random.rand(final_len)
            a0 /= a0.sum()

        # split remaining time: 80 % hard‑max, 20 % local search
        hardmax_deadline = time.time() + remaining_budget * 0.8
        a_opt, score_opt = _run_hardmax_subgrad(
            a0, final_len, hardmax_deadline, start_time,
            step0=0.5, beta=0.9, max_iter=10 ** 7
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] hardmax final improve: "
                  f"score={score_opt:.6f}")

    # --------------------------------------------------------------
    # Final stochastic local search (peak shaving)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        local_deadline = deadline
        a_loc, score_loc = _run_targeted_local_search(
            best_seq, n_final, local_deadline, start_time
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:.2f}s] final local improve: "
                  f"score={score_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (numerical drift)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
```
