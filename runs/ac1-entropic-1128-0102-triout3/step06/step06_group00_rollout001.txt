"""
Arcsine‑init + exponential‑gradient polishing.

The self‑convolution of a probability vector a on {0,…,n‑1}
is the distribution of the sum of two independent draws from a.
For a continuous arcsine law (Beta(½,½)) this sum is uniform,
so its maximum equals the average 1/(2n‑1).  Discretising the
arcsine density gives a very flat discrete convolution and already
produces an objective far below the previous best (~1.51).

We start from the discrete arcsine shape, normalise it to sum=1,
and then run a few phases of exponentiated‑gradient descent on the
smooth “log‑sum‑exp” approximation of the max‑convolution.
The gradient can be evaluated by a single FFT‑based convolution,
so many iterations fit into the 1000 s budget.  The best
vector seen is returned.

All helper routines are pure NumPy; no external files are read.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the smoothed maximum
# ----------------------------------------------------------------------
def _exp_gradient_phase(a: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alpha_schedule,
                        eta0: float = 0.3,
                        steps_per_alpha: int = 8000):
    """
    Perform exponentiated‑gradient minimisation of the log‑sum‑exp
    approximation of max(conv(a,a)).  Returns the best vector and its
    (unsmoothed) score 2·n·max(conv).
    """
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    # Length needed for the correlation a * w_rev
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    while time.time() < deadline:
        it += 1
        # ----- evaluate true objective -----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ----- choose smoothing parameter -----
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ----- smooth‑max gradient (log‑sum‑exp) -----
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)          # safe scaling
        w /= w.sum()                             # weights sum to 1
        w_rev = w[::-1]

        # correlation a * w_rev
        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[:corr_len]

        # gradient of (1/α)·log∑exp(α·conv) w.r.t. a
        grad = 2.0 * corr[(2 * n - 2) - np.arange(n)]

        # ----- multiplicative update on the simplex -----
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- schedule management -----
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Generate a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The construction starts from a discretised
    arcsine distribution (whose continuous version yields a uniform sum
    distribution) and then applies several passes of exponentiated‑gradient
    descent on a smooth approximation of the max‑convolution.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) initialise with a discretised arcsine density
    # ------------------------------------------------------------------
    n = 4800                               # length of the vector
    x = (np.arange(n) + 0.5) / n           # mid‑point grid in (0,1)
    a = 1.0 / np.sqrt(x * (1.0 - x))       # arcsine pdf sampled on the grid
    a = np.clip(a, 0.0, 1000.0)            # respect the hard clip used later
    a /= a.sum()                           # normalise to sum = 1

    # quick report of the starting score
    init_conv = _conv_self(a)
    init_score = 2.0 * n * init_conv.max()
    print(f"[{time.time() - start_time:.2f}s] initial arcsine score={init_score:.6f}")

    # ------------------------------------------------------------------
    # 2) exponential‑gradient polishing (log‑sum‑exp smoothing)
    # ------------------------------------------------------------------
    # schedule of smoothing parameters: start gentle, finish sharp
    alpha_sched = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
                   160.0, 320.0, 640.0, 1280.0]
    # give the exponentiated‑gradient about 85 % of the remaining budget
    remaining = deadline - time.time()
    phase_deadline = time.time() + max(20.0, 0.85 * remaining)

    a_opt, opt_score = _exp_gradient_phase(
        a,
        n,
        phase_deadline,
        start_time,
        alpha_schedule=alpha_sched,
        eta0=0.3,
        steps_per_alpha=8000,
    )
    if opt_score < init_score:
        print(f"[{time.time() - start_time:.2f}s] post‑exp‑grad improve: "
              f"score={opt_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(a_opt, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(v) for v in best_seq]
