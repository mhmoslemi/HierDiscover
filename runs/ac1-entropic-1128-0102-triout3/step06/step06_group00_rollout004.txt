"""
Propose a low‑value step‑function for ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – use the globally supplied ``height_sequence_1`` if
   available, otherwise start from a random Dirichlet vector.  The vector
   is up‑sampled to the working length (``n = 4800``) and normalised to
   sum = 1.

2. **Smoothed exponential‑gradient warm‑up** – minimise a
   log‑sum‑exp smoothing of the max‑convolution.  This gives a reasonably
   shaped distribution very quickly.

3. **Projected Adam on the true max‑convolution** – the exact objective
   ``2·n·max(conv)`` is convex in the sense that a sub‑gradient exists.
   We use an Adam‑style adaptive step together with Euclidean projection
   onto the probability simplex.  After each projection the vector is
   symmetrised (``a = (a + a[::-1]) / 2``) because the optimal solution
   is known to be symmetric; this reduces the effective dimensionality
   and improves convergence.

4. **Optional short annealing / smoothing pass** – a few thousand tiny
   mass‑transfer moves (or a simple local averaging) are performed while
   time permits; the best sequence seen so far is kept.

All stages respect the overall time budget (≈ 1000 s).  Throughout the
search the current best objective and a timestamp are printed so that
progress can be monitored.

The returned sequence is a list of non‑negative floats (already normalised
to sum = 1) and therefore satisfies the security checks in
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, sub‑gradient, simplex proj)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a, using all entries that are
    within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    ``{x ≥ 0, 1ᵀx = 1}``.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to ``new_n`` points while keeping total mass = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Stage 1 – smoothed exponential‑gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _run_smooth_exponential(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            alpha_schedule,
                            eta0: float = 0.4,
                            steps_per_alpha: int = 4000):
    """
    Multiplicative updates based on the log‑sum‑exp smooth approximation
    of max(conv).  Returns the best vector and its raw (unsquared‑sum) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    # pre‑allocate FFT length (sufficient for a * w_rev)
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑exp improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smooth‑max gradient ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # --- multiplicative update ---------------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a /= a.sum()

        # increase smoothing parameter periodically
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – Adam‑style projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------
def _run_adam_subgrad(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      step0: float = 0.30,
                      beta1: float = 0.90,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      restart_patience: int = 250_000):
    """
    Adam optimiser on the non‑smooth objective ``2·n·max(conv)``.
    The vector is kept on the simplex and symmetrised after every step.
    If no improvement occurs for ``restart_patience`` iterations a tiny
    random jitter is added to escape possible plateaus.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] adam improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning rate (sqrt schedule) multiplied by Adam step
        lr = step0 / math.sqrt(it)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - update
        a = _project_simplex(a)

        # enforce symmetry – the optimal solution is symmetric
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)

        # occasional restart if stuck
        if it - last_improve > restart_patience:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=1e-3, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] restart after {it:,} it")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – tiny local smoothing / annealing pass (if time permits)
# ----------------------------------------------------------------------
def _run_final_tweaks(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     steps: int = 200_000):
    """
    Very short annealing‑like pass that moves a small fraction of the mass
    between two positions.  In addition, a few smoothing sweeps are tried
    (simple moving‑average) – they often reduce the peak a little.
    """
    a = a0.copy()
    score = cur_best_score
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    # ----- mass‑transfer moves ------------------------------------------------
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # modest temperature schedule
        T = 0.02 * (1.0 - it / steps) ** 2

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] tweak improve: "
                  f"iter={it:,}, score={score:.6f}")

    # ----- gentle smoothing sweeps ------------------------------------------
    # A few passes of a 3‑point moving average (still respecting the simplex)
    for sweep in range(3):
        if time.time() >= deadline:
            break
        smooth = np.convolve(best_seq, np.array([0.25, 0.5, 0.25]), mode='same')
        smooth = np.clip(smooth, 0.0, None)
        smooth /= smooth.sum()
        smooth_score = 2.0 * n * _conv_self(smooth).max()
        if smooth_score < best_score:
            best_score = smooth_score
            best_seq = smooth
            print(f"[{time.time() - start_time:.2f}s] smoothing sweep {sweep+1} improve: "
                  f"score={smooth_score:.6f}")

    return best_seq, best_score
