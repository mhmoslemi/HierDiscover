"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s):
1. Initialise from the globally stored ``height_sequence_1`` or from a uniform
   distribution and normalise to a probability vector.
2. **Warm‑up** – a short exponentiated‑gradient run on a smooth (log‑sum‑exp)
   approximation of the max‑convolution.  This quickly moves away from the
   trivial uniform shape.
3. **LP refinement** – repeatedly solve a linear programme
      maximise   Σ g_j
      subject to (f * g)_k ≤ M   for all k,   g ≥ 0,
   where f is the current distribution, M = max(f∗f).  The LP pushes mass
   towards positions where the convolution constraints are slack, which tends
   to reduce the peak of the self‑convolution.  The solution g is rescaled to
   unit sum and blended with f (convex combination).  The process is repeated
   while time permits.
4. **Projected sub‑gradient descent** – a long momentum‑accelerated sub‑gradient
   descent on the exact objective 2·n·max(conv) (the remaining budget is spent
   here).
5. **Annealing polish** – a brief simulated‑annealing pass moves tiny fractions
   of mass between entries to escape shallow plateaus.

All iterates stay in the probability simplex, are kept non‑negative and are
re‑normalised before returning.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a using all entries within
    rel_tol of the maximum."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` while preserving the total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence() for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase‑1 : exponentiated‑gradient warm‑up (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    n = len(a)
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # --- smoothed‑max gradient ------------------------------------
        scaled = alpha * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)      # safe scaling
        w /= w.sum()
        w_rev = w[::-1]

        Af = np.fft.rfft(a, n=fft_len)
        Wrev = np.fft.rfft(w_rev, n=fft_len)
        corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
        grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

        # --- multiplicative update ---------------------------------
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : linear‑programming refinement
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the dense matrix A (shape (2n‑1, n)) such that A @ g
    equals the full convolution f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _run_lp_refinement(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       t_init: float = 0.15,
                       t_min: float = 0.01,
                       max_iters: int = 500):
    """Iteratively improve a by solving a linear programme that pushes mass
    to positions where the convolution constraint (f * g) ≤ M is slack."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _objective(a)
    t = t_init
    it = 0

    while time.time() < deadline and it < max_iters and t > t_min:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        A = _build_conv_matrix(a)
        c = -np.ones(n)                     # maximise Σ g_j  ⇔  minimise −Σ g_j
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed at iter {it}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[LP] Zero sum solution – terminating.")
            break
        g = g0 / sum_g0                     # renormalise to unit sum

        candidate = (1.0 - t) * a + t * g
        cand_obj = _objective(candidate)

        if cand_obj < best_obj:
            best_obj = cand_obj
            best_a = candidate.copy()
            a = candidate
            # increase step size a little after a success
            t = min(t_init, t * 1.03)
            print(f"[{time.time() - start_time:.2f}s] LP improve: "
                  f"iter={it:,}, t={t:.4f}, obj={cand_obj:.6f}")
        else:
            # back‑off step size
            t *= 0.7

    return best_a, best_obj


# ----------------------------------------------------------------------
# Phase‑3 : projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _run_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             step0: float = 0.25,
                             beta: float = 0.94,
                             max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : final annealing polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

    try:
        starter = np.asarray(height_sequence_1, dtype=float)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline:

            start_time,

    best_seq = np.maximum(best_seq, 0.0)

    return [float(x) for x in best_seq]
