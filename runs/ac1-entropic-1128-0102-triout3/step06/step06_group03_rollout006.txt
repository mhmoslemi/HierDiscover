def _conv_self_fft(a):

    exps = np.exp(scaled - np.max(scaled)) # stable

def _upsample(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0., 1., old_n)
    new_x = np.linspace(0., 1., new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm works in four phases:

1. **Progressive up‑sampling** – start from a very coarse vector and
   gradually increase its length (800 → 2400 → 4800 → 9600).  At each
   resolution a projected (Nesterov) gradient descent is performed on a
   temperature‑annealed soft‑max of the self‑convolution.  The temperature
   is driven down very slowly so that early iterations see a smooth
   landscape while later iterations focus on the true maximum.
2. **High‑resolution refinement** – once the 9600‑point vector has been
   obtained we keep running the same optimiser for the majority of the
   remaining time budget.
3. **Random‑restart jitter** – a few short optimisation runs are launched
   from the current best vector with a tiny random perturbation; this
   helps escape shallow basins.
4. **Targeted local search** – a Metropolis‑style “peak‑shaving’’ walk
   moves mass away from indices that contribute most to the current
   maximal convolution entry towards low‑contribution indices.

All updates are projected onto the probability simplex
{ a_i ≥ 0, Σ a_i = 1 } so the objective
    2·n·max(conv) / ( Σa )²
remains unchanged by scaling (the code works with Σa = 1).
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a (full length = 2·len(a)-1) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (full) self‑convolution of a with itself using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max weights for the convolution entries
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Return the soft‑max of conv/temperature (stable version)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # subtract max for numerical stability (already done)
    exps = np.exp(scaled)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of Σ w_k·conv_k   (w = soft‑max)   with respect to a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the temperature‑softened objective.
    It equals 2·(rev(a) * w) restricted to the central n entries,
    where * denotes linear convolution and w are the soft‑max weights.
    """
    n = a.shape[0]
    w = _softmax_weights(conv, temperature)                     # length = 2n‑1
    rev_a = a[::-1]

    # convolution rev_a (len n) with w (len 2n‑1) → length = 3n‑2
    fft_len = 1 << ((n + w.shape[0] - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + w.shape[0] - 1]

    # centre block of length n
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, idx_max: int) -> np.ndarray:
    """Sub‑gradient of conv[idx_max] w.r.t. a (sum = 1 is handled by projection)."""
    n = a.shape[0]
    grad = np.zeros_like(a)

    low = max(0, idx_max - (n - 1))
    high = min(n - 1, idx_max)

    i = np.arange(low, high + 1, dtype=int)
    j = idx_max - i

    # add a[j] to grad[i] and a[i] to grad[j] (np.add.at handles repeats)
    np.add.at(grad, i, a[j])
    np.add.at(grad, j, a[i])

    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent with annealed soft‑max
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         step0: float = 0.6,
                         beta: float = 0.9,
                         temp_start: float = 2.0e-2,
                         temp_end: float = 5.0e-5,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the smooth approximation of
    max(conv).  Returns the best point seen and its (scaled) score
    `2·n·max(conv)`.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        maxv = conv.max()
        idx_max = int(conv.argmax())
        cur_score = 2.0 * n * maxv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD it={it} n={n} score={cur_score:.6f}")

        # temperature annealing (slow decay)
        T = temp_start * (temp_end / temp_start) ** (it / 200_000.0)

        # gradient: soft‑max while T is above a tiny threshold, otherwise exact sub‑gradient
        if T > 1e-6:
            grad = _softmax_gradient(a, conv, T)
        else:
            grad = _subgradient_max(a, idx_max)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size diminishing like 1/√it
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that repeatedly moves a small fraction of mass
    from indices heavily involved in the current maximal convolution entry
    to indices that contribute little.
    Returns the best point seen and its score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start_temp = 5e-2
    end_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        maxv = conv.max()
        idx_max = int(conv.argmax())
        cur_score = 2.0 * n * maxv

        # contributions of each index to this maximal entry
        n_range = np.arange(n)
        j = idx_max - n_range
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[n_range[mask]] * a[j[mask]]

        # donor: pick proportionally to contribution (focus on hot spots)
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # receiver: pick from a low‑contribution tail (bottom 20 %)
        low_thr = np.percentile(contrib, 20)
        low_ids = np.where(contrib <= low_thr)[0]
        if low_ids.size:
            i_rec = np.random.choice(low_ids)
        else:
            i_rec = np.random.randint(n)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        # temperature schedule for Metropolis acceptance
        elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = start_temp * (end_temp / start_temp) ** elapsed
        accept = delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand()

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] LS iter={it} score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
