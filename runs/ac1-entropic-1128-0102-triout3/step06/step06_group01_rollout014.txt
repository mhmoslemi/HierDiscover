def _project_simplex(v):
    # ensure non-negative and sum=1
    # algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w
```

"""
proposal: A multi‑phase optimizer for the step‑function evaluation problem.

Phase 1 – Adam on a smooth approximation of the max‑convolution.
Phase 2 – Long projected sub‑gradient descent on the exact max‑convolution.
Phase 3 – Optional short fine‑tuning Adam with a very large smoothing
          parameter (to polish the final solution).
Phase 4 – A brief simulated‑annealing refinement.

All phases respect a strict time budget (≈1000 s) and work entirely on the
probability simplex (non‑negative entries summing to one).  The code uses
FFT‑based convolutions, a fast simplex‑projection routine and prints progress
messages.  The best sequence found is returned as a list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT, gradient, projection, annealing)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a) = (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]

    # pick the n entries that correspond to the gradient
    grad = 2.0 * corr[w_rev.shape[0] - 1 - np.arange(a.shape[0])]
    return grad


def _grad_max_conv(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Exact gradient of  max(conv)  w.r.t. a.
    mask is a vector of length 2·n‑1 with weight 1 on the active set
    (and 0 elsewhere).  The gradient equals 2·a⋆mask_rev .
    """
    n = a.shape[0]
    L = _next_pow2(n + mask.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(mask[::-1], n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + mask.shape[0] - 1]
    grad = 2.0 * corr[mask.shape[0] - 1 - np.arange(n)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

            v = a - step * momentum
            a = _project_simplex(v)

    # ------------------------------------------------------------------

            max_scaled = scaled.max()
            lse = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
            f_val = lse / alpha

            # gradient w.r.t a (still via the smooth formulation)
            grad_a = _grad_smooth_max(a, conv, alpha, fft_len) * (2.0 * n)

            grad_y = a * (grad_a - np.dot(grad_a, a))

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = 0.06 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] fine‑Adam improve: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=120_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: {score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:6.2f}s] best final score {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
