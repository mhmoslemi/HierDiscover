def _conv_fft(a, b):
    n = a.size + b.size - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)

    c = np.fft.irfft(A * B, nfft)[:n]

"""
Hybrid multi‑scale optimizer for the self‑convolution peak.

The search works on the probability simplex (∑a_i = 1) so the objective
`2·n·max(conv)` is scale‑invariant.  For each length we keep a small
population of candidates, run a fast Adam optimiser on a log‑sum‑exp
surrogate of the max, apply a greedy mass‑transfer hill‑climb that uses
the exact sub‑gradient of the max, and finish with a light simulated‑
annealing polish.  The best distribution is up‑sampled to the next
length.  A final long Adam + hill‑climb + anneal phase runs on the
largest length that fits into the time budget.

The routine respects a 1000 s budget and always returns a valid
non‑negative sequence.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, Σx=1}. """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """ Linear convolution via real FFT (fast). """
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    c = np.fft.irfft(A * B, nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """ Self‑convolution a * a (full length). """
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """ Exact objective 2·n·max(a*a) for a normalized on the simplex. """
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Gradient of the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ) w.r.t. a.
    Returns a vector of size n.
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length = 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central slice (size n)
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimizer on the smooth surrogate
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the log‑sum‑exp surrogate; returns best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.20          # λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # occasional optional symmetrisation (helps for some shapes)
        if symmetrize and (np.random.rand() < 0.2):
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

        # tiny random jitter every few thousand steps to escape flat regions
        if it % 5000 == 0:
            a = a + np.random.randn(n) * 0.001
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “swap” step (optional, only for modest n)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """Construct the (2n‑1)×n matrix A for constraints (f*g)[k] ≤ M."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 3,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """Repeated LP‑based improvement (classic LP attack)."""
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        if n > 4096:                      # avoid huge memory usage
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximise sum(b) → minimise -sum(b)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 15.0})

        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b / sum_b
        a_cur = (1.0 - step) * a_cur + step * g
        a_cur = _project_simplex(a_cur)

        if start_time is not None:
            sc = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={sc:.6f}")

    return a_cur


# ----------------------------------------------------------------------
# Greedy hill‑climb using the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------


def _hill_climb_max(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer hill‑climb.
    At each step we move a fraction of mass from the index with the
    largest sub‑gradient (most responsible for the current peak) to the
    index with the smallest sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        # mask of active constraints
        mask = (conv == M).astype(float)
        cnt = mask.sum()
        if cnt == 0:
            break
        w = mask / cnt
        a_rev = a[::-1]
        tmp = _conv_fft(w, a_rev)                # length 3n‑2
        grad = 2.0 * tmp[n - 1 : n - 1 + n]       # sub‑gradient of max

        src = np.argmax(grad)
        dst = np.argmin(grad)

        if a[src] <= 1e-12:
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = np.random.choice(nonzero)

        delta = a[src] * 0.07                      # transfer ~7 % of source mass
        if delta < 1e-12:
            break

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve n={n} score={new_score:.6f}")

        # accept only improving moves (occasionally accept worsening to escape plateaus)
        if new_score <= best_score or np.random.rand() < 0.01:
            a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing (fast polish)."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling and initial shape generators
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Generate a small, diverse collection of plausible starts."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # sinusoidal (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular (peak in the centre)
    half = n // 2
    tri = np.concatenate([
        np.linspace(1, half, half, endpoint=False),
        np.linspace(half, 1, n - half)
    ])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # random Dirichlet
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    # exponential decay (random rate)
    alpha = np.random.uniform(0.001, 0.02)
    exp = np.exp(-alpha * np.arange(n))
    exp /= exp.sum()
    variants.append(exp)

    return variants

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = start_time + budget_s - 0.5   # safety margin for the evaluator

    # Warm‑start from a previously discovered sequence, if available
    # ------------------------------------------------------------------

        starter = np.asarray(height_sequence_1, dtype=float)

            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – start modest, grow gradually, keep the best for
    # the next up‑sample step.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    # deterministic order (small → large) helps with up‑sampling
    best_seq = None
    best_score = float('inf')

    for idx, n in enumerate(lengths):

        init_candidates = _initial_variants(a0)

        # ------------------------------------------------------------------

            best_local_a = _run_lp_improve(best_local_a, n,
                                           step=0.25,

                                           deadline=lp_deadline,
                                           start_time=start_time)
            best_local_score = _objective(best_local_a)

        # ------------------------------------------------------------------
        # Greedy hill‑climb using the exact max‑subgradient
        # ------------------------------------------------------------------
        hill_deadline = time.time() + min(6.0, max(2.0, (deadline - time.time()) * 0.15))
        a_hill, score_hill = _hill_climb_max(best_local_a, n,

                                                    start_temp=1e-2,

            if score_anneal < best_local_score:

                print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} score={score_anneal:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve n={n} score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # Final intensive refinement on the largest length we managed to keep.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:

            hill_deadline = time.time() + min(8.0, max(2.0, (deadline - time.time()) * 0.25))
            a_hill, score_hill = _hill_climb_max(best_seq, n,

                best_score = score_hill
                best_seq = a_hill.copy()
                print(f"[{time.time() - start_time:6.2f}s] final hill‑climb improve score={score_hill:.6f}")

        # Final annealing polish
        if time.time() < deadline - 0.2:
            anneal_deadline = deadline - 0.1
            best_seq, best_score = _run_annealing(best_seq, n,
                                                  deadline=anneal_deadline,

                                                  cur_best_seq=best_seq,
                                                  start_temp=5e-3,
                                                  end_temp=1e-6,
                                                  steps=200000)

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return even if everything failed.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as plain Python list (the external evaluator will apply its own
    # clamping / normalisation).
    return [float(x) for x in best_seq]
