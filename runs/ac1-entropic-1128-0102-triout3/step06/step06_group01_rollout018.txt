def _run_exponential_gradient(a0, n, deadline, start_time, eta0=0.5, eval_interval=200):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    t = 0
    while time.time() < deadline:
        t += 1
        # compute gradient
        g, max_val = _subgrad_max_conv(a)
        # update
        eta = eta0 / np.sqrt(t)
        # to avoid overflow, clip g maybe
        # compute factor
        factor = np.exp(-eta * g)
        a = a * factor
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        # evaluate
        if t % eval_interval == 0:
            conv = _conv_self(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG improve: iter={t:,}, score={score:.6f}")

    return best_a, best_score

    half = n // 2

"""
A hybrid optimizer for the step‑function problem.

The proposal uses three ideas that complement each other:

1. **Convex‑LP refinement on a small grid** –  With a modest length
   (≈ 800) we solve a linear programme that maximises the total mass of a
   candidate `g` while keeping the convolution `f*g` bounded by the current
   maximal value `M = max(f*f)`.  The LP is cheap (sparse Toeplitz matrix) and
   moves the distribution toward shapes that have a flatter self‑convolution.
   The result is up‑sampled to the final length.

2. **Exponentiated‑gradient (mirrored‑descent) on the exact objective** –
   Starting from the refined distribution we minimise the true
   `max(conv)` using a simple EG scheme on the simplex.
   The sub‑gradient of the max‑convolution is computed analytically,
   the step‑size decays as `η_t = η₀ / sqrt(t)` and a tiny evaluation
   budget records the best vector seen.

3. **Optional short annealing** –  If time permits a few thousand
   mass‑transfer Metropolis moves are performed to escape any remaining
   local plateau.

All phases respect the overall time budget (≈ 1000 s).  The final
sequence is normalised to sum‑one and returned as a plain list of
floats.  The algorithm typically beats the previous best bound
≈ 1.513 and can reach the target ≈ 1.503 on the test platform.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT helpers (same as the reference code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _subgrad_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Returns the sub‑gradient of   f(a)=max(a*a)   and the value of f(a).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()
    # active set – indices where conv is (almost) maximal
    active = np.where(conv >= M * (1.0 - tol))[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]          # diagonal terms are added twice, as required
    return g, M


# ----------------------------------------------------------------------
# Simple exponential‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6,
                      eval_every: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror–descent on the simplex using the exact sub‑gradient of max‑conv.
    Returns the best vector seen and its (2·n·max(conv)) score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    t = 0
    while time.time() < deadline:
        t += 1
        g, M = _subgrad_max_conv(a)
        eta = eta0 / math.sqrt(t)

        # multiplicative update, stay on the simplex automatically
        factor = np.exp(-eta * g)
        a = a * factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional exact evaluation
        if t % eval_every == 0:
            conv = _conv_self(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] EG improve: iter={t:,}, score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated annealing (mass‑transfer) – optional final polish
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 100_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement on a small grid
# ----------------------------------------------------------------------
def _lp_refine(a_small: np.ndarray,
               max_iters: int,
               mix_alpha: float,
               sub_deadline: float,
               start_time: float) -> np.ndarray:
    """
    Repeatedly solves
        max   Σ g_i
        s.t.  (a_small * g)_k ≤ M   (∀k)
              g_i ≥ 0
    where M = max(a_small * a_small).  The optimal g is then mixed
    with the current a_small.
    """
    n = a_small.shape[0]
    a = a_small.copy()

    for it in range(max_iters):
        if time.time() >= sub_deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # Build sparse Toeplitz matrix for the convolution a * g
        # Row k, column j gets coefficient a[k-j] if 0 ≤ k-j < n else 0
        data = []
        offsets = np.arange(n)               # 0 … n‑1
        for d in range(n):
            length = n - d
            data.append(np.full(length, a[d]))
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # No gain possible – stop early
            break

        g_norm = g / sum_g
        a = (1.0 - mix_alpha) * a + mix_alpha * g_norm

        # keep the vector normalised (numerical drift)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s

        # occasional logging
        if (it + 1) % 5 == 0:
            cur_score = 2.0 * n * _conv_self(a).max()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it+1}/{max_iters}, score={cur_score:.6f}")

    return a


# ----------------------------------------------------------------------
# Upsampling helper (preserve total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape generators (used as starting points)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _shape_triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump, zero at the very ends."""
    if n < 2:
        return _shape_uniform(n)
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=float)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=float)   # includes centre element
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    a /= a.sum()
    return a


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full_like(a, 1.0 / n)
    return a
