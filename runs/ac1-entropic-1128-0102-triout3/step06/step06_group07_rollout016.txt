def _run_adam(a0, n, deadline, start_time, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0*n*max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] adam improve: n={n}, iter={t:,}, score={cur_score:.6f}")
        # subgradient
        g = _subgradient_max(a, conv)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score
```

def _solve_lp_max_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    # Build constraints for each k
    for i in range(n):
        # For each j, row = i+j, col=j
        row_start = i
        row_end = i + n
        # Append entries for all j
        rows.extend(range(row_start, row_end))
        cols.extend(range(n))
        data.extend([f[i]] * n)
    # Build sparse matrix shape (2*n-1, n)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    # Upper bounds M for each row
    b = np.full(2*n-1, M)
    # bounds for variables: (0, None)
    bounds = [(0, None)] * n
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Using linprog with method='highs' (default)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

    beta1_pow = beta1
    beta2_pow = beta2
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")
        # subgradient
        g = _subgradient_max(a, conv)
        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        # proj onto simplex
        a = _project_simplex(a)
        # enforce monotone decreasing
        a = np.sort(a)[::-1]
        # update powers
        beta1_pow *= beta1
        beta2_pow *= beta2
    return best_a, best_score

"""
Non‑negative step‑function optimisation.

The search proceeds in three phases:

1. **Stage 0 – Random restarts + Adam‑style projected optimisation**  
   A short Adam run (projected onto the probability simplex and
   optionally forced to be monotone decreasing) is performed on a modest
   size (n≈1024).  Several random restarts are tried; the best vector is
   retained.

2. **Stage 1 – Linear‑programming refinement**  
   For the best vector of stage 0 we solve a linear programme that
   maximises the sum of a second vector `g` under the constraint
   `f * g <= M` (where `M` is the current maximal self‑convolution of
   `f`).  The solution is renormalised and mixed with `f` (`f←(1‑t)f+t g`);
   repeating this a few times sharply reduces the maximal convolution.

3. **Stage 2 – Upsample + long Adam run → simulated‑annealing fine‑tuning**  
   The improved vector is up‑sampled to a larger length (≈8192) and a
   longer Adam optimisation runs.  Finally a light simulated‑annealing
   phase moves tiny amounts of mass between indices to polish the bound.

All stages respect the global time budget (default 1000 s).  The final
sequence is returned as a plain Python list of non‑negative floats
(normalised to sum = 1; this does not affect `evaluate_sequence`).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP refinement)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct diagonal handling.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_inds:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimisation (with optional monotonicity)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of  F(a)=2·n·max(conv(a))  under a≥0, Σa=1.
    The iterate is projected onto the simplex after each step and then
    sorted decreasingly (monotone shape, which empirically helps).
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        g = _subgradient_max(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex and enforce monotone decreasing order
        a = _project_simplex(a)
        a = np.sort(a)[::-1]

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programme that maximises Σ g_j  subject to  f * g ≤ M
# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve the LP for a given f and bound M.  Returns the optimal g or None."""
    if not _HAS_SCIPY:
        return None

    n = f.shape[0]

    # Build the (2n‑1)×n constraint matrix A where
    #   (A @ g)_k = Σ_i f_i * g_{k‑i}
    i = np.arange(n)[:, None]            # shape (n,1)
    j = np.arange(n)[None, :]            # shape (1,n)
    rows = (i + j).ravel()               # k = i + j
    cols = j.ravel()                     # column index of g
    data = np.repeat(f, n)               # each f_i repeats n times
    A = _sp.coo_matrix((data, (rows, cols)),
                       shape=(2 * n - 1, n)).tocsr()

    # Objective: maximise Σ g_j -> minimise -Σ g_j
    c = -np.ones(n)

    # Upper bounds for every convolution entry
    b = np.full(2 * n - 1, M)

    # Non‑negativity of g
    bounds = [(0.0, None)] * n

    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _run_lp_updates(a: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    max_updates: int = 5,
                    mix: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps on vector a.
    Each step solves the LP, rescales the solution to sum = 1,
    and mixes it into a with factor `mix`.
    """
    cur_a = a.copy()
    best_score = 2.0 * n * _conv_self(cur_a).max()
    best_a = cur_a.copy()

    for upd in range(max_updates):
        if time.time() >= deadline:
            break

        conv = _conv_self(cur_a)
        M = conv.max()
        g = _lp_max_sum(cur_a, M)
        if g is None or g.sum() == 0:
            break

        g_norm = g / g.sum()
        cur_a = (1.0 - mix) * cur_a + mix * g_norm
        cur_a = _project_simplex(cur_a)
        cur_a = np.sort(cur_a)[::-1]

        score = 2.0 * n * _conv_self(cur_a).max()
        if score < best_score:
            best_score = score
            best_a = cur_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve {upd + 1}: "
                  f"score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Light annealing that transfers a fraction of mass between indices."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source index drawn proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    # allocate ~15 % of the total budget for the initial phase
    stage0_time = min(120.0, 0.15 * (deadline - time.time()))
    stage0_deadline = time.time() + stage0_time

    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # a relatively aggressive learning rate works well on the small scale
        a_opt, score = _run_adam(
            a0, n0, stage0_deadline, start_time,
            lr=0.6, beta1=0.9, beta2=0.999,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback (unlikely)
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – cheap LP refinement (still on size n0)
    # ------------------------------------------------------------------
    if time.time() < deadline and _HAS_SCIPY:
        lp_deadline = time.time() + min(30.0, 0.08 * (deadline - time.time()))
        best_seq, best_score = _run_lp_updates(
            best_seq, n0, lp_deadline, start_time,
            max_updates=4, mix=0.3
        )

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a larger size (≈8192) and long Adam run
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    best_seq = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time = max(30.0, 0.55 * remaining)          # most of what is left
    grad_deadline = time.time() + grad_time

    a_opt, score = _run_adam(
        best_seq, n1, grad_deadline, start_time,
        lr=0.4, beta1=0.9, beta2=0.999,
    )
    if score < best_score:
        best_score = score
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            n1,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
