def _upsample(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    # linear interpolation
    old_x = np.linspace(0,1,old_n)
    new_x = np.linspace(0,1,new_n)
    new = np.interp(new_x, old_x, seq)
    # renormalize to sum=1
    new_sum = new.sum()
    if new_sum>0:
        new/=new_sum
    else:
        new = np.full(new.shape, 1.0/new.shape[0])
    return new
```

def _run_targeted_local_search(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    it = 0
    while time.time() < deadline:
        it += 1
        # recompute conv for current a
        conv = _conv_self(a)
        max_val = conv.max()
        idx_max = int(np.argmax(conv))
        # compute contributions for each index i
        # contributions only for i where 0 <= idx_max-i < n
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            donor = np.random.choice(n, p=prob)
        else:
            donor = np.random.randint(n)
        # choose recipient far from max region: indices where contribution low
        low_thr = np.percentile(contrib, 20)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size > 0:
            recipient = np.random.choice(low_candidates)
        else:
            recipient = np.random.randint(n)
        if donor == recipient or a[donor] < 1e-12:
            continue
        transfer_frac = np.random.rand() * 0.3  # up to 30%
        eps = a[donor] * transfer_frac
        a_new = a.copy()
        a_new[donor] -= eps
        a_new[recipient] += eps
        # ensure non-negative
        if a_new[donor] < 0:
            continue
        # compute new score
        new_conv = _conv_self(a_new)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] local improve n={n}, iter={it}, score={new_score:.6f}")
        # Accept if better, else maybe accept with small probability
        # Using simple deterministic acceptance to focus on improvement
        if new_score <= 2.0 * n * max_val:
            a = a_new
        else:
            # occasional uphill acceptance with decreasing probability
            elapsed_ratio = (time.time() - start_time) / (deadline - start_time + 1e-12)
            temp = 0.05 * (0.01 ** elapsed_ratio)  # start 0.05, decay to 0.0005
            if np.random.rand() < math.exp(-(new_score - 2.0 * n * max_val) / (temp + 1e-12)):
                a = a_new
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    # warm start if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            starter = None
        else:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    stage_lengths = [2000, 5000, 10000, 20000]
    total_len_weight = sum(stage_lengths)
    total_time = deadline - start_time
    grad_budget = total_time * 0.75
    local_budget = total_time - grad_budget

    best_seq = None
    best_score = float('inf')

    # gradient stages
    for n in stage_lengths:
        if time.time() >= deadline:
            break
        stage_frac = n / total_len_weight
        stage_time = max(0.2, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        a_opt, score_opt = _run_adam_gd(a0, n, stage_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] stage n={n} new best score={best_score:.6f}")
        # update remaining grad_budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - local_budget - elapsed)

    # local search
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_loc, score_loc = _run_targeted_local_search(best_seq, n_final, deadline, start_time)
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(f"[{time.time() - start_time:.2f}s] final local improvement score={best_score:.6f}")

    if best_seq is None:
        best_seq = np.full(1000, 1.0/1000.0)

    # final normalization
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(v) for v in best_seq]

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

Algorithm
---------
1. Warm‑start from the previous best construction if available,
   otherwise from a random simplex point.
2. Progressive up‑sampling: 2000 → 5000 → 10000 → 20000.
3. For each size run a projected Adam optimiser on the exact
   max‑convolution (sub‑gradient of the max).  After every improvement
   the current best is stored.
4. A final “peak‑shaving” local search moves a small amount of mass from
   indices that contribute most to the current maximal convolution entry
   to low‑contribution indices, accepting occasional uphill moves.
5. Return the best sequence seen (normalised to sum = 1).

The optimiser works directly on the simplicial domain, uses FFT for fast
convolutions and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Convolution of rev(a) with an arbitrary weight vector w
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    size = n + m - 1
    fft_len = 1 << ((size).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Gradient (sub‑gradient) of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where grad is a sub‑gradient of
    max_k (a*a)_k w.r.t. a and max_val = max_k (a*a)_k.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    # indices that attain the maximum (ties are averaged)
    idxs = np.where(conv >= max_val - 1e-12)[0]
    w = np.zeros_like(conv)
    w[idxs] = 1.0 / len(idxs)

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)           # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]          # central block
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected)
# ----------------------------------------------------------------------
def _run_adam_gd(a0: np.ndarray, n: int,
                 deadline: float, start_time: float,
                 base_lr: float = 0.2,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises 2·n·max(convolution).
    Returns the best point seen and its (un‑scaled) score = 2·n·max.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        grad, max_val = _max_gradient(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size decays like 1/√t
        step = base_lr / (np.sqrt(t) + 1e-12)

        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: n={n}, iter={t}, "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Targeted local search (peak‑shaving)
# ----------------------------------------------------------------------
def _run_targeted_local_search(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float) -> tuple[np.ndarray, float]:
    """
    Moves a small fraction of probability mass away from indices that
    contribute most to the current maximal convolution entry.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * cur_max
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        idx_max = int(np.argmax(conv))

        # contributions of each index to the maximal entry
        i_arr = np.arange(n)
        j_arr = idx_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_arr[mask]] * a[j_arr[mask]]

        # pick donor proportionally to contribution
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            i_donor = np.random.choice(n, p=prob)
        else:
            i_donor = np.random.randint(n)

        # pick a low‑contribution recipient
        low_thr = np.percentile(contrib, 20)           # bottom 20 %
        low_idx = np.where(contrib <= low_thr)[0]
        if low_idx.size > 0:
            i_rec = np.random.choice(low_idx)
        else:
            i_rec = np.random.randint(n)

        if i_donor == i_rec or a[i_donor] <= 1e-15:
            continue

        # transfer a random fraction (up to 30 %) of donor mass
        eps = a[i_donor] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i_donor] -= eps
        a_prop[i_rec]   += eps

        # evaluate proposal
        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] local improve: n={n}, iter={it}, "
                  f"score={new_score:.6f}")

        # accept downhill moves always; uphill with a simple cooling schedule
        if new_score <= 2.0 * n * max_val:
            a = a_prop
        else:
            # linear cooling from 5e‑2 to 1e‑6 over the remaining time
            elapsed = (time.time() - start_time) / (deadline - start_time + 1e-12)
            T = 5e-2 * (1e-6 / 5e-2) ** elapsed
            accept_prob = math.exp(-(new_score - 2.0 * n * max_val) / (T + 1e-12))
            if np.random.rand() < accept_prob:
                a = a_prop

    return best_a, best_score
