"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The method consists of two phases:

* Phase 0 – a quick optimisation on a medium length (≈4096) using a
  “smooth‑max” surrogate for the infinity‑norm of the self‑convolution.
  Gradients are obtained analytically and we run an Adam optimiser with
  simplex projection.

* Phase 1 – upscale the best medium‑length vector to a large length
  (≈12288) and continue the Adam optimisation with a tighter smooth‑max
  (larger temperature).  Finally a short simulated‑annealing refinement
  is performed.

All work respects the time budget (default 1000 s).  The code contains
only top‑level helper functions; no closures or file I/O are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray,
                         conv: np.ndarray,
                         tau: float) -> tuple[np.ndarray, float]:
    """
    Given a and its self‑convolution `conv`,
    compute the gradient of   f_tau(a) = (1/tau)·log Σ exp(tau·conv_k).
    Returns (grad, actual_max_conv).
    """
    # soft‑max weights over the convolution entries
    scaled = tau * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)               # shape (2n‑1,)

    # gradient = 2 * (w * a) convolution
    a_rev = a[::-1]                                    # reversed a
    L = w.shape[0]
    pad = 1 << ((L + a_rev.shape[0] - 1).bit_length())
    W = np.fft.rfft(w, n=pad)
    A_rev = np.fft.rfft(a_rev, n=pad)
    full = np.fft.irfft(W * A_rev, n=pad)              # length >= L + n - 1
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]        # shape (n,)

    return grad, conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate, with simplex projection
# ----------------------------------------------------------------------
def _run_smooth_adam(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     lr: float = 0.12,
                     beta1: float = 0.90,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     tau_start: float = 5.0,
                     tau_end: float = 200.0,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of   F(a) = max_k (a*a)_k   (smoothly approximated).
    The iterate is always projected onto the simplex.
    Returns the best vector observed and its exact evaluation score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # full self‑convolution
        conv = _conv_self_fft(a)

        # temperature schedule – linear in elapsed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        tau = tau_start * (tau_end / tau_start) ** frac

        # gradient of the smoothed max
        grad, max_conv = _smooth_max_and_grad(a, conv, tau)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep a on the simplex
        a = _project_simplex(a)

        # occasional exact evaluation (every 10 iterations)
        if it % 10 == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] ADAM iter={it:,} "
                      f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Light simulated annealing that moves a random fraction of mass from a
    source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source chosen proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear interpolation up‑sampling and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) that minimises
    `evaluate_sequence`.  The routine uses a two‑stage Adam optimisation
    on a smooth surrogate, followed by a brief annealing polish.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Stage 0 – medium size optimisation (≈4096)
    # ------------------------------------------------------------------
    n0 = 4096
    # initialise from possible historic sequence
    init_seq = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                init_seq = _upsample(tmp, n0)
        except Exception:
            init_seq = None

    if init_seq is None:
        a0 = rng.random(n0)
        a0 /= a0.sum()
    else:
        a0 = init_seq

    # allocate about 20 % of the total time to the small stage
    small_deadline = start_time + 0.20 * (deadline - start_time)

    a_mid, score_mid = _run_smooth_adam(
        a0, n0, small_deadline, start_time,
        lr=0.12, beta1=0.90, beta2=0.999,
        tau_start=5.0, tau_end=80.0
    )
    print(f"[{time.time() - start_time:6.2f}s] stage 0 (n={n0}) best score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to a large size (≈12288) and continue optimisation
    # ------------------------------------------------------------------
    n1 = 12288
    a1 = _upsample(a_mid, n1)

    # remaining time is used for the large‑scale optimisation
    a_big, score_big = _run_smooth_adam(
        a1, n1, deadline, start_time,
        lr=0.10, beta1=0.92, beta2=0.999,
        tau_start=30.0, tau_end=200.0
    )
    print(f"[{time.time() - start_time:6.2f}s] stage 1 (n={n1}) best score={score_big:.6f}")

    best_seq = a_big
    best_score = score_big

    # ------------------------------------------------------------------
    # Stage 2 – short simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq, n1, deadline, start_time,
            best_score, best_seq,
            start_temp=4e-2, end_temp=1e-6, steps=200_000
        )
        if score_ref < best_score:
            best_seq = a_ref
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal best score={score_ref:.6f}")

    # final normalisation (should already be normalized)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
