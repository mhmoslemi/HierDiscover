"""
Hybrid long‑run Adam + sub‑gradient + annealing search for low self‑convolution peak.
The routine:
  * loads an optional warm‑start “height_sequence_1”.
  * upsamples it to a moderate size (4096).
  * spends the majority of the time running a projected Adam optimiser on the
    smooth log‑sum‑exp surrogate of the max‑convolution.
  * follows with a focussed projected sub‑gradient descent on the exact max‑norm.
  * ends with a cheap simulated‑annealing mass‑transfer polish.
All steps respect the global time‑budget and keep the vector on the simplex
so that the evaluation function is scale‑invariant.
"""
import time, math, random
import numpy as np

def _project_simplex(v):
    """Project onto the probability simplex Σx=1, x≥0."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(a, b):
    """Linear convolution via FFT (real)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = _next_pow2(n)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    c = np.fft.irfft(A * B, nfft)[:n]
    return c

def _conv_self(a):
    """Self‑convolution a*a."""
    return _conv_fft(a, a)

def _objective(a):
    """Exact objective 2·n·max(conv) (assuming Σa = 1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()

def _gradient_logsumexp(a, conv, lam):
    """Gradient of λ·log Σ exp(conv/λ)."""
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w_sum = w.sum()
    w /= w_sum
    a_rev = a[::-1]
    # convolution of w (length 2n-1) with a_rev (length n)
    tmp = _conv_fft(w, a_rev)
    n = a.shape[0]
    grad = 2.0 * tmp[n-1 : n-1 + n]
    return grad

def _run_adam_opt(a0, deadline, start_time):
    """Projected Adam on the smooth surrogate until `deadline`."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # Lambda schedule parameters
    lam_factor = 1.0          # start with λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 5000
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam iter={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_logsumexp(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        step = 0.8 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score

def _subgradient_descent(a0, deadline, start_time):
    """Projected sub‑gradient descent on the exact max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n, dtype=int)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        step = 0.5 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad iter={it} score={cur_score:.6f}")

    return best, best_score

def _run_annealing(a0, deadline, start_time, cur_best_score, cur_best_seq):
    """Simple mass‑transfer simulated annealing."""
    a = a0.copy()
    n = a.shape[0]
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    # annealing schedule parameters
    start_temp = 2e-2
    end_temp = 1e-6
    total_steps = 200000
    step = 0
    while time.time() < deadline and step < total_steps:
        step += 1
        # linear temp schedule in log space
        T = start_temp * (end_temp / start_temp) ** (step / total_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = _objective(a_prop)
        delta = new_score - score
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Anneal step={step} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq, new_n):
    """Linear interpolation from seq of length m to new_n, then renormalise to sum=1."""
    m = seq.shape[0]
    if m == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, m)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # -------------------------------------------------------------
    # Warm‑start: use previously discovered sequence if present.
    # -------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.ndim == 1 and base.size > 0:
            base = np.maximum(base, 0.0)
            if base.sum() > 0.0:
                base = base / base.sum()
            else:
                base = None
        else:
            base = None
    except Exception:
        base = None

    # -------------------------------------------------------------
    # Choose a working length. 4096 gives a good trade‑off between
    # optimisation speed and expressive power.
    # -------------------------------------------------------------
    n = 4096

    if base is not None:
        a0 = _upsample(base, n)
    else:
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()

    # -------------------------------------------------------------
    # Phase 1 – long Adam optimisation (≈70 % of the budget)
    # -------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    best_a, best_score = _run_adam_opt(a0, phase1_deadline, start_time)

    # -------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max (≈20 % of budget)
    # -------------------------------------------------------------
    phase2_deadline = start_time + 0.90 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = _subgradient_descent(best_a, phase2_deadline, start_time)

    # -------------------------------------------------------------
    # Phase 3 – simulated‑annealing polish (remaining time)
    # -------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _run_annealing(best_a, deadline, start_time,
                                            cur_best_score=best_score,
                                            cur_best_seq=best_a)

    print(f"[{time.time()-start_time:.2f}s] final best score={best_score:.6f}")
    # Ensure a valid return even if something went wrong.
    if best_a is None or best_a.size == 0:
        best_a = np.full(n, 1.0 / n)

    return [float(x) for x in best_a]

"""
Hybrid long‑run Adam + sub‑gradient + annealing search for a low
self‑convolution peak.

The algorithm works on the probability simplex (the vector is projected
onto Σ a_i = 1, a_i ≥ 0) so the objective
    2·n·max(conv(a)) = evaluate_sequence(a)
is scale‑invariant.  The steps are:

1.  Load an optional warm‑start (global variable `height_sequence_1`),
    up‑sample to a moderate length (4096) and normalise.

2.  Spend ~70 % of the time running projected Adam on a smooth
    log‑sum‑exp surrogate of the max‑convolution.  The surrogate’s
    temperature λ is slowly shrunk to focus the optimisation on the true
    peak.

3.  Refine the result with projected sub‑gradient descent on the exact
    max‑norm (the true objective).

4.  Use a cheap simulated‑annealing “mass‑transfer’’ polish for the
    remaining budget.

All phases respect the global time budget and keep the vector on the
simplex, guaranteeing a valid return for `evaluate_sequence`.  The
implementation is pure NumPy (no file or network I/O) and fits within a
single CPU core.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a∗b using real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = _next_pow2(n)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    c = np.fft.irfft(A * B, nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact objective (Σa = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _gradient_logsumexp(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ).
    Returns a vector of length len(a).
    """
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]                           # reversed a
    tmp = _conv_fft(w, a_rev)                  # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]         # central slice of length n
    return grad


# ----------------------------------------------------------------------
# Projected Adam on the smooth surrogate
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray, deadline: float, start_time: float):
    """Projected Adam optimiser; returns best vector seen and its score."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 1.0          # λ = lam_factor * max(conv)
    lam_decay = 0.85
    lam_decay_interval = 5000

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it} score={cur_score:.6f}"
            )

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_logsumexp(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, start_time: float):
    """Projected sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        max_idx = np.flatnonzero(conv == M)

        grad = np.zeros_like(a)
        for k in max_idx:
            # contributions from pairs (i, k‑i)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            grad[i_range] += a[j_range]

            # contributions from pairs (k+i, i) where i≥k
            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n, dtype=int)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        step = 0.5 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Subgrad iter={it} score={cur_score:.6f}"
            )

    return best, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (mass transfer)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
):
    """Mass‑transfer annealing; keeps the best sequence seen."""
    a = a0.copy()
    n = a.shape[0]
    score = _objective(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    start_temp = 2e-2
    end_temp = 1e-6
    total_steps = 200_000
    step = 0

    while time.time() < deadline and step < total_steps:
        step += 1
        T = start_temp * (end_temp / start_temp) ** (step / total_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Anneal step={step} score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (piecewise‑linear interpolation) + normalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate seq to length new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Respects the `budget_s` time limit and returns a plain Python list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Warm start: try to reuse a previously obtained sequence.
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.ndim == 1 and base.size > 0:
            base = np.maximum(base, 0.0)
            if base.sum() > 0.0:
                base = base / base.sum()
            else:
                base = None
        else:
            base = None
    except Exception:
        base = None

    # --------------------------------------------------------------
    # Working length – 4096 gives a good trade‑off between speed and
    # expressive power for the allotted budget.
    # --------------------------------------------------------------
    n = 4096

    if base is not None:
        a0 = _upsample(base, n)
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # --------------------------------------------------------------
    # Phase 1: long Adam optimisation (≈70 % of total budget)
    # --------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 0.70 * budget_s)
    best_a, best_score = _run_adam_opt(a0, phase1_deadline, start_time)

    # --------------------------------------------------------------
    # Phase 2: sub‑gradient descent on the exact max‑norm (≈20 %)
    # --------------------------------------------------------------
    phase2_deadline = min(deadline, start_time + 0.90 * budget_s)
    if time.time() < phase2_deadline:
        best_a, best_score = _subgradient_descent(best_a, phase2_deadline, start_time)

    # --------------------------------------------------------------
    # Phase 3: simulated‑annealing polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _run_annealing(
            best_a,
            deadline,
            start_time,
            cur_best_score=best_score,
            cur_best_seq=best_a,
        )

    print(f"[{time.time() - start_time:.2f}s] final best score={best_score:.6f}")

    if best_a is None or best_a.size == 0:
        best_a = np.full(n, 1.0 / n)

    # Convert to plain Python list for the evaluator.
    return [float(x) for x in best_a]
