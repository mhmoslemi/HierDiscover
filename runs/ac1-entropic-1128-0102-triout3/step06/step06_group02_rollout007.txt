def adam_softmax(a0, deadline, start_time, best_seq, best_score, max_iter=10**6):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        maxc = conv.max()
        # temperature schedule: start high, decrease exponentially
        T = max(1e-6, 0.5 * (0.01 ** (t / 5000.0))) # e.g., start 0.5 down to ~1e-6
        # compute weights softmax
        scaled = (conv - maxc) / T
        exps = np.exp(scaled - np.max(scaled)) # stable
        w = exps / exps.sum()
        # gradient
        rev_a = a[::-1]
        # convolution rev_a * w (need central part length n)
        n = a.shape[0]
        m_len = w.shape[0]
        fft_len = 1 << ((n + m_len - 1).bit_length())
        A = np.fft.rfft(rev_a, n=fft_len)
        B = np.fft.rfft(w, n=fft_len)
        full = np.fft.irfft(A*B, n=fft_len)[: n + m_len - 1]
        grad = 2.0 * full[n-1:n-1+n]   # index size n
        # Adam moments
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # update with projection
        a = _project_simplex(a - step)

        # evaluate objective
        cur_score = 2 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] adam_softmax better score {cur_score:.6f}")
    return best_seq, best_score

    m = np.zeros_like(a); v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0 = 0.05
    t=0
    n = a.shape[0]
    while time.time()<deadline and t<max_iter:
        t+=1
        conv = _conv_self(a)
        max_val = conv.max()
        k = conv.argmax()
        # compute subgradient
        grad = np.zeros_like(a)
        if k < n:
            grad[:k+1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        v = beta2*v + (1-beta2)*(grad**2)

            conv = conv_prop

import time
import math

# ----------------------------------------------------------------------
# Simple (fast) projection onto the probability simplex:
# keep non‑negative entries and renormalise to sum = 1.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, sum x = 1} by clamping negatives and renormalising."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v.fill(1.0 / v.shape[0])
    return v

# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:

    size = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)[: 2 * n - 1]

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    return _proj_simplex(up)


# ----------------------------------------------------------------------
# Objective used in the original evaluation (sum is forced to 1 later).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _conv_self(a)
    maxc = conv.max()
    sum_a = a.sum()
    # sum_a is expected to be 1, but we keep the exact formula.
    if sum_a < 1e-12:  # avoid division by zero
        return float("inf")
    return float(2.0 * n * maxc / (sum_a ** 2))


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weighting of the convolution.
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """Return a probability vector w = softmax(conv/temperature)."""
    maxc = conv.max()
    # shift for numerical stability
    scaled = (conv - maxc) / max(temperature, 1e-12)
    # subtract the max again to avoid overflow
    shifted = scaled - scaled.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max objective (2·a * w convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    rev_a = a[::-1]
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max approximation.
# ----------------------------------------------------------------------
def _adam_softmax(a0: np.ndarray, deadline: float, start_time: float,
                  best_seq: np.ndarray, best_score: float,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.06

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)

        # temperature schedule – from 5·10⁻¹ down to 5·10⁻⁶
        T = 5e-1 * (5e-6 / 5e-1) ** (it / 20000.0)

        w = _softmax_weights(conv, T)
        grad = _softmax_gradient(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a - step)

        cur_score = 2.0 * n * conv.max()          # sum = 1 → denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam_softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the true (nonsmooth) objective – sub‑gradient of max.
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray, deadline: float, start_time: float,
                  best_seq: np.ndarray, best_score: float,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        k = conv.argmax()          # index of the active shift

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k :: -1]          # a[k‑i] for i≤k
        else:
            start = k - n + 1
            grad[start:] = 2.0 * a[start:][::-1]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _proj_simplex(a - step)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam_subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing walk (tiny mass transfers) with a slow cooling schedule.
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray, deadline: float, start_time: float,
            cur_best_seq: np.ndarray, cur_best_score: float,
            start_temp: float = 0.4, end_temp: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    # current score (sum is forced to 1 → denominator = 1)
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    it = 0
    # we do not fix a hard iteration count; we just run until the deadline.
    while time.time() < deadline:
        it += 1

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / 5_000_000)

        # pick two distinct positions
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 30 % of the mass at i towards j
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate proposal
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            conv = conv_prop
            score = new_score

        # record improvement
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    The routine:
      * starts from the previously found construction (if any) and up‑samples
        it to a target length,
      * runs Adam on a smooth soft‑max surrogate,
      * refines with Adam on the true max‑convolution,
      * finishes with a long simulated‑annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness itself
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Load the previously discovered construction (if it exists).
    # ------------------------------------------------------------------
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            starter = _proj_simplex(hs)
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a final resolution.
    # We keep it modest (≈12 k) so that many optimisation steps still fit.
    # ------------------------------------------------------------------
    n_target = 12000

    # ------------------------------------------------------------------
    # Initialise the first candidate.
    # ------------------------------------------------------------------
    if starter is not None:
        a_cur = _upsample(starter, n_target)
    else:
        a_cur = np.random.rand(n_target)
        a_cur = _proj_simplex(a_cur)

    best_seq = a_cur.copy()
    best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Partition the remaining time: ~35 % for gradient phases,
    # the rest for annealing.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    grad_budget = max(30.0, 0.35 * remaining)
    anneal_budget = remaining - grad_budget

    # ----- soft‑max Adam ------------------------------------------------
    softmax_deadline = time.time() + grad_budget * 0.5
    best_seq, best_score = _adam_softmax(
        best_seq, softmax_deadline, start_time, best_seq, best_score,
        max_iter=10 ** 7
    )

    # ----- exact sub‑gradient Adam --------------------------------------
    if time.time() < softmax_deadline:
        subgrad_deadline = softmax_deadline  # use the same finish time
        best_seq, best_score = _adam_subgrad(
            best_seq, subgrad_deadline, start_time, best_seq, best_score,
            max_iter=10 ** 7
        )

    # ------------------------------------------------------------------
    # Final annealing phase – consume the rest of the budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_seq, best_score = _anneal(
            best_seq, anneal_deadline, start_time,
            best_seq, best_score,
            start_temp=0.4, end_temp=1e-7
        )
        print(f"[{time.time() - start_time:6.2f}s] final best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Small post‑processing pass (soft‑max Adam) to polish the final answer.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        post_deadline = deadline
        best_seq, best_score = _adam_softmax(
            best_seq, post_deadline, start_time, best_seq, best_score,
            max_iter=2000
        )

    # ------------------------------------------------------------------
    # Ensure the vector is a proper probability distribution before returning.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    best_seq = _proj_simplex(best_seq)

    return [float(x) for x in best_seq]
