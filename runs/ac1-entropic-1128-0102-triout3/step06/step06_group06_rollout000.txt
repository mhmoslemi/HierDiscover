def adam_optimize(a_init, n, deadline, start_time):
    a = a_init.copy()
    # reduce some jitter? Not needed.
    y = np.log(a + 1e-12)  # initial logits
    m = np.zeros(n)
    v = np.zeros(n)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr=0.2  # maybe adjust

    # temperature schedule
    # compute initial conv max
    conv_cur = conv_self_fft(a)
    maxf = conv_cur.max()
    # start tau relative
    tau = maxf / 4.0  # smoothing factor
    tau_final = maxf / 1e6  # near zero
    tau_decay = 0.99

    best_sc = score_of(a)
    best_a = a.copy()

    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv and softmax weights
        conv = conv_self_fft(a)
        max_val = conv.max()
        # compute weights w
        if tau > 1e-12:
            shift = max_val
            w = np.exp((conv - shift) / tau)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact max
            w = np.zeros_like(conv)
            w[conv == max_val] = 1.0
            w /= w.sum()

        # compute gradient for a
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)   # length = len(w)+len(a_rev)-1 = (2n-1)+(n)-1 = 3n-2? Actually w length = 2n-1
        # idx = (n-1) - np.arange(n) selects region of length n
        idx = (n - 1) - np.arange(n)
        grad_a = 2.0 * corr[idx]

        # compute gradient w.r.t y
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # recompute a from logits
        exp_y = np.exp(y - np.max(y))  # subtract max for stability
        a = exp_y / exp_y.sum()

        # optionally enforce small epsilon to avoid zero?
        # compute score
        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={t} score={sc:.6f}")

        # anneal temperature
        tau = max(tau_final, tau * tau_decay)

    return best_a, best_sc

"""
Propose a step‑function minimizing `evaluate_sequence`.

Algorithm
---------
1. Choose a moderate length (default 4096).  This is large enough to
   approximate the continuous optimum but small enough for fast FFT
   convolutions.

2. Initialise several analytic shapes (arcsine, product‑power,
   Gaussian, symmetric Beta).  Keep the best of these as a warm‑start.

3. Optimise the shape on the probability simplex using Adam on the
   log‑it (soft‑max) parametrisation:
      a_i = exp(y_i) / Σ_j exp(y_j)

   * The objective is 2·n·max(conv(a,a)), which is invariant to scaling.
   * We smooth the max with a temperature τ and gradually anneal τ→0.
   * The gradient of the smoothed max is obtained by a cross‑correlation
     between the soft‑max weights and the reversed vector a.
   * A standard Adam update is performed on y, followed by a soft‑max
     projection back onto the simplex.

4. The optimisation runs repeatedly with random restarts while time
   permits; the best vector seen is kept globally.

5. The final vector is returned as a plain Python list.  All entries are
   non‑negative and sum to one, satisfying the security checks in
   `evaluate_sequence`.

The whole procedure respects a 1000 s budget and prints progress
information.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------


def _fft_len_needed(m: int) -> int:
    """Smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1) using power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = _fft_len_needed(m)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution of two 1‑D arrays via FFT."""
    m = len(u) + len(v) - 1
    fft_len = _fft_len_needed(m)
    Uf = np.fft.rfft(u, n=fft_len)
    Vf = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(Uf * Vf, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring (scale‑invariant)
# ----------------------------------------------------------------------


def score_of(a: np.ndarray) -> float:
    """
    Exact objective for a normalized vector (∑a = 1):
        2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_self_fft(a).max()


# ----------------------------------------------------------------------
# Parametric families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5)  raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density ∝ 1/√(i·(n‑i))."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑it parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(
    a_init: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise max(a*a) using Adam on y_i = log a_i.
    Returns best vector seen and its exact score.
    """
    # initialise log‑it variables
    y = np.log(a_init + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2  # base learning rate

    # temperature schedule for the soft‑max surrogate
    conv_cur = conv_self_fft(np.exp(y) / np.exp(y).sum())
    max_cur = conv_cur.max()
    tau = max_cur / 4.0 if max_cur > 0 else 1.0
    tau_final = max_cur / 1e6 if max_cur > 0 else 1e-8
    tau_decay = 0.985

    best_a = np.exp(y) / np.exp(y).sum()
    best_sc = score_of(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        a = np.exp(y - np.max(y))  # stability shift
        a = a / a.sum()

        # current self‑convolution
        conv = conv_self_fft(a)
        max_val = conv.max()

        # soft‑max weights over the 2·n‑1 convolution entries
        if tau > 1e-12:
            shift = max_val
            w = np.exp((conv - shift) / tau)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv)
            w[conv == max_val] = 1.0
            w /= w.sum()

        # gradient of the surrogate w.r.t. a:
        a_rev = a[::-1]
        corr = conv_fft(w, a_rev)                     # length = 3·n‑2
        idx = (n - 1) - np.arange(n)                  # pick the n relevant entries
        grad_a = 2.0 * corr[idx]

        # transform gradient to log‑it space (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate exact objective
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={sc:.6f}"
            )

        # anneal temperature
        tau = max(tau_final, tau * tau_decay)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # Target length – large enough for a fine approximation, small enough
    # for fast FFT‑based optimisation.
    n_target = 4096

    # ------------------------------------------------------------------
    # Warm‑start from a few analytic families
    # ------------------------------------------------------------------
    warm_starts = [
        shape_arcsine(n_target),
        shape_product_power(n_target, p=0.0),
        shape_product_power(n_target, p=1.0),
        shape_gaussian(n_target, sigma=n_target / 5.0),
        shape_beta(n_target, alpha=0.5, beta_par=0.5),
    ]

    best_a = None
    best_sc = float("inf")
    for idx, cand in enumerate(warm_starts):
        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] warm‑start {idx} score={sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Main optimisation loop (Adam with random restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # Choose an initial point: either the current best or a fresh random one
        if random.random() < 0.7:
            a0 = best_a.copy()
        else:
            a0 = np.random.rand(n_target)
            a0 /= a0.sum()

        # Give each optimisation run a slice of the remaining budget
        inner_deadline = time.time() + min(30.0, deadline - time.time() - 1.0)
        a_opt, sc_opt = adam_optimize(a0, n_target, inner_deadline, start_time)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST score={best_sc:.6f}"
            )

    # Fallback in the extremely unlikely case that nothing was found
    if best_a is None:
        best_a = np.full(n_target, 1.0 / n_target)

    # Convert to plain Python list (evaluate_sequence will apply its own clipping)
    return [float(x) for x in best_a]
