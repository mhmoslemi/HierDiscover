"""
Non‑negative step‑function optimiser.

The objective to minimise is
    F(a) = 2·n·max(conv(a,a))          (sum(a)=1, a ≥ 0)

`conv` is the linear self‑convolution.  The programme works in three
phases:

1.  Initialise a probability vector (the normalised step heights) – either
    from the supplied ``height_sequence_1`` or from a random start.
2.  Run a long projected‑gradient descent where the sub‑gradient of
    ``max(conv)`` is replaced by a smooth weighted average.
    The weight vector is a soft‑max of the normalised convolution values,
    i.e.  ``w_i ∝ exp(α·(c_i / max(c)‑1))``.  The temperature ``α`` is
    increased during the run, so the method starts with a very smooth
    approximation of the maximum and finishes with an almost exact
    sub‑gradient.
    Nesterov momentum and a diminishing step‑size (∝1/√t) are used;
    after each step the vector is projected onto the probability simplex.
3.  A lightweight simulated‑annealing refinement that moves a small
    amount of mass between two indices – this explores the discrete
    neighbourhood of the best continuous solution.

All phases respect the overall time budget (default 1000 s).  The best
vector found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import for fast correlation – fall back to FFT version.
# ----------------------------------------------------------------------
try:
    from scipy.signal import correlate as _scipy_correlate
except Exception:  # pragma: no cover
    _scipy_correlate = None


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Linear correlation a ⋆ w (valid part) – using SciPy when available,
# otherwise a small FFT‑based implementation.
# ----------------------------------------------------------------------
def _correlate_valid(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return the valid linear correlation of a with w (length = len(a))."""
    if _scipy_correlate is not None:
        # SciPy handles the details correctly and is fast
        return _scipy_correlate(a, w, mode='valid')
    # ----------- fallback using FFT (works for any size) -----------
    n = a.shape[0]
    lw = w.shape[0]
    L = 1 << ((n + lw - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)
    # The "valid" part starts at index lw - n and has length n
    start = lw - n
    end = start + n
    return corr_full[start:end]


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient of max(conv) – returns a vector of length n.
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute a smoothed sub‑gradient of max(conv(a,a)).
    α (alpha) controls the sharpness of the soft‑max; larger α gives a
    closer approximation to the true sub‑gradient.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    if max_c <= 0:
        # degenerate case – return zero gradient
        return np.zeros_like(a)

    # 2) soft‑max weights on the *relative* convolution values
    rel = conv / max_c                       # in [0,1]
    # w_i = exp(alpha * (rel_i - 1)).  The max entry gets weight 1.
    exp_vals = np.exp(alpha * (rel - 1.0))
    w = exp_vals / exp_vals.sum()            # length 2n‑1, sum = 1

    # 3) gradient = 2 * (a ⋆ w)   (valid correlation)
    grad = 2.0 * _correlate_valid(a, w)       # length n
    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent with a smoothed objective.
# ----------------------------------------------------------------------
def _run_smooth_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.30,
                        beta: float = 0.92,
                        init_alpha: float = 5.0,
                        max_alpha: float = 200.0,
                        alpha_factor: float = 1.5,
                        it_per_alpha: int = 40000,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise F(a)=2·n·max(conv) on the simplex using a smooth surrogate.
    Returns the best vector encountered and its (exact) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    alpha = init_alpha
    next_alpha_iter = it_per_alpha
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # exact objective (needed for reporting & final score)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve: n={n}, "
                  f"iter={it:,}, α={alpha:.1f}, score={cur_score:.6f}")

        # smoothed gradient
        grad = _smooth_gradient(a, alpha)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - step * momentum)

        # increase α according to schedule
        if it >= next_alpha_iter and alpha < max_alpha:
            alpha = min(alpha * alpha_factor, max_alpha)
            next_alpha_iter += it_per_alpha

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (identical to the earlier version
# but with a slightly larger step budget).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 350_000) -> tuple[np.ndarray, float]:
    """Fine‑tune a candidate by stochastic mass transfers."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source index proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine performs:

    * initialisation (reuse `height_sequence_1` if available),
    * several independent smoothed gradient‑descent runs,
    * a final annealing phase.

    The whole computation respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a target length – a power of two gives fast FFTs.
    # ------------------------------------------------------------------
    target_n = 12288  # ≈ 1.2·10⁴ – already used by the previous run

    # ------------------------------------------------------------------
    # Build an initial candidate (normalised to sum = 1)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = _upsample(tmp, target_n)
        except Exception:
            starter = None

    if starter is None:
        # uniform start
        starter = np.full(target_n, 1.0 / target_n)

    best_seq = starter.copy()
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – a few short random restarts for diversity
    # ------------------------------------------------------------------
    restarts = 5
    # allocate about 15 % of the budget to quick restarts
    restart_deadline = time.time() + 0.15 * (deadline - time.time())
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if r == 0:
            a0 = starter.copy()
        else:
            a0 = np.random.rand(target_n)
            a0 = _project_simplex(a0)   # already normalised

        a_opt, sc = _run_smooth_descent(
            a0,
            target_n,
            restart_deadline,
            start_time,
            step0=0.35,
            beta=0.90,
            init_alpha=4.0,
            max_alpha=120.0,
            alpha_factor=1.6,
            it_per_alpha=30000,
            max_iter=200_000,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart {r} improve: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long smooth descent from the current best vector
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # give roughly half of the remaining time to the main descent
        descent_deadline = time.time() + 0.48 * remaining

        a_opt, sc = _run_smooth_descent(
            best_seq,
            target_n,
            descent_deadline,
            start_time,
            step0=0.30,
            beta=0.92,
            init_alpha=6.0,
            max_alpha=180.0,
            alpha_factor=1.4,
            it_per_alpha=40000,
            max_iter=10 ** 7,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] main descent improve: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, sc = _run_annealing(
            best_seq,
            target_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against numerical drift) and return as Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
